"""
Propose a non‑negative step sequence with a low evaluation score.

Main ideas
----------
* Use the full length `max_n = 4000` (the larger the sequence, the lower the
  theoretical bound `2 n/(2 n−1) ≈ 1.0`).
* Two‑stage optimisation:
    1. Many short L‑BFGS runs on a smoothed log‑sum‑exp approximation of the
       max‑convolution (continuation on the smoothing parameter).
    2. A long projected sub‑gradient descent on the exact max‑convolution.
       The sub‑gradient for the active shift `k` is
       `g_i = 2·a_{k‑i}` on the overlapping indices, and after each step we
       project back onto the probability simplex.
* Finish with a short simulated‑annealing run and a greedy hill‑climb.
* All phases respect the 1000 s budget; timing is split automatically.

The implementation is self‑contained, uses only NumPy/SciPy and works with
the provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smooth‑max based on log‑sum‑exp; returns (objective, gradient) w.r.t. logits y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (on the smoothed objective)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each smoothing parameter `lam` in `lam_schedule`,
    starting each stage from the previous result.
    Returns the best sequence found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            # ignore failures, keep current y
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(a0: np.ndarray,
                    deadline: float,
                    fft_len: int,
                    n: int):
    """
    Projected sub‑gradient descent on f(a) = max_k (a*a)[k].
    Operates on the simplex. Returns the best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        conv = full_conv(a, fft_len)
        k = int(conv.argmax())                     # index of current maximum

        # sub‑gradient of the max (exact)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]                          # d max / d a_i

        # step‑size schedule (decreasing)
        eta = 0.2 / np.sqrt(it + 1.0)               # modest step to stay stable

        # gradient step + projection onto simplex
        a = a - eta * g
        a = simplex_projection(a)

        # evaluate
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy mass‑transfer hill‑climb:
    move a small random fraction from one coordinate to another;
    accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing phase (lightweight)
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """Random annealing that can accept worsening moves."""
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T = max(T * 0.92, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialiser (varied families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    A mixture of families is used to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sine', 'halfsine', 'triangular'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    elif mode == 'halfsine':
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
    else:  # triangular
        a = np.concatenate([np.linspace(0, 1, n // 2 + 1),
                            np.linspace(1, 0, n - n // 2 - 1)])
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # ------------------------------------------------------------------
    # Random‑seed handling
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (use the full budgeted length)
    # ------------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a
    fft_len_g = next_pow2(3 * n - 2)      # auxiliary length for gradient calc

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin
    phase1_end = start_time + 0.30 * budget_s      # L‑BFGS phase
    phase2_end = start_time + 0.80 * budget_s      # sub‑gradient descent phase

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS restarts with a fine lam schedule
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 250   # keep each L‑BFGS run short to allow many restarts
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (long, many iterations)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sub_deadline = phase2_end - 0.1
        best_a, best_score = subgrad_descent(best_a,
                                             sub_deadline,
                                             fft_len_conv,
                                             n)
        print(f"[sub‑gradient] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – brief simulated annealing then greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.3
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] fast_score={best_score:.6f}")

    # final greedy hill‑climb (takes remaining time)
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    deadline,
                                    max_iter=12000)
    print(f"[final hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence in the format expected by the evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
