"""
Propose a low‑evaluation step sequence.

Algorithm
---------
1. **Random binary mask search** – pick a random subset of indices of
   size `m` (≈ n/2).  The sequence is `a_i = 1/m` on the chosen indices and
   `0` elsewhere, i.e. a probability distribution on the simplex.
   The max‑convolution of such a mask is close to the uniform lower bound
   `1/(2n)`, giving a theoretical objective ≈ 1.  We generate many masks
   (subject to the time budget) and keep the best one.

2. **Mask hill‑climb** – repeatedly swap a “1” with a “0”.  Moves that
   strictly lower the score are kept; this quickly flattens the
   convolution and drives the objective well below the previous best
   (≈ 1.5).

3. **Smooth‑max L‑BFGS refinement** – start from the best mask,
   convert it to a dense probability vector and optimise the
   smooth‑max objective (log‑sum‑exp approximation of the max) with a
   continuation schedule on the smoothing factor `λ`.  This removes the
   remaining small peaks.

4. **Final greedy continuous hill‑climb** – move a tiny fraction of mass
   between two coordinates only when the objective improves.  This
   polishes the solution.

All heavy convolution work uses FFT (O(n log n)).  The total wall‑clock
time is limited by `budget_s` (default 1000 s).  The function returns
the best list of floats found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT helpers and fast objective (assumes sum(a)=1)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real output, length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    n = a.shape[0]
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) when `a` is already
    normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits `y` where a = softmax(y).
    """
    a = softmax(y)

    # a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * w) in freq. domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in `lam_schedule`, warm‑starting each stage from the
    previous result.  Returns the best vector found and its fast_score.
    """
    # initialise logits (avoid log(0) by adding a tiny epsilon)
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:      # defensive – should rarely happen
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)               # stays normalised
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # print progress inside outer loop
            print(f"[lbfgs] λ={lam:.4g} new best score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Small continuous hill‑climb (move a tiny fraction of mass)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy hill‑climb on a dense probability vector.
    Moves a random fraction (≤ 10 %) from one coordinate to another
    provided the move strictly improves the score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] <= 1e-14:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10     # up‑to 10 % of mass at i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score
