def improve_via_subgrad(a, fft_len, n, deadline):
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t=0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t+=1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # compute subgradient
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # Determine step size base
        alpha0 = 0.5 / np.sqrt(t)  # maybe larger to start
        # line search:
        alpha = alpha0
        # try several reductions
        improved=False
        for _ in range(10):
            a_candidate = project_onto_simplex(a - alpha * g)
            cand_score = fast_score(a_candidate, fft_len, n)
            if cand_score < best_score - 1e-12:   # improvement
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved=True
                break
            else:
                alpha *= 0.5
        if not improved:
            # optional: keep a unchanged
            # occasionally reduce step if not improving

                # reset to start again

def subgradient_descent_refine(a, fft_len, n, deadline):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    t = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # compute subgrad
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # step size schedule
        # We can use larger initial step size.
        alpha = 0.3 / np.sqrt(t)
        # line search: try up to 5 reductions
        improved = False
        for _ in range(5):
            a_candidate = project_onto_simplex(a - alpha * g)
            cand_score = fast_score(a_candidate, fft_len, n)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # Occasionally random perturbation to escape
            if rng.random() < 0.01:

"""
A fast, time‑limited optimizer for the max‑convolution objective.

The algorithm works in three phases on a sequence of length *n* (taken
from the previous best construction if available, otherwise 5000):
1. **Projected sub‑gradient descent with line‑search** – uses the exact
   sub‑gradient of the max‑convolution and a back‑tracking line search,
   repeatedly projecting onto the probability simplex.
2. **Simulated‑annealing hill‑climb** – randomly moves a fraction of
   mass between buckets, accepting worsening moves with a temperature
   that cools linearly.
3. **Deterministic greedy hill‑climb** – repeatedly attempts small
   mass transfers that strictly improve the fast (FFT‑based) objective.

All phases respect the global time budget (default 1000 s).  The final
sequence is returned as a plain list of non‑negative floats; it is
already normalised (∑a_i = 1) so the scaling‑invariant evaluation metric
is unchanged.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def subgradient_descent_refine(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with back‑tracking line search.
    Returns the best vector seen before ``deadline``.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # basic step‑size schedule (you can tune the constant)
        alpha = 0.5 / np.sqrt(it)

        # back‑tracking line search (up to 5 reductions)
        improved = False
        for _ in range(5):
            a_candidate = project_onto_simplex(a - alpha * g)
            cand_score = fast_score(a_candidate, fft_len, n)
            if cand_score < best_score - 1e-12:
                best = a_candidate.copy()
                best_score = cand_score
                a = a_candidate
                improved = True
                break
            else:
                alpha *= 0.5

        if not improved:
            # occasional small random "kick" to escape flat regions
            if rng.random() < 0.005:
                a = best + 0.02 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = project_onto_simplex(a)

    return best, best_score


def annealing_refine(
    a: np.ndarray,
    fft_len: int,
    n: int,
    phase_start: float,
    phase_end: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style mass redistribution.
    A temperature that decreases linearly from 0.1 to 0.001 is used.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()
    max_move = 0.05  # start with up to 5 % of the source bucket
    total_time = phase_end - phase_start
    it = 0

    while time.time() < phase_end:
        it += 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            # linear cooling
            now = time.time()
            elapsed = now - phase_start
            temp = 0.1 * (1.0 - elapsed / total_time) + 0.001
            prob = np.exp(-(cand_score - best_score) / temp)
            if rng.random() < prob:
                best = cand
                best_score = cand_score

        # slowly shrink the move magnitude
        if it % 8000 == 0:
            max_move *= 0.9
            max_move = max(max_move, 0.001)

    return best, best_score


def hill_climb_refine(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_move: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Deterministic greedy hill‑climb.
    Moves a random fraction of mass from one index to another,
    accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

        best_a, fft_len, n, deadline, max_move=0.04
    )
    print(f"[final HC] fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided harness
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # evaluate_sequence is supplied externally
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
