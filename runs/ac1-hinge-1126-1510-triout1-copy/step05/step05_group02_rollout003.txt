"""
Propose a low‑score non‑negative step sequence.

The optimizer works in five stages (all respecting Σa_i = 1):
1️⃣  Massive multi‑restart L‑BFGS on a smooth log‑sum‑exp surrogate of the
    max‑convolution (continuation on decreasing smoothing parameters).
2️⃣  Exponentiated‑gradient (mirror descent) on the exact nonsmooth objective.
3️⃣  Adam fine‑tuning on the same smooth surrogate (tiny smoothing → almost exact).
4️⃣  Short simulated‑annealing phase for escaping shallow basins.
5️⃣  Final greedy hill‑climb (pairwise mass transfers) that uses all remaining time.

All stages are time‑controlled; the total budget is 1000 s.
The search operates on the full allowed length (n = 4000) which gives a
tighter theoretical lower bound.  The code is self‑contained and uses only
numpy / scipy.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) for a distribution already
    normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialiser (diverse families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sine",
            "halfsine",
            "spike",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sine":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    elif mode == "halfsine":
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
    else:  # spike – a single tall peak plus a small uniform background
        a = np.full(n, 1e-4)
        a[rng.integers(0, n)] = 1.0
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Smoothed maximum (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth‑max surrogate based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth maximum ---------------------------------------------------
    max_c = conv_full.max()
    lam = max(max_c * lam_factor, 1e-14)  # avoid zero smoothing
    shifted = conv_full - max_c            # ≤ 0
    # protect against overflow/underflow
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. convolution entries -----------------
    w = exp_shift / Z  # d smooth_max / d conv_i, shape (2n-1,)

    # ----- back‑propagation through convolution (a*a) ----------------------
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice the part that corresponds to the derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through the soft‑max ---------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Return the best sequence found.
    """
    # initialise logits
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            # If L‑BFGS fails, keep the current y and continue
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_mirror_descent(a0: np.ndarray, deadline: float, fft_len_conv: int, n: int):
    """
    Projected exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())  # shift with current maximum

        # sub‑gradient for the max shift
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        # decreasing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update & renormalisation
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed log‑sum‑exp objective
# ----------------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    lam_factor: float = 0.003,
    lr: float = 0.12,
):
    """
    Adam optimisation of the smoothed max objective starting from logits y0.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (random mass transfers)
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray, score: float, fft_len: int, n: int, deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.025  # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12  # up to 12 % of mass at i
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 5000 == 0:
            T *= 0.93
            T = max(T, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 4 % of the mass currently at i
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000
    n = max_n                       # use the full allowed length
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient convolutions

    # ------------------------------------------------------------------
    # Initialise a starting point (warm‑start from a previous run if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            a0 = np.zeros(n)
            a0[: init_seq.size] = np.maximum(init_seq, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    phase1_end = start_time + 0.55 * budget_s  # L‑BFGS
    phase2_end = start_time + 0.75 * budget_s  # mirror‑descent
    phase3_end = start_time + 0.90 * budget_s  # Adam
    phase4_end = start_time + 0.97 * budget_s  # annealing

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with continuation on λ
    # ------------------------------------------------------------------
    lam_schedule = [
        0.30,
        0.20,
        0.15,
        0.10,
        0.07,
        0.05,
        0.03,
        0.018,
        0.010,
        0.006,
        0.003,
        0.001,
        0.0005,
        0.0002,
    ]
    maxiter_per_lam = 400

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_mirror_descent(
            best_a, phase2_end - 0.5, fft_len_conv, n
        )
        print(f"[exp‑mirror] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning on the smooth surrogate
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(
            logits,
            n,
            fft_len_conv,
            fft_len_g,
            phase3_end - 0.5,
            lam_factor=0.001,
            lr=0.18,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = random_anneal(
            best_a, best_score, fft_len_conv, n, phase4_end - 0.5
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (uses the rest of the budget)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=12000,
    )
    print(f"[final hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided black‑box function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # external function supplied by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
