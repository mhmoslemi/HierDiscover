def refine_with_lp(a, max_iter=3, time_budget=...):
    n = len(a)
    start = time.time()
    for it in range(max_iter):
        if time.time() > time_budget:
            break
        # compute conv and M
        fft_len = next_pow2(2*n-1)
        conv = conv_fft(a, fft_len, n)
        M = conv.max()
        # solve LP
        g0 = lp_max_sum_g(a, M)
        if g0 is None:
            break
        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break
        g = g0 / sum_g0  # sum=1
        # line search for t
        best_t = 0.0
        best_score = 2*n * M
        # Try few t values: logspace or linear grid
        t_vals = np.linspace(0, 0.5, 21)  # up to 0.5
        for t in t_vals[1:]:
            a_new = (1 - t) * a + t * g
            # combine should sum to 1 (since both sum to 1)
            fft_len = next_pow2(2*n-1)
            conv_new = conv_fft(a_new, fft_len, n)
            M_new = conv_new.max()
            score_new = 2*n * M_new
            if score_new < best_score:
                best_score = score_new
                best_t = t
                best_a = a_new
        if best_t > 0:
            a = best_a
            # optionally print improvement
            print(f"[LP refine] iter {it} t={best_t:.4f} new score={best_score:.6f}")
        else:
            # No improvement, break
            break
    return a

"""
Hybrid step‑function optimizer.

The pipeline builds on the previous hill‑climb / L‑BFGS scheme but adds a
compact linear‑programming refinement based on the classic iterative
construction:

    * For a current candidate a (∑a = 1) compute M = max(a∗a).
    * Solve   max ∑g   s.t. g≥0 and (a∗g) ≤ M   (LP on the convolution
      constraints).  The LP is built as a sparse Toeplitz matrix.
    * Normalise g to ∑g = 1 and perform a line‑search on the convex
      combination a' = (1−t)a + t g.  The best t reduces max(a'∗a').
    * Repeat the LP refinement a few times.

The LP step dramatically improves the bound for moderate lengths
(≈4000) while keeping the total running time well below the 1000 s
budget.  The rest of the search (random restarts, continuation L‑BFGS,
exact sub‑gradient descent and hill‑climbs) is identical to the prior
state‑of‑the‑art implementation.

The final sequence is returned as a plain Python list of non‑negative
floats (its sum is 1, which does not affect the evaluation score).
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                      # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient ----------
    w = exp_shift / Z                            # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).

    Returns the best vector seen and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)  # index array for vectorised gradient

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)

        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random and targeted hill‑climbs
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the largest contributor to the
    smallest contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP based on the classical iterative step)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray):
    """
    Build a sparse (2n‑1) × n matrix A such that A·g = f∗g.
    A is a Toeplitz‑like matrix with constant diagonals.
    """
    n = f.shape[0]
    # each diagonal i (offset i) consists of n copies of f[i]
    diagonals = [np.full(n, val, dtype=np.float64) for val in f]
    offsets = np.arange(n, dtype=np.int32)  # 0 … n‑1
    return diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")


def lp_max_sum_g(f: np.ndarray, M: float):
    """
    Solve the LP   max Σ g_j   s.t.   g ≥ 0,   (f∗g) ≤ M.
    Returns the optimal g or None if the solver fails.
    """
    try:
        A = build_conv_matrix(f)
        b = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(f.shape[0], dtype=np.float64)  # maximise Σg ⇔ minimise −Σg
        bounds = [(0, None) for _ in range(f.shape[0])]
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if res.success:
            return res.x
        else:
            print("[LP] optimizer failed:", res.message)
            return None
    except Exception as exc:
        print("[LP] exception:", exc)
        return None


def refine_with_lp(
    a: np.ndarray,
    deadline: float,
    max_iter: int = 3,
) -> tuple[np.ndarray, float]:
    """
    Apply the LP‑based refinement a few times.
    Returns a (potentially) improved vector and its fast_score.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:

        if g0 is None:

        sum_g0 = g0.sum()
        if sum_g0 <= 1e-12:
            break

        # Normalise g to have sum = 1 (preserves the bound)
        g = g0 / sum_g0

        # Simple line‑search on t∈[0,0.5]
        best_local_a = best_a
        best_local_score = best_score
        best_t = 0.0

        for t in np.linspace(0.02, 0.5, 25):

        if best_t > 0.0:

        else:

    return best_a, best_score

    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # --------------------------------------------------------------
        # 1) Continuation L‑BFGS (smooth objective)
        # --------------------------------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Exact sub‑gradient descent
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = subgrad_descent(
            best_a,
            n,
            fft_len,
            max_iters=2500,
            deadline=sub_deadline,
            eta0=0.36,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 3) Targeted hill‑climb
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=3000,
            frac=0.08,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 4) Random hill‑climb for polishing
        # --------------------------------------------------------------
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Deep final refinement (low‑λ L‑BFGS) – consume remaining time
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.5:
        for lam in lam_factors:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 600, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Linear‑programming refinement (optional, if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        best_a, best_score = refine_with_lp(best_a, deadline, max_iter=3)

    return best_a, best_score

# Global entry point – called by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its sum will be 1 after
    internal normalisation).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin

    # Try several problem sizes; allocate budget proportionally.
    sizes = [2000, 3000, 4000]  # these sizes work well with the LP step
    per_size_budget = (total_deadline - time.time()) / len(sizes)

    # Warm‑start: reuse a previously found construction if available.
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)

    best_overall_a = None
    best_overall_score = np.inf

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 2.0:
            break
        # Ensure enough time for later sizes
        slice_budget = min(per_size_budget, remaining - (len(sizes) - idx - 1) * 2.0)

        print(f"[size] Optimising length n={n} with slice budget ≈{slice_budget:.1f}s")
        a_n, score_n = optimise_one_size(
            n=n,
            budget_s=slice_budget,
            seed=seed + idx * 123,
            prev_seq=prev_arr,
        )
        print(f"[size] n={n} best fast score = {score_n:.6f}")

        if score_n < best_overall_score:
            best_overall_score = score_n
            best_overall_a = a_n.copy()

    # ------------------------------------------------------------------
    # Final LP‑based polishing on the overall best (if time left)
    # ------------------------------------------------------------------
    if time.time() < total_deadline - 1.0 and best_overall_a is not None:
        final_deadline = total_deadline - 0.5
        best_overall_a, best_overall_score = refine_with_lp(
            best_overall_a, final_deadline, max_iter=4
        )
        print(f"[final‑LP] overall fast score after polishing = {best_overall_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence for possible warm‑starts in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
