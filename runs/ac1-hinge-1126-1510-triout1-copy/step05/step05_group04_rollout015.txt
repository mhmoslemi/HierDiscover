def simulated_annealing(a, conv, cur_score, n, fft_len_conv, deadline):
    # a: probability simplex vector
    # conv: convolution a*a length 2n-1 (numpy array)
    # cur_score: 2*n*conv.max()
    # returns best a, best_score
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.3   # fraction of mass to transfer max
    # Temperature schedule: start = 0.05 * best_score; linear down to 0
    T0 = best_score * 0.05
    start = time.time()
    steps = 0
    re_sync = 0
    re_sync_interval = 5000
    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))
        # choose i based on current max conv index
        m = int(conv.argmax())
        low = max(0, m - (n - 1))
        high = min(n - 1, m)
        idx_range = np.arange(low, high + 1)
        # weights = 2*a[m - idx_range] except diagonal
        w = 2.0 * a[m - idx_range]
        if m % 2 == 0:
            mid = m // 2
            if low <= mid <= high:
                w[mid - low] = a[mid]  # replace 2*a[mid] with a[mid]
        w_sum = w.sum()
        if w_sum <= 0 or np.isnan(w_sum):
            i = rng.integers(0, n)
        else:
            probs = w / w_sum
            i = rng.choice(idx_range, p=probs)
        # pick j different from i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        a_i_old = a[i]
        a_j_old = a[j]
        if a_i_old == 0:
            continue
        delta = a_i_old * rng.random() * max_move
        if delta <= 0:
            continue

        # apply changes (in-place)
        conv[i:i+n] -= 2.0 * delta * a
        conv[j:j+n] += 2.0 * delta * a
        conv[2*i] += delta * a_i_old
        conv[2*j] -= delta * a_j_old

        a[i] = a_i_old - delta
        a[j] = a_j_old + delta

        # compute new score
        new_score = 2.0 * n * conv.max()
        # Accept?
        accept = False
        if new_score < best_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - best_score) / T):
                accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            # keep
        else:
            # revert a and conv
            a[i] = a_i_old
            a[j] = a_j_old
            # revert conv using current a (now restored)
            conv[i:i+n] += 2.0 * delta * a
            conv[j:j+n] -= 2.0 * delta * a
            conv[2*i] -= delta * a_i_old
            conv[2*j] += delta * a_j_old
        steps += 1
        re_sync += 1
        if re_sync >= re_sync_interval:
            # recompute conv from scratch to avoid drift
            conv = conv_fft(a, fft_len_conv, n)
            re_sync = 0
        # optional early break if best_score already < target
        if best_score < 1.503:
            break
    return best_a, best_score

"""Optimization of step function to minimize evaluate_sequence.

Uses multi‑restart L‑BFGS with log‑sum‑exp smoothing to obtain a good
initial distribution, then a custom simulated‑annealing hill‑climber
that updates the convolution incrementally (O(n) per move).  The
algorithm respects the simplex constraint and keeps the exact objective
(`2·n·max(conv)`) available for fast acceptance decisions.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ---- utilities ---------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    v = v.astype(float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ---- smoothed objective for L‑BFGS -------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z

    # gradient of smooth max w.r.t. conv
    # compute gradient of conv w.r.t. a via reverse convolution
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# ---- generation of starting points ------------------------------------

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums or 2 * candidate in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

# ---- basic hill climb (greedy) ----------------------------------------

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        # compute fast score after the move using full FFT (simple but safe)
        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score:
            best = new_a
            best_score = new_score
    return best, best_score

# ---- projected sub‑gradient descent ------------------------------------

def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

# ---- simulated annealing with incremental convolution update ------------

def simulated_annealing(
    a: np.ndarray,
    conv: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Custom annealer that moves a fraction of mass from one index to another.
    The convolution is updated in O(n) using the linearity of the operation."""
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.3  # up to 30 % of the source mass per move
    start = time.time()
    re_sync = 0
    re_sync_interval = 4000
    # temperature linearly decreasing from 5 % of the current best score to 0
    T0 = best_score * 0.05
    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))
        # index of current maximal convolution entry
        m = int(conv.argmax())
        low = max(0, m - (n - 1))
        high = min(n - 1, m)
        idxs = np.arange(low, high + 1)
        # contribution weights (gradient of the active peak)
        w = 2.0 * a[m - idxs]
        if m % 2 == 0:
            mid = m // 2
            if low <= mid <= high:
                w[mid - low] = a[mid]   # diagonal term has weight a[mid] instead of 2*a[mid]
        w_sum = w.sum()
        if w_sum <= 0 or np.isnan(w_sum):
            i = rng.integers(0, n)
        else:
            probs = w / w_sum
            i = rng.choice(idxs, p=probs)
        # pick a destination different from i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        a_i_old = a[i]
        a_j_old = a[j]
        if a_i_old <= 0.0:
            continue
        delta = a_i_old * rng.random() * max_move
        if delta <= 0.0:
            continue

        # --- apply the move -------------------------------------------------
        # use the current a (before change) for the convolution update
        conv[i : i + n] -= 2.0 * delta * a
        conv[j : j + n] += 2.0 * delta * a
        conv[2 * i] += delta * a_i_old
        conv[2 * j] -= delta * a_j_old

        a[i] = a_i_old - delta
        a[j] = a_j_old + delta

        new_score = 2.0 * n * conv.max()
        accept = False
        if new_score < best_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - best_score) / T):
                accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] = a_i_old
            a[j] = a_j_old
            # revert convolution using the restored a (which equals the original a)
            conv[i : i + n] += 2.0 * delta * a
            conv[j : j + n] -= 2.0 * delta * a
            conv[2 * i] -= delta * a_i_old
            conv[2 * j] += delta * a_j_old

        re_sync += 1
        if re_sync >= re_sync_interval:
            conv = conv_fft(a, fft_len, n)
            re_sync = 0
        # early exit if target already reached
        if best_score < 1.503:
            break
    return best_a, best_score

"""
Optimization of a non‑negative step function to minimise the supplied
evaluate_sequence() objective.

The algorithm proceeds in four stages:

1. **Multi‑restart L‑BFGS** on a smooth log‑sum‑exp surrogate of the
   true objective (2·n·max(conv)).  This gives a strong initial
   distribution.

2. **Projected sub‑gradient descent** (short) for extra diversification.

3. **Simulated annealing** that moves a fraction of mass from one
   index to another while updating the convolution incrementally
   (O(n) per move).  Temperature is linearly cooled; the true objective
   (max convolution) is used for acceptance decisions.

4. **Aggressive greedy hill‑climb** to polish the final result.

All operations respect the simplex constraint (non‑negative entries,
sum = 1).  The best sequence obtained before the time limit is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:

    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]

    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # derivative of conv w.r.t. a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing parameter.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Initialisation helpers

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a pre‑softmax vector for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set (size ≈ √n)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if cand + c in sums or 2 * cand in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that only keeps improving moves."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score:
            best = new_a
            best_score = new_score
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def simulated_annealing(
    a: np.ndarray,
    conv: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement with an *incremental* convolution update.
    Each move shifts a fraction of mass from i → j and updates `conv`
    in O(n).  Temperature cools linearly from 5 % of the current
    best score to 0.
    """
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.30
    start = time.time()
    re_sync = 0
    re_sync_interval = 4000
    T0 = best_score * 0.05  # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / (deadline - start))

        # index of the current maximal convolution entry
        m = int(conv.argmax())
        low = max(0, m - (n - 1))
        high = min(n - 1, m)
        idxs = np.arange(low, high + 1)

        # gradient‑like weights for the active peak
        w = 2.0 * a[m - idxs]
        if m % 2 == 0:
            mid = m // 2
            if low <= mid <= high:
                w[mid - low] = a[mid]   # diagonal term
        w_sum = w.sum()
        if w_sum <= 0 or np.isnan(w_sum):
            i = rng.integers(0, n)
        else:
            probs = w / w_sum
            i = rng.choice(idxs, p=probs)

        # destination j (different from i)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        a_i_old = a[i]
        a_j_old = a[j]
        if a_i_old <= 0.0:
            continue
        delta = a_i_old * rng.random() * max_move
        if delta <= 0.0:
            continue

        # ---- apply the move (using the *current* a for the update) ----
        conv[i : i + n] -= 2.0 * delta * a
        conv[j : j + n] += 2.0 * delta * a
        conv[2 * i] += delta * a_i_old
        conv[2 * j] -= delta * a_j_old

        a[i] = a_i_old - delta
        a[j] = a_j_old + delta

        new_score = 2.0 * n * conv.max()
        accept = False
        if new_score < best_score:
            accept = True
        else:
            if T > 0 and rng.random() < np.exp(-(new_score - best_score) / T):
                accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # ---- revert -------------------------------------------------
            a[i] = a_i_old
            a[j] = a_j_old
            conv[i : i + n] += 2.0 * delta * a
            conv[j : j + n] -= 2.0 * delta * a
            conv[2 * i] -= delta * a_i_old
            conv[2 * j] += delta * a_j_old

        re_sync += 1
        if re_sync >= re_sync_interval:
            conv = conv_fft(a, fft_len, n)
            re_sync = 0

        if best_score < 1.503:   # early stop if target already reached
            break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The algorithm consists of an L‑BFGS warm‑start,
    a short sub‑gradient phase, simulated annealing with incremental
    convolution updates, and a final greedy hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise probability vector (sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = np.maximum(tmp, 0.0)
            if a.sum() == 0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / a.sum()
        else:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    conv = conv_fft(a, fft_len_conv, n)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Multi‑restart L‑BFGS phase (≈ 55 % of budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=40))
    maxiter_per_stage = 800
    modes = ["random", "exp", "power", "edge", "sidon", "binary", "bump"]
    main_deadline = start_time + 0.55 * budget_s
    iter_cnt = 0
    while time.time() < main_deadline:
        mode = rng.choice(modes)
        y0 = generate_initial_y(mode, n, rng)
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        conv_opt = conv_fft(a_opt, fft_len_conv, n)
        score_opt = 2.0 * n * conv_opt.max()
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iter_cnt:04d}] mode={mode} improved fast score = {best_score:.6f}")
        iter_cnt += 1

    # --------------------------------------------------------------
    # Short projected sub‑gradient descent (≈ 15 % of budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.75 * budget_s)
    if time.time() < subgrad_deadline:
        print("[subgrad] projected sub‑gradient descent")
        best_a, best_score = subgradient_descent(
            best_a, fft_len_conv, n, subgrad_deadline, eta0=0.30, max_iters=200_000
        )
        print(f"[subgrad] best fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[anneal] simulated annealing")
        a = best_a.copy()
        conv = conv_fft(a, fft_len_conv, n)
        cur_score = 2.0 * n * conv.max()
        best_a, best_score = simulated_annealing(
            a, conv, cur_score, n, fft_len_conv, deadline
        )
        print(f"[anneal] best fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final aggressive hill‑climb (use any time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill] aggressive hill climb")
        a = best_a.copy()
        conv = conv_fft(a, fft_len_conv, n)
        cur_score = 2.0 * n * conv.max()
        a, best_score = hill_climb(
            a,
            cur_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
            max_move=0.30,
        )
        best_a = a
        print(f"[final hill] fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
