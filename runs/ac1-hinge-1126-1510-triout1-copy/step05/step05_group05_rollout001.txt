"""
Propose a low‑score step‑function.

The algorithm combines several cheap parametric families (including
new symmetric exponential / power‑law shapes) to collect a few promising
seeds, then refines them with a continuation L‑BFGS on a smoothed
max‑convolution objective, a long projected sub‑gradient descent on the
true max‑convolution, and a final hill‑climb / annealing polish.
All iterates stay on the probability simplex, so the scale of the
vector does not matter for the true score.
The search respects a 1000 s budget and prints progress.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1.  This equals the true score because the ratio is
    scale‑invariant.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on the log‑sum‑exp trick.
    Returns (objective, gradient) w.r.t. y (the soft‑max pre‑image).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smoothed max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    # derivative of convolution wrt a (a is symmetric)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the central n entries (index n‑1 … 2n‑2)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (on the smoothed objective)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
        # short hill‑climb after each λ (lightweight)
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via projected sub‑gradient descent.
    Uses a diminishing step size (γ / sqrt(t)) with a simple back‑tracking
    line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0
    gamma = 1.0        # initial step‑size multiplier
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max (see derivation in the comment above)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = gamma / np.sqrt(it + 1.0)
        improved = False
        # simple back‑tracking (up to 4 reductions)
        for _ in range(4):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer at most 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T = max(T * 0.92, 1e-5)

    return best_a, best_score

    add_candidate(sin_arr + 1e-3)

    # 9) V‑shaped (linear increase from centre to edges)
    centre = (n - 1) / 2.0
    v_arr = np.abs(np.arange(n) - centre) + 1.0
    v_arr = 1.0 / v_arr
    add_candidate(v_arr)

    # 10) Random Dirichlet (Gamma) draws with various concentration
    for alpha in [0.3, 0.5, 0.8, 1.0, 2.0, 4.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Include previously discovered sequence if present
    if 'height_sequence_1' in globals():
        add_candidate(np.array(globals()['height_sequence_1'], dtype=float))

    # Keep the top few seeds (they are already normalised)
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.7   # safety margin
    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS phase
    phase2_end = start_time + 0.78 * budget_s   # sub‑gradient phase
    final_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.06, 0.03,
        0.015, 0.008, 0.004, 0.002, 0.001,
        5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5
    ]
    maxiter_per_lam = 250   # a bit tighter to allow more restarts

    restarts = 0
    while time.time() < phase1_end and restarts < 80:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

        restarts += 1

        # early exit if already under target
        if best_score <= 1.503:
            print("[early] target reached during L‑BFGS")
            break

    # ------------------------------------------------------------------
    # Phase 2 – long projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.2)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing + final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < final_deadline:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           final_deadline)
        print(f"[anneal] score={best_score:.6f}")

    # final greedy polishing
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    # ------------------------------------------------------------------
    # Return the sequence (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
