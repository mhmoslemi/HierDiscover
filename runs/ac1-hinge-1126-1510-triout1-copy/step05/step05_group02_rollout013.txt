"""
Propose a low‑score non‑negative step sequence.

The optimizer works in five stages (≈1000 s total):

1. **Multi‑restart smoothed L‑BFGS** – a soft‑max approximation of the
   max‑convolution is minimised.  The smoothing factor (lam) is reduced
   geometrically in a continuation scheme; each restart starts from a
   random distribution on the simplex.

2. **Exact sub‑gradient descent** – the true max‑convolution is reduced
   using an exponentiated‑gradient (mirror‑descent) update.  All
   “almost‑maximal’’ shifts (within a small relative tolerance) are
   used to form a sub‑gradient, which gives a stronger descent direction.

3. **Fine‑tuning with Adam** on the smoothed objective (log‑sum‑exp).

4. **Simulated annealing** – random mass transfers are accepted with a
   temperature schedule to escape shallow local minima.

5. **Greedy hill‑climb + occasional swaps** – small deterministic
   moves and occasional random re‑ordering of a few coordinates are
   performed while strictly decreasing the fast score.

All steps keep the vector on the probability simplex (sum = 1) and
respect the non‑negativity constraints of `evaluate_sequence`.  The
length is set to the maximal allowed value (4000) to obtain the lowest
possible theoretical bound (≈1.0005).  The best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)                     # a ≥ 0, sum = 1

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)        # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            # ignore failures, keep current y
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated mirror descent)
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        deadline: float,
                        fft_len_conv: int,
                        n: int,
                        tol: float = 1e-5):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Uses all near‑maximal shifts (within (1+tol) of the current max) to
    build a valid sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        max_val = conv.max()
        # indices whose value is within (1+tol) of the max
        idxs = np.where(conv >= max_val * (1.0 - tol))[0]

        # sub‑gradient accumulation
        g = np.zeros_like(a)
        for k in idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i = np.arange(lo, hi + 1)
            j = k - i
            g[i] += 2.0 * a[j]

        # step size – decaying sqrt schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated (multiplicative) update
        a = a * np.exp(-eta * g)

        # renormalise to sum = 1
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # evaluate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed log‑sum‑exp objective
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective.
    Operates on logits (log‑soft‑max) so that the simplex constraint is
    satisfied automatically.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer + occasional swaps)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb with optional random swaps.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        # ----- move a small fraction of mass -----
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07   # up to 7 % of i‑mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        # ----- occasional swap of two positions (10 % of it) -----
        if (it % 10) == 0:
            s, t = rng.integers(0, n, size=2)
            if s != t:
                cand = best.copy()
                cand[s], cand[t] = cand[t], cand[s]
                cand_score = fast_score(cand, fft_len, n)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score

        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (mass transfers with temperature decay)
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing that moves a random fraction of mass between
    two coordinates.  Worsening moves are accepted with a temperature
    that decays geometrically.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.03                     # start temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # up to 12 % of i‑mass
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)

        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T = max(T * 0.89, 1e-6)   # geometric cooling
    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialiser (adds a sparse family)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector of length n.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sine', 'halfsine', 'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    elif mode == 'halfsine':
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
    else:                     # sparse – a few random positions carry all mass
        sparse_sz = max(3, int(np.sqrt(n)))          # O(√n) non‑zeros
        a = np.zeros(n)
        positions = rng.choice(n, size=sparse_sz, replace=False)
        a[positions] = rng.random(sparse_sz) + 0.1   # avoid exact zeros
    # normalise
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use the maximal allowed length)
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n                    # we keep the full length

    # warm‑start from a previously found construction if available
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size >= n:
            a0 = init_seq[:n].copy()
        else:
            a0 = random_initial(n, rng)
    else:
        a0 = random_initial(n, rng)

    # ensure feasibility
    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    a0 = a0 / s0 if s0 > 0 else np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient‑related FFTs

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    phase1_end = start_time + 0.30 * budget_s       # L‑BFGS restarts
    phase2_end = start_time + 0.60 * budget_s       # sub‑gradient descent
    phase3_end = start_time + 0.80 * budget_s       # Adam fine‑tune
    phase4_end = start_time + 0.95 * budget_s       # annealing + hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – many L‑BFGS continuations
    # --------------------------------------------------------------
    lam_schedule = np.geomspace(0.5, 0.001, num=18)   # smoother → sharper
    maxiter_per_lam = 1000
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exact sub‑gradient (mirror) descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_descent(best_a,
                                                 phase2_end,
                                                 fft_len_conv,
                                                 n,
                                                 tol=0.001)
        print(f"[sub‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning on smoothed log‑sum‑exp
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           phase3_end,
                                           lam_factor=0.002,
                                           lr=0.15)
        print(f"[Adam] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase4_end)
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb with swaps
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    print(f"[final hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the result (scaled back to raw list for evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # The external evaluator works with any scaling; we keep sum=1.
    try:
        final_eval = evaluate_sequence(final_seq)   # type: ignore
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        # In case the grader does not expose evaluate_sequence here
        final_eval = None
        print("[final] could not call evaluate_sequence:", e)

    return final_seq
