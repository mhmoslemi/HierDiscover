def exponentiated_gradient_descent(a_start, fft_len, n, deadline, eta0=0.2, min_eta=1e-5):

   a = a_start

   best_a = a

   best_score = fast_score(a)

   t = 0

   while time < deadline:

      t+=1

      conv = conv_fft(a, fft_len, n)

      m = argmax conv

      # subgradient g_i = 2*a_{m-i}
      idx = np.arange(n)
      counterpart = m - idx
      mask = (counterpart >= 0) & (counterpart < n)

      g = np.zeros_like(a)

      g[mask] = 2.0 * a[counterpart[mask]]

      # update multiplicatively

      eta = eta0 / np.sqrt(t)

      a_new = a * np.exp(-eta * g)

      a_new = a_new / a_new.sum()  # ensures sum 1

      new_score = fast_score(a_new)

      if new_score < best_score:

          best_score = new_score

          best_a = a_new.copy()

      # maybe accept if new_score < current_score else keep old?

      if new_score < fast_score(a):   # improvement

          a = a_new

      else:

          # optionally random perturb

          a = a_new  # maintain progress? hmm.

   return best_a, best_score

"""
Propose a low‑score step‑function sequence.

Algorithm
---------
* **Representation** – a probability vector `a` (non‑negative, sum = 1) is
  stored internally as `softmax(y)`.  The score that the harness uses is
  `2·n·max(conv)` where `conv = a * a` (linear convolution);
  this is scale‑invariant, so we may optimise directly on the simplex.

* **Phase 1 – smoothed L‑BFGS** – many random starts are optimised with a
  continuation scheme that minimises a log‑sum‑exp smooth approximation
  of the maximum convolution value.  The smoothing factor `λ` is reduced
  geometrically over 60 stages; each stage runs L‑BFGS with up to 800
  iterations.

* **Phase 2 – exponentiated‑gradient descent** – the true non‑smooth
  objective `max(conv)` is tackled with an exponentiated‑gradient (mirror
  descent) update on the simplex:
        a ← a ⊙ exp(−η·g)  ,   g the sub‑gradient at the active shift,
  followed by normalisation.  The stepsize η decays as 1/√t.

* **Phase 3 – targeted hill‑climb** – the shift `m` that attains the
  current maximum is identified, the most damaging pair of indices
  `(i,m−i)` is located and a small amount of mass is transferred away
  from that pair.  Random moves are accepted only if the score improves.

* **Phase 4 – final greedy hill‑climb** – a classic random mass‑transfer
  hill‑climb finishes the search with the remaining time budget.

The routine respects the 1000 s time limit, prints progress, and returns
a plain list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a of length 2·n‑1 using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective (2·n·max(a*a)) for a probability
    vector (sum = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max (log‑sum‑exp)
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z

    # propagate through convolution (g_vec = ∂conv/∂a)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def exp_gradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    min_eta: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using an exponentiated‑gradient update.
    The sub‑gradient is taken at the shift that currently attains the
    maximum of the convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # Sub‑gradient of max(conv) w.r.t. a
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = max(eta0 / np.sqrt(t), min_eta)

        # multiplicative update and re‑normalisation
        a_new = a * np.exp(-eta * g)
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            # numerical safeguard – fall back to uniform
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / a_new_sum

        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        # Accept only improving steps (otherwise keep the old a)
        if new_score < fast_score(a, fft_len, n):
            a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (focus on the worst convolution shift)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass away from the most damaging
    pair (i, m‑i) where m is the shift attaining max(conv).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # contribution of each i to the active shift
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[counterpart[mask]]

        i = int(np.argmax(contrib))
        if best[i] == 0.0:
            continue

        # choose a destination that is unlikely to increase the same shift
        # (sample until we get a different pair)
        for _ in range(10):
            j = rng.integers(0, n)
            if j != i and not (m - j >= 0 and m - j < n and best[m - j] > 0):
                break
        else:
            j = (i + n // 2) % n  # deterministic fallback

        # move a fraction of the source mass
        delta = best[i] * rng.random() * 0.08  # ≤8 % of source
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Classical random hill‑climb (used for polishing)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy random hill‑climb: repeatedly move a random fraction of mass
    from i to j, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence()
    score.  The routine works entirely in memory, respects the time budget,
    and returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0  # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size (prefer a previously discovered sequence)
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = int(min(tmp.size, 10000))
            best_a = tmp / tmp.sum()
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=60))
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon"]
    iteration = 0

    # Reserve ~55 % of the total budget for this phase
    while time.time() < deadline - 250.0:
        mode = rng.choice(modes + ["best"])
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast = {best_score:.6f}")

        # A short random hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline - 250.0, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the true max‑conv
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        eg_deadline = deadline - 70.0
        print("[exp‑grad] start")
        best_a, best_score = exp_gradient_descent(
            best_a,
            fft_len_conv,
            n,
            eg_deadline,
            eta0=0.22,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[exp‑grad] finished → fast = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline - 80.0:
        th_deadline = deadline - 40.0
        print("[targeted HC] start")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            th_deadline,
            max_iter=8000,
        )
        print(f"[targeted HC] finished → fast = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final random hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # harness supplies this
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
