def lp_improve_scipy(a, n, deadline, t_init=0.1, max_iters=12):
    a = a.copy()
    a = a / a.sum()
    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a,a)
        M = conv.max()
        # Build convolution matrix T as sparse (CSC)
        # T rows = convolution indices 0..2n-2, columns 0..n-1.
        # For each column j, entries are a[i] where i indexes a[i] and goes to row i+j.
        from scipy.sparse import diags, csr_matrix
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n)  # diag offset = i
        T = diags(data, offsets, shape=(2*n-1, n), format='csc')
        # set up linprog
        # maximize sum b -> minimize -c^T b
        c = -np.ones(n)
        A_ub = T
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve':True,'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            # maybe try again with different method
            break
        b_val = np.maximum(res.x, 0.0)
        sum_b = b_val.sum()
        if sum_b <= 0:
            break
        b_scaled = b_val / sum_b
        a_new = (1 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()
        new_score = fast_score(a_new, n, next_pow2(2*n-1))
        old_score = fast_score(a, n, next_pow2(2*n-1))
        if new_score < old_score:
            a = a_new
            t = t_init
            it += 1
        else:
            t *= 0.5
            if t < 1e-4:
                break
    return a
```

"""
Multi‑stage optimisation for the step‑function evaluation metric.

The algorithm proceeds as follows:

1.  **Parametric sweep** – generate a variety of simple families (uniform,
    linear, exponential, power‑law, sinusoidal, edge‑heavy, beta‑like) and
    random Dirichlet draws.  The best few candidates are kept as seeds.

2.  **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp approximation of
    the max‑convolution using the soft‑max parametrisation.  A decreasing
    schedule of smoothing factors is used; after each stage a short
    greedy hill‑climb polishes the result.

3.  **LP balancing** – a linear programme (solved with SciPy’s HiGHS) maximises
    the ℓ₁‑norm of a new vector `b` under the constraint
    `(a * b)_k ≤ M` where `M` is the current peak of `a*a`.  Mixing a small
    proportion of the scaled `b` back into `a` provably reduces the peak.
    Several such iterations are performed.

4.  **Projected sub‑gradient descent** – direct minimisation of the exact
    max‑convolution using sub‑gradients and a simple back‑tracking line‑search.

5.  **Final hill‑climb** – greedy mass transfers between coordinates while
    preserving the simplex, for any remaining time.

All steps respect the simplex (`sum = 1`, `a_i ≥ 0`).  The final sequence is
returned as a plain list of floats; `evaluate_sequence` (provided by the
runner) is used only for the final score print‑out.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags, csc_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a (full length 2n‑1)
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c            # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv_full
    w = exp_shift / Z                      # ∂smooth_max / ∂conv_i

    # gradient w.r.t. a via convolution with reversed a
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    # relevant slice corresponds to indices n‑1 … n‑1+n‑1
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # quick hill‑climb after each λ
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=800)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent with a simple
    back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set mask
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # transfer up to 5% of the source mass
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# LP balancing (using SciPy's HiGHS)
# ----------------------------------------------------------------------
def lp_improve_scipy(a: np.ndarray,
                     n: int,
                     deadline: float,
                     t_init: float = 0.12,
                     max_iters: int = 12):
    """
    Perform a few iterations of the classic M‑balancing linear programme:

        maximise   Σ b_j
        s.t.       b_j ≥ 0,
                   (a * b)_k ≤ M   ∀k.

    The solution is mixed back into `a` with a small factor `t`.  The
    mixing factor is halved when a step fails to improve the objective.
    """
    a = a.copy()
    a = a / a.sum()
    t = t_init
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build convolution matrix T (size (2n‑1) × n) as a sparse CSC matrix.
        # Column j contains a shifted copy of `a` with offset j.
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear programme: minimise -1ᵀb  (i.e. maximise Σb)
        c = -np.ones(n)
        A_ub = T
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        # Time limit for the solver (HiGHS respects `time_limit` in seconds)
        opts = {}
        remaining = deadline - time.time()
        if remaining > 0:
            opts["time_limit"] = remaining

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method="highs", options=opts)

        if not res.success or res.x is None:
            # Solver failed – abort LP stage.
            break

        b_val = np.maximum(res.x, 0.0)
        sum_b = b_val.sum()
        if sum_b <= 0.0:
            break

        # Scale to stay on the simplex
        b_scaled = b_val / sum_b

        # Mix and renormalise
        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a, n, next_pow2(2 * n - 1))

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init   # reset mixing factor after success
            it += 1
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1

    return a

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a

    best_score = fast_score(best_a, n, fft_len_conv)

    # ------------------------------------------------------------------
    candidates = []

        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len_conv)
        candidates.append((sc, arr))

    # Uniform
    add_candidate(np.full(n, 1.0 / n))

    # Linear decreasing / increasing
    add_candidate(np.arange(n, 0, -1, dtype=float))
    add_candidate(np.arange(1, n + 1, dtype=float))

    # Exponential families (decay from left)
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # Edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # Quadratic (symmetric bell shape)
    quad = np.arange(n, dtype=float)
    quad = quad * (n - 1 - quad)
    add_candidate(quad + 1e-3)

    # Beta‑like random shapes (a_i ∝ (i/n)^α (1‑i/n)^β )
    for _ in range(6):
        α = rng.uniform(0.2, 2.0)
        β = rng.uniform(0.2, 2.0)
        t = (np.arange(n) + 0.5) / n
        arr = (t ** α) * ((1 - t) ** β)
        add_candidate(arr + 1e-4)

    # Random Dirichlet draws (Gamma / Gamma)
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(5):
