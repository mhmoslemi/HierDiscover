```
def generate_sid_set(n, max_attempts=100000):
    rng = np.random.default_rng()
    S = []
    sums = set()
    attempts=0
    while attempts<max_attempts and len(S) < int(2*math.sqrt(n)):
        candidate = rng.integers(0, n)

        new_sums = [candidate + x for x in S] + [candidate*2]

            sums.update(new_sums)
        attempts += 1
```

assistantfinal**Solution Overview**

The problem is to minimise  

\[
F(a)=\frac{2n\;\max_k (a\ast a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where \(a\ast a\) is the ordinary (linear) convolution.  
The factor \(\sum a_i\) can be fixed to 1 because the score is invariant to
global scaling (as long as no entry is clipped to 1000).  
Hence we have to minimise the largest entry of the
self‑convolution of a probability vector.

A good vector is **very sparse**: if the support set \(S=\{i\mid a_i>0\}\)
contains few indices and pairwise sums \(i+j\) are (almost) all different,
every convolution entry receives at most one product
\(a_i a_j\).  With uniform weights on a support of size \(m\)

\[
\max_k (a\ast a)_k\approx \frac{1}{m^2},
\qquad
F(a)\approx\frac{2n}{m^{2}} .
\]

Theoretically we can reach \(m\approx\sqrt{2n}\) while keeping collisions
to at most two, which gives a score close to the absolute lower bound 1
(the average of the convolution is \(1/(2n-1)\)).  
The code below builds a *Sidon‑like* support set using a greedy 
construction that allows at most two pairs per sum, then improves the
weights by a sequence of optimisation phases:

1. **Initial sparse vector** – uniform weight on the greedy support.
2. **Smoothed max‑convolution optimisation** – L‑BFGS on a
   log‑sum‑exp smooth approximation of the max.
3. **Exact projected sub‑gradient descent** on the true max‑convolution.
4. **Targeted and random hill‑climbs** for polishing.

All phases keep the vector on the probability simplex by Euclidean
projection.  The whole pipeline runs for the whole time budget
(≤ 1000 s) and finally returns the best sequence found.

```python
"""Propose a low‑score step‑function.

The algorithm builds a sparse “Sidon‑like’’ support (few indices with
pairwise sums almost unique) and distributes the probability mass
uniformly on that support.  The initial vector is then refined by:

  * a continuation L‑BFGS on a smooth log‑sum‑exp approximation of
    max(a∗a);
  * a projected sub‑gradient descent on the exact max‑convolution;
  * a targeted greedy hill‑climb that moves mass away from the
    biggest contributors of the current peak;
  * a short random hill‑climb for final polishing.

All optimisation steps respect the simplex constraint
\sum a_i = 1 , a_i ≥ 0.  The whole search runs for the allotted time
budget (≤ 1000 s) and returns the best sequence discovered.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed)."""
    fa = np.fft.fft(a, fft_len)
    c = np.fft.ifft(fa * fa)
    return np.real(c)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x≥0, sum(x)=1}
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.30):
    """
    Returns (f, grad) where f is a smooth approximation of
    2·n·max(a*a) and grad is the gradient w.r.t. the logits y.
    """
    a = softmax(y)

    # --- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                     # max becomes zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth max w.r.t. convolution -----------------------
    w = exp_shift / Z                           # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start, n, fft_len_c, fft_len_g,
                      lam_factors, maxiter_per_stage=400):
    """Continuation L‑BFGS with decreasing smoothing."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_c, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-14,
                                "disp": False})
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(a0, n, fft_len, max_iters, deadline,
                    eta0=0.35):
    """Projected sub‑gradient descent on 2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    i = np.arange(n)  # reused index vector

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # gradient of the active entry (non‑smooth)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / math.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Hill‑climbs (random & targeted)
# ----------------------------------------------------------------------


def hill_climb(a, cur_sc, fft_len, n, deadline, max_iter=2000):
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_sc = cur_sc
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_sc = fast_score(best, fft_len, n)
        if new_sc < best_sc:
            best_sc = new_sc
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_sc


def targeted_hill_climb(a, cur_sc, fft_len, n, deadline,
                        max_moves=3000, frac=0.08):
    """Greedy move from the biggest contributor of the maximal peak."""
    best = a.copy()
    best_sc = cur_sc
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_sc = fast_score(best, fft_len, n)

        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_sc


# ----------------------------------------------------------------------
# Construction of a sparse (almost Sidon) support set
# ----------------------------------------------------------------------


def build_sparse_support(n: int,
                         target_m: int,
                         max_per_sum: int = 2,
                         max_trials: int = 200_000,
                         rng: np.random.Generator = None) -> np.ndarray:
    """
    Greedy construction of a set S⊂{0,…,n‑1} of size ≈target_m such that
    each integer s appears at most max_per_sum times as a sum i+j (i≤j,
    i,j∈S).  The algorithm is stochastic; if it cannot reach target_m it
    returns the best set found.
    """
    if rng is None:
        rng = np.random.default_rng()
    S = []
    sum_cnt = dict()          # sum → how many pairs produce it

    trials = 0
    while len(S) < target_m and trials < max_trials:
        cand = rng.integers(0, n)
        # check how many new collisions would be introduced
        ok = True
        # self‑sum
        s_self = cand + cand
        if sum_cnt.get(s_self, 0) >= max_per_sum:
            ok = False
        # sums with existing elements
        if ok:
            for v in S:
                s = cand + v
                if sum_cnt.get(s, 0) >= max_per_sum:
                    ok = False
                    break
        if ok:
            # accept candidate
            S.append(cand)
            # update counters
            sum_cnt[s_self] = sum_cnt.get(s_self, 0) + 1
            for v in S[:-1]:
                s = cand + v
                sum_cnt[s] = sum_cnt.get(s, 0) + 1
        trials += 1

    return np.array(sorted(S), dtype=int)


# ----------------------------------------------------------------------
# Optimisation driver for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(n: int,
                      budget_s: float,
                      seed: int,
                      prev_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector of length n within the given
    time slice.  Returns the best vector (sum≈1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ---- initialise -------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # build a sparse “Sidon‑like’’ start
        target_m = int(0.95 * math.sqrt(2 * n))          # aim for ~0.95·√(2n)
        S = build_sparse_support(n, target_m, max_per_sum=2, rng=rng)
        if len(S) == 0:                                # fallback
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.zeros(n)
            a0[S] = 1.0 / len(S)

    best_a = a0.copy()
    fft_len = next_pow2(2 * n - 1)
    best_sc = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # parameters for the continuation L‑BFGS
    lam_factors = np.geomspace(0.5, 5e-8, num=45)

    iteration = 0

    # ------------------------------------------------------------------
    # Main loop – many independent restarts until time expires
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # decide the type of initial logits for this restart
        mode = rng.choice(["rand", "sidon", "adv", "best"])

        if mode == "rand":
            y0 = rng.standard_normal(n)
        elif mode == "sidon":
            # perturb the deterministic sparse start a little
            y0 = np.log(best_a + 1e-12) + rng.normal(scale=0.03, size=n)
        elif mode == "adv":
            # exponential/edge biased start (helps exploring the corners)
            rate = rng.uniform(0.001, 0.025)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:   # "best"
            y0 = best_y + rng.normal(scale=0.02, size=n)

        # --------------------------------------------------------------
        # 1) Continuation L‑BFGS (smooth max) -------------------------
        # --------------------------------------------------------------
        y_opt = schedule_optimize(y_start=y0,
                                  n=n,
                                  fft_len_c=fft_len,
                                  fft_len_g=next_pow2(3 * n - 2),
                                  lam_factors=lam_factors,
                                  maxiter_per_stage=600)
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len, n)

        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_sc:.6f}")

        # --------------------------------------------------------------
        # 2) Exact projected sub‑gradient descent ----------------------
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 1.0)
        best_a, best_sc = subgrad_descent(best_a,
                                          n,
                                          fft_len,
                                          max_iters=3000,
                                          deadline=sub_deadline,
                                          eta0=0.38)
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 3) Targeted greedy hill‑climb --------------------------------
        # --------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_sc = targeted_hill_climb(best_a,
                                              best_sc,
                                              fft_len,
                                              n,
                                              tc_deadline,
                                              max_moves=3500,
                                              frac=0.075)
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 4) Short random hill‑climb -----------------------------------
        # --------------------------------------------------------------
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     deadline,
                                     max_iter=3500)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final deep L‑BFGS polishing (very small λ)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.5:
        for lam in lam_factors[::-1]:               # go again with fine λ
            if time.time() >= deadline:
                break
            res = minimize(fun=objective_and_grad,
                           x0=best_y,
                           args=(n, fft_len,
                                 next_pow2(3 * n - 2), lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": 800,
                                    "gtol": 1e-14,
                                    "disp": False})
            best_y = res.x
            best_a = softmax(best_y)
            best_sc = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] λ={lam:.2e} fast score = {best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The function respects the 1000 s budget and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives a lower theoretical bound.
    # We use a single large size but keep a fallback small size in case
    # time runs out early.
    # ------------------------------------------------------------------
    n_large = 8000                     # chosen to be comfortably inside time limit
    # If we run out of time we fall back to n=4000
    sizes = [n_large, 4000]

    prev = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev, dtype=float) if prev is not None else None

    best_overall_a = None
    best_overall_score = np.inf

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 3.0:
            break
        # Allocate a slice of the remaining budget (a few seconds for the
        # fallback size, the bulk for the large size)
        slice_budget = remaining if n == n_large else min(30.0, remaining)

        print(f"[size] Optimising length n={n} with {slice_budget:.1f}s budget")
        a_n, sc_n = optimise_one_size(n=n,
                                      budget_s=slice_budget,
                                      seed=seed + idx * 123,
                                      prev_seq=prev_arr)
        print(f"[size] n={n} best fast score = {sc_n:.6f}")

        if sc_n < best_overall_score:
            best_overall_score = sc_n
            best_overall_a = a_n.copy()

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)      # pylint: disable=undefined-variable
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence for possible warm‑starts in a next run
    globals()["height_sequence_1"] = final_seq
    return final_seq
```
