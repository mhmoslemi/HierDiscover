def hill_climb_guided(a, cur_score, fft_len, n, deadline, max_iter=5000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        # contributions w_i = best[i] * best[m-i] if valid else 0
        idx = np.arange(n)
        counterpart = m - idx
        valid = (counterpart >= 0) & (counterpart < n)
        w = np.zeros_like(best)
        w[valid] = best[idx[valid]] * best[counterpart[valid]]
        # source indices: top percentile of w
        if w.max() == 0:
            # fallback random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            # get weighted source distribution
            # choose i with probability proportional to w (normalize)
            prob_src = w / w.sum()
            i = rng.choice(n, p=prob_src)
            # destination: choose index with small a (lowest values)
            # sample from bottom 10% uniformly
            low_thresh = np.percentile(best, 20)  # bottom 20%
            low_indices = np.where(best <= low_thresh)[0]
            if low_indices.size == 0:
                j = rng.integers(0, n)
            else:
                j = rng.choice(low_indices)
        if i == j:
            continue
        # move a fraction of source mass
        delta = best[i] * rng.random() * 0.1  # up to 10%
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        # ensure nonneg (source may become zero)
        # compute new score
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Propose a non‑negative step function (list of floats) with a low
evaluate_sequence() score.

Algorithm
---------
1. Determine a problem size n (≈ 8000, or reuse a provided sequence length).
2. Initialise a probability vector on the simplex using several heuristics
   (exponential, power‑law, arcsine, random, or a perturbation of the best).
3. Refine the distribution with a continuation L‑BFGS on a smooth log‑sum‑exp
   surrogate of the max self‑convolution.  The smoothing parameter λ is
   decreased from 0.5 to 1e‑8 in 60 stages, each stage allowing up to
   800 L‑BFGS iterations.
4. Perform a guided hill‑climb that repeatedly moves mass from indices that
   contribute most to the current maximal convolution entry to indices with
   little mass.
5. Run an adaptive projected sub‑gradient descent on the true max‑convolution
   objective, adjusting the step‑size depending on whether a trial move
   improves the score.
6. Finish with a final aggressive guided hill‑climb for any remaining time.
7. Return the best vector (as a plain Python list).  The harness will evaluate
   the scale‑invariant objective.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate of max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # --- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c  # now max = 0
    exp_shift = np.exp(shifted / lam)  # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # --- propagate gradient through the convolution ----------------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)  # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Heuristic initialisers (pre‑softmax vectors)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Guided hill‑climb (uses the true max‑convolution as objective)
# ----------------------------------------------------------------------


def hill_climb_guided(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from indices strongly contributing to
    the current maximal convolution entry toward indices with little mass.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current maximal convolution entry
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # contribution of each index to that entry: w_i = a_i * a_{m-i}
        idx = np.arange(n)
        counter = m - idx
        mask = (counter >= 0) & (counter < n)
        w = np.zeros_like(best)
        w[mask] = best[idx[mask]] * best[counter[mask]]

        total_w = w.sum()
        if total_w == 0.0:
            # fallback to a random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            # source index – sample proportionally to its contribution w
            prob_src = w / total_w
            i = rng.choice(n, p=prob_src)

            # destination index – pick among the smallest a-values
            low_thresh = np.percentile(best, 20)  # bottom 20 %
            low_candidates = np.where(best <= low_thresh)[0]
            if low_candidates.size == 0:
                j = rng.integers(0, n)
            else:
                j = rng.choice(low_candidates)

        if i == j:
            continue

        delta = best[i] * rng.random() * 0.10  # up to 10 % of source mass
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)

        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Adaptive projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_adaptive(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    min_eta: float = 1e-5,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_i (a*a)_i.
    Step size η is increased when a trial move improves the score and
    decreased otherwise.  Random perturbations are occasionally injected
    to escape plateaus.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    eta = eta0
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # trial step
        a_new = project_onto_simplex(a - eta * g)
        cur_score = fast_score(a, fft_len, n)
        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        if new_score < cur_score:
            # successful move → accelerate a bit
            a = a_new
            eta = min(eta * 1.2, 1.0)
        else:
            # unsuccessful → shrink step size
            eta = max(eta * 0.5, min_eta)

        # occasional random perturbation to avoid stagnation
        if rng.random() < 0.001:
            perturb = rng.normal(scale=0.01, size=n)
            a = a + perturb
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence() score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Choose problem size.  Use a provided sequence if available,
    # otherwise fall back to n≈8000.
    # --------------------------------------------------------------
    n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability vector, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(tmp, 0.0)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)          # pre‑soft‑max parameters
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – continuation L‑BFGS on a smooth surrogate
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=60))
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]

    # keep ~250 s for later refinement phases
    while time.time() < deadline - 250.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score - 1e-7:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[L‑BFGS] mode={mode} → improved fast score = {best_score:.6f}")

        # brief guided hill‑climb to polish the L‑BFGS outcome
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb_guided(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=3000
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Stage 2 – adaptive projected sub‑gradient descent on the true objective
    # --------------------------------------------------------------
    if time.time() < deadline - 100.0:
        sub_deadline = deadline - 30.0
        print("[subgrad] start")
        best_a, best_score = subgradient_adaptive(
            best_a, fft_len_conv, n, sub_deadline, eta0=0.5
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final aggressive guided hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb_guided(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=25000
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
