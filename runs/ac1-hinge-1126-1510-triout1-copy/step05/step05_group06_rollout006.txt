from scipy.optimize import linprog
def lp_improve(...):
    a = a.copy()
    # Ensure sum=1
    a /= a.sum()

    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build constraints matrix T (2n-1)x n
        # Use Toeplitz: each row k corresponds to a shifted.
        # Use np.column_stack approach.

        # Approach: using scipy.linalg.toeplitz: the first column is a padded with zeros (length 2n-1)
        # first row is [a[0], zeros...].
        from scipy.linalg import toeplitz
        col = np.r_[a, np.zeros(n-1)]
        row = np.r_[a[0], np.zeros(2*n-2)]
        T = toeplitz(col, row)[: 2*n-1, : n]

        # T is (2n-1) x n
        # Solve: maximize sum(b) => minimize -sum(b)
        c = -np.ones(n)
        # inequality: T @ b <= M
        # Also b >= 0

        res = linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
        if not res.success:
            break
        b_val = np.maximum(res.x, 0.0)
        if b_val.sum() <= 1e-12:
            break
        b_scaled = b_val / b_val.sum()

        # Mix
        a_new = (1 - t) * a + t * b_scaled
        a_new /= a_new.sum()
        new_score = fast_score(a_new, n, fft_len_conv)
        old_score = fast_score(a, n, fft_len_conv)
        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init
        else:
            t *= 0.5
            if t < 1e-4:
                break
        it += 1
    return a
```

"""
Propose a low‑evaluation step‑function.

The objective is 2·n·max(conv(a,a)) / (sum(a)^2).  Because the ratio is
scale‑invariant we restrict ourselves to the probability simplex
(sum = 1, a_i ≥ 0).  The algorithm consists of

1.  A cheap parametric sweep (uniform, linear, exponential, power‑law,
    sinusoidal, Beta‑shapes, random Dirichlet draws) to collect a few
    promising seeds.
2.  Continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution.  A geometric λ‑schedule and many restarts are used.
3.  A linear‑programming “balancing’’ step (maximise Σb s.t.
    a∗b ≤ M); the LP is solved with SciPy’s HiGHS solver using a sparse
    convolution matrix.  The solution b is mixed back into a.
4.  Projected sub‑gradient descent on the exact max‑convolution.
5.  A greedy hill‑climb that transfers a small random mass between two
    coordinates while preserving the simplex.

All phases respect the simplex constraint and use FFT‑based convolutions
(O(n log n)).  The total budget (default 1000 s) is split automatically.
The best sequence found before the deadline is returned as a plain list.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags, csc_matrix

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – L‑BFGS inner loop
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in lam_schedule, starting each stage from the
    result of the previous stage.  Return the best sequence found.
    """
    y = np.log(a_start + 1e-12)              # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # short hill‑climb after each λ (quick local refinement)
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1200)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent with a
    simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / math.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # transfer a random fraction of the source mass (up to 5 %)
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# LP balancing step – uses SciPy's HiGHS solver on a sparse matrix
# ----------------------------------------------------------------------
def lp_improve(a: np.ndarray,
               n: int,
               deadline: float,
               t_init: float = 0.12,
               max_iters: int = 12):
    """
    Perform a few iterations of the linear‑programming balancing step:

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k    (M = max(a*a))

    After each LP we mix a small fraction of the (scaled) solution back
    into the current vector a.  The mixing factor t is halved whenever a
    step does not improve the objective.
    """
    a = a.copy()
    a = a / a.sum()                     # ensure simplex

    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build the sparse convolution matrix T (size (2n‑1)×n)
        # Column i contains a[i] on the diagonal offset i.
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        # bounds for variables: b_j ≥ 0
        bounds = [(0, None) for _ in range(n)]

        # Solve with HiGHS (sparse A_ub accepted)
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success or res.x is None:
            print("[lp] LP failed or infeasible")
            break

        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        b_scaled = b_val / s

        # Mix and project back onto the simplex (the mixture already sums to 1)
        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a, n, next_pow2(2 * n - 1))

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init                 # reset mixing factor after success
            it += 1
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1

    return a

    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # LP stage (takes a few seconds)
        lp_deadline = min(phase2_end, deadline - 0.3)
        print("[LP] starting balancing iterations")
        best_a = lp_improve(best_a, n, lp_deadline,
                            t_init=0.12, max_iters=12)
        best_score = fast_score(best_a, n, fft_len_conv)
        print(f"[LP] post‑LP score={best_score:.6f}")

        # Sub‑gradient descent (short but effective)
        if time.time() < phase2_end:
            sg_deadline = min(phase2_end, deadline - 0.2)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (as much time as left)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=15000)

    # ------------------------------------------------------------------
    # Return the sequence (plain list) and log final evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)    # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
