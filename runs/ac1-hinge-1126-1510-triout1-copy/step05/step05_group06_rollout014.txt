def propose_candidate(seed=0, budget_s=1000):
    import modules
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5 # 0.5 seconds safety
    # Choose n from previous best or default
    if height_sequence_1 exists:
        base = np.asarray(height_sequence_1, dtype=float)
        n = min(len(base), 2500)
        a = base[:n]
    else:
        n = 2500
        a = np.full(n, 1/n)
    a = np.maximum(a,0)
    total = a.sum()
    if total>0:
        a = a/total
    else:
        a = np.full(n,1/n)

    fft_len = next_pow2(2*n-1)
    best_score = fast_score(a, n, fft_len)
    best_a = a.copy()
    print("[init] score:",best_score)

    # main loop
    lp_iter = 0
    # compute initial conv and M.
    while time.time() < deadline - 2: # leave at least 2 secs for final hill climb
        # compute M
        conv = np.convolve(best_a, best_a)
        M = conv.max()
        # solve LP
        # Build T:
        c = -np.ones(n) # maximize sum b => minimize -sum(b)
        # Build Toeplitz
        c_vec = np.concatenate([best_a, np.zeros(n-1)])
        r_vec = np.zeros(n)
        r_vec[0] = best_a[0]
        T = scipy.linalg.toeplitz(c_vec, r_vec) # shape (2n-1,n)
        # define bounds
        bounds = [(0, None)] * n
        # Determine RHS
        b_ub = M * np.ones(2*n-1)
        # Solve LP
        res = scipy.optimize.linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success or res.x is None:
            print("[lp] solve failed, break")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        b_scaled = b / sum_b

        # line search for mixing factor
        improved = False
        # Evaluate at some alphas
        alphas = [0.02,0.05,0.1,0.2]
        for alpha in alphas:
            a_candidate = (1-alpha) * best_a + alpha * b_scaled
            # ensure sum=1 (should be)
            # Evaluate
            sc = fast_score(a_candidate, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a_candidate
                improved = True
                print(f"[lp] iter {lp_iter} new best score={best_score:.6f} alpha={alpha}")
                break
        if not improved:
            # Reduce mixing factor maybe gradually
            # break if no improvement for several attempts
            # For simplicity break out after few fails
            # We'll continue loop but maybe with lower alphas?
            pass

        lp_iter += 1
        # optionally run subgradient for short period to polish before next LP
        # For simplicity, after each LP run subgradient for small number of steps or until time
        sg_deadline = min(deadline-1, time.time()+0.5) # 0.5 sec
        best_a, best_score = subgradient_optimize(best_a, sg_deadline, fft_len, fft_len, n)
        # note that subgradient_optimize returns best so far.

    final_deadline = deadline

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print("[final] evaluated score =", final_eval)
    return final_seq

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm starts from the best known construction (global variable
``height_sequence_1`` if available) and repeatedly applies the classic
“M‑balancing’’ linear‑programming step of the reference paper:

    maximise   Σ b_j
    s.t.       b_j ≥ 0
               (a * b)_k ≤ M   ∀k ,   where M = max (a*a)

The LP is solved with scipy.optimize.linprog (HiGHS).  After each LP we
mix the current vector with the scaled solution using a short line‑search
over several mixing factors.  A lightweight projected sub‑gradient
descent is then run to polish the result, and finally a greedy hill‑climb
transfers small random masses between coordinates while staying on the
simplex.

All heavy convolutions are performed with FFTs, the simplex projection
uses the standard O(n) algorithm and the whole search respects the 1000 s
budget.  The function returns a plain ``list`` of floats; the external
``evaluate_sequence`` routine is called only once at the very end to obtain
the final score.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Returns 2·n·max(conv(a,a)) for a vector ``a`` that already satisfies
    sum(a)=1.  Convolution is performed via FFT for speed.
    """
    fa = fft(a, fft_len)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv_full.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0,  Σ x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_optimize(a0: np.ndarray, deadline: float,
                         fft_len_conv: int, n: int):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Simple back‑tracking line‑search; the best vector seen so far is
    retained.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current full convolution
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_conv)
        fa_rev = fft(a_rev, fft_len_conv)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line‑search
        step = 0.5 / math.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray, cur_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb on the simplex.
    Randomly moves a small fraction of mass from one coordinate to another
    and accepts strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # move up to 5 % of the source mass
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta
        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta
        i += 1
    return best, best_score


# ----------------------------------------------------------------------
# Core optimisation loop (LP + mixing + polishing)
# ----------------------------------------------------------------------
def lp_balancing_step(a: np.ndarray, n: int,
                     fft_len: int, deadline: float) -> np.ndarray:
    """
    Perform a single M‑balancing LP step:

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M  ∀k

    The solution b is scaled back to sum = 1, then mixed with ``a``.
    A small line‑search over a handful of mixing factors is performed;
    the best improving mixture (if any) is returned; otherwise ``a`` is
    returned unchanged.
    """
    # current peak M
    conv = np.convolve(a, a)
    M = conv.max()

    # build the linear convolution matrix T (size (2n‑1)×n)
    c_vec = np.concatenate([a, np.zeros(n - 1)])
    r_vec = np.zeros(n)
    r_vec[0] = a[0]
    T = toeplitz(c_vec, r_vec)          # dense (2n‑1)×n matrix

    # LP: maximise sum(b)  ⇔ minimise -sum(b)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n

    # solve
    res = linprog(c_obj, A_ub=T, b_ub=b_ub,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if not res.success or res.x is None:
        return a                     # fallback – keep the current vector

    b = np.maximum(res.x, 0.0)
    sum_b = b.sum()
    if sum_b <= 1.0 + 1e-12:
        return a                     # no increase in total mass → no improvement possible

    b_scaled = b / sum_b

    # try a short line‑search on the mixing parameter t
    best_local = a
    best_local_score = fast_score(a, n, fft_len)
    for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
        a_mix = (1.0 - t) * a + t * b_scaled
        # numerical drift → re‑normalise
        a_mix = np.maximum(a_mix, 0.0)
        a_mix /= a_mix.sum()
        sc = fast_score(a_mix, n, fft_len)
        if sc < best_local_score - 1e-12:
            best_local, best_local_score = a_mix, sc
            break
    return best_local


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a low‑evaluation step sequence.
    Returns a plain Python list of floats (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size and initialise a from the best known seed
    # --------------------------------------------------------------
    max_n = 2500                     # safe upper bound for LP solving
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)          # for all convolution work
    best_a = a0.copy()
    best_score = fast_score(best_a, n, fft_len)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Main iterative improvement (LP + sub‑gradient)
    # --------------------------------------------------------------
    start = time.time()
    overall_deadline = start + budget_s - 0.5   # safety margin
    # Reserve a few seconds for final polishing
    lp_phase_deadline = overall_deadline - 5.0

    lp_iters = 0
    while time.time() < lp_phase_deadline:
        # single LP balancing step (may improve the score)
        new_a = lp_balancing_step(best_a, n, fft_len, lp_phase_deadline)

            print(f"[LP] iter {lp_iters} improved score={best_score:.6f}")
        else:
            # no improvement → stop LP loop
            break
        lp_iters += 1

        # short sub‑gradient polishing after each LP iteration

    # --------------------------------------------------------------
    # 3. Final hill‑climb polishing (uses remaining time)
    # --------------------------------------------------------------
    final_deadline = overall_deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len, n,
                                    final_deadline,
                                    max_iter=20000)

    # --------------------------------------------------------------
    # 4. Return the final sequence
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external evaluation routine
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
