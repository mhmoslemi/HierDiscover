        if s > 1e-12:
            a = a / s
        else:

    iteration = 0

    while time.time() < deadline - 5.0:

        mode = rng.choice(['random','exp','power','edge','best'])
        if mode == 'random':

            a0 = np.exp(-rate * np.arange(n))

    print(f"[final] evaluated score = {final_val:.6f}")
    globals()["height_sequence_1"] = final_seq

"""
Propose a step‑function that minimizes
    f(a) = 2·n·max(a∗a) / (sum a)^2
using a combination of:

* L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution.
* Projected sub‑gradient descent on the exact nonsmooth objective.
* Targeted and random hill‑climbs.
* A final linear‑programming refinement that maximises the total mass
  under the current max‑convolution bound (alternating‑minimisation idea).

The algorithm is time‑aware and stops before the supplied budget.
"""
import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i ≥ 0)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """Smooth approximation of max(a*a) and its gradient w.r.t. logits y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(y_start: np.ndarray, n: int,
                      fft_len_conv: int, fft_len_g: int,
                      lam_factors, maxiter_per_stage: int = 400) -> np.ndarray:
    """Continuation L‑BFGS with a decreasing smoothing schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-14,
                                "disp": False})
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(a0: np.ndarray, n: int, fft_len: int,
                    max_iters: int, deadline: float,
                    eta0: float = 0.35) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)  # constant index array

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)

        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional perturbation to escape plateaus
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, cur_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(a: np.ndarray, cur_score: float,
                       fft_len: int, n: int,
                       deadline: float,
                       max_moves: int = 3000,
                       frac: float = 0.08) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the largest contributor
    to the smallest contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (alternating‑minimisation style)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray, fft_len: int, deadline: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to  (a * g)_k ≤ M   (for all k)
                    g_j ≥ 0
    and return a convex combination of a and the rescaled LP solution.
    """
    n = len(a)
    conv = conv_fft(a, fft_len, n)
    M = conv.max()

    rows = 2 * n - 1

    # Build sparse convolution matrix C such that C @ g = a * g
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)
    row = (j[:, None] + i[None, :]).ravel()
    col = np.repeat(j, n)
    data = np.tile(a, n)
    C = csr_matrix((data, (row, col)), shape=(rows, n))

    # LP: maximise Σ g  <=> minimise -Σ g
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(rows, M)

    res = linprog(c, A_ub=C, b_ub=b_ub,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    if not res.success:
        return a

    g = res.x
    s = g.sum()
    if s <= 0:
        return a
    g = g / s                     # rescale to sum = 1

    # convex update (small step towards g)
    t = 0.30
    new_a = (1.0 - t) * a + t * g
    new_a = project_simplex(new_a)
    return new_a


# ----------------------------------------------------------------------
# Optimisation driver for a single size
# ----------------------------------------------------------------------


def optimise_one_size(n: int,
                      budget_s: float,
                      seed: int,
                      prev_seq: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """Search for the best sequence of length n within the allocated budget."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ----- initialise ----------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a = prev_seq[:n].copy()
        a = np.clip(a, 0.0, np.inf)
        s = a.sum()
        a = a / s if s > 1e-12 else np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    best_a = a.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # smoothing schedule for L‑BFGS
    lam_factors = np.geomspace(0.5, 5e-8, num=45)

    iteration = 0

    # ----- main stochastic restart loop ----------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = (np.sqrt(n / 2.0) / n)
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # "best"
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ---- L‑BFGS continuation ---------------------------------------
        y_opt = schedule_optimize(y_start=y0,
                                  n=n,
                                  fft_len_conv=fft_len,
                                  fft_len_g=fft_len_g,
                                  lam_factors=lam_factors,
                                  maxiter_per_stage=500)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_score:.6f}")

        # ---- Sub‑gradient descent ---------------------------------------
        sub_deadline = min(deadline, time.time() + 1.5)
        best_a, best_score = subgrad_descent(best_a,
                                             n,
                                             fft_len,
                                             max_iters=10_000_000,
                                             deadline=sub_deadline,
                                             eta0=0.5)
        best_y = np.log(best_a + 1e-12)

        # ---- Targeted hill‑climb ----------------------------------------
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = targeted_hill_climb(best_a,
                                                 best_score,
                                                 fft_len,
                                                 n,
                                                 hc_deadline,
                                                 max_moves=5000,
                                                 frac=0.07)
        best_y = np.log(best_a + 1e-12)

        # ---- Random hill‑climb ------------------------------------------
        rc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        rc_deadline,
                                        max_iter=5000)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ----- final intensive refinement ------------------------------------
    while time.time() < deadline - 1.5:
        # LP refinement (alternating‑minimisation step)
        lp_deadline = min(deadline, time.time() + 2.0)
        if time.time() < lp_deadline:
            cand = lp_refine(best_a, fft_len, lp_deadline)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                best_a = cand
                best_score = cand_score
                best_y = np.log(best_a + 1e-12)
                print(f"[LP] fast score = {best_score:.6f}")

        # short L‑BFGS runs with very low λ
        for lam in np.geomspace(0.5, 1e-9, num=20):
            if time.time() >= deadline:
                break
            res = minimize(fun=objective_and_grad,
                           x0=best_y,
                           args=(n, fft_len, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": 800,
                                    "gtol": 1e-14,
                                    "disp": False})
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.2g} fast score = {best_score:.6f}")

        # extra sub‑gradient descent
        sub_deadline = min(deadline, time.time() + 1.5)
        best_a, best_score = subgrad_descent(best_a,
                                             n,
                                             fft_len,
                                             max_iters=10_000_000,
                                             deadline=sub_deadline,
                                             eta0=0.5)
        best_y = np.log(best_a + 1e-12)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (as a Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final call to evaluate_sequence()
    total_deadline = start_time + budget_s - 2.0

    # reuse previous construction if available
    prev = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev, dtype=float) if prev is not None else None

    # Choose a single problem size – larger sizes give a lower theoretical bound.
    # 4000 works well with the available memory/time budget.
    n = 4000

    remaining = total_deadline - time.time()
    if remaining < 5.0:
        # fallback to a uniform distribution if we have almost no time
        final_seq = np.full(2000, 1.0 / 2000).tolist()
        globals()["height_sequence_1"] = final_seq
        return final_seq

    print(f"[main] Optimising length n={n} (budget ≈ {remaining:.1f}s)")

    a_best, best_fast_score = optimise_one_size(n=n,
                                                budget_s=remaining,
                                                seed=seed,
                                                prev_seq=prev_arr)

    print(f"[main] best fast score after optimisation = {best_fast_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = a_best.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
