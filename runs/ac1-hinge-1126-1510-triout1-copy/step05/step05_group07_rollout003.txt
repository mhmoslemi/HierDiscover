def subgrad_momentum(a0, n, fft_len, max_iters, deadline, eta0=0.5, mu=0.9):
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    i = np.arange(n)
    for t in range(max_iters):
        if time.time() > deadline:
            break
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))
        j = k - i
        mask = (j >=0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        eta = eta0 / np.sqrt(t + 1.0)
        v = mu * v - eta * grad
        a = project_simplex(a + v)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # occasional perturb
        if (t+1) % 500 ==0 and np.random.rand() < 0.1:
            noise = np.random.normal(scale=0.01, size=n)
            a = project_simplex(a + noise)
            v = np.zeros_like(a)
    return best_a, best_score
```

def subgrad_momentum(a0, n, fft_len, max_iters, deadline, eta0=0.5, mu=0.9):
    a = a0.copy()
    velocity = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    idx = np.arange(n)
    for t in range(max_iters):
        if time.time() > deadline:
            break
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))
        # compute gradient: for each i, grad_i = 2 * a_{k-i} if index valid else 0
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]
        # step-size
        eta = eta0 / np.sqrt(t + 1.0)
        # momentum update
        velocity = mu * velocity - eta * grad
        a = project_simplex(a + velocity)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # occasional jitter
        if (t+1) % 500 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(scale=0.01, size=n)
            a = project_simplex(a + noise)
            velocity = np.zeros_like(a)
    return best_a, best_score

def hill_climb(a, cur_score, fft_len, n, deadline, max_iter=4000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.08
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Hybrid multi‑size optimizer for the min‑max self‑convolution problem.

The algorithm explores several problem sizes (up to ~12 000 entries) and
alternates four phases per restart:

1.  A continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The smoothing parameter λ is decreased on a
    geometric schedule.
2.  Projected momentum sub‑gradient descent on the true non‑smooth
    objective  f(a)=2·n·max(a*a) .
3.  A greedy “targeted” hill‑climb that moves mass from the biggest
    contributors to the smallest contributors of the current peak.
4.  A random mass‑transfer hill‑climb for polishing.

Best vectors are kept on the probability simplex (∑a=1).  After the
budget for a given size is exhausted a final low‑λ L‑BFGS refinement
is performed.  The overall best sequence (any length) is returned and
evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i ≥ 0)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                      # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z  # ∂smooth_max/∂conv, length 2n‑1

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Projected momentum sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_momentum(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.5,
    mu: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on f(a)=2·n·max(a*a).
    Returns the best vector found and its fast_score.
    """
    a = a0.copy()
    vel = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    idx = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # gradient of max at active index k:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)
        vel = mu * vel - eta * grad
        a = project_simplex(a + vel)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if (t + 1) % 500 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(scale=0.01, size=n)
            a = project_simplex(a + noise)
            vel = np.zeros_like(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random and targeted hill‑climbs (same logic as the reference code)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.08

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 5000,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the largest contributor
    to the smallest contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to random pair
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Optimisation driver for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length n within the allotted time slice.
    Returns the best vector (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ----- initialise ----------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # geometric schedule for λ (smooth max)
    lam_factors = np.geomspace(0.5, 5e-6, num=30)

    iteration = 0

    # ----- main restart loop --------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            edge_fr = rng.uniform(0.005, 0.03)
            m_edge = max(1, int(edge_fr * n))
            edge_val = 1.0 / (2 * m_edge)
            interior_val = (1.0 - 2 * m_edge * edge_val) / (n - 2 * m_edge) if n > 2 * m_edge else 0.0
            a = np.full(n, interior_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # 1) L‑BFGS continuation on the smooth approximation
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=500,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_score:.6f}")

        # 2) Projected momentum sub‑gradient descent on the exact max‑convolution
        sub_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = subgrad_momentum(
            best_a,
            n,
            fft_len,
            max_iters=8000,
            deadline=sub_deadline,
            eta0=0.5,
            mu=0.85,
        )
        best_y = np.log(best_a + 1e-12)

        # 3) Targeted greedy hill‑climb
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            tc_deadline,
            max_moves=5000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # 4) Random hill‑climb polish
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

        # stop early if we already beat the target bound
        if best_score < 1.503:
            print(f"[early stop] fast score {best_score:.6f} < 1.503")
            break

    # ----- final low‑λ L‑BFGS refinement ---------------------------------
    while time.time() < deadline - 1.5:
        for lam in lam_factors[-6:]:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 400, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.3g} fast score = {best_score:.6f}")
            if best_score < 1.503:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its sum will be 1 after
    internal normalisation).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin

    # ---- warm‑start from a previously generated sequence if available ----
    prev = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev, dtype=float) if prev is not None else None

    best_overall_a = None
    best_overall_score = np.inf

    # sizes to explore – a mix of modest and larger lengths
    sizes = [2000, 4000, 8000, 12000]
    per_size_budget = (total_deadline - time.time()) / len(sizes)

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 5.0:
            break
        slice_budget = min(per_size_budget, remaining - (len(sizes) - idx - 1) * 5.0)
        print(f"[size] Optimising length n={n} with slice budget ≈{slice_budget:.1f}s")
        a_n, score_n = optimise_one_size(
            n=n,
            budget_s=slice_budget,
            seed=seed + idx * 123,
            prev_seq=prev_arr,
        )
        print(f"[size] n={n} best fast score = {score_n:.6f}")

        if score_n < best_overall_score:
            best_overall_score = score_n
            best_overall_a = a_n.copy()

        # use the newly found best as warm start for the next (larger) size
        prev_arr = best_overall_a

        # early exit if we have already crossed the target
        if best_overall_score < 1.503:
            print("[global early stop] target achieved")
            break

    if best_overall_a is None:
        # fallback – uniform single entry
        best_overall_a = np.array([1.0])

    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence for possible reuse by a subsequent run
    globals()["height_sequence_1"] = final_seq
    return final_seq
