def init_phase(n, rng):
    fft_len = next_pow2(2 * n - 1)
    # random phases for half of spectrum due to real constraint
    # Create random complex numbers of magnitude 1 that are conjugate symmetric
    # For real IFFT, we need hermitian symmetry.
    # We'll fill half of frequencies and then set the symmetric ones.
    # Use length fft_len.
    # For real valued output, we can do:
    phase = rng.uniform(0, 2*np.pi, size=fft_len//2 - 1)  # exclude DC and Nyquist
    # Build full spectrum:
    spec = np.zeros(fft_len, dtype=complex)
    spec[0] = 0 # DC component zero because sum will be zero then later we normalize sum to 1
    spec[1:fft_len//2] = np.exp(1j * phase)
    spec[fft_len//2] = 0 if fft_len % 2 == 0 else np.exp(1j * rng.uniform(0, 2*np.pi))
    spec[fft_len//2+1:] = np.conj(spec[1:fft_len//2][::-1])
    # Inverse FFT:
    a_raw = np.fft.ifft(spec).real
    # Take absolute value because might be negative
    a = np.abs(a_raw[:n])
    a = a / a.sum()
    return np.log(a + 1e-12)

"""
Propose a low‑score non‑negative step‑function sequence.

Main ideas
----------
*   Work on the probability simplex (∑a = 1, a_i ≥ 0).  The objective
    ``evaluate_sequence`` is homogeneous, therefore the scale of the
    vector does not matter – we can keep it normalized all the time.
*   Use a continuation of L‑BFGS on a smooth approximation of the
    ``max(conv)`` term.  The smoothing parameter λ is reduced geometrically
    from a large value (easy landscape) down to almost zero (true
    max).  The schedule is finer than the reference implementation.
*   After the global L‑BFGS phase perform a long projected‑gradient descent
    on the *actual* smooth‑max (λ → 0).  This is cheap (one FFT per
    iteration) and drives the solution very close to the true optimum.
*   Finish with a focused “targeted hill‑climb” that moves mass away from
    the indices that contribute most to the current peak.
*   Several random restarts (different initialisations) are carried out
    within the time budget; the best result is kept.
*   The final sequence is stored globally as ``height_sequence_1`` for
    reuse by later runs.

The code runs entirely within the supplied time budget (default 1000 s)
and never writes to the file system.  All helper functions are top‑level
as required by the evaluation harness.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for a normalized probability vector.
    Returns ``2·n·max(conv)`` where conv = a * a (full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a (full) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max ----
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv ----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ---- back‑propagation to a (size n) ----
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient of f = 2·n·smooth_max w.r.t. a
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift that is maximal

        # identify the pair (i, k‑i) that contributes most
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]
        i = int(np.argmax(contrib))

        # pick a target index whose partner under the same shift is tiny
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        # move a fraction of mass from i → j
        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth‑max of the auto‑convolution w.r.t. a.
    Returns a vector of length n (without the outer 2·n factor).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # derivative of smooth max w.r.t. a is 2·g (outer 2·n factor applied later)
    return 2.0 * g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative heights (the best sequence found).
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # time management
    # ------------------------------------------------------------------
    # leave a small margin for the final official evaluation and printing
    deadline = time.time() + budget_s - 4.0

    # ------------------------------------------------------------------
    # problem size – reuse a previous construction if it exists
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # normalise onto the simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    best_a = best_a / best_a.sum()
    best_y = np.log(best_a + 1e-12)

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # pre‑computed FFT lengths (fixed)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # L‑BFGS continuation schedule (global search)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-8, num=45)   # finer λ‑schedule
    maxiter_per_stage = 1000                         # more work per λ‑stage

    iteration = 0
    # allocate roughly the first 60 % of the budget to the global search
    search_deadline = time.time() + (budget_s * 0.60)
    while time.time() < search_deadline and iteration < 200:
        # choose a random initialisation strategy
        mode = rng.choice(
            ["random", "exp", "power", "edge", "chi2", "phase", "best"]
        )
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            center_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, center_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "chi2":
            a0 = np.square(rng.standard_normal(n)) + 1e-12
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "phase":
            # random – phase‑only spectrum, then take absolute values
            spec = np.zeros(fft_len_conv, dtype=complex)
            # DC component – give a small constant to avoid all zeros
            spec[0] = 1.0
            # fill lower half of the spectrum with random phases (Hermitian symmetry)
            half = fft_len_conv // 2
            phases = rng.uniform(0, 2 * np.pi, size=half - 1)
            spec[1:half] = np.exp(1j * phases)
            spec[half + 1:] = np.conj(spec[1:half][::-1])
            # keep Nyquist term real if present
            if fft_len_conv % 2 == 0:
                spec[half] = rng.choice([-1, 1])
            a_raw = np.fft.ifft(spec).real[:n]
            a0 = np.abs(a_raw) + 1e-12
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # best – perturb current best
            y0 = best_y + rng.normal(scale=0.08, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode:<6} improved fast score = {best_score:.6f}"
            )

        # ----- aggressive targeted hill‑climb (short burst) -----
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.28,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Projected‑gradient descent on the smooth‑max (λ → 0)
    # ------------------------------------------------------------------
    lam_factor_pg = 1e-8          # very sharp approximation → true max
    step0 = 0.18                  # initial step size
    max_pg_iter = 8000
    pg_iter = 0
    pg_deadline = min(deadline, time.time() + (budget_s * 0.30))
    while time.time() < pg_deadline and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)   # gradient w.r.t. a
        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            # shrink λ a little if we are making progress
            lam_factor_pg = max(lam_factor_pg * 0.94, 1e-10)

    print(f"[pg] iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive targeted hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=final_deadline,
            max_iter=5000,
            move_frac=0.32,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq  # store for future runs
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
