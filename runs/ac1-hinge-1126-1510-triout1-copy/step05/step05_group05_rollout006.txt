"""
Improved step‑function search (≈1000 s budget).

Main ideas
-----------
* Start from a rich pool of parametric seeds (uniform, power‑law,
  exponential, sinusoid, double‑exponential, random Dirichlet…).
  Keep the top few candidates.
* Perform a *continuation* optimisation of a smooth log‑sum‑exp
  approximation of the max‑convolution.  The continuation uses a
  long λ‑schedule and many L‑BFGS iterations per λ.  Each stage ends with
  a short projected hill‑climb.
* Follow up with a long Adam run on the same smoothed objective.
* Finish with an exact projected sub‑gradient descent (the true
  max‑convolution is used as a sub‑gradient) with a back‑tracking line
  search, then a final greedy hill‑climb.
* All iterates stay on the probability simplex (non‑negative,
  sum = 1).  The code works for any length up to 4000 and adapts the
  FFT size automatically.

The algorithm stays completely within the 1000 s budget by splitting the
time into phases and stops early if the deadline is reached.
"""

import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the quantity 2·n·max(a*a) for a sequence already
    normalised to sum = 1.  (The true evaluation is 2·n·max(conv)/sum(a)²,
    but it is scale‑invariant.)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def smoothed_obj_and_grad(y: np.ndarray, n: int,
                          fft_len_conv: int, fft_len_g: int,
                          lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Log‑sum‑exp smoothing of max(conv).  Returns (objective, gradient)
    where the objective is 2·n·smooth_max.
    """
    # soft‑max parameters → probability vector a
    a = softmax(y)

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt conv entries
    w = exp_shift / Z                               # ∂smooth_max/∂conv_i

    # gradient wrt a via convolution w * a_rev
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    grad_a = np.real(np.fft.ifft(fw * fa_rev))
    # extract the n components that correspond to the gradient w.r.t a
    g_vec = grad_a[(n - 1):(n - 1 + n)]

    # chain rule through the softmax (∂a/∂y = diag(a) - a aᵀ)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_factors,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_factors, starting each stage from the
    result of the previous one.  After each λ a short greedy hill‑climb
    is performed.
    """
    y = np.log(a_start + 1e-12)          # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=smoothed_obj_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # L‑BFGS occasionally fails on pathological inputs – ignore.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] λ={lam:.4g} new best={best_score:.6f}")

        # Quick hill‑climb to polish after each λ
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline, max_iter=2000)

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑based continuation on the smoothed objective
# ----------------------------------------------------------------------
def adam_continuation(y0: np.ndarray,
                     lam_factors,
                     iters_per_lam,
                     deadline,
                     fft_len_conv,
                     fft_len_g,
                     n):
    """
    Adam optimiser on the same smoothed objective, sweeping through the
    λ‑schedule.  Returns the best probability vector seen.
    """
    y = y0.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.15
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            obj, grad = smoothed_obj_and_grad(y, n,
                                              fft_len_conv, fft_len_g,
                                              lam)

            # simple norm clipping for stability
            gnorm = np.linalg.norm(grad)
            if gnorm > 5.0:
                grad = grad * (5.0 / gnorm)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        print(f"[adam] after λ={lam:.4g} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Direct minimisation of 2·n·max(a*a) using a simple projected
    sub‑gradient descent with a back‑tracking line search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # compute exact convolution and the mask of max positions
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient = 2 * (mask * a_rev) convolved back
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_full = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # back‑tracking step size
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            cand = simplex_projection(a - step * g)
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1
        if it % 500 == 0:
            print(f"[sub‑grad] it={it} best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb: move a small random fraction from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07   # up to 7 % of a[i]
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature schedule.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.025
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # up to 12 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T = max(T * 0.90, 5e-6)

    return best_a, best_score


# ----------------------------------------------------------------------
# Seed generation utilities
# ----------------------------------------------------------------------
def add_candidate(arr, n, fft_len_conv, candidates):
    """Normalise a raw array, compute its fast_score and store."""
    arr = np.asarray(arr, dtype=float)
    if arr.sum() <= 0:
        return
    arr = arr / arr.sum()
    sc = fast_score(arr, fft_len_conv, n)
    candidates.append((sc, arr))


def generate_initial_seeds(n, fft_len_conv):
    """
    Produce a diverse collection of candidate distributions.
    Returns a list of numpy arrays (already normalised to sum = 1).
    """
    rng = np.random.default_rng()
    cand_data = []

    # Uniform
    add_candidate(np.full(n, 1.0 / n), n, fft_len_conv, cand_data)

    # Linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_candidate(lin, n, fft_len_conv, cand_data)

    # Exponential families
    for rate in [2e-4, 5e-4, 1e-3, 2e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)), n, fft_len_conv, cand_data)

    # Power‑law families
    for p in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)), n, fft_len_conv, cand_data)

    # Edge‑heavy (mass at both ends, varying width)
    for edge_frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr, n, fft_len_conv, cand_data)

    # Sinusoidal / raised‑cosine shapes
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-4, n, fft_len_conv, cand_data)

    # Double‑exponential (large at both ends, small in centre)
    for decay in [0.001, 0.003, 0.007]:
        left = np.exp(-decay * np.arange(n))
        right = np.exp(-decay * np.arange(n)[::-1])
        arr = left + right
        add_candidate(arr, n, fft_len_conv, cand_data)

    # Random Dirichlet draws (different concentrations)
    for alpha in [0.3, 0.6, 1.0, 2.0, 5.0]:
        for _ in range(5):
            add_candidate(rng.gamma(alpha, 1.0, size=n), n, fft_len_conv, cand_data)

    # If a previous sequence exists, include it as a seed
    if 'height_sequence_1' in globals():
        add_candidate(np.array(globals()['height_sequence_1'], dtype=float),
                      n, fft_len_conv, cand_data)

    # Sort and keep the best 12 seeds
    cand_data.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in cand_data[:12]]
    print(f"[seed] best seed score = {cand_data[0][0]:.6f}")
    return top_seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # -----------------------------
    # Problem size & initial vector
    # -----------------------------
    max_n = 2000                       # keep the problem tractable
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)    # length for a*a
    fft_len_g = next_pow2(3 * n - 2)       # length for gradient related FFTs

    # Build seed pool
    top_seeds = generate_initial_seeds(n, fft_len_conv)

    # Start from the best seed
    best_a = top_seeds[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, start_score={best_score:.6f}")

    # -----------------------------
    # Time budget split
    # -----------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.6   # a small safety margin

    # L‑BFGS phase (≈55 % of the budget)
    phase1_end = start_time + 0.55 * budget_s
    lam_schedule = np.logspace(np.log10(0.5), np.log10(0.001), num=14)
    maxiter_per_lam = 600                  # more iterations per λ

    restarts = 0
    while time.time() < phase1_end and restarts < 25:
        # jitter a good seed to diversify
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 2e-3
        a_start = seed_vec + jitter
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] restart {restarts} best={best_score:.6f}")
        restarts += 1

    # -----------------------------
    # Adam continuation phase (≈25 % of budget)
    # -----------------------------
    phase2_end = start_time + 0.80 * budget_s
    if time.time() < phase2_end:
        adam_iters_per_lam = 300
        adam_restarts = 0
        while time.time() < phase2_end and adam_restarts < 12:
            jitter = rng.random(n) * 1e-3
            a_start = best_a + jitter
            a_start = np.clip(a_start, 0.0, None)
            a_start /= a_start.sum()
            y0 = np.log(a_start + 1e-12)

            a_adam, score_adam = adam_continuation(y0,
                                                   lam_schedule,
                                                   adam_iters_per_lam,
                                                   phase2_end,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   n)
            if score_adam < best_score:
                best_score, best_a = score_adam, a_adam.copy()
                print(f"[Adam] restart {adam_restarts} best={best_score:.6f}")
            adam_restarts += 1

    # -----------------------------
    # Exact sub‑gradient descent (≈15 % of budget)
    # -----------------------------
    phase3_end = start_time + 0.95 * budget_s
    if time.time() < phase3_end:
        sg_deadline = min(phase3_end, deadline - 0.15)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑grad] after phase3 best={best_score:.6f}")

    # -----------------------------
    # Simulated annealing + final hill‑climb (remaining time)
    # -----------------------------
    if time.time() < deadline:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           deadline)
        print(f"[anneal] best={best_score:.6f}")

        # final greedy polishing
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=15000)

    # -----------------------------
    # Return result
    # -----------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
