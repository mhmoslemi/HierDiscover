def lp_step(f, M, t):
    n = len(f)
    rows = 2*n - 1
    # build A
    data = np.repeat(f, n)                     # each column repeated n times
    row_idx = np.concatenate([np.arange(i, i+n) for i in range(n)])
    col_idx = np.repeat(np.arange(n), n)
    A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(rows, n))
    # Solve LP
    b = np.full(rows, M)
    # Convert to dense for linprog? SciPy's `linprog` accepts sparse.
    res = scipy.optimize.linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'time_limit': 30})
    if not res.success:
        return f   # fallback
    g0 = res.x
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return f
    g = g0 / sum_g0
    new_f = (1 - t) * f + t * g
    return new_f
```

"""
Propose a low‑score non‑negative step sequence.

Algorithm (≈1000 s):
  1. Random restarts of a continuation L‑BFGS optimizer on a smooth
     log‑sum‑exp approximation of the max‑convolution (fast global
     search).
  2. An iterative linear‑programming smoothing step:
       given the current distribution f and its maximal convolution M,
       maximise Σ g_j subject to (f∗g)_k ≤ M (all k) and g ≥ 0.
       Rescale g to a probability vector and move a small amount of
       mass from f towards g.  Recompute M and repeat.
  3. A final greedy hill‑climb that transfers a tiny fraction of mass
     between two coordinates while keeping the sum = 1.
The best sequence found before the deadline is returned.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import minimize, linprog


# -------------------------------------------------------------
# Basic utilities
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sine",
            "halfsine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sine":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    else:  # halfsine
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)

    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# -------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# -------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in freq. domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# -------------------------------------------------------------
# Continuation L‑BFGS
# -------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """
    Runs L‑BFGS for each lambda in lam_schedule, starting each stage
    from the result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            # keep current y and continue
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------
# Linear‑programming smoothing step
# -------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2·n‑1)×n matrix A such that (A @ g) = f ∗ g.
    Row k (0‑based) gives (f∗g)[k] = Σ_i f_i g_{k‑i}.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    # each diagonal d (0 ≤ d < n) has constant value f[d] and length n
    diagonals = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)  # 0,1,2,…,n‑1
    A = sp.diags(diagonals, offsets, shape=(rows, n), format="csr")
    return A


def lp_iterate(
    a: np.ndarray,
    deadline: float,
    max_iters: int = 20,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve a linear program that maximises the total mass
    subject to the current max‑convolution bound.
    The solution is blended into the current distribution.
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current bound M
        conv = full_conv(best_a, fft_len_conv)
        M = conv.max()

        # build constraint matrix for (best_a ∗ g) ≤ M
        A = build_conv_matrix(best_a)

        # linear program: maximise Σ g_j  <=> minimise -Σ g_j
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)

        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )

        if not res.success:
            # If LP fails we stop the LP phase.
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # Rescale to the simplex (sum = 1)
        g = g0 / sum_g0

        # Step size decreases with the iteration count
        step = step0 / (it + 1)

        a_new = (1.0 - step) * best_a + step * g

        # evaluate
        score_new = fast_score(a_new, fft_len_conv, n)

        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            print(f"[LP iter {it}] fast_score={best_score:.6f}")

        # keep the updated distribution for the next LP pass
        best_a = a_new

    return best_a, best_score


# -------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# -------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer hill‑climb: move a small random fraction between two coordinates."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives a larger search space)
    # ------------------------------------------------------------------
    max_n = 3000
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n // 2
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n // 2
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)  # for a*a
    fft_len_g = next_pow2(3 * n - 2)    # for gradient‑related convolutions

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin
    phase1_end = start_time + 0.20 * budget_s  # L‑BFGS phase (≈20 %)
    phase2_end = start_time + 0.70 * budget_s  # LP smoothing phase (≈50 %)

    # ------------------------------------------------------------------
    # Phase 1 – many random L‑BFGS restarts
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.15,
        0.10,
        0.07,
        0.045,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
    ]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – linear‑programming smoothing
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = lp_iterate(best_a, phase2_end, max_iters=30)
        print(f"[LP phase] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=8000,
    )
    print(f"[final hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence (evaluate_sequence will be called by the runner)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
