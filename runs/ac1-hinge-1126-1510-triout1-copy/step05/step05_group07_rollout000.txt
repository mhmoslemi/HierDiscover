"""
Hybrid shape‑search + smooth‑max L‑BFGS + projected sub‑gradient descent.

The algorithm builds many candidate step‑functions of a fixed length
(e.g. n=4000) using simple parametric shapes (edge‑biased, exponential,
power‑law, random Dirichlet).  Each candidate is refined by a continuation
L‑BFGS on a log‑sum‑exp approximation of the true max‑convolution, then
by a long projected sub‑gradient descent on the exact non‑smooth
objective.  Finally a few greedy hill‑climbs polish the result.
All candidates are kept on the probability simplex (sum = 1) so that the
evaluation score is simply 2·n·max(conv).  The best sequence found
within the time budget is returned and stored in the global variable
``height_sequence_1`` for possible warm‑starts in later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact 2·n·max(a*a) for a probability vector a (sum = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}
    (Wang & Carreira‑Perpiñán, 2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth max‑convolution objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed approximation of max(a*a) and its gradient.
    a = softmax(y) is a probability vector (sum = 1).
    The smooth max uses a log‑sum‑exp with a temperature λ proportional
    to the current maximum value (lam_factor controls the proportionality).
    """
    a = softmax(y)

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # max becomes zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of the smooth max ---
    w = exp_shift / Z                     # ∂smooth_max/∂conv
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: np.ndarray,
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14,
                     "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on the non‑smooth objective
# ----------------------------------------------------------------------


def subgrad_descent_full(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    eta0: float = 0.40,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).
    The gradient is taken at the active convolution index.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i_arr = np.arange(n)
    t = 0
    while time.time() < deadline:
        # convolution and its max index
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of c_k = a*a at index k
        j = k - i_arr
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # step size schedule (diminishing)
        eta = eta0 / np.sqrt(t + 1.0)

        # projected gradient step
        a = project_simplex(a - eta * grad)

        # keep best
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if (t + 1) % 700 == 0 and np.random.rand() < 0.09:
            noise = np.random.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)

        t += 1
        # leave a tiny safety margin
        if deadline - time.time() < 0.12:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbs (random and targeted)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """Random pairwise mass transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.12

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.09,
) -> tuple[np.ndarray, float]:
    """Greedy move from the largest contributor of the peak to the smallest."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Parametric shapes for initialisation

def generate_sequence(
    n: int,
    shape: str,
    param,
) -> np.ndarray:
    """
    Produce a non‑negative probability vector of length n.

    Supported shapes:
        - "random": Dirichlet(1,…,1) (param ignored)
        - "edge":   param = (edge_fraction, ratio)
                    edge_fraction ∈ (0,0.5) controls number of edge entries,
                    ratio > 1 gives edge/value relative to centre.
        - "exp":    param = beta > 0   (exponential decay from edges)
        - "power":  param = p > 0    (power‑law decay from edges)
    """
    rng = np.random.default_rng()
    if shape == "random":
        a = rng.dirichlet(np.ones(n))
        return a

    if shape == "edge":
        edge_frac, ratio = param
        m_edge = max(1, int(n * edge_frac))
        r = max(1.0, ratio)
        # solve: 2*m_edge*alpha + (n-2*m_edge)*beta = 1,   alpha = r*beta
        beta = 1.0 / (2 * m_edge * r + (n - 2 * m_edge))
        alpha = r * beta
        a = np.full(n, beta)
        a[:m_edge] = alpha
        a[-m_edge:] = alpha
        return a

    if shape == "exp":
        beta = max(1e-9, param)
        idx = np.arange(n)
        dist = np.minimum(idx, n - 1 - idx)
        a = np.exp(-beta * dist)
        a = a / a.sum()
        return a

    if shape == "power":
        p = max(1e-9, param)
        idx = np.arange(n)
        dist = np.minimum(idx, n - 1 - idx) + 1.0
        a = dist ** (-p)
        a = a / a.sum()
        return a

    raise ValueError(f"Unsupported shape: {shape}")

        # 1) Choose a shape and build an initial vector
        # --------------------------------------------------------------
        mode = rng.choice(["random", "edge", "exp", "power", "best"])
        if mode == "random":
            a_start = rng.dirichlet(np.ones(n))
        elif mode == "edge":
            edge_frac = rng.uniform(0.008, 0.045)          # fraction of edge entries
            ratio = rng.uniform(2.0, 14.0)                # edge / centre ratio
            a_start = generate_sequence(n, "edge", (edge_frac, ratio))
        elif mode == "exp":
            beta = rng.uniform(0.001, 0.025)
            a_start = generate_sequence(n, "exp", beta)
        elif mode == "power":
            p = rng.uniform(0.2, 2.2)
            a_start = generate_sequence(n, "power", p)
        else:   # perturb current best
            a_start = best_a.copy()
            a_start = project_simplex(
                a_start + rng.normal(scale=0.03, size=n)
            )

        # --------------------------------------------------------------
        # 2) Continuation L‑BFGS (smooth objective)
        # --------------------------------------------------------------
        y0 = np.log(a_start + 1e-12)
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] L‑BFGS improved to {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Projected sub‑gradient descent (longer than reference)
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 1.2)   # about one second per call
        best_a, best_score = subgrad_descent_full(
            best_a,
            n,
            fft_len,
            sub_deadline,
            eta0=0.42,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 4) Targeted hill‑climb
        # --------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            tc_deadline,
            max_moves=2800,
            frac=0.09,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 5) Random hill‑climb (final polish for this iteration)
        # --------------------------------------------------------------
        rc_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            rc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Deep final refinement (use remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.2:
        # brief L‑BFGS passes with very low smoothing to polish
        for lam in lam_factors[-12:]:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 500, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.2e} score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step‑function sequence with a low

    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.5  # safety margin

    # Problem size – large enough to beat 1.503 but not too costly
    n = 4000

    # Warm‑start from a previously stored construction if available
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)

    # Allocate the whole budget to the chosen size

    # Final evaluation with the given (secure) routine
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_val:.6f}")

    # expose for potential reuse in later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq

**Explanation of improvements**

* **Diverse parametric initialisations** – edge‑biased, exponential and power‑law shapes (plus random Dirichlet) explore a much richer part of the feasible space than pure random starts.
* **Richer continuation schedule** – 45 smoothing stages (down to 5·10⁻⁹) give a finer path from a smooth surrogate to the true max‑convolution, reducing the bias of the smooth‑max approximation.
* **Longer projected sub‑gradient descent** – each call now runs for ≈1 s (instead of 0.9 s) with a slightly larger initial step size and occasional jitter, allowing the method to drive the true peak lower.
* **More aggressive hill‑climbs** – higher move fractions and more iterations improve the polishing phase.
* **Final low‑λ L‑BFGS polishing** – after the main loop, a short series of passes with very low smoothing refines the solution further.
* **Single large problem size (n = 4000)** – concentrates the computational budget where it matters most; larger n give a lower theoretical floor, but the extra FFT cost outweighs the marginal benefit within the 1000 s limit.

All helper functions are top‑level, no I/O is performed, and the code respects the time budget while aiming for a score well below the previous 1.509 → target 1.503.
