def smooth_max_and_grad_a(a, lam_factor):
    n = a.size
    # conv full
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted/lam)
    Z = exp_shift.sum()
    w = exp_shift / Z  # derivative of smooth max w.r.t conv
    # gradient w.r.t a:
    # conv_rev: reversed a conv with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]
    grad = 2.0 * g_vec  # gradient of smooth max
    f = 2.0 * n * (max_c + lam * np.log(Z))
    return f, grad
```

"""
Low‑score step‑function search (≈ 1000 s budget).

The core idea
-------------
* Initialise a non‑uniform distribution (exponential decay) – this already
  reduces the a⋆a peak compared with a uniform sequence.
* Repeatedly solve a linear programme:
      maximise   Σ g_i
      subject to  (a * g)_k ≤ M   for all shifts k,
                  g_i ≥ 0,
  where M = max(a * a).  The LP raises the ℓ₁‑norm of the sequence while
  keeping the maximal autocorrelation unchanged.
* Mix the LP solution into the current sequence (convex combination) and
  renormalise.  This strictly lowers the peak M unless the LP gives no new
  direction.
* After a few LP‑mixing rounds, polish the result with fast hill‑climbs
  (targeted and random mass‑transfer) which are cheap and improve the
  surrogate objective 2·n·max(conv).
* Finally evaluate the official objective and store the construction for
  possible reuse.

The algorithm works on a moderate size (n≈1500) – large enough to obtain
scores well below 1.51 while keeping each LP moderate (≈ 2·10⁶ non‑zeros).
All operations respect the time budget and use only NumPy / SciPy.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Hill‑climb utilities (identical to the previous state‑of‑the‑art)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb on the simplex (strictly improving)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Focuses on the shift that currently attains the maximal convolution."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = mask
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑program helper (conv(a,g) ≤ M)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n matrix for the linear constraints
      (a * g)_k = Σ_i a_i·g_{k‑i} ≤ M .
    Row k corresponds to shift k, column j to g_j.
    """
    n = a.shape[0]
    # each diagonal with offset i contains the constant a[i]
    diags = [np.full(n, a[i]) for i in range(n)]
    offsets = np.arange(n)               # offsets 0 … n‑1
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csc")
    return A

    ``height_sequence_1`` for possible reuse.
    """

    # allowing a good bound (the theoretical lower bound ≈ 1).

    n = 1500

    rate = rng.uniform(0.001, 0.02)

    a = a / a.sum()                     # ℓ₁‑normalised (sum = 1)

    best_score = fast_score(best_a, fft_len_conv, n)

    # --------------------------------------------------------------
    while time.time() < deadline - 5.0 and iteration < 30:

        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

            method="highs",

            print(f"[{iteration}] LP returned zero mass – aborting")
            break

        mixed = False

        for _ in range(5):
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = a_candidate / a_candidate.sum()
            fa_c = np.fft.fft(a_candidate, fft_len_conv)
            conv_c = np.real(np.fft.ifft(fa_c * fa_c))[: 2 * n - 1]
            M_new = conv_c.max()
            if M_new < M * 0.9999:                # require a tiny decrease
                a = a_candidate
                M = M_new
                mixed = True
                break
            t *= 0.5

        if not mixed:
            # No improvement – fall back to a quick random hill‑climb to jitter the vector
            a, _ = hill_climb(
                a,
                best_score,
                fft_len_conv,
                n,
                deadline=time.time() + 0.6,
                max_iter=800,
                max_move_frac=0.10,
            )
        # Record the best surrogate score seen so far
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration}] LP‑mix improved fast score = {best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # Final polishing using fast hill‑climbs (targeted + random)
    # --------------------------------------------------------------
    while time.time() < deadline:
        # focused hill‑climb
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline=deadline,
            max_iter=800,
            move_frac=0.30,
        )
        # generic random hill‑climb
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline=deadline,
            max_iter=2000,
            max_move_frac=0.20,
        )
        # very short safety break to avoid overrunning the budget
        if time.time() > deadline - 1.0:
            break

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
