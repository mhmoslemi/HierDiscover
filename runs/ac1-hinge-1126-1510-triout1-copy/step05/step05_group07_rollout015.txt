def simulated_annealing(a, cur_score, fft_len, n, deadline,
                        init_temp=0.01, final_temp=1e-6, max_steps=5000):
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score
    temp = init_temp
    t_start = time.time()
    step = 0
    while time.time() < deadline and step < max_steps:
        # propose new a by moving small mass between random indices
        i, j = rng.integers(0, n, size=2)
        if i==j: continue
        delta = cur_a[i] * rng.random() * 0.05
        if delta == 0: continue
        cand_a = cur_a.copy()
        cand_a[i] -= delta
        cand_a[j] += delta
        # maybe add small gaussian noise for additional randomization every few steps?
        # ensure nonneg
        # project onto simplex not necessary because sum unchanged
        cand_score = fast_score(cand_a, fft_len, n)

def simulated_annealing(a, cur_score, fft_len, n, deadline, rng,
                        init_temp=0.01, final_temp=1e-6, max_steps=5000):
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score
    temp = init_temp
    step = 0
    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.05  # move up to 5% of i's mass
        if delta <= 0:
            continue
        cand_a = cur_a.copy()
        cand_a[i] -= delta
        cand_a[j] += delta
        # projected? sum unchanged and all nonnegative given delta <= cur_a[i]
        new_score = fast_score(cand_a, fft_len, n)
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temp):
            cur_a = cand_a
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        step += 1
        # cooling schedule
        if step % (max_steps // 10) == 0:
            temp = max(final_temp, temp * 0.5)
    return best_a, best_score

    keep the best one (fast evaluation by FFT).

2.  **Intensive refinement** – a loop that repeatedly
    * continuation L‑BFGS on a smooth approximation of the max‑convolution,
    * long projected sub‑gradient descent on the exact max‑convolution,
    * a short simulated‑annealing phase,
    * targeted and random hill‑climbs.
    The loop runs until the time budget is almost exhausted.

3.  **Final polishing** – a low‑λ L‑BFGS continuation that
    pushes the solution a little further.

All operations stay on the probability simplex (sum = 1, a_i ≥ 0) so
the evaluation score is invariant under scaling.  The routine returns
the best sequence found within the allotted 1000 s.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Core FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """Smooth approximation of max(a*a) and its gradient w.r.t. logits y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max
    w = exp_shift / Z  # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """Continuation L‑BFGS with a decreasing smoothing schedule."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple hill‑climbs
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb moving mass from the biggest contributor to the
    smallest contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (small stochastic polishing)
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_temp: float = 0.01,
    final_temp: float = 1e-6,
    max_steps: int = 3000,
) -> tuple[np.ndarray, float]:
    """Very light SA that moves a tiny amount of mass between two positions."""
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score
    temp = init_temp
    step = 0

    while time.time() < deadline and step < max_steps:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.05  # up to 5 % of source
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temp):
            cur_a = cand
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()

        step += 1
        if step % (max_steps // 10) == 0:
            temp = max(final_temp, temp * 0.5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its sum will be 1 after
    internal normalisation).
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)  # for legacy np.random calls
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size – we stick to n = 4000 (a good trade‑off)
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start with any previously exported sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        if prev.size >= n:
            a0 = prev[:n].copy()
            a0 = np.clip(a0, 0.0, np.inf)
            s = a0.sum()
            if s > 1e-12:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)

    # --------------------------------------------------------------
    # Phase 1 – cheap random exploration (≈15 % of total time)
    # --------------------------------------------------------------
    explore_deadline = min(deadline, start_time + (deadline - start_time) * 0.15)
    while time.time() < explore_deadline:
        mode = rng.choice(["uniform", "edge", "power", "exp", "dirichlet"])
        if mode == "uniform":
            a = np.full(n, 1.0 / n)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
        elif mode == "power":
            p = rng.uniform(0.2, 2.5)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
        else:  # dirichlet
            alpha = rng.uniform(0.1, 5.0)
            a = rng.gamma(alpha, 1.0, size=n)
            a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[explore] mode={mode} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – intensive refinement until the budget is almost spent
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline - 5.0:
        # ---- 2.1 L‑BFGS continuation (smooth objective) ----
        y_start = np.log(best_a + 1e-12)
        lam_sched = np.geomspace(0.5, 5e-8, num=35)   # finer low‑λ tail
        y_opt = schedule_optimize(
            y_start=y_start,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_sched,
            maxiter_per_stage=500,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[refine {iteration}] L‑BFGS score={best_score:.6f}")

        # ---- 2.2 Exact sub‑gradient descent (more iterations) ----
        sub_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = subgrad_descent(
            best_a,
            n,
            fft_len,
            max_iters=8000,
            deadline=sub_deadline,
            eta0=0.36,
        )

        # ---- 2.3 Simulated annealing (short) ----
        sa_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len,
            n,
            deadline=sa_deadline,
            rng=rng,
            init_temp=0.01,
            final_temp=1e-6,
            max_steps=2500,
        )

        # ---- 2.4 Targeted hill‑climb ----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline=hc_deadline,
            max_moves=3500,
            frac=0.07,
        )

        # ---- 2.5 Random hill‑climb (polish) ----
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline=deadline,
            max_iter=4000,
        )

        iteration += 1
        if iteration % 5 == 0:
            print(f"[status] iter={iteration} best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final low‑λ L‑BFGS polishing (uses remaining time)
    # --------------------------------------------------------------
    y = np.log(best_a + 1e-12)
    final_lams = np.geomspace(0.5, 1e-9, num=50)
    for lam in final_lams:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-14, "disp": False},
        )
        y = res.x
        best_a = softmax(y)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final‑L‑BFGS] lam={lam:.3g} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence score = {final_val:.6f}")

    # expose for possible warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
