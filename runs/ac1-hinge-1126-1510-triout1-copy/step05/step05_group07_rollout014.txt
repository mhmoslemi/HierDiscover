def adam_optimize(y0, n, fft_len_conv, fft_len_g, total_steps, lr=0.1, lam_initial=0.5, lam_final=5e-6, beta1=0.9, beta2=0.999, epsilon=1e-8, deadline=None):

    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_y = y.copy()
    best_score = np.inf

    for t in range(1, total_steps+1):
        if deadline is not None and time.time() > deadline:
            break
        lam_factor = lam_initial * (lam_final/lam_initial)**(t/total_steps) # exponential schedule

        f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        # Adam update
        m = beta1*m + (1-beta1)*grad_y
        v = beta2*v + (1-beta2)*(grad_y**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + epsilon)

        # evaluate actual score occasionally
        if t % 50 == 0:
            a = softmax(y)
            current_score = fast_score(a, fft_len_conv, n)
            if current_score < best_score:
                best_score = current_score
                best_y = y.copy()
    # final best
    best_a = softmax(best_y)
    best_score = fast_score(best_a, fft_len_conv, n)
    return best_a, best_score

```python
"""
Propose a candidate step‑function vector minimizing
    2·n·max(a∗a)/(sum a)^2
using an Adam optimiser on a smooth max‑approximation, followed by
sub‑gradient descent and simple hill‑climbing.  The routine works
entirely on the probability simplex (∑a=1, a_i≥0) via a soft‑max
parameterisation, so the evaluate_sequence() scaling invariance is
preserved.  Several random initialisations are tried within the
time‑budget (≈1000 s) and the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT, trimmed to length 2n‑1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation 2·n·max(a*a) (sum a is assumed to be 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution and its gradient (w.r.t. logits y)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Returns smooth approximation of 2·n·max(a*a) and its gradient w.r.t.
    the logits y (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                      # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    # we compute convolution of a_rev and w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    total_steps: int,
    lr: float = 0.2,
    lam_initial: float = 0.5,
    lam_final: float = 5e-6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    deadline: float = None,
):
    """
    Adam optimiser on the smoothed max‑convolution objective.
    Returns the best probability vector seen and its (exact) fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_y = y.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for t in range(1, total_steps + 1):
        if deadline is not None and time.time() > deadline:
            break

        # exponential schedule for the smoothing factor
        lam_factor = lam_initial * ((lam_final / lam_initial) ** (t / total_steps))

        f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation (every 50 steps)
        if t % 50 == 0:
            a = softmax(y)
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_y = y.copy()
                best_a = a.copy()

    # final best vector
    best_a = softmax(best_y)
    best_score = fast_score(best_a, fft_len_conv, n)
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # gradient of max at active index k: (∂/∂a_i) Σ_j a_j a_{k-j}
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation to escape plateaus
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climb (polishing)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb with projection onto simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    bound.  The vector is kept on the probability simplex (sum = 1) and
    the optimisation combines Adam on a smooth max‑approximation,
    projected sub‑gradient descent, and a final random hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size – we use n=4000 (fits comfortably in memory) and
    # fall back to a larger size if time permits.
    # --------------------------------------------------------------

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # initialise from any previous construction
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)
        if prev_arr.size >= n:
            a0 = prev_arr[:n].copy()
        else:
            # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, prev_arr.size)
            new_x = np.linspace(0.0, 1.0, n)
            a0 = np.interp(new_x, old_x, prev_arr)
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_y = np.log(best_a + 1e-12)
    best_score = fast_score(best_a, fft_len_conv, n)

    iteration = 0

    # --------------------------------------------------------------
    # Main optimisation loop – repeat random restarts until time runs out
    # --------------------------------------------------------------
    while time.time() < total_deadline - 5.0:
        # pick a random initialisation mode
        mode = rng.choice(["random", "exp", "power", "edge", "perturb", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))

        # ----------------------------------------------------------

            best_a,
            n,
            fft_len_conv,

"""
Propose a candidate step‑function vector minimizing
    2·n·max(a∗a)/(sum a)^2

The algorithm works on the probability simplex (∑a=1, a_i≥0) via a
soft‑max parametrisation.  For each restart we:

1. initialise a log‑probability vector y (several heuristic modes);
2. run Adam on a smooth approximation of the max‑convolution;
3. polish with a short exact sub‑gradient descent;
4. finish with a random hill‑climb.

All phases respect a global time‑budget (≈ 1000 s).  The best
sequence found is finally evaluated with ``evaluate_sequence`` and
returned.  The implementation uses only NumPy and the standard
library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by Adam)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    Returns (value, grad_y).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                      # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z  # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    total_steps: int,
    lr: float = 0.2,
    lam_initial: float = 0.5,
    lam_final: float = 5e-6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    deadline: float = None,
):
    """
    Adam optimisation of the smoothed max‑convolution.
    Returns the best probability vector seen and its exact fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_y = y.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for t in range(1, total_steps + 1):
        if deadline is not None and time.time() > deadline:
            break

        # exponential schedule for the smoothing factor
        lam_factor = lam_initial * ((lam_final / lam_initial) ** (t / total_steps))

        _, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional exact evaluation
        if t % 50 == 0:
            a = softmax(y)
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_y = y.copy()
                best_a = a.copy()

    # final best vector
    best_a = softmax(best_y)
    best_score = fast_score(best_a, fft_len_conv, n)
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).
    Returns the best vector seen and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)  # index array for vectorised gradient

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of the max at the active index k
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional random perturbation to escape flat plateaus
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random hill‑climb (polishing)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

# Main entry point – called by the evaluation harness

    iteration = 0

    # ------------------------------------------------------------------

    while time.time() < total_deadline - 5.0:
