def param_search(n, time_budget, rng):
    best_a = None
    best_score = np.inf
    start = time.time()
    while time.time() < start + time_budget:
        # choose a random shape family
        family = rng.choice([...])
        if family == 'sym_exp':
            rate = rng.uniform(0.0005, 0.02)
            a = np.exp(-rate * np.arange(n)) + np.exp(-rate * np.arange(n-1, -1, -1))
        elif family == 'sym_power':
            p = rng.uniform(0.1, 4.0)
            a = (np.arange(1, n+1, dtype=float) ** (-p)) + (np.arange(n, 0, -1, dtype=float) ** (-p))
        elif family == 'beta':
            alpha = rng.uniform(0.1, 2.0)
            x = np.linspace(0, 1, n)
            a = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
        elif family == 'edge_mix':
            m = max(1, int(rng.uniform(0.001, 0.05) * n))
            lam_mix = rng.random()
            edge = np.zeros(n)
            edge[:m] = 1.0
            edge[-m:] = 1.0
            edge /= edge.sum()
            uniform = np.full(n, 1.0 / n)
            a = lam_mix * edge + (1 - lam_mix) * uniform
        elif family == 'arcsine':
            idx = np.arange(n, dtype=float) + 0.5
            a = 1.0 / np.sqrt(idx * (n - idx + 1))
        else:
            a = np.full(n, 1.0 / n)
        # Normalize sum to 1
        s = a.sum()
        if s <= 0:
            continue
        a /= s
        # Ensure nonneg
        a = np.maximum(a, 0)
        # Compute fast_score
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[param] family={family}, score={score:.6f}")
    return best_a, best_score

def optimise_n(n, budget_s, seed, init_a=None):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    fft_len = next_pow2(2*n-1)
    fft_len_g = next_pow2(3*n - 2)

    # initialize with init_a if given, else with param search.
    best_a = init_a if init_a is not None else np.full(n,1.0/n)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # param search time allocation might be half of remaining budget.
    param_budget = min(0.2 * (deadline - time.time()), 30.0) # e.g., 30 secs or 20%

    if param_budget > 5:
        a0, sc0 = param_search(n, param_budget, rng)
        if a0 is not None and sc0 < best_score:
            best_a = a0
            best_score = sc0
            best_y = np.log(best_a + 1e-12)
            print(f"[init] param search best fast score {best_score:.6f}")

    # L-BFGS continuation schedule
    # Choose lam_factors covering wide range.
    lam_factors = np.geomspace(0.4, 5e-7, num=25)

    # while time left for L-BFGS loops:
    while time.time() < deadline - 10.0:
        # start L-BFGS from current best_y with random perturbation?
        y_start = best_y + rng.normal(scale=0.1, size=n)  # add perturbation

        y_opt = schedule_optimize(y_start, n, fft_len, fft_len_g,
                                  lam_factors, maxiter_per_stage=300)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[L-BFGS] new best fast score {best_score:.6f}")

        # small subgradient descent
        sub_deadline = min(deadline, time.time() + 2.0)
        best_a, best_score = subgrad_descent(best_a, n, fft_len,
                                             max_iters=5000,
                                             deadline=sub_deadline,
                                             eta0=0.4)
        best_y = np.log(best_a + 1e-12)
        # targeted hill climb
        hc_deadline = min(deadline, time.time() + 1.5)
        best_a, best_score = targeted_hill_climb(best_a, best_score,
                                                 fft_len, n,

                                                 max_moves=2000)

"""
Propose a low‑score step‑function using a hybrid of
  * fast low‑dimensional parametric search,
  * continuation L‑BFGS on a smoothed max‑convolution,
  * projected sub‑gradient descent on the true max‑convolution,
  * greedy and random hill‑climbs.

The parametric families (symmetric power, symmetric exponential,
inverse‑product power, Beta, edge‑mix and arcsine) explore shapes that
push mass to the ends while keeping the centre light – a pattern known
to flatten the autocorrelation.  The best candidate of this cheap search
is then refined with the exact sub‑gradient method and several polishing
steps.  The whole routine respects the 1000 s budget and returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities (FFT convolution, score, soft‑max, projection)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using real FFT, trimmed to length 2n‑1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector a."""
    return float(2.0 * n * conv_fft(a, fft_len, n).max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape factories for the cheap parametric search
# ----------------------------------------------------------------------


def shape_sym_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p) + (n + 1 - i) ** (-p)
    a[i == (n + 1) // 2] /= 2  # compensate double count at the centre for odd n
    return a


def shape_sym_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    return a


def shape_invprod_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i * (n + 1 - i)) ** (-p)
    return a


def shape_beta(n: int, alpha: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a[np.isnan(a)] = 0.0
    return a


def shape_edge_mix(n: int, m: int, lam: float) -> np.ndarray:
    edge = np.zeros(n, dtype=float)
    edge[:m] = 1.0
    edge[-m:] = 1.0
    if edge.sum() > 0:
        edge /= edge.sum()
    uniform = np.full(n, 1.0 / n, dtype=float)
    return lam * edge + (1.0 - lam) * uniform


def shape_arcsine(n: int) -> np.ndarray:
    idx = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(idx * (n - idx + 1.0))
    return a


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """Smooth approximation of 2·n·max(a*a) and its gradient w.r.t. logits y."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient via the soft‑max chain
    w = exp_shift / Z                 # ∂smooth_max/∂conv
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors,
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """Continuation L‑BFGS with decreasing smoothing parameter."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    idx = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # gradient of the (non‑smooth) max at the active index k:
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        # keep the best vector seen so far
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional tiny random perturbation to shake out of plateaus
        if (t + 1) % 300 == 0 and np.random.rand() < 0.08:
            a = project_simplex(a + np.random.normal(scale=0.006, size=n))

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb helpers (same logic as the reference implementation)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 1500,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 2500,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Greedy move from the biggest contributor of the peak to the smallest."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # random fallback
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1
    return best, best_score


# ----------------------------------------------------------------------
# Parametric random search (very cheap – only O(n log n) per eval)
# ----------------------------------------------------------------------


def param_search(
    n: int,
    time_budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Explore a handful of hand‑crafted families and keep the best."""
    fft_len = next_pow2(2 * n - 1)
    best_a = np.full(n, 1.0 / n, dtype=float)
    best_score = fast_score(best_a, fft_len, n)

    families = ["sym_power", "sym_exp", "invprod_power", "beta", "edge_mix", "arcsine"]
    start = time.time()
    while time.time() - start < time_budget:
        family = rng.choice(families)
        if family == "sym_power":
            p = rng.uniform(0.05, 3.0)
            a = shape_sym_power(n, p)
        elif family == "sym_exp":
            rate = rng.uniform(2e-4, 0.05)
            a = shape_sym_exp(n, rate)
        elif family == "invprod_power":
            p = rng.uniform(0.05, 2.0)
            a = shape_invprod_power(n, p)
        elif family == "beta":
            a = shape_beta(n, rng.uniform(0.05, 2.5))
        elif family == "edge_mix":
            m_frac = rng.uniform(0.001, 0.06)
            m = max(1, int(m_frac * n))
            lam = rng.random()
            a = shape_edge_mix(n, m, lam)
        else:  # arcsine
            a = shape_arcsine(n)

        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total <= 0.0:
            continue
        a /= total
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[param] family={family:<12} score={sc:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Optimisation driver for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """Main optimisation loop for a fixed length n."""
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ---------- initialise ----------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = np.asarray(prev_seq[:n], dtype=float)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n, dtype=float)
    else:
        a0 = np.full(n, 1.0 / n, dtype=float)

    best_a = a0.copy()
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # ---------- cheap parametric search ----------
    remaining = deadline - time.time()
    param_budget = min(30.0, max(5.0, 0.2 * remaining))
    if param_budget > 5.0:
        a_param, sc_param = param_search(n, param_budget, rng)
        if sc_param < best_score:
            best_a, best_score, best_y = a_param, sc_param, np.log(a_param + 1e-12)
            print(f"[{n}] param‑search best fast score = {best_score:.6f}")

    # ---------- iterative refinement ----------
    fft_len_g = next_pow2(3 * n - 2)
    lam_factors = np.geomspace(0.5, 5e-7, num=30)

    while time.time() < deadline - 4.0:
        # 1) continuation L‑BFGS on a smooth approximation
        y_start = best_y + rng.normal(scale=0.07, size=n)
        y_opt = schedule_optimize(
            y_start=y_start,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_a, best_score, best_y = a_opt, score_opt, y_opt
            print(f"[{n}] L‑BFGS new best = {best_score:.6f}")

        # 2) exact sub‑gradient descent (short burst)
        sub_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = subgrad_descent(
            best_a, n, fft_len, max_iters=4000, deadline=sub_deadline, eta0=0.48
        )
        best_y = np.log(best_a + 1e-12)

        # 3) targeted greedy hill‑climb
        tc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = targeted_hill_climb(
            best_a, best_score, fft_len, n, tc_deadline, max_moves=3000, frac=0.07
        )
        best_y = np.log(best_a + 1e-12)

        # 4) random hill‑climb for polishing
        rc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len, n, rc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)

    # ---------- final low‑λ L‑BFGS sweep ----------
    while time.time() < deadline - 1.5:
        for lam in lam_factors:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 500, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[{n}] final‑L‑BFGS lam={lam:.2e} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises
    evaluate_sequence().  Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin

    # Choose a few problem sizes; larger n typically allows a lower bound.
    sizes = [4000, 5000, 6000]
    per_size_budget = (total_deadline - time.time()) / len(sizes)

    # Warm‑start with any previously stored construction.
    prev = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev, dtype=float) if prev is not None else None

    best_overall_a = None
    best_overall_score = np.inf

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 5.0:
            break
        # ensure we keep a small reserve for the next sizes
        slice_budget = min(per_size_budget, remaining - (len(sizes) - idx - 1) * 2.0)
        print(f"[size] Optimising n={n} with budget≈{slice_budget:.1f}s")

        a_n, score_n = optimise_one_size(
            n=n,
            budget_s=slice_budget,
            seed=seed + idx * 12345,
            prev_seq=prev_arr,
        )
        print(f"[size] n={n} best fast score = {score_n:.6f}")

        if score_n < best_overall_score:
            best_overall_score = score_n
            best_overall_a = a_n.copy()

    # Final evaluation using the provided evaluate_sequence()
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose for possible reuse by subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
