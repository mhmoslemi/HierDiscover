def analytic_targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=5000):

    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute conv
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        k = int(np.argmax(conv))
        # compute contributions for each i
        idx = np.arange(n)
        pj = k - idx
        mask = (pj >= 0) & (pj < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[pj[mask]]
        # pick source index among top few
        # generate list of candidate indices sorted by contrib desc
        # we can sample top 5 random among top 10
        top_indices = np.argpartition(-contrib, min(9, n-1))[:10]  # bottom of negative for top
        # filter >0
        top_indices = top_indices[contrib[top_indices] > 0]
        if top_indices.size == 0:
            break
        i = int(rng.choice(top_indices))
        # compute partner weights for shift k
        partner = np.full(n, np.inf)
        partner[mask] = best[pj[mask]]
        # pick target among indices with smallest partner weight, e.g., bottom 10
        cand_js = np.argpartition(partner, min(9, n-1))[:10]
        # ensure not equal to i
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(rng.choice(cand_js))
        # Compute maximum transferable delta: cannot exceed best[i], also target infinite? We will move fraction.
        # Use delta = best[i] * rng.random() * 0.9
        delta = best[i] * rng.random() * 0.9
        if delta <= 0.0:
            continue
        # Perform move
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def adam_optimize(a_start, initial_score, fft_len_conv, fft_len_g, n, deadline, max_iter=3000):

    a = a_start.copy()
    best_a = a.copy()
    best_score = initial_score
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    lam_factor = 1e-5  # start
    while time.time() < deadline and t < max_iter:
        t += 1
        # compute gradient of full objective (fast_score using smooth approx)
        grad = grad_smooth_max(a, lam_factor) * (2.0 * n)  # gradient of f = 2*n* smooth_max
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.5 / (np.sqrt(t))  # base step decays
        a_new = a - step * m_hat / (np.sqrt(v_hat) + epsilon)
        # Projection onto simplex
        a_new = proj_simplex(a_new)
        # Evaluate score
        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        a = a_new
        # Optionally reduce lam_factor gradually to approach true max
        lam_factor = max(lam_factor * 0.99, 1e-7)
    return best_a, best_score

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of `evaluate_sequence` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- a * a (full convolution) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a (length‑n gradient) -----
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient of f w.r.t. y (after soft‑max)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def analytic_targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive, analytically driven hill‑climb that reduces the current
    maximal convolution entry by moving a large fraction of mass from the
    strongest contributor to a location whose partner under the same shift
    is (almost) empty.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # ----- current convolution and the critical shift -----
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # ----- contribution of each index to that shift -----
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        # ----- choose a source index among the heaviest contributors -----
        if not np.any(contrib > 0):
            break
        # top‑10 contributors (or fewer if n < 10)
        cand_src = np.argpartition(-contrib, min(9, n - 1))[:10]
        cand_src = cand_src[contrib[cand_src] > 0]
        i = int(rng.choice(cand_src))

        # ----- partner weights for the chosen shift -----
        partner = np.full(n, np.inf)
        partner[mask] = best[partner_idx[mask]]

        # ----- pick a target index whose partner is minimal -----
        cand_tgt = np.argpartition(partner, min(9, n - 1))[:10]
        cand_tgt = cand_tgt[cand_tgt != i]               # avoid self‑move
        if cand_tgt.size == 0:
            continue
        j = int(rng.choice(cand_tgt))

        # ----- move a large fraction (up to 90 %) of the source mass -----
        delta = best[i] * rng.random() * 0.90
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns the (2·g) term – i.e. the gradient of `smooth_max` (without
    the outer 2·n factor used in the full objective).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)

    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # derivative of smooth max w.r.t a is 2·g
    return 2.0 * g


def adam_optimize(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on the smooth‑max objective.
    The smoothness parameter λ is slowly reduced during the run.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    t = 0
    lam_factor = 5e-5  # start a little larger than the final lam

    while time.time() < deadline and t < max_iter:
        t += 1
        # gradient of the full surrogate objective 2·n·smooth_max
        g = grad_smooth_max(a, lam_factor) * (2.0 * n)

        # Adam update on the simplex (followed by projection)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.30 / np.sqrt(t)          # a simple decaying step size
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        cur_score = fast_score(a_new, fft_len_conv, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new
        # tighten the smoothing gradually
        lam_factor = max(lam_factor * 0.97, 1e-7)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction if possible
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to a probability simplex
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # L‑BFGS continuation schedule (global search)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=45)   # finer λ‑schedule
    maxiter_per_stage = 1000                        # more work per λ‑stage

    iteration = 0
    deadline = time.time() + budget_s - 2.0  # reserve a couple of seconds

    while time.time() < deadline - 6.0:
        # -------------------- initialise a random start --------------------
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.015)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.008, 0.03)))
            edge_val = np.sqrt(n / 2.0) / n
            center_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, center_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best solution
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive analytic targeted hill‑climb -----
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = analytic_targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- random hill‑climb to escape tiny plateaus -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=5000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Adam‑style projected‑gradient descent on the smooth‑max
    # ------------------------------------------------------------------
    pg_deadline = deadline - 2.0
    if time.time() < pg_deadline:
        best_a, best_score = adam_optimize(
            a_start=best_a,
            start_score=best_score,
            fft_len_conv=fft_len_conv,
            n=n,
            deadline=pg_deadline,
            max_iter=5000,
        )
        print(f"[adam] best fast score after Adam = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing random hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        print(f"[final hill] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq  # allow reuse in a later run
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
