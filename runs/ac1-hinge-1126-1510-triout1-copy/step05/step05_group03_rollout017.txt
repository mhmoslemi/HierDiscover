"""
Propose a low‑score non‑negative step‑function sequence.

Approach
--------
1.  Initialise the probability vector `a` (∑a = 1) with a variety of
    seeds, one of which is obtained by a *spectral‑factorisation*
    of a flat target convolution (square‑root of a constant vector in
    the Fourier domain).  This gives a good starting point that already
    spreads the mass so that the convolution is almost uniform.

2.  Optimise the smooth‑max surrogate of the convolution maximum using
    a continuation of L‑BFGS on the log‑soft‑max parametrisation.
    The smoothing parameter λ is decreased geometrically
    (λ = λ_factor·max(conv)) from 0.5 down to ≈5·10⁻⁷.

3.  After each L‑BFGS run perform two cheap hill‑climbs:
        – a *targeted* climb that attacks the shift where the current
          convolution is maximal,
        – a *random* mass‑transfer climb for escaping tiny plateaus.

4.  Refine the best candidate with a projected‑gradient descent that
    uses an Adam‑style adaptive step and a very small smoothing factor.

5.  A final aggressive random hill‑climb uses any remaining time.

All operations keep the vector on the probability simplex (non‑negative,
∑a = 1).  The surrogate used during the search is
    2·n·smooth_max(a∗a) ;
the official ``evaluate_sequence`` is called only once at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (smooth max of the auto‑convolution) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a (where a = softmax(y)) and the
    gradient w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c          # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth max w.r.t. conv entries
    w = exp_shift / Z               # d(smooth_max)/d conv_i

    # back‑prop to a (gradient vector of length n)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex (strictly improving moves).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient (without the outer 2·n factor) of the smooth‑max of a*a.
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    return 2.0 * g  # 2·n factor applied later


# ----------------------------------------------------------------------
# Spectral‑factorisation initialisation (square‑root of a flat target)
# ----------------------------------------------------------------------


def init_sqrt_target(n: int) -> np.ndarray:
    """
    Produce a non‑negative vector whose auto‑convolution is (approximately)
    flat: we take the square‑root of the Fourier transform of a constant
    target and invert it.  The result is mirrored and clipped to guarantee
    non‑negativity, then normalised to a probability vector.
    """
    L = next_pow2(2 * n - 1)            # length for circular convolution
    target = np.zeros(L, dtype=float)
    target[: 2 * n - 1] = 1.0 / (2 * n - 1)    # flat target (linear convolution)

    t_fft = np.fft.fft(target)
    # sqrt in polar form: sqrt(|z|) * exp(i * arg(z)/2)
    amp = np.abs(t_fft)
    phase = np.angle(t_fft)
    a_fft = np.sqrt(amp) * np.exp(1j * phase / 2.0)

    a_full = np.real(np.fft.ifft(a_fft))
    # keep the first n entries, enforce symmetry and non‑negativity
    a = a_full[:n]
    a = (a + a[::-1]) / 2.0           # force symmetry (helps the convolution)
    a = np.maximum(a, 0.0)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – try to reuse a previous construction, otherwise use 8 k
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and len(prev) >= 8000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 8000
        best_a = None

    # ------------------------------------------------------------------
    # Initialise a probability vector
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = init_sqrt_target(n)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len_conv, n)

    print(f"[init] n={n}, fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS continuation schedule (global optimisation)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 5e-7, num=40)   # finer λ‑schedule
    maxiter_per_stage = 800

    iteration = 0
    while time.time() < deadline - 5.0:
        # ----------- pick a start point ---------------------------------
        mode = rng.choice(["random", "exp", "power", "edge", "sqrt", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.008, 0.025)))
            a0 = np.full(n, 1.0 / n)
            a0[:m_edge] = a0[-m_edge:] = 0.5 / m_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "sqrt":
            a0 = init_sqrt_target(n)
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----------- continuation L‑BFGS -------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score={best_score:.6f}")

        # ------------------------------------------------------------------
        # Aggressive targeted hill‑climb
        # ------------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ------------------------------------------------------------------
        # Random hill‑climb (helps escape tiny plateaus)
        # ------------------------------------------------------------------
        rc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=6000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Projected‑gradient descent with Adam‑style momentum (λ → 0)
    # ------------------------------------------------------------------
    lam_factor_pg = 1e-5
    step0 = 0.28
    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    pg_iter = 0
    while time.time() < deadline - 2.0 and pg_iter < 2500:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)   # gradient without outer 2·n
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** pg_iter)
        v_hat = v / (1 - beta2 ** pg_iter)

        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * m_hat / (np.sqrt(v_hat) + eps))
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            lam_factor_pg = max(lam_factor_pg * 0.9, 1e-7)

    print(f"[pg] projected‑gradient iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq          # store for later runs
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
