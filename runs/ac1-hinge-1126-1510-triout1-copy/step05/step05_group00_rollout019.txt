def subgradient_restart(start_a, max_iters, deadline):
    a = start_a.copy()
    best_local_a = a.copy()
    best_local_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    while it < max_iters and time.time() < deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12 * max_val).astype(np.float64)
        # gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n-1):(n-1+n)]
        v = beta * v + (1-beta) * g
        step = 0.5 / np.sqrt(it+1)
        a_candidate = simplex_projection(a - step*v)
        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_local_score:
            best_local_score = cand_score
            best_local_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_local_a, best_local_score
```

def lp_step(a, fft_len_conv, n, t=0.3):
    # compute convolution
    conv = full_conv(a, fft_len_conv)
    max_val = conv.max()
    # select active constraints indices
    tol = max_val * 1e-6   # small tolerance
    active_idx = np.where(conv >= max_val - tol)[0]
    if active_idx.size < 1:
        active_idx = np.array([np.argmax(conv)], dtype=int)
    # Build constraint matrix
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_idx):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # for j from start to end => f_{k-j}
        # vectorization: j = start..end
        js = np.arange(start, end+1)
        # corresponding f indices = k - js
        f_vals = a[k - js]
        rows.extend([row_idx] * len(js))
        cols.extend(js.tolist())
        data.extend(f_vals.tolist())
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), max_val)
    # Solve LP
    c = -np.ones(n)
    # Using scipy.linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback simple: use current a
        return a.copy()
    g0 = res.x
    sum_g = g0.sum()
    if sum_g <= 0:
        return a.copy()
    g = g0 / sum_g  # normalize to sum 1
    # blend
    a_new = (1 - t) * a + t * g
    # project onto simplex (should remain sum=1)
    a_new = simplex_projection(a_new)
    return a_new

"""
Propose a step‑function sequence that minimises the evaluation metric.

Algorithm Overview
------------------
1. Initialise a probability vector `a` (sum = 1) – either from the supplied
   `height_sequence_1` or a random “edge‑heavy’’ start.
2. Perform many cheap sub‑gradient restarts: a few hundred short
   projected sub‑gradient runs from random starts to explore the landscape.
3. Refine the current best with a long sub‑gradient‑with‑momentum phase.
4. Optional L‑BFGS continuation on a smoothed log‑sum‑exp approximation
   (helps escape shallow basins).
5. Linear‑programming “LP‑step’’ – solve a small LP that finds a vector `g`
   maximising its ℓ₁‑norm under the current max‑convolution bound.
   Blend `a ← (1‑t)·a + t·g` and repeat a few times.  This is the
   deterministic version of the LP‑based improvement described in the
   problem statement.
6. A short simulated‑annealing phase followed by an aggressive greedy
   hill‑climb with many moves.
7. Return the best list of floats; the external `evaluate_sequence`
   verifies the final score.

All operations are performed on the probability simplex
(`a_i ≥ 0, Σ a_i = 1`) so the evaluation function is scale‑invariant.
FFT based convolutions give O(n log n) speed (n = 4000 by default).
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when Σa = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max re‑parameterisation (used by L‑BFGS)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of a*a via log‑sum‑exp; returns (obj, grad) w.r.t. y."""
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑prop through convolution (a_rev * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Continuation L‑BFGS on the smoothed objective."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Very rare – just skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """Long projected sub‑gradient descent with momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def subgradient_limited(a_start: np.ndarray,
                        iters: int,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int):
    """A short sub‑gradient run with a fixed number of iterations."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9

    for it in range(iters):
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)
        a = simplex_projection(a - step * v)

        cand_score = fast_score(a, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """Adam on the smoothed log‑sum‑exp objective."""
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """Simple simulated annealing that moves a random fraction of mass."""
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 15000):
    """Greedy hill‑climb (strictly improving moves)."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse set of random starting distributions."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'spike'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike – a single heavy point plus uniform background
        a = np.full(n, 1.0 / n)
        pos = rng.integers(0, n)
        extra = 0.5
        a[pos] += extra
        a /= a.sum()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def lp_improve_step(a: np.ndarray,
                    fft_len_conv: int,
                    n: int,
                    tol_factor: float = 1e-6,
                    blend_t: float = 0.30) -> np.ndarray:
    """
    One LP‑based improvement iteration.
    Solve   max Σ g_j   s.t.   (a * g)_k ≤ M   for active k,
    g ≥ 0, then blend g into a.
    """
    conv = full_conv(a, fft_len_conv)
    max_val = conv.max()
    # Identify constraints that are (almost) active
    tol = max_val * tol_factor
    active_idx = np.where(conv >= max_val - tol)[0]
    if active_idx.size == 0:
        active_idx = np.array([np.argmax(conv)], dtype=int)

    # Build sparse A_ub where each row k corresponds to
    # Σ_i a_i * g_{k-i} ≤ max_val
    rows = []
    cols = []
    data = []
    for row_id, k in enumerate(active_idx):
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        js = np.arange(start_j, end_j + 1, dtype=int)
        f_vals = a[k - js]                # a_i where i = k-j
        rows.extend([row_id] * len(js))
        cols.extend(js.tolist())
        data.extend(f_vals.tolist())

    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    b_ub = np.full(len(active_idx), max_val, dtype=float)

    # LP: maximise Σ g_j  ⇔ minimise −Σ g_j
    c = -np.ones(n, dtype=float)

    res = linprog(c,
                  A_ub=A_ub,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        # If LP fails, return the original vector unchanged
        return a

    g = res.x
    # Blend a with the LP solution and re‑project onto the simplex
    a_new = (1.0 - blend_t) * a + blend_t * g
    a_new = simplex_projection(a_new)
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimizing the given evaluation.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise a (maybe from a provided construction)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # interpolate / stretch to length n
            old_n = init_seq.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Phase A – many cheap sub‑gradient restarts
    phase_a_end = start_time + 0.25 * budget_s
    while time.time() < phase_a_end:
        a_start = random_initial(n, rng)
        # run a short sub‑gradient (few hundred iterations)
        a_local, score_local = subgradient_limited(a_start,
                                                   iters=400,
                                                   fft_len_conv=fft_len_conv,
                                                   fft_len_g=fft_len_g,
                                                   n=n)
        if score_local < best_score:
            best_score = score_local
            best_a = a_local.copy()
            print(f"[rand‑subgrad] new best {best_score:.6f}")

    # Phase B – long sub‑gradient with momentum
    phase_b_end = start_time + 0.60 * budget_s
    if time.time() < phase_b_end:
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                          phase_b_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # Phase C – L‑BFGS continuation (smoothed max)
    phase_c_end = start_time + 0.80 * budget_s
    if time.time() < phase_c_end:
        lam_schedule = [0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                        0.08, 0.05, 0.03, 0.018, 0.01,
                        0.006, 0.003, 0.001, 0.0005, 0.0002]
        best_a, best_score = lbfgs_continuation(best_a,
                                                lam_schedule,
                                                maxiter_per_lam=800,
                                                deadline=phase_c_end,
                                                fft_len_conv=fft_len_conv,
                                                fft_len_g=fft_len_g,
                                                n=n)
        print(f"[L‑BFGS] score={best_score:.6f}")

    # Phase D – LP based deterministic improvement (few iterations)
    phase_d_end = start_time + 0.95 * budget_s
    while time.time() < phase_d_end:
        a_new = lp_improve_step(best_a,
                                fft_len_conv=fft_len_conv,
                                n=n,
                                tol_factor=1e-6,
                                blend_t=0.30)
        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[LP] new best {best_score:.6f}")
        else:
            # Stop if LP step does not improve after a couple of tries
            # (prevent endless looping when we are already at a local optimum)
            break

    # Phase E – final stochastic refinements
    if time.time() < deadline:
        anneal_deadline = deadline - 0.30
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] score={best_score:.6f}")

        # final greedy hill‑climb, allow many moves
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=20000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
