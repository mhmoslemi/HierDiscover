def gradient_descent_line_search(a_start, fft_len, n, deadline, max_iters=200000):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv and gradient of max entry
        conv = conv_fft(a, fft_len, n)
        idx = int(np.argmax(conv))
        # compute gradient g_i = 2 * a[idx - i] (if in bounds)
        g = np.zeros_like(a)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        # i range [lo, hi]
        # for i in that range: idx - i within [0,n-1]
        # fill:
        g[lo:hi+1] = 2 * a[idx - np.arange(lo, hi+1)[::-1]]
        # compute max possible step
        pos = g > 1e-15
        if not np.any(pos):
            break
        alpha_max = np.min(a[pos] / g[pos])
        # generate candidate alphas
        if alpha_max <= 0:
            break
        found = False
        for factor in (0.9, 0.5, 0.2, 0.1, 0.05):
            alpha = factor * alpha_max
            a_new = a - alpha * g
            # project onto simplex
            a_new = project_onto_simplex(a_new)
            score = fast_score(a_new, fft_len, n)
            if score < best_score - 1e-9:
                best = a_new
                best_score = score
                a = a_new
                found = True
                break
        if not found:
            # reduce step size maybe
            a = project_onto_simplex(a - 0.1 * alpha_max * g)
        it += 1
    return best, best_score

def simulated_annealing(a_start, fft_len, n, deadline, start_temp=0.01, end_temp=1e-5):
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    elapsed = 0
    while time.time() < deadline:
        temp = start_temp * ((end_temp/start_temp) ** (elapsed / total_time))
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.2
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegative
        if a_new[i] < 0: continue
        new_score = fast_score(a_new, fft_len, n)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / temp):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score
        elapsed = time.time() - (deadline - total_time)
    return best, best_score

"""
Propose a step‑function (list of non‑negative floats) with a low
evaluation score.

Algorithm
---------
* Choose a moderate length n (5000‑8000).  All optimisation is performed
  on the probability simplex (∑a = 1) – the scale cancels out in the
  objective.
* Multi‑restart smoothed L‑BFGS (continuation on a log‑sum‑exp smooth
  approximation of the max‑convolution).  Several random initialisers
  (Gaussian, exponential, power‑law, edge‑heavy, Sidon‑set, binary,
  single Gaussian bump) are tried and the best result is kept.
* After the L‑BFGS phase a *targeted* projected sub‑gradient descent
  with an adaptive line‑search is run.  The sub‑gradient of the active
  convolution entry is computed analytically, the step size is chosen
  as a fraction of the largest feasible move (so that non‑negativity is
  preserved) and the iterate is projected back onto the simplex.
* The final stage consists of a short simulated‑annealing / greedy
  hill‑climb polishing until the time budget is exhausted.
* The best sequence found is returned; the harness will call
  ``evaluate_sequence`` on it.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast proxy for the evaluation function when ∑a = 1:
    2·n·max(conv) .
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0,∑x=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a sequence of L‑BFGS optimisations with
    decreasing smoothing parameter λ.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_center = 1.0 / np.sqrt(2.0) / n
        a_edge = np.sqrt(n / 2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if cand + c in sums:
                    conflict = True
                    break
            if conflict or (2 * cand in sums):
                continue
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        a0 = np.zeros(n)
        if chosen:
            a0[chosen] = 1.0 / len(chosen)
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: move a random fraction of mass from i to j.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def max_conv_gradient(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Gradient of the *active* convolution entry conv[idx] w.r.t. a.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    lo = max(0, idx - (n - 1))
    hi = min(n - 1, idx)
    # indices i = lo … hi
    i = np.arange(lo, hi + 1)
    g[i] = 2.0 * a[idx - i]
    return g


def gradient_descent_line_search(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with an adaptive line‑search.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    iteration = 0

    while time.time() < deadline and iteration < max_iters:
        conv = conv_fft(a, fft_len, n)
        idx = int(np.argmax(conv))

        g = max_conv_gradient(a, idx)
        pos = g > 1e-15
        if not np.any(pos):
            break

        # Largest step that keeps all entries non‑negative
        alpha_max = np.min(a[pos] / g[pos])
        if alpha_max <= 0:
            break

        improved = False
        # try a few fractions of the maximal step
        for factor in (0.9, 0.5, 0.2, 0.1, 0.05):
            alpha = factor * alpha_max
            a_new = a - alpha * g
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len, n)
            if new_score < best_score - 1e-9:
                best = a_new
                best_score = new_score
                a = a_new
                improved = True
                break

        if not improved:
            # tiny step to keep moving
            alpha = 0.02 * alpha_max
            a = project_onto_simplex(a - alpha * g)

        iteration += 1

    return best, best_score


def simulated_annealing(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    start_temp: float = 0.02,
    end_temp: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing / stochastic hill‑climb.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    total = deadline - time.time()
    while time.time() < deadline:
        elapsed = time.time() - (deadline - total)
        # exponential cooling
        temp = start_temp * ((end_temp / start_temp) ** (elapsed / total))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * 0.15
        if delta == 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # projection is not needed because we subtract a fraction of a[i]
        new_score = fast_score(a_new, fft_len, n)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / temp):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best = a_new.copy()
                best_score = new_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    # Medium‑large n gives a rich search space while keeping FFT cheap.
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Multi‑restart smoothed L‑BFGS  (~55 % of the budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-7, num=35))
    iteration = 0
    modes = [
        "random", "exp", "power", "edge",
        "sidon", "binary", "bump", "uniform", "best"
    ]
    l_bfgs_deadline = start_time + 0.55 * budget_s

    while time.time() < l_bfgs_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}"
            )

        # short polishing hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # 2️⃣  Targeted projected sub‑gradient descent with line‑search
    #      (~35 % of the budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 5.0, start_time + 0.90 * budget_s)
    print("[subgrad] start targeted sub‑gradient descent")
    best_a, best_score = gradient_descent_line_search(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final aggressive stochastic polishing (~remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] simulated annealing + hill‑climb")
        best_a, best_score = simulated_annealing(
            best_a,
            fft_len_conv,
            n,
            final_deadline,
            start_temp=0.02,
            end_temp=1e-6,
        )
        # one last deterministic hill‑climb
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=8000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
