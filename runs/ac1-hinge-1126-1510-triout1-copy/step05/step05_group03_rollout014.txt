def subgradient_descent(a, n, fft_len_conv, fft_len_g, deadline, max_iter=20000, eps_factor_start=0.1, eps_factor_end=1e-8):

    iter = 0

    eps_factor = eps_factor_start

    while time.time() < deadline and iter < max_iter:

        # compute conv full

        fa = np.fft.fft(a, fft_len_conv)

        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]

        max_c = conv.max()

        eps = max(eps_factor * max_c, 1e-12)

        # active mask

        active = (conv >= max_c - eps).astype(float)

        # gradient via convolution of a_rev and active

        a_rev = a[::-1]

        fa_rev = np.fft.fft(a_rev, fft_len_g)

        f_active = np.fft.fft(active, fft_len_g)

        grad_full = np.real(np.fft.ifft(fa_rev * f_active))

        g = 2.0 * grad_full[(n-1) : (n-1 + n)]

        # step size base = 0.5

        step = 0.5 / (np.sqrt(iter+1) * (np.linalg.norm(g) + 1e-12))

        # line search

        improved = False

        cand = proj_simplex(a - step * g)

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < fast_score(a, fft_len_conv, n) - 1e-12:

            a = cand

            improved = True

        else:

            # try smaller step

            for factor in [0.5, 0.25, 0.125]:

                cand = proj_simplex(a - step * factor * g)

                cand_score = fast_score(cand, fft_len_conv, n)

                if cand_score < fast_score(a, fft_len_conv, n) - 1e-12:

                    a = cand

                    improved = True

                    break

        if not improved:

            # if not improved, maybe increase tolerance or break

            eps_factor = min(eps_factor * 2, 0.5)

        else:

            # reduce eps_factor gradually

            eps_factor = max(eps_factor * 0.9, eps_factor_end)

        iter += 1

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm
---------
* Re‑use the best sequence seen so far (global ``height_sequence_1``) if it
  exists, otherwise start from a uniform distribution.
* Global optimisation – a continuation of L‑BFGS on the log‑soft‑max
  parametrisation with a finer λ‑schedule and many restarts.
* Fine‑tuning – projected gradient descent on the *smooth‑max* of the
  auto‑convolution (λ → 0) using an explicit step‑size schedule.
* Aggressive local moves – short random‑/targeted‑hill‑climbs and a final
  unrestricted hill‑climb.
* All steps respect the simplex (non‑negative, sum = 1).  The official
  ``evaluate_sequence`` is called only once at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` for a probability vector.
    (sum(a) = 1 ⇒ score = 2·n·max(conv)).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max = lam·log sum exp((conv−max)/lam) + max
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv (soft‑max weights)
    w = exp_shift / Z

    # back‑prop to a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)   # chain rule through soft‑max
    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple hill‑climbs (keep only strictly improving moves)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that attacks the most offending convolution shift.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift (used to reject bad targets)
        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (used in the final polishing phase)
# ----------------------------------------------------------------------


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth‑max of the auto‑convolution w.r.t. a
    (without the outer factor 2·n).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    return 2.0 * g   # derivative of smooth_max w.r.t a (outer 2·n omitted later)


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent (final polishing)
# ----------------------------------------------------------------------


def smooth_descent(
    a: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    lam_factors: list[float],
    steps_per_lam: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth‑max objective.
    The step size is scaled by the current maximal convolution value.
    """
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam_factor in lam_factors:
        for it in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score

            # gradient of the smooth‑max (excluding the outer 2·n)
            g = grad_smooth_max(best_a, lam_factor) * (2.0 * n)

            grad_norm = np.linalg.norm(g)
            if grad_norm == 0.0:
                continue

            max_c = best_score / (2.0 * n)          # current maximal convolution
            step = 0.5 * max_c / (grad_norm + 1e-12)   # sensible magnitude
            step /= np.sqrt(it + 1)                  # decay within a λ‑stage

            cand = proj_simplex(best_a - step * g)
            cand_score = fast_score(cand, fft_len_conv, n)

            if cand_score < best_score - 1e-12:
                best_a = cand
                best_score = cand_score
                # tighten λ as we improve
                # (no explicit action needed – the outer loop will lower lam_factor)

    return best_a, best_score

            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors_lbfgs,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast_score={best_score:.6f}")

        # ----- short targeted hill‑climb (helps escape tiny plateaus) -----
        tc_deadline = min(outer_deadline, time.time() + 0.7)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=800,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- short random hill‑climb -----
        rc_deadline = min(outer_deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=3000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Fine‑tuning: smooth‑max projected gradient descent
    # ------------------------------------------------------------------
    pg_deadline = time.time() + 250.0   # reserve ~4 min for this stage
    lam_factors_pg = np.geomspace(0.5, 1e-9, num=60)
    best_a, best_score = smooth_descent(
        a=best_a,
        n=n,
        fft_len_conv=fft_len_conv,
        fft_len_g=fft_len_g,
        deadline=pg_deadline,
        lam_factors=lam_factors_pg,
        steps_per_lam=200,
    )
    print(f"[pg] after smooth descent best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use whatever time is left)
    # ------------------------------------------------------------------
    final_deadline = time.time() + budget_s - 2.0   # keep 2 s margin for final eval
    if time.time() < final_deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=final_deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        print(f"[final hill‑climb] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq     # store for possible reuse
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
