    fm = np.fft.fft(mask, p)
    fa_rev = np.fft.fft(a_rev, p)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * conv_g[n-1:n-1+n]

"""
Improved step‑function optimizer.

The algorithm performs:
1. A broad parametric sweep (uniform, linear, exponential, power‑law,
   edge‑heavy, sinusoid, Dirichlet, and a few custom Gaussian bumps) and
   keeps the best few candidates.
2. A long projected sub‑gradient descent on the true ℓ∞ convolution
   objective (exact max, not a smooth surrogate).  The step‑size decays
   as 1/√k and a back‑tracking line‑search accepts only strictly
   improving moves.
3. Intermittent random hill‑climbs and a final simulated‑annealing
   polishing phase.

All loops stop when the allotted 1000 s budget is exhausted.
The returned list is a feasible non‑negative sequence whose
`evaluate_sequence` value is (hopefully) ≤ 1.503.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# helper utilities
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is already a vertex of the simplex
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) for a already normalised (sum=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def gradient_max_conv(a: np.ndarray, fft_len: int):
    """
    Exact sub‑gradient of max(a*a) with respect to a.
    Returns (g, max_val) where g is the sub‑gradient vector.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * conv_g[n - 1 : n - 1 + n]
    return g, max_val

def subgradient_descent(a0: np.ndarray,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 10**9):
    """
    Projected sub‑gradient descent with back‑tracking line‑search.
    Returns the best vector and its score found before `deadline`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iter:
        g, max_val = gradient_max_conv(a, fft_len)
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(4):
            cand = simplex_projection(a - step * g)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                best_a, best_score = cand, cand_score
                a = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)
        it += 1
    return best_a, best_score

def hill_climb_one_step(a: np.ndarray, fft_len: int, n: int, rng: np.random.Generator):
    """
    Perform a single random mass‑transfer move and accept it iff it strictly
    improves the score.
    """
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return a, False
    delta = a[i] * rng.random() * 0.05
    if delta <= 0.0:
        return a, False
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    cand = simplex_projection(cand)   # keep exact sum=1
    new_score = fast_score(cand, fft_len, n)
    cur_score = fast_score(a, fft_len, n)
    if new_score < cur_score:
        return cand, True
    return a, False

def hill_climb(a: np.ndarray,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """Greedy hill‑climb using many random mass‑transfer steps."""
    rng = np.random.default_rng()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        # try the move
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = simplex_projection(cand)
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best, best_score = cand, new_score
        it += 1
    return best, best_score

def random_anneal(a: np.ndarray,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing / random perturbation phase.
    Accepts worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T = 0.02
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = simplex_projection(cand)
        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T = max(T * 0.92, 1e-5)
    return best_a, best_score

def lp_improve(a: np.ndarray,
               max_val: float,
               n: int):
    """
    Solve the linear program:
        max   Σ b_i
        s.t.  (a * b)_k ≤ max_val for all k,
              b_i ≥ 0.
    Returns a normalised vector g = b / Σ b (or None if LP fails).
    """
    # Build convolution matrix A where A @ b = a * b
    c_vec = np.concatenate([a, np.zeros(n - 1)])
    r_vec = np.zeros(n)
    r_vec[0] = a[0]
    A = toeplitz(c_vec, r_vec)   # shape (2*n-1, n)
    b_ub = np.full(2 * n - 1, max_val)
    # objective: maximize sum(b)  ==> minimize -sum(b)
    res = linprog(c=-np.ones(n),
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',

"""
Improved step‑function optimizer (≈1000 s budget).

The algorithm
1. Performs a cheap parametric sweep (uniform, linear, exponential,
   power‑law, edge‑heavy, sinusoid, two Gaussian bumps, Dirichlet draws)
   and keeps the best few seeds.
2. Executes a long projected sub‑gradient descent on the exact
   ℓ∞‑convolution objective.  The step size decays as 1/√k and a
   back‑tracking line‑search accepts only strictly improving moves.
3. Finishes with a greedy hill‑climb and a short simulated‑annealing
   polishing stage.

All phases respect the probability simplex (non‑negative, sum = 1)
and use FFT‑based convolutions (O(n log n)).  The best sequence found
before the deadline is returned.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities – all are top‑level, no closures.
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) for a already normalised (sum=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def gradient_max_conv(a: np.ndarray, fft_len: int):
    """
    Exact sub‑gradient of max(a*a) w.r.t a.
    Returns (g, max_val) where g is a sub‑gradient vector.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * conv_g[n - 1 : n - 1 + n]
    return g, max_val


def subgradient_descent(a0: np.ndarray,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 10**9):
    """
    Projected sub‑gradient descent with back‑tracking line‑search.
    Returns the best vector (and its score) found before `deadline`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iter:
        g, _ = gradient_max_conv(a, fft_len)
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(4):
            cand = simplex_projection(a - step * g)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                best_a, best_score = cand, cand_score
                a = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)
        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy hill‑climb: repeatedly transfer a small random fraction of mass
    from one coordinate to another, keeping only strictly improving moves.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = simplex_projection(cand)
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best, best_score = cand, new_score
        it += 1
    return best, best_score


def random_anneal(a: np.ndarray,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated‑annealing phase: accept worsening moves with a decreasing
    temperature to escape shallow local minima.
    """
    cur_a = a.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T = 0.02
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = simplex_projection(cand)
        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T = max(T * 0.92, 1e-5)
    return best_a, best_score


def normalise(arr: np.ndarray, n: int) -> np.ndarray:
    """Clip to non‑negative and scale to sum = 1."""
    arr = np.asarray(arr, dtype=float)
    arr[arr < 0] = 0.0
    s = arr.sum()
    if s <= 0:
        return np.full_like(arr, 1.0 / n)
    return arr / s

        add_candidate(np.array(globals()['height_sequence_1'], dtype=float), n, fft_len, candidates)

    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # Initialise from the best seed
    # --------------------------------------------------------------
    best_a = top_seeds[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – long projected sub‑gradient descent (≈70 % of budget)
    # --------------------------------------------------------------
    subgrad_deadline = start_time + 0.70 * budget_s
    best_a, best_score = subgradient_descent(best_a,
                                             fft_len,
                                             n,
                                             subgrad_deadline)
    print(f"[sub‑gradient] best score after descent = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – greedy hill‑climb (≈20 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = start_time + 0.90 * budget_s
        best_a, best_score = hill_climb(best_a,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        max_iter=12000)
        print(f"[hill‑climb] score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing for final polishing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_anneal(best_a,
                                           fft_len,
                                           n,
                                           deadline)
        print(f"[anneal] score = {best_score:.6f}")

    # final safety projection onto simplex
    best_a = simplex_projection(best_a)

    final_seq = best_a.tolist()
    # evaluate with the external function (clipping, sum‑check, etc.)
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
