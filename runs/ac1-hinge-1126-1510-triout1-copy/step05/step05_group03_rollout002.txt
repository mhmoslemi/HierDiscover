def local_random_search(a, best_score, deadline):
    rng = np.random.default_rng()
    n = len(a)
    while time.time() < deadline:
        # choose random indices i != j
        i = rng.integers(n)
        j = rng.integers(n)

        # choose delta as random fraction of a[i]
        fraction = rng.uniform(0.001, 0.1)

        a[i] -= delta
        a[j] += delta
        score = fast_score(a, fft_len_conv, n)

            best_score = score
            # keep change
        else:
            # revert
            a[i] += delta
            a[j] -= delta

def adam_optimize(y0, n, fft_len_conv, fft_len_g, lam_start, lam_end, steps, deadline):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_y = y.copy()
    best_score = fast_score(softmax(y), fft_len_conv, n)
    for t in range(1, steps+1):
        if time.time() >= deadline:
            break
        # linear interpolation for lam_factor
        lam = lam_start * ((lam_end / lam_start) ** ((t-1)/(steps-1))) # geometric schedule
        f, g = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=lam)
        # Adam update
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        y -= 0.05 * m_hat/(np.sqrt(v_hat)+epsilon)
        # compute current score via fast_score after projecting? Actually a = softmax(y) automatically normalized.
        a = softmax(y)
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_y = y.copy()
    return best_y, best_score
```

"""
propose_candidate – heuristic optimizer for the “max‑convolution” score.

The algorithm mixes global optimisation (L‑BFGS on a smooth‐max surrogate
with a geometric λ‑schedule) with several stochastic refinement phases:

* many different initial shapes (uniform, random, sparse, β‑distribution,
  power‑law, edge‑biased) are used for multi‑restart L‑BFGS;
* the λ‑schedule is finer (≈ 45 stages) and each stage runs a generous number
  of L‑BFGS iterations;
* after each L‑BFGS run we apply a fast targeted hill‑climb that removes the
  largest contribution to the current peak of the convolution, followed by a
  generic mass‑transfer hill‑climb;
* the best solution found is further polished by an Adam optimiser (gradient
  descent on the smooth‑max) with a decreasing λ, then a final long random
  hill‑climb.

All operations work on the probability simplex (∑a = 1) and use the fast
surrogate `fast_score = 2·n·max(a∗a)`.  The official ``evaluate_sequence``
is called only once at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective (used by L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c  # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z  # d(smooth_max)/d conv_i

    # ----- back‑propagation to a (gradient length n) -----
    # Compute (a ⊛ w) where ⊛ denotes cross‑correlation.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient w.r.t. y (through soft‑max)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)
    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb (strictly improving moves only).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (FFT)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift (used to pick a good target)
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]

        # pick a target whose partner under the critical shift is tiny
        cand_js = rng.integers(0, n, size=7)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def init_sparse(rng: np.random.Generator, n: int, density: float = 0.02) -> np.ndarray:
    """Log‑soft‑max vector for a sparse uniform distribution."""
    m = max(2, int(n * density))
    indices = rng.choice(n, size=m, replace=False)
    a = np.zeros(n)
    a[indices] = 1.0 / m
    return np.log(a + 1e-12)


def init_beta(rng: np.random.Generator, n: int) -> np.ndarray:
    """Log‑soft‑max vector where a follows a symmetric Beta(α,α) pdf."""
    # α drawn log‑uniformly in [0.2, 5]
    log_alpha = rng.uniform(np.log(0.2), np.log(5.0))
    alpha = np.exp(log_alpha)
    x = np.linspace(0.0, 1.0, n)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return np.log(a + 1e-12)


def init_powerlaw(rng: np.random.Generator, n: int) -> np.ndarray:
    """Log‑soft‑max for a power‑law shape a_i ∝ (i+1)^{-p}."""
    p = rng.uniform(0.3, 2.0)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return np.log(a + 1e-12)


def init_edgebiased(rng: np.random.Generator, n: int) -> np.ndarray:
    """A simple shape that gives a little extra weight to the two edges."""
    m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
    edge_val = np.sqrt(n / 2.0) / n
    center_val = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, center_val)
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a /= a.sum()
    return np.log(a + 1e-12)


# ----------------------------------------------------------------------
# Adam optimiser (projected via soft‑max)
# ----------------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_start: float,
    lam_end: float,
    max_steps: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam style optimisation on the smooth‑max surrogate.
    The λ‑factor follows a geometric schedule from lam_start to lam_end.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.05  # base learning rate for the log‑probabilities
    best_y = y.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for t in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        # geometric λ‑schedule
        lam = lam_start * ((lam_end / lam_start) ** ((t - 1) / (max_steps - 1)))

        f, g = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=lam)

        # Adam update on the unconstrained parameters y
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y -= alpha * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_y = y.copy()
    return best_y, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step‑function sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3.0  # safety margin for final evaluation

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to probability simplex
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}   fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS global search (multi‑restart)
    # ------------------------------------------------------------------
    # a fairly fine λ‑schedule
    lam_factors = np.geomspace(0.5, 1e-6, num=45)
    maxiter_per_stage = 700  # enough work per λ‑stage without being too slow
    iteration = 0

    while time.time() < deadline - 15.0:
        # Choose an initialisation strategy
        mode = rng.choice(
            ["uniform", "random", "sparse", "beta", "powerlaw", "edgebiased", "perturb"]
        )
        if mode == "uniform":
            y0 = np.log(np.full(n, 1.0 / n) + 1e-12)
        elif mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "sparse":
            y0 = init_sparse(rng, n, density=rng.uniform(0.015, 0.04))
        elif mode == "beta":
            y0 = init_beta(rng, n)
        elif mode == "powerlaw":
            y0 = init_powerlaw(rng, n)
        elif mode == "edgebiased":
            y0 = init_edgebiased(rng, n)
        else:  # perturb the current incumbent
            y0 = best_y + rng.normal(scale=0.09, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode}  improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb (brief) -----
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- generic random hill‑climb (brief) -----
        rc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=5000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Fine L‑BFGS refinement with a very sharp λ‑schedule
    # ------------------------------------------------------------------
    fine_lambda = np.geomspace(1e-6, 1e-11, num=15)
    best_y = schedule_optimize(
        y_start=best_y,
        n=n,
        fft_len_conv=fft_len_conv,
        fft_len_g=fft_len_g,
        lam_factors=fine_lambda,
        maxiter_per_stage=500,
    )
    best_a = softmax(best_y)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[fine] after sharp λ‑schedule   fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam‑style projected gradient descent on the smooth‑max (λ → 0)
    # ------------------------------------------------------------------
    lam_start = 0.4    # relatively large smoothing at the beginning
    lam_end = 2e-7     # very sharp at the end
    max_adam_steps = 3000
    best_y, best_score = adam_optimize(
        y0=best_y,
        n=n,
        fft_len_conv=fft_len_conv,
        fft_len_g=fft_len_g,
        lam_start=lam_start,
        lam_end=lam_end,
        max_steps=max_adam_steps,
        deadline=deadline - 5.0,
    )
    best_a = softmax(best_y)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[adam] after Adam optimisation   fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final long random hill‑climb (as much time as remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq  # keep for possible reuse
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
