```
def gradient_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=5000, move_frac=0.05):
    best = a.copy()
    best_score = cur_score
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_max, g = max_conv_and_subgrad(best, fft_len, n)
        src = np.argmax(g)
        tgt = np.argmin(g)
        if src == tgt:
            # fallback to random
            i, j = rng.integers(0, n, size=2)
            src, tgt = i, j
        delta = best[src] * move_frac * rng.random()
        if delta <= 0:
            continue
        best[src] -= delta
        best[tgt] += delta
        # renorm not needed because we moved mass, sum unchanged
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[src] += delta
            best[tgt] -= delta
    return best, best_score
```

# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(p*p)   (p = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    p = softmax(y)

    # convolution p * p
    fa = np.fft.rfft(p, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c        # max → 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt conv
    w = exp_shift / Z             # ∂smooth_max/∂conv_i

    # propagate through convolution (grad wrt p)
    p_rev = p[::-1]
    fa_rev = np.fft.rfft(p_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(p, g_vec)
    grad_y = p * (g_vec - mean_g)          # (diag(p) - p p^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Sub‑gradient utilities for the true max‑convolution objective
# ----------------------------------------------------------------------


def max_conv_and_subgrad(a: np.ndarray, fft_len: int, n: int):
    """Return (max_convolution, sub‑gradient vector) for a."""
    conv = conv_fft(a, fft_len, n)
    idx = int(np.argmax(conv))
    # sub‑gradient of conv_idx w.r.t. a:
    # ∂/∂a[i] = 2 * a[idx - i]   (if 0 ≤ idx-i < n), else 0
    g = np.zeros_like(a)
    i = np.arange(n)
    j = idx - i
    mask = (j >= 0) & (j < n)
    g[mask] = 2.0 * a[j[mask]]
    return conv[idx], g, idx


def mirror_subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.12,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Entropic mirror descent on the true max‑convolution objective.
    Uses the sub‑gradient of the active peak, updates with
    a_i ← a_i·exp(-η·g_i) and renormalises.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1
        _, g, _ = max_conv_and_subgrad(a, fft_len, n)
        eta = eta0 / np.sqrt(it)
        # exponential (mirror) update
        a = a * np.exp(-eta * g)
        # renormalise to simplex
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # optional symmetrisation (helps reduce peak)
        a = (a + a[::-1]) * 0.5
        a /= a.sum()

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


def gradient_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    move_frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always moves mass from the index with the
    largest sub‑gradient (most responsible for the current peak) to the
    index with the smallest sub‑gradient.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        _, g, _ = max_conv_and_subgrad(best, fft_len, n)

        src = int(np.argmax(g))
        tgt = int(np.argmin(g))
        if src == tgt:
            # fallback to a random pair to guarantee progress
            i, j = rng.integers(0, n, size=2)
            src, tgt = int(i), int(j)

        delta = best[src] * move_frac * rng.random()
        if delta <= 0.0:
            continue

        best[src] -= delta
        best[tgt] += delta

        # symmetrise after the move (keeps the structure balanced)
        best = (best + best[::-1]) * 0.5
        best /= best.sum()

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move (undo symmetrisation as well)
            best[tgt] -= delta
            best[src] += delta
            # re‑symmetrise to stay feasible
            best = (best + best[::-1]) * 0.5
            best /= best.sum()

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluate_sequence() score.  The function respects the time budget,
    runs a multi‑stage L‑BFGS optimisation, followed by mirror
    sub‑gradient descent and a final gradient‑guided hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.  Prefer an existing construction if available.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 12000)
            best_a = np.maximum(tmp, 0.0)
            best_a /= best_a.sum()
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)

    # Ensure a feasible simplex point
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)  # parameters for soft‑max

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Smoothed L‑BFGS phase (major part of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-7, num=45))
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]
    iteration = 0

    # keep ~300 s for the mirror‑subgradient stage, rest for L‑BFGS
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Short deterministic gradient hill‑climb to polish L‑BFGS result
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = gradient_hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline,
            max_iter=2000, move_frac=0.08,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Mirror sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 60.0
        print("[mirror subgrad] start")
        best_a, best_score = mirror_subgradient_descent(
            best_a, fft_len_conv, n, sub_deadline, eta0=0.12
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[mirror subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive gradient hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = gradient_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=5000,
            move_frac=0.05,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)   # provided by the harness
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)
    return final_seq
