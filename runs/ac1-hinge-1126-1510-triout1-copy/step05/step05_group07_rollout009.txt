"""
Propose a low‑score step‑function sequence.

The search combines:
* L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution,
* Projected sub‑gradient descent on the exact max‑convolution,
* A small linear‑programming “flatten‑convolution” refinement that
  maximises the total mass of a secondary sequence while keeping the
  current max‑convolution bound,
* Targeted and random hill‑climbs for final polishing.

All variables live on the probability simplex (sum = 1, non‑negative),
so the evaluation score 2·n·max(a∗a) equals the true objective
`evaluate_sequence`.  The algorithm runs several problem sizes
(≈2–4 k) within the 1000‑s budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT (real part, trimmed to length 2n‑1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation of the objective for a probability vector a:
    2·n·max(a*a).  (Sum(a)=1, so denominator is 1.)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the simplex
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max used by L‑BFGS (log‑sum‑exp smoothing of the max)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a (linear, via FFT)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max
    w = exp_shift / Z  # d(smooth_max)/d conv

    # back‑prop through convolution: g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).
    Returns the best vector seen and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max at index k
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)

        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # small random perturbations occasionally
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random and targeted hill‑climbs
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the biggest contributor
    to the smallest contributor of the current peak convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (flatten the convolution)
# ----------------------------------------------------------------------


def lp_refine_step(
    a: np.ndarray,
    n: int,
    fft_len: int,
    topk: int,
    mix_t: float = 0.25,
) -> np.ndarray:
    """
    One refinement iteration:
    * compute current max M,
    * take the topk convolution entries as constraints,
    * solve max Σ g_j   s.t.   (a*g)_k ≤ M  (active ks only), g ≥ 0,
    * normalise g to sum 1 and mix it with a.
    """
    conv = conv_fft(a, fft_len, n)
    M = conv.max()
    # pick the most critical ks
    if topk >= conv.shape[0]:
        rows = np.arange(conv.shape[0])
    else:
        rows = np.argpartition(-conv, topk - 1)[:topk]
    rows = np.sort(rows)

    # build sparse constraint matrix A (|rows| × n)
    data = []
    r_ind = []
    c_ind = []
    for ridx, k in enumerate(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_vals = np.arange(lo, hi + 1, dtype=int)
        # coefficient for g_i is a_i
        data.extend(a[i_vals].tolist())
        r_ind.extend([ridx] * len(i_vals))
        c_ind.extend(i_vals.tolist())
    A = sp.csr_matrix((data, (r_ind, c_ind)), shape=(len(rows), n))
    b = np.full(len(rows), M)

    # linear program: maximise sum(g)  <=> minimise -sum(g)
    res = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=b,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return a
    g = res.x
    if g.sum() == 0.0:
        return a
    g = g / g.sum()
    new_a = (1.0 - mix_t) * a + mix_t * g
    new_a = project_simplex(new_a)
    return new_a


# ----------------------------------------------------------------------
# Main optimisation driver for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length n within a time slice.
    Returns the best vector (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ------------------------------------------------------------------
    # Initialise a from previous best if possible
    # ------------------------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].astype(float).copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    fft_len_g = next_pow2(3 * n - 2)

    # more aggressive smoothing schedule
    lam_factors = np.geomspace(0.5, 5e-8, num=45)
    maxiter_per_stage = 600

    iteration = 0

    # ------------------------------------------------------------------
    # Main restart / improvement loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # --------------------------------------------------------------
        # 1) L‑BFGS on a smooth approximation of the max
        # --------------------------------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Exact sub‑gradient descent on the true max‑convolution
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = subgrad_descent(
            best_a,
            n,
            fft_len,
            max_iters=3000,
            deadline=sub_deadline,
            eta0=0.36,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 3) Linear‑programming refinement (only for moderate n)
        # --------------------------------------------------------------
        if n <= 2500:
            lp_deadline = min(deadline, time.time() + 0.8)
            lp_a = best_a.copy()
            # number of top convolution entries to keep as constraints
            topk = max(20, int(0.02 * n))
            for _ in range(3):
                if time.time() > lp_deadline:
                    break
                lp_a = lp_refine_step(lp_a, n, fft_len, topk=topk, mix_t=0.25)
                cur_score = fast_score(lp_a, fft_len, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = lp_a.copy()
                    best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 4) Targeted hill‑climb (quick greedy refinement)
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=3000,
            frac=0.08,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 5) Random hill‑climb for polishing
        # --------------------------------------------------------------
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Deep final L‑BFGS passes with a very small smoothing parameter
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.5:
        for lam in lam_factors:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 800, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its sum will be 1 after
    internal normalisation).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin

    # try a few lengths (all modest enough for the LP step)
    sizes = [2000, 3000, 4000]
    per_size_budget = (total_deadline - time.time()) / len(sizes)

    # warm‑start with any previously discovered construction
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)

    best_overall_a = None
    best_overall_score = np.inf

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 2.0:
            break
        slice_budget = min(per_size_budget, remaining - (len(sizes) - idx - 1) * 2.0)

        print(f"[size] Optimising length n={n} with slice budget ≈{slice_budget:.1f}s")
        a_n, score_n = optimise_one_size(
            n=n,
            budget_s=slice_budget,
            seed=seed + idx * 123,
            prev_seq=prev_arr,
        )
        print(f"[size] n={n} best fast score = {score_n:.6f}")

        if score_n < best_overall_score:
            best_overall_score = score_n
            best_overall_a = a_n.copy()

    # ------------------------------------------------------------------
    # Final evaluation using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence to the global namespace for possible reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
