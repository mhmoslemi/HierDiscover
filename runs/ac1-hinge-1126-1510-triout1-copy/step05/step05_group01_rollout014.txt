def distribution_from_params(mode, params, n):
    if mode=="arcsine":
        alpha = params[0]
        a = 1.0/np.sqrt(np.arange(n)+alpha) + 1.0/np.sqrt(np.arange(n)[::-1]+alpha)
    elif mode=="power":
        p,alpha = params
        i = np.arange(n,dtype=float)
        a = (i+alpha)**(-p) + ( (n-1-i)+alpha )**(-p)
    elif mode=="exp":
        lam = params[0]
        a = np.exp(-lam*np.arange(n)) + np.exp(-lam*np.arange(n)[::-1])
    elif mode=="mix":
        # mixture of uniform and arcsine
        w = params[0]; alpha = params[1]
        arcs = 1.0/np.sqrt(np.arange(n)+alpha) + 1.0/np.sqrt(np.arange(n)[::-1]+alpha)
        arcs /= arcs.sum()
        uni = np.full(n, 1./n)
        a = w * arcs + (1-w) * uni
    # else random

"""
Improved search for a low‑score step‑function.
The algorithm combines several ideas:

1. **Shape‑parameter sweep** – random families (exponential, power‑law,
   “arcsine”, mixtures) are sampled and the best is kept as a warm‑start.
2. **Continuation L‑BFGS** on a smooth log‑sum‑exp proxy for the max‑convolution.
   The smoothing factor is decreased geometrically (≈50 stages).
3. **Projected sub‑gradient descent** on the true max‑convolution objective.
4. **Aggressive hill‑climbing** – moves mass between two coordinates when it
   strictly improves the fast score.

The code respects the 1000 s budget, always maintains a probability simplex
(sum = 1) and returns a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Utility functions
# -------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns probabilities that sum to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using real FFT.
    a must be length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv.real

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# -------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# -------------------------------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is ∂f/∂y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)    # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient wrt conv ------------------------------------------------
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # ---- propagate through convolution (grad wrt a) -----------------------
    # g_vec[j] = Σ_i a_i * w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # the slice (n‑1) … (n‑1+n‑1) corresponds to j = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max --------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)   # (diag(a) - a aᵀ)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing factor.  Returns the final y (pre‑soft‑max) vector.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# -------------------------------------------------------------
# Initial distribution generators (parameterised families)
# -------------------------------------------------------------

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑soft‑max) for a given heuristic mode.
    Modes:
        - "random": standard normal vector.
        - "exp": one‑sided exponential decay.
        - "exp_sym": symmetric exponential heavy at both ends.
        - "power": power‑law decay from the left.
        - "power_sym": symmetric power‑law.
        - "arcsine": a_i ∝ 1/√i + 1/√(n‑1‑i).
        - "mix": mixture of uniform and arcsine (parameterised by w).
        - "sidon": greedy Sidon‑set (sparse) distribution.
        - "best": perturbation of the current best.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power_sym":
        p = rng.uniform(0.3, 2.0)
        i = np.arange(n, dtype=float)
        a0 = (i + 1) ** (-p) + (n - i) ** (-p)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "mix":
        w = rng.random()
        alpha = rng.uniform(0.1, 5.0)
        i = np.arange(n, dtype=float)
        arcs = 1.0 / np.sqrt(i + alpha) + 1.0 / np.sqrt((n - 1 - i) + alpha)
        arcs /= arcs.sum()
        uniform = np.full(n, 1.0 / n)
        a0 = w * arcs + (1 - w) * uniform
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = any(((cand + c) in sums) or ((2 * cand) in sums) for c in chosen)
            if conflict:
                continue
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.4):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "best":
        # placeholder – caller will overwrite with perturbed best_y
        raise ValueError("mode 'best' must be handled by the caller")

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# -------------------------------------------------------------
# Local optimisation utilities
# -------------------------------------------------------------

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb.  Moves a random fraction (≤10 %) of mass
    from a source index to a target index and keeps the move if the fast
    score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.22,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective:
        f(a) = max_i (a*a)_i .
    The step size is ηₜ = η₀ / √t and we Euclidean‑project onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score
