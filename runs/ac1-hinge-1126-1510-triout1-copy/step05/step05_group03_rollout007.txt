def simulated_annealing(a0, max_time, n, fft_len):
    best_a = a0 copy.
    best_score = fast_score(best_a, fft_len, n)
    cur_a = best_a copy
    cur_score = best_score
    rng = np.random.default_rng()
    start = time.time()
    # Initial temperature based on relative improvement range.
    # We'll schedule t decreasing exponentially.
    # Determine initial temperature: may be 0.01 * cur_score? Use heuristic.

    while time.time() < max_time:
        # proportion of elapsed
        fraction = (time.time() - start) / (max_time - start)
        T = max(0.001, (1 - fraction) )
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # propose delta proportional to cur_a[i] * random *
        max_move = cur_a[i] * 0.2
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        # apply move
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure non-negative
        if new_a[i] < 0: continue
        # compute score
        new_score = fast_score(new_a, fft_len, n)
        # accept if better or probabilistically
        if new_score < cur_score:
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_a = new_a
                best_score = new_score
        else:
            # acceptance probability
            # delta = new_score - cur_score (positive)
            # p = exp(-delta / (T * best_score))
            # Use small scaling.
            prob = np.exp(- (new_score - cur_score) / (T * cur_score * 0.1 + 1e-12))
            if rng.random() < prob:
                cur_a = new_a
                cur_score = new_score

    return best_a, best_score

def subgradient_descent(a, cur_score, n, fft_len, deadline, max_iter=20000):

    best_a = a.copy()
    best_score = cur_score
    t = 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute convolution
        fa = np.fft.fft(best_a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        # find max indices
        max_val = conv.max()
        # if max_val close to best_score/(2n)? Actually best_score = 2n * max_val.
        # compute subgradient for random max index
        max_k = np.flatnonzero(conv == max_val)

        # gradient w.r.t a: g_i = 2 * a[k-i] if 0 <= k-i < n else 0.
        # can compute by shifting a
        g = np.zeros_like(best_a)
        # compute indexes where i in [0,n-1] and idx = k-i in [0,n-1]
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)  # inclusive
        # For i in [i_start,i_end], partner = k-i

"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm (≈1000 s budget)
--------------------------------
1.  Initialise a probability vector `a` (∑a = 1).  Use the previous result
    if available, otherwise start from a uniform or a few hand‑crafted shapes.

2.  Global optimisation:
    * Multi‑restart L‑BFGS on the log‑soft‑max parametrisation of `a`.
    * A geometric continuation schedule for the smooth‑max parameter λ
      (from 0.5·max down to 1e‑9·max) is used to approach the true max
      ``max(a∗a)``.
    * Each run uses a few hundred L‑BFGS iterations per λ‑stage.

3.  Local refinement – three complementary hill‑climbs:
    * **Targeted hill‑climb** – as in the previous state‑of‑the‑art.
    * **Random hill‑climb** – classic random mass‑transfer move.
    * **Projected sub‑gradient descent** – uses the exact sub‑gradient of
      ``max(a∗a)`` (the gradient of the active convolution shift) and
      projects back onto the probability simplex.

4.  Simulated annealing (accepting occasional worsening moves) for a
    final escape from local plateaus.

5.  One last aggressive random hill‑climb, then evaluate with the official
    ``evaluate_sequence`` and return the best sequence found.

All steps keep the vector on the probability simplex (non‑negative,
∑a = 1).  The fast surrogate ``2·n·max(conv)`` is used throughout the
search; the official evaluator is called only once at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of `evaluate_sequence` for a probability vector.
    Returns 2·n·max(a∗a).  No scaling by the sum is required because we
    keep the vector normalised (∑a = 1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a (full) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max ----
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max w.r.t. conv ----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ---- chain rule back to a (gradient vector of length n) ----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    In each step the shift giving the maximal value is identified, the most
    contributing index pair is found, and a fraction of mass is moved away
    from the worst contributor to a location whose partner under the same
    shift is (almost) empty.  Only improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        # choose the worst contributor (largest product)
        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift (used to reject bad targets)
        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        # sample a few candidate target indices and pick the one whose partner
        # under the critical shift is smallest
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        # transfer a fraction of mass from i → j
        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep move
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 30000,
    init_step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max of the auto‑convolution.
    The gradient is taken at a randomly selected maximising shift.
    """
    best = a.copy()
    best_score = cur_score
    t = 1
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute full convolution once per iteration
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        # pick a random maximising shift (helps when there are ties)
        max_idxs = np.flatnonzero(conv == max_val)
        k = rng.choice(max_idxs)

        # build the sub‑gradient for this shift:
        # g_i = 2 * a_{k-i} for all i where the partner is in range
        g = np.zeros_like(best)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low <= high:
            i_range = np.arange(low, high + 1)
            partner = k - i_range
            g[i_range] = 2.0 * best[partner]

        # step size decreasing like 1/√t
        step = init_step / np.sqrt(t)
        t += 1

        cand = proj_simplex(best - step * g)
        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # keep the improvement
    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 200000,
    start_temp: float = 0.08,
    end_temp: float = 0.001,
    max_move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Classic Metropolis‑Hastings simulated annealing on the simplex.
    Moves that increase the surrogate objective are occasionally accepted
    according to a linear temperature schedule.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return best, best_score

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # linear temperature decay
        elapsed = time.time() - start_time
        frac = elapsed / total_time
        T = start_temp * (1 - frac) + end_temp * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of the mass at i to j
        delta = cur[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-dE / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta

    return best, best_score

    lam_factors = np.geomspace(0.5, 1e-9, num=45)   # finer λ‑schedule
    maxiter_per_stage = 800                         # robust per λ‑stage

    # allocate roughly 60 % of the time budget to this phase
    phase1_deadline = time.time() + (deadline - time.time()) * 0.60
    iteration = 0
    while time.time() < phase1_deadline - 5.0:
        # pick an initialisation strategy
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            center_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, center_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best solution
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb -----
        tc_deadline = min(phase1_deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1500,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- fallback random hill‑climb -----
        rc_deadline = min(phase1_deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=5000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 8.0:
        sub_deadline = time.time() + remaining * 0.30
        best_a, best_score = subgradient_descent(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=sub_deadline,
            max_iter=40000,
            init_step=0.6,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for final escape from local minima
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sa_deadline = deadline
        best_a, best_score = simulated_annealing(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=sa_deadline,
            max_iter=200000,
            start_temp=0.08,
            end_temp=0.001,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[SA] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final aggressive random hill‑climb (use any leftover time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq  # store for possible reuse
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
