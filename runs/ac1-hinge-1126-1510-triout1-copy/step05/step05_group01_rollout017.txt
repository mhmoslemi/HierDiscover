def simulated_annealing_hc(a, cur_score, fft_len, n, deadline, init_temp=0.001, decay=0.999, max_iter=20000):
   best = a.copy()
   best_score = cur_score
   temp = init_temp
   rng = np.random.default_rng()
   while time.time() < deadline and max_iter>0:
       max_iter -= 1
       i, j = rng.integers(0,n, size=2)
       if i == j: continue
       delta = best[i] * rng.random() * 0.1
       if delta <= 0: continue
       new_a = best.copy()
       new_a[i] -= delta
       new_a[j] += delta
       # ensure nonnegative
       if new_a[i] < 0: continue
       new_score = fast_score(new_a, fft_len, n)
       diff = new_score - best_score
       if diff < 0 or rng.random() < np.exp(-diff / temp):
           best = new_a
           best_score = new_score
       temp *= decay
   return best, best_score

"""
Hybrid optimization for the self‑convolution max‑norm problem.

The objective (up to an irrelevant scaling) is to minimise
    2·n·max_i (a*a)_i   with a_i≥0, Σ a_i = 1 .
The algorithm proceeds in four stages:

1. **Parametric grid search** – cheap scans of a few one‑parameter families
   (exponential, symmetric exponential, power law, arcsine, Catalan)
   to obtain a good “seed” distribution.

2. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The smoothing parameter λ is decreased geometrically over
   many stages, each stage performing a modest number of L‑BFGS iterations.

3. **Projected sub‑gradient descent** on the true non‑smooth objective,
   using a 1/√t step‑size and Euclidean projection onto the probability simplex.

4. **Simulated‑annealing hill‑climb** – a final greedy phase that occasionally
   accepts worsening moves, allowing the search to escape shallow local minima.

The whole search respects a 1000 s wall‑clock budget and returns the best sequence
found.  All numbers are non‑negative; scaling is irrelevant because the evaluation
function is scale‑invariant.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import gammaln

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector, sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    a is assumed to have length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (Σa=1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of the max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution (a → conv) -----------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n−1)…(n−1+n−1) correspond to i = 0…n−1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the previous.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Parametric families for cheap initialisation
# ----------------------------------------------------------------------


def catalan_distribution(n: int) -> np.ndarray:
    """a_i ∝ (2i choose i) / 4^i  (the discrete arcsine / Catalan law)."""
    i = np.arange(n, dtype=float)
    # log of the coefficient to avoid overflow
    log_c = gammaln(2 * i + 1) - 2 * gammaln(i + 1) - i * np.log(4.0)
    a = np.exp(log_c)
    a[0] = 1.0  # ensure exact 1 for i=0 (log_c may be -inf for i=0 due to gammaln)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑soft‑max) for a given heuristic mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "catalan":
        a0 = catalan_distribution(n)
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

# Local improvement utilities

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only strict improvements.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a) = 2·n·max_i (a*a)_i   (Σa = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing_hc(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.001,
    decay: float = 0.9995,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb.  A random mass transfer is accepted
    with probability exp(−Δ/temperature).  The temperature decays geometrically.
    """
    best = a.copy()
    best_score = cur_score
    temp = init_temp
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue

        new_a = best.copy()
        new_a[i] -= delta
        new_a[j] += delta
        if new_a[i] < 0.0:  # safeguard – should not happen
            continue

        new_score = fast_score(new_a, fft_len, n)
        diff = new_score - best_score
        if diff < 0.0 or rng.random() < np.exp(-diff / max(temp, 1e-12)):
            best = new_a
            best_score = new_score

        temp *= decay

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluate_sequence() score.  The routine respects the given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for harness

    # --------------------------------------------------------------
    # Choose a problem size.
    # --------------------------------------------------------------
    n = 8000
    # reuse a previous construction if present and not too small
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Light parametric grid search (cheap)
    # --------------------------------------------------------------
    param_modes = ["exp", "exp_sym", "power", "arcsine", "catalan"]
    for mode in param_modes:
        # a few random parameter choices per mode
        for _ in range(12):
            if mode in ("exp", "exp_sym"):
                rate = rng.uniform(0.001, 0.02)
                if mode == "exp":
                    a0 = np.exp(-rate * np.arange(n))
                else:
                    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
                    a0 = np.exp(-rate * d)
            elif mode == "power":
                p = rng.uniform(0.3, 2.0)
                a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            elif mode == "arcsine":
                i = np.arange(n, dtype=float)
                a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
            elif mode == "catalan":
                a0 = catalan_distribution(n)
            else:
                continue

            a0 = a0 / a0.sum()
            sc = fast_score(a0, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a0.copy()
                best_y = np.log(a0 + 1e-12)
                print(f"[grid] mode={mode} → improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Continuation L‑BFGS (major part of the budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=50))
    iteration = 0
    modes = ["random", "exp", "exp_sym", "power", "arcsine", "catalan", "best"]
    # reserve about 240 s for later refinement phases
    while time.time() < deadline - 240.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # a brief hill‑climb after each L‑BFGS run
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline - 120.0:
        sub_deadline = deadline - 60.0
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.25,
            max_iters=600_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final SA] start")
        best_a, best_score = simulated_annealing_hc(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_temp=0.001,
            decay=0.9993,
            max_iter=30_000,
        )
        print(f"[final SA] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
