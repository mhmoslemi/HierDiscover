def gradient_descent_line_search(a_start, fft_len, n, end_time):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    # initial step size
    eta = 0.5
    while time.time() < end_time:
        it += 1
        # compute conv and max index
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # subgradient g (2*a_{m-i}) as array length n
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # line search: try decreasing step sizes
        # ensure step size not too large
        step = eta
        improved = False
        while step > 1e-8:
            a_new = a - step * g
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len, n)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                # accept
                a = a_new
                improved = True
                break
            step *= 0.5
        if not improved:
            # if no improvement, reduce eta
            eta *= 0.5
            if eta < 1e-8:
                # random restart maybe
                a = best_a.copy()
                eta = 0.5

        # optional: increase eta gradually when improvement
        if improved and eta < 1.0:
            eta *= 1.05

    return best_a, best_score

"""
Propose a non‑negative step‑function sequence with a low evaluate_sequence score.

Algorithm Overview
------------------
1. **Problem size** – Choose a moderate length (≈12000‑15000).  
   Larger n gives more freedom to lower the L∞‑norm of the self‑convolution.

2. **Multi‑start smoothed L‑BFGS** –  
   *Start from a variety of heuristic distributions* (exponential,
   symmetric exponential, power‑law, arcsine‑law, edge‑heavy, Sidon‑set,
   random and perturbations of the current best).  
   For each start we minimise a *smooth* surrogate of the max‑convolution:
   `smooth_max = max + λ·log Σexp((conv‑max)/λ)`.  
   The smoothing factor λ is decreased geometrically (continuation method)
   over ~45 stages.  Each stage runs L‑BFGS‑B with at most 300 iterations.

3. **Projected gradient descent with line‑search** –  
   Using the true objective `f(a)=2·n·max(conv(a))` we perform a
   sub‑gradient step and a simple back‑tracking line‑search, always
   projecting back onto the probability simplex.

4. **Aggressive hill‑climbing** –  
   Randomly relocate a fraction of mass from one coordinate to another,
   accepting only strictly improving moves.  Repeated until the time budget
   is exhausted.

5. **Return** – Convert the best probability vector to a plain Python
   list; the harness will scale it internally.  The final score is printed
   for verification.

The code respects the 1000 s budget, prints progress, and avoids
closures and lambda functions as required.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    Implementation based on the algorithm by Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # ---- propagate gradient through the convolution ----------------------
    # Compute g_vec = ∂conv/∂a (size n)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a aᵀ)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def gradient_descent_line_search(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    end_time: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a) = 2·n·max(conv(a)).
    Uses a simple back‑tracking line search and Euclidean projection onto
    the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    eta = 0.5  # initial step size
    it = 0

    while time.time() < end_time:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of max entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # line search
        step = eta
        improved = False
        while step > 1e-8:
            a_new = a - step * g
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len, n)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            step *= 0.5

        if not improved:
            # no progress – shrink step size
            eta *= 0.5
            if eta < 1e-8:
                # restart from the current best
                a = best_a.copy()
                eta = 0.5
        else:
            # successful step – allow a modest increase
            eta = min(1.0, eta * 1.05)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size (n) and FFT lengths
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        base_len = len(init_seq)
        n = min(max(8000, base_len), 15000)
    else:
        n = 12000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate on the probability simplex
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / tmp.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Multi‑start smoothed L‑BFGS phase
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.25, 1e-6, num=45))
    modes = [
        "random", "exp", "exp_sym", "power", "arcsine", "edge",
        "sidon", "mix", "best"
    ]

    # Reserve time for later refinement phases
    while time.time() < deadline - 200.0:
        mode = rng.choice(modes)

        # ----- generate starting point y0 (pre‑soft‑max) -----------------
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        elif mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.0005, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "exp_sym":
            rate = rng.uniform(0.0005, 0.02)
            d = np.minimum(np.arange(n), n - 1 - np.arange(n))
            a0 = np.exp(-rate * d)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 3.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "arcsine":
            p = rng.uniform(0.1, 2.0)
            i = np.arange(n, dtype=float)
            a0 = (i + 1) ** (-p) + (n - i) ** (-p)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.008, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "mix":
            # mixture of arcsine and edge‑heavy components
            p = rng.uniform(0.2, 2.0)
            arcs = (np.arange(1, n + 1) ** (-p) + np.arange(n, 0, -1) ** (-p))
            arcs = arcs / arcs.sum()
            m_edge = max(1, int(n * rng.uniform(0.008, 0.03)))
            extra = np.zeros(n)
            extra[:m_edge] = 1.0
            extra[-m_edge:] = 1.0
            extra = extra / extra.sum()
            a0 = 0.7 * arcs + 0.3 * extra
            a0 = np.maximum(a0, 0)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "sidon":
            # greedy Sidon‑set construction
            chosen = []
            sums = set()
            for cand in range(n):
                conflict = False
                for c in chosen:
                    if (cand + c) in sums or (2 * cand) in sums:
                        conflict = True
                        break
                if not conflict:
                    chosen.append(cand)
                    for c in chosen:
                        sums.add(cand + c)
                if len(chosen) >= int(np.sqrt(n) * 1.4):
                    break
            k = len(chosen)
            a0 = np.zeros(n)
            a0[chosen] = 1.0 / k
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:
            y0 = rng.standard_normal(n)

        # ----- L‑BFGS optimisation ---------------------------------------
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage=300
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{time.time():.1f}] mode={mode} → new best fast score = {best_score:.6f}")

        # ----- short hill‑climb for polishing ----------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Refine with projected gradient descent + line search
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        print("[refine] projected gradient descent with line search")
        refine_deadline = deadline - 60.0
        best_a, best_score = gradient_descent_line_search(
            best_a, fft_len_conv, n, refine_deadline
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[refine] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # evaluate_sequence is supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
