"""
Propose a low‑evaluation step sequence.

Algorithm
---------
1. Choose a problem size `n` (default 4000).  Larger `n` gives a finer
   discretisation but does not change the evaluation cost dramatically.
2. Initialise the candidate with a variety of parametric families
   (uniform, exponential, power‑law, Beta, Gaussian, ramps, edge‑spikes).
   Each candidate is normalised to sum = 1 and scored with a fast FFT‑based
   estimator of `2·n·max(a*a)`.  The best shape from this cheap random
   search becomes the starting point for the subsequent optimisation.
3. **Continuation L‑BFGS** on a log‑sum‑exp (smooth‑max) surrogate.
   A decreasing schedule of smoothing factors `lam` drives the optimiser
   from a coarse approximation to the true max‑convolution.
   Restarts with Gaussian noise and with fresh random shapes keep the
   search escaping shallow basins.
4. **Projected sub‑gradient descent with momentum** on the exact max‑convolution.
   The sub‑gradient is obtained from the active (max‑valued) indices.
5. **Adam** on the smooth‑max objective for fine‑grained polishing.
6. **Simulated annealing** followed by an aggressive greedy hill‑climb
   (mass moves between two coordinates only when the score strictly
   improves).  These stochastic phases explore the neighbourhood of the
   best solution found so far.
7. Return the final sequence as a Python list; the outer driver will
   evaluate it with `evaluate_sequence`.

All steps respect the non‑negative, sum‑invariant nature of the problem
and use only NumPy / SciPy.  The time budget (default 1000 s) is split
automatically among the phases.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (FFT helpers, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z    # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in F domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Very rare failure – just skip this lam.
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best score={best_score:.6f} (lam={lam:.5f})")
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with a simple momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing that moves a random fraction of mass between two
    coordinates.  Accepts worsening moves with a temperature that decays.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10            # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a random fraction (up to 12 %) of the mass at i
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another.  Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative distribution on the simplex using a
    diverse collection of parametric families.
    """
    mode = rng.choice(['uniform', 'exp', 'power', 'beta',
                       'gaussian', 'ramp', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        lam = rng.uniform(0.0005, 0.02)
        a = np.exp(-lam * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.0, 4.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'beta':
        alpha = rng.uniform(0.0, 4.0)
        beta = rng.uniform(0.0, 4.0)
        x = (np.arange(n) + 0.5) / n
        a = (x ** alpha) * ((1 - x) ** beta)
    elif mode == 'gaussian':
        mu = rng.uniform(0.2, 0.8) * (n - 1)
        sigma = rng.uniform(0.05, 0.2) * n
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    elif mode == 'ramp':
        # decreasing linear ramp
        a = np.linspace(1.0, 0.0, n)
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1e-6)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal (already used in the original code)
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000
    # Use an existing global sequence as a possible starter (if present)
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = max_n
            old_n = init_seq.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            s0 = a0.sum()
            a0 = a0 / s0 if s0 > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.80 * budget_s   # sub‑gradient + Adam
    phase3_end = start_time + 0.92 * budget_s   # annealing
    # remaining time for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 0 – cheap shape exploration (random parametric families)
    # ------------------------------------------------------------------
    shape_deadline = start_time + 0.15 * budget_s
    while time.time() < shape_deadline:
        a = random_shape(n, rng)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[shape‑search] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005]
    maxiter_per_lam = 200   # modest per‑lam budget – many restarts

    while time.time() < phase1_end:
        # Occasionally perturb the current best to stay in its basin
        if best_score < np.inf and rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_shape(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑restart] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning on the smoothed max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           phase3_end,
                                           lam_factor=0.002,
                                           lr=0.15)
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.3
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # One final evaluation with the provided external function (slow but only once)
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
