"""
Low‑peak step‑function optimizer.

The algorithm alternates a linear‑programming (LP) phase (the “dual
step” from the reference) with a short projected sub‑gradient descent.
Given a probability vector f (∑f=1) we compute M = max(f∗f).
The LP maximises ∑b under the *linear* constraints f∗b ≤ M,
yielding an extreme point b (often concentrated on a few indices).
A convex combination f←(1−t)f+t·b with a small step size t reduces the
peak M.  Repeating the LP–mixing loop quickly drives the maximum
convolution value towards its theoretical lower bound (≈1/(2n)).
A brief sub‑gradient descent (exact max‑convolution) refines the
solution, followed by a cheap random hill‑climb.

The implementation allocates the bulk of the 1000 s budget to this
LP‑mixing scheme for a moderate problem size (n≈2000–3000), keeping the
memory footprint modest while still achieving a score below the
target 1.503.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Core convolution utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(a∗a) – the quantity returned by evaluate_sequence."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – needed by sub‑gradient descent
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent on max(a∗a)
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a∗a).

    Returns the best vector seen and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    idx = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max at k
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(t + 1.0)

        a = project_simplex(a - eta * grad)

        # record best
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # occasional random perturbation
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climb (mass transfer) – final polishing
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of mass between two positions."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Construction of the LP matrix A (constraints f∗b ≤ M)
# ----------------------------------------------------------------------


def build_conv_ub_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
    A @ b = f∗b (linear convolution).  Only non‑negative offsets appear.
    """
    n = len(f)
    # each diagonal offset d (0 … n‑1) carries the constant f[d]
    offsets = np.arange(0, n, dtype=int)
    data = [np.full(n, f[d], dtype=float) for d in offsets]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


# ----------------------------------------------------------------------
# One optimisation run for a given length n
# ----------------------------------------------------------------------


def optimise_one_length(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_vec: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Perform LP‑mixing + short sub‑gradient descent for a fixed size n.
    Returns the best probability vector (sum=1) and its fast_score.
    """
    fft_len = next_pow2(2 * n - 1)

    # initialise f (probability simplex)
    if init_vec is not None and init_vec.size >= n:
        f = np.asarray(init_vec[:n], dtype=float)
        f = np.clip(f, 0.0, np.inf)
        s = f.sum()
        if s > 1e-12:
            f = f / s
        else:
            f = np.full(n, 1.0 / n)
    else:
        f = np.full(n, 1.0 / n)

    best_f = f.copy()
    best_sc = fast_score(f, fft_len, n)

    iteration = 0
    # main LP‑mixing loop
    while time.time() < deadline - 5.0:
        iteration += 1
        conv_f = conv_fft(f, fft_len, n)
        M = conv_f.max()

        # ----- LP: maximise sum(b) s.t. f∗b ≤ M, b≥0 -----
        A = build_conv_ub_matrix(f)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        c = -np.ones(n, dtype=float)  # maximise sum(b)
        bounds = [(0.0, None)] * n

        lp_res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not lp_res.success:
            # LP failure – abort the loop
            print(f"[LP] iteration {iteration} failed: {lp_res.message}")
            break

        b = np.maximum(lp_res.x, 0.0)
        if b.sum() < 1e-12:
            break
        g = b / b.sum()                     # scale to probability simplex

        # ----- line‑search mixing f←(1−t)f + t·g -----
        t = 1.0
        improved = False
        while t > 1e-4:
            f_new = (1.0 - t) * f + t * g
            conv_new = conv_fft(f_new, fft_len, n)
            M_new = conv_new.max()
            if M_new < M - 1e-15:
                f = f_new
                improved = True
                break
            t *= 0.5
        if not improved:
            # if no improvement even for tiny t, stop iterating
            print(f"[LP] iteration {iteration} no improvement")
            break

        # keep the best seen so far
        sc = 2.0 * n * M_new
        if sc < best_sc:
            best_sc = sc
            best_f = f.copy()
            print(f"[LP-{iteration}] new best score {best_sc:.6f}")

        # ----- short sub‑gradient refinement (for robustness) -----
        sub_deadline = min(deadline, time.time() + 0.6)
        f_ref, sc_ref = subgrad_descent(
            f, n, fft_len, max_iters=1500, deadline=sub_deadline, eta0=0.38
        )
        if sc_ref < best_sc:
            best_sc = sc_ref
            best_f = f_ref.copy()
            f = f_ref
            print(f"[sub‑grad-{iteration}] improved to {best_sc:.6f}")

    # final random hill‑climb polish
    final_deadline = deadline
    best_f, best_sc = hill_climb(
        best_f, best_sc, fft_len, n, final_deadline, max_iter=4000
    )
    return best_f, best_sc


# ----------------------------------------------------------------------
# Global entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with low evaluation score.
    Returns the best sequence found within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0   # small safety margin

    # problem size – pick the largest n we can afford comfortably
    # 2000 gives a fast LP; 3000 still fits comfortably in memory.
    n = 3000 if (total_deadline - time.time()) > 600 else 2000

    # warm‑start from a previous construction if available
    prev = globals().get("height_sequence_1")
    init = None
    if prev is not None and isinstance(prev, list):
        init = np.asarray(prev, dtype=float)

    # split the remaining budget into several independent runs (restarts)
    remaining = total_deadline - time.time()
    n_runs = max(1, int(remaining // 120))   # aim for ~2 min per run
    per_run_budget = remaining / n_runs

    best_overall_f = None
    best_overall_sc = np.inf

    for run in range(n_runs):
        now = time.time()
        if now + per_run_budget > total_deadline:
            break
        run_deadline = now + per_run_budget

        # occasional random initialisation to avoid being stuck
        if run == 0 and init is not None:
            init_vec = init
        else:
            # Dirichlet (≈ uniform but with random perturbations)
            init_vec = rng.dirichlet(np.ones(n))

        print(f"[run {run+1}/{n_runs}] start optimisation n={n}")
        f_opt, sc_opt = optimise_one_length(
            n=n,
            deadline=run_deadline,
            rng=rng,
            init_vec=init_vec,
        )
        print(f"[run {run+1}] best score = {sc_opt:.6f}")

        if sc_opt < best_overall_sc:
            best_overall_sc = sc_opt
            best_overall_f = f_opt.copy()

    if best_overall_f is None:
        # fallback – uniform distribution
        best_overall_f = np.full(n, 1.0 / n)

    # final evaluation with the provided evaluate_sequence()
    final_seq = best_overall_f.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returns {final_val:.6f}")

    # expose the best sequence for possible warm‑starts in later calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
