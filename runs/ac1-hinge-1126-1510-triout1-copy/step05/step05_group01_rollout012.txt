    k_max = np.argmax(conv)
    # picks i from contributors
    lo = max(0, k_max - (n-1))
    hi = min(n-1, k_max)
    # choose i uniformly from lo..hi
    i = rng.integers(lo, hi+1)
    # choose j random any index not i
    j = rng.integers(0, n)
    if j == i: continue
    delta = min(epsilon * a[i], a[i])

    a[i] -= delta
    a[j] += delta
    new_score = fast_score(a, fft_len, n)
    if new_score < cur_score:
        cur_score = new_score
    else:
        # revert
        a[i] += delta
        a[j] -= delta

def targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=20000, max_move=0.05):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))
        lo = max(0, k_max - (n-1))
        hi = min(n-1, k_max)
        if lo >= hi:
            # cannot move; just random
            i = rng.integers(0, n)
        else:
            i = rng.integers(lo, hi+1)
        j = rng.integers(0, n)
        if i == j:
            continue
        # shift some mass
        delta = best[i] * rng.random() * max_move
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Propose a step‑function (list of non‑negative floats) with a low
evaluate_sequence() score.

Algorithm (≈ 1000 s budget)
--------------------------------
1. Choose a problem size n (≈ 5000‑8000).  If a previous construction
   (global *height_sequence_1*) exists we reuse its length.
2. Initialise a probability vector a (∑a=1) with several heuristics:
   uniform, exponential, symmetric exponential, power‑law, edge‑heavy,
   arcsine, Sidon‑set‑style, inverse‑square‑root.
3. For each start run a continuation L‑BFGS on a smooth log‑sum‑exp
   surrogate of the max‑convolution.  The smoothing factor λ is reduced
   geometrically over ~45 stages.
4. After each L‑BFGS run perform a short random hill‑climb (move a
   random fraction of mass between two coordinates) to polish the
   solution.
5. When the time budget leaves ~300 s, run a long projected
   sub‑gradient descent on the true max‑convolution objective (step‑size
   ηₜ=η₀/√t, Euclidean projection onto the simplex).  Up to a few
   hundred‑thousand iterations are performed.
6. Finally a *targeted* hill‑climb is used: repeatedly find the
   convolution index with maximal value and move a small amount of mass
   from a contributing coordinate to a random other coordinate; only
   improving moves are kept.
7. Return the best probability vector (the harness will apply the
   scaling‑invariant evaluation).

The whole pipeline respects the 1000 s time limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ---- propagate gradient through the convolution ----------------------
    # we need grad_a = ∂f/∂a ; then ∂a/∂y = diag(a) - a a^T
    # First compute g_vec = ∂conv/∂a   (size n)
    #   g_vec[i] = Σ_j a_j * w_{i+j}   (valid indices only)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "inv_sqrt":
        # a_i ∝ 1/√(i+1)
        idx = np.arange(1, n + 1, dtype=float)
        a0 = 1.0 / np.sqrt(idx)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: randomly move a fraction of mass from i to j.
    Accept only moves that strictly improve the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
        f(a) = max_i (a*a)_i .
    The sub‑gradient of the active convolution entry m is:
        g_i = 2·a_{m‑i}   for indices i with 0 ≤ m‑i < n, else 0.
    Steps use ηₜ = η₀/√t and Euclidean projection onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
    max_move: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that always attacks the currently worst convolution entry.
    A random fraction (≤ max_move) of mass is moved from a coordinate that
    participates in the maximal sum to a random other coordinate.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # indices that contribute to k_max
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        if lo > hi:
            # fallback to random move
            i = rng.integers(0, n)
        else:
            i = rng.integers(lo, hi + 1)

        j = rng.integers(0, n)
        if i == j:
            continue

        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        # move delta from i to j
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Choose problem size.
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        n = max(4000, min(int(tmp.size), 8000))
    else:
        n = 5000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(0.0, tmp)
            best_a = best_a / best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Smoothed L‑BFGS phase (major part of the budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=45))
    iteration = 0
    modes = [
        "random",
        "exp",
        "exp_sym",
        "power",
        "edge",
        "arcsine",
        "sidon",
        "inv_sqrt",
        "best",
    ]

    # Reserve about 300 s for later phases
    while time.time() < deadline - 300.0:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # short polishing hill‑climb
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Projected sub‑gradient descent on the true max‑conv objective
    # --------------------------------------------------------------
    if time.time() < deadline - 150.0:
        sub_deadline = deadline - 80.0
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.22,
            max_iters=800_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final targeted hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final target HC] start")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[final target HC] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
