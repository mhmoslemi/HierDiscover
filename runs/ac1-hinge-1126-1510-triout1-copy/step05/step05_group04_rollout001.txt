def subgrad_adaptive(a, fft_len, n, deadline):
    a = a.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    step = 0.5 # initial step size maybe 0.2 based on expectation.
    while time.time() < deadline:
        conv = conv_fft(a, fft_len, n)
        max_idx = int(np.argmax(conv))
        # subgradient
        idx = np.arange(n)
        counterpart = max_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

def anneal(a, current_score, fft_len, n, deadline):
    a = a.copy()
    best_a = a.copy()
    best_score = current_score
    temp = 0.01 # starting temperature determined empirically.
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Move fraction of mass from i to j
        frac = rng.random() * 0.3
        delta = a[i] * frac
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        new_score = fast_score(a, fft_len, n)
        # Accept if improve or with probability exp(-(new_score-current_score)/temp)
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-12)):
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        # annealing schedule
        temp *= 0.9999
    return best_a, best_score
```

"""
Propose a non‑negative step‑function (list of floats) with a low
evaluate_sequence() score.

Algorithm
---------
1. Choose a problem size n (≈ 5‑8 k).  If a previous construction
   ``height_sequence_1`` is available we reuse its length, otherwise we
   take n=8000.  All work is performed on the probability simplex
   (sum = 1) – the objective is scale‑invariant.

2. Multi‑restart continuation L‑BFGS:
      – many random initialisations (Gaussian, exponential decay,
        power‑law, edge‑heavy, Sidon‑set, binary mask, Gaussian bump,
        and a perturbed copy of the current best).
      – a schedule of decreasing smoothing parameters
        (log‑sum‑exp approximation of the max‑convolution).
      – each stage runs a relatively large number of LBFGS iterations.
   After each run we evaluate the true (fast) score and keep the best
   candidate.  A short greedy hill‑climb is applied after every L‑BFGS
   run.

3. Targeted projected sub‑gradient descent on the true objective
   f(a)=2 n max(conv(a,a)).
   The gradient of the max entry is known analytically; we take adaptive
   step sizes and always project back onto the simplex.

4. Simulated‑annealing style random mass‑transfer moves:
   we continue to explore the neighbourhood of the current best,
   accepting occasional worsening moves to escape shallow local minima.

5. Final aggressive hill‑climb (larger move size) until the time budget
   expires.

All phases respect the global time budget (≈ 1000 s).  The final
sequence is returned as a plain Python list; the harness will call
``evaluate_sequence`` on it.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # back‑propagation of smooth_max
    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Run a continuation L‑BFGS optimisation with decreasing smoothing.
    """
    y = y_start.copy()
    for lam_factor in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam_factor),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent_adaptive(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    step_init: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv) with an adaptive step size.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    step = step_init
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = conv_fft(a, fft_len, n)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        counterpart = max_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # trial step
        a_trial = a - step * g
        a_trial = project_onto_simplex(a_trial)
        trial_score = fast_score(a_trial, fft_len, n)

        if trial_score < best_score:
            best_score = trial_score
            best_a = a_trial.copy()
            a = a_trial
            # increase step modestly after a success
            step = min(step * 1.2, 1.0)
        else:
            # no improvement: shrink step
            step *= 0.7
            if step < 1e-6:
                step = step_init
    return best_a, best_score


def annealing_optimize(
    a_start: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    temp_start: float = 0.02,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style random mass transfers.
    Accepts worsening moves with probability exp(-Δ/ T).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = cur_score
    cur = cur_score
    rng = np.random.default_rng()
    T = temp_start
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = a[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta
        new_score = fast_score(a, fft_len, n)
        accept = False
        if new_score < cur:
            accept = True
        else:
            prob = np.exp(-(new_score - cur) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert the move
            a[i] += delta
            a[j] -= delta
        # exponential cooling
        T *= 0.99995
        if T < 1e-7:
            T = 1e-7
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    # Prefer to reuse a previously found sequence if available.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)
        else:
            n = 8000
    else:
        n = 8000

    # FFT lengths (powers of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate on the simplex (sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / tmp.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with many restarts (≈ 55 % budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-12, num=45))
    main_phase_deadline = start_time + 0.55 * budget_s
    iteration = 0
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "binary",
        "bump",
        "uniform",
        "best",
    ]

    while time.time() < main_phase_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # Continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # Short greedy hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – adaptive projected sub‑gradient descent (≈ 30 % budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 15.0, start_time + 0.85 * budget_s)
    print("[subgrad] starting adaptive projected sub‑gradient descent")
    best_a, best_score = subgradient_descent_adaptive(
        best_a,
        fft_len_conv,

    )

    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing (≈ 10 % budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        print("[anneal] simulated‑annealing phase")
        best_a, best_score = annealing_optimize(
            best_a,
            best_score,
            fft_len_conv,
            n,
            anneal_deadline,
            temp_start=0.02,
            move_frac=0.30,
        )

        print(f"[anneal] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final aggressive hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:

        best_a, best_score = hill_climb(

    # --------------------------------------------------------------

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # Provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
