"""
Hybrid optimisation for the step‑function score.

The algorithm runs three phases within the 1000 s budget:

1. **Global search** – several random restarts of a smooth‑max
   L‑BFGS optimisation on the log‑soft‑max parametrisation.
   The smoothness parameter λ (expressed as a fraction of the current
   maximal convolution) is decreased geometrically (continuation).

2. **Long projected‑gradient refinement** – after the global search the
   current best sequence is further minimised by projected gradient descent
   on the smoothed max of the auto‑convolution for a sequence of λ‑values
   that approach zero.  The step size follows a 1/√t rule and is
   adaptively reduced when a trial does not improve the fast surrogate.

3. **Local escaping moves** – a final aggressive random hill‑climb (mass
   transfer) is performed with the remaining time to explore tiny
   plateaus.

All iterates stay on the probability simplex (non‑negative, ℓ₁‑norm 1).
The fast surrogate `fast_score` ( = 2·n·max(conv) ) is used during the
search; the official `evaluate_sequence` is called only once at the end.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities – top‑level only
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a (full) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max ----
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv ----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ---- back‑prop to a (gradient vector of length n) ----
    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient w.r.t. y (soft‑max Jacobian included)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = opt.minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer hill‑climb (strictly improving moves only)."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-13:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weights under the same shift
        partner_weight = np.full(n, np.inf)
        in_range = mask
        partner_weight[in_range] = best[partner_idx[in_range]]

        # choose a target where the partner is smallest
        cand_js = rng.integers(0, n, size=5)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-13:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns the derivative of the smooth max (without the outer 2·n factor).
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # derivative of smooth max w.r.t. a (outer 2·n factor handled later)
    return 2.0 * g

        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            center_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, center_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # "best"
            y0 = best_y + rng.normal(scale=0.05, size=n)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lbfgs_lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart {restart}] mode={mode} improved fast score = {best_score:.6f}")

        # brief targeted and random hill‑climbs to escape tiny plateaus
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=500,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=3000,
            max_move_frac=0.18,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 2 – extended projected‑gradient descent on smooth‑max
    # ------------------------------------------------------------------
    pg_start = time.time()
    lam_factors_pg = np.geomspace(0.5, 1e-7, num=28)  # finer near zero
    step0 = 0.28
    pg_iter = 0

    while time.time() < deadline - 3.0:
        for lam_factor in lam_factors_pg:
            if time.time() >= deadline - 3.0:
                break
            # gradient of the smooth‑max (without outer 2·n factor)
            grad = grad_smooth_max(best_a, lam_factor)
            pg_iter += 1
            step = step0 / np.sqrt(pg_iter)
            cand = proj_simplex(best_a - step * grad)
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score - 1e-13:
                best_a = cand
                best_score = cand_score
                best_y = np.log(best_a + 1e-12)
                # slightly increase step size when we make progress
                step0 = min(step0 * 1.02, 0.5)
            else:
                # reduce step size on failure to keep stability
                step0 *= 0.92
                if step0 < 1e-4:
                    step0 = 1e-4

    print(f"[pg] iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final aggressive random hill‑climb (uses any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
