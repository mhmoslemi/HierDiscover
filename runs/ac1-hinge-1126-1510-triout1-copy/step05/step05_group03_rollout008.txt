    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
    M = conv.max()
    # select constraints with conv >= M - eps
    eps = M * 0.01   # 1% tolerance
    idx = np.where(conv >= M - eps)[0]
    if idx.size == 0:

    # limit to max_constraints highest values
    if idx.size > max_constraints:
        # sort by conv descending and take top
        top = np.argsort(-conv[idx])[:max_constraints]
        idx = idx[top]
    # Build A_ub (len(idx) x n)
    K = idx.size
    A = np.zeros((K, n), dtype=float)
    for row, k in enumerate(idx):

        end_i = min(k, n - 1)

"""
Propose a non‑negative step‑function sequence that minimises the
``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)
--------------------------------------------------------------------
1. Initialise a probability vector a (∑a = 1).  If a previous construction
   ``height_sequence_1`` exists it is reused, otherwise a smooth exponential
   shape is used.

2. Global optimisation:
   * Optimise the log‑soft‑max parameters y = log(a) with L‑BFGS.
   * A geometric continuation on the smooth‑max parameter λ (from 0.5·M
     down to 1e‑9·M) approaches the true max of the autocorrelation.
   * Many random restarts are performed while time permits.

3. After each L‑BFGS run two refinements are applied:
   * Targeted hill‑climb that reduces the current maximal convolution entry
     by moving mass away from the worst contributing index pair.
   * Random mass‑transfer hill‑climb (classic “move a fraction of mass”.

4. Dual LP step (inspired by the state‑of‑the‑art iterative scheme):
   * Compute M = max(a∗a) and collect the shifts where the convolution is
     close to M.
   * Solve a linear program that maximises Σ gᵢ under g ≥ 0 and the
     constraints (a∗g)[k] ≤ M for the selected shifts.
   * Blend the obtained g with a (convex combination) and keep the blend
     if it reduces the fast surrogate ``2·n·max(conv)``.

5. Projected‑gradient descent on the smooth‑max objective (λ → 0) for final
   polishing.

6. A final aggressive random hill‑climb finishes the search.

All steps keep the vector on the probability simplex, work on a fixed
length (default n = 5000) and respect the given time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a (full) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max ----
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max w.r.t. conv ----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ---- chain rule back to a (gradient vector of length n) ----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient of f = 2·n·smooth_max w.r.t. a is 2·n·(∂smooth_max/∂a) = 2·n·g_vec
    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        # pick worst contributor
        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under same shift – used to pick a cheap target
        partner_weight = np.full(n, np.inf)
        in_range = mask
        partner_weight[in_range] = best[partner_idx[in_range]]

        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """Gradient of the smooth max of the auto‑convolution w.r.t. a."""
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]
    return 2.0 * g  # derivative of smooth max w.r.t. a (outer 2·n omitted)


def lp_improve_step(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_constraints: int = 12,
    eps_frac: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Solve the linear‑programming sub‑problem described in the state‑of‑the‑art
    iteration and blend the solution with the current vector if it improves
    the fast surrogate.
    """
    # current convolution and max
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    M = conv.max()

    # choose a small set of critical shifts
    eps = M * eps_frac
    cand = np.where(conv >= M - eps)[0]
    if cand.size == 0:
        cand = np.array([np.argmax(conv)], dtype=int)
    # keep at most max_constraints most critical shifts
    if cand.size > max_constraints:
        order = np.argsort(-conv[cand])[:max_constraints]
        cand = cand[order]

    K = cand.size
    A = np.zeros((K, n), dtype=float)
    for r, k in enumerate(cand):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        A[r, j_vals] = a[i_vals]          # (a * g)[k] = Σ a_i * g_{k-i}

    b = np.full(K, M)

    # maximise Σ g_i   ⇔   minimise -Σ g_i
    c = -np.ones(n)

    # Solve the LP (highs is fast and handles moderate size problems)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
    except Exception:
        return a, cur_score

    if not res.success:
        return a, cur_score

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return a, cur_score

    # normalise g to keep the simplex constraint after blending
    g_norm = g / sum_g

    # try a few blend ratios and keep the best improving one
    improved = False
    best_local_a = a
    best_local_score = cur_score
    for t in (0.25, 0.5, 0.75):
        cand_a = (1.0 - t) * a + t * g_norm
        cand_score = fast_score(cand_a, fft_len, n)
        if cand_score < best_local_score - 1e-12:
            best_local_a = cand_a
            best_local_score = cand_score
            improved = True

    # fallback: a small step towards g_norm
    if not improved:
        t = 0.1
        cand_a = (1.0 - t) * a + t * g_norm
        cand_score = fast_score(cand_a, fft_len, n)
        if cand_score < best_local_score - 1e-12:
            best_local_a = cand_a
            best_local_score = cand_score
            improved = True

    if improved:
        return best_local_a, best_local_score
    else:
        return a, cur_score

    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to the simplex
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}  fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # L‑BFGS continuation schedule (global search)
    # ------------------------------------------------------------------
    lam_factors = list(np.geomspace(0.5, 1e-9, 45))   # finer λ‑schedule
    maxiter_per_stage = 800                         # more work per λ‑stage

    iteration = 0
    # outer loop = many random restarts + local refinements
    while time.time() < deadline - 5.0:
        # --------------------------------------------------------------
        # 1) Choose an initialisation strategy for the log‑soft‑max
        # --------------------------------------------------------------
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            center_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, center_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best solution
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # --------------------------------------------------------------
        # 2) Continuation L‑BFGS
        # --------------------------------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score - 1e-12:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) LP‑based dual improvement (occasionally)
        # --------------------------------------------------------------
        if time.time() < deadline - 10.0:
            best_a, best_score = lp_improve_step(
                a=best_a,
                cur_score=best_score,
                fft_len=fft_len_conv,
                n=n,
                deadline=deadline,
                rng=rng,
                max_constraints=12,
                eps_frac=0.01,
            )
            best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 4) Targeted hill‑climb
        # --------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1500,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 5) Random hill‑climb (helps to escape tiny plateaus)
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=5000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # 6) Projected‑gradient descent on the smooth‑max (λ → 0)
    # ------------------------------------------------------------------
    lam_factor_pg = 1e-5
    step0 = 0.30
    max_pg_iter = 4000
    pg_iter = 0
    while time.time() < deadline - 2.0 and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)   # gradient without outer 2·n
        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            lam_factor_pg = max(lam_factor_pg * 0.9, 1e-7)

    print(f"[pg] projected‑gradient iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
