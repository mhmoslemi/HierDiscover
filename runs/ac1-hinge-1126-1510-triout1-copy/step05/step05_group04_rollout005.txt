"""
Propose a non‑negative step‑function (list of floats) whose evaluation score
2·n·max(conv)/sum² is as low as possible.

Strategy
--------
1. Choose a moderately large problem size (n≈8000) and work on the
   probability simplex (sum = 1) – the objective is scale‑invariant.
2. Perform a fast “shape” sweep over a few symmetric families
   (exponential, power‑law, cosine‑based) that concentrate mass at the
   edges.  The best shape gives a very good starting point.
3. Refine the candidate with a continuation L‑BFGS on a smooth
   log‑sum‑exp approximation of the max‑convolution.  The schedule
   gradually reduces the smoothing factor.
4. Continue optimisation with a projected (sub‑)gradient descent on the
   true max‑convolution (full gradient of the active entry) and a short
   aggressive hill‑climb.
5. Return the final distribution; evaluate_sequence will scale it
   automatically.

The code respects the 1000 s budget, prints progress, and uses only
numpy/scipy which are available in the execution environment.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv w.r.t a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def projected_gradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the true objective 2·n·max(conv).
    Uses the exact gradient of the active convolution entry.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))                       # index of maximal entry

        idx = np.arange(n)
        counterpart = m - idx
        valid = (counterpart >= 0) & (counterpart < n)

        g = np.zeros_like(a)
        # full gradient of conv[m] w.r.t a_i = a_{m-i} + a_i (if valid)
        g[valid] = a[counterpart[valid]] + a[idx[valid]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()

        a = a_new
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Determine problem size n
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Baseline uniform distribution
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Fast sweep over simple symmetric families
    # --------------------------------------------------------------
    idx = np.arange(n)
    half = n - 1
    # ----- exponential (edge‑heavy) -----
    for lam in np.logspace(-4, -1.0, num=30):
        if time.time() > deadline:
            break
        d = np.exp(-lam * np.minimum(idx, half - idx))
        a = d / d.sum()
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_score, best_a = s, a
            print(f"[shape] exp lam={lam:.2e} score={s:.6f}")

    # ----- power‑law (edge‑heavy, positive exponent) -----
    for p in np.linspace(0.2, 5.0, num=30):
        if time.time() > deadline:
            break
        d = (np.minimum(idx, half - idx) + 1) ** p
        a = d / d.sum()
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_score, best_a = s, a
            print(f"[shape] pow+ p={p:.2f} score={s:.6f}")

    # ----- power‑law (center‑heavy, negative exponent) -----
    for p in np.linspace(0.2, 5.0, num=30):
        if time.time() > deadline:
            break
        d = (np.minimum(idx, half - idx) + 1) ** (-p)
        a = d / d.sum()
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_score, best_a = s, a
            print(f"[shape] pow- p={p:.2f} score={s:.6f}")

    # ----- cosine‑based (edges 1, centre 0) -----
    base_cos = np.maximum(0.0, 1 + np.cos(np.pi * idx / half))
    for p in np.linspace(0.5, 5.0, num=30):
        if time.time() > deadline:
            break
        a = (base_cos ** p)
        a = a / a.sum()
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_score, best_a = s, a
            print(f"[shape] cos p={p:.2f} score={s:.6f}")

    # --------------------------------------------------------------
    # 2) Continuation L‑BFGS (smooth approximation of max)
    # --------------------------------------------------------------
    best_y = np.log(best_a + 1e-12)  # pre‑softmax parameters
    lam_schedule = list(np.geomspace(0.5, 1e-7, num=30))

    main_deadline = start_time + 0.55 * budget_s
    it = 0
    while time.time() < main_deadline:
        # start from a slight perturbation of the current best
        y0 = best_y + rng.normal(scale=0.03, size=n)
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len,
            fft_len,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        s_opt = fast_score(a_opt, fft_len, n)
        if s_opt < best_score:
            best_score, best_a, best_y = s_opt, a_opt, y_opt
            print(f"[lbfgs] it={it:04d} improved fast score = {s_opt:.6f}")

        # short hill‑climb polishing
        hc_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_dead,
            max_iter=2500,
            max_move=0.15,
        )
        best_y = np.log(best_a + 1e-12)
        it += 1

    # --------------------------------------------------------------
    # 3) Projected (sub‑)gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    grad_deadline = min(deadline - 5.0, start_time + 0.85 * budget_s)
    print("[grad] start projected gradient descent")
    best_a, best_score = projected_gradient_descent(
        best_a,
        fft_len,
        n,
        grad_deadline,
        eta0=0.30,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[grad] finished, fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final aggressive hill‑climb using the remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=20000,
            max_move=0.30,
        )

    # --------------------------------------------------------------
    # 5) Return the sequence (scale does not matter – keep sum≈1)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
