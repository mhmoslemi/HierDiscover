"""
Propose a low‑evaluation non‑negative step sequence.

Algorithm overview
-------------------
1. **Parameter sweep** – generate a handful of diverse seeds
   (uniform, linear, exponential, power‑law, edge‑heavy, sinusoid,
   Dirichlet draws).  Keep the best few according to the exact score.

2. **Continuation L‑BFGS** – optimise the smooth log‑sum‑exp
   approximation of the max‑convolution on the soft‑max
   (probability‑simplex) parametrisation.
   A fine λ‑schedule together with many restarts explores the landscape.

3. **Linear‑programming balancing** – maximise ∑b subject to
   b≥0 and (a∗b)≤M (M=current peak).  The LP is solved with
   SciPy’s high‑performance ‘highs’ solver (sparse matrix).  The
   solution b is scaled to the simplex and mixed back into a
   with a small factor *t*.  Several iterations are performed.

4. **Projected sub‑gradient descent** – directly minimise the exact
   objective 2·n·max(a∗a) using the sub‑gradient of the max‑convolution,
   with a simple back‑tracking line‑search.

5. **Greedy hill‑climb** – random mass transfers between two coordinates
   (preserving the simplex) accept only strictly improving moves.
   Run until the time budget expires.

All stages respect the simplex constraint, use FFT‑based convolutions
(O(n log n)), and split the total 1000 s budget automatically.
The best sequence found before the deadline is returned as a plain list.
"""

import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # Gradient of smooth max
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # convolution of w with reversed a (to get derivative wrt a)
    a_rev = a[::-1]
    fw = fft(w, fft_len)
    fa_rev = fft(a_rev, fft_len)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple λ values)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best vector found.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # quick hill‑climb after each λ to escape possible plateaus
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hc_deadline,
                                        max_iter=1200)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len: int,
                         n: int):
    """
    Directly minimise 2·n·max(a∗a) via sub‑gradient descent
    with a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len)
    it = 0

    while time.time() < deadline:
        # current convolution and active mask
        fa = fft(a, fft_len)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len)
        fa_rev = fft(a_rev, fft_len)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta
        i += 1

    return best, best_score

# LP balancing step using SciPy’s linprog (sparse)
# ----------------------------------------------------------------------
def lp_balancing(a: np.ndarray,
                 n: int,
                 deadline: float,
                 t_init: float = 0.12,
                 max_iters: int = 12):
    """
    Perform a few iterations of the classic M‑balancing LP:
        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k

    The LP is solved with scipy.optimize.linprog (highs, sparse).
    After each solve we mix a fraction t of the scaled solution back
    into a.  The mixing factor is halved whenever a step does not improve.
    """
    a = a.copy()
    a = a / a.sum()
    t = t_init
    fft_len = next_pow2(2 * n - 1)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current peak M
        M = fast_score(a, n, fft_len) / (2 * n)

        # Build sparse convolution matrix T (size (2n‑1) × n)
        rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
        cols = np.repeat(np.arange(n), n)
        data = np.repeat(a, n)
        T = coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsc()

        # Linear programme: maximise sum(b) ↔ minimise -sum(b)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        # Solve
        res = linprog(c=c,
                      A_ub=T,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success or res.x is None:
            print("[lp] linprog failed or not successful")
            break

        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        b_scaled = b_val / s

        a_new = (1.0 - t) * a + t * b_scaled
        a_new = simplex_projection(a_new)

        new_score = fast_score(a_new, n, fft_len)

        if new_score < fast_score(a, n, fft_len) - 1e-12:
            a = a_new
            t = t_init
            it += 1
            # print progress (optional)
            print(f"[lp] iter {it}, improved score={new_score:.6f}")
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1

    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step sequence.
    Returns a plain Python list of non‑negative floats (sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    max_n = 2000                     # moderate size – good trade‑off
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0[a0 < 0] = 0.0
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)   # for all FFT‑based convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, n, fft_len)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (collect seeds)
    # --------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len)
        candidates.append((sc, arr))

    # uniform
    add_candidate(np.full(n, 1.0 / n))

    # linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # exponential families
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # random Dirichlet (Gamma) draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # keep the best few seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:6]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin
    phase1_end = start + 0.50 * budget_s      # L‑BFGS
    phase2_end = start + 0.80 * budget_s      # LP + sub‑gradient

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule = [
        0.8, 0.6, 0.4, 0.2, 0.1, 0.08, 0.05,
        0.03, 0.02, 0.01, 0.008, 0.005,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05
    ]
    maxiter_per_lam = 300
    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
        restarts += 1

    # --------------------------------------------------------------
    # Phase 2 – LP balancing + projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        # LP balancing (takes a few seconds)
        lp_deadline = min(phase2_end, deadline - 0.3)
        print("[LP] starting balancing iterations")
        best_a = lp_balancing(best_a, n, lp_deadline,
                              t_init=0.12, max_iters=12)
        best_score = fast_score(best_a, n, fft_len)
        print(f"[LP] post‑LP score={best_score:.6f}")

        # Sub‑gradient descent (longer)
        if time.time() < phase2_end:
            sg_deadline = min(phase2_end, deadline - 0.2)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len,
                                                      n)
            print(f"[sub‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final hill‑climb polishing (as much time as left)
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len,
                                    n,
                                    final_deadline,
                                    max_iter=15000)

    # --------------------------------------------------------------
    # Return the sequence (plain list)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # sanity check – the external evaluator will also clip negatives etc.
    final_eval = evaluate_sequence(final_seq)    # provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
