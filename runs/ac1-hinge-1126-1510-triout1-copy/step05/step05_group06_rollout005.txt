"""
Search for a low‑evaluation non‑negative step function.

Algorithm
---------
1. **Parameter sweep** – generate a handful of simple candidate shapes
   (uniform, Gaussian bells, cosine bells, beta‑like, Dirichlet draws and the
   previous best sequence if available).  Keep the few best seeds
   (according to the fast 2·n·max(conv) score).

2. **Adam optimisation on a smooth surrogate** – minimise the smoothed
   max‑convolution  f(a)=2·n·smooth_max(a*a)  where the max is replaced by a
   log‑sum‑exp with a decreasing smoothing factor λ.
   The gradient of the smooth surrogate is obtained analytically via FFTs.
   After each Adam update the vector is projected onto the probability simplex
   and symmetrised (a ← (a+rev(a))/2) – the problem is symmetric, and this
   reduces the search space.

3. **Intensive hill‑climb** – once the Adam phase finishes, perform a large
   number of random mass‑transfer moves (with a simulated‑annealing acceptance
   rule) while staying on the simplex.  This frequently improves the exact
   max‑convolution score a few more units.

All phases respect the 1000 s time budget; the best sequence found is
returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len_conv: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1 (exact objective)."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a_i = a_{n-1-i} while preserving the sum."""
    b = (a + a[::-1]) * 0.5
    # sum is unchanged by averaging with the reversed copy
    return b


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its exact gradient
# ----------------------------------------------------------------------
def smooth_obj_and_grad(a: np.ndarray,
                       lam_factor: float,
                       fft_len_conv: int,
                       fft_len_g: int) -> tuple[float, np.ndarray]:
    """
    Compute the smoothed objective f = 2·n·smooth_max(a*a) and its gradient.
    lam_factor controls the smoothing (λ = lam_factor * max(conv)).
    Returns (f, grad) where grad = ∂f/∂a.
    """
    n = a.shape[0]

    # ----- convolution a*a -----
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]
    max_c = conv_full.max()

    # ----- smoothing parameters -----
    lam = max_c * lam_factor + 1e-12
    shifted = (conv_full - max_c) / lam          # ≤0 for all entries
    exp_shift = np.exp(shifted)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t a -----
    # weights w_k = ∂smooth_max/∂conv_k  = exp_shift / Z
    w = exp_shift / Z

    # gradient ∂smooth_max/∂a_i = 2 * Σ_k w_k * a_{k-i}
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev_full = np.real(ifft(fa_rev * fw))
    # slice that corresponds to indices i = 0..n-1
    g_vec = 2.0 * conv_rev_full[(n - 1):(n - 1 + n)]

    grad = 2.0 * n * g_vec          # chain rule for f = 2·n·smooth_max
    return f, grad


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def adam_optimize(seed_vec: np.ndarray,
                  lam_schedule: list[float],
                  max_iters_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that runs through the smoothing schedule.
    Returns the best vector (on the simplex) and its exact fast_score.
    """
    a = seed_vec.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12                     # base learning rate; will be scaled automatically

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t_it = 0

    for lam_fac in lam_schedule:
        if time.time() >= deadline:
            break
        # limit iterations for this λ
        for _ in range(max_iters_per_lam):
            if time.time() >= deadline:
                break
            t_it += 1
            f, grad = smooth_obj_and_grad(a, lam_fac,
                                          fft_len_conv, fft_len_g)

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_it)
            v_hat = v / (1.0 - beta2 ** t_it)

            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = a - step

            # projection onto simplex + optional symmetrisation
            a = simplex_projection(a)
            a = symmetrize(a)

            # exact evaluation
            cur_score = fast_score(a, n, fft_len_conv)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # after finishing this λ we optionally do a short hill‑climb
        # (a few hundred cheap moves) to escape possible plateaus
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=800)
        a = best_a.copy()
        # reset Adam moments after a large perturbation
        m.fill(0.0)
        v.fill(0.0)
        t_it = 0

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with optional simulated‑annealing acceptance
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000,
               temperature: float = 0.0) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb.
    If temperature>0 a Metropolis acceptance is used.
    Returns improved (a, score).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            it += 1
            continue
        # transfer a random fraction of the mass at i1 to i2
        delta = best[i1] * rng.random() * 0.07   # up to 7 % moves
        if delta <= 0.0:
            it += 1
            continue
        cand = best.copy()
        cand[i1] -= delta
        cand[i2] += delta
        cand = simplex_projection(cand)
        cand = symmetrize(cand)

        new_score = fast_score(cand, n, fft_len)

        accept = False
        if new_score < best_score - 1e-12:
            accept = True
        elif temperature > 0.0:
            # Metropolis criterion
            prob = math.exp(-(new_score - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            best = cand
            best_score = new_score
        it += 1
    return best, best_score
