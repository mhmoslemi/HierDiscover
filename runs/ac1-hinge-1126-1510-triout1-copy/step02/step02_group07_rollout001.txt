def propose_candidate(seed=42, budget_s=1000, **kwargs):
    # Setup
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = kwargs.get('n', 1200)  # default length

    # base
    if 'height_sequence_1' in globals():
        base = np.array(height_sequence_1, dtype=float)
        if base.size != n:
            # if different size adjust or interpolate; for simplicity, truncate or pad uniformly
            if base.size > n:
                base = base[:n]
            else:
                base = np.pad(base, (0, n-base.size), 'constant')
                base[n-base.size:] = 0.0
        base = base / base.sum()
    else:
        base = _half_order_series(n)

    # Perform param sweep: tilt delta and mixing alpha for better start
    best_seq = base.copy()
    best_score = float('inf')
    # compute eval function using evaluate_sequence for final but for speed compute own bound 
    # because it's same if sum=1: score=2*n*max(conv)
    for delta in np.linspace(0.0, 0.7, 15):
        tapered = base * (np.arange(1, n+1) ** (-delta))
        tapered /= tapered.sum()
        conv = np.convolve(tapered, tapered)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = tapered.copy()
    uniform = np.full(n, 1.0/n)
    for alpha in np.linspace(0.0, 1.0, 11):
        mixed = (1-alpha) * best_seq + alpha * uniform
        mixed /= mixed.sum()
        conv = np.convolve(mixed, mixed)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = mixed.copy()

    cur_seq = best_seq.copy()
    cur_score = best_score
    best_seq_global = cur_seq.copy()
    best_score_global = cur_score

    # Parameter schedules
    total_time = deadline - start
    iteration = 0
    # steps for gradient descent
    gamma = 0.1  # initial step size
    gamma_min = 1e-6
    gamma_max = 1.0

    # temperature for metropolis acceptance for random perturbations
    init_temp = cur_score * 0.1  # around 0.1 of score
    final_temp = 1e-12

    # Softmax alpha schedule
    alpha = 4.0  # initial
    alpha_max = 400.0

    while time.time() < deadline:

        now = time.time()

        # update schedules
        # increase alpha exponentially to final

        # temperature decreasing

        if np.random.rand() < 0.3:

            new_weights /= new_weights.sum()

            cand_seq = cur_seq.copy()
            cand_seq[idx] = total_mass * new_weights
            # ensure no negative (not needed)
            # evaluate

            cand_score = 2.0 * n * conv.max()
            accept = False

                    if np.random.rand() < prob:
                        accept = True
            if accept:

"""
Propose a candidate step‑function sequence.

The algorithm combines three ideas:
1.  A good deterministic start – the “half‑order” (central‑binomial) series,
    optionally tilted and mixed with the uniform distribution.
2.  Stochastic mass‑redistribution moves (random Dirichlet reshuffling of a
    small subset of coordinates) with a Metropolis acceptance rule.
3.  Gradient descent on a smooth approximation of the max‑convolution:
    we replace max x_k by (1/α)·log Σ exp(α·x_k) and compute its exact gradient.
    The gradient step is projected onto the probability simplex.

Alpha (smoothness) and the Metropolis temperature are increased/decreased
linearly in time, while the step size for the gradient descent adapts
automatically.  The best sequence seen so far is kept and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i) / 4**i,
    normalised so that sum(a) == 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    if np.all(v <= 0):
        # project to uniform point
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises
        2·n·max(conv(a,a)) / (sum(a)²)
    (the evaluate_sequence() score).  The routine runs for at most
    ``budget_s`` seconds and returns the best sequence found as a plain
    Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin so the caller can still stop us cleanly
    deadline = start_time + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # 1) length of the sequence & initial base
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        # use the provided construction if its length matches the request
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = base.size
        base = base / base.sum()
    else:
        n = int(kwargs.get("n", 1200))          # default length
        base = _half_order_series(n)

    # ------------------------------------------------------------------
    # 2) cheap parametric sweep to obtain a decent starting point
    # ------------------------------------------------------------------
    best_seq = base.copy()
    best_score = float("inf")

    # tilt the tail by a power law (exponent delta)
    for delta in np.linspace(0.0, 0.7, 15):
        taper = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = base * taper
        cand /= cand.sum()
        conv = np.convolve(cand, cand)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # convex combination with uniform distribution
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha_mix in np.linspace(0.0, 1.0, 11):
        cand = (1.0 - alpha_mix) * best_seq + alpha_mix * uniform
        cand /= cand.sum()
        conv = np.convolve(cand, cand)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) main optimisation loop
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    best_seq_global = cur_seq.copy()
    best_score_global = cur_score

    # schedules

    iteration = 0

    # gradient‑descent step size (projected onto the simplex)
    step_size = 0.1
    step_min = 1e-6
    step_max = 1.0

    # Metropolis temperature for random moves
    init_temp = cur_score * 0.1          # warm start
    final_temp = 1e-12

    # smooth‑max parameter α (controls how close we are to the true max)
    alpha_start = 1.0
    alpha_end = 400.0

    # probability of performing a random redistribution move
    prob_random = 0.3

    while time.time() < deadline:
        iteration += 1
        now = time.time()
        progress = (now - start_time) / total_time

        # update schedules
        temp = init_temp * (final_temp / init_temp) ** progress
        alpha = alpha_start + (alpha_end - alpha_start) * progress
        # --------------------------------------------------------------
        # 3a) random redistribution move
        # --------------------------------------------------------------
        if np.random.rand() < prob_random:
            # pick a small random subset of indices (2‑6 elements)
            k = np.random.randint(2, 7)
            idx = np.random.choice(n, k, replace=False)
            mass = cur_seq[idx].sum()
            if mass > 0.0:
                new_weights = np.random.rand(k)
                new_weights /= new_weights.sum()
                cand_seq = cur_seq.copy()
                cand_seq[idx] = mass * new_weights
                # evaluate candidate
                conv = np.convolve(cand_seq, cand_seq)
                cand_score = 2.0 * n * conv.max()
                accept = False
                if cand_score < cur_score:
                    accept = True
                else:
                    # Metropolis acceptance for uphill moves
                    if temp > 0.0:
                        prob = math.exp(-(cand_score - cur_score) / temp)
                        if np.random.rand() < prob:
                            accept = True
                if accept:
                    cur_seq = cand_seq
                    cur_score = cand_score
                    if cur_score < best_score_global:
                        best_score_global = cur_score
                        best_seq_global = cur_seq.copy()
                        print(f"[{iteration:6d}] random move improved to {best_score_global:.6f}")

        # --------------------------------------------------------------
        # 3b) smooth‑max gradient descent
        # --------------------------------------------------------------
        else:
            conv = np.convolve(cur_seq, cur_seq)
            max_val = conv.max()
            # soft‑max weights (stable version)
            shifted = conv - max_val                # non‑positive, max = 0
            exp_vals = np.exp(alpha * shifted)      # 1 for the max entry
            w = exp_vals / exp_vals.sum()

            # gradient of the smooth max (2 · cross‑correlation)
            grad = 2.0 * np.correlate(w, cur_seq, mode='valid')

            # try a few step‑size reductions (back‑tracking)
            improved = False
            trial_step = step_size
            for _ in range(4):
                cand_seq = cur_seq - trial_step * grad
                cand_seq = _project_simplex(cand_seq)
                conv_cand = np.convolve(cand_seq, cand_seq)
                cand_score = 2.0 * n * conv_cand.max()
                if cand_score < cur_score:
                    cur_seq = cand_seq
                    cur_score = cand_score
                    improved = True
                    # modestly increase step size for next iteration
                    step_size = min(step_size * 1.2, step_max)
                    break
                trial_step *= 0.5
            if not improved:
                # shrink step size when we cannot improve
                step_size = max(step_size * 0.7, step_min)
                # occasional jitter to escape tiny plateaux
                if iteration % 400 == 0:
                    sigma = 0.002
                    noisy = cur_seq * (1.0 + np.random.randn(n) * sigma)
                    noisy = np.clip(noisy, 0.0, None)
                    noisy /= noisy.sum()
                    cur_seq = _project_simplex(noisy)
                    conv = np.convolve(cur_seq, cur_seq)
                    cur_score = 2.0 * n * conv.max()

            # record global best
            if cur_score < best_score_global:
                best_score_global = cur_score
                best_seq_global = cur_seq.copy()
                print(f"[{iteration:6d}] gradient step improved to {best_score_global:.6f}")

        # occasional progress report
        if iteration % 1000 == 0:
            elapsed = now - start_time
            print(f"[{iteration:6d}] elapsed {elapsed:.1f}s, best {best_score_global:.6f}, cur {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) return best sequence found
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score_global:.6f}")
    return best_seq_global.tolist()
