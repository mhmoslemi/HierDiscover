    iteration = 0
    last_print = time.time()

        # compute conv and its FFT
        Lc = next_pow2(2*n)
        fa = np.fft.rfft(a, n=Lc)
        conv = np.fft.irfft(fa * fa, n=Lc)[:2*n-1]
        maxc = conv.max()

        # Determine active tolerance: relative tolerance allows multiple peaks.
        eps_rel = 0.002  # maybe 0.2%

        # generate w mask

        if active.size == 0:

"""
Propose a candidate step‑function (non‑negative sequence) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1️⃣  **Diversified random sampling** – many families of shapes
    (power‑law, exponential, two‑sided power‑law, Beta‑like,
    Gaussian) with random lengths and parameters are generated,
    normalised to sum = 1 and scored with a fast FFT‑based evaluator.
    The best sequence found so far is kept.

2️⃣  **Active‑set exponentiated‑gradient refinement** – starting from the
    best candidate we repeatedly reduce the peak of the self‑convolution.
    In each iteration we
        * evaluate the full convolution via FFT,
        * collect all indices whose value is within a tiny relative
          tolerance of the current maximum (the “active set”),
        * compute the sub‑gradient (average over the active set) by a
          correlation performed with FFT,
        * perform an exponentiated‑gradient step on the probability
          simplex (multiplicative update + renormalisation) with a
          diminishing step size,
        * keep the globally best sequence.
    This “active‑set” gradient is more informative than using a single
    maximal index and the multiplicative update avoids the costly
    Euclidean projection.

3️⃣  **Final verification** – the best sequence is converted to a plain
    Python list and evaluated with the provided `evaluate_sequence`
    function.

The whole search respects the 1000 s budget (a small safety margin is
kept).  The implementation is self‑contained, requires only NumPy and
the standard library and returns a list of floats ready for scoring.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (sum of seq is assumed > 0)."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # zero‑pad to a power of two for speed
    L = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_conv = float(np.max(conv))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^{-p_left} + w_right * (n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta‑like shape:
        a_i ∝ (i+1)^{alpha-1} * (n-i)^{beta-1},
    which is unimodal and vanishes at the ends for alpha,beta>1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """
    Discrete Gaussian centred in the middle of the interval.
    """
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - centre) / sigma) ** 2)


# ----------------------------------------------------------------------
# Core refinement routine (active‑set exponentiated gradient)
# ----------------------------------------------------------------------


def refine_sequence(
    seq: np.ndarray, deadline: float, start_time: float, eps_rel: float = 0.002
) -> np.ndarray:
    """
    Starting from `seq` (already normalised to sum = 1), perform many
    exponentiated‑gradient steps using the active set of maximal
    convolution entries.  Returns the best sequence discovered before
    `deadline`.
    """
    a = seq.copy()
    n = a.shape[0]

    # pre‑allocate FFT length that can accommodate both convolution and correlation
    L = next_pow2(3 * n)  # >= n + (2n‑1) – 1

    best_a = a.copy()
    best_score = fast_score(a)

    iteration = 0
    last_report = start_time

    while time.time() < deadline:
        iteration += 1

        # ----- convolution of a with itself -----
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_c = conv_full.max()

        # ----- active set (indices within a tiny relative tolerance) -----
        tol = max_c * eps_rel
        active_idx = np.where(conv_full >= max_c - tol)[0]
        if active_idx.size == 0:               # fallback – should never happen
            active_idx = np.array([int(np.argmax(conv_full))])

        # ----- build weight mask (average over the active set) -----
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[active_idx] = 1.0 / float(active_idx.size)

        # ----- correlation a ⋆ w (g = 2 * a ⋆ w) -----
        fw = np.fft.rfft(w, n=L)
        # correlation via conj(FA) * FW
        corr = np.fft.irfft(np.conj(fa) * fw, n=L)[:n]   # linear correlation, first n entries
        g = 2.0 * corr

        # ----- exponentiated‑gradient step -----
        # diminishing step size (tuned constant)
        eta = 0.8 / math.sqrt(iteration)

        # to keep numbers stable, clip the exponent arguments
        max_exp = 50.0
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -max_exp, max_exp)
        a = a * np.exp(exp_arg)
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            # numerical trouble – fall back to the best known vector
            a = best_a.copy()
            s = a.sum()
        a = a / s

        # ----- occasional evaluation and logging -----
        if iteration % 200 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"[{time.time() - start_time:.1f}s] refine it {iteration}: "
                    f"new best {cur_score:.6f}"
                )
            # very cheap periodic check that we are not running out of time
            if time.time() > deadline - 1.0:
                break

    return best_a
