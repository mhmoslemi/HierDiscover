def compute_subgradient(a):
    n = len(a)
    fft_len = next_pow2(2*n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    max_val = conv.max()
    epsilon = 1e-12
    active = np.where(conv >= max_val - epsilon)[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        i_range = np.arange(i_start, i_end+1)
        j_range = k - i_range
        g[i_range] += 2.0 * a[j_range]   # double loop but vectorized
    # if double counting includes diagonal correctly: for each i=j, we still add 2*a[i]; which matches derivative of a_i^2 => 2*a_i.
    return conv, g
```

def simulated_annealing(a, time_left):
    start = time.time()
    deadline = start + time_left
    n = a.shape[0]
    cur_a = a.copy()
    conv_cur = compute_conv(cur_a)
    cur_score = 2 * n * conv_cur.max()
    best_a = cur_a.copy()
    best_score = cur_score
    temp_start = 1.0
    temp_end = 1e-4
    # define temperature schedule based on elapsed ratio.
    while time.time() < deadline:
        elapsed = time.time() - start
        t = elapsed / time_left
        temp = temp_start * (temp_end / temp_start) ** t
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * (np.random.rand() * 0.1) # up to 10% of value
        if delta <= 0:
            continue
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegative (i) may become zero
        # evaluate conv
        conv_new = compute_conv(a_new)
        new_score = 2 * n * conv_new.max()
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score)/temp):
            cur_a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score
```

"""
Propose a candidate sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm follows three phases:

1️⃣ **Broad random exploration** – generate many candidates from a
   rich collection of families (power‑law, exponential, two‑sided,
   β‑shapes, Gaussian, cosine, power‑law with offset).  Each candidate
   is normalised to sum = 1 and scored with a fast FFT‑based routine.
   The best few candidates are retained.

2️⃣ **Deterministic refinement** – starting from the retained candidates
   we apply two first‑order methods on the simplex:
   * Euclidean projected sub‑gradient descent with momentum.
   * Mirror descent (exponentiated‑gradient) on the probability simplex.
   Both use the exact sub‑gradient of the peak of the self‑convolution
   (the active indices of the maximal convolution entry) and run
   until the allocated time slice expires.

3️⃣ **Stochastic fine‑tuning** – a simple simulated‑annealing scheme
   that repeatedly transfers a small amount of mass between two random
   coordinates, accepting moves according to a temperature schedule.

The whole search respects the global time budget (default 1000 s).  The
final sequence is returned as a plain Python list; it is also verified
with the provided ``evaluate_sequence`` function before returning.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Cheap evaluation using FFT; assumes non‑negative and sum > 0."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_power_law_offset(n: int, p: float, c: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + c."""
    return generate_power_law(n, p) + c


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_sum_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^{-p_left} + w_right * (n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    a_i = (i+1)^{-alpha} * (n-i)^{-beta}.
    Symmetric β‑shape when alpha == beta.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (-alpha)
    right = (n - i + 1) ** (-beta)
    return left * right


def generate_gaussian(n: int, sigma_factor: float) -> np.ndarray:
    """Centered Gaussian (non‑negative).  sigma = sigma_factor * n."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    sigma = sigma_factor * n
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_raised_cosine(n: int) -> np.ndarray:
    """a_i = ½·(1 − cos(2πi/(n‑1))) – non‑negative, zero at the ends."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


# ----------------------------------------------------------------------
# Convolution + sub‑gradient utilities
# ----------------------------------------------------------------------


def compute_conv_and_subgrad(a: np.ndarray):
    """
    Returns
        conv  – self‑convolution (length 2n‑1)
        g     – sub‑gradient of the peak `max(conv)` w.r.t. `a`
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    # active indices (within a tiny tolerance)
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vectorised contribution for this k
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        g[i_range] += 2.0 * a[j_range]   # derivative = 2*a_{k-i}
    return conv, g


def compute_conv(a: np.ndarray):
    """Return self‑convolution only (FFT based)."""
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Descent / refinement methods
# ----------------------------------------------------------------------


def euclidean_descent(init_a: np.ndarray, time_budget: float):
    """Projected sub‑gradient descent with momentum (Euclidean)."""
    deadline = time.time() + time_budget
    a = init_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    conv, _ = compute_conv_and_subgrad(a)
    best_score = 2.0 * n * conv.max()

    v = np.zeros_like(a)
    mu = 0.9   # momentum coefficient
    it = 0
    while time.time() < deadline:
        it += 1
        conv, g = compute_conv_and_subgrad(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # diminishing step size
        eta = 1.0 / (np.sqrt(it) + 1.0)
        # heavy‑ball momentum on the gradient
        v = mu * v + eta * g
        a_candidate = project_simplex(a - v)
        a = a_candidate
    return best_a, best_score


def mirror_descent(init_a: np.ndarray, time_budget: float):
    """Exponentiated‑gradient (mirror) descent on the simplex."""
    deadline = time.time() + time_budget
    a = init_a.copy()
    n = a.shape[0]

    best_a = a.copy()
    conv, _ = compute_conv_and_subgrad(a)
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        it += 1
        conv, g = compute_conv_and_subgrad(a)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        eta = 1.0 / (np.sqrt(it) + 1.0)
        # multiplicative update in log‑space
        log_a = np.log(np.maximum(a, 1e-12))
        log_a -= eta * g
        a = np.exp(log_a)
        a /= a.sum()
    return best_a, best_score


def simulated_annealing(init_a: np.ndarray, time_budget: float):
    """Very simple annealing that transfers a small mass between two indices."""
    deadline = time.time() + time_budget
    a = init_a.copy()
    n = a.shape[0]

    conv = compute_conv(a)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score

    # linear temperature schedule
    T_start = 0.2
    T_end = 1e-6

    while time.time() < deadline:
        # propose a move
        i, j = random.sample(range(n), 2)
        if a[i] <= 0.0:
            continue
        # transfer a random fraction (up to 15%) of a[i] to a[j]
        frac = random.random() * 0.15
        delta = a[i] * frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv_new = compute_conv(a_new)
        new_score = 2.0 * n * conv_new.max()

        # compute temperature based on elapsed fraction
        elapsed = time.time() - (deadline - time_budget)
        frac_time = elapsed / time_budget
        T = T_start * (1.0 - frac_time) + T_end * frac_time

        # Metropolis acceptance
        if new_score < cur_score or random.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score

        "twosided_sum",
        "beta",
        "gauss",
        "cosine",
    ]

    # allocate roughly 70 % of total time for random search
    random_deadline = deadline - 0.3 * (deadline - start)

    while time.time() < random_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.30, 0.80)
            seq = generate_power_law(n, p)

        elif family == "power_offset":
            p = random.uniform(0.30, 0.80)
            c = random.uniform(0.0, 0.5)
            seq = generate_power_law_offset(n, p, c)

        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided_sum":
            p_l = random.uniform(0.30, 0.80)
            p_r = random.uniform(0.30, 0.80)
            w_l = random.uniform(0.1, 5.0)
            w_r = random.uniform(0.1, 5.0)
            seq = generate_two_sided_sum_power_law(n, p_l, p_r, w_l, w_r)

        elif family == "beta":
            a = random.uniform(0.30, 0.80)
            b = random.uniform(0.30, 0.80)
            seq = generate_beta_shape(n, a, b)

        elif family == "gauss":
            sigma_factor = random.uniform(0.05, 0.4)
            seq = generate_gaussian(n, sigma_factor)

        else:   # "cosine"
            seq = generate_raised_cosine(n)

        # normalise and guard against pathological sequences
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total < 1e-12:
            continue
        seq = seq / total

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[rand] NEW best ({family}) n={n} score={sc:.6f}")

        # keep a short list of top candidates
        if len(top_candidates) < TOP_K or sc < max(s for s, _ in top_candidates):
            top_candidates.append((sc, seq.copy()))
            top_candidates = sorted(top_candidates, key=lambda x: x[0])[:TOP_K]

    # -----------------------------------------------------------------
    # 2️⃣  Deterministic refinements (Euclidean, Mirror, Annealing)
    # -----------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        # allocate 60 % Euclidean, 30 % Mirror, 10 % Annealing
        eu_time = remaining * 0.60
        mi_time = remaining * 0.30
        an_time = remaining * 0.10

        # Euclidean descent on the overall best sequence
        print(f"[refine] Euclidean descent (time≈{eu_time:.1f}s)")
        a_eu, sc_eu = euclidean_descent(best_seq, eu_time)
        if sc_eu < best_score:
            best_score, best_seq = sc_eu, a_eu.copy()
            print(f"[euclid] improvement to {sc_eu:.6f}")

        # Mirror descent starting from the current best
        print(f"[refine] Mirror descent (time≈{mi_time:.1f}s)")
        a_mi, sc_mi = mirror_descent(best_seq, mi_time)
        if sc_mi < best_score:
            best_score, best_seq = sc_mi, a_mi.copy()
            print(f"[mirror] improvement to {sc_mi:.6f}")

        # Simulated annealing as a final polish
        print(f"[refine] Simulated annealing (time≈{an_time:.1f}s)")
        a_an, sc_an = simulated_annealing(best_seq, an_time)
        if sc_an < best_score:
            best_score, best_seq = sc_an, a_an.copy()
            print(f"[anneal] improvement to {sc_an:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final verification with the official checker
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Final score reported by evaluate_sequence = {final_score:.6f}")

    return final_seq
