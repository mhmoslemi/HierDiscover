def random_moves(a, n, L, phase, best_obj, deadline):
    while time.time() < deadline:
        # choose move type 0: mass transfer; 1: swap indices
        if np.random.rand() < 0.5:
            # swap
            i, j = np.random.choice(n, 2, replace=False)
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:
            # transfer
            i = np.random.choice(n, p=a)
            j = np.random.randint(n)
            delta = np.random.rand() * 0.03 * a[i]
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        # evaluate quickly via FFT
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[:2*n-1]
        obj_new = conv_new.max()
        if obj_new < best_obj - 1e-12:

"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a three‑stage search:

1.  **Broad parametric sweep** – generate many families of shapes
    (power‑law, exponential, symmetric beta‑distributions, random Dirichlet).
    For each shape we also try a few random permutations.  Every candidate is
    normalised to unit L1‑norm and its convolution peak is evaluated by an FFT.
    The best candidate (across a few different lengths) is kept.

2.  **Adam optimisation on a smooth‑max surrogate** – the true objective
    `max(conv)` is replaced by a soft‑max
    `log‑sum‑exp(α·conv)/α`.  The gradient of this surrogate is obtained
    efficiently with FFTs (correlation of the current vector with the
    soft‑max weights).  Adam with Euclidean projection onto the simplex is
    run for as many iterations as the remaining time permits.  The smoothing
    parameter `α` is increased from a small to a very large value, so the
    optimisation finally focuses on the true supremum norm.

3.  **Intensive local moves** – after the gradient phase we perform cheap
    hill‑climbing moves that either swap two entries or transfer a small
    amount of mass from a heavy entry to a random one.  Any improvement is
    kept.  This step helps to escape plateaus of the Adam phase.

All steps respect the global time budget (default 1000 s).  The returned
sequence is scaled so that `sum = sqrt(2·n)`, which is the normalisation used
by the reference code; the evaluation function is scale‑invariant, so any
positive scaling would be equivalent.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max_norm(a: np.ndarray, L: int) -> float:
    """
    Return the maximum value of the full convolution a*a,
    assuming a already sums to 1.  The FFT length L must be ≥ 2·len(a)-1.
    """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv.max()


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, L: int, phase: np.ndarray,
                   n: int, deadline: float,
                   max_iters: int = 30000) -> tuple[np.ndarray, float]:
    """
    Adam descent on the soft‑max approximation of max(conv).
    Returns the best vector seen and its (non‑smoothed) convolution max.
    """
    a = a0.copy()
    best_a = a.copy()
    best_obj = _conv_max_norm(a, L)

    # Adam hyper‑parameters
    lr0 = 0.07
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule – start gentle, end very sharp
    alpha_start, alpha_end = 2.0, 800.0

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # convolution of current a
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

        # soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        scaled = alpha * conv
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()               # length 2n‑1, sum = 1

        # gradient = 2 * correlation( a , w_rev )
        # w_rev is w reversed – achieved by multiplying the FFT of a with the
        # phase factor that implements a reversal.
        fft_w = np.fft.fft(w, n=L)
        fft_a_rev = np.conj(fft_a) * phase                 # reversed a in Fourier domain
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n - 1: n - 1 + n]                # extract the n relevant entries

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        obj_new = _conv_max_norm(a_new, L)
        if obj_new < best_obj:
            best_obj, best_a = obj_new, a_new.copy()
            print(f"    Adam it {it:5d}: improved obj={obj_new:.8e} score={2*n*obj_new:.6f}")

        a = a_new

    return best_a, best_obj


# ----------------------------------------------------------------------
def _random_local_search(a0: np.ndarray, L: int, n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing that combines swaps and small mass transfers.
    Returns the best vector seen and its convolution max.
    """
    a = a0.copy()
    best_obj = _conv_max_norm(a, L)

    # pre‑compute a distribution for source‑index sampling (proportional to mass)
    while time.time() < deadline:
        if np.random.rand() < 0.6:      # swap move (more frequent)
            i, j = np.random.choice(n, size=2, replace=False)
            a_new = a.copy()
            a_new[i], a_new[j] = a_new[j], a_new[i]
        else:                           # mass‑transfer move
            i = np.random.choice(n, p=a)    # heavy entries more likely
            j = np.random.randint(n)
            if i == j:
                continue
            delta = np.random.rand() * 0.03 * a[i]   # ≤ 3 % of source mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

        obj_new = _conv_max_norm(a_new, L)
        if obj_new < best_obj - 1e-12:           # strict improvement
            a, best_obj = a_new, obj_new
            print(f"    move improved: obj={obj_new:.8e} score={2*n*obj_new:.6f}")

    return a, best_obj

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects the given time budget (seconds) and returns a plain
    Python list of floats (scaled to sum = sqrt(2·n)).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin

    # ------------------------------------------------------------------
    # Choose a set of candidate lengths.
    # Prefer the length of a previously found construction if it exists.

        lengths = []

    # add a few hand‑picked lengths (must be modest to keep FFT cheap)
    lengths.extend([3000, 4500, 6000, 7500, 9000])
    # ensure uniqueness and sort
    lengths = sorted(set(lengths))

    best_global_seq = None
    best_global_obj = float("inf")
    best_global_n = None

    # ------------------------------------------------------------------
    # Main loop over the candidate lengths
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Length n = {n} ===")
        L = 1 << ((2 * n - 2).bit_length())          # FFT length ≥ 2n‑1
        phase = np.exp(2j * np.pi * np.arange(L) / L)   # phase factor for reversed FFT

        # --------------------------------------------------------------
        # 0.  Warm‑start: use the previous construction (if same length)
        if prev_norm is not None and len(prev_norm) == n:
            a_best = prev_norm.copy()
            obj = _conv_max_norm(a_best, L)
            best_seq_n = a_best
            best_obj_n = obj
            print(f"    warm‑start from previous: obj={obj:.8e} score={2*n*obj:.6f}")
        else:
            best_seq_n = None
            best_obj_n = float("inf")

        # --------------------------------------------------------------
        # 1.  Parametric sweep ------------------------------------------------
        def _maybe_update(vec):
            nonlocal best_seq_n, best_obj_n
            vec_norm = vec / vec.sum()
            obj = _conv_max_norm(vec_norm, L)
            if obj < best_obj_n:
                best_obj_n, best_seq_n = obj, vec_norm.copy()
                print(f"    new sweep best: obj={obj:.8e} score={2*n*obj:.6f}")

        # power‑law (both increasing and decreasing)
        exponents = np.linspace(0.0, 2.0, 21)
        for p in exponents:
            if time.time() > deadline:
                break
            base = np.arange(1, n + 1, dtype=np.float64) ** p
            _maybe_update(base)                 # increasing
            _maybe_update(base[::-1])           # decreasing

        # pure exponential decay (and its mirror)
        lambdas = np.linspace(0.0, 5.0, 21)
        for lam in lambdas:
            if time.time() > deadline:
                break
            exp_shape = np.exp(-lam * np.arange(n, dtype=np.float64))
            _maybe_update(exp_shape)            # decreasing
            _maybe_update(exp_shape[::-1])      # increasing

        # symmetric beta distributions (α = β)
        alphas = np.concatenate((
            np.linspace(0.2, 1.0, 9),   # U‑shaped to uniform
            np.linspace(1.0, 4.0, 7)[1:]  # uniform to mound‑shaped
        ))
        idx = np.arange(n, dtype=np.float64) + 0.5
        x = idx / n
        for a in alphas:
            if time.time() > deadline:
                break
            beta_shape = (x ** (a - 1)) * ((1 - x) ** (a - 1))
            # avoid NaNs when a < 1 (division by zero at the ends)
            beta_shape = np.nan_to_num(beta_shape, nan=0.0, posinf=0.0, neginf=0.0)
            if beta_shape.sum() == 0:
                continue
            _maybe_update(beta_shape)

        # random Dirichlet draws (moderate concentration)
        for _ in range(30):
            if time.time() > deadline:
                break
            rand_vec = np.random.exponential(scale=1.0, size=n)
            _maybe_update(rand_vec)

        # for each of the top‑few shapes also try a few random permutations
        # (permutations do not change the uniform case, but they affect all
        # non‑symmetric shapes).
        if best_seq_n is not None:
            for _ in range(12):
                if time.time() > deadline:
                    break
                perm = np.random.permutation(n)
                _maybe_update(best_seq_n[perm])

        # --------------------------------------------------------------
        # 2.  Adam optimisation starting from the best sweep candidate -----
        if best_seq_n is None:
            # fallback: start from a uniform vector
            best_seq_n = np.full(n, 1.0 / n)

        a_opt, obj_opt = _adam_optimize(best_seq_n, L, phase, n, deadline,
                                        max_iters=40000)
        if obj_opt < best_obj_n:
            best_seq_n, best_obj_n = a_opt, obj_opt

        # --------------------------------------------------------------
        # 3.  Intensive local random moves ---------------------------------
        print(f"    start local moves: obj={best_obj_n:.8e} score={2*n*best_obj_n:.6f}")
        a_final, obj_final = _random_local_search(best_seq_n, L, n, deadline)
        if obj_final < best_obj_n:
            best_seq_n, best_obj_n = a_final, obj_final

        # --------------------------------------------------------------
        # Keep the best across all lengths
        if best_obj_n < best_global_obj:
            best_global_obj, best_global_seq, best_global_n = best_obj_n, best_seq_n.copy(), n
            print(f">>> New global best (n={n}): obj={best_obj_n:.8e} score={2*n*best_obj_n:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the reference normalisation (sum = sqrt(2·n))
    if best_global_seq is None:

    return final_seq
