Exponential‑gradient descent on a smooth approximation of the max‑convolution.
The algorithm works on the probability simplex via a soft‑max (log‑it) parametrisation,
optimises a log‑sum‑exp surrogate of max {(a∗a)_k} with an annealing schedule

convolutions keep the inner loop lightweight, allowing many thousands of
iterations in the 1000 s budget.  The best sequence seen is returned.

import math
import numpy as np

def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i,
    normalised so that sum(a)=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * ((2.0*i-1.0)*(2.0*i)) / (4.0*i*i)
    a /= a.sum()
    return a

def _softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that maps y∈ℝⁿ to the probability simplex."""
    y_max = np.max(y)
    e = np.exp(y - y_max)
    return e / e.sum()

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the
    half‑order bound.  The routine uses an exponentiated‑gradient optimiser
    on a smooth approximation of the max‑convolution objective.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 1000))

    # Length needed for a linear correlation of a (len=n) with w (len=2n‑1)

    best_a = a0.copy()

    print(f"Finished after {it} iterations (best at {best_iter}), best score = {best_score:.6f}")
    # Convert to plain Python list of floats as required by the evaluator
    return best_seq.tolist()
