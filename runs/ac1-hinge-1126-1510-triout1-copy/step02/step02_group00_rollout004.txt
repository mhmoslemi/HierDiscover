"""
Propose a step‑function sequence that minimizes
    E(a) = 2·n·max(a*a) / (sum a)^2
using a three‑stage optimisation:

1. Stochastic L‑BFGS on a smooth log‑sum‑exp approximation of the
   max‑convolution, with many random restarts and brief post‑L‑BFGS
   hill‑climbs.

2. Simulated‑annealing on the full convolution (exact max) that moves
   small amounts of mass between positions while keeping the sum fixed.
   Incremental O(n) updates of the convolution are used, allowing
   millions of cheap proposals.

3. A final FFT‑based greedy hill‑climb to polish the best candidate.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found (list of non‑negative floats).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float           – 2·n·smooth_max(conv)
    grad_y : np.ndarray – gradient w.r.t. y (shape n)
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c            # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max (optional)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (FFT‑based)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000):
    """
    Greedy mass‑transfer hill‑climb that keeps sum = 1.
    Moves a small random fraction of mass between two positions.
    Accepts only improvements.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the exact max‑convolution
# ----------------------------------------------------------------------
def simulated_annealing(a_start: np.ndarray,
                        conv_start: np.ndarray,
                        n: int,
                        deadline: float):
    """
    Incremental SA that moves small amounts of probability mass.
    Maintains an exact convolution array (size 2·n‑1) and updates it
    in O(n) per move.
    Returns the best sequence seen.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_c = conv_start.copy()
    cur_max = cur_c.max()
    best_a = cur_a.copy()
    best_c = cur_c.copy()
    best_max = cur_max

    accept_cnt = 0
    iter_cnt = 0
    start_time = time.time()
    total_budget = deadline - start_time

    while time.time() < deadline:
        # linear schedule for temperature and step size
        elapsed = time.time() - start_time
        frac = 1.0 - elapsed / total_budget   # 1 → 0
        # temperature is a fraction of the current max
        T = cur_max * max(1e-12, 0.05 * frac)
        # step size (max fraction of source mass to move)
        step_factor = 0.3 * frac + 0.01   # starts ~0.31, ends ~0.01

        # pick source index weighted by its mass
        src = rng.choice(n, p=cur_a / cur_a.sum())
        if cur_a[src] <= 0.0:
            continue
        # pick a different target uniformly
        tgt = rng.integers(0, n)
        while tgt == src:
            tgt = rng.integers(0, n)

        delta = cur_a[src] * rng.random() * step_factor
        if delta <= 0.0:
            continue

        # ---------- incremental convolution update ----------
        a_old = cur_a
        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[tgt] += delta

        new_c = cur_c.copy()
        # linear part
        new_c[src:src + n] -= 2.0 * delta * a_old
        new_c[tgt:tgt + n] += 2.0 * delta * a_old
        # quadratic part
        new_c[2 * src] += delta * delta
        new_c[2 * tgt] += delta * delta
        new_c[src + tgt] -= 2.0 * delta * delta

        new_max = new_c.max()
        delta_obj = new_max - cur_max

        # acceptance test
        accept = False
        if delta_obj <= 0.0:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp(-delta_obj / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a = new_a
            cur_c = new_c
            cur_max = new_max
            accept_cnt += 1
            if cur_max < best_max:
                best_max = cur_max
                best_a = cur_a.copy()
                best_c = cur_c.copy()

        # periodic re‑normalisation via FFT to wipe out accumulated error
        if accept_cnt % 4000 == 0 and accept_cnt > 0:
            cur_c = full_conv(cur_a, next_pow2(2 * n - 1))
            cur_max = cur_c.max()

        iter_cnt += 1

    # Return best observed candidate (as probability vector)
    return best_a, best_c, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initial distribution
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            max_n = 4000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            init_seq[init_seq < 0] = 0.0
            if init_seq.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
        else:
            n = 4000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)

    # Pre‑computed FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    # ------------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS + brief hill‑climbs
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # allocate ~70 % of the budget to the L‑BFGS phase
    phase1_deadline = start_time + budget_s * 0.70

    rng = np.random.default_rng(seed)
    iteration = 0
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04]   # decreasing smoothing

    while time.time() < phase1_deadline:
        lam = lam_schedule[iteration % len(lam_schedule)]

        # choose a starting point for the L‑BFGS run
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy heuristic
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # run L‑BFGS on the smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250,
                                'gtol': 1e-10,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] L‑BFGS lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(phase1_deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing on the exact max
    # ------------------------------------------------------------------
    # recompute exact convolution for the current best_a
    cur_conv = full_conv(best_a, fft_len_conv)
    phase2_deadline = start_time + budget_s * 0.95
    print(f"[phase2] starting SA; current max_convolution={cur_conv.max():.8f}")

    best_a, _, best_max = simulated_annealing(best_a, cur_conv,
                                              n, phase2_deadline)

    best_score = float(2.0 * n * best_max)
    print(f"[phase2] SA finished best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polishing hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    final_deadline,
                                    max_iter=5000)

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
