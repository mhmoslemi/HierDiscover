def adam_optimize(y0, n, fft_len, lam_factor, steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    for t in range(1, steps+1):
        f, grad = objective_and_grad(y, n, fft_len, lam_factor)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return y

"""
Optimise a non‑negative step sequence to minimise the evaluation score.
The method combines:
  * a low‑dimensional Adam optimisation of the smooth‑max objective on the soft‑max logits,
  * a decreasing temperature schedule (smooth‑max width),
  * aggressive targeted hill‑climbing that moves mass away from the convolution entry that is currently maximal,
  * many random restarts and simple perturbations.
All heavy lifting uses only NumPy, no external solvers.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector (sum==1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using an FFT of length fft_len."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

# Smooth‑max objective and Adam optimiser
# ----------------------------------------------------------------------


def smooth_max_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                        lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Compute 2·n·smooth_max(conv) and its gradient w.r.t. the unconstrained logits y.

    ``lam_factor`` multiplies the current maximum convolution value to obtain the
    smoothing width ``lam``.
    """
    # probability vector
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max via log‑sum‑exp
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    loss = 2.0 * n * smooth_max

    # ----- gradient of smooth‑max w.r.t. conv -----
    w = exp_shift / Z                 # (∂smooth_max)/(∂conv)

    # ----- back‑propagation through the convolution -----
    # we need g_i = Σ_j a_j * w_{i+j},  i = 0..n‑1
    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.fft.ifft(fa_rev * fw).real
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return loss, grad_y


def adam_optimize(y0: np.ndarray, n: int, fft_len_conv: int,
                  lam_factor: float, steps: int,
                  lr: float = 0.01,
                  beta1: float = 0.9, beta2: float = 0.999,
                  eps: float = 1e-8) -> np.ndarray:
    """Simple Adam optimiser for the smooth‑max objective."""
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    for t in range(1, steps + 1):
        loss, grad = smooth_max_and_grad(y, n, fft_len_conv, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y

def targeted_hill_climb(a: np.ndarray, cur_score: float,
                        fft_len: int, n: int,
                        deadline: float,
                        max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly steals a small fraction of mass from the
    index that contributes most to the currently maximal convolution entry.
    The move is kept only if it strictly lowers the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current convolution and its worst index
        conv = conv_fft(best, fft_len, n)
        s = int(np.argmax(conv))

        # indices (i, j) with i+j == s
        i_min = max(0, s - (n - 1))
        i_max = min(n - 1, s)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = s - idx_i

        # contribution = a[i] * a[j] (ignoring the factor 2 for i!=j,
        # the ordering does not affect which pair is largest)
        prod = best[idx_i] * best[idx_j]

        # pick the pair with biggest product – it is the “heaviest” contributor
        k = int(np.argmax(prod))
        i_src = int(idx_i[k])
        j_src = int(idx_j[k])

        # decide from which of the two we will move mass
        src = i_src if best[i_src] >= best[j_src] else j_src

        # pick a random destination different from src
        tgt = rng.integers(0, n)
        if tgt == src:
            continue

        # move a small fraction of the mass at src to tgt
        delta = best[src] * rng.random() * 0.05  # up to 5 % of source mass
        if delta <= 0.0:
            continue

        new_a = best.copy()
        new_a[src] -= delta
        new_a[tgt] += delta

        # evaluate the move using the cheap fast_score
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score - 1e-12:
            best = new_a
            best_score = new_score
            # keep the improvement – no need to recompute conv now, it will be
            # recomputed at the top of the next loop iteration
        # otherwise discard the move and continue

    return best, best_score
