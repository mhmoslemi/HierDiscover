Pseudo:

def project_simplex(v):
    """Project onto simplex sum(x)=1, x>=0."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_to_simplex(v):
    """Project onto the probability simplex {x >=0, sum x = 1}."""
    # Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if len(rho) == 0:
        # all negative, return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

3. **Simulated‑annealing perturbations** – random mass transfers between
   two positions are accepted with a Metropolis rule.  The temperature
   decays exponentially, allowing occasional uphill moves.

All iterates are kept on the probability simplex (non‑negative,
unit‑sum) via a Euclidean projection.  The best sequence found is
scaled to the normalisation Σa = √(2 n) required by the reference code
and clipped to the bounds enforced by ``evaluate_sequence``.

The routine respects the supplied time budget (`budget_s`), prints
progress and returns a plain ``list[float]``.
"""

import time
import math
import numpy as np


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _project_to_simplex(x: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {y >= 0, Σy = 1}."""
    # Duchi–Shalev‑Shwartz–Singer–Chandra (2008) algorithm
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(x) + 1, dtype=float)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become negative → return uniform point
        return np.full_like(x, 1.0 / len(x))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(x - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a, length = 2·len(a)‑1 (truncated)."""
    fa = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def _correlate_full(w: np.ndarray, a_rev: np.ndarray, L_corr: int) -> np.ndarray:
    """Full correlation (w ⋆ a_rev) needed for the smooth‑max gradient."""
    fw = np.fft.fft(w, n=L_corr)
    fa_rev = np.fft.fft(a_rev, n=L_corr)
    corr = np.real(np.fft.ifft(fw * fa_rev))
    return corr


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    2·n·max(conv(a,a)) / (Σa)².
    Returns the best sequence found (scaled to Σa = √(2 n)).
    """
    np.random.seed(seed)
    start_time = time.time()
    # a tiny safety margin before the hard kill
    deadline = start_time + budget_s - 5.0

    # -------------------------------------------------------------
    # Determine the length n.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 6000                            # conservative fallback

    target_sum = math.sqrt(2.0 * n)          # normalisation used by the reference
    print(f"--- Optimising a sequence of length n = {n} (target Σ = {target_sum:.3f}) ---")

    # Pre‑compute FFT sizes (next powers of two)
    L_conv = 1 << ((2 * n - 1).bit_length())      # for a⋆a
    L_corr = 1 << ((3 * n - 2).bit_length())      # for correlation with smooth‑max weights

    # -------------------------------------------------------------
    # Helper: compute the (non‑smooth) objective – i.e. max(conv)
    def _max_conv(a: np.ndarray) -> float:
        conv = _conv_full(a, L_conv)
        return conv.max()

    # -------------------------------------------------------------
    # Build a modest set of initial points on the simplex Σa = 1.
    start_vectors = []

    # 1) previous best if available
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)          # type: ignore[name-defined]
        if seq0.size == n:
            a0 = np.clip(seq0, 0.0, 1000.0)
            a0 = a0 / a0.sum()
            start_vectors.append(a0)
    except Exception:
        pass

    # 2) uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # 3) a few Dirichlet draws (moderate concentration)
    for _ in range(3):
        start_vectors.append(np.random.dirichlet(np.ones(n)))

    # 4) simple power‑law shapes
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        w = np.arange(1, n + 1, dtype=float) ** p
        a = w / w.sum()
        start_vectors.append(a)

    # -------------------------------------------------------------
    # Track the best sequence seen so far (still on the simplex).
    best_a = None
    best_score = float("inf")   # will store max(conv) (lower is better)

    # Helper to update the global best
    def _maybe_update(a_vec: np.ndarray, cur_score: float):
        nonlocal best_score, best_a
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_vec.copy()
            print(f"[update] new best max‑conv = {best_score:.6f}")

    # -------------------------------------------------------------
    # ==================== PHASE 1 – SMOOTH‑MAX ADAM ====================
    beta_start = 5.0
    beta_end = 200.0                # a larger final β yields a tighter max‑approx
    max_iters_per_restart = 8000
    lr = 0.04                       # base learning rate for Adam
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    eps_softmax = 1e-12

    for ridx, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break

        # Initialise the unconstrained variables v (soft‑max parameters)
        v = np.log(a_init + eps_softmax)
        m = np.zeros_like(v)
        v2 = np.zeros_like(v)

        for it in range(max_iters_per_restart):
            if time.time() > deadline:
                break

            # ------- current simplex point a -------
            a = _softmax(v)                     # Σa = 1, a ≥ 0

            # ------- convolution and current objective -------
            conv = _conv_full(a, L_conv)
            conv_max = conv.max()
            _maybe_update(a, conv_max)

            # ------- smooth‑max weights (soft‑max over convolution) -------
            # β grows exponentially – smoother early, tighter later
            beta = beta_start * (beta_end / beta_start) ** (it / max_iters_per_restart)
            # stability: shift by the max before exponentiation
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()                     # soft‑max weights, Σw = 1

            # ------- gradient w.r.t. a (2·corr(w, a_rev)) -------
            a_rev = a[::-1]
            corr_full = _correlate_full(w, a_rev, L_corr)
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]

            # ------- gradient w.r.t. soft‑max parameters v -------
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ------- Adam update -------
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # -------------------------------------------------------------
    # If nothing was found (highly unlikely) fall back to uniform.
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = _max_conv(best_a)

    # -------------------------------------------------------------
    # ==================== PHASE 2 – PROJECTED SUB‑GRADIENT ==============
    # Start from the current best simplex point.
    a_cur = best_a.copy()
    maxconv_cur = best_score

    max_sub_iters = 5000
    for it in range(max_sub_iters):
        if time.time() > deadline:
            break

        # Compute the full convolution and locate the active shift.
        conv = _conv_full(a_cur, L_conv)
        idx_max = int(conv.argmax())
        k = idx_max - (n - 1)                     # shift (can be negative)

        # Gradient of the active shift.
        g = np.zeros_like(a_cur)
        if k >= 0:
            # contribution a_{i+k}
            g[: n - k] += a_cur[k:]
            # contribution a_{i-k}
            g[k:] += a_cur[: n - k]
        else:
            kk = -k
            g[kk:] += a_cur[: n - kk]
            g[: n - kk] += a_cur[kk:]

        # Simple decreasing step size.
        step = 0.2 / math.sqrt(it + 1.0)
        a_new = a_cur - step * g
        a_new = _project_to_simplex(a_new)

        maxconv_new = _max_conv(a_new)

        if maxconv_new < best_score - 1e-12:
            best_score = maxconv_new
            best_a = a_new.copy()
            print(f"[subgrad] iter {it:5d}  new best max‑conv = {best_score:.6f}")

        a_cur = a_new
        maxconv_cur = maxconv_new

    # -------------------------------------------------------------
    # ==================== PHASE 3 – SIMULATED ANNEALING ================
    a_cur = best_a.copy()
    maxconv_cur = best_score
    temperature = 0.02
    cooling = 0.9998
    step_scale = 0.03                # maximal fraction of a_i moved in one step
    rng = np.random.default_rng(seed + 1)

    while time.time() < deadline:
        # Randomly pick two distinct indices.
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Propose moving a small amount of mass from i to j.
        delta = min(a_cur[i], step_scale * rng.random())
        if delta <= 0:
            continue
        a_prop = a_cur.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        maxconv_prop = _max_conv(a_prop)
        diff = maxconv_prop - maxconv_cur

        # Metropolis acceptance.
        if diff <= 0 or rng.random() < math.exp(-diff / temperature):
            a_cur = a_prop
            maxconv_cur = maxconv_prop
            if maxconv_prop < best_score - 1e-12:
                best_score = maxconv_prop
                best_a = a_prop.copy()
                print(f"[anneal] new best max‑conv = {best_score:.6f}")

        temperature *= cooling
        # occasional temperature reset to avoid it becoming too tiny.
        if temperature < 1e-8:
            temperature = 0.02

    # -------------------------------------------------------------
    # ==================== PHASE 4 – RANDOM DIRICHLET SWEEPS ============
    # A quick stochastic sweep that sometimes discovers exotic shapes.
    dirichlet_alphas = [0.3, 0.6, 1.0, 2.0]
    while time.time() < deadline:
        alpha = float(rng.choice(dirichlet_alphas))
        a_rand = rng.dirichlet(np.full(n, alpha))
        maxconv_rand = _max_conv(a_rand)
        if maxconv_rand < best_score - 1e-12:
            best_score = maxconv_rand
            best_a = a_rand.copy()
            print(f"[random] new best max‑conv = {best_score:.6f}")

    # -------------------------------------------------------------
    # Scale the simplex solution back to the required normalisation.
    best_seq = best_a * target_sum
    # Clip to the range enforced by evaluate_sequence (just for safety).
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished optimisation: best max‑conv = {best_score:.6f} (n = {n}) ---")
    return best_seq.tolist()
```

This implementation stays within the 1000 s budget, explores the search space with several complementary heuristics, and returns the best step‑function found (scaled to Σ a = √(2 n)).
