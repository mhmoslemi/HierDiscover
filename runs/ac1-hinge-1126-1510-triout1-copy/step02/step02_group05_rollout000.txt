def compute_max_conv(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A*A, fft_len)[:2*n-1]
    return conv.max()

"""
Hybrid projected sub‑gradient descent + stochastic local refinement.

The score to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Because the expression is homogeneous we keep the vector a on the
probability simplex (a_i ≥ 0, Σ a_i = 1).  The core optimiser is a
projected sub‑gradient method that follows the gradient of the presently
active peak of the self‑convolution.  After a descent run we apply a
light simulated‑annealing style local search that moves a tiny amount of
mass between two coordinates; this often escapes shallow local minima.
Multiple restarts with different seeds, lengths and initial shapes
(power‑law, uniform, perturbed best) are performed until the time budget
expires.  The best sequence ever observed is returned as a plain Python
list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the full linear convolution a*a and its maximum entry.
    Uses a power‑of‑two FFT length for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper that returns only the maximal convolution entry."""
    _, mx = compute_conv_and_max(a)
    return mx


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the simplex
# ----------------------------------------------------------------------
def subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run projected sub‑gradient descent on the simplex.

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (will be normalised to sum‑one).
    max_iters : int
        Upper bound on iterations.
    step0 : float
        Base step‑size (scaled by 1/√iteration).
    deadline : float
        Unix timestamp after which the routine must stop.

    Returns
    -------
    best_seq : np.ndarray
        Best sequence encountered.
    best_score : float
        Corresponding evaluation score.
    """
    a = a0.copy()
    n = a.shape[0]
    a = a / a.sum()                       # enforce Σa = 1

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # full convolution and its peak
        conv, max_b = compute_conv_and_max(a)

        # score = 2 * n * max_b  (since Σa = 1)
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # occasional progress report
            print(f"[subgrad it {it}] new best={score:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient of the *active* peak
        # --------------------------------------------------------------
        k = int(np.argmax(conv))           # index where the peak occurs
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        # grad_i = 2 * a_{k-i} for all i that contribute
        grad[idx] = 2.0 * a[k - idx]

        # diminishing step size (standard sub‑gradient schedule)
        step = step0 / np.sqrt(it)

        # gradient step followed by simplex projection
        a = project_simplex(a - step * grad)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a cheap random walk that moves a small amount of mass between
    random coordinates, accepting improvements (or occasional uphill moves).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02                      # initial acceptance temperature

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # Choose a direction: move a fraction of mass from src → dst
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # Amount to move – up to 5 % of the source mass
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        # Numerical safety
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        # Accept if better, otherwise with Metropolis probability
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # Cool down slowly
        temperature *= 0.9999
        temperature = max(temperature, 1e-6)

    return best_seq, best_score
