    # function to run LP refinement:
    def lp_refine(a, max_iters=10, t_initial=0.5):
        n = a.size
        # Precompute convolution matrix A (Toeplitz) using SciPy perhaps.
        c = np.concatenate([a, np.zeros(n-1)])
        r = np.concatenate([a[0:1], np.zeros(n-1)])  # first row
        # Build A as dense? Use np.zeros maybe high memory; use scipy.linalg.toeplitz
        # We'll use dense to create quickly; then convert to sparse.

        A = scipy.linalg.toeplitz(c, r)  # shape (2n-1, n)
        # We'll cast as float64.
        # Since we will solve with same A each iteration maybe we should rebuild each iteration as a changes.
        # That's heavy; but we only need A for each iteration.

        # Acquire b initial maybe uniform.
        b = a.copy()
        best_local = a.copy()
        best_local_score = fast_score(a)

        t = t_initial

        for it in range(max_iters):
            # compute self-convolution of a and get M
            conv_a = np.fft.ifft(np.fft.fft(a, n=2*n-2) * np.fft.fft(a, n=2*n-2)).real
            M = conv_a.max()

            # Setup linprog: maximize sum(b) => minimize -sum(b)
            c_obj = -np.ones(n)
            # inequality: A @ b <= M
            bounds = [(0, None)] * n
            # Solve
            res = scipy.optimize.linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
            if not res.success:
                # fallback: break
                break

            b_sol = res.x
            # Normalize to sum 1
            if b_sol.sum() == 0:
                b_norm = np.full(n, 1.0 / n)
            else:
                b_norm = b_sol / b_sol.sum()

            # mixture
            a = (1 - t) * a + t * b_norm

            # optionally reduce t
            t *= 0.9

            # evaluate score
            sc = fast_score(a)
            if sc < best_local_score:
                best_local_score = sc
                best_local = a.copy()
        return best_local, best_local_score

    # For each length, generate candidate families and refine.

    while time.time() < deadline - 5:
        # sample random length and random family
        n = random.choice(lengths)
        # choose family

            seq = generate_power_law(n, p)
        elif family_type == 'exp':
            lam = random.uniform(1e-4, 2e-2)
            seq = generate_exponential(n, lam)
        elif family_type == 'twosided':
            p_l = random.uniform(0.4, 0.7)
            p_r = random.uniform(0.4, 0.7)
            w_l = random.uniform(0.3, 3.0)
            w_r = random.uniform(0.3, 3.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)
        elif family_type == 'gauss':

            x = np.arange(n)
            center = (n-1)/2
            seq = np.exp(-0.5 * ((x - center)/sigma)**2)
        else:  # triangular
            peak = random.uniform(0.2, 0.8)
            seq = np.concatenate([np.linspace(0, 1, int(peak*n), endpoint=False),
                                  np.linspace(1, 0, n - int(peak*n))])

        # normalize to sum 1
        seq = np.maximum(seq, 0)
        if seq.sum() == 0:
            continue
        seq = seq / seq.sum()

        # initial quick evaluate
        sc0 = fast_score(seq)
        if sc0 < best_score:
            best_score = sc0
            best_seq = seq.copy()
            print(f"[{time.time()-start:.1f}s] init candidate {family_type} n={n} score={sc0:.6f}")

        if time.time() < deadline - 30:  # ensure enough time

                print(f"[{time.time()-start:.1f}s] LP refined {family_type} n={n} score={refined_sc:.6f}")

        # optionally run subgradient refinement further if time left and seq improved

    # final evaluation
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"Finished. Best final score = {final_score:.6f}")
    return final_seq

"""
Propose a candidate step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm has three phases:

1️⃣  *Diverse random families* – generate many candidates from several
    parameterised families (power‑law, exponential, two‑sided power‑law,
    Gaussian, triangular) with lengths up to ~5000.  Keep the best
    element seen.

2️⃣  *Focused parameter hill‑climbing* – for the two‑sided power‑law
    family perform a lightweight coordinate‑descent on the four
    parameters (two exponents and two weights) on a few different
    lengths.  The hill‑climber accepts any improvement and shrinks the
    step‑size when progress stalls.

3️⃣  *Adaptive sub‑gradient refinement* – starting from the best vector,
    repeatedly move against the sub‑gradient of the current peak of the
    self‑convolution and project onto the probability simplex.
    A simple back‑tracking line search chooses the step‑size, and
    occasional small random jitters help escape plateaux.

All work respects a 1000 s budget; the final sequence is checked with the
provided ``evaluate_sequence`` function before being returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – required by the task)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quickly evaluate the objective using an FFT convolution.
    ``seq`` must be a 1‑D non‑negative array that sums to 1 (scaling does not
    affect the score, but the normalisation makes the value comparable).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # normalise – not strictly necessary but keeps numbers stable
    seq = seq / s
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.real.max())
    return float(2.0 * n * max_conv)


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    mask = u - cssv / ind > 0
    if not np.any(mask):
        return np.full_like(v, 1.0 / n)
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^{-p_left} + w_right * (n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Discrete symmetric Gaussian centred in the middle of the vector."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    return np.exp(-0.5 * ((x - centre) / sigma) ** 2)


def generate_triangular(n: int, peak_frac: float) -> np.ndarray:
    """
    Simple triangular shape.
    ``peak_frac`` is the fraction of the length at which the peak occurs.
    """
    k = max(1, int(round(peak_frac * n)))
    left = np.linspace(0, 1, k, endpoint=False)
    right = np.linspace(1, 0, n - k)
    return np.concatenate([left, right])

                max(1e-6, w_right + random.uniform(-step_w, step_w)),
            )
            seq = generate_two_sided_power_law(n, *cand)
            seq = np.maximum(seq, 0.0)
            tot = seq.sum()
            if tot < 1e-12:
                continue
            seq = seq / tot
            sc = fast_score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                p_left, p_right, w_left, w_right = cand
                print(f"[{time.time() - start_time:.1f}s] hill‑climb improve n={n} score={sc:.6f}")
                # slowly shrink steps after a success
                step_p *= 0.99
                step_w *= 0.99
            else:
                # if we have gone a while without improvement, tighten steps
                if it % 100 == 0:
                    step_p *= 0.98
                    step_w *= 0.98

    # Run the hill‑climber on a few lengths while we have time
    while time.time() < deadline - 180:
        n_hc = random.choice([1200, 1600, 2000, 2500, 3000])
        hill_climb_one_length(n_hc, deadline - 180)

    # --------------------------------------------------------------
    # 3️⃣  Adaptive sub‑gradient refinement on the best candidate
    # --------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    iter_count = 0

    while time.time() < deadline - 5:
        iter_count += 1

        # self‑convolution via FFT
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        max_idx = int(conv.argmax())

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            j_range = max_idx - i_range
            g[i_range] = 2.0 * a[j_range]

        # back‑tracking line search for a suitable step size
        eta = 1.0
        improved = False
        while eta > 1e-7:
            cand = project_simplex(a - eta * g)
            sc = fast_score(cand)
            if sc < best_score - 1e-12:
                a = cand
                best_score = sc
                best_seq = a.copy()
                improved = True
                if iter_count % 50 == 0:
                    print(f"[{time.time() - start_time:.1f}s] iter {iter_count} step {eta:.2e} score {sc:.6f}")
                break
            eta *= 0.5

        if not improved:
            # tiny move to keep the dynamics alive
            a = project_simplex(a - 0.1 * g)

        # occasional jitter to escape flat plateaux
        if iter_count % 1000 == 0 and time.time() < deadline - 10:
            jitter = np.random.normal(scale=0.004, size=n)
            a = project_simplex(a + jitter)
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.time() - start_time:.1f}s] jitter improvement score {sc:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)          # type: ignore[name-defined]
    print(f"Finished. Best final score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
