"""
Propose a step‑function (non‑negative sequence) that minimises the
self‑convolution supremum.

Algorithm
---------
1. Initialise a set of promising candidates.
   * Half‑order series (C(2i,i)/4**i) – the classic start.
   * Its reversal, the uniform vector and many convex mixtures of the three.
   * Power‑law tilts and Beta‑distribution shapes (i+1)^{α‑1}(n‑i)^{β‑1}.
   All candidates are simplex‑projected and the best (lowest 2·n·max (a*a))
   is kept.

2. Refine the best candidate with a smooth‑max gradient descent.
   * The true objective is  max_k (a*a)_k .  We approximate the max by a
     soft‑max  M_γ = (1/γ)·log Σ exp(γ·c_k)  where c = a*a.
   * The gradient of M_γ w.r.t. a is
         ∇M_γ = 2·(p * a_rev)   (convolution),
     with p_k = exp(γ·c_k)/Σ exp(γ·c) and a_rev = a[::-1].
   * A projected gradient step is taken on the simplex; a simple back‑track
     line‑search guarantees a non‑increase of the true max‑convolution.
   * The temperature γ is increased exponentially from a small value to a
     large one, so the algorithm first follows a smooth landscape and then
     focusses on the exact supremum.

3. Occasionally perform a stochastic “pair‑exchange’’ move
   (transfer a small random amount of mass between two positions) to
   escape shallow plateaux.

The whole search runs for the allotted time budget (default 1000 s)
and returns the best sequence found.  All numbers are non‑negative and
sum to 1, so the score equals 2·n·max(conv(a,a)).
"""

import time
import math
import numpy as np


def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights (length n) that minimises
    `evaluate_sequence`.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) Parameters
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1000))
    total_budget = deadline - start_time

    # ------------------------------------------------------------------
    # 2) Generate a diverse pool of initial candidates
    # ------------------------------------------------------------------
    base = _half_order_series(n)               # classic half‑order series
    rev = base[::-1].copy()                    # its reversal
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    candidates = [base, rev, uniform]

    # convex mixes of the three
    for w1 in np.linspace(0.0, 1.0, 11):
        for w2 in np.linspace(0.0, 1.0 - w1, 11):
            w3 = 1.0 - w1 - w2
            mix = w1 * base + w2 * rev + w3 * uniform
            mix /= mix.sum()
            candidates.append(mix)

    # power‑law tilts of the half‑order series (delta sweep)
    for delta in np.linspace(0.0, 0.6, 13):
        tilt = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = base * tilt
        cand /= cand.sum()
        candidates.append(cand)

    # Beta‑shape candidates: a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    for _ in range(150):
        alpha = np.random.uniform(0.2, 2.5)
        beta = np.random.uniform(0.2, 2.5)
        idx = np.arange(1, n + 1, dtype=np.float64)
        cand = (idx ** (alpha - 1.0)) * ((n + 1 - idx) ** (beta - 1.0))
        # protect against overflow / infinities
        cand = np.nan_to_num(cand, nan=0.0, posinf=0.0, neginf=0.0)
        if cand.sum() == 0.0:
            continue
        cand /= cand.sum()
        candidates.append(cand)

    # a few random Dirichlet draws
    for _ in range(200):
        alphas = np.random.rand(n) + 0.05
        cand = np.random.gamma(alphas, 1.0)
        if cand.sum() == 0.0:
            continue
        cand /= cand.sum()
        candidates.append(cand)

    # ------------------------------------------------------------------
    # 3) Pick the best initial candidate (cheap score evaluation)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for cand in candidates:
        conv = np.convolve(cand, cand)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
    # sanity: ensure we have a sequence
    if best_seq is None:
        best_seq = uniform.copy()
        best_score = 2.0 * n * np.convolve(best_seq, best_seq).max()

    print(f"[init] best initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Gradient‑descent refinement with soft‑max smoothing
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_conv = np.convolve(cur_seq, cur_seq)
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max

    # hyper‑parameters for the outer loop
    iteration = 0
    step = 0.08               # initial projected‑gradient step size
    step_min = 1e-7
    step_max = 0.5
    fail_cnt = 0              # number of consecutive non‑improving steps

    # schedule for the soft‑max temperature (γ)
    gamma_start = 0.2
    gamma_end = 200.0

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        progress = elapsed / total_budget

        # ------------------------------------------------------------------
        # 4a) soft‑max temperature (γ) – slowly increase to focus on true max
        # ------------------------------------------------------------------
        gamma = gamma_start * ((gamma_end / gamma_start) ** progress)

        # recompute convolution for the current sequence
        conv = np.convolve(cur_seq, cur_seq)
        cur_max = conv.max()

        # ------------------------------------------------------------------
        # 4b) soft‑max distribution over convolution entries
        # ------------------------------------------------------------------
        exps = gamma * conv
        # numeric stabilisation
        exps_shift = exps - np.max(exps)
        p = np.exp(exps_shift)
        p_sum = p.sum()
        if p_sum == 0.0:
            p = np.full_like(p, 1.0 / p.shape[0])
        else:
            p /= p_sum

        # ------------------------------------------------------------------
        # 4c) gradient of the smooth max (see analysis)
        # ------------------------------------------------------------------
        a_rev = cur_seq[::-1]
        full = np.convolve(p, a_rev)               # length = 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]          # slice to length n

        # ------------------------------------------------------------------
        # 4d) back‑tracking line‑search on the true objective (max‐conv)
        # ------------------------------------------------------------------
        accepted = False
        trial_step = step
        for _ in range(5):
            cand = cur_seq - trial_step * grad
            cand = _project_simplex(cand)
            cand_conv = np.convolve(cand, cand)
            cand_max = cand_conv.max()
            if cand_max < cur_max - 1e-14:          # strict improvement
                cur_seq = cand
                cur_max = cand_max
                cur_score = 2.0 * n * cur_max
                step = min(step * 1.1, step_max)
                accepted = True
                fail_cnt = 0
                break
            trial_step *= 0.5

        if not accepted:
            # no improvement – shrink step and possibly jitter later
            step = max(step * 0.7, step_min)
            fail_cnt += 1

        # ------------------------------------------------------------------
        # 4e) occasional stochastic pair‑exchange to escape plateaux
        # ------------------------------------------------------------------
        if fail_cnt > 30 and (iteration % 5 == 0):
            i, j = np.random.choice(n, 2, replace=False)
            move = min(cur_seq[i], 0.05) * np.random.rand()
            cand_seq = cur_seq.copy()
            cand_seq[i] -= move
            cand_seq[j] += move
            cand_seq = _project_simplex(cand_seq)   # ensure exact simplex
            cand_max = np.convolve(cand_seq, cand_seq).max()
            if cand_max < cur_max:
                cur_seq = cand_seq
                cur_max = cand_max
                cur_score = 2.0 * n * cur_max
                fail_cnt = 0

        # ------------------------------------------------------------------
        # 4f) keep the best ever seen
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{iteration:6d}] new best score = {best_score:.6f}")

        # periodic status update (lightweight)
        if iteration % 2000 == 0:
            print(f"[{iteration:6d}] best so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence in plain Python list form
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
