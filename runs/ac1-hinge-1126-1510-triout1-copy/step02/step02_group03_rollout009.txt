"""
Propose a low‑score step function.

The algorithm minimises a smooth approximation of the peak autocorrelation
max (a∗a) using projected / exponentiated gradient descent on the simplex.
A log‑sum‑exp (soft‑max) surrogate of the max is employed; the temperature
β is increased (annealed) so the optimisation gradually focuses on the true
infinity‑norm.  Convolutions and correlations are computed with FFTs for
speed, allowing thousands of iterations on sequences of length ≈1500
within the 1000 s budget.  Several random restarts are performed and a final
polishing phase swaps small amounts of mass between two positions.
The best sequence found is scaled to the [0, 1000] range required by the
evaluator.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear convolution a*a using FFT (real input → real output)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself via FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two >= 2n‑1
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: smooth max (log‑sum‑exp) of the convolution and its gradient.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(a: np.ndarray, beta: float):
    """
    Compute a smooth approximation of max(a*a) and the gradient w.r.t. a.

    Returns
    -------
    smooth_val : float
        (1/β) * log Σ exp(β * (a*a)[k]).
    grad : np.ndarray, shape (n,)
        ∂smooth_val/∂a.
    conv : np.ndarray, shape (2*n‑1,)
        The full convolution a*a (needed for diagnostics only).
    """
    n = a.shape[0]

    # 1) convolution a*a
    conv = _conv_fft(a)                     # length 2n‑1

    # 2) soft‑max weights w_k = exp(β conv_k) / Σ exp(β conv)
    bc = beta * conv
    max_bc = np.max(bc)
    exp_bc = np.exp(bc - max_bc)            # subtract max for stability
    Z = np.sum(exp_bc)
    w = exp_bc / Z                          # Σ w = 1

    # 3) gradient = 2 * correlation(a, w)
    # correlation (a ⋆ w)[i] = Σ_j a_j * w_{i+j}
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())   # ≥ 3n‑2
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w, size)
    corr_full = np.fft.irfft(A * np.conj(W), size)
    offset = (2 * n - 1) - 1                    # index where shift i=0 starts
    grad = 2.0 * corr_full[offset: offset + n]

    # 4) smooth value (not strictly needed for the optimiser)
    smooth_val = (max_bc + np.log(Z)) / beta

    return smooth_val, grad, conv

# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step function using a smoothed max‑convolution
    objective, exponential‑gradient updates on the simplex, and a polishing
    phase.  Returns a list of heights in the range [0, 1000].
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a small safety margin

    # ------------------------------------------------------------------
    # Determine the length of the sequence (reuse previous solution if present)
    # ------------------------------------------------------------------
    seq0 = globals().get('height_sequence_1')
    if seq0 is not None:
        n = len(seq0)
        a0 = np.array(seq0, dtype=float)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() == 0.0:
            a0 = np.ones_like(a0)
        a0 = a0 / a0.sum()
    else:
        n = 1542
        a0 = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    beta_schedule = [0.5, 2.0, 8.0, 32.0, 128.0, 512.0, 2048.0]
    max_iter_per_beta = 1500                # iterations for each temperature
    step0_base = 0.2                        # base step size for exponentiated updates
    restart = 0

        # ----- Local polishing (pairwise mass swaps) -------------------------
        if best_seq is not None:

                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # stop if time is almost exhausted
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        a_fallback = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        best_seq = a_fallback / a_fallback.sum()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
