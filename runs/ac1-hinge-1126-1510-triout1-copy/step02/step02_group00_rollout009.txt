def hill_climb(a, max_iter, deadline):
    n = len(a)
    best = a.copy()
    best_score = fast_evaluation(best)
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j: continue
        delta = best[i] * rng.random() * 0.1  # up to 10% of i's mass
        if delta == 0: continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_evaluation(best)
        if new_score < best_score:
            best_score = new_score
            # keep
        else:
            best[i] += delta
            best[j] -= delta
        max_iter -= 1
    return best, best_score
```

def lp_refine(a, n, fft_len_conv, deadline, best_score):
    from scipy.sparse import csr_matrix
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    best_local = a.copy()
    best_local_score = best_score
    # limit number of LP iterations
    iter_lp = 0
    while time.time() < deadline and iter_lp < 20:
        conv = np.convolve(a, a)
        M = conv.max()
        # Build sparse matrix A_ub (2n-1) x n
        rows = 2 * n - 1
        data = []
        row_ind = []
        col_ind = []
        for k in range(rows):
            start_j = max(0, k - (n - 1))
            end_j = min(n - 1, k)
            # this includes indices j where k-j in [0,n-1]
            js = np.arange(start_j, end_j + 1)
            f_vals = a[k - js]
            data.extend(f_vals)
            row_ind.extend([k] * len(js))
            col_ind.extend(js.tolist())
        A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
        b_ub = np.full(rows, M, dtype=np.float64)

        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=(0, None), method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g
        # Update a with small step
        t = 0.2
        a = (1 - t) * a + t * g
        a = np.maximum(a, eps)
        a /= a.sum()
        # evaluate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local = a.copy()
            print(f"[LP iter {iter_lp}] score={best_local_score:.6f}")
        iter_lp += 1
    return best_local, best_local_score

  1. Stochastic L‑BFGS on a log‑sum‑exp smooth approximation of the
     max‑convolution (as in the reference implementation).  Many
     random restarts with a decreasing smoothing parameter explore the
     non‑convex landscape.
  2. A linear‑programming refinement step that implements the
     “alternating projection” idea from the problem statement:
        max(g)  s.t.  f * g ≤ M  (M = max(f * f))
     The LP‑solution g is normalised to the simplex and blended with
     the current f.  Repeating this step drives the peak of the
     self‑convolution down further.
  3. Final polishing by a projected sub‑gradient method and a fast
     greedy hill‑climb.

All phases respect the 1000 s time budget, keep the sequence
non‑negative and normalised to sum = 1 (scale invariance makes this a
safe normalisation).  The best sequence found is returned as a plain
list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps ℝⁿ onto the probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the numerator 2·n·max(a*a) for a distribution that
    already sums to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp and its gradient w.r.t. y.
    Returns (objective value, gradient).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a simple sub‑gradient method.
    Each iterate is projected onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        # convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: 2·(mask * a_rev) restricted to original n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep best iterate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps the simplex)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy transfer of a small fraction of mass between two coordinates.
    Accepts only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 8 % of the mass currently at i
        delta = best[i] * rng.random() * 0.08
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# LP‑based refinement (alternating projection)
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              n: int,
              fft_len_conv: int,
              deadline: float,
              current_best_score: float):
    """
    Perform a few iterations of the LP‑based alternating‑projection step.
    Each iteration solves:
        max Σ g_j   s.t.   f * g ≤ M   ,   g ≥ 0
    where M = max(f * f).  The solution g is normalised to the simplex
    and blended with the current f.
    """
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()
    best_a = a.copy()
    best_score = current_best_score
    it = 0
    max_lp_iters = 15

    while time.time() < deadline and it < max_lp_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build sparse matrix for constraints: (f * g)[k] ≤ M
        rows = 2 * n - 1
        data = []
        row_ind = []
        col_ind = []
        for k in range(rows):
            start = max(0, k - (n - 1))
            end = min(n - 1, k)
            js = np.arange(start, end + 1, dtype=int)
            vals = a[k - js]
            data.extend(vals)
            row_ind.extend([k] * len(js))
            col_ind.extend(js.tolist())
        A_ub = csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
        b_ub = np.full(rows, M, dtype=np.float64)

        # maximise Σ g_j  ⇒  minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=(0, None), method='highs')
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g  # project to simplex

        # convex blend (step size decreasing with iterations)
        t = 0.4 / np.sqrt(it + 1.0)
        a = (1 - t) * a + t * g
        a = np.maximum(a, eps)
        a /= a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[LP {it}] improved fast_score={best_score:.6f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  The function respects the supplied time budget and
    returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # small safety margin

    # --------------------------------------------------------------
    # Choose problem size and an initial distribution
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        max_n = 2000
        if init_seq.size > max_n:
            init_seq = init_seq[:max_n]
        n = init_seq.shape[0]
        init_seq[init_seq < 0] = 0.0
        if init_seq.sum() == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = init_seq / init_seq.sum()
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # pre‑computed FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS on a smoothed max
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    iteration = 0
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04]   # decreasing smoothing

    while time.time() < deadline:
        lam = lam_schedule[iteration % len(lam_schedule)]

        # choose a starting point for L‑BFGS
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:          # edge‑heavy heuristic
            edge_frac = rng.uniform(0.01, 0.04)
            m_edge = max(1, int(edge_frac * n))
            a_center = 1.0 / np.sqrt(2.0) / n
            a_edge_val = np.sqrt(n / 2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge_val
            a_tmp[-m_edge:] = a_edge_val
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # L‑BFGS optimisation
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] L‑BFGS lam={lam:.3f} mode={mode} "
                  f"score={best_score:.6f}")

        # short hill‑climb to polish this L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

        iteration += 1

        # occasional sub‑gradient refinement
        if iteration % 15 == 0 and time.time() < deadline - 5.0:
            sg_deadline = min(deadline, time.time() + 5.0)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[{iteration}] sub‑gradient refine score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – LP‑based alternating projection (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        lp_deadline = deadline - 2.0
        best_a, best_score = lp_refine(best_a, n, fft_len_conv,
                                       lp_deadline, best_score)
        print(f"[LP] after refinement fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final polishing with an extended hill‑climb
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    final_deadline,
                                    max_iter=3000)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
