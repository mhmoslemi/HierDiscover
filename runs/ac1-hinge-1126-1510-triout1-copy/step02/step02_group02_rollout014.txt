"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The routine works in four stages:

1. **Initialisation** – use the previous global solution if present,
   otherwise start from a discrete arcsine (U‑shaped) distribution
   (the continuous arcsine law has a *flat* self‑convolution).

2. **Broad parametric sweep** – generate many simple families
   (power‑law, exponential, symmetric power‑law, beta‑distributions,
   the arcsine shape) normalised to unit sum and keep the best one.

3. **Adam descent on a smooth approximation of the max**
   – the `max` is replaced by a soft‑max  
   `w = softmax(α·conv)`.  The gradient is
   `2 * correlation(w, a)`.  α is increased from a small value to a
   large one (≈ 500) while the learning‑rate is slowly decayed.
   After each step the vector is projected onto the probability simplex.

4. **Fine‑grained sub‑gradient descent on the true max** – after the
   soft‑max phase we switch to the exact sub‑gradient of the
   supremum norm.  The active set (indices where the convolution attains
   its current maximum) is used to compute a cheap gradient via FFT.
   A simple back‑tracking step‑size scheme is employed.

5. **Local random moves** – cheap hill‑climbing moves that transfer a
   small fraction of mass from a heavy entry to a random one.

All stages respect a 1000 s budget, the best sequence found is finally
scaled so that `sum = sqrt(2·n)` (the evaluation function is scale
invariant).  The function prints progress information and returns a
plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)-1) using FFT of length L.
    Returns a real array of length 2·n-1.
    """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_i = exp(α·conv_i) / Σ exp(α·conv_i).
    The subtraction of the maximum improves numerical stability.
    """
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


# ----------------------------------------------------------------------
def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                           L: int, phase: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ w_i * conv_i  w.r.t. a:
        grad = 2 * correlation(w, a)
    Correlation is computed via a single FFT using a pre‑computed phase factor.
    Returns a vector of length n.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_rev_a = np.conj(fft_a) * phase               # FFT of a reversed (zero‑padded)
    fft_w = np.fft.fft(w, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev_a))
    # the relevant slice is n‑1 … n‑1+n‑1
    return 2.0 * corr[n - 1: n - 1 + n]


# ----------------------------------------------------------------------
def _subgradient_active_set(a: np.ndarray, active: np.ndarray,
                            L: int, phase: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the supremum norm of the convolution.
    `active` is a boolean mask of length 2·n‑1 selecting the indices
    where the convolution attains its maximum (within a tiny tolerance).
    """
    n = a.shape[0]
    s = np.zeros(2 * n - 1, dtype=np.float64)
    s[active] = 1.0
    fft_a = np.fft.fft(a, n=L)
    fft_rev_a = np.conj(fft_a) * phase
    fft_s = np.fft.fft(s, n=L)
    corr = np.real(np.fft.ifft(fft_s * fft_rev_a))
    return 2.0 * corr[n - 1: n - 1 + n]


# ----------------------------------------------------------------------
def _evaluate_and_update(a_raw: np.ndarray, n: int, L: int,
                         best_seq: np.ndarray, best_obj: float) -> tuple:
    """
    Normalise a_raw to unit sum, compute max(conv) and keep the better
    of the current best and the candidate.
    Returns possibly updated (best_seq, best_obj).
    """
    if a_raw.sum() == 0:
        return best_seq, best_obj
    a = a_raw / a_raw.sum()
    conv = _conv_full(a, L)
    obj = conv.max()
    if obj < best_obj:
        best_obj, best_seq = obj, a.copy()
        print(f"  new best | obj={obj:.9f} | score={2 * n * obj:.6f}")
    return best_seq, best_obj

    a = best_seq.copy()                # start from the best parametric candidate
    # recompute best_obj to be sure (might have changed during sweep)
    best_obj = _conv_full(a, L).max()
    best_seq = a.copy()

    max_iters = 40000
    lr0 = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha_start, alpha_end = 4.0, 500.0   # smoothing schedule (log‑linear)

    print(f"--- Phase 2: Adam descent (max_iters≈{max_iters}) ---")
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # current convolution
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

        # soft‑max weights
        alpha = alpha_start * (alpha_end / alpha_start) ** (it / max_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth objective
        grad = _gradient_from_weights(a, w, L, phase)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (simple decay)
        lr = lr0 * (1.0 - it / max_iters) ** 0.5
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate new point
        conv_new = np.real(np.fft.ifft(np.fft.fft(a_new, n=L) *
                                       np.fft.fft(a_new, n=L)))[: 2 * n - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj, best_seq = obj_new, a_new.copy()
            print(f"Iter {it:5d}: improved | obj={obj_new:.9f} | score={2*n*obj_new:.6f}")

        a = a_new

    # --------------------------------------------------------------
    # 3. Sub‑gradient descent on the true max‑norm.
    print("--- Phase 3: sub‑gradient descent on true max ---")
    a_current = best_seq.copy()
    # step‑size for the sub‑gradient phase
    sub_lr = 0.15
    sub_iter = 0
    while time.time() < deadline:
        sub_iter += 1
        conv = _conv_full(a_current, L)
        M = conv.max()
        # active set: indices with value within 1e‑12 of the maximum
        tol = 1e-12
        active_mask = conv >= (M - tol)

        # compute sub‑gradient using the active set
        grad = _subgradient_active_set(a_current, active_mask, L, phase)

        # try a simple back‑tracking line‑search
        step = sub_lr / math.sqrt(sub_iter)
        accepted = False
        for _ in range(5):
            a_new = a_current - step * grad
            a_new = _project_simplex(a_new)
            obj_new = _conv_full(a_new, L).max()
            if obj_new < best_obj:
                best_obj, best_seq, a_current = obj_new, a_new.copy(), a_new
                print(f"  sub‑grad iter {sub_iter:5d} improved | obj={obj_new:.9f} | score={2*n*obj_new:.6f}")
                # slightly increase step size for next iteration
                sub_lr = min(sub_lr * 1.2, 1.0)
                accepted = True
                break
            else:
                step *= 0.5
        if not accepted:
            # if none of the trial steps improved, shrink learning rate
            sub_lr *= 0.7
        # stop early if learning rate becomes tiny
        if sub_lr < 1e-5:
            break

    # --------------------------------------------------------------
    # 4. Cheap random local moves (hill‑climbing).
    print("--- Phase 4: random local moves ---")
    a_current = best_seq.copy()
    while time.time() < deadline:
        i = np.random.choice(n, p=a_current)                 # source (biased)
        j = np.random.randint(0, n)                          # destination
        if i == j:
            continue
        delta = np.random.rand() * 0.02 * a_current[i]      # up to 2 % transfer
        if delta <= 0.0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        obj_new = _conv_full(a_new, L).max()
        if obj_new < best_obj:
            best_obj, best_seq, a_current = obj_new, a_new.copy(), a_new
            print(f"  move improved | obj={obj_new:.9f} | score={2*n*obj_new:.6f}")

    # --------------------------------------------------------------
    # Final scaling to the reference normalisation (sum = sqrt(2·n))
    final_seq = (best_seq * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)   # just for the final log line
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) after {it} Adam iters ---")
    return final_seq
