def compute_gradient(a, w):
    n = len(a)
    L = 1 << ((3*n - 2).bit_length())
    A = np.fft.rfft(a, n=L)
    w_rev = w[::-1]

    grad = 2.0 * grad_slice
    return grad

"""
Improved search for a step‑function with a low self‑convolution peak.

The algorithm is a *smoothed projected gradient descent* on the simplex.
At each iteration we compute the full self‑convolution b = a*a,
form a soft‑max weight vector w that concentrates on the large entries of b,
and take a gradient step w.r.t. the smoothed objective

    φ(a) = ε·log Σ_i exp(b_i/ε),

whose gradient is a weighted sum of the individual convolution gradients.
After the step we project back onto the probability simplex (non‑negative
entries summing to one).  The step size decays as η_t = η₀/√t.
Multiple random restarts (uniform, power‑law, perturbed best) are performed
until the time budget expires; the best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (O(n) Duchi et al. 2008)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, real‑valued)
# ----------------------------------------------------------------------


def full_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """
    Return the linear (non‑circular) convolution a*a.
    The result has length 2*len(a)-1.
    """
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[: 2 * a.shape[0] - 1]


def gradient_wrt_softmax(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Compute ∇_a  Σ_i w_i (a*a)_i  efficiently via FFT.
    The gradient has length len(a).
    """
    n = a.shape[0]
    # w_rev = reverse(w)  (length 2n-1)
    w_rev = w[::-1]
    A = np.fft.rfft(a, n=L)
    W = np.fft.rfft(w_rev, n=L)
    conv = np.fft.irfft(A * W, n=L)[: 3 * n - 2]   # length 3n-2
    # required slice: indices n-1 .. 2n-2 (inclusive), then reverse
    slice_part = conv[n - 1 : 2 * n - 1][::-1]
    return 2.0 * slice_part


# ----------------------------------------------------------------------
# Core optimiser (smoothed projected gradient descent)
# ----------------------------------------------------------------------


def smoothed_descent(
    a0: np.ndarray,
    max_iters: int,
    eta0: float,
    eps_factor: float,
    eps_min: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run a monotone smoothed projected gradient descent on the simplex.

    Parameters
    ----------
    a0 : np.ndarray
        Starting distribution (will be normalised to sum 1).
    max_iters : int
        Upper bound on iteration count for this restart.
    eta0 : float
        Initial step size (scaled by 1/√t).
    eps_factor : float
        Relative factor to compute the smoothing parameter ε = eps_factor * max(b).
    eps_min : float
        Minimum ε to avoid division by zero.
    deadline : float
        Unix time after which the routine must stop.

    Returns
    -------
    best_a : np.ndarray
        The best sequence encountered during the descent.
    best_score : float
        Corresponding value of evaluate_sequence (i.e. 2·n·max_conv).
    """
    a = np.copy(a0)
    a = np.maximum(a, 1e-15)   # avoid exact zeros for numerical stability
    a = a / a.sum()            # enforce Σ a_i = 1
    n = a.shape[0]

    # FFT size large enough for the longest convolution (a * w_rev)
    L = 1 << ((3 * n - 2).bit_length())

    # initial convolution and score
    b = full_convolution(a, L)
    max_b = b.max()
    cur_score = 2.0 * n * max_b

    best_a = a.copy()
    best_score = cur_score

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- smoothing ----------
        eps = max(max_b * eps_factor, eps_min)

        # soft‑max weights w_i ∝ exp((b_i - max_b)/eps)
        # subtraction of max_b ensures the largest entry gets weight 1.
        w_exp = np.exp((b - max_b) / eps)
        w = w_exp / w_exp.sum()

        # ---------- gradient ----------
        grad = gradient_wrt_softmax(a, w, L)

        # ---------- step ----------
        step = eta0 / np.sqrt(it)
        a_candidate = project_simplex(a - step * grad)

        # ---------- evaluate candidate ----------
        b_cand = full_convolution(a_candidate, L)
        max_b_cand = b_cand.max()
        cand_score = 2.0 * n * max_b_cand

        # accept only if we made progress (monotone descent)
        if cand_score <= cur_score:
            a = a_candidate
            b = b_cand
            max_b = max_b_cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                print(f"[iter {it}] new best score={cand_score:.6f}")
        else:
            # try a reduced step size (up to three halvings)
            improved = False
            step_tmp = step * 0.5
            for _ in range(3):
                a_tmp = project_simplex(a - step_tmp * grad)
                b_tmp = full_convolution(a_tmp, L)
                max_b_tmp = b_tmp.max()
                tmp_score = 2.0 * n * max_b_tmp
                if tmp_score <= cur_score:
                    a = a_tmp
                    b = b_tmp
                    max_b = max_b_tmp
                    cur_score = tmp_score
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_a = a.copy()
                        print(f"[iter {it}] new best (after step backtrack) score={tmp_score:.6f}")
                    improved = True
                    break
                step_tmp *= 0.5
            if not improved:
                # No improvement – keep the current a (no update)
                pass

    return best_a, best_score

                a0 /= a0.sum()

        # ----- allocate iteration budget for this restart ------------------------
        time_left = deadline - time.time()
        # conservative estimate: ~0.003 s per iteration for n≈8000
        it_budget = int(min(max_iters_per_restart, max(1, time_left / 0.003)))
        if it_budget <= 0:
            break

        # ----- run the optimiser ------------------------------------------------
        seq, score = smoothed_descent(
            a0,
            it_budget,
            eta0,
            eps_factor,
            eps_min,
            deadline,
        )

        # ----- update global best -----------------------------------------------
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(
                f"[restart {restart}] new global best score={score:.6f} "
                f"(iters={it_budget})"
            )

        # slowly decay the step‑size parameter if we see no improvement
        if restart % 7 == 0:
            eta0 = max(0.3, eta0 * 0.9)

    # ----- safety net -------------------------------------------------------------
    if global_best_seq is None:
        # simple power‑law fallback
        idx = np.arange(1, n + 1, dtype=np.float64)
        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()

    # Convert to plain Python list as required by the external harness.
    return global_best_seq.tolist()
