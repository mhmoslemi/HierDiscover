"""
Propose a low‑score non‑negative step sequence.

The score is  E = 2·n·max(a*a) / (sum a)² .
Because the scale cancels we may restrict the search to the probability simplex
(sum = 1).  The optimizer works in three stages:

1.  Randomly initialise a soft‑max parameter vector `y` (a = softmax(y)).
    For a decreasing schedule of smooth‑max parameters `lam` we run a few
    L‑BFGS optimisations (trust‑region on a log‑sum‑exp approximation of the
    maximum).  Warm‑starting with the previous `y` implements a continuation
    method that drives the solution towards the true non‑smooth problem.

2.  A short projected sub‑gradient refinement (the true max is used, not the
    smoothed version).  The sub‑gradient is obtained analytically via FFT
    convolutions and the iterate is projected back onto the simplex.

3.  A greedy hill‑climb that repeatedly transfers a small fraction of mass
    between two coordinates, accepting only moves that improve the fast
    score.  The climb is run many times and the best solution is symmetrised
    (a ← (a + rev(a))/2) before the final evaluation.

All phases run until the global time‑budget (`budget_s`) expires – the
algorithm adapts the amount of work allocated to each phase so that it makes
full use of the ~1000 s limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) using FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) under the constraint sum(a)=1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen for non‑negative v)
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    # Convolution a_rev * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5          # initial step‑size multiplier
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active (binding) indices – tolerance avoids tiny gaps
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to the original n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 5 % of the mass at i (smaller steps near the end)
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Shape initialisers (used for diverse random restarts)
# ----------------------------------------------------------------------
def initialise_shape(shape: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalized non‑negative vector of length n according to ``shape``."""
    if shape == "uniform":
        a = np.full(n, 1.0 / n)
    elif shape == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
    elif shape == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
    elif shape == "sin":
        # sin on (0,π) – zero at boundaries, positive inside
        x = np.arange(n) + 0.5
        a = np.sin(np.pi * x / n)
        a = np.clip(a, 0, None)
        a /= a.sum()
    elif shape == "cos":
        # cos shifted to be non‑negative
        x = np.arange(n) + 0.5
        a = np.cos(np.pi * (x - n / 2) / n) + 1.0
        a = np.clip(a, 0, None)
        a /= a.sum()
    elif shape == "quad":
        # symmetric quadratic hump, zero at ends
        i = np.arange(n)
        a = (i + 1) * (n - i)
        a = np.clip(a, 0, None)
        a /= a.sum()
    elif shape == "tri":
        # simple triangle (increase then decrease)
        half = n // 2
        left = np.linspace(1, 2, half, endpoint=False)
        right = left[::-1] if n % 2 == 0 else np.append(left, [2])[:half]
        a = np.concatenate([left, right])
        a /= a.sum()
    elif shape == "random":
        a = rng.random(n)
        a /= a.sum()
    else:   # fallback to random
        a = rng.random(n)
        a /= a.sum()
    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the global time budget (≈1000 s) and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5   # safety margin for final I/O

    # ------------------------------------------------------------------
    # Choose problem size and an initial distribution
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            max_n = 4000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            init_seq[init_seq < 0] = 0.0
            if init_seq.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
        else:
            n = 4000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parameter schedules
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    shape_pool = ['uniform', 'exp', 'power', 'sin', 'cos',
                  'quad', 'tri', 'random', 'prev']

    iteration = 0
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Pick a start shape and initialise soft‑max parameters y0
        # --------------------------------------------------------------
        mode = rng.choice(shape_pool)
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        else:
            a_start = initialise_shape(mode, n, rng)
            y0 = np.log(a_start + 1e-12)

        # --------------------------------------------------------------
        # 2) Continuation L‑BFGS across decreasing smooth‑max parameters
        # --------------------------------------------------------------
        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            # L‑BFGS on the smooth objective
            res = minimize(fun=objective_and_grad,
                           x0=y0,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 250,
                                    'gtol': 1e-10,
                                    'disp': False})
            y0 = res.x                                   # warm‑start next lam
            a_opt = softmax(y0)
            # occasional symmetrisation (helps reducing the peak)
            a_opt = (a_opt + a_opt[::-1]) * 0.5
            a_opt /= a_opt.sum()                         # re‑normalise after symmetrisation
            score_opt = fast_score(a_opt, fft_len_conv, n)

            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt.copy()
                print(f"[{iteration}] L‑BFGS lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # --------------------------------------------------------------
        # 3) Brief sub‑gradient refinement (few seconds)
        # --------------------------------------------------------------
        if time.time() < deadline - 5.0:
            sub_deadline = min(deadline, time.time() + 3.0)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sub_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[{iteration}] sub‑gradient refined score={best_score:.6f}")

        # --------------------------------------------------------------
        # 4) Hill‑climb polishing
        # --------------------------------------------------------------
        if time.time() < deadline - 2.0:
            hc_deadline = min(deadline, time.time() + 1.5)
            best_a, best_score = hill_climb(best_a,
                                            best_score,
                                            fft_len_conv,
                                            n,
                                            hc_deadline,
                                            max_iter=2500)
            # optional symmetrisation after hill‑climb
            best_a = (best_a + best_a[::-1]) * 0.5
            best_a /= best_a.sum()
            # recompute exact score after symmetrisation
            best_score = fast_score(best_a, fft_len_conv, n)
            print(f"[{iteration}] hill‑climb polished score={best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final intensive polishing (use remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        # Longer sub‑gradient phase
        sub_deadline = time.time() + remaining * 0.6
        best_a, best_score = subgradient_optimize(best_a,
                                                  sub_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[final‑sub] refined score={best_score:.6f}")

        # Final hill‑climb with many iterations
        hc_deadline = time.time() + remaining * 0.35
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=6000)
        best_a = (best_a + best_a[::-1]) * 0.5
        best_a /= best_a.sum()
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑hill] polished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best found sequence (list of floats)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # evaluate with the supplied black‑box to obtain the official score
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated external score = {final_eval:.6f}")
    return final_seq
