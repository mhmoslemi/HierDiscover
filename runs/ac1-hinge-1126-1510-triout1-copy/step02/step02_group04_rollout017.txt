def proj_simplex(v):
    # from Duchi et al 2008

import time
import math
import numpy as np

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()

def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length L (>=2n-1)."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[:2 * a.size - 1]
    return conv

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex {x >=0, sum(x)=1}."""
    # Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho
    mask = u - cssv / ind > 0
    if not np.any(mask):
        # all zero
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(mask)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed: int = 123, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that (approximately) minimises
    2·n·max(conv(a,a)) / (sum(a)**2).  The optimisation works directly on the
    simplex using a projected Adam update based on the sub‑gradient of the
    (non‑smooth) max‑convolution.  Several diverse initialisations are tried
    and the best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin before the hard kill
    deadline = start_time + budget_s - 5.0

    # Choose length of the sequence.  Use the length of the previous
    # construction if it exists, otherwise fall back to a slightly larger
    # size (8000) – longer sequences often give a lower bound but keep the
    # FFT size reasonable.
    try:
        n = len(height_sequence_1)   # type: ignore
        print(f"Using length of previous construction: n={n}")
    except Exception:
        n = 8000
        print(f"No previous construction, using default length n={n}")

    target_sum = math.sqrt(2.0 * n)      # normalisation used by the reference
    # ------------------------------------------------------------------
    # Pre‑compute FFT length – next power of two >= 2n‑1
    L_conv = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    def _objective(a: np.ndarray) -> float:
        """Return 2·n·max(conv(a,a)) (a already sums to 1)."""
        conv = _conv_full(a, L_conv)
        return 2.0 * n * conv.max()

    # Build a set of starting points on the simplex
    start_vectors = []

        a0 = seq0 / seq0.sum()

    for lam in (0.0005, 0.001, 0.002, 0.005):

            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1, dtype=int)
            grad_a = np.zeros_like(a)
            # grad_i = 2 * a_{k-i} for all valid i
            grad_a[idx] = 2.0 * a[k - idx]

            # ---------- (optional) blend with a smoothed max ----------
            # A soft‑max weight vector w (temperature beta) gives a smoother
            # direction early in the run and becomes one‑hot at the end.
            # The blend is linear: w = (1‑alpha)·soft + alpha·hard, where
            # alpha grows with the iteration count.
            beta = beta_start + (beta_end - beta_start) * it / max_iters_per_restart
            # soft weights
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()
            # blend coefficient (linear from 0 to 1)
            alpha = it / max_iters_per_restart
            w = (1.0 - alpha) * w + alpha * (np.eye(1, 2 * n - 1, k, dtype=float).ravel())

            # The smoothed gradient would be 2 * correlation(w, a_rev);
            # we keep the hard sub‑gradient and add a tiny contribution from the
            # smoothed one to avoid getting stuck on flat plateaus.
            if alpha > 0.0:  # when we have any soft component
                # compute correlation of w with reversed a using FFT (cost similar to a conv)
                L_corr = 1 << ((3 * n - 2).bit_length())
                # we reuse the same routine but with different inputs
                fft_w = np.fft.fft(w, n=L_corr)
                fft_rev = np.fft.fft(a[::-1], n=L_corr)
                corr_full = np.real(np.fft.ifft(fft_w * fft_rev))
                corr = corr_full[n - 1 : n - 1 + n]
                grad_a = (1.0 - alpha) * grad_a + alpha * (2.0 * corr)

            # ---------- gradient w.r.t. soft‑max parameters v ----------
            dot = float(np.dot(grad_a, a))
            grad_v = a * (grad_a - dot)

            # ---------- Adam update ----------
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            # Adaptive learning rate – we slowly decay it to aid convergence.
            lr = base_lr / (1.0 + 1e-4 * it)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

            # ---------- occasional random local perturbation ----------
            # Every 500 iterations try a cheap random move that swaps a tiny
            # amount of mass between two coordinates. Accept only if it
            # improves the objective.
            if it % 500 == 0 and it > 0:
                # make a copy to try the move
                a_try = a.copy()
                # pick two distinct indices
                i, j = np.random.choice(n, size=2, replace=False)
                # transfer δ = η * a_i where η is a small fraction
                delta = 0.01 * a_try[i]   # 1 % of the source mass
                if delta > 0:
                    a_try[i] -= delta
                    a_try[j] += delta
                    # re‑normalise – the move preserves the sum, only numerical errors
                    a_try = a_try / a_try.sum()
                    # evaluate
                    conv_try = _conv_full(a_try, L_conv)
                    cur_score_try = 2.0 * n * conv_try.max()
                    if cur_score_try < cur_score:
                        # Accept the move – update v so that softmax(v) ≈ a_try
                        # Simple way: replace v by the log‑odds of a_try.
                        v = np.log(a_try + eps)
                        # Reset Adam moments to avoid stale direction after the jump
                        m.fill(0.0)
                        v2.fill(0.0)

        # End of this restart
        # If we have exhausted time after this restart, break out.
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # If nothing was found (should not happen) fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Clip to the range enforced by evaluate_sequence (just to be safe)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished optimisation: best score = {best_score:.6f} (n={n}) ---")
    return best_seq.tolist()

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, guaranteeing a ≥ 0 and Σa = 1."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) using FFT of length L."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv
