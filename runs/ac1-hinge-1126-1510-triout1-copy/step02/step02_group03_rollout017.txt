def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Mirror‑descent optimizer for the autocorrelation‑peak objective.

The objective (with sum(a)=1) is   score = 2·n·max_k (a*a)[k] .
It is a convex problem on the probability simplex.
Instead of the Euclidean projection used in the reference implementation,
this version uses an *exponential‑gradient / mirror‑descent* update:

    a_{new} ∝ a ∘ exp( -η·g )      (∘ = elementwise product)

where g is a sub‑gradient of the peak (the average sub‑gradient over all
active indices).  This update stays in the simplex automatically (normalise
by the ℓ₁‑norm) and often makes larger, more focused steps than a plain
projected sub‑gradient descent.

The optimiser runs many short random restarts, each consisting of a few
thousand iterations.  After every restart a tiny “polishing’’ phase swaps
small amounts of mass between two positions to escape shallow plateaus.
If a global variable `height_sequence_1` exists the previous best length is
re‑used, otherwise the default length 1542 is taken.

The returned sequence is scaled to the interval [0,1000]; this scaling does
not affect the score because the evaluation function is homogeneous.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex (kept for fallback)
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear convolution using FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (real inputs, real output)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    size = 1 << ((x.shape[0] + y.shape[0] - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Helper: sub‑gradient of the maximal autocorrelation entry
# ----------------------------------------------------------------------
def _subgrad_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of (a*a) over the active peak indices."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if active_idx.size == 0:
        return g
    for k in active_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
    g /= float(active_idx.size)
    return g


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function using mirror‑descent on the simplex.
    Returns a list of floats (heights) scaled to [0, 1000].
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # Determine sequence length (reuse previous length if possible)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1542

    # --------------------------------------------------------------
    # Global best tracking
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optimisation parameters
    # --------------------------------------------------------------
    max_iter_per_restart = 4000
    step0 = 0.6               # initial step size (mirrored gradient step)
    epsilon_rel = 1e-6        # tolerance for the active set
    restart = 0

    # --------------------------------------------------------------
    # Main loop over random restarts
    # --------------------------------------------------------------
    while time.time() < deadline:
        restart += 1

        # ----------- initialise a random candidate ------------------
        init_mode = np.random.choice(["sqrtdiff", "power", "random"])
        if init_mode == "sqrtdiff":

                a_new = a_local.copy()
                a_new[i] -= eps
                a_new[j] += eps
                # re‑normalise (still on simplex)
                a_new = a_new / a_new.sum()
                conv = _conv_fft(a_new)
                s = 2.0 * n * conv.max()
                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # ---------------------------------------------------------
        # Continue with another restart if time permits
        # ---------------------------------------------------------
        if time.time() >= deadline:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    return final_seq.tolist()
