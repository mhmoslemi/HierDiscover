"""
Non‑negative step‑function optimizer.

The evaluation metric is
    E(a) = 2·n·max(a*a) / (sum a)^2 .

Scaling a does not change the score, so we work on the probability simplex
(sum = 1).  The algorithm consists of three stages:

1.  Initialise a probability vector (use the provided construction if any,
    otherwise a simple exponential decay).

2.  Repeatedly solve a tiny linear program that finds a feasible direction
    d (with −a ≤ d ≤ a and Σd=0) that reduces the current maximal
    convolution entry.  Because each active convolution index gives a linear
    inequality  ∇(a*a)_i·d ≤ t, the LP minimizes t; a negative t guarantees a
    strict improvement.  When the active set becomes large we fall back to a
    fast greedy hill‑climb.

3.  A final stochastic hill‑climb that moves a few percent of the mass
    between random positions while keeping the sum fixed.

All heavy work is done with FFT‑based convolutions (O(n log n)).  The whole
search respects the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast computation of 2·n·max(a*a) for a distribution that already sums to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb that respects the simplex.
    A random fraction (≤5 %) of the mass at i is moved to j.
    Only moves that improve the fast score are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score

# LP based refinement (key novelty)

def active_set(conv: np.ndarray) -> np.ndarray:
    """Return indices where the convolution attains (almost) its maximum."""
    mx = conv.max()
    eps = mx * 1e-12 + 1e-12
    return np.where(conv >= mx - eps)[0]


def build_active_gradients(a: np.ndarray, act: np.ndarray) -> np.ndarray:
    """
    For each active convolution index i return the gradient
        g_i = ∇_a (a*a)_i = 2 * [a_{i-j}]_{j}
    as a row of a dense matrix (shape |act| × n).
    """
    n = a.shape[0]
    m = act.size
    G = np.zeros((m, n), dtype=np.float64)
    for r, i in enumerate(act):
        # indices j where a[i‑j] is defined
        j_start = max(0, i - (n - 1))
        j_end = min(i, n - 1)
        # vector of k = i‑j
        ks = i - np.arange(j_start, j_end + 1, dtype=np.int64)
        G[r, j_start:j_end + 1] = 2.0 * a[ks]
    return G


def lp_step(a: np.ndarray, G: np.ndarray) -> tuple[np.ndarray, float, bool]:
    """
    Solve   min t
            s.t. G·d ≤ t
                 Σ d = 0
                -a ≤ d ≤ a
    Returns (new_a, t, success_flag).
    """
    n = a.shape[0]
    m = G.shape[0]

    # Objective: minimise t (the last variable)
    c = np.zeros(n + 1, dtype=np.float64)
    c[-1] = 1.0

    # Inequalities G d - t ≤ 0
    A_ub = np.hstack([G, -np.ones((m, 1), dtype=np.float64)])
    b_ub = np.zeros(m, dtype=np.float64)

    # Equality Σ d = 0
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0], dtype=np.float64)

    # Variable bounds: -a_j ≤ d_j ≤ a_j, t free
    bounds = [( -a[j], a[j]) for j in range(n)] + [(None, None)]

    # Use the HiGHS interior‑point method (fast for large n, few constraints)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': None})

    if not res.success:
        return a, 0.0, False

    d = res.x[:n]
    t = res.x[-1]

    # No improvement if t is not negative enough
    if t >= -1e-12:
        return a, t, False

    new_a = a + d
    # Numerical safety: clip tiny negatives that can appear due to rounding
    new_a = np.clip(new_a, 0.0, None)
    # Renormalise (should already sum to 1 because Σd=0, but correct drift)
    s = new_a.sum()
    if s > 0:
        new_a /= s
    else:
        # fallback to uniform – should never happen
        new_a = np.full_like(new_a, 1.0 / new_a.size)

    return new_a, t, True

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence that minimises the evaluation metric.
    The method combines random initialisation, a linear‑programming based
    refinement that directly reduces the maximal convolution entry, and a final
    stochastic hill‑climb.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.5  # safety margin

    # ------------------------------------------------------------------
    # 1️⃣ Initialise a probability vector
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init = np.array(globals()['height_sequence_1'], dtype=float)
        if init.ndim == 1 and init.size > 0:
            n = min(init.size, 4000)
            a = init[:n].copy()
        else:
            n = 4000
            a = np.full(n, 1.0 / n)
    else:
        n = 4000
        a = np.full(n, 1.0 / n)

    # Clean and normalise
    a[a < 0] = 0.0
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    # Occasionally replace a completely random start if the provided one
    # does not look promising (empirical safeguard).
    if a.max() > 0.5:                     # too spiky → randomise a bit
        rng = np.random.default_rng(seed + 1)
        a = np.exp(-rng.uniform(0.001, 0.02, size=n))
        a = a / a.sum()

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2️⃣ Generic fast hill‑climb to escape possible poor local minima
    # ------------------------------------------------------------------
    cur_score = fast_score(a, fft_len, n)
    a, cur_score = hill_climb(a, cur_score, fft_len, n,
                              deadline=min(deadline, time.time() + 5.0),
                              max_iter=2500)
    best_a = a.copy()
    best_score = cur_score
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ LP‑based active‑set refinement (core of the new approach)
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        conv = full_conv(a, fft_len)
        mx = conv.max()
        act = active_set(conv)

        # Too many active constraints → fall back to hill‑climb for a while
        if act.size > 200:
            a, cur_score = hill_climb(a, fast_score(a, fft_len, n), fft_len, n,
                                      deadline=deadline, max_iter=1200)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[iter {iteration}] hill‑climb improved score={best_score:.6f}")
            continue

        # Build the gradient matrix for the active set
        G = build_active_gradients(a, act)

        # Solve the small LP that finds the best feasible direction
        a_new, t, ok = lp_step(a, G)

        if not ok:
            # No improving LP direction – try a brief hill‑climb then give up
            a, cur_score = hill_climb(a, fast_score(a, fft_len, n), fft_len, n,
                                      deadline=deadline, max_iter=800)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[iter {iteration}] post‑LP hill‑climb improved score={best_score:.6f}")
            break

        a = a_new
        cur_score = fast_score(a, fft_len, n)

        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[iter {iteration}] LP step improved score={best_score:.6f} (t={t:.2e})")

        # Occasionally inject a tiny random jitter to avoid stagnation
        if iteration % 25 == 0 and time.time() < deadline - 2.0:
            rng = np.random.default_rng(seed + iteration)
            jitter = rng.normal(scale=0.002, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()
            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[iter {iteration}] jitter improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final polishing with an unrestricted hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        fast_score(best_a, fft_len, n),
                                        fft_len, n,
                                        deadline=deadline,
                                        max_iter=4000)
        print(f"[final hill] polished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Return the best sequence (the evaluation routine will re‑scale it)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
