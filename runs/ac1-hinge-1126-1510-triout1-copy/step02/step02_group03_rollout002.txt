"""
Smooth‑max Adam optimizer for the autocorrelation peak.

The algorithm works on the probability simplex (non‑negative entries that sum to 1)
and minimizes the evaluation function

    2 * n * max(conv(a,a)) / (sum(a)**2)

which is invariant under scaling, so we keep the sum equal to 1.
A smooth approximation of the ℓ∞‑norm,
    f_λ(a) = (1/λ)·log( Σ_k exp(λ·(a*a)_k) ),
is used.  Its gradient can be written in closed form via a convolution,
allowing an O(n log n) implementation with FFTs.
We employ the Adam adaptive‑step scheme and Euclidean projection onto the simplex.
Multiple random restarts (including the previously found sequence
`height_sequence_1` when available) are performed; a short polishing phase
based on random mass transfers follows each restart.

The final sequence is scaled to the range [0,1000] (the scaling does not change
the score) and returned as a plain Python list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # Duchi et al. (2008) – O(n) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Helper: full linear convolution a*a using FFT (real → real)
# ----------------------------------------------------------------------
def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution of a with itself (full length, real FFT)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]

# ----------------------------------------------------------------------
# Helper: gradient of the smooth max (log‑sum‑exp) of the autocorrelation
# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Compute ∇_a  (1/λ)·log Σ_k exp( λ·(a*a)_k ).
    Returned gradient has shape (n,).
    """
    n = a.shape[0]

    # 1) self‑convolution a*a
    conv = _conv_full(a)                     # length 2n-1

    # 2) soft‑max weights (shifted for numerical stability)
    max_b = conv.max()
    exp_shifted = np.exp(lam * (conv - max_b))
    Z = exp_shifted.sum()
    w = exp_shifted / Z                      # normalized weights, length 2n-1

    # 3) convolution a with the reverse of w
    w_rev = w[::-1]                          # length 2n-1
    size = 1 << ((n + (2 * n - 1) - 1).bit_length())
    fa = np.fft.rfft(a, size)
    fw = np.fft.rfft(w_rev, size)
    conv_full = np.fft.irfft(fa * fw, size)[: n + (2 * n - 1) - 1]

    # slice the part that corresponds to the gradient (see analysis)
    # conv_full[n-1 : 2n-1] has length n, but it is in reverse order
    grad = 2.0 * conv_full[n - 1 : 2 * n - 1][::-1]
    return grad

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step function using a smooth‑max Adam optimizer.
    The best sequence found within the time budget is returned (scaled to [0,1000]).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # --------------------------------------------------------------
    # Determine the length of the sequence.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)            # type: ignore[name-defined]
    except Exception:
        n = 1542                               # fallback length used previously

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Adam hyper‑parameters (these are fairly robust)
    # --------------------------------------------------------------
    alpha0 = 0.15          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Smooth‑max schedule
    lam_start = 0.5
    lam_end = 50.0
    lam_ramp_iters = 3000   # after this many steps λ reaches lam_end

    restart_counter = 0

    # --------------------------------------------------------------
    # Helper: initialise a sequence (different modes)
    # --------------------------------------------------------------
    def _initialise(mode: str) -> np.ndarray:
        if mode == "provided" and 'height_sequence_1' in globals():
            a = np.array(height_sequence_1, dtype=float)
        elif mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        elif mode == "power":
            p = np.random.uniform(0.1, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        else:  # "random"
            a = np.random.rand(n)

        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.ones(n, dtype=float)
        a = a / a.sum()                 # enforce sum = 1 (simplex)
        return a

    # --------------------------------------------------------------
    # Main restart loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        restart_counter += 1
        # Choose an initialization mode (occasionally reuse the known best)
        if best_seq is not None and np.random.rand() < 0.15:
            a = best_seq.copy()
        else:
            mode = np.random.choice(["provided", "sqrtdiff", "power", "random"])
            a = _initialise(mode)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        iteration = 0
        # ----------------------------------------------------------
        # Inner optimisation loop (Adam + smooth‑max)
        # ----------------------------------------------------------
        while time.time() < deadline:
            # linear λ schedule
            lam = lam_start + (lam_end - lam_start) * min(1.0, iteration / lam_ramp_iters)

            # gradient of smooth max
            g = _smooth_gradient(a, lam)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)

            t_hat = iteration + 1
            m_hat = m / (1.0 - beta1**t_hat)
            v_hat = v / (1.0 - beta2**t_hat)

            # adaptive step (we move opposite to the gradient)
            step = alpha0 / (np.sqrt(v_hat) + eps)
            a_candidate = a - step * m_hat

            # projection onto the simplex
            a_candidate = _project_to_simplex(a_candidate)

            # evaluate true objective (this is the only expensive call)
            score = evaluate_sequence(a_candidate.tolist())
            if score < best_score:
                best_score = score
                best_seq = a_candidate.copy()
                elapsed = time.time() - start_time
                print(f"[Restart {restart_counter}] iter {iteration:05d} – "
                      f"new best {best_score:.6f} (t={elapsed:.1f}s)")

            # unconditionally accept the candidate (helps escaping plateaus)
            a = a_candidate
            iteration += 1

            # simple stopping criterion for this restart
            if iteration >= 8000:      # generous per‑restart budget
                break

        # ----------------------------------------------------------
        # Polishing phase – tiny random mass swaps
        # ----------------------------------------------------------
        if best_seq is not None:
            a_local = best_seq.copy()
            polish_deadline = time.time() + min(8.0, deadline - time.time())
            while time.time() < polish_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # move a random fraction (up to 5 %) from i to j
                eps_move = a_local[i] * np.random.uniform(0.0, 0.05)
                if eps_move <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps_move
                a_new[j] += eps_move
                a_new = _project_to_simplex(a_new)
                s = evaluate_sequence(a_new.tolist())
                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # ----------------------------------------------------------
        # End of this restart – see if we have time for another one
        # ----------------------------------------------------------
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        a_fallback = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        best_seq = a_fallback / a_fallback.sum()

    # ------------------------------------------------------------------
    # Scale to the evaluator's [0,1000] range (does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(f"Returning sequence – best score {best_score:.6f} after {total_elapsed:.1f}s")
    return final_seq.tolist()
