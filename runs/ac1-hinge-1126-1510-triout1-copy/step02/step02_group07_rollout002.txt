def solve_lp(f, M):
    n = len(f)
    # Build A_ub shape (2n-1, n)
    # Use np.zeros then fill.
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # For each i, j = k - i
        # coefficient of b_j = f_i
        # So we can vectorize but loops okay.
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]  # assignment repeated if multiple i? Actually each j appears exactly once per i.
    # Note: there may be multiple i contributions for same j? Actually for each k, each j corresponds to i = k - j. So each j appears exactly once. So fine.
    # Create bounds
    bounds = [(0, None)] * n
    # c = -1 vector
    c = -np.ones(n)
    # b_ub vector all = M
    b_ub = np.full(rows, M)
    # Solve LP
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # If fails, fallback to zero vector.
        return np.zeros(n)
    b = res.x
    # Ensure non-negative
    b = np.maximum(b, 0)
    return b

"""
Hybrid LP‑driven optimizer for the self‑convolution bound.

1. Initialise a candidate from the half‑order (central binomial) series.
   A small parametric sweep (power‑tilt, mixing with the uniform vector and
   exponentiation of the base) is performed to obtain a good starting point.

2. Main loop (until the time budget expires):
   – Solve a linear programme
       maximise   Σ b_j
       subject to (f * b)_k ≤ M   for all k,
                  b ≥ 0,
     where f is the current sequence and M = max(f*f).  The constraints are
     linear in b (convolution with a fixed f) and the LP yields a vector b
     that is “as large as possible” while respecting the current self‑convolution
     bound M.

   – Normalise b to sum to 1, obtaining g = b / Σb.
     Because (f * b) ≤ M, we also have (f * g) ≤ M / Σb ≤ M.
     We then move from f towards g by a convex combination
         f←(1‑t)·f + t·g ,   t∈(0,1].
     The step size t is reduced (halved) until the new maximal convolution
     does not exceed the previous M, guaranteeing a monotone descent.

   – Occasionally perform a cheap random pair‑exchange move (as in the
     previous submission) to escape plateaux.

   – Keep the best sequence (lowest evaluation score) seen so far and print
     progress.  The loop stops early if the target score 1.503 is reached.

The algorithm mixes the global improvement power of a linear programme with
fast stochastic hill‑climbing, and fits comfortably into the 1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _half_order_series(n: int) -> np.ndarray:
    """Central‑binomial (half‑order) coefficients, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x≥0, Σx=1}. """
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_ub_matrix(f: np.ndarray) -> np.ndarray:
    """Build the matrix A such that (A @ b)[k] = (f * b)[k] (full convolution)."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j corresponds to b_j; it appears in rows j … j+n‑1 with weight f
    for j in range(n):
        A[j : j + n, j] = f
    return A


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray:
    """Linear programme:
         maximise Σ b_j
         s.t. (f * b)_k ≤ M   (k = 0 … 2n‑2),
              b ≥ 0 .
       Returns the optimal b (may be all‑zeros if the solver fails)."""
    n = f.shape[0]
    A = _build_ub_matrix(f)
    c = -np.ones(n)                     # linprog minimises, so -Σb
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    # use the modern HiGHS backend – fast and robust
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return np.zeros(n, dtype=np.float64)
    return np.maximum(res.x, 0.0)


def _random_pair_exchange(seq: np.ndarray, max_move: float = 0.02):
    """Move a small amount of mass from one index to another."""
    n = seq.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    max_delta = min(seq[i], max_move)
    if max_delta <= 0:
        return seq, False, np.inf
    delta = np.random.rand() * max_delta
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    cand_max = np.convolve(cand, cand).max()
    return cand, True, cand_max

                     budget_s: float = 1000,

    evaluate_sequence().  The routine mixes a parametric sweep,
    a linear‑programming driven Frank‑Wolfe style update,
    and occasional stochastic pair‑exchanges.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # 1) problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 800))      # length of the sequence

    # ------------------------------------------------------------------
    base = _half_order_series(n)

    best_M = np.convolve(best_seq, best_seq).max()
    best_score = 2.0 * n * best_M

    for delta in np.linspace(0.0, 0.8, 17):          # 0 … 0.8 step 0.05

        M = np.convolve(cand, cand).max()
        score = 2.0 * n * M

            print(f"[init] tilt δ={delta:.2f} → score {best_score:.6f}")

        M = np.convolve(cand, cand).max()

        if np.random.rand() < prob_lp_step:
            b = _solve_lp(cur_seq, cur_M)
            sum_b = b.sum()
            if sum_b > 0:
                g = b / sum_b                     # normalise to Σg = 1
                # try a convex combination, backing off the step size if needed
                t = 0.5
                for _ in range(6):
                    cand_seq = (1.0 - t) * cur_seq + t * g
                    cand_seq = _project_simplex(cand_seq)
                    cand_M = np.convolve(cand_seq, cand_seq).max()
                    if cand_M <= cur_M * (1.0 + 1e-12):
                        cur_seq = cand_seq
                        cur_M = cand_M
                        cur_score = 2.0 * n * cur_M
                        break
                    t *= 0.5
        else:
            # --------------------------------------------------------------
            # 3b) cheap stochastic pair‑exchange (hill‑climbing)
            # --------------------------------------------------------------
            cand_seq, moved, cand_M = _random_pair_exchange(cur_seq)
            if moved:
                accept = False
                if cand_M < cur_M:
                    accept = True
                else:
                    # Metropolis‑type acceptance with a tiny temperature
                    temp = cur_M * 0.01
                    prob = math.exp(-(cand_M - cur_M) / temp)
                    if np.random.rand() < prob:
                        accept = True
                if accept:
                    cur_seq = cand_seq
                    cur_M = cand_M
                    cur_score = 2.0 * n * cur_M

        # --------------------------------------------------------------
        # 4) bookkeeping
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-9:
            best_score = cur_score
            best_seq = cur_seq.copy()
            no_improve = 0
            print(f"[{iteration:6d}] New best score: {best_score:.6f}")
        else:
            no_improve += 1

        # occasional jitter to shake out of shallow plateaux
        if no_improve >= 1500:
            sigma = 0.001
            noise = np.random.randn(n) * sigma
            cur_seq = cur_seq * (1.0 + noise)
            cur_seq = np.clip(cur_seq, 0.0, None)
            cur_seq = _project_simplex(cur_seq)
            cur_M = np.convolve(cur_seq, cur_seq).max()
            cur_score = 2.0 * n * cur_M
            no_improve = 0

        # periodic status report
        if iteration % 500 == 0:
            elapsed = time.time() - start
            print(f"[{iteration:6d}] elapsed {elapsed:.1f}s best {best_score:.6f}")

        # early exit if target already reached
        if best_score <= 1.503:
            print(f"Target reached after {iteration} iterations.")
            break

    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    # guarantee a plain list of floats for the evaluator
    return best_seq.tolist()
