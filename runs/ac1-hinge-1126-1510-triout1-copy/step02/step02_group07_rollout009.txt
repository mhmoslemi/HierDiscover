"""
Linearized-convolution LP refinement for the self‑convolution bound.

The algorithm:
1. Build an initial non‑negative sequence (half‑order series, optionally tilted
   and mixed with the uniform distribution) that already gives a decent bound.
2. Repeatedly linearise the non‑convex constraints
   b_k = (a*a)_k ≤ M around the current iterate a.
   The linearisation yields the linear constraints
       Σ_i a_i a_cur_{k-i} - 0.5 M ≤ ½ b_cur_k .
   These constraints together with a_i ≥ 0, Σ a_i = 1 define a linear program.
   Solving the LP gives a new candidate a that (hopefully) reduces the true
   maximal convolution value.
3. If the LP step does not improve the true objective we fall back to a
   small random mass‑exchange move (simulated‑annealing style) to escape
   plateaux.
4. The best sequence seen so far is kept and returned when the time budget
   expires.
"""

import time
import math
import numpy as np

from scipy.linalg import toeplitz
from scipy.optimize import linprog

def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i, normalised to sum 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a

def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix C where C[k,i] = a_{k-i} (zero if out of range).
    This matrix satisfies C @ x = np.convolve(a, x) for zero‑padded x.
    """
    n = a.shape[0]
    # first column: a followed by n‑1 zeros
    col = np.concatenate((a, np.zeros(n - 1, dtype=np.float64)))
    # first row: a_0 followed by n‑1 zeros
    row = np.concatenate((a[:1], np.zeros(n - 1, dtype=np.float64)))
    C = toeplitz(col, row)  # shape (2n‑1, n)
    return C

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The routine uses a linear‑program based refinement
    (see the module docstring) together with occasional stochastic
    pair‑exchange moves.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Problem size and initial sequence
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1000))
    # try to use a previously found sequence if it exists
    init_seq = None
    try:
        # `height_sequence_1` is expected to be a global variable containing a list
        if isinstance(height_sequence_1, (list, tuple, np.ndarray)):
            tmp = np.asarray(height_sequence_1, dtype=np.float64)
            if tmp.ndim == 1 and tmp.size >= 2:
                if tmp.size != n:
                    # truncate or pad with zeros, then renormalise
                    if tmp.size > n:
                        tmp = tmp[:n]
                    else:
                        extra = np.zeros(n - tmp.size, dtype=np.float64)
                        tmp = np.concatenate((tmp, extra))
                init_seq = tmp.copy()
    except Exception:
        init_seq = None

    if init_seq is None:
        # start from the half‑order series
        init_seq = _half_order_series(n)

    # make sure it is a probability vector
    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0:
        init_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        init_seq = init_seq / init_seq.sum()

    # ------------------------------------------------------------------
    # 2. Light parametric sweep (exponent tilt + uniform mixing)
    # ------------------------------------------------------------------
    best_seq = init_seq.copy()
    conv = np.convolve(best_seq, best_seq)
    best_max = conv.max()
    best_score = 2.0 * n * best_max

    # exponent tilt sweep
    for delta in np.linspace(0.0, 0.8, 17):   # 0 … 0.8 step 0.05
        taper = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = init_seq * taper
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        conv = np.convolve(cand, cand)
        m = conv.max()
        score = 2.0 * n * m
        if score < best_score:
            best_score = score
            best_max = m
            best_seq = cand.copy()
            print(f"[init tilt] delta={delta:.2f} → score {best_score:.6f}")

    # uniform mixing sweep
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in np.linspace(0.0, 1.0, 13):   # 0 … 1 step 0.0833
        cand = (1.0 - alpha) * init_seq + alpha * uniform
        cand_sum = cand.sum()
        if cand_sum == 0:
            continue
        cand /= cand_sum
        conv = np.convolve(cand, cand)
        m = conv.max()
        score = 2.0 * n * m
        if score < best_score:
            best_score = score
            best_max = m
            best_seq = cand.copy()
            print(f"[init mix] alpha={alpha:.2f} → score {best_score:.6f}")

    cur_seq = best_seq.copy()
    cur_max = best_max
    cur_score = best_score

    # ------------------------------------------------------------------
    # 3. Main optimisation loop: LP refinement + random exchange
    # ------------------------------------------------------------------
    iteration = 0
    no_improve_counter = 0
    # Parameters for the fallback random move
    max_move_fraction = 0.02   # max proportion of mass moved in one exchange
    jitter_sigma = 0.003       # magnitude of occasional jitter

    while time.time() < deadline:
        iteration += 1
        # compute fresh convolution and bound
        conv = np.convolve(cur_seq, cur_seq)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # ------------------------------------------------------------------
        # 3a. Linearised LP step
        # ------------------------------------------------------------------
        # Build Toeplitz matrix representing convolution with current seq
        C = _build_convolution_matrix(cur_seq)          # shape (2n-1, n)
        rows = C.shape[0]
        # Assemble inequality matrix: C @ a - 0.5 * t <= 0.5 * conv
        A_ub = np.empty((rows, n + 1), dtype=np.float64)
        A_ub[:, :n] = C
        A_ub[:, n] = -0.5
        # Right‑hand side with a tiny slack for numerical stability
        b_ub = 0.5 * conv + 1e-14

        # Equality constraint Σ a_i = 1
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0])

        # Bounds: a_i ≥ 0, t ≥ 0

    a = np.empty(n, dtype=np.float64)

    a /= a.sum()
