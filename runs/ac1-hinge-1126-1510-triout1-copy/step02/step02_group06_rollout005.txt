"""
Propose a candidate step‑function that (hopefully) beats the current
best bound 1.527945.  

The algorithm has two phases:

1. **Broad stochastic search** – many random shapes (power‑law,
   exponential, two‑sided power‑law and discretised Beta‑pdf) of
   assorted lengths are generated, normalised to sum = 1 and scored
   with a cheap FFT‑based evaluation.  The top‑few candidates are kept.

2. **Momentum sub‑gradient refinement** – each top candidate is refined
   by projected sub‑gradient descent on the simplex.  The sub‑gradient
   is the sum of the contributions of *all* convolution entries that
   are (numerically) at the current maximum, not just a single index.
   A Nesterov‑style momentum term, a diminishing step‑size and occasional
   random noise (to escape plateaus) are used.  The refinement runs
   until the allotted time expires.

The best sequence found is finally verified with the supplied
`evaluate_sequence` function and returned as a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quickly evaluate 2 * n * max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT convolution.  The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # convolution via FFT, zero‑padded to a power of two
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))

# Sub‑gradient refinement with momentum
# ----------------------------------------------------------------------

    Returns the best sequence found and its fast_score.
    """

    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * float(conv.max())

    v = np.zeros_like(a)
    it = 0
    no_improve = 0

        it += 1

        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- 2) compute sub‑gradient of the peak (all active indices) ----------
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # ----- 3) momentum / step size ---------------------------------------
        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * g
        cand = a - eta * v
        a = project_simplex(cand)

        # ----- 4) occasional random perturbation to escape plateaus ----------
        if no_improve >= 3000:
            sigma = 0.03 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact score for the best candidate
    best_score = fast_score(best_seq)
    return best_seq, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score height sequence within `budget_s` seconds.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5       # small safety margin

    # ------------------------------------------------------------------
    # 0) initial candidate – try to reuse a previously discovered one
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        # fallback to a modest two‑sided power‑law
        init_len = 3000
        init_seq = generate_two_sided_power_law(init_len, 0.55, 0.55, 1.0, 1.0)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0:
        init_seq = np.full(init_seq.shape, 1.0 / init_seq.shape[0])
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Randomised family search
    # ------------------------------------------------------------------
    # possible lengths – keep them moderate so that each evaluation stays fast
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    # keep a pool of the best few candidates (score, sequence)
    top_k = [(best_score, best_seq.copy())]   # start with the init

    # allocate ~60 % of the total time to the random search
    rand_deadline = start + 0.6 * budget_s
    families = ["power", "exp", "twosided", "beta"]
    while time.time() < rand_deadline:
        n = random.choice(length_choices)

        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.10, 1.20)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            p_l = random.uniform(0.10, 1.20)
            p_r = random.uniform(0.10, 1.20)
            w_l = random.uniform(0.2, 5.0)
            w_r = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        else:  # "beta"
            alpha = random.uniform(0.05, 1.50)
            beta = random.uniform(0.05, 1.50)
            seq = generate_beta_shape(n, alpha, beta)

        # normalise and safeguard
        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] new global best ({family}) n={n} score={sc:.6f}")

        # maintain a small elite set (size ≤ 5)
        if len(top_k) < 5:
            top_k.append((sc, seq.copy()))
        else:
            # replace the worst elite if this one is better
            worst_idx = max(range(len(top_k)), key=lambda i: top_k[i][0])
            if sc < top_k[worst_idx][0]:
                top_k[worst_idx] = (sc, seq.copy())

    # sort elite candidates by score (ascending)
    top_k.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 2) Momentum sub‑gradient refinement on elite candidates
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining <= 0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Time exhausted. Final score = {final_score:.6f}")
        return final_seq

    # give each elite candidate an equal share of the remaining budget
    per_candidate = time_remaining / len(top_k)

    for idx, (score, seq) in enumerate(top_k):
        if time.time() > deadline:
            break
        print(f"[refine] starting elite #{idx+1} (score={score:.6f}, len={seq.size})")
        refined_seq, refined_score = refine_sequence(seq, per_candidate, time.time())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
