def gradient_descent_smooth_max(a_init, time_limit):
    a = a_init.copy()
    n = a.shape[0]
    L = next_pow2(2*n)
    best_a = a.copy()
    best_score = fast_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2
    t = 0
    start = time.time()
    while time.time() < time_limit:
        t += 1
        alpha = min(500.0, 10.0 + t * 0.005)  # increase slowly
        Fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2*n - 1]
        # softmax weights
        # To avoid overflow, subtract max
        max_val = np.max(alpha * conv)
        w = np.exp(alpha * conv - max_val)
        w_sum = np.sum(w)
        w /= w_sum
        # gradient w.r.t a: 2 * alpha * correlation(a,w)
        Fw = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(Fa * np.conj(Fw), n=L)[:n]
        grad = 2.0 * alpha * np.real(corr)
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a_candidate)
        # periodic evaluation
        if t % 100 == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[{time.time() - start:.1f}s] iter {t}, new best {sc:.6f}")
    return best_a, best_score
```

    a = a_init.copy()
    n = a.shape[0]
    L = next_pow2(2 * n)
    best_a = a.copy()

    beta2 = 0.999

    base_lr = 0.3  # adjust if needed
    iter_count = 0
    last_improve_iter = 0
    while time.time() < deadline - 0.5:
        iter_count += 1
        # temperature schedule
        alpha = 10.0 + (iter_count - 1) * 0.004  # slowly up to ~500 after 125k iters, but we will stop earlier
        # limit alpha
        if alpha > 500.0:
            alpha = 500.0

        # FFT of a
        Fa = np.fft.rfft(a, n=L)
        # self-convolution
        conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        # compute softmax weights
        # to improve numeric stability, subtract max
        max_val = np.max(alpha * conv)
        w = np.exp(alpha * conv - max_val)
        w_sum = np.sum(w)
        if w_sum == 0:
            # fallback uniform weights
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient via correlation a * w
        Fw = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(Fa * np.conj(Fw), n=L)[:n]
        # gradient (real)
        grad = 2.0 * alpha * np.real(corr)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** iter_count)
        v_hat = v / (1 - beta2 ** iter_count)
        # adaptive learning rate (optional: decay)
        lr = base_lr / (np.sqrt(iter_count) + 1)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a = project_simplex(a_new)

        # periodic evaluation
        if iter_count % 200 == 0:
            cur_sc = fast_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

            # optional early break if stagnated for long time
            if iter_count - last_improve_iter > 5000:
                # no improvement for many iterations; break
                # but we must still respect time; we can continue a bit
                pass

    return best_a, best_sc

"""
Improved search for a low‑score step‑function.

Algorithm
---------
1. **Exploratory random sampling** (≈15 % of the time budget):
   - Sample many candidate shapes (power‑law, exponential, two‑sided power‑law,
     Hann, Kaiser, Tukey, Gaussian, DPSS when SciPy is available) of a
     relatively large length (≈30 000).
   - Normalise each to sum = 1 and evaluate quickly with an FFT‑based
     approximation (`fast_score`).  Keep the best candidate.

2. **Smooth‑max descent** (≈85 % of the budget):
   - Optimise the smooth approximation
         f(a)=log Σ_k exp(α·(a*a)_k)
     whose gradient is
         ∇f = 2·α·corr(a, w)   with   w_k ∝ exp(α·(a*a)_k) .
   - The correlation is computed by a single FFT‑based cross‑correlation.
   - An Adam optimiser with a diminishing learning‑rate is applied.
   - After each update the point is projected back onto the probability
     simplex (non‑negative entries summing to 1).

3. The best sequence found is finally evaluated with the provided
   `evaluate_sequence` and returned.

The whole routine respects a 1000 s wall‑clock limit and prints progress
messages.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – required by the task)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Very fast surrogate for ``evaluate_sequence`` using FFT."""
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (i = 0 … n‑1)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Gaussian centred window."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    return np.exp(-0.5 * ((i - centre) / sigma) ** 2)

except Exception:  # pragma: no cover
    SCIPY_WINDOWS = False

# Smooth‑max (soft‑max) descent on the simplex
# ----------------------------------------------------------------------


def smooth_max_descent(a_init: np.ndarray, deadline: float, start_time: float):
    """
    Perform Adam‑style descent on the smooth approximation of the max
    self‑convolution.  Returns the best sequence seen and its (fast) score.
    """
    a = a_init.copy()
    n = a.shape[0]
    L = next_pow2(2 * n)

    best_a = a.copy()
    best_sc = fast_score(a)

    # Adam hyper‑parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.3

    iter_cnt = 0
    last_improve = 0

    while time.time() < deadline - 0.5:
        iter_cnt += 1

        # temperature (smoothness) schedule
        alpha = 10.0 + 0.004 * (iter_cnt - 1)  # grows very slowly
        if alpha > 500.0:
            alpha = 500.0

        # FFT of current a
        Fa = np.fft.rfft(a, n=L)

        # self‑convolution (full length 2n‑1)
        conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]

        # soft‑max weights w_k ∝ exp(α·conv_k)
        z = alpha * conv_full
        z_max = np.max(z)
        w = np.exp(z - z_max)
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # correlation a ⋆ w (lag 0 … n‑1)
        Fw = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(Fa * np.conj(Fw), n=L)[:n]  # real values

        # gradient of the smooth‑max objective
        grad = 2.0 * alpha * np.real(corr)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        lr = base_lr / (np.sqrt(iter_cnt) + 1.0)
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection back onto the simplex
        a = project_simplex(a_candidate)

        # periodic evaluation of the true objective
        if iter_cnt % 200 == 0:
            cur_sc = fast_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                last_improve = iter_cnt
                print(
                    f"[{time.time() - start_time:.1f}s] iter {iter_cnt}: "
                    f"new best {cur_sc:.6f}"
                )
            # optional early‑stop if stagnated for a long time
            if iter_cnt - last_improve > 5000:
                # no improvement for many steps – continue a little longer
                pass

    return best_a, best_sc
