"""
Propose a high‑quality step‑function sequence.

The algorithm proceeds in three stages:

1.  **Parametric warm‑start** – a small grid of parametric families
    (half‑order, arcsine‑type and Beta‑shaped densities) is evaluated
    on the simplex and the best sequence is kept.

2.  **Smoothed projected gradient descent** – the objective
    `max(a*a)` (the maximal self‑convolution entry) is approximated
    by a soft‑max (log‑sum‑exp) with a temperature that slowly
    anneals from a smooth regime (λ≈1) to a near‑max regime
    (λ≈30).  The gradient of this surrogate is
    `∇f(a) = 2 * corr(a, w)` where `w` are the soft‑max weights.
    After each step the iterate is projected back onto the probability
    simplex.

3.  **Occasional random jitter** – to escape shallow plateaus a small
    Gaussian perturbation is applied every few hundred iterations.

All convolutions are performed with the radix‑2 real FFT for speed
(`O(n log n)`).  The routine respects the 1000‑second budget,
keeps the best sequence seen and finally returns it as a plain list
of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        # projection is the uniform vector
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _half_order_series(n: int) -> np.ndarray:
    """Half‑order series a_i = C(2i,i) / 4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape (i+1)^{α-1} (n-i)^{β-1} on {0,…,n‑1}, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    shape[shape < 0] = 0.0
    shape /= shape.sum()
    return shape


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine‑like density 1/√(i (n+1-i)), normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = 1.0 / np.sqrt(i * (n + 1 - i))
    shape /= shape.sum()
    return shape


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution using a real FFT (fast for n up to several‑thousand)."""
    m = x.shape[0] + y.shape[0] - 1
    nfft = 1 << (m - 1).bit_length()          # next power of two
    X = np.fft.rfft(x, n=nfft)
    Y = np.fft.rfft(y, n=nfft)
    Z = X * Y
    conv = np.fft.irfft(Z, n=nfft)[:m]
    return conv


def _score_from_seq(seq: np.ndarray) -> float:
    """Return 2·n·max(conv) for a probability vector (sum≈1)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative sequence that minimises
    `evaluate_sequence`.  The routine combines a parametric
    warm‑start with a softened projected gradient descent.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1) problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1500))          # default length; adjustable

    # ------------------------------------------------------------------
    # 2) Initial candidates from simple parametric families
    # ------------------------------------------------------------------
    candidates = []

    # (a) half‑order series
    candidates.append(_half_order_series(n))

    # (b) arcsine‑type shape
    candidates.append(_arcsine_shape(n))

    # (c) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # (d) a small grid of Beta‑shapes
    alphas = [0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
    betas = [0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0]
    for a in alphas:
        for b in betas:
            candidates.append(_beta_shape(n, a, b))

    # (e) exponent‑tilted half‑order (a_i *= (i+1)^{-δ})
    base_half = _half_order_series(n)
    for delta in np.linspace(0.0, 0.8, 9):
        tilt = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = base_half * tilt
        cand /= cand.sum()
        candidates.append(cand)

    # (f) convex combinations with uniform (mixing factor α)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in np.linspace(0.0, 1.0, 11):
        cand = (1.0 - alpha) * base_half + alpha * uniform
        cand /= cand.sum()
        candidates.append(cand)

    # (g) possibly reuse a previously discovered construction
    if "height_sequence_1" in globals():
        seq = np.array(height_sequence_1, dtype=np.float64)
        if seq.shape[0] >= n:
            seq = seq[:n]
        else:
            seq = np.concatenate([seq, np.zeros(n - seq.shape[0])])
        seq = np.maximum(seq, 0.0)
        if seq.sum() > 0:
            seq /= seq.sum()
            candidates.append(seq)

    # evaluate all warm‑start candidates (cheap, only O(#candidates))
    best_seq = None
    best_score = math.inf
    for cand in candidates:
        sc = _score_from_seq(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[init] Warm‑start best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Smoothed projected gradient descent
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # step‑size control (in ℓ₂‑sense)
    step_len = 0.05
    step_min = 1e-6
    step_max = 0.5

    # temperature (soft‑max) schedule
    lam_start = 1.0
    lam_end = 30.0

    iteration = 0
    print_interval = 1000

    while time.time() < deadline:
        iteration += 1
        progress = (time.time() - start_time) / (deadline - start_time)
        lam = lam_start + (lam_end - lam_start) * progress

        # ----- compute current convolution and soft‑max weights -----
        conv = _conv_fft(cur_seq, cur_seq)          # length 2n‑1
        cur_max = conv.max()

        # soft‑max weights w_i ∝ exp(λ (conv_i - cur_max))
        exp_vals = np.exp(lam * (conv - cur_max))
        w = exp_vals / exp_vals.sum()               # length 2n‑1

        # ----- gradient of the smoothed objective -----
        # grad = 2 * corr(cur_seq, w)  (active shifts 0…n‑1)
        corr_full = _conv_fft(cur_seq, w[::-1])      # length 3n‑2
        start_idx = 2 * n - 2
        # grad_k = 2 * corr_full[start_idx - k] for k=0..n‑1
        grad = 2.0 * corr_full[start_idx - np.arange(n)]

        # ----- take a projected gradient step -----
        g_norm = np.linalg.norm(grad) + 1e-16
        direction = grad / g_norm
        cand_seq = cur_seq - step_len * direction
        cand_seq = _project_simplex(cand_seq)

        # evaluate candidate (exact objective)
        cand_score = _score_from_seq(cand_seq)

        # accept if we made progress
        if cand_score < cur_score - 1e-12:
            cur_seq = cand_seq
            cur_score = cand_score
            # enlarge step a bit
            step_len = min(step_len * 1.15, step_max)
        else:
            # shrink step size
            step_len = max(step_len * 0.70, step_min)

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{iteration:6d}] New best score: {best_score:.6f}")

        # occasional random jitter to escape plateaux
        if iteration % 500 == 0:
            sigma = 0.001
            noise = np.random.randn(n) * sigma
            jitter = cur_seq * (1.0 + noise)
            jitter = np.clip(jitter, 0.0, None)
            jitter /= jitter.sum()
            cur_seq = _project_simplex(jitter)
            # evaluate jittered point (cheap)
            cur_score = _score_from_seq(cur_seq)

        # periodic logging
        if iteration % print_interval == 0:
            print(f"[{iteration:6d}] Current best: {best_score:.6f}  step={step_len:.4g}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
