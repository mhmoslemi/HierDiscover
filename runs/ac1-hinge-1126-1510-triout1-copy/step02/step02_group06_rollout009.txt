def gradient_descent(a, total_time_seconds, start_time):
    n = a.shape[0]
    L = next_pow2(3*n)
    iteration = 0
    best_seq = a.copy()
    best_score = fast_score(a)
    while time.time() < start_time + total_time_seconds - 0.5:
        iteration += 1
        # compute FFT of a
        FA = np.fft.fft(a, L)
        # compute self-convolution (full)
        conv_full = np.fft.ifft(FA * FA).real
        # extract length 2n-1 segment
        conv = conv_full[: 2*n - 1]
        max_val = conv.max()
        # create mask of near-max indices
        # tolerance: relative tolerance maybe 1e-4 of max or absolute 1e-8
        # set tol = max_val * 1e-4
        tol = max(1e-12, max_val * 1e-4)
        mask = conv >= max_val - tol
        selected = np.where(mask)[0]
        # Build weight vector w
        w = np.zeros_like(conv)
        if selected.size > 0:
            w[selected] = 1.0 / selected.size
        else:
            # fallback to single max index
            i_max = int(np.argmax(conv))
            w[i_max] = 1.0
        # compute correlation gradient
        FW = np.fft.fft(w, L)
        corr_full = np.fft.ifft(FA * np.conj(FW)).real
        grad = 2.0 * corr_full[:n]
        # step size
        eta = 0.5 / np.sqrt(iteration)
        # optional decrease factor based on max_val magnitude
        # candidate step
        a_candidate = a - eta * grad
        a_proj = project_simplex(a_candidate)
        # evaluate
        sc = fast_score(a_proj)
        if sc < best_score:
            best_score = sc
            best_seq = a_proj.copy()
            print(f"[{time.time() - start:.1f}s] grad it {iteration}, new best {sc:.6f}")
        # accept if improved else maybe use momentum?
        if sc <= fast_score(a):   # accept if not worse
            a = a_proj
        # else maybe do nothing (remain same a)
        # occasional random perturb
        if iteration % 500 == 0:
            # jitter
            sigma = 0.01 / np.sqrt(iteration)
            a_pert = a + sigma * np.random.randn(n)
            a_pert = np.maximum(a_pert, 0.0)

                print(f"[{time.time() - start:.1f}s] jitter improve {scp:.6f}")
            if scp < fast_score(a):
                a = a_pert
    return best_seq, best_score

"""
Propose a candidate sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1. **Diversified random search** – generate many candidate shapes
   (power‑law, exponential, two‑sided power‑law) of various lengths,
   normalise them to sum = 1 and keep the best one according to a fast
   FFT‑based evaluation.

2. **Smoothed sub‑gradient descent on the simplex** – starting from the
   best random candidate, repeatedly reduce the peak of the self‑
   convolution.  At each iteration the gradient of the (softened) peak
   is computed via an FFT correlation (so it is O(n log n)).  The step
   size diminishes as 1/√iteration and the iterate is projected back
   onto the probability simplex.  Every few hundred iterations the
   exact score is checked and the best sequence is stored.

3. **Post‑processing** – tiny trailing entries are trimmed and the
   result is re‑normalised; a final call to the provided
   ``evaluate_sequence`` verifies the score.

The whole routine respects a 1000 s budget and prints progress.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute the objective value 2*n*max(conv(a,a))/sum(a)^2 quickly.
    The sequence is assumed to be non‑negative; the sum is taken as‑is.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # zero‑pad to a power of two for an FFT‑based convolution
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} for i = 0 … n‑1 (not normalised)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i) for i = 0 … n‑1 (not normalised)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left * (i+1)^{-p_left} + w_right * (n-i)^{-p_right}
    – not normalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _gradient_descent(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform the smoothed sub‑gradient descent described in the module
    docstring.  Returns the best sequence found and its fast_score.
    """
    n = a.shape[0]
    L = next_pow2(3 * n)                # large enough for convolution + correlation
    best_seq = a.copy()
    best_score = fast_score(best_seq)

    rel_tol = 1e-4                      # how many near‑max indices we treat as active
    iteration = 0

    while time.time() < deadline - 1.0:
        iteration += 1

        # ----- self‑convolution (FFT) -----
        FA = np.fft.fft(a, L)
        conv_full = np.fft.ifft(FA * FA).real
        conv = conv_full[: 2 * n - 1]   # linear convolution, length 2n‑1
        max_val = conv.max()
        cur_score = 2.0 * n * max_val   # sum(a) == 1 by construction

        # keep the best sequence seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.1f}s] iter {iteration} improved score={cur_score:.6f}")

        # ----- active set for the maximal convolution entry -----
        tol = max(rel_tol * max_val, 1e-12)
        mask = conv >= max_val - tol
        active = np.where(mask)[0]
        if active.size == 0:            # sanity fallback
            active = np.array([int(np.argmax(conv))])

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size    # uniform weight on the active indices

        # ----- gradient of the weighted peak (correlation) -----
        FW = np.fft.fft(w, L)
        corr_full = np.fft.ifft(FA * np.conj(FW)).real
        grad = 2.0 * corr_full[:n]       # ∂/∂a of Σ_i w_i (a*a)[i]

        # ----- simple diminishing step size -----
        eta = 0.5 / np.sqrt(iteration)

        # projected sub‑gradient step
        a_candidate = a - eta * grad
        a = project_simplex(a_candidate)

        # ----- occasional random perturbation to escape flat regions -----
        if iteration % 500 == 0:
            sigma = 0.02 / np.sqrt(iteration)
            a_pert = a + sigma * np.random.randn(n)
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = project_simplex(a_pert)

            # cheap evaluation of the perturbed point
            FA_pert = np.fft.fft(a_pert, L)
            conv_pert = np.fft.ifft(FA_pert * FA_pert).real[: 2 * n - 1]
            pert_score = 2.0 * n * conv_pert.max()
            if pert_score < best_score:
                best_score = pert_score
                best_seq = a_pert.copy()
                print(f"[{time.time():.1f}s] perturb improved score={pert_score:.6f}")

            if pert_score <= cur_score:
                a = a_pert                   # accept the perturbation

    return best_seq, best_score
