def adam_descent(a0, max_iters, lr, beta1, beta2, deadline):
    a = a0.copy()
    n = len(a)
    m = np.zeros(n)
    v = np.zeros(n)
    best_seq = a.copy()
    best_score = float('inf')
    fft_size = 1 << ((2*n-1).bit_length())
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute conv and max
        b, max_b = conv_full_max(a, fft_size)
        sum_a = a.sum()
        if sum_a == 0:
            break
        score = 2.0 * n * max_b / (sum_a**2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # subgradient
        k = int(np.argmax(b))
        grad = np.zeros(n)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n-1)
        # vectorized:
        idx = np.arange(i_low, i_high+1)
        jdx = k - idx
        grad[idx] = 2.0 * a[jdx]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step_vec = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = a - step_vec  # descent
        # project onto simplex
        a = project_simplex(a)
    return best_seq, best_score

"""
Propose a low‑score step‑function using three phases:

1. **Random Power‑law/offset search** –  sample many (p,offset) pairs,
   evaluate the objective quickly with an FFT‐based convolution, and keep the
   best shape found.

2. **Projected Adam (mirror‑descent) refinement** – start from the best shape
   (or other random starts) and run a few thousand Adam‑style sub‑gradient
   steps, projecting after each step onto the probability simplex.

3. **Light simulated‑annealing polishing** – with the remaining time,
   perform small random mass‑shifts between two coordinates and accept
   improvements (or uphill moves with a cooling temperature).

All helper routines are top‑level, use only NumPy and the built‑in `time`
module, and respect the 1000 s time‑budget.  The returned list satisfies the
non‑negativity constraints required by `evaluate_sequence`.
"""

import time
import numpy as np


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, Σ x = 1} (Duchi et al., 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non–positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_conv_max(a: np.ndarray, fft_size: int) -> tuple[np.ndarray, float]:
    """
    Full (linear) convolution a*a via FFT, returning the full vector
    and its maximal entry.
    """
    A = np.fft.rfft(a, n=fft_size)
    b = np.fft.irfft(A * A, n=fft_size)[: 2 * a.shape[0] - 1]
    return b, float(b.max())


def adam_descent(
    a0: np.ndarray,
    max_iters: int,
    lr: float,
    beta1: float,
    beta2: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the peak of the self‑convolution.
    The objective is f(a)=2·n·max(conv(a,a)) / (∑a)².
    Only the sub‑gradient of the active peak is used.
    """
    a = a0.copy()
    n = a.shape[0]
    # normalise – the scale does not affect the score
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum

    # pre‑compute FFT size for the whole run
    fft_size = 1 << ((2 * n - 1).bit_length())

    m = np.zeros(n)      # first moment estimate
    v = np.zeros(n)      # second moment estimate

    best_seq = a.copy()
    best_score = float("inf")

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # convolution and active peak
        b, max_b = compute_conv_max(a, fft_size)

        # current score (∑a = 1 after the projection)
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # occasional progress report
            print(f"[adam it {t}] new best score={score:.6f}")

        # sub‑gradient of the active peak
        k = int(np.argmax(b))
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        grad = np.zeros(n, dtype=np.float64)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        jdx = k - idx
        grad[idx] = 2.0 * a[jdx]

        # Adam update (note the descent direction is -grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        a = a - step
        a = project_simplex(a)      # keep us on the simplex

    # score reconstruction (in case the last iteration was never evaluated)
    b, max_b = compute_conv_max(best_seq, fft_size)
    final_score = 2.0 * n * max_b
    return best_seq, final_score
