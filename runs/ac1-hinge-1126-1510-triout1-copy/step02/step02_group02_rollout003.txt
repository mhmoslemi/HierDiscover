def _adam_opt(a_init, L, phase, max_iters, deadline, alpha_start, alpha_end, lr0):
    a = a_init.copy()
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    # compute objective of a init
    conv = np.real(np.fft.ifft(np.fft.fft(a, n=L) * np.fft.fft(a, n=L)))[:2*n-1]
    best_obj = conv.max()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
        cur_max = conv.max()
        # compute smooth weights
        t = it / max_iters
        alpha = alpha_start * (alpha_end/alpha_start)**t
        # compute softmax
        scaled = alpha * conv
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()
        fft_w = np.fft.fft(w, n=L)
        # compute gradient
        grad = _gradient(a, fft_a, w, fft_w, phase)
        # Adam parameters
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)
        # evaluate new
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[:2*n-1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            print(f"Adam iter {it}: improved obj={obj_new:.8g}, score={2*n*obj_new:.6f}")
        a = a_new
    return best_seq, best_obj

def _subgrad_opt(a_init, L, phase, max_iters, deadline, lr0, epsilon):
    a = a_init.copy()
    n = len(a)
    best_seq = a.copy()
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
    best_obj = conv.max()
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv again (maybe reuse)
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
        max_val = conv.max()
        # define mask indices within tolerance
        mask = conv >= max_val - epsilon
        # binary weights uniform over mask
        w = np.zeros_like(conv)
        if np.any(mask):
            w[mask] = 1.0 / mask.sum()
        else:
            w[conv.argmax()] = 1.0
        # gradient
        fft_w = np.fft.fft(w, n=L)
        grad = _gradient(a, fft_a, w, fft_w, phase)
        # simple gradient descent with decreasing lr
        lr = lr0 / np.sqrt(it)
        a_new = a - lr * grad
        a_new = _project_simplex(a_new)
        # evaluate new
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[:2*n-1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            print(f"Subgrad iter {it}: improved obj={obj_new:.8g}, score={2*n*obj_new:.6f}")
        a = a_new
    return best_seq, best_obj

def _local_search(a_init, L, deadline):
    a_current = a_init.copy()
    n = len(a_current)
    fft_a = np.fft.fft(a_current, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
    best_obj = conv.max()
    best_seq = a_current.copy()
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # select source idx i weighted by mass
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = a_current[i] * np.random.rand() * 0.02  # up to 2%
        if delta <= 0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # projection not needed as mass conserved and nonneg ensured (a_new[i] >= 0)
        # evaluate
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[:2*n-1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            a_current = a_new
            print(f"Local move {iter_count}: improved obj={obj_new:.8g}, score={2*n*obj_new:.6f}")
    return best_seq, best_obj

"""
Propose a step‑function minimizing

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is divided into four stages:

1. **Broad parametric sweep** – explores a variety of simple analytic families
   (power‑law, exponential, Gaussian, sin‑powered, …) and a few random
   Dirichlet draws.  Every candidate is normalised to unit sum and the best
   (smallest convolution maximum) is kept.

2. **Adam optimisation on a smooth approximation of the ∞‑norm** – the true
   objective `max(conv)` is replaced by a soft‑max
   `log‑sum‑exp(α·conv)/α`.  The smoothing parameter α is increased
   exponentially (2 → 800) so the landscape starts smooth and ends
   essentially hard.  Adam updates are performed on the probability simplex
   (projection after every step).  The number of iterations is chosen
   adaptively from the remaining time budget.

3. **Sub‑gradient descent on the exact max** – after the soft‑max phase we
   switch to a true sub‑gradient (weights are uniform on the active
   constraints where the convolution attains its maximum).  A simple
   decreasing step‑size (`lr ∝ 1/√t`) together with simplex projection
   further polishes the solution.

4. **Cheap local random moves** – a hill‑climbing phase that repeatedly
   transfers a tiny random fraction of mass from a heavy entry to a random
   entry; any improvement is kept.  This helps to escape tiny plateaus.

All stages respect the global time budget (default 1000 s).  The final
sequence is scaled to the normalisation used by the reference code
(`sum = sqrt(2·n)`) and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int):
    """Return (max(conv(a,a)), full convolution) using FFT of length L."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv.max(), conv

# ----------------------------------------------------------------------
def _gradient(a: np.ndarray, fft_a: np.ndarray, w: np.ndarray,
              fft_w: np.ndarray, phase: np.ndarray) -> np.ndarray:
    """
    Gradient of the (soft‑)max of the convolution w.r.t. a.
    Uses the identity grad = 2 * corr(w, a) computed via FFT.
    Returns a vector of length n.
    """
    n = a.shape[0]
    fft_a_rev = np.conj(fft_a) * phase                # FFT of reversed a
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]

# ----------------------------------------------------------------------
def _adam_opt(a_init: np.ndarray, L: int, phase: np.ndarray,
              max_iters: int, deadline: float,
              alpha_start: float, alpha_end: float, lr0: float):
    """Adam on the soft‑max approximation of max(conv)."""
    a = a_init.copy()
    n = a.shape[0]
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # keep best ever
    best_seq = a.copy()
    best_obj, _ = _conv_max(a, L)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current convolution
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

        # smooth‑max weights (log‑sum‑exp trick)
        t = it / max_iters
        alpha = alpha_start * (alpha_end / alpha_start) ** t
        scaled = alpha * conv
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()
        fft_w = np.fft.fft(w, n=L)

        # gradient
        grad = _gradient(a, fft_a, w, fft_w, phase)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _project_simplex(a - step)

        # evaluate new point
        obj_new, _ = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj, best_seq = obj_new, a_new.copy()
            print(f"Adam iter {it:5d}: improved obj={obj_new:.8e}  score={2*n*obj_new:.6f}")

        a = a_new

    return best_seq, best_obj

# ----------------------------------------------------------------------
def _subgrad_opt(a_init: np.ndarray, L: int, phase: np.ndarray,
                 max_iters: int, deadline: float,
                 lr0: float, eps_mask: float):
    """Sub‑gradient descent on the true max(conv) (binary active set)."""
    a = a_init.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_obj, conv = _conv_max(a, L)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # recompute convolution (needed because a changes)
        obj, conv = _conv_max(a, L)
        max_val = conv.max()
        # active set (indices where conv is (almost) maximal)
        mask = np.isclose(conv, max_val, atol=eps_mask)
        w = np.zeros_like(conv)
        if mask.any():
            w[mask] = 1.0 / mask.sum()
        else:
            w[conv.argmax()] = 1.0

        fft_a = np.fft.fft(a, n=L)
        fft_w = np.fft.fft(w, n=L)
        grad = _gradient(a, fft_a, w, fft_w, phase)

        # decreasing step size
        lr = lr0 / math.sqrt(it)
        a_new = _project_simplex(a - lr * grad)

        obj_new, _ = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj, best_seq = obj_new, a_new.copy()
            print(f"Subgrad iter {it:5d}: improved obj={obj_new:.8e}  score={2*n*obj_new:.6f}")

        a = a_new

    return best_seq, best_obj

# ----------------------------------------------------------------------
def _local_search(a_init: np.ndarray, L: int, deadline: float):
    """Very cheap hill‑climbing by moving a tiny fraction of mass."""
    a_cur = a_init.copy()
    n = a_cur.shape[0]
    best_seq = a_cur.copy()
    best_obj, _ = _conv_max(a_cur, L)
    move_cnt = 0

    while time.time() < deadline:
        move_cnt += 1
        # pick source proportional to its mass (large entries more likely)
        i = np.random.choice(n, p=a_cur)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = a_cur[i] * np.random.rand() * 0.02   # ≤2 % of a[i]
        if delta <= 0:
            continue
        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta

        obj_new, _ = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj, best_seq = obj_new, a_new.copy()
            a_cur = a_new
            print(f"Local move {move_cnt:5d}: improved obj={obj_new:.8e}  score={2*n*obj_new:.6f}")

    return best_seq, best_obj
