def smoothed_gradient(a, b, gamma):
    # b = a*a convolution length 2n-1
    max_b = b.max()
    # subtract max for numerical stability
    w = np.exp(gamma * (b - max_b))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(b, 1.0 / b.size)
    else:
        w = w / w_sum
    # w is length 2n-1
    # compute gradient = 2 * cross-correlation of a with w
    # use conv of reversed a and w
    a_rev = a[::-1]
    size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
    A = np.fft.rfft(a_rev, n=size)
    W = np.fft.rfft(w, n=size)
    conv = np.fft.irfft(A * W, n=size)[:len(w)+len(a_rev)-1]
    # gradient entries:
    grad = 2.0 * conv[n-1:n-1+len(a)]
    return grad

def exp_gradient_update(a, grad, eta):
    # a>0 maybe zero entries; we treat zero as epsilon to avoid stagnation
    a = np.maximum(a, 1e-12)
    a = a * np.exp(-eta * grad)
    total = a.sum()
    if total == 0:
        a = np.full_like(a, 1.0/len(a))
    else:
        a /= total
    return a

import time
import numpy as np

def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Compute the full (linear) convolution a * a via FFT."""
    n = a.shape[0]
    fft_sz = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_sz)
    b = np.fft.irfft(A * A, n=fft_sz)[: 2 * n - 1]
    return b

def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Given a (length n) and weight vector w (length 2n‑1) compute
    ∇_a  Σ_k w_k (a*a)_k = 2 * (a ⋆ w) where ⋆ denotes cross‑correlation."""
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = len(w) + len(a_rev) - 1
    fft_sz = 1 << (conv_len.bit_length())
    A_rev = np.fft.rfft(a_rev, n=fft_sz)
    W = np.fft.rfft(w, n=fft_sz)
    conv = np.fft.irfft(A_rev * W, n=fft_sz)[:conv_len]
    # cross‑correlation value for index i is conv[n‑1+i]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad

def exp_gradient_optimize(a0: np.ndarray, deadline: float,
                          max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the smoothed max of the self‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = np.inf
    it = 0
    # pre‑compute a few constants
    while it < max_iters and time.time() < deadline:
        it += 1
        # convolution and current score
        b = full_convolution(a)
        max_b = b.max()
        # (a.sum() is kept equal to 1 by construction)
        score = 2.0 * n * max_b / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[exp_gd it {it}] new best score = {score:.6f}")

        # ==== smooth approximation of the max ====
        # gamma grows slowly – early iterations get a very smooth proxy,
        # later iterations concentrate on the true peak.
        gamma = min(200.0, 8.0 + 0.02 * it)
        # numeric stabilisation – subtract the maximum before exponentiating
        w = np.exp(gamma * (b - max_b))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = w / w_sum

        # gradient of the smooth objective
        grad = gradient_from_weights(a, w)

        # exponentiated‑gradient step (step size decays like 1/√t)
        eta = 0.5 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total
    return best_seq, best_score

def subgradient_descent(a0: np.ndarray, max_iters: int,
                        step0: float, deadline: float) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact peak (the code from the
    baseline is kept largely unchanged)."""
    a = a0.copy()
    n = a.shape[0]
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    fft_sz = 1 << ((2 * n - 1).bit_length())
    best_seq = a.copy()
    best_score = np.inf

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A = np.fft.rfft(a, n=fft_sz)
        b = np.fft.irfft(A * A, n=fft_sz)[: 2 * n - 1]
        max_b = b.max()
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[subgrad it {it}] new best score = {score:.6f}")

        # sub‑gradient of the active peak
        k = int(np.argmax(b))
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        for i in range(i_low, i_high + 1):
            j = k - i
            grad[i] = 2.0 * a[j]

        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

    return best_seq, best_score

"""Optimization of a non‑negative step‑function to minimise
    2·n·max(conv(a,a)) / (sum(a)**2).

The core search mixes two complementary phases:

* a smooth exponentiated‑gradient descent that minimises a
  soft‑max approximation of the convolution peak.  The soft‑max
  temperature (γ) grows slowly, letting the optimiser first flatten
  the whole spectrum and later concentrate on the actual peak.

* a short projected sub‑gradient polish (the baseline method) that
  directly reduces the currently active peak.

Several independent restarts are performed until the allotted wall‑clock
time expires.  The best sequence found is returned as a plain list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Geometry helpers
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using FFT."""
    n = a.shape[0]
    fft_sz = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_sz)
    b = np.fft.irfft(A * A, n=fft_sz)[: 2 * n - 1]
    return b


def gradient_from_weights(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Given a (length n) and a weight vector w (length 2n‑1) compute the
    gradient of Σ_k w_k (a*a)_k, i.e.  ∇_a = 2·(a ⋆ w) where ⋆ denotes
    cross‑correlation.  The result has length n.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    conv_len = len(w) + len(a_rev) - 1
    fft_sz = 1 << (conv_len.bit_length())
    A_rev = np.fft.rfft(a_rev, n=fft_sz)
    W = np.fft.rfft(w, n=fft_sz)
    conv = np.fft.irfft(A_rev * W, n=fft_sz)[:conv_len]
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def exp_gradient_optimize(
    a0: np.ndarray, deadline: float, max_iters: int = 25000
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smoothed max of the
    self‑convolution.  Returns the best sequence encountered and its
    corresponding score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = np.inf
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1

        # current convolution and score
        b = full_convolution(a)
        max_b = b.max()
        score = 2.0 * n * max_b / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[exp_gd it {it}] new best score = {score:.6f}")

        # ----- smooth approximation of the max -----
        gamma = min(200.0, 8.0 + 0.02 * it)          # grow slowly
        w = np.exp(gamma * (b - max_b))            # numeric stability
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            w = w / w_sum

        # gradient of the smooth objective
        grad = gradient_from_weights(a, w)

        # exponentiated‑gradient step (step size ≈ 1/√t)
        eta = 0.5 / np.sqrt(it)
        a = a * np.exp(-eta * grad)

        total = a.sum()
        if total <= 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / total

    return best_seq, best_score


def subgradient_descent(
    a0: np.ndarray, max_iters: int, step0: float, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent directly on the peak of the
    self‑convolution (the baseline routine).
    """
    a = a0.copy()
    n = a.shape[0]
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    fft_sz = 1 << ((2 * n - 1).bit_length())
    best_seq = a.copy()
    best_score = np.inf

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        A = np.fft.rfft(a, n=fft_sz)
        b = np.fft.irfft(A * A, n=fft_sz)[: 2 * n - 1]
        max_b = b.max()
        sum_a = a.sum()
        score = 2.0 * n * max_b / (sum_a * sum_a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[subgrad it {it}] new best score = {score:.6f}")

        # sub‑gradient of the active peak
        k = int(np.argmax(b))
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        for i in range(i_low, i_high + 1):
            j = k - i
            grad[i] = 2.0 * a[j]

        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

    return best_seq, best_score
