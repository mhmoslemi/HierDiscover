    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}.

    Implementation of the O(n) algorithm from
    Duchi, Shalev‑Shwartz, Singer, Chandra (2008).
    """
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]

    return w
