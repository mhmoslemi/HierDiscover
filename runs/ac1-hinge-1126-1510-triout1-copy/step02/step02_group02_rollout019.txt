def adam_descent(a, L, max_iters, deadline):
    n = len(a)
    best_seq = a.copy()
    best_obj = np.real(np.fft.ifft(np.fft.fft(a, n=L) * np.fft.fft(a, n=L)))[:2*n-1].max()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps=1e-8
    lr0=0.07
    alpha_start, alpha_end = 4.0, 250.0
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute fft etc.
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
        cur_max = conv.max()
        # compute softmax weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        scaled = alpha * conv
        shift = scaled.max()
        w = np.exp(scaled - shift)
        w /= w.sum()
        # compute reversed fft of a (phase factor)
        phase = np.exp(2j * np.pi * np.arange(L) / L)
        fft_a_rev = np.conj(fft_a) * phase
        fft_w = np.fft.fft(w, n=L)
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n-1:n-1+n]   # length n
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad*grad)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_simplex(a_new)
        # evaluate
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[:2*n-1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            print(f"Iter {it}: improved obj={obj_new:.8f} score={2*n*obj_new:.6f}")
        a = a_new
    return best_seq, best_obj

        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
        k_max = conv.argmax()
        # compute contributions
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >=0) & (j_vals < n)
        i_mask = i_vals[mask]
        j_mask = j_vals[mask]
        if i_mask.size == 0:
            continue
        contrib = a[i_mask] * a[j_mask]
        idx = contrib.argmax()
        i_star = i_mask[idx]
        # optional j_star = j_mask[idx] if needed
        # choose a random target index
        t = np.random.randint(0, n)
        if t == i_star:
            continue
        delta = 0.05 * a[i_star]  # move 5%
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i_star] -= delta
        a_new[t] += delta
        # ensure nonnegativity (already ensured)
        # evaluate
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[:2*n-1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            a = a_new
            print(f"Coord step {step}: improved obj={obj_new:.8f} score={2*n*obj_new:.6f}")

"""
Search for a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in several stages:

1.  **Initialisation** – a few candidate lengths are tried (including the
    length of a previously found solution, if any).  For each length a large
    family of parametric shapes (power‑law, exponential, double‑exponential,
    sinusoidal, cosine‑squared) as well as a handful of random Dirichlet draws
    are generated, normalised to the simplex and evaluated.  The best of these
    becomes the starting point for the next stage.

2.  **Adam optimisation of a smooth‑max** – the non‑smooth `max(conv)`
    is approximated by a soft‑max
    `log‑sum‑exp(α·conv) / α`.  The gradient of this surrogate is a
    correlation that can be computed by a few FFTs.  Adam updates are applied
    on the probability simplex (Euclidean projection after each step).  The
    smoothing parameter α is increased linearly from a small to a large value,
    turning the landscape from smooth to the true supremum norm.

3.  **Greedy coordinate descent** – after Adam, the current maximal
    convolution entry is identified.  The pair of indices contributing most
    to that entry is located and a small amount of mass is moved away from the
    larger index to a random location.  Any improvement is kept; this
    aggressively flattens the peak.

4.  **Cheap random hill‑climbing** – finally many tiny random transfers of
    mass (up to 2 % of a source entry) are performed; improvements are kept.
    This helps to escape the remaining plateaus.

All stages respect the supplied time budget (default 1000 s).  The best
sequence found (scaled so that `sum = sqrt(2·n)`) is returned as a plain
Python list, ready to be handed to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _max_conv(a: np.ndarray, L: int) -> float:
    """Return max(conv(a, a)) for a vector a that sums to 1."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv.max()


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Smooth‑max weights: softmax(alpha * conv).  Returns a vector that sums to 1."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    w = exp_scaled / exp_scaled.sum()
    return w


# ----------------------------------------------------------------------
def _adam_descent(a: np.ndarray,
                  L: int,
                  max_iters: int,
                  deadline: float,
                  lr0: float = 0.08,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  alpha_start: float = 4.0,
                  alpha_end: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smooth‑max surrogate.
    Returns the best vector seen and its corresponding `max(conv)`.
    """
    n = a.shape[0]
    phase = np.exp(2j * np.pi * np.arange(L) / L)          # reversal factor
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # current best
    best_a = a.copy()
    best_obj = _max_conv(a, L)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # convolution of a with itself
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

        # smooth‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of smooth‑max (2 * correlation)
        fft_a_rev = np.conj(fft_a) * phase          # reversed a in Fourier domain
        fft_w = np.fft.fft(w, n=L)
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n - 1: n - 1 + n]          # same length as a

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate new point
        obj_new = _max_conv(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            print(f"Adam iter {it:5d}: improved max(conv)={obj_new:.8e} "
                  f"score={2 * n * obj_new:.6f}")

        a = a_new
    return best_a, best_obj


# ----------------------------------------------------------------------
def _coordinate_descent(a: np.ndarray,
                        L: int,
                        max_steps: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy coordinate‑descent that repeatedly removes mass from the index
    most responsible for the current maximal convolution entry.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_obj = _max_conv(a, L)

    for step in range(max_steps):
        if time.time() > deadline:
            break
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        k_max = int(conv.argmax())

        # all pairs (i, j) with i + j = k_max
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        i_mask = i_vals[mask]
        j_mask = j_vals[mask]
        if i_mask.size == 0:
            break

        # contribution of each pair
        contrib = a[i_mask] * a[j_mask]
        idx = int(contrib.argmax())
        i_star = int(i_mask[idx])

        # move a small fraction of mass from i_star to a random position
        tgt = np.random.randint(0, n)
        if tgt == i_star:
            continue
        delta = 0.05 * a[i_star]          # 5 % of the source entry
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i_star] -= delta
        a_new[tgt] += delta
        if a_new[i_star] < 0.0:
            continue

        obj_new = _max_conv(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            print(f"Coord step {step:4d}: improved max(conv)={obj_new:.8e} "
                  f"score={2 * n * obj_new:.6f}")
        else:
            # keep the old a (no acceptance)
            pass
    return best_a, best_obj


# ----------------------------------------------------------------------
def _random_hill_climb(a: np.ndarray,
                       L: int,
                       deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic hill‑climbing: move a tiny random amount of mass from a
    source drawn proportionally to its weight to a uniformly random target.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_obj = _max_conv(a, L)

    while time.time() < deadline:
        # source index – proportional to current mass
        src = np.random.choice(n, p=a)
        tgt = np.random.randint(0, n)
        if src == tgt:
            continue
        delta = np.random.rand() * 0.02 * a[src]   # ≤ 2 % of source
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        if a_new[src] < 0.0:
            continue

        obj_new = _max_conv(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            print(f"Random move: improved max(conv)={obj_new:.8e} "
                  f"score={2 * n * obj_new:.6f}")
    return best_a, best_obj

def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) with a low evaluation
    score.  The function respects the supplied time budget and returns a plain
    Python list of floats scaled to the reference normalisation (sum = sqrt(2·n)).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Determine the lengths that will be examined.
    lengths = [2000, 3000, 4000, 5659, 8000]        # a few diverse choices
    try:
        # If a previous construction exists, prefer its length first.
        prev = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        lengths = [len(prev)] + [l for l in lengths if l != len(prev)]
    except Exception:
        prev = None

    best_global_seq = None
    best_global_score = float("inf")
    best_global_n = None

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (sum normalised to 1) and possibly update global best.
    def _maybe_update(a_norm: np.ndarray, n: int):
        nonlocal best_global_seq, best_global_score, best_global_n
        obj = _max_conv(a_norm, L=n_fft[n])
        score = 2.0 * n * obj
        if score < best_global_score:
            best_global_score = score
            best_global_seq = a_norm.copy()
            best_global_n = n
            print(f"New global best: n={n}  max(conv)={obj:.8e}  score={score:.6f}")

    # Pre‑compute FFT lengths for all n that will be used.
    n_fft = {}
    for n in lengths:
        # length must be at least 2n‑1; we use a power of two a bit larger.
        L = 1 << ((3 * n - 2).bit_length())
        n_fft[n] = L

    # ------------------------------------------------------------------
    # Main loop over the selected lengths.
    for n in lengths:
        if time.time() > deadline:
            break
        L = n_fft[n]
        target_sum = math.sqrt(2.0 * n)

        # ---- 0. start from a previous solution if it matches the length ----
        if prev is not None and len(prev) == n:
            a0 = prev.astype(np.float64)
            a0 /= a0.sum()
            _maybe_update(a0, n)
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
            _maybe_update(a0, n)

        # ---- 1. parametric sweep -------------------------------------------------
        print(f"\n--- Length {n}: parametric sweep ---")
        # power‑law (decreasing & increasing)
        for p in np.linspace(0.0, 4.0, 41):
            if time.time() > deadline:
                break
            base = np.arange(1, n + 1, dtype=np.float64) ** (-p)
            base /= base.sum()
            _maybe_update(base, n)
            _maybe_update(base[::-1], n)

        # pure exponential decay (and its mirror)
        for lam in np.linspace(0.0, 6.0, 31):
            if time.time() > deadline:
                break
            exp_shape = np.exp(-lam * np.arange(n, dtype=np.float64))
            exp_shape /= exp_shape.sum()
            _maybe_update(exp_shape, n)
            _maybe_update(exp_shape[::-1], n)

        # double exponential (U‑shaped)
        for lam in np.linspace(0.0, 6.0, 21):
            if time.time() > deadline:
                break
            left = np.exp(-lam * np.arange(n, dtype=np.float64))
            right = left[::-1]
            u_shape = left + right
            u_shape /= u_shape.sum()
            _maybe_update(u_shape, n)

        # sinusoidal power‑law
        for p in np.linspace(1.0, 4.0, 7):
            if time.time() > deadline:
                break
            s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
            s /= s.sum()
            _maybe_update(s, n)

        # cosine‑squared family
        for p in np.linspace(1.0, 4.0, 7):
            if time.time() > deadline:
                break
            c = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1))) ** p
            c /= c.sum()
            _maybe_update(c, n)

        # random Dirichlet draws
        for _ in range(30):
            if time.time() > deadline:
                break
            r = np.random.rand(n)
            r /= r.sum()
            _maybe_update(r, n)

        # ------------------------------------------------------------------
        # 2. Adam descent on smooth‑max ----------------------------------------
        print(f"--- Length {n}: Adam descent ---")
        a_adam, obj_adam = _adam_descent(best_global_seq if best_global_n == n else a0,
                                         L,
                                         max_iters=20000,
                                         deadline=deadline,
                                         lr0=0.07,
                                         alpha_start=4.0,
                                         alpha_end=300.0)
        _maybe_update(a_adam, n)

        # ------------------------------------------------------------------
        # 3. Greedy coordinate descent -----------------------------------------
        print(f"--- Length {n}: coordinate descent ---")
        a_coord, obj_coord = _coordinate_descent(a_adam, L,
                                                max_steps=4000,
                                                deadline=deadline)
        _maybe_update(a_coord, n)

        # ------------------------------------------------------------------
        # 4. Random local moves ------------------------------------------------
        print(f"--- Length {n}: random moves ---")
        a_final, obj_final = _random_hill_climb(a_coord, L, deadline)
        _maybe_update(a_final, n)

    # ------------------------------------------------------------------
    # No candidate found (should not happen) – fall back to uniform.
    if best_global_seq is None:
        n = 2000
        best_global_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_global_n = n

    # ------------------------------------------------------------------
    # Scale to the reference normalisation (sum = sqrt(2·n)) and return.
    final_seq = (best_global_seq * math.sqrt(2.0 * best_global_n)).tolist()
    # One final evaluation just for the log line (can be omitted if not needed)
    try:
        final_score = evaluate_sequence(final_seq)  # type: ignore
        print(f"\n--- Finished: best score = {final_score:.6f} (n={best_global_n}) ---")
    except Exception:
        pass
    return final_seq
