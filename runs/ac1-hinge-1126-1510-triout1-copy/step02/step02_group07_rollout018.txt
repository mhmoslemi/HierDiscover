Hybrid optimiser for the self–convolution bound.

* Initialise from the best known construction (height_sequence_1 if present,
  otherwise the half‑order series).
* Main loop (≈1000 s) alternates three moves:
    – *projected gradient descent* on a smooth approximation of the max
      self‑convolution (soft‑max weighting, back‑tracking line‑search).
    – *random pair exchange* (small mass move between two positions, with a
      warm Metropolis acceptance).
    – *global multiplicative perturbation* (small log‑Gaussian jitter).
* Every few thousand iterations a linear‑programming step (CVXPY) finds a
  direction g that maximises the total mass under the current worst‑case
  convolution bound; the current sequence is mixed with g.
* The best sequence (lowest score) seen so far is kept and printed.
"""

def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i,
    normalised so that sum(a)=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max on the convolution vector b.
    Returns a probability vector w (sums to 1)."""
    maxb = b.max()
    # exponent is non‑positive → no overflow
    w = np.exp(alpha * (b - maxb))
    w_sum = w.sum()
    if w_sum == 0.0:
        # fall back to uniform weights
        return np.full_like(w, 1.0 / w.shape[0])
    return w / w_sum


def _convolution_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the (soft‑max) objective w.r.t. a:
        g_i = 2 * sum_j w_{i+j} * a_j.
    Implemented via a single convolution.
    """
    # convolution of the weight vector with the reversed a
    conv = np.convolve(w, a[::-1])
    n = a.shape[0]
    # slice gives exactly the required n elements
    grad = 2.0 * conv[n - 1 : n - 1 + n]
    return grad


def _lp_direction(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the linear programme:
        maximise   sum(g)
        subject to  f * g ≤ M   (component‑wise convolution)
                    g ≥ 0
    Returns the optimiser g (not normalised). If the LP fails,
    returns None.
    """
    import cvxpy as cp

    n = f.shape[0]
    # Build the Toeplitz matrix A where (A @ g)_k = (f * g)_k
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = f  # place f on the i‑th column
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        return None
    if g.value is None:
        return None
    g_opt = np.maximum(g.value, 0.0)
    return g_opt

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low self‑convolution bound.
    Returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    n = int(kwargs.get("n", 1000))

    # ------------------------------------------------------------------
    # 1) Initialise ----------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # use the globally supplied construction if it exists
        init = np.asarray(height_sequence_1, dtype=np.float64)
        if init.shape[0] != n:
            if init.shape[0] > n:
                init = init[:n]
            else:
                init = np.concatenate([init, np.zeros(n - init.shape[0])])
    except NameError:
        init = _half_order_series(n)

    init = np.maximum(init, 0.0)
    s = init.sum()
    if s == 0.0:
        cur_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        cur_seq = init / s

    # initial evaluation
    cur_conv = np.convolve(cur_seq, cur_seq)
    cur_max = cur_conv.max()
    cur_score = 2.0 * n * cur_max
    best_seq = cur_seq.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 2) Optimisation parameters
    # ------------------------------------------------------------------
    iteration = 0
    # move probabilities (must sum ≤ 1)
    prob_pair = 0.30         # small random mass exchange
    prob_global = 0.10       # multiplicative jitter move
    # gradient descent step‑size
    step = 0.10
    step_min = 1e-8
    step_max = 0.5
    # soft‑max temperature schedule (alpha grows from α₀ to α₁)
    alpha_start = 5.0
    alpha_end = 200.0
    # global jitter magnitude (decays with time)
    sigma0 = 0.02
    # linear‑programming mixing interval
    lp_interval = 20000      # iterations
    t_mix = 0.20             # mixing weight with the LP direction

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        total_allowed = deadline - start_time
        progress = elapsed / total_allowed

        # update the smoothing parameter
        alpha = alpha_start + (alpha_end - alpha_start) * progress

        # ----------------------------------------------------------------
        # 3a) Choose a move type
        # ----------------------------------------------------------------
        r = np.random.rand()
        if r < prob_pair:
            # ----- random pair exchange ---------------------------------
            i, j = np.random.choice(n, 2, replace=False)
            max_delta = min(cur_seq[i], 0.02)  # at most 2 % of total mass
            if max_delta > 0.0:
                delta = np.random.rand() * max_delta
                cand_seq = cur_seq.copy()
                cand_seq[i] -= delta
                cand_seq[j] += delta
                cand_conv = np.convolve(cand_seq, cand_seq)
                cand_score = 2.0 * n * cand_conv.max()
                # accept always if improvement, otherwise Metropolis
                if cand_score < cur_score or np.random.rand() < 0.001:
                    cur_seq = cand_seq
                    cur_score = cand_score

        elif r < prob_pair + prob_global:
            # ----- global multiplicative perturbation -------------------
            sigma = sigma0 * (1.0 - progress)  # decay
            noise = np.random.randn(n) * sigma
            cand_seq = cur_seq * np.exp(noise)   # stay positive
            cand_seq = cand_seq / cand_seq.sum()
            cand_conv = np.convolve(cand_seq, cand_seq)
            cand_score = 2.0 * n * cand_conv.max()
            if cand_score < cur_score or np.random.rand() < 0.001:
                cur_seq = cand_seq
                cur_score = cand_score

        else:
            # ----- projected gradient descent on smoothed max ------------
            # compute current convolution and soft‑max weights
            b = np.convolve(cur_seq, cur_seq)
            w = _softmax_weights(b, alpha)

            # gradient of the smoothed objective
            grad = _convolution_gradient(cur_seq, w)

            # back‑tracking line search on the true max‑objective
            trial_step = step
            improved = False
            for _ in range(5):
                cand_seq = cur_seq - trial_step * grad
                cand_seq = _project_simplex(cand_seq)
                cand_conv = np.convolve(cand_seq, cand_seq)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < cur_score:
                    improved = True
                    cur_seq = cand_seq
                    cur_score = cand_score
                    step = min(step * 1.1, step_max)
                    break
                trial_step *= 0.5
            if not improved:
                step = max(step * 0.7, step_min)

        # ----------------------------------------------------------------
        # 3b) Occasionally solve the LP direction and mix in
        # ----------------------------------------------------------------
        if iteration % lp_interval == 0:
            # current worst‑case bound
            M = np.convolve(cur_seq, cur_seq).max()
            g_opt = _lp_direction(cur_seq, M)
            if g_opt is not None and g_opt.sum() > 0:
                # normalise the direction
                g_opt = g_opt / g_opt.sum()
                cand_seq = (1.0 - t_mix) * cur_seq + t_mix * g_opt
                cand_seq = cand_seq / cand_seq.sum()
                cand_conv = np.convolve(cand_seq, cand_seq)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < cur_score:
                    cur_seq = cand_seq
                    cur_score = cand_score
                    print(
                        f"[{iteration:6d}] LP mix improvement: score {cand_score:.6f}"
                    )

        # ----------------------------------------------------------------
        # 3c) Record the best solution found so far
        # ----------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{iteration:6d}] New best score: {best_score:.6f}")

        # occasional progress report
        if iteration % 5000 == 0:
            print(
                f"[{iteration:6d}] elapsed {elapsed:.1f}s / {budget_s:.0f}s, "
                f"best {best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 4) Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
