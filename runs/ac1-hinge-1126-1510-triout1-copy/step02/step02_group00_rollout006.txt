"""
Propose a low‑score non‑negative step sequence.

Algorithm overview
------------------
1. **Initialisation** – use a provided global sequence (`height_sequence_1`) or a uniform
   distribution, normalised to the probability simplex (sum = 1).

2. **Smoothed‑max L‑BFGS** – repeatedly run L‑BFGS on a log‑sum‑exp
   approximation of the max‑convolution.  The smoothing factor (`lam_factor`)
   is gradually reduced (0.5 → 0.01) and many random restarts are performed.
   After each optimisation a short greedy hill‑climb is applied.

3. **Exponentiated‑gradient sub‑gradient descent** – a mirror‑descent step
   on the true max‑objective, using a multiplicative update that stays on the
   simplex.  The step‑size decays like 1/√t.

4. **Active‑set linear‑program refinement** – for the current sequence `f`
   collect the *active* convolution indices (those attaining the maximum).
   Solve a tiny LP that maximises the total mass of a new sequence `b` under
   the linear constraints `f * b ≤ M` (where `M` is the current max).  Mixing
   `b` into `f` (convex combination) reduces the maximal convolution value.

5. **Final greedy hill‑climb** – transfer small fractions of mass between
   coordinates while preserving the sum, accepting only improving moves.

All phases respect the global time budget (default 1000 s).  The final
sequence is returned as a plain Python list; `evaluate_sequence` will be
called by the host to obtain the true score.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution with sum = 1."""
    conv = full_conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float
        2·n·smooth_max(conv)
    grad_y : np.ndarray
        Gradient w.r.t. y (shape n)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv
    w = exp_shift / Z
    # gradient w.r.t. a: convolution of w with reversed a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fw * fa_rev))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient descent on the true max
# ----------------------------------------------------------------------
def exp_gradient_descent(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Mirror‑descent on the true max‑convolution using a multiplicative update.
    Each iteration projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active mask (binding constraints)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max (see main code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step‑size schedule (η ↓)
        eta = 0.5 / np.sqrt(it + 1.0)

        # multiplicative (exponentiated) update, clipped to avoid overflow
        delta = -eta * g
        delta = np.clip(delta, -50, 50)
        a = a * np.exp(delta)
        # renormalise to the simplex
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # keep best
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another, accept only
    improvements of the fast (max‑convolution) score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Active‑set linear‑program refinement
# ----------------------------------------------------------------------
def lp_refine(f: np.ndarray,
              M: float,
              act_idx: np.ndarray,
              n: int) -> np.ndarray | None:
    """
    Solve a tiny LP that maximises Σ b_j subject to the linear constraints
    (f * b)_k ≤ M for all k in the active set `act_idx`.
    Returns the optimal vector `b` (may be None if LP fails).
    """
    if act_idx.size == 0:
        return None

    rows = []
    for k in act_idx:
        # Build one row of the constraint matrix:
        # (f * b)_k = Σ_i f_i * b_{k-i}
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = k - i_vals
        row = np.zeros(n, dtype=np.float64)
        row[j_vals] = f[i_vals]
        rows.append(row)

    A = np.vstack(rows)                # shape (|act|, n)
    b_ub = np.full(A.shape[0], M)
    c = -np.ones(n)                    # maximise Σ b_j → minimise -Σ b_j

    # Use the high‑performance HiGHS interior‑point solver
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs')
    if not res.success:
        return None
    return res.x

    # ------------------------------------------------------------------

            a_tmp[:m_edge] = a_edge

                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS #{iteration}] lam={lam:.3f} mode={mode} "
                  f"score={best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient sub‑gradient descent
    # ------------------------------------------------------------------
    grad_deadline = deadline - 150.0   # keep ~150 s for LP + final hill‑climb
    best_a, best_score = exp_gradient_descent(best_a,
                                              grad_deadline,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
    print(f"[exp‑grad] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – active‑set LP refinement (few iterations)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 80.0   # keep ~80 s for final hill‑climb
    while time.time() < lp_deadline:
        # current convolution and max
        fa = np.fft.fft(best_a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()
        eps = M * 1e-12 + 1e-12
        act_idx = np.where(conv >= M - eps)[0]

        # solve tiny LP
        b_opt = lp_refine(best_a, M, act_idx, n)
        if b_opt is None:
            break

        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-9:
            # No improvement possible
            break

        # normalise to keep sum = 1
        g = b_opt / sum_b

        # try a few mixing coefficients
        improved = False
        for t in [0.5, 0.3, 0.1]:
            cand = (1.0 - t) * best_a + t * g
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score:
                best_a = cand
                best_score = cand_score
                improved = True
                print(f"[LP] mixing t={t:.2f} score={best_score:.6f}")
                break
        if not improved:
            # LP gave no benefit; exit the loop
            break

    # ------------------------------------------------------------------
    # Phase 4 – final aggressive hill‑climb
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=5000)
    if final_score < best_score:
        best_a = final_a
        best_score = final_score

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # Optional sanity evaluation (the host will do this anyway)
    try:
        eval_score = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {eval_score:.6f}")
    except Exception as e:
        print(f"[final] evaluation failed: {e}")

    return final_seq
