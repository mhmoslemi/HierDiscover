"""
Hybrid geometric‑power‑window search + exponentiated‑gradient descent.

The routine

* builds a diverse set of candidate step functions (half‑order,
  exponential tilt, power‑law tilt, sine / cosine windows, triangular,
  and mixtures with the uniform distribution),
* picks the best among them,
* then refines the winner by a smooth‑max / exponentiated‑gradient
  descent on the probability simplex (the smooth max is the
  log‑sum‑exp of the self‑convolution values, annealed with a decreasing
  temperature schedule),
* finally decorates the result with a few stochastic pair‑exchanges to
  escape tiny plateaux.

All convolutions are performed with FFT (O(n log n)) and the vector is
kept on the simplex by a simple Euclidean projection.  The algorithm
stops automatically when the allocated time budget (default 1000 s) is
exhausted and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """First n coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full self‑convolution using real FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[: 2 * n - 1]
    # numerical noise may produce tiny negatives – clip
    np.maximum(conv, 0, out=conv)
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution."""
    return _conv_full(a).max()


def _score(a: np.ndarray, n: int) -> float:
    """Exact score used by evaluate_sequence (sum‑normalised)."""
    s = a.sum()
    if s < 0.01:
        return float("inf")
    M = _conv_max(a)
    return 2.0 * n * M / (s * s)


def _softmax(x: np.ndarray, tau: float) -> np.ndarray:
    """Numerically stable softmax of x / tau."""
    # shift by max to avoid overflow
    shift = x.max()
    y = np.exp((x - shift) / tau)
    return y / y.sum()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a low
    evaluate_sequence score.  The routine respects the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Parameterisation (length of the sequence)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1000))
    # In practice n≈1000 – larger n gives more freedom but costs more.
    # The algorithm will automatically stop when the time budget expires.

    # ------------------------------------------------------------------
    # 1) Rich family of hand‑crafted candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_sc = float("inf")
    best_M = None

    def consider(candidate: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_sc, best_M
        cand = candidate.astype(np.float64)
        cand = np.clip(cand, 0, None)
        tot = cand.sum()
        if tot < 1e-12:
            return
        cand /= tot  # normalise to sum = 1
        sc = _score(cand, n)
        if sc < best_sc:
            best_sc = sc
            best_seq = cand.copy()
            best_M = _conv_max(cand)
            print(f"[init] {tag:30s} → score {sc:.6f}")

    # half‑order (central binomial) baseline
    base = _half_order_series(n)
    consider(base, "half_order")

    # exponent tilt of the half‑order series: a_i *= (i+1)^(-δ)
    for delta in np.linspace(0.0, 1.2, 13):
        tilt = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = base * tilt
        consider(cand, f"half_tilt δ={delta:.2f}")

    # pure exponential tilt (geometric series)
    for beta in np.linspace(0.0, 0.01, 11):               # a_i ∝ exp(-beta·i)
        expo = np.exp(-beta * np.arange(n, dtype=np.float64))
        consider(expo, f"exp_tilt β={beta:.4f}")

    # power‑law tilt (a_i ∝ (i+1)^(-p))
    for p in np.linspace(0.2, 2.0, 19):
        pw = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        consider(pw, f"pow_tilt p={p:.2f}")

    # sine window (positive sine)
    sine = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    consider(sine, "sine_window")

    # raised‑cosine (Hann) window
    hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    consider(hann, "hann_window")

    # triangular window
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    consider(tri, "triangular")

    # mixtures with the uniform distribution
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in np.linspace(0.0, 1.0, 11):
        cand = (1.0 - alpha) * best_seq + alpha * uniform
        consider(cand, f"mix_uniform α={alpha:.2f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max exponentiated‑gradient descent on the simplex
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_sc = best_sc
    cur_M = best_M
    # temperature (smooth‑max) schedule – decreasing geometric progression
    tau_schedule = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    # global step size scaling (η = τ * factor)
    eta_factor = 1.0

    for tau in tau_schedule:
        # limit the inner loop by time, not by a fixed iteration count
        it = 0
        while time.time() < deadline:
            it += 1
            # full convolution and smooth‑max weights
            conv = _conv_full(a)
            M = conv.max()
            # keep the best ever seen
            sc = 2.0 * n * M / (a.sum() ** 2)
            if sc < best_sc:
                best_sc = sc
                best_seq = a.copy()
                best_M = M
                print(f"[grad] tau={tau:.3f} it={it:5d} → new best {best_sc:.6f}")

            # softmax over convolution entries (smooth approximation of max)
            w = _softmax(conv, tau)          # shape (2n‑1,)
            # gradient of the smooth max w.r.t. a:
            #    ∂/∂a_i  Σ_k w_k (a*a)_k  = 2 * Σ_k w_k a_{k‑i}
            grad = 2.0 * np.correlate(w, a, mode='valid')   # length n

            # exponentiated‑gradient update (mirror descent on the simplex)
            eta = eta_factor * tau
            a_new = a * np.exp(-eta * grad)
            a_new = _project_simplex(a_new)

            # accept if improvement, otherwise keep old a (no back‑track)
            new_M = _conv_max(a_new)
            if new_M <= M:
                a = a_new
                cur_M = new_M
                cur_sc = 2.0 * n * new_M / (a.sum() ** 2)
            else:
                # occasional stochastic acceptance (simulated‑annealing style)
                temp = tau  # using the same τ as a temperature
                prob = math.exp(-(new_M - M) / (temp + 1e-12))
                if np.random.rand() < prob:
                    a = a_new
                    cur_M = new_M
                    cur_sc = 2.0 * n * new_M / (a.sum() ** 2)

            # after a few hundred inner iterations we break to the next τ
            if it % 400 == 0:
                # occasionally inject a tiny random jitter to avoid stagnation
                if np.random.rand() < 0.2:
                    jitter = np.random.randn(n) * 1e-4
                    a = a * (1.0 + jitter)
                    a = np.clip(a, 0, None)
                    a = _project_simplex(a)
                # check time once per block
                if time.time() >= deadline:
                    break

        # decrease step‑size factor a little when temperature drops
        eta_factor = max(0.1, eta_factor * 0.9)

    # ------------------------------------------------------------------
    # 3) Final polishing with cheap random pair‑exchange moves
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_sc = best_sc
    cur_M = best_M
    move_prob = 0.7
    while time.time() < deadline:
        if np.random.rand() > move_prob:
            continue
        i, j = np.random.choice(n, 2, replace=False)
        # move a modest fraction of mass from i to j
        max_delta = min(cur_seq[i], 0.01)  # up to 1 % of total mass
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        cand = cur_seq.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_M = _conv_max(cand)
        cand_sc = 2.0 * n * cand_M / (cand.sum() ** 2)
        if cand_sc < cur_sc:
            cur_seq = cand
            cur_sc = cand_sc
            cur_M = cand_M
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_seq = cur_seq.copy()
                print(f"[swap] improved to {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (plain Python list)
    # ------------------------------------------------------------------
    print(
        f"Finished – best score = {best_sc:.6f} (max conv {best_M:.6f}) after {time.time() - start_time:.1f}s"
    )
    # Ensure we obey the evaluation routine's clipping (negative -> 0, >1000 -> 1000)
    result = np.clip(best_seq, 0.0, 1000.0).tolist()
    return result
