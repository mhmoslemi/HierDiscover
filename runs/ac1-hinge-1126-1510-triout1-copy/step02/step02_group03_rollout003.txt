def compute_gradient(a, alpha):
    b = _conv_fft(a)  # length 2N-1
    # compute softmax weights
    maxb = np.max(b)  # for numerical stability we subtract max before exponent
    exp_term = np.exp(alpha * (b - maxb))
    w = exp_term / exp_term.sum()

    g = 2.0 * corr[:len(a)]

import time

import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a using FFT (real inputs). Returns length 2n-1 array."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    prod = fa * fa
    conv = np.fft.irfft(prod, size)
    return conv[: 2 * n - 1]

def _corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Linear correlation of a (length n) with w (length 2n-1). Returns first n entries."""
    n = a.shape[0]
    L = 1 << ((n + len(w) - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    corr = np.fft.irfft(fa * np.conj(fw), L)
    # correlation at non‑negative lags starts at index 0

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear convolution a*a using FFT (real -> real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a*a (length 2·n‑1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())           # next power of two
    fa = np.fft.rfft(a, size)
    prod = fa * fa                                   # element‑wise product
    conv = np.fft.irfft(prod, size)
    return conv[: 2 * n - 1]                         # keep needed part


# ----------------------------------------------------------------------
# Helper: linear correlation of a (len n) with w (len 2·n‑1)
# ----------------------------------------------------------------------
def _corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Linear correlation (non‑negative lags) of a with w.
    Returns the first n entries, i.e.  (a ⋆ w)[j] = Σ_i a[i]·w[i+j] .
    """
    n = a.shape[0]
    L = 1 << ((n + len(w) - 1).bit_length())        # power of two >= n+len(w)-1
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    corr = np.fft.irfft(fa * np.conj(fw), L)
