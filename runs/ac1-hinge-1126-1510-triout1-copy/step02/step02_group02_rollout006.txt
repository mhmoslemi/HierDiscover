def adam_optimize(a_init, L, deadline, max_iters=100000, alpha_start=4.0, alpha_end=300.0, lr0=0.08):
    a = a_init.copy()
    best_seq = a.copy()
    best_obj = _conv_max(a, L)   # compute conv max

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it=0
    while time.time() < deadline and it < max_iters:
        it+=1
        # convolution
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
        cur_max = conv.max()
        # smoothing parameter
        alpha = alpha_start + (alpha_end-alpha_start)*(it/max_iters)  # linear
        # softmax
        scaled = alpha * conv
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()
        # gradient: 2 * correlation of w and a
        # compute correlation via FFT: corr = ifft(fft(w) * conj(fft(a))*phase)
        fft_w = np.fft.fft(w, n=L)
        # compute reversed a via conj(fft_a) * phase
        fft_a_rev = np.conj(fft_a) * phase
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n-1 : n-1 + n]    # length n
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)
        # eval new
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[:2*n-1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            print(f"Iter {it}: improved obj={obj_new:.8e} score={2*n*obj_new:.6f}")
        a = a_new
        # maybe occasionally increase lr? Not needed.

    return best_seq, best_obj, it
```

"""
Propose a step‑function (non‑negative vector) that minimises

        2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three independent stages that together
try to push the objective well below the previous best (≈1.5235).

1. **Broad parametric sweep + random Dirichlet seeds**  
   Generate a large pool of simple families (power‑law, exponential,
   double‑exponential, Gaussian, triangular, etc.) and a number of
   random Dirichlet samples.  All candidates are normalised to sum = 1
   and the best few (according to the exact max‑convolution) are kept
   for the next stage.

2. **Long Adam optimisation on a smooth approximation of the max**  
   For a chosen candidate we minimise a soft‑max of the self‑convolution
   (soft‑max(α·conv)) using Adam on the probability simplex.  The
   smoothing parameter α is increased from a small value (≈4) to a large
   value (≈500) so the method first sees a smooth landscape and then
   focuses on the true supremum.  Up to 150 000 Adam steps are performed
   (or as many as time permits) with a decaying learning‑rate.
   The best sequence seen during the run is retained.

3. **Hard‑max sub‑gradient refinement + cheap random hill‑climbing**  
   Starting from the best Adam result we run a plain sub‑gradient method
   that uses the *exact* max‑convolution (gradient = 2·reversed(a) shifted
   to the maximising index) together with a simple step‑size schedule.
   Afterwards a cheap “move a little mass’’ neighbourhood search is
   performed until the time budget expires.

All vectors are kept on the probability simplex (non‑negative,
sum = 1); the final answer is scaled to the problem’s normalisation
`sum = sqrt(2·n)`.  The code respects the 1000 s wall‑clock limit
by checking the deadline after every major loop.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray, L: int) -> float:
    """
    Compute max(self‑convolution) of a (which is already normalised to sum=1).
    """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv.max()


# ----------------------------------------------------------------------
def _conv_array(a: np.ndarray, L: int) -> np.ndarray:
    """Full self‑convolution (real part) – length 2n‑1."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights that sum to 1 – numerically stable."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   L: int,
                   phase_factor: np.ndarray,
                   deadline: float,
                   max_iters: int = 150_000,
                   alpha_start: float = 4.0,
                   alpha_end: float = 500.0,
                   lr0: float = 0.07) -> tuple[np.ndarray, float, int]:
    """
    Adam descent on the soft‑max of the convolution.
    Returns (best_sequence, best_obj, performed_iterations).
    """
    n = a0.shape[0]
    a = a0.copy()
    best_seq = a.copy()
    best_obj = _conv_max(a, L)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-9

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # convolution of current a
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]

        # soft‑max approximation of the max
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of soft‑max = 2 * correlation(w, a)
        fft_w = np.fft.fft(w, n=L)
        fft_a_rev = np.conj(fft_a) * phase_factor
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n - 1: n - 1 + n]

        # Adam update (projected onto simplex)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate decay – simple linear schedule
        lr = lr0 * (1.0 - it / max_iters)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate new point
        obj_new = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            print(f"Adam it {it:5d}: improved obj={obj_new:.8e}  score={2*n*obj_new:.6f}")

        a = a_new

    return best_seq, best_obj, it


# ----------------------------------------------------------------------
def _hard_max_subgradient(a0: np.ndarray,
                          L: int,
                          deadline: float,
                          max_iters: int = 30_000,
                          eta0: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the *exact* max‑convolution.
    Uses a simple decaying step‑size and projects onto the simplex.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_seq = a.copy()
    best_obj = _conv_max(a, L)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        conv = _conv_array(a, L)
        k = int(np.argmax(conv))               # index where max occurs

        # sub‑gradient of the max‑convolution
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # decaying step size (inverse square‑root schedule)
        eta = eta0 / math.sqrt(it)

        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        obj_new = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            best_seq = a_new.copy()
            print(f"Sub‑grad it {it:5d}: improved obj={obj_new:.8e}  score={2*n*obj_new:.6f}")

        a = a_new

    return best_seq, best_obj


# ----------------------------------------------------------------------
def _random_moves(a_start: np.ndarray,
                  L: int,
                  deadline: float,
                  max_iter: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small random fraction of mass from
    one index to another.  Accepts only improving moves.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    best_obj = _conv_max(a, L)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # source index – choose proportionally to its mass (large entries move more often)
        i = np.random.choice(n, p=a)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = np.random.rand() * 0.02 * a[i]      # up to 2 % of source mass
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        obj_new = _conv_max(a_new, L)
        if obj_new < best_obj:
            best_obj = obj_new
            a = a_new
            print(f"Random move {it:5d}: improved obj={obj_new:.8e}  score={2*n*obj_new:.6f}")

    return a, best_obj
