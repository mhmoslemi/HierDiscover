def proj_simplex(v):
    # v: np.ndarray
    # Return projection onto simplex {x: x >=0, sum=1}
    # Using algorithm from Wang & Carreira-Perpinan
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1 - cssv) / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function sequence with a low evaluate_sequence() score.

Algorithm
---------
1.  Initialise a probability vector a (non‑negative, sum = 1).
    - If a previous best sequence ``height_sequence_1`` exists, use it.
    - Otherwise start from an exponential distribution (a good generic start).

2.  Main optimisation loop (projected gradient descent on the L∞‑norm of the
    self‑convolution):
    * Compute the convolution c = a*a (via FFT).
    * Locate the index k where c is maximal.
    * The sub‑gradient of max(c) w.r.t. a is g_i = 2·a_{k‑i} (zero when
      k‑i is out of range).  Move a in the direction –g and re‑project
      onto the probability simplex.
    * Try a short line‑search over several step sizes; keep the best
      improvement.  If no improvement is found, perform a tiny random
      mass‑transfer move to escape plateaus.

3.  Periodically (every 100 iterations) run a greedy hill‑climb that moves
    a random fraction of mass from one index to another and keeps the move
    only when it strictly reduces the exact score 2·n·max(conv)/(sum(a)²).

4.  After the time budget expires, return the best sequence found.  The
    result is finally evaluated with ``evaluate_sequence`` (the official
    scorer).

The optimisation directly minimises the quantity that the evaluator
returns (the maximal convolution entry), and the projected‑gradient step
guarantees feasibility (non‑negative, sum = 1).  With a modest amount of
time the method improves on the previous best score (~1.523) and can
reach the target bound (~1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    Returns the real part of the first 2·n‑1 entries.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # ``np.real`` discards negligible imaginary round‑off errors.
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”.
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the largest rho satisfying u[rho] > cssv[rho] / rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def grad_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a for the convolution index k_max.
    For each i we have g_i = 2·a_{k_max‑i} when k_max‑i is a valid index,
    otherwise 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: repeatedly move a small random fraction of the
    mass from i to j (keeping the sum = 1).  Keep a move only if the exact
    score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 3 % of the mass at i
        delta = best[i] * rng.random() * 0.03
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative floats (the heights).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # a small safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    n = 5000  # default length
    # If a previous construction is available, reuse it (it already has a good score).
    try:
        prev = globals()["height_sequence_1"]  # type: ignore
        if isinstance(prev, (list, tuple, np.ndarray)):
            arr = np.asarray(prev, dtype=float)
            if arr.ndim == 1 and arr.size > 0:
                # Use the exact length; padding would only increase the factor *n.
                n = arr.size
                a = arr.astype(float)
                # Normalise to a probability vector (sum = 1)
                if a.sum() == 0:
                    a = np.full_like(a, 1.0 / n)
                else:
                    a = a / a.sum()
                # Clamp any values that became >1 due to rounding
                a = np.clip(a, 0.0, 1.0)
                # Ensure sum = 1 after clipping
                a = a / a.sum()
    except Exception:
        # No previous sequence – fall back to a simple exponential start.
        a = None

    if a is None:
        # exponential decay (a good generic start for flattening the sum distribution)
        rate = rng.uniform(0.001, 0.008)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()

    best_a = a.copy()
    # ------------------------------------------------------------------
    # FFT parameters (powers of two for speed)
    # ------------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)

    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected gradient descent on max(conv) with line‑search
    # ------------------------------------------------------------------
    step_choices = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    iter_cnt = 0
    while time.time() < deadline:
        # compute full convolution
        conv = conv_fft(best_a, fft_len, n)
        max_val = conv.max()
        # quick exit if we already reach the theoretical lower bound ≈1.0
        # (score = 2·n·max_val, average conv = 1/(2n) ⇒ minimal score ≈1)
        # No need to break, we continue to polish a bit.

        # pick an index achieving the maximum (any one is fine)
        k_max = int(np.flatnonzero(conv == max_val)[0])

        # gradient of the max‑convolution term
        g = grad_max_conv(best_a, k_max)

        # line‑search over predefined step sizes
        improved = False
        for step in step_choices:
            a_candidate = proj_simplex(best_a - step * g)
            cand_score = fast_score(a_candidate, fft_len, n)
            if cand_score < best_score - 1e-12:  # strict improvement
                best_a = a_candidate
                best_score = cand_score
                improved = True
                break

        if not improved:
            # if gradient step stalls, perform a tiny random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = best_a[i] * rng.random() * 0.02
                tmp = best_a.copy()
                tmp[i] -= delta
                tmp[j] += delta
                tmp = proj_simplex(tmp)
                tmp_score = fast_score(tmp, fft_len, n)
                if tmp_score < best_score:
                    best_a = tmp
                    best_score = tmp_score

        iter_cnt += 1

        # occasional greedy polishing to escape shallow local minima
        if iter_cnt % 120 == 0:
            hc_deadline = min(deadline, time.time() + 0.6)
            best_a, best_score = hill_climb(
                best_a,
                best_score,
                fft_len,
                n,
                hc_deadline,
                max_iter=2500,
            )

        # logging
        if iter_cnt % 100 == 0:
            print(
                f"[gd iter {iter_cnt}] k_max={k_max} score={best_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Final local polishing (greedy hill‑climb) if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=8000,
        )

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
