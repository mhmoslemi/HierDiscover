def gradient_smooth(a, mu):
    n = len(a)
    conv = compute_conv(a) # length 2n-1
    # compute weights with numerically stable exponentials
    max_val = conv.max()
    exp_weights = np.exp(mu * (conv - max_val))
    p = exp_weights / exp_weights.sum()
    # compute g = 2 * correlation p with a
    # use FFT cross-correlation.
    L = next_pow2(len(p) + n - 1)  # ensure power of 2
    fp = np.fft.rfft(p, n=L)
    fa = np.fft.rfft(a, n=L)
    # cross-correlation of p and a: corr = ifft(conj(fp) * fa)
    corr = np.fft.irfft(np.conj(fp) * fa, n=L)
    # The correlation array length L; we need indices from n-1 to n-1+n-1 = 2n-2 maybe.
    # As per formula: corr[i + (len(a)-1)] = sum_j p_{i+j} * a_j .
    # So g_i = 2 * corr[i + n-1] for i in [0,n-1]
    start = n - 1
    g = 2.0 * corr[start : start + n]
    return g

def refine_candidate(a, deadline):
    n = a.shape[0]
    best = a.copy()
    best_score = fast_score(a)
    # Precompute FFT length for conv
    fft_len_conv = next_pow2(2*n)  # for self-convolution compute.
    # We'll compute conv inside each iteration.
    # For smoothing gradient we need conv already.
    # Use schedule for mu: linearly increase from mu0 to mu_max.
    mu0 = 1.0  # start low smoothing
    mu_max = 100.0
    eta0 = 0.8
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv for a
        fa = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2*n - 1]
        # compute smoothed gradient
        max_conv = conv.max()
        # compute weighted exponentials
        # To avoid overflow, subtract max.
        exp_arg = mu0 + (mu_max - mu0) * min(1.0, (iteration / 5000.0))  # gradually increase
        # compute p weights
        mu = exp_arg
        # compute p:
        # Use shift for numerical stability:
        shifted = mu * (conv - max_conv)  # convex below zero, safe
        exp_weights = np.exp(shifted)
        p = exp_weights / exp_weights.sum()
        # compute correlation gradient
        # using FFT cross-correlation:
        L = next_pow2(len(p) + n - 1)
        fp = np.fft.rfft(p, n=L)
        fa2 = np.fft.rfft(a, n=L)  # reuse spectral of a but need length L
        corr = np.fft.irfft(np.conj(fp) * fa2, n=L)
        start = n - 1
        g = 2.0 * corr[start : start + n]
        # step size
        eta = eta0 / np.sqrt(iteration)
        # update and project
        a_candidate = a - eta * g
        a = project_simplex(a_candidate)
        # periodically evaluate actual score
        if iteration % 200 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{time.time() - start_time:.1f}s] refine iter {iteration} best score {cur_score:.6f}")
    return best, best_score

"""
A fast two‑phase optimiser for the self‑convolution bound.

Phase 1 – randomised family search:
    Generates diverse candidates (power‑law, exponential, two‑sided,
    Gaussian, cosine / Hann windows).  Every shape is normalised to
    sum = 1 and scored with a cheap FFT‑based evaluator.  The best
    candidate is kept.

Phase 2 – smoothed‑max projected gradient descent:
    Starting from the best candidate, we minimise the quantity

        2·n·max(conv(a,a)) / (sum(a)²)

    (the denominator is constant after normalisation).  The
    non‑smooth max is replaced by the log‑sum‑exp (a smooth
    approximation) whose gradient can be written as a weighted
    cross‑correlation and computed entirely with FFTs.
    After each gradient step the vector is projected onto the
    probability simplex (non‑negative entries summing to 1).
    The temperature of the soft‑max is increased during the run so
    that the optimisation begins with a smooth landscape and ends
    with a tight focus on the true maximum.  Progress is logged
    and the best sequence found is returned.

The whole routine respects the 1000 s budget; if time runs out it
still returns the best sequence discovered so far.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x must be > 0)."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective using an FFT convolution.
    `seq` is assumed to be a 1‑D non‑negative array with sum > 0.
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (all return un‑normalised non‑negative arrays)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_gaussian(n: int, sigma_factor: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle, σ = sigma_factor·n."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    return np.exp(-0.5 * ((i - centre) / sigma) ** 2)


def generate_cosine_window(n: int, power: float = 1.0) -> np.ndarray:
    """Raised cosine (0 at the two ends, maximum in the centre)."""
    i = np.arange(n, dtype=np.float64)
    w = (1.0 - np.cos(np.pi * i / (n - 1))) ** power
    return w


def generate_hann_window(n: int, power: float = 1.0) -> np.ndarray:
    """Standard Hann window raised to a power."""
    i = np.arange(n, dtype=np.float64)
    w = (0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))) ** power
    return w


# ----------------------------------------------------------------------
# Gradient based refinement (smooth‑max)
# ----------------------------------------------------------------------


def smooth_max_gradient(a: np.ndarray, mu: float) -> np.ndarray:
    """
    Returns the gradient of the smooth approximation
        f_mu(a) = (1/μ)·log Σ_k exp( μ·conv_k(a) )
    where conv is the self‑convolution of a.
    The gradient is
        g_i = 2· Σ_k p_k·a_{k-i}   with   p_k = exp( μ·conv_k ) / Σ_k exp( μ·conv_k )
    Computed via FFTs.
    """
    n = a.shape[0]
    L = next_pow2(3 * n)               # enough for all convolutions

    # ----- self‑convolution -----
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]    # length 2n‑1

    # ----- soft‑max weights p_k -----
    max_val = conv_full.max()
    # shift for numerical stability (max_val ≤ conv ≤ max_val)
    shifted = mu * (conv_full - max_val)
    p = np.exp(shifted)
    p /= p.sum()                     # normalised probabilities, shape (2n‑1,)

    # ----- gradient = 2· correlation(p, a) at negative lags -----
    # correlation needed at lag –i, i.e. g_i = 2· Σ_j p_j·a_{j-i}
    # This equals 2· (p_rev * a) evaluated at indices (2n‑2 – i)
    p_rev = p[::-1]                  # reverse to obtain the required lag direction
    p_rev_fft = np.fft.rfft(p_rev, n=L)

    # convolution of p_rev with a (linear, length 3n‑2)
    conv_rev = np.fft.irfft(p_rev_fft * a_fft, n=L)[: 3 * n - 2]

    idx = 2 * n - 2 - np.arange(n)    # map i → (2n‑2‑i)
    g = 2.0 * conv_rev[idx]
    return g


def refine_sequence(
    init_seq: np.ndarray, deadline: float, log_prefix: str = "[refine]"
) -> np.ndarray:
    """
    Projected gradient descent on the smoothed max.
    Returns the best sequence seen before the deadline.
    """
    a = init_seq.copy()
    n = a.shape[0]

    best_a = a.copy()
    best_score = fast_score(best_a)

    iteration = 0
    eta0 = 0.8                     # initial step‑size factor
    mu_start = 0.5
    mu_end = 50.0                  # final temperature (more focussed)
    mu_ramp_iters = 5000          # number of iterations over which μ is increased

    while time.time() < deadline:
        iteration += 1

        # temperature schedule (linear increase)
        mu = mu_start + (mu_end - mu_start) * min(1.0, iteration / mu_ramp_iters)

        # gradient of the smooth max
        g = smooth_max_gradient(a, mu)

        # diminishing step size
        eta = eta0 / np.sqrt(iteration)

        # projected gradient step onto the simplex
        a_candidate = a - eta * g
        a = project_simplex(a_candidate)

        # periodic exact evaluation
        if iteration % 300 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(
                    f"{log_prefix} t={time.time():.1f}s it={iteration} "
                    f"μ={mu:.2f} η={eta:.4f} score={cur_score:.6f}"
                )
    return best_a

def random_candidate(length_choices):
    """Pick a random family, random parameters and return a normalised sequence."""
    n = random.choice(length_choices)
    family = random.choice(
        ["power", "exp", "twosided", "gaussian", "cosine", "hann"]
    )
    if family == "power":
        p = random.uniform(0.30, 0.85)
        seq = generate_power_law(n, p)
    elif family == "exp":
        # log‑uniform rate between 1e‑4 and 3e‑2
        lam = 10 ** random.uniform(-4, -1.5)
        seq = generate_exponential(n, lam)
    elif family == "twosided":
        pl = random.uniform(0.30, 0.85)
        pr = random.uniform(0.30, 0.85)
        wl = random.uniform(0.2, 4.0)
        wr = random.uniform(0.2, 4.0)
        seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
    elif family == "gaussian":
        sigma_factor = random.uniform(0.08, 0.45)   # fairly broad shapes
        seq = generate_gaussian(n, sigma_factor)
    elif family == "cosine":
        power = random.uniform(0.5, 3.5)
        seq = generate_cosine_window(n, power)
    else:   # "hann"
        power = random.uniform(0.5, 3.5)
        seq = generate_hann_window(n, power)

    # make non‑negative, normalise to sum = 1
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s < 1e-12:
        return None
    return seq / s
