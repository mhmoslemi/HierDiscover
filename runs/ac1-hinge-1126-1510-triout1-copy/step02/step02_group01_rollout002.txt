def simulated_annealing(a, score, fft_len, n, deadline, initial_temp=0.5, final_temp=0.01):
    cur_a = a.copy()
    cur_score = score
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        t = (deadline - time.time()) / (deadline - start_time)  # fraction remaining
        temp = final_temp + (initial_temp - final_temp) * t**0.5
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.1
        if delta <= 0:
            continue
        cur_a[i] -= delta
        cur_a[j] += delta
        new_score = fast_score(cur_a, fft_len, n)
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / temp):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta
    return best_a, best_score
```

def hill_climb_product(a, cur_score, fft_len, n, deadline, max_iter=3000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter>0:
        max_iter -= 1
        # compute conv and max index
        conv = conv_fft(best, fft_len, n)
        max_idx = int(np.argmax(conv))
        # compute contributions product
        idx = np.arange(n)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        prod = np.zeros(n)
        prod[mask] = best[idx[mask]] * best[partner[mask]]
        # compute sum to sample
        total = prod.sum()
        if total <= 0:
            # fallback to random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            probs = prod / total
            i = rng.choice(n, p=probs)
            # pick j from low contribution region
            low_mask = prod < np.median(prod)
            if low_mask.any():
                j = rng.choice(np.where(low_mask)[0])
            else:
                j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.1  # up to 10%
        if delta <= 0:
            continue
        # move mass
        best[i] -= delta
        best[j] += delta
        # compute new score
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def simulated_annealing(a, cur_score, fft_len, n, deadline,
                        init_temp=0.5, final_temp=0.01):
    cur = a.copy()
    cur_score = cur_score
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    while time.time() < deadline:
        # compute temperature based on progress
        elapsed = time.time() - start_time
        total = deadline - start_time
        if total <= 0:
            break
        t_frac = max(0.0, 1.0 - elapsed / total)  # decreasing
        temp = final_temp + (init_temp - final_temp) * (t_frac ** 0.5)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Choose delta scale adaptively
        delta = cur[i] * rng.random() * 0.1  # up to 10% of mass at i
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        dscore = new_score - cur_score
        if dscore < 0 or rng.random() < np.exp(-dscore / temp):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm is a hybrid optimiser:
* Random‑restart L‑BFGS on a smooth approximation of max(a*a) (the
  usual “smooth‑max” surrogate).  A continuation schedule gradually
  tightens the smoothing.
* A product‑weighted greedy hill‑climb that always attacks the
  convolution entry that is currently maximal and moves probability
  mass away from the indices that contribute most to it.
* A lightweight simulated‑annealing phase that explores the
  neighbourhood of the current best solution.

The three phases are interleaved until the time budget runs out,
followed by a short final L‑BFGS refinement and another hill‑climb.
All sequences are kept on the probability simplex (sum = 1) – the
evaluation function is invariant under uniform scaling, so this
normalisation does not affect the final score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x* (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using the FFT.
    *a* is assumed to be length *n* and *fft_len* a power of two
    at least 2·n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the attacker’s score for a probability vector *a*.
    Returns 2·n·max(a*a) (the denominator (sum a)^2 is 1 because a sums to 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max surrogate for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max(conv) and its gradient w.r.t. the
    unconstrained vector *y* (the soft‑max parameters).  The smoothing
    width is lam = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z   # shape (2*n-1,)

    # ----- gradient w.r.t. a (convolution of a with w) -----
    # we need (a * w) evaluated at indices n‑1 … n‑1+n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through the soft‑max -----
    mean_g = np.dot(a, g_vec)          # <a, grad_a>
    grad_y = a * (g_vec - mean_g)      # ∂f/∂y

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 180,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Product‑weighted greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb_product(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the current maximal convolution
    entry.  In each iteration the algorithm:
    * finds the index `k` where (a*a) is maximal,
    * computes the contribution a[i]·a[k‑i] for every i,
    * picks an index `i` with probability proportional to that contribution,
    * picks a destination index `j` where the partner contribution is small,
    * moves a random fraction (≤10 %) of mass from i to j,
    * accepts the move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current convolution and its arg‑max
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        # contribution of each index to the maximal entry
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros(n)
        contrib[mask] = best[idx[mask]] * best[partner[mask]]

        total = contrib.sum()
        if total <= 0.0:
            # fall back to a completely random move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            probs = contrib / total
            i = rng.choice(n, p=probs)

            # destination should have a small partner value at index k
            low_mask = contrib < np.median(contrib)
            if low_mask.any():
                j = rng.choice(np.where(low_mask)[0])
            else:
                j = rng.integers(0, n)

        if i == j:
            continue

        # move a fraction of the mass at i to j
        delta = best[i] * rng.random() * 0.10   # ≤10 % of a[i]
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing explorer
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.5,
    final_temp: float = 0.01,
) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing on the simplex.
    Moves mass from a random source to a random destination;
    acceptance follows the Metropolis rule with an exponential cooling.
    """
    cur = a.copy()
    cur_score = cur_score
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()

    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        if total <= 0.0:
            break
        # linear cooling in sqrt‑time
        t_frac = max(0.0, 1.0 - elapsed / total)
        temp = final_temp + (init_temp - final_temp) * (t_frac ** 0.5)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.10   # up to 10 % of source mass
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        dscore = new_score - cur_score
        accept = dscore < 0.0 or rng.random() < np.exp(-dscore / temp)

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The algorithm combines L‑BFGS on a
    smooth‑max surrogate, product‑weighted greedy hill‑climbing,
    and simulated annealing.  It adapts its behaviour to the remaining
    time budget and finally returns the best sequence found as a plain
    list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (larger if a previous construction is available)
    # ------------------------------------------------------------------
    n_default = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.array(init_seq, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = max(n_default, arr.size)
            best_a = np.maximum(arr, 0.0)
            best_a = best_a / best_a.sum()
            if best_a.size > n:
                best_a = best_a[:n]
            elif best_a.size < n:
                best_a = np.pad(best_a, (0, n - best_a.size))
        else:
            n = n_default
            best_a = np.full(n, 1.0 / n)
    else:
        n = n_default
        best_a = np.full(n, 1.0 / n)

    if best_a.sum() < 1e-12:                     # guard against pathological start
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)          # parameters for L‑BFGS

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003]
    iteration = 0

    # ------------------------------------------------------------------
    # Main search loop – interleave the three strategies
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 1.0:
            break

        mode = rng.choice(["lbfgs", "hill", "sa"])

        # --------------------------------------------------------------
        # L‑BFGS phase (continuation on the smooth‑max surrogate)
        # --------------------------------------------------------------
        if mode == "lbfgs" and remaining > 8.0:
            start_type = rng.choice(["best", "random", "exp", "power", "edge"])
            if start_type == "best":
                y0 = best_y + rng.normal(scale=0.05, size=n)
            elif start_type == "random":
                y0 = rng.standard_normal(n)
            elif start_type == "exp":
                rate = rng.uniform(0.001, 0.02)
                a0 = np.exp(-rate * np.arange(n))
                a0 /= a0.sum()
                y0 = np.log(a0 + 1e-12)
            elif start_type == "power":
                p = rng.uniform(0.3, 2.0)
                a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
                a0 /= a0.sum()
                y0 = np.log(a0 + 1e-12)
            else:  # edge
                m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
                a_edge = np.sqrt(n / 2.0) / n
                a_center = 1.0 / np.sqrt(2.0) / n
                a0 = np.full(n, a_center)
                a0[:m_edge] = a_edge
                a0[-m_edge:] = a_edge
                a0 /= a0.sum()
                y0 = np.log(a0 + 1e-12)

            y_opt = schedule_optimize(
                y0,
                n,
                fft_len_conv,
                fft_len_g,
                lam_schedule,
                maxiter_per_stage=180,
            )
            a_opt = softmax(y_opt)
            score_opt = fast_score(a_opt, fft_len_conv, n)

            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt.copy()
                best_y = y_opt.copy()
                print(f"[{iteration}] lbfgs ({start_type}) improved fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Product‑weighted greedy hill‑climb
        # --------------------------------------------------------------
        elif mode == "hill" and remaining > 6.0:
            sub_deadline = min(deadline, time.time() + min(6.0, remaining * 0.5))
            a_hc, score_hc = hill_climb_product(
                best_a,
                best_score,
                fft_len_conv,
                n,
                sub_deadline,
                max_iter=4000,
            )
            if score_hc < best_score:
                best_score = score_hc
                best_a = a_hc.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[{iteration}] hill‑climb improved fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Simulated annealing explorer
        # --------------------------------------------------------------
        elif mode == "sa" and remaining > 12.0:
            sub_deadline = min(deadline, time.time() + min(12.0, remaining * 0.5))
            a_sa, score_sa = simulated_annealing(
                best_a,
                best_score,
                fft_len_conv,
                n,
                sub_deadline,
                init_temp=0.5,
                final_temp=0.01,
            )
            if score_sa < best_score:
                best_score = score_sa
                best_a = a_sa.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[{iteration}] annealing improved fast score = {best_score:.6f}")

        else:
            # Not enough time for the chosen mode – exit the loop.
            break

        iteration += 1

    # ------------------------------------------------------------------
    # Final refinement – a few extra L‑BFGS passes with the same schedule
    # ------------------------------------------------------------------
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 250, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[refine] lam={lam:.4f} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # One last greedy polishing if time permits
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_product(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=8000
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
