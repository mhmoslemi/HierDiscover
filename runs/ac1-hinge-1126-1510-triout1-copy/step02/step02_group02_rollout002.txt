Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of several independent phases that explore the
search space from different angles:

1️⃣  **Parametric sweep** – a rich collection of one‑parameter families
    (power‑law, exponential, beta‑distribution, Gaussian, sinusoid,
    piecewise‑constant, U‑shaped sum of exponentials).  Every candidate
    is normalised to unit sum and its convolution max is evaluated.
    The best shape is kept.

2️⃣  **Random Dirichlet starts** – a few hundred random vectors drawn
    from Dirichlet distributions with various concentration parameters.
    They serve as diverse seeds for the subsequent continuous optimisation.

3️⃣  **Adam on a smoothed max** – the non‑smooth `max` is replaced by a
    soft‑max (`log‑sum‑exp`).  The smoothing parameter α is increased
    from 4 to 500 while an Adam optimiser updates the vector on the
    probability simplex (Euclidean projection after each step).

4️⃣  **Sub‑gradient descent on the true max** – once α becomes large the
    soft‑max gradient approximates the sub‑gradient of the true max.
    A short Adam‑style sub‑gradient phase with α fixed at the final
    value tightens the solution further.

5️⃣  **Targeted local moves** – the current peak index `p` of the
    convolution is identified.  Mass is repeatedly transferred from
    positions that contribute heavily to `conv[p]` to random positions.
    Any move that lowers the maximum is accepted.

All phases respect the overall time budget (default 1000 s).  The final
vector is scaled so that `sum = sqrt(2·n)` as required by the scoring
function; because the objective is scale‑invariant this scaling does
not affect optimality.

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (x ≥ 0, Σx = 1). """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, L: int) -> tuple[float, np.ndarray]:
    """Return (max(conv), full convolution) for a real vector a (unit sum)."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return float(conv.max()), conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights (length 2n‑1) that sum to 1."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _gradient_from_weights(a: np.ndarray,
                           w: np.ndarray,
                           fft_a: np.ndarray,
                           phase: np.ndarray,
                           L: int) -> np.ndarray:
    """
    Compute ∇(w·conv(a,a)) = 2 * correlation(w, a).
    This uses the FFT of a (fft_a) and the pre‑computed phase vector for reversal.
    Returns a gradient of length n.
    """
    fft_w = np.fft.fft(w, n=L)
    # fft of reversed a (no extra FFT)
    fft_a_rev = np.conj(fft_a) * phase
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    # correlation result is centred at index n‑1
    n = a.size
    return 2.0 * corr[n - 1 : n - 1 + n]

    try:
        # `height_sequence_1` may be present from a previous run.
        init_global = np.array(height_sequence_1, dtype=np.float64)  # type: ignore
        n = init_global.size
        init_norm = init_global / init_global.sum() if init_global.sum() > 0 else None
    except Exception:
        init_norm = None
        n = 8000                                 # a reasonable default size

    target_sum = math.sqrt(2.0 * n)
    L = 1 << ((3 * n - 2).bit_length())          # FFT length >= 3n‑2
    phase = np.exp(2j * np.pi * np.arange(L) / L)  # phase factor for reversed FFT

    # Best solution (unit‑sum) and its objective (max(conv))
    best_seq = [np.full(n, 1.0 / n)]  # mutable container for closure‑free updates
    best_obj = [float("inf")]

    # ------------------------------------------------------------------
    # 0.  Initialise from a possible previous construction.
    if init_norm is not None:
        obj, _ = _conv_max(init_norm, L)
        best_seq[0] = init_norm.copy()
        best_obj[0] = obj
        print(f"Initial global candidate: obj={obj:.9f} | score={2*n*obj:.6f}")

    # ------------------------------------------------------------------
    # 1.  Parametric sweep over a diverse family of shapes.
    print(f"--- Phase 1: parametric sweep (n={n}) ---")

    # ---- Power‑law ----------------------------------------------------
    exponents = np.linspace(-2.0, 2.0, 41)
    for p in exponents:
        if time.time() > deadline:
            break
        base = np.arange(1, n + 1, dtype=np.float64) ** p
        _update_best(base, L, best_seq, best_obj, n)          # increasing
        _update_best(base[::-1], L, best_seq, best_obj, n)   # decreasing

    # ---- Exponential --------------------------------------------------
    lambdas = np.linspace(0.0, 5.0, 41)
    for lam in lambdas:
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * np.arange(n, dtype=np.float64))
        _update_best(exp_shape, L, best_seq, best_obj, n)
        _update_best(exp_shape[::-1], L, best_seq, best_obj, n)

    # ---- Beta‑distribution (flexible U‑shape) -------------------------
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    betas = alphas
    x = np.arange(n, dtype=np.float64) / (n - 1)
    eps = 1e-12
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta_shape = (x + eps) ** (a - 1) * (1 - x + eps) ** (b - 1)
            _update_best(beta_shape, L, best_seq, best_obj, n)

    # ---- Gaussian (single peak) ---------------------------------------
    mus = [n / 2]
    sigmas = np.linspace(n / 20, n / 4, 12)
    for mu in mus:
        for sigma in sigmas:
            if time.time() > deadline:
                break
            gauss = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
            _update_best(gauss, L, best_seq, best_obj, n)

    # ---- Double‑Gaussian (high at both ends) -------------------------
    sigs = np.linspace(n / 30, n / 8, 8)
    for sigma in sigs:
        if time.time() > deadline:
            break
        left = np.exp(-0.5 * ((np.arange(n) - 0) / sigma) ** 2)
        right = left[::-1]
        double = left + right
        _update_best(double, L, best_seq, best_obj, n)

    # ---- Sinusoidal (single bump) ------------------------------------
    powers = np.linspace(0.5, 4.0, 8)
    for p in powers:
        if time.time() > deadline:
            break
        sinu = np.sin(np.pi * (np.arange(1, n + 1) / (n + 1))) ** p
        _update_best(sinu, L, best_seq, best_obj, n)

    # ---- Piecewise constant (two blocks) -----------------------------
    ratios = np.linspace(0.1, 0.9, 9)   # split position
    for r in ratios:
        if time.time() > deadline:
            break
        k = int(r * n)
        if k == 0 or k == n:
            continue
        # t is the total mass assigned to the first block
        t_vals = np.linspace(0.1, 0.9, 9)
        for t in t_vals:
            if time.time() > deadline:
                break
            a = np.empty(n, dtype=np.float64)
            a[:k] = t / k
            a[k:] = (1.0 - t) / (n - k)
            _update_best(a, L, best_seq, best_obj, n)

    # ------------------------------------------------------------------
    # 2.  Random Dirichlet starting points (diversity).
    print("--- Phase 2: random Dirichlet seeds ---")
    dirichlet_alphas = [0.3, 0.7, 1.0, 2.0, 5.0]
    n_dirichlet = 300
    for alpha in dirichlet_alphas:
        for _ in range(n_dirichlet):
            if time.time() > deadline:
                break
            a_rand = np.random.dirichlet(np.full(n, alpha))
            _update_best(a_rand, L, best_seq, best_obj, n)

    # ------------------------------------------------------------------
    # 3.  Adam descent on a smoothed max (soft‑max).
    print("--- Phase 3: Adam with smooth‑max ---")
    a = best_seq[0].copy()
    # Adam hyper‑parameters
    max_iters = 30000
    lr0 = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    alpha_start, alpha_end = 4.0, 500.0   # more aggressive smoothing

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # -------- current convolution ----------
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        cur_max = conv.max()

        # -------- soft‑max weights -------------
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)

        # -------- gradient ---------------------
        grad = _gradient_from_weights(a, w, fft_a, phase, L)

        # -------- Adam step --------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # -------- evaluate new point ------------
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[: 2 * n - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj[0]:
            best_obj[0] = obj_new
            best_seq[0] = a_new.copy()
            print(f"Iter {it:5d}: improved | obj={obj_new:.9f} | score={2*n*obj_new:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # 4.  Sub‑gradient (soft‑max with α fixed at the final, large value)
    print("--- Phase 4: sub‑gradient descent (α fixed) ---")
    a = best_seq[0].copy()
    alpha_fixed = alpha_end
    max_sub_iters = 5000
    sub_it = 0
    while time.time() < deadline and sub_it < max_sub_iters:
        sub_it += 1
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        cur_max = conv.max()
        # identify all near‑max indices (tolerance relative to cur_max)
        tol = 1e-12 * max(1.0, cur_max)
        active = np.where(conv >= cur_max - tol)[0]
        # uniform weight on the active set
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size
        grad = _gradient_from_weights(a, w, fft_a, phase, L)

        # simple Adam‑style update (re‑using the same hyper‑params)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** sub_it)
        v_hat = v / (1.0 - beta2 ** sub_it)
        step = lr0 * 0.5 * m_hat / (np.sqrt(v_hat) + eps)  # smaller LR

        a_new = a - step
        a_new = _project_simplex(a_new)

        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[: 2 * n - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj[0]:
            best_obj[0] = obj_new
            best_seq[0] = a_new.copy()
            print(f"  sub‑iter {sub_it:4d}: improved | obj={obj_new:.9f} | score={2*n*obj_new:.6f}")
        a = a_new

    # ------------------------------------------------------------------
    # 5.  Targeted local random moves.
    print("--- Phase 5: targeted random moves ---")
    a_current = best_seq[0].copy()
    # pre‑compute a few constants for speed
    max_move_iters = 200000
    move_it = 0
    while time.time() < deadline and move_it < max_move_iters:
        move_it += 1
        # current convolution to locate the worst (peak) index
        fft_a = np.fft.fft(a_current, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        p_peak = int(conv.argmax())

        # contributions of each i to conv[p_peak] = Σ_i a_i a_{p_peak-i}
        # zero where the partner index is out of range
        contrib = np.zeros_like(a_current)
        lo = max(0, p_peak - (n - 1))
        hi = min(n - 1, p_peak)
        idx = np.arange(lo, hi + 1, dtype=int)
        partner = p_peak - idx
        contrib[idx] = a_current[idx] * a_current[partner]

        if contrib.sum() == 0.0:
            continue
        prob_source = contrib / contrib.sum()
        i = np.random.choice(n, p=prob_source)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # transfer a modest fraction (up to 3 %) of the source mass
        delta = np.random.rand() * 0.03 * a_current[i]
        if delta <= 0.0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:
            continue

        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[: 2 * n - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj[0]:
            best_obj[0] = obj_new
            best_seq[0] = a_new.copy()
            a_current = a_new
            print(f"  move #{move_it:6d} improved | obj={obj_new:.9f} | score={2*n*obj_new:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the reference normalisation (sum = sqrt(2·n))
    final_seq = (best_seq[0] * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) after {it} Adam iters ---")
    return final_seq
