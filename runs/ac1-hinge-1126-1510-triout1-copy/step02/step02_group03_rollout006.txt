def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time
    from scipy.optimize import linprog

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5 # safety

    # Choose n
    n = 800   # moderate size

    # Initialize a shape using sqrt diff, they used earlier. Use a = sqrt(i+1) - sqrt(i)
    a = np.sqrt(np.arange(1,n+1, dtype=float)) - np.sqrt(np.arange(0,n, dtype=float))
    a = np.clip(a, 0, None)
    a = a / a.sum()  # sum = 1

    best_seq = a.copy()
    best_score = evaluate_sequence((best_seq*1000).tolist())
    print(f"initial score {best_score}")

    iter_count = 0
    while time.time() < deadline:
        # compute convolution fast
        conv = np.convolve(a, a) # small n, fine
        M = conv.max()
        # Build matrix A
        # get all constraints rows length = 2*n-1
        m = 2*n - 1
        # Preallocate matrix
        A = np.zeros((m, n), dtype=float)
        # Fill using f
        for i in range(n):
            A[i:i+n, i] = a  # f placed in column i

        # Solve LP
        c = -np.ones(n)  # maximize sum(b)
        # bounds: b >= 0
        bounds = [(0, None)] * n
        # Using HiGHS
        res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds, method='highs')
        if not res.success:
            print("LP failed")
            break
        b = res.x
        s = b.sum()
        if s <= 1 + 1e-12:
            # cannot increase sum, converge
            break
        # update a
        a = b / s
        # optionally compute new score
        score = evaluate_sequence((a*1000).tolist())
        iter_count += 1
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Iter {iter_count}: new best {best_score}")
        # continue loop

    # final scaling
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Returning best score {best_score} after {time.time() - start}s")
    return final_seq.tolist()

"""
Proposed candidate generator – Linear‑programming alternating reduction.

The algorithm works with a probability vector `a` (non‑negative, sum = 1).
For a fixed `a` we compute the autocorrelation peak `M = max(a*a)`.
The next vector `b` is obtained by solving the linear program

    maximize    sum(b)
    subject to  a * b  ≤  M                (pointwise convolution)
                b ≥ 0

Because `b = a` is always feasible, the optimal objective value `s = sum(b)`
satisfies `s ≥ 1`.  Scaling `b` back to a probability vector (`b / s`)
strictly reduces the peak: the new peak becomes `M / s`.  Repeating this step
produces a monotone decrease of the evaluation score
    score = 2·n·max(a*a)   (the denominator ∑a = 1 is fixed).

Multiple random restarts (different initial shapes) are performed until the
time budget expires; the best sequence found is returned after scaling to the
required [0, 1000] range (scaling does not affect the score).
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: build the convolution matrix  C where (C @ b)[k] = (a * b)[k]
# ----------------------------------------------------------------------
def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """Return the full linear convolution matrix of a with a variable vector."""
    n = a.shape[0]
    m = 2 * n - 1
    C = np.zeros((m, n), dtype=a.dtype)
    # each column i contains a shifted copy of a
    for i in range(n):
        C[i:i + n, i] = a
    return C


# ----------------------------------------------------------------------
# Main entry point – the function invoked by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low autocorrelation peak using an
    alternating linear‑programming scheme.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for final work

    # ------------------------------------------------------------------
    # Choose a length.  If a previous construction is available, reuse its
    # length; otherwise fall back to a moderately large size.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        n = 1542                       # default used in earlier runs

    # ------------------------------------------------------------------
    # Keep the best solution seen so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Different ways to initialise a candidate probability vector
    # ------------------------------------------------------------------
    init_modes = ["sqrtdiff", "power", "uniform", "random"]
    restart = 0

    while time.time() < deadline:
        restart += 1
        mode = np.random.choice(init_modes)

        if mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        elif mode == "power":
            p = np.random.uniform(0.1, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == "uniform":
            a = np.ones(n, dtype=float)
        else:   # "random"
            a = np.random.rand(n)

        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.ones(n)
        a = a / a.sum()                     # enforce ∑a = 1

        # --------------------------------------------------------------
        # Alternating LP improvement – stop after a few iterations or when
        # the improvement stalls.
        # --------------------------------------------------------------
        inner_iter = 0
        while time.time() < deadline and inner_iter < 30:
            # current autocorrelation peak
            conv = np.convolve(a, a)
            M = conv.max()                  # max_{k} (a*a)[k]

            # store best progress
            cur_score = 2.0 * n * M          # exact score when ∑a = 1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start
                print(f"[Restart {restart}] iter {inner_iter:02d} – "
                      f"new best {best_score:.6f} (t={elapsed:.1f}s)")

            # Build the convolution matrix for the current a
            C = _conv_matrix(a)

            # The LP: maximise sum(b)  s.t. C @ b ≤ M  and b ≥ 0.
            # To avoid tiny numbers we normalise the RHS to 1.
            A_ub = C / M
            b_ub = np.ones(2 * n - 1, dtype=float)

            # Linear objective – maximise sum(b) ⇒ minimise -sum(b)
            c = -np.ones(n, dtype=float)

            # Solve with HiGHS (fast interior‑point/simplex hybrid)
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
            if not res.success:
                # LP failed – abort this restart
                break

            b = res.x
            s = b.sum()
            if s <= 1.0 + 1e-12:            # no increase ⇒ convergence
                break

            # Normalise b back to a probability vector.
            a = b / s

            inner_iter += 1

        # --------------------------------------------------------------
        # If there is still time, perform a tiny random polishing phase.
        # --------------------------------------------------------------
        if best_seq is not None and time.time() < deadline:
            a_local = best_seq.copy()
            polish_deadline = time.time() + min(4.0, deadline - time.time())
            while time.time() < polish_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # move a small fraction from i to j
                eps = min(a_local[i], a_local[i] * np.random.uniform(0.0, 0.05))
                if eps <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps
                a_new[j] += eps
                a_new = a_new / a_new.sum()          # re‑normalise
                new_score = 2.0 * n * np.convolve(a_new, a_new).max()
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # --------------------------------------------------------------
        # Continue with another restart if time permits
        # --------------------------------------------------------------
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – in the extremely unlikely case no candidate was found
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n, dtype=float) / n

    # ------------------------------------------------------------------
    # Scale to the evaluator's [0, 1000] range (does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_time = time.time() - start
    print(f"Returning sequence – best score {best_score:.6f} after {total_time:.1f}s")
    return final_seq.tolist()
