"""
propose_candidate
-----------------

Search for a non‑negative step function (list of heights) that minimises

    2 * n * max(conv(a, a))   with Σ a = √(2 n)

The implementation builds many diverse initial shapes (uniform,
triangular, cosine windows, Gaussians, exponentials, Dirichlet draws,
the previous best) and refines each of them with a smooth‑max Adam
optimizer.  Improvements over the reference implementation come from

* a richer set of starting points,
* a longer optimisation (up to 15 000 iterations per restart),
* a more aggressive β‑schedule (β_end = 200 instead of 80) that
  approaches the true max more closely,
* a linearly decreasing learning‑rate (0.1 → 0.01),
* early‑stopping once the global time budget (≈ 1000 s) is exhausted.

The optimiser works on the probability simplex by parametrising the
vector with a soft‑max of unconstrained logits `v`.  Gradients are
computed via FFT‑based correlation, exactly as in the reference code.
The best sequence found is finally rescaled to Σ a = √(2 n) and clipped
to the range accepted by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers (no closures, no lambdas)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a, length = 2·len(a)‑1 (truncated to that length)."""
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """Full correlation (w ⋆ a_rev) needed for the gradient."""
    fft_w = np.fft.fft(w, n=L)
    fft_rev = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev))
    return corr


# ----------------------------------------------------------------------
# Optimisation routine (runs one restart)
# ----------------------------------------------------------------------


def _run_adam(
    a_init: np.ndarray,
    n: int,
    L_conv: int,
    L_corr: int,
    timeout: float,
    best_state: dict,
) -> None:
    """
    Perform Adam optimisation on the simplex starting from ``a_init``.
    Updates ``best_state`` (keys: 'score' and 'a') in place if a better
    candidate is found during the run.
    """
    # hyper‑parameters (tuned for this problem)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    eps_log = 1e-12

    # schedule parameters
    max_iters = 15000
    beta_start = 5.0
    beta_end = 200.0          # more aggressive than the reference
    lr_start = 0.10
    lr_end = 0.01

    # initialise the unconstrained logits
    v = np.log(a_init + eps_log)
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    for it in range(max_iters):
        # time‑check (leave a safety margin)
        if time.time() > timeout:
            break

        # ---- current point on the simplex ----
        a = _softmax(v)                     # a ≥ 0, Σa = 1

        # ---- convolution and current objective ----
        conv = _conv_full(a, L_conv)
        conv_max = conv.max()
        cur_score = 2.0 * n * conv_max      # exactly the value returned by evaluate_sequence

        # ---- keep the best global candidate ----
        if cur_score < best_state["score"]:
            best_state["score"] = cur_score
            best_state["a"] = a.copy()
            print(f"[restart] iter {it:5d}  new best = {cur_score:.6f}")

        # ---- smooth‑max (log‑sum‑exp) weights ----
        # linear β schedule (more weight to the true max as we progress)
        frac = it / max_iters
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start + (lr_end - lr_start) * frac

        # stability: shift by the max before exponentiation
        w_exp = np.exp(beta * (conv - conv_max))
        w = w_exp / w_exp.sum()                 # soft‑max over the 2 n‑1 entries

        # ---- gradient w.r.t. a (2·corr(w, a)) ----
        # correlation is performed via FFT; we need the segment a[n‑1 : 2n‑1]
        corr_full = _correlation_full(w, a[::-1], L_corr)
        grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ---- gradient w.r.t. soft‑max logits v ----
        dot = np.dot(grad_a, a)                 # ⟨grad_a , a⟩
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        t = it + 1
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v2_hat = v2 / (1.0 - beta2 ** t)

        v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    start_vectors.append(hann / hann.sum())

    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    start_vectors.append(hamming / hamming.sum())

    blackman = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    start_vectors.append(blackman / blackman.sum())

    # 4) Gaussian bumps with different widths
    for sigma_frac in (0.15, 0.20, 0.25):
        sigma = sigma_frac * n
        gauss = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
        start_vectors.append(gauss / gauss.sum())

    # 5) exponential decays (forward and backward)
    for rate in (0.001, 0.002, 0.003):
        exp_fwd = np.exp(-rate * i)
        start_vectors.append(exp_fwd / exp_fwd.sum())
        exp_rev = np.exp(-rate * (n - 1 - i))
        start_vectors.append(exp_rev / exp_rev.sum())

    # 6) a few random Dirichlet draws
    for _ in range(5):
        start_vectors.append(np.random.dirichlet(np.ones(n)))

    # -------------------------------------------------------------
    # Global state for the best sequence discovered so far
    best_state = {"score": float("inf"), "a": None}

    # -------------------------------------------------------------
    # Run a separate Adam optimisation for each initial vector.
    # The loop respects the global deadline.
    for idx, a0 in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break
        print(f"--- Restart {idx}/{len(start_vectors)} ---")
        _run_adam(
            a_init=a0,
            n=n,
            L_conv=L_conv,
            L_corr=L_corr,
            timeout=deadline,
            best_state=best_state,
        )

    # -------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    if best_state["a"] is None:
        best_state["a"] = np.full(n, 1.0 / n)

    # Scale to the required sum √(2 n) and clip to the limits accepted by evaluate_sequence
    best_seq = best_state["a"] * target_sum
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished: best score = {best_state['score']:.6f} (n={n}) ---")
    return best_seq.tolist()
