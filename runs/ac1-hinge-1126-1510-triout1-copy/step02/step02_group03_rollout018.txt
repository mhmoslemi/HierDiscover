def conv_full_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << ((n-1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[:n]

the simplex using FFT‑based convolutions.  Occasional evaluations with the

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using FFT (real inputs)."""
    n = len(x) + len(y) - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[:n]


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a (full length 2*len(a)-1) using FFT."""
    return _conv_full_fft(a, a)

    """Soft‑max weights for the smooth‑max surrogate of the peak."""
    # shift by the maximum to keep exponent non‑positive
    shift = conv - conv.max()
    # exponentiate with scaling alpha

    # normalize to sum to 1
    return ex / ex.sum()

    For each k, d conv_k / d a_i = 2 * a_{k-i} (if index valid).
