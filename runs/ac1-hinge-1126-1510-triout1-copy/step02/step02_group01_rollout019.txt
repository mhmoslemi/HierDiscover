"""
Hybrid optimization for the self‑convolution step‑function problem.

The algorithm repeatedly starts from diverse random initialisations,
optimises a smooth approximation of  max(conv)  with L‑BFGS‑B on the
soft‑max parametrisation (ensuring non‑negativity and sum‑to‑one),
applies an aggressive greedy hill‑climb, and finally performs a short
sub‑gradient descent on the *exact* max‑convolution.  Time is split
between many cheap restarts and deeper refinements so that the search
continues up to the provided budget (default 1000 s).  The best sequence
found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (output sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the first 2*n‑1 entries (real part).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1).
    This coincides with evaluate_sequence when sum(a)=1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max version of 2·n·max(conv) and its gradient w.r.t. y.
    The smoothing width is lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth maximum -----
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                 # max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    # Compute via FFT (reverse a then convolve with w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 180,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS‑B optimisations with decreasing smoothing.
    Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # the smooth‑max function uses the current lam through the `lam_factor`
        # argument of `objective_and_grad`.
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive greedy hill‑climb on the probability simplex.
    Randomly moves a fraction of mass from one index to another and
    keeps the move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # parameters for the move size
    max_frac = 0.08             # up to 8 % of the source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_refine(
    y: np.ndarray,
    n: int,
    fft_len: int,
    steps: int,
    deadline: float,
    initial_lr: float = 0.5,
) -> np.ndarray:
    """
    Sub‑gradient descent on the true max‑convolution.
    Works directly on the unconstrained `y` (soft‑max variables).
    """
    rng = np.random.default_rng()
    lr = initial_lr
    for t in range(steps):
        if time.time() >= deadline:
            break
        a = softmax(y)
        conv = conv_fft(a, fft_len, n)
        k_max = int(np.argmax(conv))

        # compute sub‑gradient w.r.t a for the active index k_max
        g = np.zeros_like(a)
        idx_i = np.arange(n)
        idx_j = k_max - idx_i
        mask = (idx_j >= 0) & (idx_j < n)
        g[mask] = 2.0 * a[idx_j[mask]]

        # gradient w.r.t y (soft‑max chain rule)
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # simple step‑size schedule (inverse‑time)
        lr = initial_lr / (t + 1.0)

        # random jitter occasionally to escape plateaus
        if rng.random() < 0.005:
            grad_y += 0.001 * rng.standard_normal(n)

        y = y - lr * grad_y
    return y

        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode} new best fast score = {best_score:.6f}"
            )

        # ---- aggressive hill climbing ----------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final refinement using exact sub‑gradient steps
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_y = subgradient_refine(
            best_y,
            n,
            fft_len,
            steps=600,
            deadline=deadline,
            initial_lr=0.6,
        )
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[post‑subgrad] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy polishing (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=12000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Return the result in the format required by evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
