def convolve_full(a, fft_len):
    # Using rfft for real input
    Fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(Fa * Fa, fft_len)
    return conv[:2*len(a)-1]

def hill_climb_fast(a, fft_len_conv, n, deadline, batch=10000):

        i = rng.integers(0, n)
        j = rng.integers(0, n)

        # optional early revert if max increased

            a[i] += delta

"""
Search for a non‑negative step sequence that minimises
    E(a) = 2·n·max(a*a) / (sum a)^2
using a multi‑phase strategy:
1.  Sweep a few parametric families (exponential, power‑law, linear,
    cosine) and keep the best shape.
2.  Refine the best shape with Adam on a smooth approximation of the
    max‑convolution (log‑sum‑exp) while staying on the simplex via a
    soft‑max re‑parameterisation.
3.  Perform a multiplicative‑gradient descent (mirror‑descent like)
    on the true max with a decaying step–size.
4.  Finish with an aggressive stochastic hill‑climb that transfers
    small amounts of mass between two positions.
All phases respect the overall time budget (default 1000 s) and
periodically check the wall‑clock to stop before the deadline.
The returned list is guaranteed to be non‑negative; the external
`evaluate_sequence` function will enforce the scaling limits.
"""

import time
import math
import numpy as np

def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)

def full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a on the simplex (sum=1)."""
    conv = full_conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())

def smooth_max_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smooth‑max of the convolution using log‑sum‑exp and its gradient w.r.t. y.
    Returns (value, gradient) where gradient has shape (n,).
    """
    a = softmax(y)                 # a lies on the simplex
    # ----- convolution a*a -----
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smooth approximation of max
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient -----
    w = exp_shift / Z                  # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, n=fft_len)
    # extract the part that aligns with original indices (n‑1 … n‑1+n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def adam_optimize(y0: np.ndarray, n: int, fft_len: int, deadline: float,
                  max_iters: int = 400, lam_init: float = 0.30):
    """
    Adam optimisation of the smooth‑max objective.
    Returns the (softmax‑projected) sequence a and its fast score.
    """
    # Adam hyper‑parameters
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    lam = lam_init
    while time.time() < deadline and it < max_iters:
        f_val, grad = smooth_max_and_grad(y, n, fft_len, lam_factor=lam)
        # Adam update on y (unconstrained)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasionally re‑evaluate the true (non‑smoothed) score
        if (it & 31) == 0:   # every 32 iterations
            a = softmax(y)
            scr = fast_score(a, fft_len, n)
            if scr < best_score:
                best_score = scr
                best_a = a.copy()

        # simple schedule for the smoothing parameter
        if it % 50 == 0 and lam > 0.01:
            lam *= 0.9

        it += 1

    return best_a, best_score

def multiplicative_refine(a: np.ndarray, n: int, fft_len: int,
                         deadline: float, init_step: float = 0.2,
                         decay: float = 0.997):
    """
    Mirror‑descent‑like refinement using sub‑gradient of the true max.
    Keeps a on the simplex by a multiplicative update.
    Returns refined a and its score.
    """
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    step = init_step
    it = 0
    while time.time() < deadline:
        # current convolution and active set mask
        conv = full_conv_fft(best_a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient g = 2 * (mask * a_rev) restricted to n entries
        a_rev = best_a[::-1]
        fm = np.fft.rfft(mask, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv_g = np.fft.irfft(fa_rev * fm, n=fft_len)
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # multiplicative update (mirrors KL‑projection)
        upd = np.exp(-step * g)
        a_new = best_a * upd
        a_new /= a_new.sum()
        best_a = a_new

        # evaluate
        cur_score = fast_score(best_a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score

        # decay step size
        step *= decay
        if step < 1e-6:
            break
        it += 1

    return best_a, best_score

def hill_climb(a: np.ndarray, n: int, fft_len: int,
               deadline: float, max_iter: int = 3000):
    """
    Stochastic hill‑climbing: move a random fraction of mass from one index
    to another, keep the move if it reduces the fast score.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

    """
    Apply a single linear‑programming refinement on a down‑sampled version
    of `a`.  Uses cvxpy if available.  Returns refined a (upsampled back to
    original size) and its fast score.  If cvxpy is missing or the solve
    exceeds the remaining time, the original a is returned unchanged.
    """
    try:
        import cvxpy as cp
    except Exception:
        return a, fast_score(a, fft_len, n)

    # down‑sample to at most 800 points (still enough resolution)
    max_small = 800
    if n <= max_small:
        return a, fast_score(a, fft_len, n)
    factor = int(math.ceil(n / max_small))
    n_small = int(math.ceil(n / factor))
    # average pooling
    a_small = a[: n_small * factor].reshape(n_small, factor).mean(axis=1)
    # compute M for the small problem
    fft_small = next_pow2(2 * n_small - 1)
    M = full_conv_fft(a_small, fft_small).max()

    # build convolution matrix A (shape (2*n_small-1, n_small))
    rows = 2 * n_small - 1
    A = np.zeros((rows, n_small))
    for j in range(n_small):
        A[j:j + n_small, j] = a_small

    g = cp.Variable(n_small, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        return a, fast_score(a, fft_len, n)

    if g.value is None:
        return a, fast_score(a, fft_len, n)

    g_opt = np.maximum(g.value, 0.0)
    # normalise to sum 1
    if g_opt.sum() == 0:
        g_norm = a_small
    else:
        g_norm = g_opt / g_opt.sum()

    # optional mixing with current a_small
    t = 0.2
    a_mix = (1 - t) * a_small + t * g_norm
    a_mix /= a_mix.sum()

    # upsample back to original length by repeating each entry `factor` times
    a_up = np.repeat(a_mix, factor)[:n]
    a_up /= a_up.sum()
    return a_up, fast_score(a_up, fft_len, n)

"""
Search for a non‑negative step sequence that minimises

    E(a) = 2·n·max(a*a) / (sum a)^2

The algorithm runs four phases within the given time budget:

1.  A quick sweep over a few parametric families (uniform, linear,
    exponential, power‑law, Hann) plus a handful of random Dirichlet
    draws.  The best shape from this sweep seeds the later phases.

2.  Adam optimisation of a smooth approximation of the max‑convolution.
    The sequence is re‑parameterised by a soft‑max (so it always stays on
    the probability simplex) and Adam with a decaying smoothing factor
    refines the solution.

3.  A mirror‑descent‑like multiplicative refinement that works on the
    true max (no smoothing).  The sub‑gradient is obtained from the
    active set of the convolution and a multiplicative update keeps the
    iterate non‑negative and normalised.

4.  An aggressive stochastic hill‑climb that repeatedly transfers a
    small fraction of mass between two positions and keeps the move
    only if it reduces the fast (FFT‑based) score.

Optionally a single linear‑programming refinement (cvxpy) on a
down‑sampled version of the current sequence can be performed before
phase 3 – it often yields a useful “jump” in quality.

All heavy work uses FFT‑based convolutions (O(n log n)).  The routine
adaptively checks the remaining wall‑clock time and stops before the
deadline, guaranteeing that at least one valid sequence is returned.

The returned list is ready for the external ``evaluate_sequence`` call.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)


def full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) using a real FFT.
    The result is truncated to the needed length.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a already normalised (sum = 1)."""
    conv = full_conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max and its gradient (used by Adam)
# ----------------------------------------------------------------------
def smooth_max_and_grad(y: np.ndarray, n: int, fft_len: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth‑max of the convolution and gradient w.r.t. y.
    Returns (value, gradient) where gradient has shape (n,).
    """
    a = softmax(y)                     # a lives on the simplex

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # smooth approximation of max
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient -----
    w = exp_shift / Z                       # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, n=fft_len)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(y0: np.ndarray, n: int, fft_len: int,
                  deadline: float, max_iters: int = 500,
                  lam_init: float = 0.30):
    """
    Adam optimisation of the smooth‑max objective.
    Returns the best sequence (soft‑max of y) and its fast score.
    """
    lr = 0.05
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    lam = lam_init
    while time.time() < deadline and it < max_iters:
        f_val, grad = smooth_max_and_grad(y, n, fft_len, lam_factor=lam)

        # Adam update on the unconstrained logits
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasionally evaluate the *true* (non‑smoothed) score
        if (it & 31) == 0:
            a = softmax(y)
            scr = fast_score(a, fft_len, n)
            if scr < best_score:
                best_score = scr
                best_a = a.copy()

        # slowly tighten the smoothing
        if it % 50 == 0 and lam > 0.01:
            lam *= 0.9
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Multiplicative (mirror‑descent) refinement on the true max
# ----------------------------------------------------------------------
def multiplicative_refine(a: np.ndarray, n: int, fft_len: int,
                         deadline: float,
                         init_step: float = 0.2,
                         decay: float = 0.997):
    """
    Mirror‑descent‑like refinement using a sub‑gradient of the true max.
    The update is multiplicative, which automatically respects non‑negativity
    and the simplex after renormalisation.
    """
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    step = init_step
    while time.time() < deadline:
        conv = full_conv_fft(best_a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max (2·(mask * a_rev) restricted to n entries)
        a_rev = best_a[::-1]
        fm = np.fft.rfft(mask, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv_g = np.fft.irfft(fa_rev * fm, n=fft_len)
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # multiplicative update and re‑normalisation
        upd = np.exp(-step * g)
        a_new = best_a * upd
        a_new /= a_new.sum()
        best_a = a_new

        cur_score = fast_score(best_a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score

        step *= decay
        if step < 1e-6:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray, n: int, fft_len: int,
               deadline: float, max_iter: int = 3000):
    """
    Randomly move a small fraction of mass from one index to another.
    Keep the move only if it reduces the fast score.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Optional linear‑programming refinement (cvxpy, down‑sampled)
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray, n: int, fft_len: int, deadline: float):
    """
    One LP step on a down‑sampled version of ``a``.
    Maximises sum(g) under the constraint (a_small * g) ≤ M,
    where M is the current max‑convolution.  The result is up‑sampled
    back to the original length and returned.
    If cvxpy is unavailable or the solve would exceed the remaining time,
    the original ``a`` is returned unchanged.
    """
    try:
        import cvxpy as cp
    except Exception:
        return a, fast_score(a, fft_len, n)

    # down‑sample to at most 800 points
    max_small = 800
    if n <= max_small:
        return a, fast_score(a, fft_len, n)

    factor = int(math.ceil(n / max_small))
    n_small = int(math.ceil(n / factor))

    # simple block averaging
    trimmed = a[: n_small * factor]
    a_small = trimmed.reshape(n_small, factor).mean(axis=1)

    # current bound M on the small problem
    fft_small = next_pow2(2 * n_small - 1)
    M = full_conv_fft(a_small, fft_small).max()

    # build the convolution matrix A (shape (2*n_small-1, n_small))
    rows = 2 * n_small - 1
    A = np.zeros((rows, n_small))
    for j in range(n_small):
        A[j:j + n_small, j] = a_small

    g = cp.Variable(n_small, nonneg=True)
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except Exception:
        return a, fast_score(a, fft_len, n)

    if g.value is None:
        return a, fast_score(a, fft_len, n)

    g_opt = np.maximum(g.value, 0.0)
    if g_opt.sum() == 0:
        g_norm = a_small
    else:
        g_norm = g_opt / g_opt.sum()

    # a small convex combination of the old and the LP solution
    t = 0.2
    a_mix = (1 - t) * a_small + t * g_norm
    a_mix /= a_mix.sum()

    # up‑sample by repeating each entry ``factor`` times
    a_up = np.repeat(a_mix, factor)[:n]
    a_up /= a_up.sum()
    return a_up, fast_score(a_up, fft_len, n)


# ----------------------------------------------------------------------
# Parametric families for quick initial guesses
# ----------------------------------------------------------------------
def init_shape_uniform(n):
    return np.full(n, 1.0 / n)


def init_shape_linear(n):
    # strictly decreasing linear ramp
    x = np.arange(n, 0, -1, dtype=float)
    return x / x.sum()


def init_shape_exponential(n, rate):
    x = np.exp(-rate * np.arange(n))
    return x / x.sum()


def init_shape_power(n, p):
    x = (np.arange(1, n + 1, dtype=float) ** (-p))
    return x / x.sum()


def init_shape_hann(n):
    x = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    return x / x.sum()


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (reproducibility).
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            max_n = 4000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            init_seq[init_seq < 0] = 0.0
            if init_seq.sum() == 0.0:
                a0 = init_shape_uniform(n)
            else:
                a0 = init_seq / init_seq.sum()
        else:
            n = 4000
            a0 = init_shape_uniform(n)
    else:
        n = 4000
        a0 = init_shape_uniform(n)

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Phase 0 – quick sweep over analytic families.
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[phase0] initial fast_score = {best_score:.6f}")

    candidates = [
        init_shape_uniform(n),
        init_shape_linear(n),
        init_shape_hann(n)
    ]

    # power‑law sweep
    for p in np.linspace(0.2, 2.5, 12):
        candidates.append(init_shape_power(n, p))

    # exponential sweep (log‑spaced rates)
    for rate in np.geomspace(0.001, 0.02, 12):
        candidates.append(init_shape_exponential(n, rate))

    # a few random Dirichlet draws
    rng = np.random.default_rng(seed + 999)
    for _ in range(8):
        candidates.append(rng.dirichlet(np.ones(n)))

    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[phase0] new best from family {idx} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on a smoothed max.
    # ------------------------------------------------------------------
    eps = 1e-12
    y0 = np.log(best_a + eps)
    adam_runs = 0
    while time.time() < deadline - 5.0:
        if rng.random() < 0.3:
            # perturb the current logits a bit
            y_start = np.log(np.maximum(best_a, eps)) + rng.normal(scale=0.5, size=n)
        else:
            y_start = rng.standard_normal(n)

        a_opt, sc_opt = adam_optimize(
            y_start, n, fft_len,
            deadline=deadline,
            max_iters=500,
            lam_init=0.30
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[adam] run {adam_runs} improved score={best_score:.6f}")
        adam_runs += 1

    # ------------------------------------------------------------------
    # Optional LP refinement (if we still have time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        best_a, best_score = lp_refine(best_a, n, fft_len, deadline)
        print(f"[lp] post‑LP score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – multiplicative (mirror‑descent) refinement on the true max.
    # ------------------------------------------------------------------
    mul_deadline = min(deadline, time.time() + 30.0)
    best_a, best_score = multiplicative_refine(
        best_a, n, fft_len,
        deadline=mul_deadline,
        init_step=0.15,
        decay=0.995
    )
    print(f"[mult] post‑refine score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb polish.
    # ------------------------------------------------------------------
    best_a, best_score = hill_climb(
        best_a, n, fft_len,
        deadline=deadline,
        max_iter=5000
    )
    print(f"[final] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert to list, evaluate once (required by the outer driver) and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
