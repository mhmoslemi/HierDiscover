Parametric optimizer for the self‑convolution bound.

- Build families a_i ∝ (i+β)^{-p}, p∈[0.55,0.68], β∈[0,5].
- Normalise to unit L1 norm.
- Compute the objective 2·n·max(a * a) with a fast FFT based convolution.
- Keep the best family and optionally polish it with a simple random
  mass‑exchange hill‑climber.
- Return the best sequence found within the time budget.
"""

import time
import numpy as np

def _max_self_conv(seq: np.ndarray) -> float:
    """Maximum entry of the linear convolution of `seq` with itself.
    Uses FFT for O(n log n) time.
    """
    n = seq.shape[0]
    L = 2 * n - 1
    # real FFT of zero‑padded sequence
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    # The result is real; tiny negative values may appear because of
    # floating‑point noise – clip them.
    return float(conv.max())

def _score(seq: np.ndarray) -> float:
    """Score defined by evaluate_sequence for a normalised sequence."""
    # seq should already satisfy sum(seq)==1 (or very close)
    n = seq.shape[0]
    max_b = _max_self_conv(seq)
    return 2.0 * n * max_b

def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence of non‑negative heights with a low self‑convolution
    bound.  The routine explores a two‑parameter family
        a_i ∝ (i + β)^{‑p}
    and then performs a short stochastic hill‑climb.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    n = int(kwargs.get("n", 2000))   # length of the sequence, default 2000

    best_seq = None
    best_score = float("inf")

    # -----------------------------------------------------------
    # 1) Grid search over (p, beta)
    # -----------------------------------------------------------
    # exponent p in (0.55, 0.68) – the region around the half‑order value.
    # beta shifts the whole curve to the right, reducing the weight of a_0.
    p_values = np.linspace(0.55, 0.68, 14)   # step ≈0.009
    beta_values = np.linspace(0.0, 5.0, 11)  # 0,0.5,...,5

    # pre‑compute the index array once
    idx = np.arange(1, n + 1, dtype=np.float64)   # start at 1 to avoid 0⁰

    for p in p_values:
        for beta in beta_values:
            # construct the raw (unnormalised) coefficients
            raw = (idx + beta) ** (-p)
            total = raw.sum()
            if total == 0.0:
                continue
            seq = raw / total   # now sum(seq) == 1
            sc = _score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[grid] p={p:.4f}, beta={beta:.2f} → score {sc:.6f}")
            # early stop if we already beat the target
            if best_score < 1.45:
                # we are comfortably below the required 1.503,
                # continue a little for robustness but we can break later.
                pass

            if time.time() >= deadline:
                print("Time limit reached during grid search")
                return best_seq.tolist()

    # -----------------------------------------------------------
    # 2) Simple hill‑climbing from the best grid point
    # -----------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    n = cur_seq.shape[0]

    move_fraction = 0.02      # up to 2 % of the source mass is transferred
    attempts = 0
    max_attempts = 20000      # limited by the remaining time

    while time.time() < deadline and attempts < max_attempts:

        i, j = np.random.choice(n, 2, replace=False)
        # amount we are willing to move from i to j
        max_delta = cur_seq[i] * move_fraction
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta
        cand_seq = cur_seq.copy()
        cand_seq[i] -= delta
        cand_seq[j] += delta
        # recompute the score (no need to renormalise – sum unchanged)
        cand_score = _score(cand_seq)
        if cand_score < cur_score:
            cur_seq = cand_seq
            cur_score = cand_score
            if cur_score < best_score:

                best_seq = cur_seq.copy()

            cur_seq = jitter
            cur_score = _score(cur_seq)

    print(f"Finished after {attempts} hill‑climb steps, best score = {best_score:.6f}")

    # shift beta ∈ [0, 5] – moves mass away from the very first entry
    beta_vals = np.linspace(0.0, 5.0, 11)         # 0,0.5,…,5

    # Pre‑compute the index vector i = 1,…,n (using 1‑based indices avoids
    # division by zero when beta = 0).
    idx = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for beta in beta_vals:
            # raw (unnormalised) coefficients a_i ∝ (i + beta)^{-p}
            raw = (idx + beta) ** (-p)
            total = raw.sum()
            if total == 0.0:
                continue
            seq = raw / total               # normalise to unit L1‑norm
            score = _score(seq)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[grid] p={p:.4f}, beta={beta:4.1f} → score {score:.6f}")

            # stop early if we have already beat the target comfortably
            if best_score < 1.45:
                # we keep scanning the grid a little longer to gather a good
                # starting point for the hill‑climb, but this flag can be used
                # to break out if desired.
                pass

            if time.time() >= deadline:
                print("Time limit reached during grid search.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # 2) Random hill‑climbing (mass‑exchange moves)
    # ------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # the hill‑climber moves at most `move_frac` of the source mass
    move_frac = 0.02
    attempts = 0
    max_attempts = 20000                     # limited by remaining time

    while time.time() < deadline and attempts < max_attempts:
        attempts += 1
        # pick two distinct indices i → source, j → destination
        i, j = np.random.choice(n, 2, replace=False)
        max_delta = cur_seq[i] * move_frac
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta

        cand_seq = cur_seq.copy()
        cand_seq[i] -= delta
        cand_seq[j] += delta

        cand_score = _score(cand_seq)
        if cand_score < cur_score:
            cur_seq = cand_seq
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur_seq.copy()
                print(f"[hill] iter {attempts:05d} → score {best_score:.6f}")

        # occasional jitter to escape shallow plateaux
        if attempts % 5000 == 0:
            noise = np.random.randn(n) * 1e-4
            jitter = cur_seq * (1.0 + noise)
            jitter = np.clip(jitter, 0.0, None)
            jitter /= jitter.sum()
            cur_seq = jitter
            cur_score = _score(cur_seq)

    print(f"Finished after {attempts} hill‑climb iterations, best score = {best_score:.6f}")
    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
```
