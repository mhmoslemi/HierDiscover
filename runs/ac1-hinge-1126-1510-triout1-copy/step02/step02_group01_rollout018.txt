"""
Propose a low‑evaluation step sequence.

The algorithm is a multi‑stage optimiser for the objective

    2·n·max(a*a) / (sum(a)^2)

with a_i ≥ 0 (the sum is irrelevant because the score is scale‑invariant).
We work on the probability simplex (sum = 1) by parametrising

    a = softmax(y)

so the constraints are automatically satisfied.

Stage 1 – many random restarts:
    - initialise y from several families (uniform, exponential, power‑law,
      “edge‑heavy”, or a noisy copy of the current best).
    - run a continuation of L‑BFGS‑B on a smooth approximation of the
      max‑convolution, gradually decreasing the smoothing factor.

Stage 2 – targeted greedy hill‑climb:
    - repeatedly identify the shift k where the convolution attains its
      maximum.
    - move a small fraction of mass from an index that participates in the
      peak (chosen with probability proportional to a_i·a_{k‑i}) to a random
      index, accepting the move only if the true max‑convolution improves.
    - this direct optimisation of the true objective often yields
      additional gains beyond the smooth L‑BFGS phase.

Stage 3 – final refinement:
    - a short extra L‑BFGS continuation followed by a quick random hill‑climb.

All heavy lifting (convolutions, smooth‑max and its gradient) is performed
with FFTs for speed; the true max‑convolution used in hill‑climbing is also
computed via FFT (as a fast equivalent of `np.convolve`).  The routine runs
for the allotted budget (default 1000 s) and finally returns the best
sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max objective and its gradient w.r.t. the unconstrained vector y.
    The smoothing width is lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth‑max of the convolution
    shifted = conv - max_c          # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z               # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 120,
) -> np.ndarray:
    """Run a continuation of L‑BFGS‑B for a list of smoothing factors."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


def hill_climb_random(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb: move a random fraction of mass from i→j.
    Accept only strictly improving moves (according to the exact max‑convolution).
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def hill_climb_targeted(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    move_frac: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always attacks the shift where the convolution peaks.
    A small amount of mass (up to ``move_frac`` of the current entry) is moved
    from an index participating in the peak to a random other index.
    Accept only if the true max‑convolution improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # compute full convolution
        conv = conv_fft(best, fft_len, n)
        k = int(conv.argmax())               # shift where max occurs
        # indices that contribute to this shift
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idxs = np.arange(i_start, i_end + 1)

        # compute contribution weights a_i * a_{k-i}
        contrib = best[idxs] * best[k - idxs]
        if contrib.sum() == 0.0:
            # fallback to random move
            i = rng.integers(0, n)
        else:
            # pick i with probability proportional to its contribution
            probs = contrib / contrib.sum()
            i = rng.choice(idxs, p=probs)

        # choose a destination index j (different from i)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The algorithm aims to finish a little
        before the deadline to avoid the safety cut‑off.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 8000  # default length – larger than the previous 5000
    # reuse a previously discovered sequence if it is longer
    try:
        init_seq = globals()["height_sequence_1"]
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            arr = np.asarray(init_seq, dtype=float)
            if arr.ndim == 1 and arr.size > n:
                n = int(arr.size)
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial candidates (uniform and a few analytic shapes)
    # ------------------------------------------------------------------
    uniform_a = np.full(n, 1.0 / n)
    best_a = uniform_a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    # exponential shape with c≈1 (ε = c/n)
    c0 = 1.0
    eps = c0 / n
    r = 1.0 - eps
    exp_a = (1 - r) * (r ** np.arange(n))
    exp_a /= exp_a.sum()
    exp_y = np.log(exp_a + 1e-12)

    # power‑law shape
    p0 = 1.0
    pow_a = (np.arange(1, n + 1, dtype=float) ** (-p0))
    pow_a /= pow_a.sum()
    pow_y = np.log(pow_a + 1e-12)

    print(f"[init] n={n}, uniform score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007, 0.003, 0.001]
    maxiter_per_stage = 150

    iteration = 0
    # Phase 1 – many L‑BFGS restarts
    while time.time() < deadline:
        # choose a starting point
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            y0 = exp_y.copy()
            # add a tiny random jitter to avoid staying stuck
            y0 += rng.normal(scale=0.05, size=n)
        elif mode == "power":
            y0 = pow_y.copy()
            y0 += rng.normal(scale=0.05, size=n)
        elif mode == "edge":
            # push a modest amount of mass to the first and last 2 % of indices
            m_edge = max(1, int(n * rng.uniform(0.015, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # L‑BFGS continuation
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved score={best_score:.6f}")

        # occasional aggressive random hill‑climb
        if iteration % 5 == 0:
            hc_deadline = min(deadline, time.time() + 0.8)
            best_a, best_score = hill_climb_random(
                best_a,
                best_score,
                fft_len_conv,
                n,
                hc_deadline,
                max_iter=3000,
            )
            best_y = np.log(best_a + 1e-12)

        iteration += 1
        # leave a small margin for the final refinement
        if time.time() + 30.0 > deadline:
            break

    # ------------------------------------------------------------------
    # Phase 2 – targeted greedy refinement (focus on the current peak)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hc_deadline = deadline - 2.0
        best_a, best_score = hill_climb_targeted(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=8000,
            move_frac=0.03,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 3 – final L‑BFGS pass with a finer smoothing schedule
    # ------------------------------------------------------------------
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 250, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[refine] lam={lam:.4f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Very short random hill‑climb if any time remains
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_random(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=6000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Return the best sequence (as plain list) – the official evaluator
    # will compute the real score.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # The following call is only for logging; the harness will call it again.
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception:
        # In case `evaluate_sequence` is not imported in the local test harness.
        pass
    return final_seq
