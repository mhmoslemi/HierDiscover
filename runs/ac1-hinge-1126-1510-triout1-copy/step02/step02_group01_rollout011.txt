def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    n=5000. Possibly adapt based on height_sequence_1 if bigger.

    fft_len_conv = next_pow2(2*n-1)
    fft_len_g = next_pow2(3*n-2)

    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print("[init] uniform fast score:", best_score)

    # Define lam schedule more refined:
    lam_schedule = [0.5,0.3,0.15,0.07,0.035,0.018,0.009,0.004,0.002,0.001]

    # Number of restarts and while loop
    restart_count = 0
    while time.time() < deadline - 5:  # keep time for hill climb final
        # choose init mode
        mode = rng.choice(["random","exp","power","edge","dirichlet","best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001,0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3,2.0)
            a0 = (np.arange(1, n+1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01,0.04)))
            a_center = 1.0 / np.sqrt(2.0) / n
            a_edge = np.sqrt(n/2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "dirichlet":
            concentration = rng.uniform(0.01, 0.5)  # low concentration encourages sparsity
            a0 = rng.gamma(shape=concentration, scale=1.0, size=n)
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # Run schedule optimization
        y_opt = schedule_optimize(y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage=200)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[restart {restart_count}] mode={mode} improved fast score={best_score:.6f}")

"""
Propose a low‑evaluation step sequence.

The search consists of three phases:

1. **Exploratory L‑BFGS restarts** – many random starts on the
   probability simplex are refined with a continuation schedule that
   minimises a smooth approximation of the max‑convolution.  A short
   greedy hill‑climb is applied after each run.

2. **Targeted stochastic hill‑climb** – the current best sequence is
   improved by repeatedly moving a small fraction of the mass from an
   index that contributes heavily to the maximal convolution entry to an
   index that contributes little.  An annealing acceptance rule allows
   occasional uphill moves.

3. **Final polishing** – a few extra L‑BFGS passes with a very small
   smoothing parameter and another short hill‑climb are performed.

All operations are done on the simplex (sum‑to‑one) so the official
evaluator returns ``2·n·max(a*a)``.  The algorithm stays well within the
1000 s budget and attempts to beat the previous bound of 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping to the probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (zero‑padded) using FFT.
    Returns an array of length 2·n‑1 (the valid part of the linear conv).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the official objective for a probability vector.
    Returns 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t.
    the unconstrained logit vector y.

    ``lam_factor`` scales the smoothing width relative to the current
    maximum convolution entry.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                           # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    # we need a reversed copy of a for correlation
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 120,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS‑B optimisations.
    Each stage uses a smaller smoothing factor.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: repeatedly move a random fraction of mass
    from i to j (keeping the sum = 1).  Keep a move only if the fast
    score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def compute_weights(a: np.ndarray, max_indices: np.ndarray) -> np.ndarray:
    """
    For each position i compute
        w_i = Σ_{k∈max_indices} a_{k‑i}
    (terms outside the range are ignored).  This is exactly the total
    contribution of a_i to the currently maximal convolution entries.
    """
    n = a.size
    w = np.zeros_like(a)
    for k in max_indices:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # vector of i values
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        w[i_vals] += a[j_vals]
    return w


def directed_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    step_factor: float = 0.10,
    temp_start: float = 1e-3,
    temp_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climb.  At each iteration the algorithm
    identifies the indices that contribute most/least to the maximal
    convolution entry(s) and moves a small fraction of mass from the
    former to the latter.  An annealing acceptance rule lets the search
    escape shallow local minima.
    """
    best = a.copy()
    best_score = cur_score
    T = temp_start

    while time.time() < deadline:
        # current convolution & its maximum
        conv = conv_fft(best, fft_len, n)
        max_val = conv.max()
        # indices where the maximum is attained (within tiny tolerance)
        max_idx = np.where(conv >= max_val - 1e-12)[0]

        # compute contribution weights
        w = compute_weights(best, max_idx)

        # source: heavy contributor that also carries mass
        src_candidates = np.where(best > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src_scores = w[src_candidates] * best[src_candidates]
        i = src_candidates[np.argmax(src_scores)]

        # destination: light contributor (avoid choosing i)
        dst_candidates = np.delete(np.arange(n), i)
        j = dst_candidates[np.argmin(w[dst_candidates])]

        # move a small fraction of mass from i to j
        delta = best[i] * step_factor * rng.random()
        if delta <= 0.0:
            continue
        new = best.copy()
        new[i] -= delta
        new[j] += delta

        new_score = fast_score(new, fft_len, n)

        if new_score < best_score:
            best = new
            best_score = new_score
        else:
            # annealing acceptance
            prob = np.exp(-(new_score - best_score) / (T + 1e-15))
            if rng.random() < prob:
                best = new
                best_score = new_score

        # schedule updates
        T *= temp_decay
        step_factor *= 0.9999  # slowly shrink step size

    return best, best_score
