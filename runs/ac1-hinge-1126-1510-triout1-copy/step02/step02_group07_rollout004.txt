"""
Hybrid stochastic‑gradient optimizer for the self‑convolution bound.

The algorithm works with a probability simplex representation (a_i ≥ 0, Σa_i = 1)
because the evaluation function is homogeneous of degree zero.
It combines three complementary move types:

1. **Pair exchange** – move a small fraction of mass from one index to another.
2. **Multi‑index redistribution** – pick a few indices, collect their mass and
   split it again according to a Dirichlet draw.
3. **Projected (soft) sub‑gradient step** – compute a smooth approximation of the
   max‑convolution using a soft‑max (temperature β) and take a back‑tracking
   line‑search step on the simplex.

An annealing temperature gradually cools the acceptance of non‑improving
candidates.  The best sequence found up to the deadline is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """Central binomial (half‑order) coefficients, normalised to Σa = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v <= 0):
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score(seq: np.ndarray) -> float:
    """Exact evaluation of the objective (2·n·max(conv) / sum^2)."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights on the convolution entries (numerically stable)."""
    # shift for stability
    z = beta * conv
    z -= np.max(z)
    expz = np.exp(z)
    return expz / expz.sum()


def _soft_gradient(seq: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the soft‑max approximation of max(conv).
    Implements the formula
        grad_i = 2 Σ_j a_j * w_{i+j}
    using a single convolution (correlation) for O(n log n) time.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)                     # length 2n‑1
    w = _softmax_weights(conv, beta)                # same length
    w_rev = w[::-1]                                 # reverse for correlation
    # correlation via convolution with reversed weight
    corr = np.convolve(seq, w_rev)                  # length 3n‑2
    grad = 2.0 * corr[(2 * n - 2) - np.arange(n)]   # pick the needed slice
    return grad

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    evaluate_sequence(seq).  The routine runs for at most `budget_s` seconds
    and returns the best sequence found as a plain Python list of floats.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Parameters (can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 1000))

    # Temperature schedule
    init_temp_factor = 0.5
    final_temp = 1e-12

    # Gradient‑step parameters
    beta = 5.0            # initial soft‑max temperature
    beta_max = 200.0
    beta_inc = 1.02      # multiplicative increase per 1000 iterations
    step_g = 0.2         # initial step size for projected gradient
    step_g_min = 1e-8
    step_g_max = 0.5

    # Move probabilities
    prob_pair = 0.4
    prob_multi = 0.2
    prob_grad = 0.4

    # ------------------------------------------------------------------
    # Initialise a collection of candidate start points
    # ------------------------------------------------------------------
    starters = []

    # (A) half‑order series
    starters.append(_half_order_series(n))

    # (B) a few random Dirichlet draws (diverse shapes)
    for _ in range(3):
        starters.append(np.random.dirichlet(np.ones(n)))

    # (C) optional global best if provided by the framework
    if "height_sequence_1" in globals():
        hs = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if hs.size == n:
            starters.append(hs / hs.sum())
        else:
            # adapt length if possible (crop or pad with zeros)
            if hs.size > n:
                starters.append(hs[:n] / hs[:n].sum())
            else:
                pad = np.zeros(n, dtype=np.float64)
                pad[: hs.size] = hs
                starters.append(pad / pad.sum())

    # (D) simple sweeps over exponent tilt and uniform mixing (inspired by prior code)
    base = _half_order_series(n)
    for delta in np.linspace(0.0, 0.8, 9):
        taper = (np.arange(1, n + 1, dtype=np.float64) ** (-delta))
        cand = base * taper
        cand /= cand.sum()
        starters.append(cand)

    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    for alpha in np.linspace(0.0, 1.0, 5):
        cand = (1.0 - alpha) * base + alpha * uniform
        cand /= cand.sum()
        starters.append(cand)

    # Pick the best among them as the initial working sequence
    best_seq = None
    best_val = np.inf
    for cand in starters:
        val = _score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    cur_seq = best_seq.copy()
    cur_val = best_val
    cur_max = np.convolve(cur_seq, cur_seq).max()

    best_seq = cur_seq.copy()
    best_val = cur_val
    best_max = cur_max

    # ------------------------------------------------------------------
    # Main optimisation loop (simulated annealing + gradient steps)
    # ------------------------------------------------------------------
    iteration = 0
    total_time = deadline - start

    while time.time() < deadline:
        iteration += 1
        now = time.time()
        elapsed = now - start
        progress = elapsed / total_time

        # Annealing temperature for Metropolis acceptance
        init_temp = cur_max * init_temp_factor
        temp = init_temp * (final_temp / init_temp) ** progress

        # Choose a move type
        r = np.random.rand()
        if r < prob_pair:
            # --------------------------------------------------------------
            # 1) Pair exchange – move a small fraction from i to j
            # --------------------------------------------------------------
            i, j = np.random.choice(n, 2, replace=False)
            if cur_seq[i] <= 0:
                continue
            max_delta = min(cur_seq[i], 0.03)  # up to 3 % of the source mass
            delta = np.random.rand() * max_delta
            cand_seq = cur_seq.copy()
            cand_seq[i] -= delta
            cand_seq[j] += delta
            # compute candidate score
            conv = np.convolve(cand_seq, cand_seq)
            cand_max = conv.max()
            cand_val = 2.0 * n * cand_max

        elif r < prob_pair + prob_multi:
            # --------------------------------------------------------------
            # 2) Multi‑index redistribution (Dirichlet)
            # --------------------------------------------------------------
            m = np.random.randint(2, 6)                     # number of indices
            idx = np.random.choice(n, m, replace=False)
            total_mass = cur_seq[idx].sum()
            if total_mass <= 0:
                continue
            # new proportions from a Dirichlet distribution
            new_props = np.random.dirichlet(np.ones(m))
            cand_seq = cur_seq.copy()
            cand_seq[idx] = total_mass * new_props
            conv = np.convolve(cand_seq, cand_seq)
            cand_max = conv.max()
            cand_val = 2.0 * n * cand_max

        else:
            # --------------------------------------------------------------
            # 3) Projected soft‑gradient step (smooth approximation of max)
            # --------------------------------------------------------------
            # compute soft‑max weights on the current convolution
            conv_cur = np.convolve(cur_seq, cur_seq)
            w = _softmax_weights(conv_cur, beta)

            # gradient of the smooth objective
            grad = _soft_gradient(cur_seq, beta)

            # back‑tracking line search on the simplex
            improved = False
            trial_step = step_g
            for _ in range(5):
                cand_seq = cur_seq - trial_step * grad
                cand_seq = _project_simplex(cand_seq)
                conv = np.convolve(cand_seq, cand_seq)
                cand_max = conv.max()
                cand_val = 2.0 * n * cand_max
                if cand_max < cur_max:
                    improved = True
                    break
                # optionally accept worse move via Metropolis rule
                if temp > 0 and np.random.rand() < math.exp(-(cand_max - cur_max) / temp):
                    improved = True
                    break
                trial_step *= 0.5

            if not improved:
                # shrink global step size if we failed to improve
                step_g = max(step_g * 0.7, step_g_min)
                # occasional jitter to escape plateaus
                if iteration % 500 == 0:
                    sigma = 0.001
                    noise = np.random.randn(n) * sigma
                    jitter = cur_seq * (1.0 + noise)
                    jitter = np.clip(jitter, 0.0, None)
                    jitter = jitter / jitter.sum()
                    cur_seq = _project_simplex(jitter)
                    cur_max = np.convolve(cur_seq, cur_seq).max()
                    cur_val = 2.0 * n * cur_max
                continue

            # successful gradient step – adapt step size upwards
            step_g = min(step_g * 1.1, step_g_max)

        # ------------------------------------------------------------------
        # Acceptance decision (Metropolis for non‑improving moves)
        # ------------------------------------------------------------------
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            if temp > 0 and np.random.rand() < math.exp(-(cand_val - cur_val) / temp):
                accept = True

        if accept:
            cur_seq = cand_seq
            cur_val = cand_val
            cur_max = cand_max

        # ------------------------------------------------------------------
        # Keep global best
        # ------------------------------------------------------------------
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq.copy()
            best_max = cur_max
            print(f"[{iteration:7d}] New best score: {best_val:.6f}")

        # ------------------------------------------------------------------
        # Heuristic schedule updates
        # ------------------------------------------------------------------
        if iteration % 1000 == 0:
            # slowly increase the soft‑max sharpness β
            beta = min(beta * beta_inc, beta_max)
            # occasional diagnostic
            print(f"[{iteration:7d}] best={best_val:.6f}, β={beta:.2f}, temp={temp:.2e}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_val:.6f}")
    return best_seq.tolist()
