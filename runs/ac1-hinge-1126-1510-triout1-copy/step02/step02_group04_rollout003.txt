"""
Multi‑start Adam + sub‑gradient polishing for the step‑function bound.

The algorithm works on the simplex Σ a_i = 1 (the objective is scale‑invariant).
For each start vector it runs Adam on the soft‑max parameters of a, using a
smooth log‑sum‑exp approximation of the max‑convolution.  The smoothness
parameter β is increased from a small value to a large one, so the landscape
becomes sharp near the end.

After the Adam phase a short sub‑gradient descent (exact max, not smooth) is
performed on the simplex (projected gradient descent).  Finally a tiny random
local‑search phase tries to squeeze a little more out of the best candidate.

All steps respect the 1000 s time budget, print occasional progress messages,
and return the best sequence (scaled to Σ a = √(2 n) as required by the
reference code).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector (sum = 1)."""
    vmax = v.max()
    e = np.exp(v - vmax)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a (full length = 2·len(a)‑1) using FFT of length L."""
    fa = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a_rev).  The result has length L; the relevant slice
    for the gradient is [n‑1 : n‑1+n].
    """
    fw = np.fft.fft(w, n=L)
    fa = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fw * fa))
    return corr


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # Sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find rho = max { j | u_j > (cumsum_j - 1)/j }
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_score(a: np.ndarray, n: int, L_conv: int) -> float:
    """
    Returns the original evaluation (2·n·max(conv(a,a))) for a on the simplex.
    """
    conv = _conv_full(a, L_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

        2·n·max(conv(a,a)) / (sum(a)²)

    Returns the best sequence found (scaled so that Σ a = √(2 n)).
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a few seconds buffer for interpreter shutdown
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Choose the length of the sequence.
    # ------------------------------------------------------------------
    try:
        n0 = len(height_sequence_1)  # type: ignore
    except Exception:
        n0 = 6000

    # Use a slightly larger length (up to ~12000) – longer sequences tend
    # to give a tighter asymptotic bound while staying fast to FFT.
    if n0 < 8000:
        n = 10000
    elif n0 < 12000:
        n = 12000
    else:
        n = n0

    target_sum = math.sqrt(2.0 * n)          # normalisation used later
    print(f"--- Optimising a sequence of length n={n} (target Σ={target_sum:.3f}) ---")

    # Pre‑compute FFT lengths (next powers of two)
    L_conv = 1 << ((2 * n - 1).bit_length())   # for a⋆a
    L_corr = 1 << ((3 * n - 2).bit_length())   # for correlation with weights

    # ------------------------------------------------------------------
    # Helper to keep track of the best solution so far.
    # ------------------------------------------------------------------
    best_seq = None          # will hold the *scaled* sequence
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Build a collection of diverse starting points on the simplex.
    # ------------------------------------------------------------------
    start_vectors = []

    # 1) previous best (interpolated if lengths differ)
    try:
        prev = np.asarray(height_sequence_1, dtype=float)   # type: ignore
        if prev.size != n:
            # linear interpolation onto the new grid
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            prev = np.interp(x_new, x_old, prev)
        prev = np.maximum(prev, 0.0)
        prev /= prev.sum()
        start_vectors.append(prev)
    except Exception:
        pass

    # 2) uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # 3) random Dirichlet (moderately concentrated)
    for _ in range(2):
        start_vectors.append(np.random.dirichlet(np.ones(n)))

    # 4) sparse Dirichlet (α < 1 gives spiky vectors)
    for _ in range(2):
        alpha = np.full(n, 0.3)
        start_vectors.append(np.random.dirichlet(alpha))

    # 5) simple power‑law decreasing shapes
    for p in (0.2, 0.5, 1.0, 1.5, 2.0, 3.0):
        w = np.arange(1, n + 1, dtype=float) ** (-p)   # decreasing
        a = w / w.sum()
        start_vectors.append(a)

    # 6) exponential decays (r close to 1)
    for r in (0.995, 0.998, 0.999):
        w = r ** np.arange(n)
        a = w / w.sum()
        start_vectors.append(a)

    # ------------------------------------------------------------------
    # Adam optimiser (soft‑max parametrisation)
    # ------------------------------------------------------------------
    base_beta_start = 0.5
    base_beta_end = 300.0
    base_lr = 0.06
    max_iters_per_restart = 12000

    restart_id = 0
    total_restarts = len(start_vectors)

    for a_init in start_vectors:
        if time.time() > deadline:
            break
        restart_id += 1

        # Randomise a few hyper‑parameters per restart to increase diversity
        beta_start = base_beta_start * (0.5 + np.random.rand())
        beta_end = base_beta_end * (0.5 + np.random.rand())
        lr = base_lr * (0.5 + np.random.rand())

        # Initialise soft‑max parameters v
        eps = 1e-14
        v = np.log(a_init + eps)

        m = np.zeros_like(v)          # first moment
        v2 = np.zeros_like(v)         # second moment (Adam's v)

        for it in range(max_iters_per_restart):
            if time.time() > deadline:
                break

            # ----- a on the simplex (soft‑max) -----
            a = _softmax(v)                     # a ≥ 0, Σ a = 1

            # ----- current objective (exact, no smoothing) -----
            conv = _conv_full(a, L_conv)
            conv_max = conv.max()
            cur_score = 2.0 * n * conv_max

            # ----- keep the best sequence seen so far -----
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a * target_sum          # scale to required Σ
                print(f"[restart {restart_id}/{total_restarts}] iter {it:5d}  new best = {best_score:.6f}")

            # ----- smooth‑max weights (log‑sum‑exp) -----
            beta = beta_start + (beta_end - beta_start) * (it / max_iters_per_restart)
            # stability shift by the maximum before exponentiation
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()               # soft‑max over convolution entries

            # ----- gradient w.r.t. a (2·corr(w,a)) -----
            # correlation returns length L_corr; keep the central n entries.
            corr_full = _correlation_full(w, a[::-1], L_corr)
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]

            # ----- gradient w.r.t. soft‑max parameters v -----
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ----- Adam update (with simple learning‑rate decay) -----
            t = it + 1
            m = 0.9 * m + 0.1 * grad_v
            v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

            m_hat = m / (1.0 - 0.9 ** t)
            v2_hat = v2 / (1.0 - 0.999 ** t)

            # optional decay of the base learning rate
            lr_t = lr * (1.0 - it / max_iters_per_restart) ** 0.5
            v -= lr_t * m_hat / (np.sqrt(v2_hat) + 1e-12)

    # ------------------------------------------------------------------
    # Sub‑gradient polishing (exact max, projected gradient descent)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fall back to uniform if nothing better was found
        best_seq = np.full(n, target_sum / n)

    # Work on the simplex again (sum = 1)
    a = best_seq / target_sum

    sub_iters = 5000
    for it in range(sub_iters):
        if time.time() > deadline:
            break

        conv = _conv_full(a, L_conv)
        max_val = conv.max()
        # Indicator for all maximising indices (may be more than one)
        w = (conv == max_val).astype(float)
        w_sum = w.sum()
        if w_sum == 0:
            break
        w /= w_sum

        # Gradient of the exact max‑convolution
        corr_full = _correlation_full(w, a[::-1], L_corr)
        grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]

        # Simple decreasing step size
        eta = 0.5 * (1.0 - it / sub_iters)

        # Projected gradient descent on the simplex
        a = _project_simplex(a - eta * grad_a)

        # Track improvement
        cur_score = 2.0 * n * _conv_full(a, L_conv).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a * target_sum
            if (it % 500) == 0:
                print(f"[sub‑grad] iter {it:4d}  new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Light random‑local‑search around the current best
    # ------------------------------------------------------------------
    a = best_seq / target_sum      # back to the simplex representation
    rnd_iters = 2000
    for it in range(rnd_iters):
        if time.time() > deadline:
            break
        # multiplicative noise (log‑normal perturbation)
        noise = np.exp(0.05 * np.random.randn(n))
        a_candidate = a * noise
        a_candidate /= a_candidate.sum()

        cur_score = 2.0 * n * _conv_full(a_candidate, L_conv).max()
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_candidate * target_sum
            a = a_candidate                # accept the move
            print(f"[local‑search] iter {it:4d}  new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final post‑processing: clipping to the limits enforced by evaluate_sequence
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished optimisation: best score = {best_score:.6f} (n={n}) ---")
    return best_seq.tolist()
