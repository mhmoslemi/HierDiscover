def gradient_softmax(a, b, beta):
    # b is full conv result length 2n-1.
    # compute p weights.
    max_b = b.max()
    # compute w = exp(beta*(b - max_b))
    w = np.exp(beta * (b - max_b))
    sum_w = w.sum()
    p = w / sum_w
    # compute gradient g = 2 * correlation(p, a)
    # Use FFT: convolve a with reversed p.
    n = a.shape[0]
    # compute size for FFT: need length >= n + (2n-1) - 1 = 3n-2
    size = 1 << ((3 * n - 2).bit_length())
    # compute FFT of a and p_rev
    fa = np.fft.rfft(a, size)
    fp_rev = np.fft.rfft(p[::-1], size)
    conv = np.fft.irfft(fa * fp_rev, size)
    # conv length = size
    # slice conv[n-1:2*n-1]
    slice_start = n - 1
    slice_end = 2 * n - 1
    g_slice = conv[slice_start:slice_end][::-1]  # reverse
    g = 2 * g_slice
    return g

""" 
Softmax-weighted gradient descent with simplex projection to flatten the
autocorrelation of a non‑negative step function.  Random restarts and a
final polishing phase (small mass transfers) are performed within the
time budget.  The algorithm works with a probability vector (sum = 1),
optimising the L∞ norm of its self‑convolution, which directly minimises
the score returned by ``evaluate_sequence``.  The code uses only NumPy
and runs on a single CPU core.
"""

import time
import math
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. (2008) O(n) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgrad_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_{k∈active}(a*a)_k with respect to a."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
    g *= 2.0 / max(1, len(active_idx))
    return g

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    return np.convolve(a, a)

def _gradient_from_weights(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """Gradient of  Σ_k p_k (a*a)_k  w.r.t. a  (p weights sum to 1)."""
    n = a.shape[0]
    # convolution a with reversed p
    conv = np.convolve(a, p[::-1])
    # slice part that corresponds to the correlation
    slice_start = n - 1
    slice_end = 2 * n - 1   # exclusive
    g_slice = conv[slice_start:slice_end][::-1]
    return 2.0 * g_slice

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a low‑score step function using a smoothed max‑gradient
    descent on the self‑convolution, random restarts and a polishing
    phase.  The returned list is scaled to the [0,1000] range required
    by ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose length of the sequence.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        init_seq = np.array(height_sequence_1, dtype=float)
        init_seq = np.clip(init_seq, 0.0, None)
        if init_seq.sum() == 0.0:
            init_seq = np.ones_like(init_seq)
        init_seq = init_seq / init_seq.sum()
    except Exception:
        # a moderate size that allows many iterations
        n = 800
        init_seq = None

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")   # internal objective = 2*n*max_b (sum=1)

    # ------------------------------------------------------------------
    # Optimisation parameters
    # ------------------------------------------------------------------
    max_iter_per_restart = 4000
    step0 = 0.2            # initial step scale
    gamma = 0.9            # momentum coefficient
    beta_start = 5.0
    beta_max = 200.0
    beta_factor = 1.005

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ------------------------------------------------------------------
        # Initialise a candidate distribution
        # ------------------------------------------------------------------
        mode = np.random.choice(["prev", "sqrtdiff", "power", "random", "spike"])
        if mode == "prev" and best_seq is not None:
            a = best_seq.copy()
        elif mode == "prev" and init_seq is not None:
            a = init_seq.copy()
        elif mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        elif mode == "power":
            pexp = np.random.uniform(0.1, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-pexp))
        elif mode == "spike":
            a = np.zeros(n, dtype=float)
            k_spikes = max(2, int(np.sqrt(n)))
            idx = np.random.choice(n, k_spikes, replace=False)
            a[idx] = np.random.rand(k_spikes) + 0.1
        else:   # random
            a = np.random.rand(n)

        # Normalise to a probability vector
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.ones(n)
        a = a / a.sum()

        # ------------------------------------------------------------------
        # Local optimisation (soft‑max gradient descent)
        # ------------------------------------------------------------------
        m = np.zeros_like(a)            # momentum term
        beta = beta_start
        for it in range(max_iter_per_restart):
            if time.time() >= deadline:
                break

            # full convolution and current max
            b = _conv_full(a)
            max_b = b.max()

            # soft‑max weights over the convolution entries
            w = np.exp(beta * (b - max_b))
            sum_w = w.sum()
            if sum_w == 0.0:
                # fall back to the exact sub‑gradient on the active set
                active = np.where(b >= max_b - 1e-12)[0]
                grad = _subgrad_active(a, active)
            else:
                p = w / sum_w               # weighting vector (len = 2n-1)
                grad = _gradient_from_weights(a, p)

            # momentum update
            m = gamma * m + (1.0 - gamma) * grad

            # diminishing step size
            step = step0 / math.sqrt(it + 1)
            a_candidate = a - step * m
            a_candidate = _project_to_simplex(a_candidate)

            # internal objective (exact score, sum(a)=1)
            b_candidate = _conv_full(a_candidate)
            score_candidate = 2.0 * n * b_candidate.max()

            if score_candidate < best_score:
                best_score = score_candidate
                best_seq = a_candidate.copy()
                elapsed = time.time() - start_time
                print(f"[Restart {restart}] iter {it:04d} – new best {best_score:.6f} (t={elapsed:.1f}s)")

            a = a_candidate

            # increase beta slowly – makes the soft‑max concentrate on the peak
            if beta < beta_max:
                beta = min(beta_max, beta * beta_factor)

        # ------------------------------------------------------------------
        # Polishing: tiny random mass transfers with acceptance on true score
        # ------------------------------------------------------------------
        if best_seq is not None:
            a_local = best_seq.copy()
            polish_deadline = time.time() + min(8.0, deadline - time.time())
            while time.time() < polish_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                eps = a_local[i] * np.random.uniform(0.0, 0.05)
                if eps <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps
                a_new[j] += eps
                a_new = _project_to_simplex(a_new)
                # true score via evaluate_sequence (rarely called)
                s = evaluate_sequence(a_new.tolist())
                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        a_fallback = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        a_fallback = a_fallback / a_fallback.sum()
        best_seq = a_fallback

    # ------------------------------------------------------------------
    # Scale to the required [0,1000] range (does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(f"Returning sequence – best internal score {best_score:.6f} after {total_elapsed:.1f}s")
    return final_seq.tolist()

non‑negative step function.

The score returned by ``evaluate_sequence`` is
    2·n·max_k (a*a)_k   (with ∑a_i = 1)

so minimising the L∞‑norm of the self‑convolution directly minimises the

optimises the smoothed objective

where β is increased slowly so that p concentrates on the current peak.
A Euclidean projection onto the simplex enforces the constraints.

within the time budget.  The final sequence is scaled to the [0,1000] range
required by the evaluator (the scaling does not affect the score).

    w = np.maximum(v - theta, 0.0)
    return w


def _subgrad_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximum autocorrelation entry."""
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active_idx:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=int)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
    # the true gradient would have a factor 2, but the scaling is absorbed
    # by the step size – we keep the same convention as the original code.
    if active_idx.size > 0:
        g /= float(active_idx.size)
    return g


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    return np.convolve(a, a)


def _gradient_from_weights(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k p_k (a*a)_k w.r.t. a.
    p has length 2·n‑1 and sums to 1.
    """
    n = a.shape[0]
    # correlation a ⋆ p  =  convolution of a with reversed p
    conv = np.convolve(a, p[::-1])               # length 3·n‑2
    slice_start = n - 1
    slice_end = 2 * n - 1                         # exclusive
    g_slice = conv[slice_start:slice_end][::-1]   # reverse to obtain correct order
    return 2.0 * g_slice


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step function using soft‑max gradient descent,
    random restarts, and a polishing phase.

    The returned list is scaled to the [0,1000] range required by
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose length of the sequence (use previous construction if available)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]
        init_seq = np.array(height_sequence_1, dtype=float)
        init_seq = np.clip(init_seq, 0.0, None)
        if init_seq.sum() == 0.0:
            init_seq = np.ones_like(init_seq)
        init_seq = init_seq / init_seq.sum()
    except Exception:
        n = 800               # a size that permits many iterations
        init_seq = None

    # ------------------------------------------------------------------
    # Global best tracking
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")   # internal objective = 2·n·max_b (∑a_i = 1)

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    max_iter_per_restart = 4000
    step0 = 0.2                # initial step scale
    gamma = 0.9                # momentum coefficient
    beta_start = 5.0           # initial soft‑max inverse temperature
    beta_max = 200.0
    beta_factor = 1.005        # multiplicative increase per iteration

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ------------------------------- initialise -------------------------------
        mode = np.random.choice(["prev", "sqrtdiff", "power", "random", "spike"])
        if mode == "prev" and best_seq is not None:
            a = best_seq.copy()
        elif mode == "prev" and init_seq is not None:
            a = init_seq.copy()
        elif mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        elif mode == "power":
            exp = np.random.uniform(0.1, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-exp))
        elif mode == "spike":
            a = np.zeros(n, dtype=float)
            k_sp = max(2, int(np.sqrt(n)))
            idx = np.random.choice(n, k_sp, replace=False)
            a[idx] = np.random.rand(k_sp) + 0.1
        else:  # random
            a = np.random.rand(n)

        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.ones(n)
        a = a / a.sum()                     # keep sum exactly 1

        # -------------------------- soft‑max gradient descent -------------------
        m = np.zeros_like(a)                # momentum accumulator
        beta = beta_start

        for it in range(max_iter_per_restart):
            if time.time() >= deadline:
                break

            # full self‑convolution and peak
            b = _conv_full(a)
            max_b = b.max()

            # soft‑max weights over all lag positions
            w = np.exp(beta * (b - max_b))
            sum_w = w.sum()
            if sum_w == 0.0:                     # fallback – unlikely
                active = np.where(b >= max_b - 1e-12)[0]
                grad = _subgrad_active(a, active)
            else:
                p = w / sum_w                    # probability vector, length 2n‑1
                grad = _gradient_from_weights(a, p)

            # momentum update
            m = gamma * m + (1.0 - gamma) * grad

            # diminishing step
            step = step0 / math.sqrt(it + 1)
            a_candidate = a - step * m
            a_candidate = _project_to_simplex(a_candidate)

            # internal (exact) objective; sum(a_candidate) = 1
            b_cand = _conv_full(a_candidate)
            score_cand = 2.0 * n * b_cand.max()

            if score_cand < best_score:
                best_score = score_cand
                best_seq = a_candidate.copy()
                elapsed = time.time() - start_time
                print(f"[Restart {restart}] iter {it:04d} – new best {best_score:.6f} "
                      f"(t={elapsed:.1f}s)")

            a = a_candidate

            # slowly raise the inverse temperature (focus on the true peak)
            if beta < beta_max:
                beta = min(beta_max, beta * beta_factor)

        # ----------------------------- polishing stage -------------------------
        if best_seq is not None:
            a_local = best_seq.copy()
            polish_deadline = time.time() + min(8.0, deadline - time.time())
            while time.time() < polish_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                eps = a_local[i] * np.random.uniform(0.0, 0.05)
                if eps <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps
                a_new[j] += eps
                a_new = _project_to_simplex(a_new)

                # true score via the provided evaluator (costly, called rarely)
                s = evaluate_sequence(a_new.tolist())
                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        if time.time() >= deadline:
            break

    # ------------------------------- fallback ---------------------------------
    if best_seq is None:
        a_fallback = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        a_fallback = a_fallback / a_fallback.sum()
        best_seq = a_fallback

    # --------------------------- final scaling --------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(f"Returning sequence – best internal score {best_score:.6f} after "
          f"{total_elapsed:.1f}s")
    return final_seq.tolist()
