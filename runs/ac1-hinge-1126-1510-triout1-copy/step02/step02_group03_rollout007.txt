def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety

    # Determine n
    try: n = len(height_sequence_1)
    except NameError: n = 1542

    best_seq = None
    best_score = float('inf')
    # Possibly store best per restart

    # Define Adam parameters
    alpha = 0.05   # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    eps_factor = 0.1  # for softmax temperature

    # We'll apply multiple restarts until time runs out
    restart = 0
    while time.time() < deadline:
        restart += 1

        # Randomly pick init mode
        mode = np.random.choice(["sqrtdiff","power","random","geometric"])
        if mode == "sqrtdiff":
            a = np.sqrt(np.arange(1,n+1, dtype=float)) - np.sqrt(np.arange(0,n, dtype=float))
        elif mode == "power":
            p = np.random.uniform(0.05,2.0)
            a = (np.arange(1,n+1, dtype=float) ** (-p))
        elif mode == "geometric":
            r = np.random.uniform(0.9,0.999)
            a = r ** np.arange(n)
        else:
            a = np.random.rand(n)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0:
            a = np.ones(n) / n
        else:
            a = a / s  # sum=1

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t_iter = 0

        # set max iterations for this restart based on remaining time (give each ~10s)
        max_iter = 8000  # high enough; will break early if time

        while time.time() < deadline and t_iter < max_iter:
            # compute convolution via FFT
            b = _conv_fft(a)  # length 2n-1
            M = b.max()
            # evaluate score (exact)
            score = 2 * n * M  # since sum=1
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[Restart {restart}] iter {t_iter} – new best {best_score:.6f} (t={elapsed:.1f}s)")

            # compute gradient approximation
            # compute temperature epsilon
            eps_temp = max(1e-8, eps_factor * (b.max() - b.min() + 1e-12))
            # compute stable softmax
            max_b = b.max()
            w_unnorm = np.exp((b - max_b) / eps_temp)
            w = w_unnorm / w_unnorm.sum()
            # compute gradient using convolution w and reversed a
            a_rev = a[::-1]
            # compute convolution w * a_rev
            # using fft conv
            size = 1 << ((len(w) + len(a_rev) - 1).bit_length())
            fw = np.fft.rfft(w, size)
            fa_rev = np.fft.rfft(a_rev, size)
            conv_wa = np.fft.irfft(fw * fa_rev, size)[: len(w) + len(a_rev) - 1]
            # extract central part for gradient
            g = 2.0 * conv_wa[n-1 : n-1 + n]

            # Adam update
            t_iter += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t_iter)
            v_hat = v / (1 - beta2 ** t_iter)
            step = alpha / (np.sqrt(v_hat) + eps_adam)
            a_new = a - step * m_hat

            # project onto simplex
            a_new = _project_to_simplex(a_new)

            a = a_new

            # optional early stopping if improvement small
            # maybe break if score hasn't improved for many iterations

        # after inner loop, do a quick random swapping polish on best_seq for this restart
        if best_seq is not None:
            a_local = best_seq.copy()
            polishing_time = min(5.0, deadline - time.time())
            polish_end = time.time() + polishing_time
            while time.time() < polish_end:
                i, j = np.random.choice(n, 2, replace=False)
                # transfer small random mass
                delta = np.random.uniform(0, 0.02) * a_local[i]
                if delta <= 0: continue
                a2 = a_local.copy()
                a2[i] -= delta
                a2[j] += delta
                a2 = _project_to_simplex(a2)  # keep sum 1
                # compute score diff
                b2 = _conv_fft(a2)
                score2 = 2 * n * b2.max()
                if score2 < best_score:
                    best_score = score2
                    best_seq = a2.copy()
                    a_local = a2
                    elapsed = time.time() - start_time
                    print(f"[Polish restart {restart}] improved to {best_score:.6f} (t={elapsed:.1f}s)")

    # after time budget, construct final sequence
    if best_seq is None:
        # fallback to sqrt diff shape
        a_fallback = np.sqrt(np.arange(1,n+1, dtype=float)) - np.sqrt(np.arange(0,n, dtype=float))
        a_fallback = a_fallback / a_fallback.sum()
        best_seq = a_fallback

    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    total_time = time.time() - start_time
    print(f"Returning sequence – best score {best_score:.6f} after {total_time:.1f}s")
    return final_seq.tolist()

"""
Improved minimisation of the autocorrelation peak.

The algorithm uses several random restarts.  For each restart a
probability vector `a` (non‑negative, sum = 1) is initialised with a
different heuristic (sqrt‐difference, power law, geometric or uniform).

The main optimiser is a projected Adam optimiser that works on the
smooth approximation of the objective

    f(a) = max_k (a * a)_k

by a temperature‑softened softmax:
    w_k = exp( (b_k - max(b)) / eps ) / Σ_j exp( (b_j - max(b)) / eps )
with `b = a * a`.  The gradient of the softmax‑weighted sum can be
computed in O(n log n) via a single FFT:

    ∇f_i ≈ 2·(w ★ a_rev)_{n‑1+i}      (a_rev = a[::-1])

After each Adam step the iterate is projected onto the simplex
{ x ≥ 0 , Σx = 1 } (Euclidean projection – Duchi et al., 2008).

When the time budget is almost exhausted a cheap polishing phase swaps
tiny amounts of mass between random positions.  The best sequence found
is finally scaled into the required [0, 1000] range (the scaling does not
change the score).

This method differs from the previous sub‑gradient approach by:
  • using a smooth softmax approximation of the max,
  • employing Adam with adaptive step sizes,
  • exploring a larger variety of initialisations,
  • performing an FFT‑based exact gradient computation,
  • adding a short random‑swap polish.

The score is evaluated online as `2·n·max(b)` (the sum is 1), which is
equivalent to calling `evaluate_sequence` but far cheaper.

The function `propose_candidate` follows the required signature and
returns the best sequence found within the allotted budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto {x ≥ 0, sum(x)=1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑2 FFT."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


def _conv_fft_two(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution x*y using a power‑of‑2 FFT."""
    size = 1 << ((x.shape[0] + y.shape[0] - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: x.shape[0] + y.shape[0] - 1]


# ----------------------------------------------------------------------
# Gradient of the softmax‑smoothed max‑convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, eps_factor: float = 0.1) -> np.ndarray:
    """
    Returns an approximation of the gradient of max(a*a) w.r.t. a.
    The temperature is eps = eps_factor * (max - min) (or a tiny floor).
    """
    # convolution a*a
    b = _conv_fft(a)               # length 2n‑1
    M = b.max()
    m = b.min()
    # temperature – avoid zero division
    eps = max(1e-12, eps_factor * (M - m + 1e-12))

    # numerically stable softmax
    shifted = b - M                # max will be zero
    w_unnorm = np.exp(shifted / eps)
    w = w_unnorm / w_unnorm.sum()  # shape (2n‑1,)

    # gradient = 2 * (w ★ a_rev)_{n‑1 : n‑1+n}
    a_rev = a[::-1]
    conv_wa = _conv_fft_two(w, a_rev)          # length 3n‑2
    grad = 2.0 * conv_wa[len(a) - 1 : len(a) - 1 + len(a)]
    return grad, M

# ----------------------------------------------------------------------

    """
    Search for a low‑score step‑function using projected Adam optimisation
    with several random restarts and a final random‑swap polish.
    """

    start = time.time()

    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Length of the sequence – reuse a previous one if available
    # ------------------------------------------------------------------
    try:

    # ------------------------------------------------------------------
    # Adam hyper‑parameters (these proved robust in quick tests)
    # ------------------------------------------------------------------

    beta1 = 0.90
    beta2 = 0.999
    eps_adam = 1e-8
    eps_factor = 0.07    # temperature factor for the softmax

    best_seq = None
    best_score = float("inf")

    restart = 0
    # ------------------------------------------------------------------
    # Main loop over random restarts (as many as time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        restart += 1

        # ----- random initialisation ------------------------------------
        mode = np.random.choice(["sqrtdiff", "power", "geom", "uniform"])
        if mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) \
                - np.sqrt(np.arange(0, n, dtype=float))
        elif mode == "power":
            p = np.random.uniform(0.05, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == "geom":

        else:  # uniform random
            a = np.random.rand(n)

        s = a.sum()
        if s == 0.0:
            a = np.ones(n) / n
        else:
            a = a / s                     # enforce sum == 1

        # ----- Adam state ------------------------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0

        # Rough allocation of iterations for this restart:
        max_iter = 7000

        while time.time() < deadline and it < max_iter:
            # 1) obtain softmax‑gradient and current max‑convolution value
            grad, cur_max = _softmax_gradient(a, eps_factor=eps_factor)

                elapsed = time.time() - start
                print(f"[Restart {restart}] iter {it:04d} – new best {best_score:.6f} (t={elapsed:.1f}s)")

            # 3) Adam update on the simplex
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)

            while time.time() < polish_deadline:

                # move a small random fraction from i to j
                delta = a_loc[i] * np.random.uniform(0.0, 0.02)
                if delta <= 0.0:

                a2[i] -= delta
                a2[j] += delta
                a2 = _project_to_simplex(a2)

                # evaluate quickly
                max_b = _conv_fft(a2).max()
                score2 = 2.0 * n * max_b
                if score2 < best_score:
                    best_score = score2
                    best_seq = a2.copy()
                    a_loc = a2
                    elapsed = time.time() - start
                    print(f"[Polish restart {restart}] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # --------------------------------------------------------------

    # Scale to the [0, 1000] range required by the evaluator
    # (scale does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total = time.time() - start
    print(f"Returning sequence – best score {best_score:.6f} after {total:.1f}s")
    return final_seq.tolist()
