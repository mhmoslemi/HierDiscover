"""
Propose a candidate step‑function (list of heights) that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The score for a sequence normalised to Σa = sqrt(2·n) is simply
max(conv(a,a)), so we work on the probability simplex Σa = 1.
A smooth approximation of the max (log‑sum‑exp) is used together with
the Adam optimiser on the unconstrained soft‑max parameters.
Several diverse initialisations are tried; the learning‑rate and the
smoothing parameter β are scheduled aggressively so that the optimiser
explores the landscape early and refines the solution later.
When the time budget allows, a short stochastic local‑search (mass‑swap)
is run on the best sequence found to squeeze a little more improvement.
All steps respect the 1000 s wall‑time limit and the returned list
contains only non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)

def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; returns a probability vector (sum = 1)."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()


def _conv_full(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the first (2*len(a)-1) entries (the valid part).
    """
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv


def _correlation_full(w: np.ndarray, a_rev: np.ndarray, L: int) -> np.ndarray:
    """
    Full correlation (w ⋆ a_rev) = ifft(fft(w) * fft(a_rev)).
    Returns a length‑L array; the useful slice is
    indices [n-1 : n-1+n] which correspond to the gradient w.r.t. a.
    """
    fft_w = np.fft.fft(w, n=L)
    fft_rev = np.fft.fft(a_rev, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev))
    return corr


def _objective_max_conv(a: np.ndarray, L_conv: int) -> float:
    """
    Return max(conv(a,a)) for a that already satisfies Σa = 1.
    """
    conv = _conv_full(a, L_conv)
    return conv.max()


def _local_search(a: np.ndarray,
                  L_conv: int,
                  target_sum: float,
                  time_deadline: float,
                  max_swaps: int = 2000) -> np.ndarray:
    """
    Simple stochastic hill‑climbing: repeatedly move a tiny amount of mass
    from one index to another while the max‑convolution does not increase.
    Runs until the time budget is exhausted or max_swaps attempts are made.
    """
    n = a.shape[0]
    best = a.copy()
    best_score = _objective_max_conv(best, L_conv)

    # scale to the sum‑1 simplex (the optimiser works on the simplex)
    # but the local search works with the same representation.
    while max_swaps > 0 and time.time() < time_deadline:
        # pick two distinct positions
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a small epsilon from i to j
        eps = 1e-5 * best[i]  # relative move; keep non‑negative
        if eps == 0.0:
            continue
        cand = best.copy()
        cand[i] -= eps
        cand[j] += eps
        # keep positivity (the eps is tiny, but guard against rounding)
        if cand[i] < 0:
            continue
        # normalise again (should still sum to 1, but due to rounding)
        cand /= cand.sum()
        cand_score = _objective_max_conv(cand, L_conv)
        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        max_swaps -= 1
    return best

# Main entry point required by the competition framework

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The function respects the 1000 s time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # a safety margin

    # --------------------------------------------------------------
    # Determine the length n of the candidate (use the previous best if present)
    try:
        n = len(height_sequence_1)                 # type: ignore
    except Exception:
        n = 5600                                    # a reasonably large fallback

    target_sum = math.sqrt(2.0 * n)                # normalisation used in the reference
    print(f"--- Optimising a sequence of length n={n} (target sum={target_sum:.3f}) ---")

    # pre‑compute FFT lengths (next powers of two for speed)
    L_conv = 1 << ((2 * n - 1).bit_length())      # length used for a*a
    L_corr = 1 << ((3 * n - 2).bit_length())      # length used for correlation with weights

    # --------------------------------------------------------------
    # Helper that evaluates the *unscaled* objective (Σa = 1)
    def _raw_score(a_vec: np.ndarray) -> float:
        """2·n·max(conv(a,a)) for a on the simplex."""
        conv = _conv_full(a_vec, L_conv)
        return 2.0 * n * conv.max()

    best_seq = None          # will hold the best *scaled* sequence (sum = target_sum)
    best_score = float("inf")

    # --------------------------------------------------------------
    # Build a diverse pool of starting points on the simplex
    start_vectors = []

    # 1) previous best if it exists
    try:
        seq0 = np.asarray(height_sequence_1, dtype=float)
        a0 = seq0 / seq0.sum()
        start_vectors.append(a0)
    except Exception:
        pass

    # 2) uniform
    start_vectors.append(np.full(n, 1.0 / n))

    # 3) random Dirichlet draws (a few)
    for _ in range(3):
        start_vectors.append(np.random.dirichlet(np.ones(n)))

    # 4) power‑law shapes (p < 1 gives heavy tail, p > 1 gives front‑loaded)
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        w = np.arange(1, n + 1, dtype=float) ** p
        start_vectors.append(w / w.sum())

    # 5) exponential decays (different rates)
    for lam in (0.0005, 0.001, 0.002, 0.004):
        w = np.exp(-lam * np.arange(n))
        start_vectors.append(w / w.sum())

    # --------------------------------------------------------------
    # Adam hyper‑parameters (shared across restarts)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    base_lr = 0.07                 # a slightly larger base learning rate
    max_iters_per_restart = 120000    # generous budget; actual count limited by time

    # schedule for the smooth‑max parameter β (log‑sum‑exp)
    beta_start = 2.0
    beta_end   = 400.0

    for restart_id, a_init in enumerate(start_vectors, 1):
        if time.time() > deadline:
            break

        # initialise unconstrained logits v that give a = softmax(v)
        eps = 1e-12
        v = np.log(a_init + eps)

        # Adam moment estimates
        m = np.zeros_like(v)
        v_hat = np.zeros_like(v)

        # bookkeeping for stagnation detection
        last_improve_iter = 0
        last_best_score = best_score

        for it in range(max_iters_per_restart):
            now = time.time()
            if now > deadline:
                break

            # ----- simplex variable a (probability vector) -----
            a = _softmax(v)                       # a ≥ 0, Σa = 1

            # ----- convolution and current raw score -----
            conv = _conv_full(a, L_conv)
            conv_max = conv.max()
            cur_score = 2.0 * n * conv_max          # same as evaluate_sequence when Σa=1

            # ----- record best -----
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a * target_sum             # scale to required L1 norm
                print(f"[restart {restart_id}] iter {it:6d}  new best = {best_score:.6f}")

                # reset stagnation counter
                last_improve_iter = it
                last_best_score = cur_score

            # ----- smooth‑max weights (soft‑max over convolution entries) -----
            # β grows from beta_start to beta_end (exponential schedule for smoothness)
            progress = it / max_iters_per_restart
            beta = beta_start * (beta_end / beta_start) ** progress
            # shift by the max to keep exponentials ≤ 1
            exp_arg = beta * (conv - conv_max)
            w_exp = np.exp(exp_arg)
            w_sum = w_exp.sum()
            if w_sum == 0.0:                     # underflow – fall back to hard max
                w = np.zeros_like(conv)
                w[conv.argmax()] = 1.0
            else:
                w = w_exp / w_sum                 # soft‑max weights over convolution entries

            # ----- gradient w.r.t. a (2·corr(w , a)) -----
            # correlation of w with reversed a
            corr_full = _correlation_full(w, a[::-1], L_corr)
            # slice the part that corresponds to the gradient (length n)
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]

            # ----- gradient w.r.t. logits v (softmax Jacobian) -----
            dot = np.dot(grad_a, a)                # ⟨grad_a , a⟩
            grad_v = a * (grad_a - dot)            # ∂obj/∂v = a ∘ (grad_a – dot)

            # ----- Adam update with a simple learning‑rate decay -----
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat_hat = v_hat / (1.0 - beta2 ** t)

            # optional decay of the learning rate (linear in progress)
            lr = base_lr * (1.0 - progress) + 0.005   # never go below ~0.005

            v -= lr * m_hat / (np.sqrt(v_hat_hat) + eps_adam)

            # ----- early‑restart if we have been stuck for a while -----
            if it - last_improve_iter > 3000 and best_score < float("inf"):
                # perturb v a little to escape a flat region
                perturb = np.random.normal(scale=0.05, size=v.shape)
                v += perturb
                last_improve_iter = it          # reset counter

        # end of one restart
        # ----------------------------------------------------------

    # --------------------------------------------------------------
    # If time permits, perform a brief stochastic local search on the best
    # sequence obtained from the gradient phase.
    if best_seq is not None and time.time() < deadline:
        print("--- Starting short stochastic local search ---")
        # work on the simplex representation (divide by target_sum)
        a_best = best_seq / target_sum
        a_refined = _local_search(a_best,
                                  L_conv,
                                  target_sum,
                                  deadline,
                                  max_swaps=5000)
        refined_score = _raw_score(a_refined)   # raw (2·n·max) score
        if refined_score < best_score:
            best_score = refined_score
            best_seq = a_refined * target_sum
            print(f"Local search improved best to {best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Clip to the range enforced by evaluate_sequence (just to be safe)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished optimisation: best score = {best_score:.6f} (n={n}) ---")
    return best_seq.tolist()
