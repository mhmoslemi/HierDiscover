"""
Propose a low‑score step‑function using a hybrid of
* Differential Evolution on the simplex,
* a short simulated‑annealing polish,
* and a final projected sub‑gradient refinement.

The algorithm works for any length.  If a previous solution
`height_sequence_1` exists it is inserted into the initial population.
All vectors are kept on the probability simplex (non‑negative, sum = 1);
the objective is scale‑invariant, so the final sequence is simply scaled
to the [0, 1000] range required by `evaluate_sequence`.
"""

import time
import math

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: convolution max using FFT (fast even for n > 2000)
# ----------------------------------------------------------------------
def conv_max_fft(a: np.ndarray) -> float:
    """Return max_{k} (a * a)[k] using an FFT‑based convolution."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    # keep only the first (2n‑1) entries of the linear convolution
    return float(conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Objective: the value returned by evaluate_sequence (without the
# security overhead).  The function is scale‑invariant, we keep sum=1.
# ----------------------------------------------------------------------
def objective(a: np.ndarray) -> float:
    """Compute 2*n*max(conv)/sum(a)^2 for a non‑negative vector a."""
    sum_a = a.sum()
    if sum_a < 1e-8:
        return float("inf")
    max_b = conv_max_fft(a)
    return 2.0 * a.shape[0] * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------

# active set).  Used only in the final refinement stage.
# ----------------------------------------------------------------------
def subgrad_active(a: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of the maximal autocorrelation entry."""
    n = a.shape[0]
    g = np.zeros_like(a)
    if active_idx.size == 0:
        return g
    for k in active_idx:
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        i_vals = np.arange(i_lo, i_hi + 1, dtype=int)
        j_vals = k - i_vals
        g[i_vals] += a[j_vals]
    g /= float(active_idx.size)
    return g

            break

    # ------------------------------------------------------------------
    # Phase 2 – Simulated‑annealing polish (≈20 % of the time budget)
    # ------------------------------------------------------------------
    anneal_deadline = start_time + 0.90 * budget_s
    current = best_seq.copy()
    temperature_start = 0.12
    while time.time() < anneal_deadline:
        # linear cooling schedule
        remaining = anneal_deadline - time.time()
        total_anneal = anneal_deadline - (start_time + 0.70 * budget_s)
        temp = temperature_start * (remaining / total_anneal)

        # propose a small mass transfer between two positions
        i, j = np.random.choice(n, 2, replace=False)
        if current[i] == 0.0:
            continue
        # move a random fraction (up to 20 %) of mass from i to j
        eps = current[i] * np.random.uniform(0.0, 0.20)
        if eps <= 0.0:
            continue
        trial = current.copy()
        trial[i] -= eps
        trial[j] += eps
        # keep the sum exactly 1 (the transfer already preserves it)
        s_trial = objective(trial)

        if s_trial < best_score:
            best_score = s_trial
            best_seq = trial.copy()
            current = trial.copy()
            t = time.time() - start_time
            print(f"[Anneal] improved to {best_score:.6f} (t={t:.1f}s)")
        else:
            # Metropolis acceptance
            if temp > 0 and np.random.rand() < math.exp(-(s_trial - best_score) / temp):
                current = trial.copy()

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient refinement (remaining time)
    # ------------------------------------------------------------------
    refine_deadline = deadline
    a = best_seq.copy()
    t = 0
    max_iter = 3000
    step0 = 1.0
    epsilon_rel = 1e-6
    while time.time() < refine_deadline and t < max_iter:
        conv = np.convolve(a, a)
        M = conv.max()
        tol = max(epsilon_rel * M, 1e-12)
        active = np.where(conv >= M - tol)[0]
        g = subgrad_active(a, active)
        step = step0 / math.sqrt(t + 1)
        a_candidate = a - step * g
        a_candidate = project_to_simplex(a_candidate)
        s_candidate = objective(a_candidate)

        if s_candidate < best_score:
            best_score = s_candidate
            best_seq = a_candidate.copy()
            a = a_candidate.copy()
            elapsed = time.time() - start_time
            print(f"[SubGrad] new best {best_score:.6f} (t={elapsed:.1f}s)")
        else:
            a = a_candidate.copy()

        t += 1
        if time.time() >= refine_deadline:
            break

    # ------------------------------------------------------------------
    # Scale to the [0, 1000] range expected by the evaluator.
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(f"Returning sequence – best score {best_score:.6f} after {total_elapsed:.1f}s")
    return final_seq.tolist()
