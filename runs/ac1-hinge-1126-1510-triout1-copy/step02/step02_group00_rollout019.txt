def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    from scipy.optimize import minimize

    np.random.seed(seed)
    deadline = time.time() + budget_s - 1

    # Determine n
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=float)
        n0 = init_seq.shape[0]
        # use at most 4000 baseline, but also try larger n later
        n = max(2000, min(8000, n0*2)) # maybe enlarge
        # if sequence is shorter than n, pad zeros
        if n0 < n:
            a0 = np.zeros(n)
            a0[:n0] = init_seq
            a0 = np.maximum(a0, 0.0)
        else:
            a0 = init_seq[:n]
        # normalize to sum 1
        if a0.sum() == 0:
            a0 = np.full(n, 1.0/n)
        else:
            a0 = a0 / a0.sum()
    else:
        n = 4000
        a0 = np.full(n, 1.0/n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} score={best_score:.6f}")

    # generate diverse initial points
    def generate_initial():
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'sin', 'prev'])
        if mode == 'prev':
            return best_a.copy()
        elif mode == 'random':
            v = rng.random(n)
            return v / v.sum()
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            v = np.exp(-rate * np.arange(n))
            return v / v.sum()
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            v = (np.arange(1, n+1, dtype=float) ** (-p))
            return v / v.sum()
        elif mode == 'edge':
            edge_frac = rng.uniform(0.01, 0.08)
            m_edge = max(1, int(n * edge_frac))
            a_center = 1.0 / n
            a_edge = np.sqrt(n/2.0) / n
            v = np.full(n, a_center)
            v[:m_edge] = a_edge
            v[-m_edge:] = a_edge
            return v / v.sum()
        else:  # sin shape
            x = np.arange(n) + 0.5
            v = np.sin(np.pi * x / n)
            v = np.maximum(v, 0)
            return v / v.sum()
    rng = np.random.default_rng(seed+1)

    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.001]
    iteration = 0

    while time.time() < deadline:
        # pick new starting point
        a_start = generate_initial()
        # convert to logit variable
        y0 = np.log(a_start + 1e-12)

        for lam in lam_schedule:
            if time.time() >= deadline:
                break
            # run L-BFGS
            res = minimize(fun=objective_and_grad,
                           x0=y0,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 200,
                                    'gtol': 1e-12,
                                    'disp': False})
            y0 = res.x   # warm start for next lam
            a_opt = softmax(y0)
            score_opt = fast_score(a_opt, fft_len_conv, n)

            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt.copy()
                print(f"[{iteration}] L-BFGS lam={lam:.3f} score={best_score:.6f}")

        # after lam schedule, perform short gradient refinement
        sg_deadline = min(deadline, time.time() + 3.0)
        best_a, best_score = subgradient_optimize(best_a, sg_deadline, fft_len_conv, fft_len_g, n)

        # hill climb for quick polishing
        hc_deadline = min(deadline, time.time() + 2.0)
        best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000)

        iteration += 1

    # final long hill climb
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n, final_deadline, max_iter=5000)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq

"""
Propose a step‑function (non‑negative vector) that minimises

    E(a) = 2·n·max(a*a) / (sum a)^2

for a list of heights a.  The search uses several complementary
techniques within the given time budget:

1.  Randomly generated starting points (uniform, exponential,
    power‑law, edge‑heavy, sinusoid, previous best).

2.  A continuation of smooth‑max L‑BFGS optimisations.
    The smoothing parameter (lam) is decreased progressively,
    each stage warm‑starting the next.

3.  A projected sub‑gradient descent that works on the true
    non‑smooth objective.

4.  An aggressive greedy hill‑climb that transfers a small
    fraction of mass between two positions while keeping the sum
    unchanged (simple “mass‑transfer” moves).

The routine adapts the number of iterations to the supplied
`budget_s` (default 1000 s) and finally returns the best sequence
found.  All helpers are top‑level functions – no closures or
lambda‑expressions are used.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max: maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    summing to 1.  This is exactly the value returned by
    `evaluate_sequence` when `sum(a) == 1`.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    # `cond` is always True for a non‑negative vector, but we guard anyway
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective based on a log‑sum‑exp approximation and its
    gradient w.r.t. `y` (the log‑odds vector before the soft‑max).

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    # map to simplex
    a = softmax(y)

    # -------- convolution a*a (full) ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # -------- gradient of smooth max ----------
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    # convolution a_rev * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # indices (n‑1) … (n‑1+n‑1) give the required n‑length vector
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient scheme.
    Each step is projected back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0          # a slightly larger initial step size
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active (binding) indices – tolerance avoids numerical gaps
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to the n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed (reproducibility).
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    if 'height_sequence_1' in globals():
        init_seq = np.asarray(globals()['height_sequence_1'], dtype=float)

                a0[:init_seq.size] = init_seq
            else:
                a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            if a0.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            n = 4000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)

    # Pre‑computed FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Helper to generate diverse starting points

    def generate_initial():
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'sin', 'prev'])
        if mode == 'prev':
            return best_a.copy()
        elif mode == 'random':
            v = rng.random(n)
            return v / v.sum()
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            v = np.exp(-rate * np.arange(n))
            return v / v.sum()
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            v = (np.arange(1, n + 1, dtype=float) ** (-p))
            return v / v.sum()
        elif mode == 'edge':
            edge_frac = rng.uniform(0.01, 0.08)
            m_edge = max(1, int(n * edge_frac))
            a_center = 1.0 / n
            a_edge = np.sqrt(n / 2.0) / n
            v = np.full(n, a_center)
            v[:m_edge] = a_edge
            v[-m_edge:] = a_edge
            return v / v.sum()
        else:       # sinusoidal shape (always non‑negative on (0,π))
            x = np.arange(n, dtype=float) + 0.5
            v = np.sin(np.pi * x / n)
            v = np.maximum(v, 0.0)
            return v / v.sum()

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04,
                    0.02, 0.01, 0.005, 0.001]   # decreasing smoothing
    iteration = 0
    deadline = time.time() + budget_s - 1.0   # a tiny safety margin

    while time.time() < deadline:
        # ---- initialise a new starting point ---------------------------------
        a_start = generate_initial()
        y0 = np.log(a_start + 1e-12)   # log‑odds for the soft‑max map

        # ---- continuation of L‑BFGS with decreasing smoothing ---------------
        for lam in lam_schedule:
            if time.time() >= deadline:

            res = minimize(fun=objective_and_grad,

        # ---- occasional random perturbation to escape local minima ----------
        if iteration % 5 == 0 and time.time() < deadline - 2.0:

        print("[final] evaluation unavailable in the local environment")
    return final_seq
