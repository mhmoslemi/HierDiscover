def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5 # margin

    # Determine n:
    n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim==1 and init_seq.size>0:
            if init_seq.size > n:
                init_seq = init_seq[:n]
            # normalize to sum=1
            if init_seq.sum()==0:
                a0 = np.full(init_seq.shape[0], 1.0/init_seq.shape[0])
            else:
                a0 = init_seq / init_seq.sum()
            n = a0.shape[0]
        else:
            a0 = np.full(n, 1.0/n)
    else:
        a0 = np.full(n, 1.0/n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} score={best_score}")

    # L-BFGS phase
    # We'll do a number of restarts based on time left.
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04]  # smoothing
    iteration = 0
    rng = np.random.default_rng(seed)

    while time.time() < deadline - 30:  # reserve at least 30 sec for annealing+final hill climb
        lam = lam_schedule[iteration % len(lam_schedule)]

        # Choose mode
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n+1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:  # edge-heavy
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n/2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # Run L-BFGS
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'gtol': 1e-12,
                                'disp': False})
        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L-BFGS {iteration}] mode={mode} lam={lam:.3f} score={best_score:.6f}")

        # Short hill climb to polish
        # Using hill_climb (fast greedy) for limited iterations
        hc_deadline = min(deadline - 10, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score, fft_len, n,
                                        hc_deadline, max_iter=800)
        iteration += 1

    # Annealing phase
    # parameters
    T0 = 0.01  # initial temperature for relative score changes (score difference)
    T_final = 1e-6
    total_anneal = deadline - time.time()
    if total_anneal > 0:
        cur_a = best_a.copy()
        cur_score = best_score
        start_time = time.time()
        while time.time() < deadline:
            elapsed = time.time() - start_time
            # linear schedule of T
            T = T0 * (1 - elapsed / total_anneal) + T_final * (elapsed / total_anneal)
            # propose move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # move some fraction up to 10% of current mass at i
            delta = cur_a[i] * rng.random() * 0.1
            if delta <= 0:
                continue
            new_a = cur_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            # feasiblity: ensure sum constant (it is)
            new_score = fast_score(new_a, fft_len, n)
            # accept
            if new_score < cur_score:
                cur_a, cur_score = new_a, new_score
                if new_score < best_score:
                    best_score, best_a = new_score, new_a.copy()
            else:
                # metropolis acceptance
                prob = np.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    cur_a, cur_score = new_a, new_score

    # Final greedy hill climb with more iterations
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len, n,
                                    final_deadline,
                                    max_iter=4000)

    print(f"[final] best_score={best_score}")
    return best_a.tolist()

    rows, cols precomputed static.
    while time.time() < deadline and iteration < max_iterations:
        # Solve LP
        data = np.tile(f, n)
        A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
        b_ub = np.full(2*n-1, M)

"""
Propose a step‑function (a non‑negative list of heights) with a low
evaluation score

The score is  E(a)=2·n·max(a*a) / (sum a)^2.
Because the objective is scale‑invariant we keep the vector on the
probability simplex (sum = 1).  The algorithm consists of three stages:

1.  Stochastic L‑BFGS on a smooth (log‑sum‑exp) approximation of the
    max.  Many random restarts with different initial shapes are tried.
    After each run a short greedy hill‑climb (mass transfer between two
    coordinates) is performed.

2.  A linear‑programming refinement.  For the current candidate f we
    compute M = max(f*f).  Solving the LP
          maximise Σ g_i
          subject to  f * g ≤ M ,   g ≥ 0
    yields a vector g with a larger ℓ₁‑norm while keeping the convolution
    below the current bound.  After normalising g to sum 1 we mix it
    with f (f←(1‑t)f+t g).  Repeating this step strictly reduces M.

3.  Simulated‑annealing / greedy polishing.  Small random mass transfers
    are accepted if they improve the score or with a Metropolis
    probability that slowly cools down.  Finally a longer greedy
    hill‑climb is executed.

The whole routine respects the 1000 s time budget, prints intermediate
scores and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score for a vector already on the simplex (sum = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float                – 2·n·smooth_max(conv)
    grad_y : np.ndarray      – gradient w.r.t. y (shape n)
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy hill‑climb that moves a small fraction of mass from one
    coordinate to another while preserving the sum.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 5 % of the current mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “alternating LP” step)
# ----------------------------------------------------------------------


def build_conv_constraint_matrix(f: np.ndarray) -> diags:
    """
    Return a sparse matrix C (shape (2n‑1, n)) such that C @ g = f * g
    (linear convolution).  The matrix is built from the diagonals of f.
    """
    n = f.shape[0]
    # each diagonal with offset –d contains the constant f[d] repeated n times
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = [-d for d in range(n)]
    C = diags(diagonals, offsets, shape=(2 * n - 1, n), format='csc')
    return C


def lp_refine(f: np.ndarray, M: float, n: int,
              fft_len: int, deadline: float,
              max_steps: int = 6, mix_factor: float = 0.2):
    """
    Perform a few alternating‑LP updates.  Each step solves

        maximise Σ g_i
        s.t.    f * g ≤ M ,   g ≥ 0

    rescales the solution to sum 1 and mixes it into f.
    Returns the improved f and its new max‑convolution value.
    """
    cur_f = f.copy()
    cur_M = M
    step = 0

    while time.time() < deadline and step < max_steps:
        # build the (f * g) ≤ M constraints
        C = build_conv_constraint_matrix(cur_f)
        b_ub = np.full(2 * n - 1, cur_M, dtype=np.float64)

        # maximise Σ g_i  → minimise –Σ g_i
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=C, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if not res.success:
            # LP failed – abort further refinement
            break

        g0 = res.x
        s0 = g0.sum()
        if s0 <= 0:
            break

        # normalise to the simplex
        g = g0 / s0

        # convex combination
        new_f = (1.0 - mix_factor) * cur_f + mix_factor * g
        conv = full_conv(new_f, fft_len)
        new_M = conv.max()

        if new_M < cur_M - 1e-12:
            cur_f, cur_M = new_f, new_M
            print(f"[LP] step {step} improved M to {cur_M:.6f}")
        else:
            # no improvement – stop the LP loop
            break
        step += 1

    return cur_f, cur_M

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function will return before the
        budget expires).

    Returns
    -------
    list[float]
        The best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # --------------------------------------------------------------
    # Choose problem size and initial distribution
    # --------------------------------------------------------------
    default_n = 4000
    if 'height_sequence_1' in globals():
        init = np.array(globals()['height_sequence_1'], dtype=float)
        if init.ndim == 1 and init.size > 0:
            n = min(default_n, init.size)
            init = init[:n]
            init[init < 0] = 0.0
            if init.sum() == 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init / init.sum()
        else:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS on a smoothed max
    # --------------------------------------------------------------
    rng = np.random.default_rng(seed)
    lam_sched = [0.5, 0.3, 0.15, 0.08, 0.04]
    it = 0

    # reserve ~200 s for the LP refinement and the remaining time for
    # annealing / final hill‑climb
    lp_reserved = 200.0
    while time.time() < deadline - lp_reserved:
        lam = lam_sched[it % len(lam_sched)]

        # pick a starting point
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy heuristic
            ef = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * ef))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # L‑BFGS optimisation
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'gtol': 1e-12,
                                'disp': False})
        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS {it}] mode={mode} lam={lam:.3f} score={best_score:.6f}")

        # short hill‑climb to polish each L‑BFGS result
        hc_dead = min(deadline - lp_reserved, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, n,
                                        hc_dead, max_iter=800)
        it += 1

    # --------------------------------------------------------------
    # Phase 2 – linear‑programming refinement
    # --------------------------------------------------------------
    cur_M = full_conv(best_a, fft_len).max()
    lp_deadline = min(deadline, time.time() + lp_reserved)
    best_a, cur_M = lp_refine(best_a, cur_M, n, fft_len,
                             lp_deadline, max_steps=6, mix_factor=0.2)
    best_score = fast_score(best_a, fft_len, n)

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing + final hill‑climb
    # --------------------------------------------------------------
    # annealing parameters
    T_start = 0.02
    T_end = 1e-6
    start_time = time.time()
    cur_a = best_a.copy()
    cur_score = best_score

    while time.time() < deadline:
        elapsed = time.time() - start_time
        # linear cooling schedule
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + \
            T_end * (elapsed / (deadline - start_time))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the mass at i
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = fast_score(new_a, fft_len, n)

            if new_score < best_score:
                best_a, best_score = new_a.copy(), new_score
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                cur_a, cur_score = new_a, new_score

    # final greedy polishing
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len, n,
                                    final_deadline,
                                    max_iter=4000)

    print(f"[final] best_score={best_score:.6f}")
    return best_a.tolist()
