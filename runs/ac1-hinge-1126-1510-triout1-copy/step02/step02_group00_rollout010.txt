def lp_descent_optimize(a0, p, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    step0 = 0.5  # initial step size
    it = 0
    while time.time() < deadline:
        # conv
        conv = full_conv(a, fft_len_conv) # length 2n-1
        # compute gradient of sum(conv**p)
        w = conv ** (p - 1)
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_grad = np.real(np.fft.ifft(fw * fa_rev))
        g = 2.0 * p * conv_grad[(n - 1):(n - 1 + n)]
        # step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        # project onto simplex
        a = simplex_projection(a_new)
        # evaluate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def simulated_annealing(a, current_score, fft_len, n, deadline, max_iter=100000):
    best = a.copy()
    best_score = current_score
    cur = a.copy()
    cur_score = current_score
    rng = np.random.default_rng()
    T0 = cur_score * 0.01  # initial temp about 1% of score?
    iter = 0
    while time.time() < deadline and iter < max_iter:
        # linear cooling
        T = T0 * max(0.0, 1.0 - iter / max_iter)
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.1  # up to 10% of source mass
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        # evaluate
        new_score = fast_score(cur, fft_len, n)
        # accept if better or with prob
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
            cur_score = new_score
            # keep move
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
        iter += 1
    return best, best_score
```

"""
Search for a low‑evaluation non‑negative step sequence.

The algorithm combines several complementary optimisation phases
within the given time budget (default 1000 s):

1.  Stochastic L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution, with a continuation schedule for the smoothing
    factor.
2.  A short projected sub‑gradient refinement (the true max is used).
3.  A fast greedy hill‑climb that moves a small fraction of mass
    between two positions.
4.  A simulated‑annealing phase that occasionally accepts worsening
    moves to escape local minima.
5.  A final polishing hill‑climb.

All iterates live on the probability simplex (sum = 1), enforced via
the soft‑max parametrisation for L‑BFGS and an Euclidean simplex
projection for the sub‑gradient phase.

The best sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max: maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    summing to 1 (i.e. exactly the quantity returned by
    ``evaluate_sequence`` after normalisation).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float
        2·n·smooth_max(conv)
    grad : np.ndarray
        gradient w.r.t. y (shape n)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                      # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Direct minimisation of 2·n·max(a*a) using a classic sub‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep best
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """Greedy hill‑climb moving a small fraction of mass between two slots."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing refinement
# ----------------------------------------------------------------------
def simulated_annealing(a: np.ndarray,
                        current_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 20000):
    """
    Annealing that occasionally accepts worsening moves.
    Moves transfer up to 10 % of the source mass.
    """
    cur = a.copy()
    cur_score = current_score
    best = cur.copy()
    best_score = current_score
    rng = np.random.default_rng()
    # initial temperature ≈ 2 % of the current score
    T0 = max(cur_score * 0.02, 1e-6)

    it = 0
    while time.time() < deadline and it < max_iter:
        # linear cooling
        T = T0 * max(0.0, 1.0 - it / max_iter)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # accept with probability exp((old-new)/T)
            if rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best = cur.copy()
            else:
                # revert
                cur[i] += delta
                cur[j] -= delta

        it += 1
    return best, best_score
