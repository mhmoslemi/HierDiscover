"""
Propose a step sequence that minimises

    E(a) = 2·n·max(a*a) / (sum a)^2

The algorithm consists of three stages:

1. **Random parametric sampling** – generate a large number of candidate
   probability vectors using simple one‑parameter families (beta
   distributions, U‑shaped power laws, exponential edges, cosine powers,
   and mixtures).  The vectors are normalised to sum = 1 and evaluated
   with a fast FFT‑based convolution that computes 2·n·max(a*a).

2. **Refinement by a projected sub‑gradient method** – starting from the
   best candidate, repeatedly move in the direction of a sub‑gradient of
   the true (non‑smooth) objective and project back onto the simplex.

3. **Greedy hill‑climb** – finally perform small random mass‑transfers
   between two positions while keeping the sum fixed; only moves that
   improve the fast score are accepted.

All steps respect the 1000 s budget, use only NumPy/SciPy, and avoid
closures or lambdas (required by the competition framework).
"""

import time
import numpy as np
from scipy.special import betaln
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities (fast O(n log n) convolution)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summed to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
# ----------------------------------------------------------------------
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (all return a normalised probability vector)
# ----------------------------------------------------------------------
def gen_power_family(n: int, p: float) -> np.ndarray:
    """U‑shaped / bell‑shaped family a_i ∝ (x·(1‑x))**p."""
    x = (np.arange(n) + 0.5) / n
    raw = (x * (1.0 - x)) ** p
    raw = np.maximum(raw, 1e-15)        # avoid exact zero for p>0
    a = raw / raw.sum()
    return a


def gen_beta_family(n: int, a: float, b: float) -> np.ndarray:
    """Beta distribution PDF evaluated at mid‑points."""
    x = (np.arange(n) + 0.5) / n
    # log‑pdf to stay stable for extreme parameters
    log_pdf = (a - 1.0) * np.log(x) + (b - 1.0) * np.log(1.0 - x) - betaln(a, b)
    pdf = np.exp(log_pdf)
    pdf = np.maximum(pdf, 1e-15)
    a_vec = pdf / pdf.sum()
    return a_vec


def gen_exp_edges(n: int, lam: float) -> np.ndarray:
    """Mass decays exponentially from both ends."""
    i = np.arange(n, dtype=np.float64)
    raw = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    a = raw / raw.sum()
    return a


def gen_cos_power(n: int, p: float) -> np.ndarray:
    """Cosine‑shaped family a_i ∝ |cos(π·(i+0.5)/n)|**p."""
    theta = np.pi * (np.arange(n) + 0.5) / n
    raw = np.abs(np.cos(theta)) ** p
    raw = np.maximum(raw, 1e-15)
    a = raw / raw.sum()
    return a


def gen_mixed(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of two simple families."""
    w = rng.random()
    a1 = gen_power_family(n, rng.uniform(-0.45, 0.45))
    a2 = gen_exp_edges(n, rng.uniform(0.001, 0.02))
    a = w * a1 + (1.0 - w) * a2
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser for the true max
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """Minimise 2·n·max(a*a) using a simple projected sub‑gradient scheme."""
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 0.5
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep best observed
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer between two positions)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """Greedy local search: transfer a small fraction of mass between two indices."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05   # up to 5 % of mass at i
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score

        else:
            n = 4000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – random parametric exploration
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)
    families = ['power', 'beta_sym', 'beta_asym', 'exp_edge', 'cos_pow', 'mixed']
    it = 0
    # we reserve ~30 % of the budget for local refinement
    param_deadline = deadline - 30.0
    while time.time() < param_deadline:
        it += 1
        choice = rng.choice(families)

        if choice == 'power':
            p = rng.uniform(-0.48, 0.48)
            a_cand = gen_power_family(n, p)

        elif choice == 'beta_sym':
            a_param = rng.uniform(0.1, 10.0)
            a_cand = gen_beta_family(n, a_param, a_param)

        elif choice == 'beta_asym':
            a_param = rng.uniform(0.1, 10.0)
            b_param = rng.uniform(0.1, 10.0)
            a_cand = gen_beta_family(n, a_param, b_param)

        elif choice == 'exp_edge':
            lam = rng.uniform(0.001, 0.02)
            a_cand = gen_exp_edges(n, lam)

        elif choice == 'cos_pow':
            p = rng.uniform(0.1, 6.0)
            a_cand = gen_cos_power(n, p)

        else:  # mixed
            a_cand = gen_mixed(n, rng)

        # evaluate quickly
        s = fast_score(a_cand, fft_len_conv, n)
        if s < best_score:
            best_score = s
            best_a = a_cand
            print(f"[{it}] new best via {choice} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – sub‑gradient refinement (≈20 s)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 10.0, time.time() + 20.0)
    if subgrad_deadline > time.time():
        best_a, best_score = subgradient_optimize(best_a,
                                                  subgrad_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑grad] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final greedy hill‑climb (use the rest of the time)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    final_deadline,
                                    max_iter=4000)
    print(f"[final] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence; ask the true evaluator for the final exact score
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
