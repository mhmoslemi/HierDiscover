def gradient_smoothed(a, alpha):
    n = len(a)
    L = next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n - 1]
    max_val = conv.max()
    exp_args = alpha * (conv - max_val)  # max arg zero
    # To avoid overflow/underflow, use np.exp with clip
    w = np.exp(exp_args)  # w_max = 1
    w_sum = w.sum()
    w /= w_sum
    fw = np.fft.rfft(w, n=L)
    # correlation via ifft( fa * conj(fw) )
    corr = np.fft.irfft(fa * np.conj(fw), n=L)[:n]
    grad = 2.0 * np.real(corr)
    return grad, conv

def optimize_sequence(a_initial, time_limit):
    n = len(a_initial)
    # initialize logits z = np.log(a_initial + 1e-12)
    z = np.log(a_initial + 1e-12)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    total_iters = 5000
    start = time.time()
    best_local_seq = a_initial.copy()
    best_local_score = fast_score(best_local_seq)
    for t in range(1, total_iters+1):
        # time check
        if time.time() > time_limit:
            break
        # compute a
        a = np.exp(z - np.max(z))
        a = a / a.sum()
        # compute gradient
        alpha = 50000 + (250000) * (t / total_iters)  # schedule
        grad_a, conv = gradient_smoothed(a, alpha)
        # transform to z-gradient
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        # periodically evaluate
        if t % 200 == 0:
            cur_score = fast_score(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local_seq = a.copy()
                print(f"[{time.time() - start:.1f}s] local best iter {t} score {cur_score:.6f}")
    # optionally subgradient refinement
    a = best_local_seq.copy()

"""
Propose a low‑score step‑function using a hybrid of

1.  Diverse random initialization (power‑law, exponential, two‑sided
    power‑law) with lengths up to 12 000.
2.  Smoothed‑gradient descent on the simplex via an Adam optimiser.
    The smoothed objective is  (1/α)·log∑exp(α·(a∗a)),
    whose gradient can be written as a cheap cross‑correlation.
    A large α concentrates the weight on the current peak, reproducing a
    sub‑gradient of the original L∞‑objective while giving a well‑behaved
    gradient for Adam.
3.  A final Euclidean projected sub‑gradient refinement (as in the
    reference implementation) to polish the result.
4.  A short random‑perturbation sweep for possible last‑minute gains.

All sequences are kept on the probability simplex (non‑negative,
sum = 1) so that the scale‑invariant objective
    2·n·max(conv(a,a)) / (∑a)²
reduces to 2·n·max(conv(a,a)).  The routine obeys the 1000‑second budget
and returns the best list of heights found.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quick evaluation of the competition objective using FFT.
    The sequence is assumed to be non‑negative and already normalised
    (sum == 1).  The returned value is exactly the quantity that
    ``evaluate_sequence`` computes (up to the rounding performed there).
    """
    n = seq.shape[0]
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for random initialisation
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (i = 0 … n‑1) – not normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    return seq


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i) – not normalised."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    return seq


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n‑i)^{-p_right}
    – not normalised.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


# ----------------------------------------------------------------------
# Smoothed gradient of the L∞‑objective
# ----------------------------------------------------------------------


def gradient_smoothed(a: np.ndarray, alpha: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns (grad, conv) where

    * conv is the full self‑convolution a∗a (length 2·n‑1);
    * grad = ∂/∂a ( (1/α)·log∑exp(α·conv) ) .

    The gradient can be expressed as a cross‑correlation and is
    obtained via FFTs.
    """
    n = a.shape[0]
    L = next_pow2(3 * n - 2)                      # length needed for linear correlation
    fa = np.fft.rfft(a, n=L)                     # FFT of a
    conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]

    # ------- smoothed max via log‑sum‑exp ---------------------------------
    max_val = conv_full.max()
    # shift for numerical stability (max becomes 0)
    exp_args = alpha * (conv_full - max_val)
    # exp is safe because the largest argument is 0
    w = np.exp(exp_args)
    w_sum = w.sum()
    w /= w_sum                                    # weights that sum to 1

    # ------- gradient = 2·(a ⋆ w) -----------------------------------------
    fw = np.fft.rfft(w, n=L)
    # cross‑correlation via ifft(FA * conj(FW))
    corr = np.fft.irfft(fa * np.conj(fw), n=L)[:n]
    grad = 2.0 * np.real(corr)
    return grad, conv_full


# ----------------------------------------------------------------------
# Optimisation core (Adam + sub‑gradient polishing)
# ----------------------------------------------------------------------


def optimise_sequence(
    a0: np.ndarray, time_deadline: float, max_adam_iters: int = 6000,
    max_subgrad_iters: int = 4000
) -> tuple[np.ndarray, float]:
    """
    Refines a normalised initial sequence ``a0``.
    * First part: Adam optimiser on the smoothed objective.
    * Second part: projected sub‑gradient descent on the true L∞‑objective.
    Returns the best sequence encountered and its fast_score.
    """
    n = a0.shape[0]
    # initialise logits for a soft‑max parametrisation
    eps = 1e-12
    z = np.log(a0 + eps)                 # logits (real numbers)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.1
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    best_seq = a0.copy()
    best_sc = fast_score(best_seq)

    start = time.time()
    t = 0
    # ----------- ADAM PHASE ----------------------------------------------
    while time.time() < time_deadline - 5.0 and t < max_adam_iters:
        t += 1
        # compute current a from logits (soft‑max)
        a = np.exp(z - np.max(z))
        a /= a.sum()

        # schedule for α (smoothness).  Start moderate, end large.
        alpha_start = 5e4
        alpha_end = 2.5e5
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_adam_iters)

        grad_a, _ = gradient_smoothed(a, alpha)

        # translate gradient w.r.t. a to gradient w.r.t. logits z
        # ∂J/∂z = a ⊙ (∂J/∂a – (a·∂J/∂a)·1)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional evaluation
        if t % 200 == 0 or t == max_adam_iters:
            cur_sc = fast_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_seq = a.copy()
                print(f"[{time.time() - start:.1f}s] Adam iter {t} --> score {cur_sc:.6f}")

    # ----------- SUB‑GRADIENT POLISHING -----------------------------------
    a = best_seq.copy()
    it = 0
    while time.time() < time_deadline - 2.0 and it < max_subgrad_iters:
        it += 1
        L = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        max_idx = int(np.argmax(conv_full))

        # sub‑gradient of the active peak
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        if i_start <= i_end:
            i_range = np.arange(i_start, i_end + 1)
            j_range = max_idx - i_range
            g[i_range] = 2.0 * a[j_range]

        # diminishing step size (classical sub‑gradient schedule)
        eta = 0.6 / np.sqrt(it)
        a_candidate = a - eta * g
        a = project_simplex(a_candidate)

        if it % 250 == 0 or it == max_subgrad_iters:
            cur_sc = fast_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_seq = a.copy()
                print(f"[{time.time() - start:.1f}s] Sub‑grad iter {it} --> score {cur_sc:.6f}")

    return best_seq, best_sc
