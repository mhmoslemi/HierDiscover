```
def smooth_max_grad(a, alpha):
    n = len(a)
    conv_a = fast_convolve(a, a)  # length 2n-1
    # compute softmax weights
    maxv = np.max(alpha * conv_a)
    # if alpha large, may overflow; compute stable
    exp_vals = np.exp(alpha * conv_a - maxv)
    p = exp_vals / np.sum(exp_vals)
    # conv of p and a
    conv_pa = fast_convolve(p, a)  # length 3n-2
    # compute gradient
    offset = n-1
    g = 2.0 * conv_pa[offset:offset+n]
    # subtract diag term
    # diag indices: 2*i for i from 0..n-1, ensure within p length
    diag_mask = np.arange(n) * 2
    # we need p at those indices
    diag_vals = np.zeros_like(a)
    valid = diag_mask < len(p)
    diag_vals[valid] = p[diag_mask[valid]]
    g -= a * diag_vals
    return g, conv_a, p
```

def refine_sequence(a0, time_budget):
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = fast_score(a)
    start = time.time()
    # adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    t = 0
    alpha = 1.0
    alpha_max = 60.0
    alpha_factor = 1.05
    eval_every = 200
    while time.time() - start < time_budget:
        t += 1
        # compute convolution a*a
        conv_a = fft_convolve(a, a)  # length 2n-1
        # compute softmax weights
        # np.exp(alpha*conv) stable
        scaled = alpha * conv_a
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        p = exp_scaled / np.sum(exp_scaled)
        # compute convolution p * a
        conv_pa = fft_convolve(p, a)   # length 3n-2
        offset = n - 1
        g = 2.0 * conv_pa[offset:offset+n]
        # diag term
        diag_idx = np.arange(n) * 2
        g -= a * p[diag_idx]
        # adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a = project_simplex(a_candidate)
        # gradually increase alpha
        if alpha < alpha_max and t % 500 == 0:
            alpha = min(alpha * alpha_factor, alpha_max)
        if t % eval_every == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                # print improvement
                print(f"[refine] t={t}, alpha={alpha:.2f}, score={cur_score:.6f}")
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm has two stages:

1️⃣  Randomised family search – many candidate shapes (power‑law,
    exponential, two‑sided power‑law, discrete Beta, sine‑power) with
    a variety of lengths are generated, normalised to sum 1 and scored
    with a fast FFT‑based evaluation.  The ten best candidates are kept.

2️⃣  Smooth‑max Adam refinement – each of the best candidates is
    improved by minimising a smooth approximation of the
    ``max(conv)`` term:

        L(a) = (1/α)·log ∑ₖ exp(α·cₖ)   with  c = a∗a

    The gradient of L is computed analytically via two FFT‑based
    convolutions, an Adam step is taken, and the iterate is projected
    onto the probability simplex.  The smoothing parameter α is
    increased gradually (continuation) so the optimiser first sees a
    global landscape and later sharpens to the true max.  The best
    refined sequence is finally evaluated with the provided
    ``evaluate_sequence`` function.

The whole routine respects the 1000 s budget, prints progress and
returns the best sequence (as a plain list of floats)."""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real 1‑D arrays using FFT.
    Returns the full convolution (len(x)+len(y)-1).
    """
    n = len(x) + len(y) - 1
    f_len = next_pow2(n)
    X = np.fft.rfft(x, f_len)
    Y = np.fft.rfft(y, f_len)
    conv = np.fft.irfft(X * Y, f_len)[:n]
    return conv


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective (the expensive part is the convolution).
    The sequence is assumed to be non‑negative and sums to 1 (scale‑invariant).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = fft_convolve(seq, seq)          # length 2n‑1
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ---------- families for the random phase ---------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑like shape: (i+1)^{α-1}·(n-i)^{β-1}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return seq


def generate_sine_power(n: int, gamma: float) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)^γ – zero at the boundaries, smooth interior."""
    i = np.arange(n, dtype=np.float64)
    seq = np.sin(np.pi * (i + 0.5) / n) ** gamma
    return seq


# ----------------------------------------------------------------------
# Smooth‑max Adam refinement
# ----------------------------------------------------------------------


def refine_sequence_adam(
    a0: np.ndarray, time_budget: float, min_alpha: float = 1.0, max_alpha: float = 60.0
) -> tuple[np.ndarray, float]:
    """
    Refine a probability vector `a0` by minimising a smooth approximation of
    ``max(conv(a,a))`` using Adam and projection onto the simplex.
    The function runs for at most ``time_budget`` seconds and returns the
    best vector found together with its fast_score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_sc = fast_score(a)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.12

    start = time.time()
    t = 0
    alpha = min_alpha
    # schedule: multiply α every `alpha_step` iterations
    alpha_step = 500
    alpha_factor = 1.05
    eval_every = 200

    while time.time() - start < time_budget:
        t += 1

        # ---- compute smooth max -------------------------------------------------
        conv_a = fft_convolve(a, a)                 # length 2n‑1
        scaled = alpha * conv_a
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)    # stability
        p = exp_scaled / np.sum(exp_scaled)         # softmax weights, sum=1

        # ---- gradient of the smoothed objective ---------------------------------
        conv_pa = fft_convolve(p, a)                # length 3n‑2
        offset = n - 1
        g = 2.0 * conv_pa[offset : offset + n]      # main term
        diag_idx = np.arange(n) * 2
        g -= a * p[diag_idx]                        # subtract diagonal contribution

        # ---- Adam update --------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a = project_simplex(a_candidate)

        # ---- continuation for α -------------------------------------------------
        if alpha < max_alpha and t % alpha_step == 0:
            alpha = min(alpha * alpha_factor, max_alpha)

        # ---- occasional true evaluation -----------------------------------------
        if t % eval_every == 0:
            cur_sc = fast_score(a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                print(
                    f"[refine] it={t:,} α={alpha:.2f} score={cur_sc:.6f}",
                    flush=True,
                )

    return best_a, best_sc

    # 2️⃣  Refine the best candidates with Adam + smooth max
    # -------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left <= 0.0:
        # no time left – just return the current best
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Finished (no refinement). Score={final_score:.6f}", flush=True)
        return final_seq

    # allocate roughly equal time to each candidate (but guard the total)
    per_candidate = max(0.0, time_left / len(top_candidates))

    global_best_seq = best_seq.copy()
    global_best_score = best_score

    for idx, (init_sc, init_seq) in enumerate(top_candidates, start=1):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        budget = min(per_candidate, remaining)
        if budget < 1.0:          # less than a second – skip further refinements
            continue
        print(
            f"[refine] start candidate {idx}/{len(top_candidates)} (init score={init_sc:.6f})",
            flush=True,
        )
        refined_seq, refined_sc = refine_sequence_adam(init_seq, budget)
        if refined_sc < global_best_score:
            global_best_score = refined_sc
            global_best_seq = refined_seq.copy()
            print(
                f"[global] improved after refinement: score={refined_sc:.6f}",
                flush=True,
            )

    # -------------------------------------------------------------------
    # 3️⃣  Final verification with the provided evaluate_sequence
    # -------------------------------------------------------------------
    final_seq = global_best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(
        f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}",
        flush=True,
    )
    return final_seq
