def gradient_smooth_max(a, p):
    # a is length n
    # p length 2n-1
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(a_rev) + len(p) - 1).bit_length())
    fa = np.fft.rfft(a_rev, L)
    fp = np.fft.rfft(p, L)
    conv = np.fft.irfft(fa * fp, L)
    # conv length = L; we only need slice n-1 : n-1+n
    g = conv[n-1 : n-1 + n]
    return 2.0 * g

"""
Propose a low‑score step‑function using a smooth‑max (log‑sum‑exp) objective,
projected Adam optimisation on the probability simplex, and a short
probabilistic polishing phase.  The method works entirely in the space of
probability vectors (∑aᵢ=1, aᵢ≥0) because the evaluation score is scale‑invariant.
A continuation on the smoothness parameter β gradually turns the smooth‑max
into the true max, helping the optimiser escape shallow local minima.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Projection onto the simplex {x >= 0, sum(x) = 1}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho_max = rho[-1]
        theta = cssv[rho_max] / (rho_max + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full linear convolution via FFT (real → real)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Convolution of a with itself using FFT (full length 2·len(a)-1)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Convolution of two arbitrary real vectors via FFT
# ----------------------------------------------------------------------
def _conv_fft_general(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of x and y using FFT."""
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the self‑convolution
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute grad_a 2·Σ_k p_k a_{k-i} where
    p_k = exp(beta * (a*a)[k]) / Σ_j exp(beta * (a*a)[j]).
    Returns a vector of length n (same as a).
    """
    n = a.shape[0]

    # 1) self‑convolution
    conv = _conv_fft(a)                     # length 2n‑1

    # 2) soft‑max over the convolution entries
    scaled = beta * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)   # stability
    p = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

    # 3) cross‑correlation a ↔ p  (g_i = 2 Σ_j a_j p_{i+j})
    a_rev = a[::-1]                           # reversed a
    full = _conv_fft_general(a_rev, p)        # length 3n‑2
    # slice that corresponds to i = 0 … n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad

# Main entry point – optimisation driver
# ----------------------------------------------------------------------

    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Uses projected Adam on a smooth‑max surrogate of the maximal
    convolution entry, several random restarts, and a brief polishing
    stage based on random mass transfers.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for the caller

    # --------------------------------------------------------------
    # Length of the sequence – reuse previous best if available
    # --------------------------------------------------------------
    try:

    except Exception:
        n = 1542                     # historic default

    # --------------------------------------------------------------
    # Global best tracking
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Optimisation hyper‑parameters
    # --------------------------------------------------------------
    max_iter_per_restart = 20000
    beta_start = 5.0          # smooth‑max temperature at the start
    beta_end   = 30.0         # temperature at the end of a restart
    lr0 = 1.0                 # base learning rate for Adam
    beta1 = 0.9               # Adam first‑moment decay
    beta2 = 0.999             # Adam second‑moment decay
    eps_adam = 1e-8
    noise_start = 1e-2        # magnitude of injected noise (relative)
    eval_every = 30           # how often to call `evaluate_sequence`

    restart = 0
    while time.time() < deadline:
        restart += 1

        # ------------------- initialise a candidate --------------------
        init_mode = np.random.choice(["sqrtdiff", "power", "random"])
        if init_mode == "sqrtdiff":
            a = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        elif init_mode == "power":
            p_exp = np.random.uniform(0.1, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-p_exp))
        else:  # "random"
            a = np.random.rand(n)

        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.ones(n)
        a = a / a.sum()                 # simplex initialisation

        # -------------------- Adam state ------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # -------------------- optimisation loop -----------------------
        t = 0
        while time.time() < deadline and t < max_iter_per_restart:
            # smooth‑max temperature (linear continuation)
            beta = beta_start + (beta_end - beta_start) * (t / max_iter_per_restart)

            # gradient of smooth‑max
            grad = _smooth_max_grad(a, beta)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            # bias‑corrected estimates
            m_hat = m / (1.0 - beta1 ** (t + 1))
            v_hat = v / (1.0 - beta2 ** (t + 1))

            # learning‑rate schedule (diminishing)
            lr = lr0 / math.sqrt(t + 1)

            # parameter step
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # optional small random perturbation to aid exploration
            if t % 200 == 0 and t > 0:
                sigma = noise_start * (1.0 - t / max_iter_per_restart)
                perturb = sigma * np.random.randn(n) * a
                a += perturb
                a = np.clip(a, 0.0, None)

            # project back onto the probability simplex
            a = _project_to_simplex(a)

            # evaluate occasionally (costly)
            if t % eval_every == 0:
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    elapsed = time.time() - start_time
                    print(f"[Restart {restart}] iter {t:05d} – new best {best_score:.6f} (t={elapsed:.1f}s)")

            t += 1

        # -------------------- local polishing -------------------------
        if best_seq is not None:
            a_local = best_seq.copy()
            polish_deadline = time.time() + min(6.0, deadline - time.time())
            while time.time() < polish_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # move a tiny fraction of mass from i to j
                eps_move = min(a_local[i],
                               a_local[i] * np.random.uniform(0.0, 0.05))
                if eps_move <= 0.0:
                    continue
                a_new = a_local.copy()
                a_new[i] -= eps_move
                a_new[j] += eps_move
                a_new = _project_to_simplex(a_new)
                s = evaluate_sequence(a_new.tolist())
                if s < best_score:
                    best_score = s
                    best_seq = a_new.copy()
                    a_local = a_new
                    elapsed = time.time() - start_time
                    print(f"[Polish] improved to {best_score:.6f} (t={elapsed:.1f}s)")

        # --------------------------------------------------------------
        # Stop if we ran out of time
        # --------------------------------------------------------------
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        a_fallback = np.sqrt(np.arange(1, n + 1, dtype=float)) - np.sqrt(np.arange(0, n, dtype=float))
        best_seq = a_fallback / a_fallback.sum()

    # ------------------------------------------------------------------
    # Scale to the [0, 1000] range required by the evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    total_elapsed = time.time() - start_time
    print(f"Returning sequence – best score {best_score:.6f} after {total_elapsed:.1f}s")
    return final_seq.tolist()
