"""
Improved search for a low‑value step sequence.

The evaluation metric  E(a)=2·n·max(a*a)/(sum a)^2  is scale‑invariant,
so we restrict the search to the probability simplex (sum = 1, a_i≥0).

Algorithm
---------

1. **Initialisation** – Load the previously discovered sequence (if any) and
   normalise it; otherwise start from a uniform distribution of moderate size
   (≈2000).

2. **Smooth optimisation (L‑BFGS)** – Repeatedly minimise a smooth
   approximation of the max (log‑sum‑exp) using L‑BFGS with a continuation
   schedule for the smoothing parameter λ.  Randomised starting points
   (exponential, power‑law, edge‑heavy, or the current best) are used
   to explore many basins.

3. **Projected sub‑gradient refinement** – Directly minimise the true
   max‑convolution using a sub‑gradient step with Euclidean projection onto
   the simplex.  A longer dedicated phase is allocated to this method.

4. **Simulated annealing** – Starting from the best point found so far,
   repeatedly move a random fraction of mass from one coordinate to another.
   Moves that improve the score are always accepted; uphill moves are
   accepted with a Boltzmann probability that gradually cools.  This
   non‑local search helps escape remaining local minima.

5. **Final greedy hill‑climb** – A fast local optimiser that transfers a
   small fraction of mass between two positions while keeping the sum fixed.
   It polish‑es the result before returning it.

All convolutions are performed via FFT (O(n log n)) and the simplex
projection is exact (Euclidean).  The whole routine respects the supplied
time budget (default 1000 s) and prints progress information.

The returned list is a non‑negative step sequence ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c            # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # g_vec_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0          # larger initial step size
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active (binding) indices – tolerance avoids numeric gaps
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to the original n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple simulated annealing that moves a fraction of mass between two
    positions.  Accepts improving moves always, uphill moves with a
    temperature‑dependent probability.  Temperature decays geometrically.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # initial temperature as a small fraction of the current score
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995  # geometric cooling factor
    it = 0

    while time.time() < deadline:
        # pick a source with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue

        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a random fraction of the source mass
        max_delta = cur_a[i] * 0.30  # allow up to 30 % of the source mass
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # apply the move
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        # Metropolis acceptance
        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert move
                cur_a[i] += delta
                cur_a[j] -= delta
                # continue without updating best
                it += 1
                T *= decay
                continue

        # record best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        # temperature schedule
        T *= decay
        # occasional reheating if temperature becomes too tiny
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)

        it += 1

    return best_a, best_score

        init_seq = np.array(globals()['height_sequence_1'], dtype=float)

    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    iteration = 0

    while time.time() < phase1_end:
        lam = lam_schedule[iteration % len(lam_schedule)]

        # Select a random start mode
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0005, 0.02)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy heuristic
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # Run L‑BFGS on the smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS {iteration}] lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # Short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(phase1_end, time.time() + 0.35)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – longer sub‑gradient refinement
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase2_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for fine‑grained exploration
    # ------------------------------------------------------------------
    phase3_end = deadline
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=3000)

    # Evaluate with the provided evaluator for reporting
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception as e:   # evaluate_sequence might not be imported in local testing
        eval_val = float('nan')
    print(f"[final] evaluated score = {eval_val:.6f}")

    return final_a.tolist()
