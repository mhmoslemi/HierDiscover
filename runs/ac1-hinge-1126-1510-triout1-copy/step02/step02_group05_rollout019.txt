def adam_smooth_opt(a0, max_iters, deadline, beta_start=5.0, beta_end=60.0, lr=0.05):
    n = len(a0)
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b_len = 2*n - 1

    # Precompute FFT size for conv and correlation
    N = 1 << ((3*n).bit_length())  # power of two >= 3n

    best_seq = a.copy()
    best_score = float('inf')
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute convolution b and max
        A = np.fft.rfft(a, n=N)
        b = np.fft.irfft(A * A, n=N)[:b_len]
        max_b = b.max()
        # compute score
        score = 2.0 * n * max_b / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # compute smooth weights
        beta = beta_start + (beta_end - beta_start) * it / max_iters
        # using scaled subtraction for numeric stability
        b_max = max_b
        # compute exp(beta*(b-b_max)) to avoid overflow
        exp_vals = np.exp(beta * (b - b_max))
        w = exp_vals / np.sum(exp_vals)
        # compute gradient via correlation
        # correlation of a and w:
        # Compute FFT for a and w length N
        W = np.fft.rfft(w, n=N)
        corr_full = np.fft.irfft(A * np.conj(W), n=N)  # length N
        # extract slice for indices 0..n-1
        start = (b_len - 1)  # 2n-2
        grad = 2.0 * corr_full[start : start + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        # Project onto simplex
        a = project_simplex(a)
    return best_seq, best_score

def adam_optimize(a0, max_iters, deadline):

    n = a.shape[0]
    # Ensure positivity and sum 1
    a = np.maximum(a, 0)
    s = a.sum()
    if s == 0: a = np.full(n, 1.0 / n)
    else: a = a / s

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr = 0.05 # may adjust

    # Precompute FFT size
    N = 1 << ((3 * n).bit_length()) # Next power-of-two >= 3n

    # Variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_score = float('inf')

    # schedule for beta smoothing
    beta_start = 5.0
    beta_end = 80.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # compute convolution for actual max
        A = np.fft.rfft(a, n=N)
        b = np.fft.irfft(A * A, n=N)[: (2 * n - 1)]
        max_b = b.max()
        # Compute score (since sum a approx 1)
        score = 2.0 * n * max_b / (a.sum() ** 2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # optionally print progress
            # print(f"[iter {it}] best score={score:.6f}")

        # compute smoothed weights
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max_iters
        # for numerical stability use max
        b_max = max_b
        # compute exp(beta*(b - b_max))
        exp_vals = np.exp(beta * (b - b_max))
        w = exp_vals / np.sum(exp_vals)  # shape (2n-1)

        # compute gradient via correlation of a and w
        # compute FFT of w
        W = np.fft.rfft(w, n=N)
        corr_full = np.fft.irfft(A * np.conj(W), n=N)  # length N
        start = (2 * n - 2)  # index where correlation begins for 0 shift
        grad = 2.0 * corr_full[start : start + n]  # shape n

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        # bias corrections
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # Adam step
        step = lr * m_hat / (np.sqrt(v_hat) + epsilon)

        a = a - step

        # Project to simplex (clip & renorm)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

    return best_seq, best_score

"""
Fast Adam‑based optimizer for minimizing the self‑convolution peak.

The algorithm works on the probability simplex (a_i ≥ 0, Σ a_i = 1) and
minimises a smooth approximation of the objective
    2·n·max(a*a) / (Σ a)^2 .
A soft‑max (log‑sum‑exp) with an annealing parameter β turns the
non‑differentiable max into a smooth function.
The gradient of the smooth objective is obtained by a single FFT‑based
correlation, giving O(n log n) per iteration.
We run many short Adam optimisations from diverse random starts
(symmetric power‑law, Dirichlet, uniform, perturbations of the best
found so far) until the time budget expires and keep the best sequence.
A final tiny projected sub‑gradient step is applied as a polish.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – projection onto the simplex (clip‑and‑renorm is enough here)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σ x = 1} using the O(n) algorithm."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser that smooths the max with a log‑sum‑exp (soft‑max)
# ----------------------------------------------------------------------
def adam_smooth_opt(
    a0: np.ndarray,
    max_iters: int,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 80.0,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the simplex that minimises a smoothed version of
    max(conv(a,a)).  Returns the best sequence (as an ndarray) and the
    corresponding (exact) score 2·n·max/Σa² observed during the run.
    """
    a = a0.copy()
    n = a.shape[0]

    # ensure feasibility
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    # first‑order moments for Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # FFT size – must be ≥ 3·n to obtain a linear correlation
    N = 1 << ((3 * n).bit_length())

    best_seq = a.copy()
    best_score = float("inf")
    b_len = 2 * n - 1
    corr_start = b_len - 1            # == 2·n‑2, start index of linear correlation

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- convolution and exact score ----------
        A = np.fft.rfft(a, n=N)
        b = np.fft.irfft(A * A, n=N)[:b_len]      # full linear self‑convolution
        max_b = b.max()
        score = 2.0 * n * max_b / (a.sum() ** 2)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # occasional progress report
            if it % 500 == 0:
                print(f"[adam] iter {it}  best score = {score:.6f}")

        # ---------- soft‑max weighting ----------
        # anneal β linearly from beta_start to beta_end
        beta = beta_start + (beta_end - beta_start) * (it - 1) / max_iters
        # use max_b for numerical stability
        exp_vals = np.exp(beta * (b - max_b))
        w = exp_vals / exp_vals.sum()               # shape (2n‑1)

        # ---------- gradient (correlation a ⋆ w) ----------
        W = np.fft.rfft(w, n=N)
        corr_full = np.fft.irfft(A * np.conj(W), n=N)   # linear correlation a⋆w
        grad = 2.0 * corr_full[corr_start : corr_start + n]   # length n

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---------- feasibility (clip & renorm) ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
# A tiny projected sub‑gradient polish (optional, applied at the end)
# ----------------------------------------------------------------------
def polish_subgradient(a: np.ndarray, max_iters: int = 200, lr: float = 0.2) -> np.ndarray:
    """
    One‑shot projected sub‑gradient descent that directly reduces the
    *active* convolution peak.  Used only as a final polish.
    """
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # size for the self‑convolution only

    for _ in range(max_iters):
        # full convolution
        A = np.fft.rfft(a, n=N)
        b = np.fft.irfft(A * A, n=N)[: 2 * n - 1]
        k = int(np.argmax(b))          # active peak index
        # gradient of the active peak (2·a_{k‑i} on the overlapping window)
        grad = np.zeros_like(a)
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        for i in range(i_low, i_high + 1):
            j = k - i
            grad[i] = 2.0 * a[j]

        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
    return a

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine launches several Adam‑based optimisations from diverse
    random starts (including a symmetric power‑law family and a perturbation
    of the best sequence seen so far).  The best sequence encountered before
    the time limit is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (use the length of the previous solution if
    # it exists, otherwise fall back to 4000).
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))  # type: ignore[name-defined]  # noqa: F821
        if n < 1000:
            n = 4000
    except Exception:
        n = 4000

    # ------------------------------------------------------------------
    # Global best record
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    # If a previous construction is available, evaluate it immediately.
    try:
        cand = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]  # noqa: F821
        if cand.shape[0] == n:
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            score = evaluate_sequence(cand.tolist())
            if score < global_best_score:
                global_best_seq = cand.copy()
                global_best_score = score
                print(f"[init] previous sequence score = {score:.6f}")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Main search loop – many short restarts with diverse seeds
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1

        # ---- initialise a fresh starting point -------------------------
        r = np.random.rand()
        if r < 0.15:
            # Uniform
            a0 = np.full(n, 1.0 / n, dtype=float)
        elif r < 0.45:
            # Symmetric power‑law: a_i ∝ (i+1)^(-p) + (n-i)^(-p)
            p = np.random.uniform(0.45, 0.80)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p) + (n + 1 - idx) ** (-p)
            a0 /= a0.sum()
        elif r < 0.75:
            # Dirichlet‑like random (Gamma) vector
            a0 = np.random.gamma(shape=1.0, scale=1.0, size=n)
            a0 /= a0.sum()
        else:
            # Perturbation of the current global best (if any)
            if global_best_seq is not None:
                a0 = global_best_seq.copy()
                eps = 0.02 * np.random.randn(n)
                a0 = np.maximum(a0 + eps, 0.0)
                s = a0.sum()
                if s == 0.0:
                    a0 = np.full(n, 1.0 / n)
                else:
                    a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n, dtype=float)

        # ---- decide how many iterations we can afford -----------------
        time_left = deadline - time.time()
        # Empirical cost per iteration ≈ 0.002 s (conservative)
        it_budget = int(min(25000, max(2000, time_left / 0.002)))
        if it_budget <= 0:
            break

        # ---- run the Adam‑based optimiser -----------------------------
        seq, approx_score = adam_smooth_opt(
            a0,
            max_iters=it_budget,
            deadline=deadline,
            beta_start=5.0,
            beta_end=80.0,
            lr=0.05,
        )

        # Evaluate the exact score (avoid any rounding artefacts)
        exact_score = evaluate_sequence(seq.tolist())
        if exact_score < global_best_score:
            global_best_score = exact_score
            global_best_seq = seq.copy()
            print(
                f"[restart {restart}] new global best score = {exact_score:.6f} "
                f"(iters≈{it_budget})"
            )

        # Occasionally shrink the learning‑rate a little if progress stalls
        if restart % 5 == 0:
            # not a strict rule – simply a placeholder for manual tuning
            pass

    # ------------------------------------------------------------------
    # Final polish with a short projected sub‑gradient descent
    # ------------------------------------------------------------------
    if global_best_seq is not None:
        polished = polish_subgradient(global_best_seq.copy(), max_iters=300, lr=0.15)
        polished_score = evaluate_sequence(polished.tolist())
        if polished_score < global_best_score:
            global_best_seq = polished
            global_best_score = polished_score
            print(f"[polish] improved score = {polished_score:.6f}")

    # ------------------------------------------------------------------
    # Safety net – if nothing succeeded, fall back to a modest power‑law.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        idx = np.arange(1, n + 1, dtype=float)
        global_best_seq = idx ** (-0.55)
        global_best_seq /= global_best_seq.sum()
        print("[fallback] using simple power‑law seed")

    # Convert to plain Python list as required by the harness.
    return global_best_seq.tolist()
