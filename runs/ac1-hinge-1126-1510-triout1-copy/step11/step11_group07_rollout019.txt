```
def targeted_hill_climb(a, deadline, rng, max_frac_initial=0.02):
    best = a.copy()
    best_score = fast_score(best, fft_len)
    conv = np.convolve(best, best)
    while time.time() < deadline:
        # compute gradient at max index
        fa = np.fft.fft(best, fft_len)
        conv_arr = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_idx = int(conv_arr.argmax())
        a_rev = best[::-1]
        # gradient = 2 * convolution of a_rev with mask where only at max_idx:
        mask = np.zeros(2*n-1)
        mask[max_idx] = 1.0
        grad_full = np.real(np.fft.ifft(np.fft.fft(a_rev, fft_len) * np.fft.fft(mask, fft_len)))[:n]
        g = 2.0 * grad_full
        src = np.argmax(g)  # largest gradient component
        dst = np.argmin(g)  # smallest gradient component
        if src == dst: # pick random fallback
            src = rng.choice(n)
            dst = rng.choice(n)
        delta = best[src] * rng.random() * max_frac_initial
        if delta <= 0:
            continue
        # update trial
        new = best.copy()
        new[src] -= delta
        new[dst] += delta
        # update conv incrementally
        diff = np.zeros_like(conv)
        diff[src:src+n] -= best
        diff[dst:dst+n] += best
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0*n*conv_new.max()
        if new_score < best_score:
            best = new
            conv = conv_new
            best_score = new_score
        # adapt max_frac decreasing over time
        # maybe adjust with schedule
```

"""
Improvements over the previous baseline:
- Uses a targeted hill‑climb that moves mass from the index contributing
  most to the current maximum convolution to the index contributing the least.
- Gradient for the max‑convolution is computed analytically (O(n) per step),
  avoiding the costly FFT for the sub‑gradient.
- A simple simulated‑annealing acceptance is added to escape plateaus.
- The overall search budget is allocated mainly to this fine‑grained
  local optimisation after an initial exponentiated‑gradient warm‑up.
"""

import time
import math
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int) -> float:
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def arcsine_initial(n: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def targeted_hill_climb(a_start: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always moves mass from the index that
    most harms the current maximum convolution to the index that
    harms it the least.  The convolution is updated each iteration
    by a fresh FFT (still cheap for n≈4000) and a simple simulated‑annealing
    acceptance rule lets the walk escape shallow plateaus.
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    a = a_start.copy()
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    best = a.copy()
    best_score = float(2 * n * conv.max())

    start_time = time.time()
    last_improve = start_time

    while time.time() < deadline:
        # current max convolution index
        k = int(conv.argmax())

        # gradient of the max‑convolution w.r.t. a (closed form)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        g[i] = 2.0 * a[j]                      # g_i = 2 * a_{k-i}

        # choose source (largest gradient) and destination (smallest gradient)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback to a random move if the deterministic choice is illegal
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(0, n)
            if src == dst or a[src] <= 0.0:
                continue

        # step size – a decreasing fraction of the source mass
        elapsed = time.time() - start_time
        fraction = 1.0 - elapsed / (deadline - start_time)
        max_frac = 0.02 * fraction + 0.001          # shrink to a few ‰ at the end
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # recompute convolution (full FFT – still cheap)
        fa_new = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa_new * fa_new))[: 2 * n - 1]
        new_score = float(2 * n * conv_new.max())

        # deterministic improvement test
        improve = new_score < best_score

        # simulated‑annealing acceptance if not directly improving
        if not improve:
            # temperature proportional to remaining fraction of the budget
            T = max_frac * 0.5 + 1e-12
            prob = math.exp(-(new_score - best_score) / T)
            if rng.random() < prob:
                improve = True

        if improve:
            a = a_new
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                last_improve = time.time()
        else:
            # occasionally inject a tiny jitter to get unstuck
            if time.time() - last_improve > 5.0:
                jitter = rng.normal(scale=1e-5, size=n)
                a = a + jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                fa = np.fft.fft(a, fft_len)
                conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
                last_improve = time.time()

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    The routine builds an arcsine‑shaped start, performs a short

    hill‑climb that always moves mass from the most harmful
    index to the least harmful one.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 works well for the given budget
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1.  Initialise with the arcsine distribution (heavy on the borders)

    # ------------------------------------------------------------------

    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient warm‑up
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on max(conv)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (local refinement)
# ----------------------------------------------------------------------


def targeted_hill_climb(a_start: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always moves mass from the index that
    most harms the current maximum convolution to the index that
    harms it the least.  The convolution is recomputed each step
    (still cheap via FFT) and a tiny simulated‑annealing rule helps
    escape plateaus.
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    a = a_start.copy()
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    best = a.copy()
    best_score = float(2 * n * conv.max())

    start_time = time.time()
    last_improve = start_time

    while time.time() < deadline:
        # index where the convolution attains its maximum
        k = int(conv.argmax())

        # analytic gradient of that entry w.r.t. a:
        #   g_i = 2 * a_{k-i}   (if the index k-i lies inside the array)
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        g[i] = 2.0 * a[j]

        # choose source (largest gradient) and destination (smallest gradient)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback: random move respecting the simplex
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(0, n)
            if src == dst or a[src] <= 0.0:
                continue

        # step size – a decreasing fraction of the source mass
        elapsed = time.time() - start_time
        fraction = 1.0 - elapsed / (deadline - start_time)
        max_frac = 0.02 * fraction + 0.001   # gently shrink near the deadline
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # recompute convolution from scratch (fast enough for n≈4000)
        fa_new = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa_new * fa_new))[: 2 * n - 1]
        new_score = float(2 * n * conv_new.max())

        # deterministic improvement?
        improve = new_score < best_score

        # simulated‑annealing acceptance if not improving
        if not improve:
            T = max_frac * 0.5 + 1e-12
            if rng.random() < math.exp(-(new_score - best_score) / T):
                improve = True

        if improve:
            a = a_new
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                last_improve = time.time()
        else:
            # occasional jitter to get unstuck
            if time.time() - last_improve > 5.0:
                jitter = rng.normal(scale=1e-5, size=n)
                a = a + jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                fa = np.fft.fft(a, fft_len)
                conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
                last_improve = time.time()

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 works well for the given budget
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1.  Initialise with an arcsine‑shaped distribution
    # --------------------------------------------------------------
    a = arcsine_initial(n)
    best_score = fast_score(a, fft_len)
    print(f"[init] arcsine fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2.  Warm‑up – short exponentiated‑gradient run (≈10 % of time)
    # --------------------------------------------------------------
    now = time.time()
    deadline = now + budget_s - 0.05                # tiny safety margin
    warm_up_deadline = now + 0.1 * (deadline - now)  # 10 % of the total budget
    a, best_score = exp_gradient_optimize(
        a, warm_up_deadline, n, fft_len, rng, eta0=30.0
    )
    print(f"[exp‑grad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Focused hill‑climb (the main optimisation phase)
    # --------------------------------------------------------------
    a, best_score = targeted_hill_climb(a, deadline, rng)
    print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
