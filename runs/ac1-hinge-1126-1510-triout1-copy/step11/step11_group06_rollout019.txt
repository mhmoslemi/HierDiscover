    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

        # compute conv via FFT for speed
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient (subgradient) of max conv: (mask * a_rev) convolution
        a_rev = a[::-1]
        grad = 2.0 * np.real(np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a_rev, fft_len)))[:n]
        # multiplicative update
        eta = eta0 / (np.sqrt(it+1))
        a = a * np.exp(-eta * grad)

        if s == 0: a = best_a.copy()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def projected_gradient_descent(a_start, deadline, n, fft_len, lr=0.01):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

         fa = np.fft.fft(a, fft_len)
         conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
         max_val = conv.max()
         eps = max_val * 1e-12 + 1e-12
         mask = (conv >= max_val - eps).astype(np.float64)
         grad = 2.0 * np.real(np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a[::-1], fft_len)))[:n]
         # gradient direction to reduce max: move opposite direction
         a_new = a - lr * grad
         a_new = np.clip(a_new, 0, None)  # nonnegative

         a_new = a_new / a_new_sum
         new_score = fast_score(a_new, fft_len)
         if new_score < best_score:
             best_score = new_score
             best_a = a_new.copy()
             a = a_new

"""
Propose a low‑evaluation step function.

The search combines several stages:
  1. Many short L‑BFGS runs on a smoothed log‑sum‑exp approximation of
     max(a∗a) with a geometric λ‑schedule.
  2. Adam optimisation on the same smoothed objective.
  3. Exponentiated‑gradient descent on the *true* max‑convolution
     (sub‑gradient is obtained by convolving the mask of maximal
     entries with the reversed vector).
  4. Additive projected‑gradient descent (mirror of the previous stage)
     for fine‑tuning.
  5. Tiny random local mass‑transfer moves to escape possible plateaus.

All stages respect the simplex (non‑negative entries, sum = 1) and
periodically recompute the exact convolution via FFT.  The algorithm
runs for up to the allotted budget (≈1000 s) and returns the best
sequence found.  The final sequence is also stored in the global variable
`height_sequence_1` for possible reuse in later calls.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for the true objective when ∑a = 1:
        score = 2·n·max(conv(a,a))
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:                     # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max‑convolution)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # soft‑max to stay on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient w.r.t. a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the *hard* max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Multiplicative (exponentiated) gradient descent directly on the true
    max‑convolution.  Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # full convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        grad_full = np.real(np.fft.ifft(
            np.fft.fft(mask, fft_len) *
            np.fft.fft(a_rev, fft_len)))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Additive projected‑gradient descent (fine‑tuning)
# ----------------------------------------------------------------------


def projected_gradient_descent(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               lr: float = 0.02):
    """
    Simple projected gradient descent on the true max‑convolution.
    Keeps the iterate on the simplex after each additive step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        grad_full = np.real(np.fft.ifft(
            np.fft.fft(mask, fft_len) *
            np.fft.fft(a_rev, fft_len)))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        a_new = a - lr * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0.0:
            break
        a_new /= s

        new_score = fast_score(a_new, fft_len)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        # optional: slowly decay learning rate (not essential)
        lr = max(lr * 0.9999, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Small random local refinements
# ----------------------------------------------------------------------


def random_local_refine(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator):
    """Move tiny amounts of mass between random pairs – keep improvements."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    while time.time() < deadline:
        src, dst = rng.integers(0, n, size=2)
        if src == dst or a[src] < 1e-12:
            continue
        # transfer 1 % of src's mass (or less, if src is tiny)
        delta = a[src] * 0.01
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # mass is conserved → sum stays 1, positivity ensured by the test above
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
    return best_a, best_score

    n = 4000
    fft_len = next_pow2(2 * n - 1)

    if 'height_sequence_1' in globals():

            if old.size != n:

                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:

    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
