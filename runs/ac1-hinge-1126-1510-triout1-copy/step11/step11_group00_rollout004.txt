"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() value.

Algorithm
---------
1. Build a diverse pool of feasible probability vectors (spectral guess,
   Hann, Blackman, Gaussian, random Dirichlet‑style, etc.).  Keep the
   best (exact fast_score = 2·n·max(conv) for sum‑1 vectors).

2. Phase 1 – many short L‑BFGS runs on a smoothed max‑convolution
   (log‑sum‑exp) with a decreasing smoothing schedule.  Each run starts
   from a random convex combination of the current best and a fresh
   random vector.

3. Phase 2 – Adam optimiser on the *smooth* max‑convolution.  The
   smoothing factor is halved every few thousand iterations so the
   optimiser gradually focuses on the true infinity norm.

4. Phase 3 – Simple simulated‑annealing with mass‑transfer moves to
   escape possible local minima.

5. Phase 4 – greedy hill‑climb (strictly improving pairwise mass swaps).

The code respects the 1000 s budget, adapts the time split automatically
and works for any sequence length (chosen here as 8192 for extra
degrees of freedom).  All helpers are top‑level functions – no closures
or lambdas are used.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score for probability vectors (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(self‑convolution) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of a*a using log‑sum‑exp.
    Arguments:
        y           – logits (softmax will produce a)
        n           – length of a
        fft_len_conv, fft_len_g – zero‑padded FFT lengths
        lam_factor  – λ = lam_factor * max(conv)
    Returns (objective, gradient) w.r.t. y.
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smoothed max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral guess obtained by taking a square‑root of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalised to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Adam optimiser on the *smoothed* max‑convolution.
    The smoothing factor λ = lam_factor * max(conv) starts at 0.2 and is
    halved every 5000 iterations, eventually focussing on the true max.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    lam_factor = 0.2  # initial smoothing

    while time.time() < deadline:
        it += 1

        # adapt smoothing schedule
        if it % 5000 == 0:
            lam_factor = max(0.001, lam_factor * 0.5)

        # compute smoothed objective and its gradient w.r.t. logits
        f, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam_factor=lam_factor)

        # Adam biased‑corrected update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.4 / np.sqrt(it)  # slowly decaying learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional random jitter to escape plateaus
        if it % 3000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # evaluate exact score (fast_score) and keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def simulated_annealing_opt(a_start: np.ndarray,
                            deadline: float,
                            n: int,
                            fft_len: int,
                            rng: np.random.Generator,
                            init_temp: float = 5e-4,
                            final_temp: float = 1e-8,
                            max_moves: int = 150000):
    """
    Simple SA with mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    start = time.time()
    move = 0
    while time.time() < deadline and move < max_moves:
        # linear temperature schedule
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.12
        if delta <= 0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # avoid tiny negative due to rounding
        if a_candidate[i] < 0:
            move += 1
            continue

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb_opt(a_start: np.ndarray,

                   fft_len: int,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_iter: int = 120000,
                   delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb based on pairwise mass swaps.
    """
    best = a_start.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
        # occasional tiny jitter to avoid stagnation
        if it % 5000 == 0:
            jitter = rng.random(n) * 1e-5
            best += jitter
            best = np.maximum(best, 0.0)
            best /= best.sum()
            best_score = fast_score(best, fft_len, n)
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a power‑of‑two for fast FFTs.
    # Larger n gives more freedom; 8192 fits easily into 1 GB RAM.
    # --------------------------------------------------------------
    n = 8192
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates (all are probability vectors)
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution is available, up‑sample it.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Randomised families
    for _ in range(5):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # --------------------------------------------------------------
    # Choose the best among the pool (fast_score is exact for sum=1)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Phase split (roughly 20% L‑BFGS, 60% Adam, rest SA + hill‑climb)
    phase1_end = start_time + 0.20 * budget_s
    phase2_end = start_time + 0.80 * budget_s

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS refinements with a smoothing schedule
    # ------------------------------------------------------------------
    lam_schedule = [0.35, 0.25, 0.15, 0.08, 0.04, 0.02,
                    0.01, 0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.35
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam‑smooth] start exact optimisation")
        best_a, best_score = adam_smooth_optimize(best_a,
                                                  deadline,
                                                  n,
                                                  fft_len,
                                                  rng)
        print(f"[Adam‑smooth] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing for escaping possible plateaus
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[SA] start")
        best_a, best_score = simulated_annealing_opt(best_a,
                                                     deadline,
                                                     n,
                                                     fft_len,
                                                     rng)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + remaining
        best_a, best_score = hill_climb_opt(best_a,
                                            best_score,
                                            fft_len,
                                            n,
                                            hill_deadline,
                                            rng,
                                            max_iter=200000,
                                            delta_frac=0.05)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (evaluate_sequence will rescale internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
