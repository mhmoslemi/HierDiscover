"""
A compact re‑implementation of the search for a low‑score non‑negative step
sequence.

Overview
--------
The evaluation function is invariant under scaling, therefore we can work
with vectors that sum to one (a simplex).  The score is
    score = 2·n·max_k (a*a)[k]    (fast_score)
and the true score differs only by the factor 1/(∑a)², which is 1 for the
simplex.

Two observations guide the design:

1.  For a pure exponential a_i ∝ exp(‑α i) the maximum of the
    convolution can be written analytically (no FFT needed) and the
    dependence on α can be explored with a very fine grid.

2.  The objective is non‑smooth (a max).  Its sub‑gradient can be computed
    in closed form:
        g_i = 2·a_{k*‑i}   (k* = argmax a*a)
    and the gradient of the full objective is
        ∇f = (2·n / S²)·( g – 2·M/S·1 )
    where S=∑a and M is the current maximal convolution value.
    A projected (simplex) sub‑gradient descent with a decaying step size
    quickly reduces the maximum and works well as a final polishing step.

The algorithm proceeds in four stages (all respecting a global deadline
that defaults to 1000 s):

    0.  Initialise from the uniform vector and from any historic best.
    1.  Very fine grid search over a single exponential family (α∈[2e‑4,
        1.5e‑3]) using the analytic formula for the max convolution.
    2.  Random search over a mixture of two exponentials
        a = w·exp(‑α₁·i) + (1‑w)·exp(‑α₂·i) (α₁,α₂∈[1e‑4,2e‑3],
        w∈[0,1]) – each candidate is normalised to the simplex.
    3.  Projected sub‑gradient descent (with momentum) starting from the
        best candidate found so far.  The descent works on the exact
        convolution (FFT) and terminates when the deadline is reached.
    4.  One final L‑BFGS refinement on the smooth log‑sum‑exp surrogate
        (as in the previous code) to polish any remaining slack.

All phases keep the best vector seen so far; the final vector is passed
to ``evaluate_sequence`` and returned as a plain list of floats.

The code uses only numpy and scipy (no file I/O) and respects the
1000 s global budget.  Print statements give a short progress log.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate for the official evaluation.
    a is assumed to be on the simplex (∑a = 1), therefore the denominator is 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic max‑convolution for pure exponential
# ----------------------------------------------------------------------


def exp_max_convolution(alpha: float, n: int) -> float:
    """
    Returns the maximal entry of the convolution of the (unnormalised)
    exponential vector e_i = exp(-alpha * i), i=0..n‑1.
    The convolution formula is:
        b_k = (k+1)·exp(-alpha·k)          for k < n
        b_k = (2·n-1-k)·exp(-alpha·k)      for k ≥ n
    """
    k = np.arange(2 * n - 1, dtype=np.float64)
    # counts of pairs contributing to each sum
    counts = np.where(k < n, k + 1, 2 * n - 1 - k)
    b = counts * np.exp(-alpha * k)
    return b.max()


def exp_fast_score(alpha: float, n: int, fft_len: int) -> float:
    """
    Fast evaluation of the score for a pure exponential distribution.
    The vector is normalised to the simplex before computing the score.
    """
    # raw max of the convolution (before normalisation)
    raw_max = exp_max_convolution(alpha, n)
    # raw sum of the exponential
    raw_sum = (1.0 - np.exp(-alpha * n)) / (1.0 - np.exp(-alpha))
    # normalised max = raw_max / raw_sum**2
    score = 2.0 * n * raw_max / (raw_sum * raw_sum)
    return float(score)

def generate_exp(n: int, rate: float) -> np.ndarray:
    """Pure exponential decay, normalised to the simplex."""
    a = np.exp(-rate * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def generate_mix_two_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Mixture of two exponentials with independent rates and a mixing weight.
    All three parameters are drawn uniformly from sensible intervals.
    """
    rate1 = rng.uniform(5e-05, 0.0020)      # α₁ (slow decay)
    rate2 = rng.uniform(rate1, 0.0025)      # α₂ (equal or faster)
    w = rng.uniform(0.0, 1.0)               # mixing proportion
    i = np.arange(n, dtype=float)
    a = w * np.exp(-rate1 * i) + (1.0 - w) * np.exp(-rate2 * i)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def generate_random_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Very sparse binary mask (useful to explore extreme configurations)."""
    prob = rng.uniform(0.02, 0.25)
    mask = rng.random(n) < prob
    if not mask.any():
        mask[0] = True
    a = mask.astype(float)
    a /= a.sum()
    return a

# Projected sub‑gradient descent on the true max‑convolution

def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
    init_lr: float = 0.5,
    decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Classical sub‑gradient descent with projection onto the simplex.
    Uses the exact convolution (FFT) to obtain the active index k*.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    lr = init_lr
    while time.time() < deadline:
        it += 1
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # sub‑gradient of the max (ordered pairs)
        g = np.zeros_like(a)
        low = max(0, k_star - (n - 1))
        high = min(k_star, n - 1)
        # i ranges over low..high, j = k_star - i
        idx_i = np.arange(low, high + 1, dtype=int)
        idx_j = k_star - idx_i
        g[idx_i] = 2.0 * a[idx_j]               # factor 2 for ordered pairs

        # gradient of the whole objective (scale invariant)
        S = a.sum()
        grad = (2.0 * n / (S * S)) * (g - (2.0 * max_val / S) * np.ones_like(a))

        # simple SGD update + projection
        a = a - lr * grad
        a = simplex_projection(a)

        # evaluate (fast) score
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[sgd] iter={it}  new best fast_score={best_score:.6f}")

        # decay learning rate
        lr *= decay

    return best_a, best_score


# ----------------------------------------------------------------------
# L‑BFGS refinement on a smoothed log‑sum‑exp surrogate
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the self‑convolution where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z
    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_refine(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    L‑BFGS continuation over a decreasing schedule of smoothing parameters.
    Starts from the vector a_start (already on the simplex).
    """
    y = np.log(a_start + 1e-12)          # logits – stay on simplex
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")
    return best_a, best_score

    # --------------------------------------------------------------
    print("[phase0] exhaustive exponential grid")
    rates = np.linspace(2e-4, 1.5e-3, 401)   # step 3.5e‑6 → very fine
    for rate in rates:
        if time.time() >= deadline:
            break
        sc = exp_fast_score(rate, n, fft_len_conv)
        if sc < best_score:
            best_score = sc
            best_a = generate_exp(n, rate)
            print(f"[grid] exp rate={rate:.6e} → fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Random mixture of two exponentials
    # --------------------------------------------------------------
    print("[phase0] random two‑exponential mixtures")
    while time.time() < deadline:
        a = generate_mix_two_exp(n, rng)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[mix2] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Projected sub‑gradient descent (exact max)
    # --------------------------------------------------------------
    # Reserve roughly half of the remaining time for this robust phase.
    phase3_end = time.time() + 0.45 * (deadline - time.time())
    print("[phase1] sub‑gradient descent start")
    best_a, best_score = subgradient_descent(
        best_a, phase3_end, fft_len_conv, n, init_lr=0.8, decay=0.9996
    )
    print(f"[phase1] best fast_score after SGD={best_score:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Final L‑BFGS refinement on a smoothed surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[phase2] final L‑BFGS polishing")
        lam_schedule = [
            0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
            0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
            0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        ]
        best_a, best_score = lbfgs_refine(
            best_a,
            lam_schedule,
            maxiter_per_lam=1500,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        print(f"[final‑lbfgs] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation (the only place where evaluate_sequence is used)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
