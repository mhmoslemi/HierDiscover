            cols.append(j)
            data.append(coeff)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
    b_ub = np.full(m, M)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return None, False
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None, False
    g = b / sum_b  # normalized to sum 1
    return g, True

"""
optimize_step.py

A search for a low‑score non‑negative step sequence.

Algorithm
---------
1.  Build a pool of deterministic and randomized initialisations
    (spectral, triangular, Hann, Blackman, Gaussian, inverse‑sqrt,
    arcsine, and several random draws).  The best of the pool is kept
    as the starting point.

2.  Main refinement – active‑set linear programming.
    For the current distribution f we compute its self‑convolution
    and its maximum M = max(f*f).  Only the most critical convolution
    entries (the top k values) are kept as an active set.
    On this active set we solve the linear programme

        maximise   sum g_j
        subject to f * g ≤ M   (only on active indices)
                   g_j ≥ 0

    The solution g is normalised to a probability vector and blended
    with f:  f ← (1‑t)·f + t·g  (t slowly decays).  This Frank–Wolfe/
    cutting‑plane step provably never increases the maximum
    convolution and often yields a strict improvement.

3.  Optional L‑BFGS polishing on the smoothed max‑objective
    (log‑sum‑exp).  The optimiser works on the soft‑max logits so
    the positivity constraint is automatically satisfied.

4.  A final greedy hill‑climb (mass‑transfer moves) to polish any
    remaining tiny improvements.

The whole procedure respects the 1000 s time budget and returns the
best sequence found (as a plain Python list of floats).  The final
score is obtained with the official ``evaluate_sequence`` routine.
"""

import time
import numpy as np
from scipy import fft
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = fft.rfft(a, fft_len)
    conv = fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def compute_full_convolution(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Return the full self‑convolution a*a (length 2·n‑1)."""
    fa = fft.rfft(a, fft_len)
    conv = fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Smooth max objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of self‑convolution; returns (obj, grad) w.r.t. y."""
    a = softmax(y)

    # full self‑convolution
    fa = fft.rfft(a, fft_len_conv)
    conv_full = fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = fft.rfft(a_rev, fft_len_g)
    fw = fft.rfft(w, fft_len_g)
    conv_rev = fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y

# Initialisation helpers (families of candidate distributions)

def spectral_init(n: int) -> np.ndarray:
    """Spectral guess: inverse FFT of a flat target spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_power_init(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(alpha,alpha) density on the discrete grid."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = (x * (1.0 - x)) ** (alpha - 1.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

def lbfgs_polish(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len_conv: int,
                 fft_len_g: int,
                 n: int):
    """
    L‑BFGS polishing on the smoothed max‑objective.
    Returns best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[lbfgs] improvement → {best_score:.8f}")
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 100000,
               delta_frac: float = 0.06):
    """Strictly improving greedy hill‑climb."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    print(f"[hill‑climb] final score = {best_score:.8f}")
    return best, best_score


# ----------------------------------------------------------------------
# Active‑set LP refinement (core of this solution)
# ----------------------------------------------------------------------


def solve_lp_active(f: np.ndarray, M: float, active_idxs: np.ndarray):
    """
    Solve   max 1ᵀg   s.t.   f * g ≤ M   (only on active indices),   g ≥ 0 .
    Returns a normalised probability vector g (sum = 1) on success,
    otherwise (None, False).
    """
    n = f.size
    m = active_idxs.size
    # Build sparse matrix A (shape m × n)
    rows = []
    cols = []
    data = []
    for row, k in enumerate(active_idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # columns j = i_start … i_end
        # coefficient is f[k - j]
        col_range = np.arange(i_start, i_end + 1, dtype=int)
        coeffs = f[k - col_range]
        rows.extend([row] * col_range.size)
        cols.extend(col_range.tolist())
        data.extend(coeffs.tolist())
    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    b_ub = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum(g) → minimise -sum(g)

    # Linear programme using HiGHS (fast for large sparse problems)
    res = linprog(c, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        return None, False
    g_raw = res.x
    total = g_raw.sum()
    if total <= 0:
        return None, False
    g = g_raw / total
    return g, True


def lp_active_refinement(f0: np.ndarray,
                         fft_len: int,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 30,
                         max_active: int = 200,
                         t0: float = 0.5):
    """
    Repeated active‑set LP improvement.
    Returns the best vector found and its fast_score.
    """
    f = f0.copy()
    best_score = fast_score(f, fft_len, n)
    best_f = f.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = compute_full_convolution(f, fft_len, n)
        M = conv.max()
        # Select top entries as active constraints
        k = min(max_active, int(0.02 * conv.size))
        if k <= 0:
            k = max_active
        active_idxs = np.argpartition(conv, -k)[-k:]

        g, ok = solve_lp_active(f, M, active_idxs)
        if not ok:
            print("[LP] solver failed – terminating LP phase")
            break

        # mixing factor, decreasing schedule
        t = t0 / (1.0 + it)
        f = (1.0 - t) * f + t * g
        f /= f.sum()          # numerical safety

        sc = fast_score(f, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
            print(f"[LP] iter {it:02d} → score {best_score:.8f}")
        # early stop if improvement is tiny
        if abs(best_score - sc) < 1e-9:
            break
        it += 1
    return best_f, best_score

        maxiter_per_lam = 1200
        best_a, best_score = lbfgs_polish(best_a,
                                          lam_schedule,
                                          maxiter_per_lam,
                                          phase2_end,
                                          fft_len, fft_len,
                                          n)
        print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (≈30 % of total time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=150000,
                                        delta_frac=0.07)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (the official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
