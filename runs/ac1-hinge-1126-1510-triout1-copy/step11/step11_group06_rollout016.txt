"""
Improved step‑function search for the evaluation function.

Approach
--------
1. Initialise a large number of random candidates on the simplex.
2. Optimise each candidate with a smoother log‑sum‑exp objective using
   L‑BFGS and a long Adam schedule (many iterations, finer λ‑schedule).
3. Refine the best candidate with several *independent* runs of a
   gradient‑guided hill‑climb.  Each run starts from a slightly perturbed
   version of the current best sequence, runs until its own sub‑deadline,
   and uses a slowly decaying step‑size.
4. After hill‑climbing, symmetrise the sequence (average with its reverse)
   – this often lowers the peak of the self‑convolution.
5. Return the best sequence found before the overall deadline.

The code is deliberately modular and contains only top‑level functions,
so it can be imported directly by the evaluation harness.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) – the quantity minimised by the search."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random distribution on the simplex using a variety of modes."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the convolution) – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Returns (smooth objective, gradient) for the variable y where
    a = softmax(y).  The smooth objective is a log‑sum‑exp approximation
    to max(conv(a,a)).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a using FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient of smooth max w.r.t. a

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the smooth objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (new component)
# ----------------------------------------------------------------------


def guided_hill_climb(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      start_factor: float = 0.30,
                      end_factor: float = 0.001,
                      recompute_every: int = 500):
    """
    Gradient‑guided hill‑climb that moves a fraction of mass from the index
    with the largest sub‑gradient to the index with the smallest.
    The step size decays linearly with the remaining time.
    The exact convolution is recomputed after a fixed number of accepted moves
    to keep numerical drift under control.
    Returns the best a and its fast_score found before the deadline.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact convolution (length 2n‑1)
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    phase_start = time.time()
    total = deadline - phase_start
    accepted = 0
    steps = 0

    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]   # sub‑gradient (length n)

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback: random source with non‑negligible mass
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero,
                                 p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        delta = min(delta, a[src])

        # keep copies for rollback
        a_old = a.copy()
        conv_old = conv.copy()

        # incremental update of the convolution (linear part only)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[dst:dst + n] += 2.0 * delta * a_old

        # move the mass
        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score:
            # accept the move
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject – revert
            a = a_old
            conv = conv_old

        steps += 1

        # periodic exact recompute to keep the linear‑approximation honest
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        if steps % 10000 == 0:
            elapsed = time.time() - phase_start
            print(f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Small post‑processing helper – symmetrisation
# ----------------------------------------------------------------------


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Average the sequence with its reversal and renormalise to the original sum."""
    s = a.sum()
    a_sym = (a + a[::-1]) * 0.5
    a_sym *= s / a_sym.sum()
    return a_sym


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – a few thousand entries give a good trade‑off
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # initialise from a stored construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – extensive L‑BFGS bursts from random starts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.12 * budget_s          # 12 % of the budget
    lam_schedule = np.geomspace(0.5, 1e-7, num=20)    # finer λ‑schedule
    maxiter_per_lam = 500
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective (long schedule)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = list(np.geomspace(0.5, 1e-7, num=30))
        best_a, best_score = adam_optimize(
            y0, lam_sched, max_steps_per_lam=4000,
            deadline=deadline, n=n, fft_len=fft_len, lr0=0.14
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, rng, eta0=12.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – several independent guided hill‑climbs
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # split the remaining budget into a handful of hill‑climb trials
        trials = max(1, int(remaining // 100))   # roughly 100 s per trial
        trial_deadline = time.time() + remaining / trials
        for t in range(trials):
            if time.time() >= deadline:
                break
            # start each trial from a lightly perturbed version of the current best
            a_start = best_a.copy()
            noise = rng.normal(scale=0.015, size=n)
            a_start = a_start + noise
            a_start = np.clip(a_start, 0, None)
            a_start /= a_start.sum()
            start_score = fast_score(a_start, fft_len)

            sub_deadline = min(deadline, time.time() + (deadline - time.time()) / 2)
            a_res, score_res = guided_hill_climb(
                a_start, start_score, sub_deadline, rng,
                start_factor=0.30, end_factor=5e-4, recompute_every=300
            )
            if score_res < best_score:
                best_score = score_res
                best_a = a_res.copy()
                print(f"[guided_hc trial {t+1}] new best fast_score={best_score:.6f}")

            # after each trial also try a simple symmetrisation – often beneficial
            a_sym = symmetrize(best_a)
            sym_score = fast_score(a_sym, fft_len)
            if sym_score < best_score:
                best_score = sym_score
                best_a = a_sym.copy()
                print(f"[symmetrize] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
