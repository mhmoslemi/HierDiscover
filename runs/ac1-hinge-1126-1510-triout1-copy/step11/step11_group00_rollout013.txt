"""
Propose a low‑score step function.

Algorithm
---------
1. **Conditional‑gradient (Frank‑Wolfe) on a small instance** –  
   Solve a linear program that maximises the ℓ₁‑norm of a direction
   g under the constraint (f * g)[k] ≤ M, where M = max(f * f).  The
   update f←(1‑α)f+α g with α=2/(t+2) reduces the maximal convolution
   value.  A few hundred cheap LP steps on n≈256 produce a shape that
   dramatically flattens the self‑convolution.

2. **Upsample** the small‑scale solution to the target size (n≈4096) and
   normalise to a probability vector.

3. **L‑BFGS refinement** – minimise a smooth log‑sum‑exp approximation
   of the max‑convolution, using a decreasing schedule of the smoothing
   parameter λ.  Random blends with fresh Dirichlet‑like vectors keep the
   search diverse.

4. **Exact Adam optimisation** on the true objective
   2·n·max(a * a) (a lives on the simplex via softmax).

5. **Greedy hill‑climb polishing** – strictly improving mass‑transfer
   moves finish the run.

All steps respect the 1000 s budget and the non‑negativity/normalisation
requirements of `evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import minimize, linprog
# evaluate_sequence is assumed to be imported in the execution environment


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (sum(a)=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (random draws on the simplex)
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y so that a = softmax(y) stays on the simplex.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient of max conv w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # gradient of the objective (2·n·max_conv)
        grad_a = 2.0 * n * g_a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam bias‑corrected updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        # decaying learning rate
        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaux
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Conditional‑gradient (Frank‑Wolfe) optimisation on a small instance
# ----------------------------------------------------------------------
def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A such that (f * g)[k] = (A @ g)[k].
    Shape of A is (2·n‑1, n).
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def conditional_gradient_optimize_small(n: int,
                                        iters: int,
                                        rng: np.random.Generator) -> np.ndarray:
    """
    Frank‑Wolfe style optimisation:
    maximise sum(g) s.t. (f * g)[k] ≤ M = max(f * f) and g≥0.
    """
    f = np.full(n, 1.0 / n)   # start uniform
    for it in range(iters):
        conv = np.convolve(f, f)
        M = conv.max()

        A = build_conv_matrix(f)
        b = np.full(2 * n - 1, M)
        c = -np.ones(n)                       # minimise -∑g  ⇔ maximise ∑g

        # linear programme (highs solver works with sparse A)
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=(0, None), method='highs',
                      options={'presolve': True})
        if not res.success:
            break
        g = res.x
        sumg = g.sum()
        if sumg <= 0:
            break
        g = g / sumg                       # renormalise to sum=1

        # step size α_t = 2/(t+2) (standard FW schedule)
        alpha = 2.0 / (it + 2.0)
        f = (1.0 - alpha) * f + alpha * g
        # keep feasibility
        f = np.maximum(f, 0.0)
        f /= f.sum()
    return f


def upsample(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from a_small to length n_large."""
    x_small = np.linspace(0, 1, a_small.size)
    x_large = np.linspace(0, 1, n_large)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.clip(a_large, 0.0, None)
    a_large /= a_large.sum()
    return a_large


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1) Small‑scale conditional‑gradient optimisation
    # ------------------------------------------------------------------
    n_small = 256
    cg_iters = 200
    print("[CG] optimizing on small instance (n=%d) …" % n_small)
    a_small = conditional_gradient_optimize_small(n_small, cg_iters, rng)
    print("[CG] done – best small‑scale fast_score = %.8f"
          % fast_score(a_small, next_pow2(2 * n_small - 1), n_small))

    # ------------------------------------------------------------------
    # 2) Upsample to the target size
    # ------------------------------------------------------------------
    n = 4096                         # target length – power of two
    fft_len = next_pow2(2 * n - 1)   # length for zero‑padded FFTs

    a = upsample(a_small, n)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] up‑sampled to n={n}, fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – L‑BFGS continuation on a smoothed max
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s   # 20 % of total time
    lam_schedule = [0.30, 0.20, 0.10, 0.05, 0.02, 0.01,
                    0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4, 5e-5]
    maxiter_per_lam = 800

    while time.time() < phase1_deadline:
        # diversify start point a little
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = a_start / a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – Exact Adam optimisation
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_score = adam_optimize(best_a,
                                           deadline,
                                           n,
                                           fft_len,
                                           rng)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # 5) Phase 3 – Final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 40.0)  # up to 40 s
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        rng,
                                        max_iter=60000,
                                        delta_frac=0.05)
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
