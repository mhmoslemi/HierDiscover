"""
Improved step‑function search for the evaluation metric.

Main ideas
----------
* Work with a *probability simplex*: the objective is scale invariant,
  so we normalise the vector to sum‑to‑one and minimise
  2·n·max(conv(a,a)) (fast_score).
* Enlarge the family of handcrafted candidates (beta‑shapes,
  inverse‑power, raised‑cosine, Hann/Hamming windows, etc.).
* Run a multi‑phase optimisation:
    0. systematic grid + cheap random sweep,
    1. many restarts of L‑BFGS on the smooth log‑sum‑exp (LSE) surrogate,
    2. projected Adam on the true max‑convolution (sub‑gradient),
    3. simulated annealing with richer moves,
    4. final L‑BFGS polish with a very small smoothing factor.
* All phases respect a global deadline (default ≈ 1000 s).  The best
  vector seen at any moment is returned.

The code uses only numpy / scipy, no I/O, and obeys the competition
rules.  The finalisation step evaluates the returned sequence with the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·len(a)-1 entries (the linear part)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of evaluate_sequence for sum(a)=1:
    2·n·max(conv(a,a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (logits → probabilities)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution (full linear conv)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (vector of length n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)  # logits – stays on the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={
                    "maxiter": maxiter_per_lam,
                    "gtol": 1e-12,
                    "disp": False,
                },
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution (sub‑gradient of max).
    Very robust for long runs.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.4

    while time.time() < deadline:
        it += 1
        # true convolution a*a
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()  # any convex combo is a valid sub‑gradient

        # sub‑gradient of the max (linear in mask)
        a_rev = a[::-1]
        fw = np.fft.rfft(mask, fft_len_g)
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        conv_g = np.fft.irfft(fa_rev * fw, fft_len_g)
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update (project afterwards)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with square‑root cooling.
    Moves:
      – transfer a random fraction of mass from i → j,
      – swap two entries,
      – redistribute mass on three random positions.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # temperature initialization
    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # square‑root cooling
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            # simple transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            # triple redistribution (keep sum)
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_raised_cosine(n: int, width: float) -> np.ndarray:
    """Cosine bell of given width (positive, ≤ n)."""
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    a = np.cos(np.pi * i / width)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like shape on [0,1] using centre‑point discretisation."""
    # centre points avoid exact 0/1 singularities
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def generate_inverse_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = 1.0 / (i ** p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Systematic deterministic sweep over many families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential
    for rate in np.linspace(5e-4, 1.5e-3, 25):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5g} → {best_sc:.6f}")

    # power law (both directions)
    for p in np.linspace(-4.5, -0.5, 21):
        if time.time() >= deadline:
            return best_a, best_sc
        for inc in (True, False):
            a = generate_power(n, p, inc)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                dirc = "inc" if inc else "dec"
                print(f"[grid] power p={p:.2f} {dirc} → {best_sc:.6f}")

    # Gaussian width sweep
    for sigma in np.linspace(200.0, 2000.0, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_gauss(n, sigma)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] gauss sigma={sigma:.0f} → {best_sc:.6f}")

    # Hann window (single)
    a = generate_hann(n)
    sc = fast_score(a, fft_len, n)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
        print(f"[grid] hann → {best_sc:.6f}")

    # Hamming window
    a = generate_hamming(n)
    sc = fast_score(a, fft_len, n)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
        print(f"[grid] hamming → {best_sc:.6f}")

    # Raised cosine width sweep
    for width in np.linspace(0.15 * n, 0.45 * n, 13):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_raised_cosine(n, width)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] raised_cos width={width:.0f} → {best_sc:.6f}")

    # Beta family (symmetric α=β)
    for a_par in np.linspace(0.3, 3.0, 15):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_beta(n, a_par, a_par)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] beta α=β={a_par:.2f} → {best_sc:.6f}")

    # Inverse power law
    for p in np.linspace(0.3, 2.0, 18):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_inverse_power(n, p)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] inv_power p={p:.2f} → {best_sc:.6f}")

    return best_a, best_sc


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "exp":
        rate = rng.uniform(4e-4, 1.6e-3)
        a = np.exp(-rate * i)
    elif family == "power":
        p = rng.uniform(-5.0, -0.4)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "gauss":
        sigma = rng.uniform(250.0, 1800.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "hann":
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    elif family == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    elif family == "raised_cos":
        width = rng.uniform(0.15 * n, 0.45 * n)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "beta":
        alpha = rng.uniform(0.2, 4.0)
        beta = rng.uniform(0.2, 4.0)
        x = (i + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    elif family == "inv_power":
        p = rng.uniform(0.2, 2.5)
        a = 1.0 / ((i + 1.0) ** p)
    elif family == "double_exp":
        rate = rng.uniform(4e-4, 1.4e-3)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "random":
        a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Quick random exploration over many families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10  # 10 % of total budget

    families = [
        "exp",
        "power",
        "gauss",
        "hann",
        "hamming",
        "raised_cos",
        "beta",
        "inv_power",
        "double_exp",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives a finer discretisation.
    # ------------------------------------------------------------------
    n = 5000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # ------------------------------------------------------------------
    # Initial candidate (uniform) and optional historic best
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # Use a previously saved construction if available
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] historic fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global timing
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – deterministic grid + cheap random sweep
    # ------------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time allocation for the remaining phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS restarts
    phase2_end = start_time + 0.75 * budget_s   # Adam sub‑gradient
    phase3_end = start_time + 0.95 * budget_s   # Simulated annealing
    # remaining time → final polishing L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.8, 0.6, 0.4, 0.25, 0.15, 0.09, 0.06, 0.04,
        0.025, 0.015, 0.008, 0.004, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 2e-05, 1e-05,
        5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # jitter the current best or start from a fresh family
        if rng.random() < 0.35:
            a_start = best_a * rng.normal(1.0, 0.03, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            fam = rng.choice(["exp", "power", "gauss", "beta", "inv_power"])
            a_start = generate_family(fam, n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] Adam best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing (rich move set)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – short L‑BFGS run with ultra‑small λ
    # ------------------------------------------------------------------
    if time.time() < deadline:
        tiny_sched = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08, 1e-08]
        a_final, sc_final = lbfgs_continuation(
            best_a,
            tiny_sched,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_score:
            best_score, best_a = sc_final, a_final.copy()
            print(f"[final‑lbfgs] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (scale‑invariant, but we keep sum≈1)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
