"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm Overview
------------------
1. **Diverse initial pool** – generate many candidate vectors using random
   families, classic windows and custom parametric shapes.  All candidates are
   normalised to the probability simplex (sum = 1).

2. **Aggressive L‑BFGS cascade** – minimise a smooth log‑sum‑exp
   approximation of the true L∞ objective.  The λ‑schedule is long and ends
   at 5e‑8, i.e. the surrogate becomes virtually the true max.  Each L‑BFGS
   run starts from a random pool element (occasionally perturbed by Gaussian
   noise).  The best vector found is kept.

3. **Mirrored exponentiated‑gradient descent** – a long run of multiplicative
   EG updates using the exact sub‑gradient of the max‑convolution.  The step
   size decays as 1/√t.

4. **Fine L‑BFGS** – a short low‑λ L‑BFGS pass (λ down to 1e‑8) to polish
   the solution.

5. **Metropolis annealing** – simple mass‑transfer / swap moves with a
   linear cooling schedule.

6. **Projected sub‑gradient descent** – final additive sub‑gradient steps
   with Euclidean projection onto the simplex for the last few seconds.

All stages respect the global 1000 s budget.  The best vector is stored in
the global variable `height_sequence_1` and its official score is printed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) assuming sum(a) == 1.
    Uses FFT of length fft_len ≥ 2·n‑1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros (should not happen); return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    # re‑normalise (projection onto simplex)
    return simplex_projection(noisy)

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative vector (sum = 1) from several styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian‑shaped density on the simplex."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power, clipped at zero."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass at the two edges, tiny interior."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a

def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Variables:
        y – unconstrained vector (log‑its of a)
        a = softmax(y) → lies on the simplex.
    Returns:
        f   – value of the smooth surrogate (2·n·smooth_max).
        grad_y – gradient w.r.t. y.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14  # avoid zero

    shifted = conv_full - max_c            # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)  # safely ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i
    # back‑prop through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]      # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS optimisations over decreasing λ."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (mirrored SGD)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored exponentiated‑gradient descent on the exact L∞ objective.
    Runs until the deadline.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the hard max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (enhanced)
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types:
        * mass transfer between two arbitrary positions
        * swapping two entries
        * local neighbour transfer (distance ≤ 5)
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25  # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # transfer
                j = rng.integers(0, n)
            if i == j:
                continue
            # move mass from larger to smaller entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        # fast evaluation (sum is preserved by the moves)
        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected additive sub‑gradient descent (final polishing)
# ----------------------------------------------------------------------


def subgradient_descent_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 eta0: float = 0.1):
    """
    Classic sub‑gradient descent on the exact L∞ objective,
    followed by Euclidean projection onto the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * g
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # 0) Re‑use the previously best construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) Classic windows (if SciPy available)
    try:
        from scipy.signal import chebwin, kaiser
    except Exception:
        chebwin = None
        kaiser = None

    if chebwin is not None:
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    if kaiser is not None:
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

    # 2) Random families
    for _ in range(6):
        pool.append(random_initial(n, rng))

    # 3) Parametric families
    for _ in range(4):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # Allocate 50 % to the aggressive L‑BFGS phase
    phase1_end = start_time + 0.50 * budget_s
    # Next 35 % to long exponentiated‑gradient descent
    phase2_end = start_time + 0.85 * budget_s

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
        5e-6, 2e-6, 1e-6, 5e-7, 2e-7, 1e-7,
        5e-8
    ]
    maxiter_per_lam = 800  # more iterations per λ than the baseline

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        # Occasionally add a small perturbation to escape plateaus
        if rng.random() < 0.30:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.01), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    if best_a is None:
        # fallback – uniform vector (should never happen)
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=25.0
        )
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with very low λ (final polishing)
    # --------------------------------------------------------------
    if time.time() < deadline - 60.0:
        fine_lam_sched = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
                         5e-7, 2e-7, 1e-7, 5e-8]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                        fine_lam_sched,
                                        maxiter_per_lam=1200,
                                        deadline=deadline - 30.0,
                                        fft_len=fft_len,
                                        n=n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – short simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 5 – projected additive sub‑gradient descent (last few secs)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_score = subgradient_descent_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            eta0=0.12
        )
        print(f"[sub‑grad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer and expose the result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
