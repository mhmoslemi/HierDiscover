def compute_subgradient(a):
    # compute conv = np.convolve(a,a) length 2n-1
    conv = np.convolve(a,a)
    max_c = conv.max()
    lam = max_c * 0.01 + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted/lam)
    Z = exp_shift.sum()
    w = exp_shift / Z
    # compute gradient: 2 * conv_rev(w, a)
    a_rev = a[::-1]
    g = 2 * np.real(np.fft.ifft(np.fft.fft(w, fft_len) * np.fft.fft(a_rev, fft_len))) # length 2n-1
    gradient = g[n-1:n-1+n]
    return gradient

def projected_subgradient(a, num_iters, deadline, initial_eta):
    n = a.shape[0]
    eta = initial_eta
    # compute fft_len for conv
    fft_len = next_pow2(2*n-1)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    for it in range(num_iters):
        if time.time() >= deadline:
            break
        # compute gradient
        conv = np.convolve(a, a) # O(n^2) might be heavy; use FFT: conv = np.real(np.fft.ifft(np.fft.fft(a,fft_len)**2))[:2*n-1]
        # we need conv anyway. Use FFT.
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_c = conv.max()
        lam = max_c * 0.01 + 1e-12  # smoothing parameter
        shifted = conv - max_c
        exp_shift = np.exp(shifted / lam)
        Z = exp_shift.sum()
        w = exp_shift / Z
        # gradient of smooth max = 2 * (a_rev convolve w)
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fw))
        g = 2 * grad_full[n-1:n-1+n]
        # step
        a_new = a - eta * g
        # project onto simplex
        a_new = simplex_projection(a_new)
        # evaluate
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        # optionally decay eta
        eta *= 0.9999
    return best_a, best_score

"""
Optimize a non‑negative step‑function (probability vector) to minimise

    2·n·max(conv(a,a)) / (sum(a)^2)

by a combination of random restarts, exponentiated‑gradient on the true
max‑convolution, smooth‑subgradient descent on a log‑sum‑exp surrogate,
and a final gradient‑guided hill‑climb.  The search runs for up to the
provided time budget (≈1000 s) and returns the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent directly on max(convolution).
    Returns best sequence and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # active set

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Smooth‑subgradient descent (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Gradient descent on a smooth approximation of max(conv) with
    projection onto the simplex every iteration.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_c = conv.max()
        # smoothing parameter (slowly decreasing)
        lam_factor = max(0.01, 0.5 * np.exp(-it / 2000.0))
        lam = max_c * lam_factor + 1e-12

        # log‑sum‑exp weights
        shifted = conv - max_c
        exp_shift = np.exp(shifted / lam)
        w = exp_shift / exp_shift.sum()

        # gradient of smooth max w.r.t a
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fw))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        # step size (inverse‑sqrt schedule)
        eta = 0.6 / np.sqrt(it + 1.0)
        a_new = a - eta * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = a.copy()

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (aggressive version)
# ----------------------------------------------------------------------


def guided_hill_climb(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      start_factor: float = 0.5,
                      end_factor: float = 0.001,
                      recompute_every: int = 1000) -> tuple[np.ndarray, float]:
    """
    Moves a fraction of mass from the index with the largest sub‑gradient
    to the index with the smallest sub‑gradient, accepting only non‑increasing
    moves.  Step size decays linearly with remaining time.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact convolution (length 2n‑1)
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score

    phase_start = time.time()
    total = deadline - phase_start
    accepted = 0
    steps = 0

    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fall back to a random move
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz, p=a[nz] / a[nz].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linear decay of step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        delta = min(delta, a[src])

        a_old = a.copy()
        conv_old = conv.copy()

        # linear update of convolution (first‑order)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[dst:dst + n] += 2.0 * delta * a_old

        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score:
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a = a_old
            conv = conv_old

        steps += 1

        # periodic exact recompute to avoid drift
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        if steps % 5000 == 0:
            elapsed = time.time() - phase_start
            print(f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} best_score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns the best list of non‑negative floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size: a moderately large n gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a stored construction (if any) and also from uniform.
    # ------------------------------------------------------------------
    init_candidates = []
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            init_candidates.append(a0)
    init_candidates.append(np.full(n, 1.0 / n))   # uniform baseline

    best_a = None
    best_score = np.inf
    for cand in init_candidates:
        sc = fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts of exponentiated‑gradient
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    eta_choices = [5.0, 8.0, 12.0, 16.0, 20.0]
    while time.time() < phase1_end:
        a0 = random_initial(n, rng)
        eta0 = rng.choice(eta_choices)
        a_opt, sc_opt = exp_gradient_optimize(
            a0, phase1_end, n, fft_len, rng, eta0=eta0
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[exp‑grad] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth sub‑gradient descent (log‑sum‑exp surrogate)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    while time.time() < phase2_end:
        # 50 % chance to start from the current best, otherwise a fresh random start
        if rng.random() < 0.5:
            a_start = best_a.copy()
        else:
            a_start = random_initial(n, rng)
        a_opt, sc_opt = projected_subgradient_optimize(
            a_start, phase2_end, n, fft_len, rng
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[subgrad] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = guided_hill_climb(
            best_a, best_score, deadline, rng,
            start_factor=0.5, end_factor=0.001, recompute_every=1000
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[guided_hc] final best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official security‑checked function.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # type: ignore  # provided by the harness
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
