def generate_start_vectors(n, rng):
   pool = []
   # uniform
   pool.append(np.full(n, 1.0/n))
   # beta distributions random params
   for _ in range(8):
       a = rng.uniform(0.2, 5.0)
       b = rng.uniform(0.2, 5.0)
       x = np.arange(1, n+1)
       w = (x**a) * ((n+1 - x)**b)
       w /= w.sum()
       pool.append(w)
   # exponential decay
   for rate in rng.uniform(0.0005, 0.02, size=6):
       w = np.exp(-rate * np.arange(n))
       w /= w.sum()
       pool.append(w)
   # power law
   for p in rng.uniform(0.1, 3.0, size=6):
       w = (np.arange(1, n+1) ** (-p))
       w /= w.sum()
       pool.append(w)
   # double-sided exponential (symmetric)
   for rate in rng.uniform(0.001, 0.02, size=4):
       left = np.exp(-rate * np.arange(n//2))
       right = left[::-1]
       w = np.concatenate([left, right]) if n%2==0 else np.concatenate([left, [0], right])
       w = w[:n]
       w /= w.sum()
       pool.append(w)
   # previous best
   if 'height_sequence_1' in globals():
       old = np.array(globals()['height_sequence_1'], dtype=float)
       if old.ndim == 1 and old.size>0:
           x_old = np.linspace(0,1,old.size)
           x_new = np.linspace(0,1,n)
           a0 = np.interp(x_new,x_old,old)
           a0 = np.maximum(a0, 0)
           a0 /= a0.sum()
           pool.append(a0)
   # random
   for _ in range(5):
       a = rng.random(n)
       a /= a.sum()
       pool.append(a)
   return pool

"""
Proposes a step‑function (non‑negative vector) that minimises the official score.

Algorithm
---------

* **Phase 0 – Diverse seeds** –  several parametric families
  (uniform, beta, exponential, power‑law, symmetric‑exponential) are
  built, interpolated from a previous best and slightly perturbed.  The
  `fast_score` (exact value of the official objective when the vector
  sums to 1) is computed and the best few seeds are kept.

* **Phase 1 – Coarse L‑BFGS cascade** – each seed is refined by a
  cascade of L‑BFGS optimisations on a smooth log‑sum‑exp surrogate.
  The λ‑schedule starts rather large (≈0.4) and is decreased down to
  5e‑5.  The best vector found so far is retained.

* **Phase 2 – Adaptive exponentiated‑gradient descent** – the true
  `max`‑convolution objective is tackled directly.  The exact sub‑gradient
  of the maximal convolution entry is obtained via FFT, and an
  exponentiated‑gradient (mirror‑SGD on the probability simplex) with an
  adaptive step size and a tiny back‑tracking line‑search is run until
  the time budget expires.

* **Phase 3 – Greedy local mass‑transfer** – while time permits we
  repeatedly move a small fraction of mass from a high‑mass index to a
  random other index; the move is kept only if the `fast_score` improves.
  This simple hill‑climb can still squeeze a few percent out of the
  solution.

* **Phase 4 – Final low‑λ L‑BFGS polish** – a short L‑BFGS run with an
  even finer λ‑schedule (down to 1e‑6) finishes the optimisation.

All steps respect the overall 1000 s budget.  The best vector found is
stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats.  The official scorer `evaluate_sequence` is called
once at the very end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Exact value of the official score when sum(a)==1:
        2·n·max(conv(a,a))
    using an FFT for the convolution.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def generate_start_vectors(n: int, rng: np.random.Generator):
    """Build a heterogeneous pool of probability vectors."""
    pool = []

    # 1) uniform
    pool.append(np.full(n, 1.0 / n))

    # 2) beta‑like: (i+1)^α (n-i)^β
    for _ in range(8):
        α = rng.uniform(0.2, 5.0)
        β = rng.uniform(0.2, 5.0)
        i = np.arange(1, n + 1, dtype=float)
        w = (i ** α) * ((n + 1 - i) ** β)
        w /= w.sum()
        pool.append(w)

    # 3) pure exponential decay
    for rate in rng.uniform(0.0005, 0.02, size=6):
        w = np.exp(-rate * np.arange(n))
        w /= w.sum()
        pool.append(w)

    # 4) power‑law
    for p in rng.uniform(0.1, 3.0, size=6):
        w = (np.arange(1, n + 1, dtype=float) ** (-p))
        w /= w.sum()
        pool.append(w)

    # 5) symmetric double‑exponential (mass on both ends)
    for rate in rng.uniform(0.001, 0.02, size=4):
        half = np.exp(-rate * np.arange(n // 2 + 1))
        if n % 2 == 0:
            w = np.concatenate([half, half[-2::-1]])
        else:
            w = np.concatenate([half, half[-1:0:-1]])
        w = w[:n]
        w /= w.sum()
        pool.append(w)

    # 6) random vectors (already fairly diverse)
    for _ in range(5):
        w = rng.random(n)
        w /= w.sum()
        pool.append(w)

    # 7) interpolate previous best (if available)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            w = np.interp(x_new, x_old, old)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

    return pool

# Smoothed log‑sum‑exp surrogate (used by L‑BFGS)

def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns smoothed max‑convolution value and its gradient w.r.t. the
    log‑variables y (a = softmax(y)).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                                 # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]               # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Runs a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase together with its fast_score.
    """
    y = np.log(a_start + 1e-12)                     # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:          # safety net – continue with next λ
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def maxconv_subgradient(a: np.ndarray, n: int, fft_len: int):
    """
    Returns (conv_array, max_val, subgradient_vector, mask) where
    subgradient_vector = ∂max_k (a*a)[k] / ∂a.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(float)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]   # length n
    return conv, max_val, g, mask

def exp_gradient_refine(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       eta0: float = 15.0):
    """
    Mirror‑SGD on the exact max‑convolution with a tiny back‑tracking line–search.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len)
    best_a = a.copy()
    best_score = cur_score
    it = 0

    while time.time() < deadline:
        _, _, g, _ = maxconv_subgradient(a, n, fft_len)

        # basic decreasing schedule, but we will also try a few smaller trial steps
        eta = eta0 / np.sqrt(it + 1.0)
        improved = False

        for factor in (1.0, 0.5, 0.25, 0.125):
            eta_try = eta * factor
            exp_term = np.exp(np.clip(-eta_try * g, -50, 50))
            a_new = a * exp_term
            s = a_new.sum()
            if s == 0.0:
                continue
            a_new /= s

            new_score = fast_score(a_new, fft_len)
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                improved = True
                break   # accept first successful shrink

        if not improved:
            # optional tiny random jitter to escape flat regions
            if rng.random() < 0.01:
                a = add_noise(a, sigma=rng.uniform(5e-4, 2e-3), rng=rng)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score

def greedy_local_refine(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       move_frac: float = 0.15):
    """
    Randomly move a small fraction of mass from a source index to a destination.
    Keep the move only if it reduces the objective.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len)
    best_a = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        # source chosen with probability proportional to its mass
        src = rng.choice(n, p=a / a.sum())
        if a[src] <= 1e-15:

        dst = rng.integers(0, n)
        if dst == src:

        delta = a[src] * move_frac * rng.random()
        if delta <= 0.0:

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        # sum unchanged, all entries stay non‑negative by construction
        new_score = fast_score(cand, fft_len)

        if new_score < cur_score:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    # Store globally for the next run
    globals()['height_sequence_1'] = final_seq
    return final_seq
