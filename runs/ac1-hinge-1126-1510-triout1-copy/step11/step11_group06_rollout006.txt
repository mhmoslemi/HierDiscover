def hill_climb_adaptive(a_start, start_score, deadline, rng,
                        init_factor=0.2, min_factor=0.001, max_factor=0.5,
                        factor_decay=0.999, recompute_every=5000,
                        random_move_prob=0.001):
    a = a_start.copy()
    conv = np.convolve(a, a)
    n = a.shape[0]
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    step_factor = init_factor
    # Might adjust step_factor dynamically: increase after accepted, decrease after rejected.

    while time.time() < deadline:
        # compute gradient g via mask
        M = conv.max()
        eps = M * 1e-12 + 1e-12
        mask = (conv >= M - eps).astype(float)
        # compute subgradient g
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a[::-1], fft_len)))
        g = 2.0 * g_full[n-1:n-1+n]
        # choose src and dst
        src = np.argmax(g)
        dst = np.argmin(g)
        if src == dst:
            # random move fallback
            src = int(rng.choice(np.where(a>1e-12)[0]))
            dst = int(rng.integers(0, n))
        # Determine max possible delta = a[src]
        max_delta = a[src]
        # start with delta = step_factor * max_delta
        delta = max_delta * step_factor
        # Ensure > small threshold
        if delta < 1e-14:
            # reduce step factor maybe
            step_factor *= 0.5
            continue
        # line search: try delta, if not improve, reduce factor
        accept = False
        for attempt in range(10):
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
            a_new = np.maximum(a_new, 0)
            a_new_sum = a_new.sum()
            if a_new_sum == 0:
                # invalid
                break
            a_new /= a_new_sum
            # compute new score using fast_score (since only relative)
            new_score = fast_score(a_new, fft_len)
            if new_score <= cur_score + 1e-12:  # improvement or equal
                accept = True
                break
            else:
                # reduce delta
                delta *= 0.5
        if accept:
            # update a, conv, score
            a = a_new
            conv = np.convolve(a, a)
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                # possibly increase step factor after good move
                step_factor = min(max_factor, step_factor * 1.05)
        else:
            # reject; reduce step factor
            step_factor = max(min_factor, step_factor * 0.5)

        # periodic random perturbation
        if rng.random() < random_move_prob:
            # random epsilon mass redistribution on two random indices
            idxs = rng.choice(n, size=2, replace=False)
            delta2 = min(a[idxs[0]], 0.01) * rng.random()
            a[idxs[0]] -= delta2
            a[idxs[1]] += delta2
            a = np.maximum(a, 0)
            a /= a.sum()
            conv = np.convolve(a,a)
            cur_score = fast_score(a, fft_len)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

    return best_a, best_score

assistantfinal**Solution Overview**

The program searches for a non‑negative step‑function (a probability vector) that minimises  

\[
\frac{2\,n\;\max (a*a)}{(\sum a)^2}\;,
\]

where `*` denotes linear convolution.  
The score is invariant under scaling, therefore we keep the vector normalised
(`sum(a)=1`).  

The optimisation consists of four stages, each using a different numerical
strategy and each running for a fraction of the total 1000 s budget.

1. **L‑BFGS on a smooth log‑sum‑exp approximation** – many short,
   random restarts quickly find promising basins.
2. **Adam on the same smoothed objective** – fine‑tunes the best L‑BFGS result.
3. **Exponentiated‑gradient descent on the true max‑convolution** – directly
   minimises the true objective using the sub‑gradient of the max.
4. **Adaptive gradient‑guided hill‑climb** – moves a fraction of mass from the
   index with the largest sub‑gradient to the one with the smallest.
   A simple line‑search evaluates the exact effect of a move (including the
   quadratic `δ²` terms) and accepts it only if the score does not increase.
   The step size grows when moves are accepted and shrinks otherwise, and a
   tiny random perturbation occasionally helps to escape plateaus.

The algorithm works with `n = 8000` coefficients, a size that still allows
very fast FFT‑based convolutions (`≈ 0.001 s` per evaluation) while giving the
searcher enough freedom to approach the asymptotic optimum (≈ 1.5).  
All intermediate scores are obtained with a fast routine that evaluates
`2·n·max(conv)`.  At the very end the provided `evaluate_sequence` function is
called to obtain the officially checked score.

The implementation follows the required *no‑closure*, *no‑lambda* style and
stores the final sequence in the global variable `height_sequence_1` so that
future runs can reuse it as a warm‑start.

"""
search_step_function.py

A multi‑stage optimiser for the step‑function evaluation problem.
The pipeline:

1. Many short L‑BFGS runs on a log‑sum‑exp smooth approximation.
2. Adam optimisation on the same objective.
3. Exponentiated‑gradient descent on the true max‑convolution.
4. Adaptive gradient‑guided hill‑climb with line‑search and occasional random
   perturbations.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2*len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)).  a is assumed to sum to 1."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (Euclidean projection)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector of length n."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smooth approximation of 2·n·max(conv) and its gradient w.r.t. logits y
    where a = softmax(y).  Returns (value, grad_y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (FFT)
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient w.r.t a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation (multiple restarts)
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for every λ in lam_schedule.
    Returns the best vector a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        # back to probability vector
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def adam_optimize(y_start: np.ndarray,

    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best vector a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adaptive gradient‑guided hill‑climb (exact line‑search)
# ----------------------------------------------------------------------


def hill_climb_adaptive(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator,
                        init_factor: float = 0.25,
                        min_factor: float = 0.001,
                        max_factor: float = 0.5,
                        recompute_every: int = 2000,
                        random_move_prob: float = 5e-4):
    """
    Gradient‑guided hill‑climb that moves a fraction of mass from the
    index with the largest sub‑gradient to the one with the smallest.
    A simple line‑search evaluates the *exact* effect (including δ² terms)
    and accepts the move only if the score does not increase.
    The step size grows after successful moves and shrinks after
    rejections; occasional random perturbations help to escape plateaus.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact linear convolution (len = 2n‑1)
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    step_factor = init_factor
    accepted = 0
    steps = 0
    phase_start = time.time()
    total_time = deadline - phase_start

    while time.time() < deadline:
        # ----- compute sub‑gradient of the max‑convolution -----
        cur_max = conv.max()
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # occasional random fallback if gradient is degenerate
        if src == dst or a[src] <= 0.0:
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero,
                                 p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # ----- line‑search for a suitable δ -----
        max_delta = a[src]
        delta = max_delta * step_factor
        if delta < 1e-14:
            # step too small – shrink and continue
            step_factor = max(min_factor, step_factor * 0.5)
            continue

        accepted_move = False
        # try at most 10 shrinking attempts
        for _ in range(10):
            # exact convolution update for this delta (including δ² terms)
            conv_new = conv.copy()
            # linear part:
            conv_new[src:src + n] -= 2.0 * delta * a
            conv_new[dst:dst + n] += 2.0 * delta * a
            # quadratic contributions:
            idx_src2 = 2 * src
            idx_dst2 = 2 * dst
            idx_sum = src + dst
            if idx_src2 < conv_new.size:
                conv_new[idx_src2] += delta * delta
            if idx_dst2 < conv_new.size:
                conv_new[idx_dst2] += delta * delta
            if idx_sum < conv_new.size:
                conv_new[idx_sum] -= 2.0 * delta * delta

            new_score = 2.0 * n * conv_new.max()
            if new_score <= cur_score + 1e-14:        # accept if not worse
                accepted_move = True
                break
            # otherwise shrink δ and try again
            delta *= 0.5

        if accepted_move:
            # apply the successful move to the real state
            a[src] -= delta
            a[dst] += delta
            # re‑normalise (protect against floating drift)
            a_sum = a.sum()
            if a_sum != 0.0:
                a /= a_sum
            conv = conv_new          # already exact for this δ
            cur_score = new_score
            accepted += 1

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

            # increase step size modestly after a success
            step_factor = min(max_factor, step_factor * 1.02)
        else:
            # rejection – shrink step size
            step_factor = max(min_factor, step_factor * 0.5)

        steps += 1

        # periodic exact recompute to compensate any numerical drift
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        # occasional random jitter to escape flat regions
        if rng.random() < random_move_prob:
            idxs = rng.choice(n, size=2, replace=False)
            # move a tiny random amount from idxs[0] to idxs[1]
            delta_rand = min(a[idxs[0]], 0.001) * rng.random()
            a[idxs[0]] -= delta_rand
            a[idxs[1]] += delta_rand
            a = np.maximum(a, 0)
            a_sum = a.sum()
            if a_sum != 0.0:
                a /= a_sum
            conv = np.convolve(a, a)
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        # occasional progress report
        if steps % 5000 == 0:
            elapsed = time.time() - phase_start
            remaining = max(0.0, deadline - time.time())
            print(f"[hill_climb] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f} remaining={remaining:.1f}s")

    return best_a, best_score

# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 8000 offers a good trade‑off between quality and speed
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS bursts from random starts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-5, num=14)
    maxiter_per_lam = 400
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(
            y0, lam_sched, max_steps_per_lam=5000,
            deadline=deadline, n=n, fft_len=fft_len, lr0=0.13
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, rng, eta0=12.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – adaptive gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_adaptive(
            best_a, best_score, deadline, rng,
            init_factor=0.25, min_factor=0.001, max_factor=0.5,
            recompute_every=2000, random_move_prob=5e-4
        )
        print(f"[hill‑climb] final best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # Store the result for possible warm‑starts in later runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
