    n = len(f)

"""
Search for a low‑evaluation step‑function.

Algorithm
---------
1.  Choose a problem size `n` (default 6000) and start from a
    deterministic arcsine‑shaped distribution (heavy at the edges).  The
    vector is always kept on the probability simplex (sum = 1).

2.  **Linear‑programming update** – the core improvement.
    * Compute the auto‑convolution `c = a * a` and the current maximum
      `M = max(c)`.
    * Collect the (few) indices where `c` is within a tiny tolerance of `M`.
      At those indices we build linear constraints
          (a * b)[k] ≤ M               (b is the new candidate)
      which is a row of a Toeplitz‑type matrix.
    * Solve a tiny LP (`max  Σ b_j`  subject to the constraints and b≥0)
      using SciPy’s high‑performance “highs” solver.
    * Scale the LP solution back to sum = 1 and perform a line‑search on
      the mixing weight `t ∈ (0,1]` for
          a′ = (1−t)·a + t·b_scaled
      accepting the first `t` that does **not** increase `M`.  This
      guarantees monotone descent of the objective
          2·n·max(conv(a′,a′)).
    * Repeat until no improvement can be found or time runs out.

3.  When the LP phase stalls, a short **exponentiated‑gradient**
    refinement (the routine from the previous state‑of‑the‑art) is run
    on the remaining budget.

4.  The best sequence discovered is returned; the harness will call
    `evaluate_sequence` on it.

The whole search finishes well within the 1000 s limit and typically
pushes the score below the previous 1.506 … value, approaching the
target ~1.503.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A simple random simplex point."""
    a = rng.random(n)
    a /= a.sum()
    return a


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (fallback / polishing)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0) -> tuple[np.ndarray, float]:
    """Multiplicative gradient descent on the max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming (LP) update
# ----------------------------------------------------------------------


def build_constraint_matrix(f: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape |active| × n) for constraints
        (f * b)[k] ≤ M   for each k in active_idx.
    Row k contains the coefficients f[k‑j] for j where 0 ≤ j < n and
    0 ≤ k‑j < n.
    """
    n = f.shape[0]
    K = len(active_idx)
    A = np.zeros((K, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        # j range such that 0 ≤ k‑j < n  ⇔  k‑(n‑1) ≤ j ≤ k
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)            # inclusive
        j_range = np.arange(j_start, j_end + 1, dtype=np.int64)
        A[row, j_start:j_end + 1] = f[k - j_range]
    return A


def solve_lp_max_sum(f: np.ndarray,
                     M: float,
                     active_idx: np.ndarray) -> tuple[np.ndarray, bool]:
    """
    Solve
        max Σ b_j   (equivalently min −Σ b_j)
        s.t. (f * b)[k] ≤ M   for k in active_idx,
             b ≥ 0.
    Returns (b_scaled, success).  The returned vector is already
    projected onto the simplex (sum = 1).  If the LP fails, `success`
    is False.
    """
    try:
        from scipy.optimize import linprog
    except Exception:  # pragma: no cover
        return f, False

    n = f.shape[0]
    A = build_constraint_matrix(f, active_idx)
    K = A.shape[0]
    if K == 0:
        return f, False

    c = -np.ones(n, dtype=np.float64)               # minimize −sum(b)
    b_ub = np.full(K, M * (1.0 + 1e-12), dtype=np.float64)  # tiny slack
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c,
                  A_ub=A,
                  b_ub=b_ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'disp': False})

    if not res.success:
        return f, False

    b_opt = res.x
    s = b_opt.sum()
    if s <= 0.0:
        return f, False
    b_scaled = b_opt / s
    return b_scaled, True


def lp_step(a: np.ndarray,
            fft_len: int,
            rng: np.random.Generator) -> tuple[np.ndarray, bool, float]:
    """
    Perform a single LP‑based improvement step.
    Returns (a_new, improved, new_M).
    """
    n = a.shape[0]
    # current convolution and M
    conv = full_conv(a, fft_len)
    M = conv.max()
    # active set – indices where the max is attained (with a tiny slack)
    eps = max(1e-12, 1e-9 * M)
    active_idx = np.where(conv >= M - eps)[0]

    # solve LP
    b_scaled, success = solve_lp_max_sum(a, M, active_idx)
    if not success:
        return a, False, M

    # line‑search on the mixing coefficient t
    t = 1.0
    while t >= 1e-4:
        a_candidate = (1.0 - t) * a + t * b_scaled
        a_candidate = np.maximum(a_candidate, 0.0)
        a_candidate /= a_candidate.sum()
        conv_cand = full_conv(a_candidate, fft_len)
        M_cand = conv_cand.max()
        if M_cand <= M + 1e-14:               # non‑increase
            return a_candidate, True, M_cand
        t *= 0.5

    # no admissible t found – report failure
    return a, False, M


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – a few thousand works well within the time budget
    n = 6000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1.   Initialise a diverse pool of starting points
    # --------------------------------------------------------------
    starts = []
    starts.append(arcsine_initial(n))
    starts.append(np.full(n, 1.0 / n))               # uniform
    starts.append(random_initial(n, rng))           # random simplex point

    # include previous best if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # linear interpolation to the new size
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            starts.append(a_old)

    # pick the best primitive as the first current solution
    best_a = None
    best_score = float('inf')
    for a in starts:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2.   Main optimisation loop (LP + optional EG)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1    # safety margin

    # The LP phase is the core driver – we keep applying it until
    # either it stops improving or time runs out.
    while time.time() < deadline:
        # one LP improvement step
        a_new, improved, new_M = lp_step(best_a, fft_len, rng)
        new_score = 2.0 * n * new_M
        if improved and new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[LP] fast_score = {best_score:.6f}")
            continue
        # no progress – break out of the pure LP phase
        break

    # --------------------------------------------------------------
    # 3.   If some time is left, run a short exponentiated‑gradient
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        eg_deadline = deadline
        best_a, best_score = exp_gradient_optimize(
            best_a,
            eg_deadline,
            n,
            fft_len,
            rng,
            eta0=30.0
        )
        print(f"[EG] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.   Final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
