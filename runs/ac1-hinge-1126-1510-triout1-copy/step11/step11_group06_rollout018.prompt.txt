def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_optimize(a_start: np.ndarray,

def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):

    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:

    return best_a, best_score

def guided_hill_climb(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      start_factor: float = 0.20,
                      end_factor: float = 0.001,
                      recompute_every: int = 2000):
    """
    Gradient‑guided hill‑climb that moves a fraction of mass from the index
    with the largest sub‑gradient (most responsible for the current peak) to
    the index with the smallest sub‑gradient.  The step size shrinks linearly
    with the remaining time.  The convolution is updated incrementally; the
    exact convolution is recomputed periodically to avoid drift.
    Returns the best a and its fast_score found before the deadline.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact linear convolution, length 2n‑1
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    phase_start = time.time()
    total = deadline - phase_start
    accepted = 0
    steps = 0

    while time.time() < deadline:
        # current max and mask
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]   # length n

        # indices to move mass between
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback random move when gradient information is degenerate
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero,
                                 p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        if delta > a[src]:
            delta = a[src]

        # keep copies for possible rejection
        a_old = a.copy()
        conv_old = conv.copy()

        # incremental convolution update (linear term only)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[dst:dst + n] += 2.0 * delta * a_old

        # move the mass
        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score:
            # accept
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject – revert to previous state
            a = a_old
            conv = conv_old

        steps += 1

        # periodic exact recompute to counteract linear‑approximation drift
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        # occasional progress report
        if steps % 10000 == 0:
            elapsed = time.time() - phase_start
            print(f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f}")

    return best_a, best_score
