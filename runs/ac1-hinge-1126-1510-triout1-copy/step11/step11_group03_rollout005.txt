"""
Improved optimiser for the step–function L∞‑convolution problem.

Algorithm
---------
1. Build a diverse pool of starting vectors (random styles, classic windows,
   Gaussian / cosine‑power families, edge‑heavy, Tukey, Blackman‑Harris…).
   All candidates are projected onto the probability simplex.

2. **Aggressive L‑BFGS cascade** – minimise a log‑sum‑exp smooth
   approximation of the true max‑convolution.  A long λ‑schedule (down to
   5·10⁻⁷) and a larger per‑λ iteration budget allow the optimiser to
   converge much closer to the genuine ‖a*a‖∞.

3. **Adam on the smooth surrogate** – the soft‑max parametrisation
   `a = softmax(y)` is optimiser with Adam (adaptive moments).  The λ‑schedule
   is continued inside Adam, giving a high‑precision refinement.

4. **Exact sub‑gradient (mirrored EG)** – use the true max‑convolution
   sub‑gradient with a decreasing 1/√t stepsize to push the solution to
   a (near) stationary point of the original nonsmooth objective.

5. **Short Metropolis anneal** – local mass‑transfer moves (global,
   neighbour, swap) accept downhill changes and occasionally uphill
   changes to escape tiny plateaus.

The whole pipeline respects the 1000 s budget, prints intermediate scores

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Return 2·n·max(conv(a,a)) for a that is already normalised to sum = 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector from a diverse set of styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal',
         'sparse', 'triangular', 'tukey']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'triangular':
        a = 1.0 - np.abs(np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0)
    elif mode == 'tukey':
        if tukey is not None:
            a = tukey(n, alpha=rng.uniform(0.2, 0.8))
        else:
            a = rng.random(n)
    else:   # fallback
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian‑shaped density."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power (non‑negative)."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass at the two edges, light uniform interior."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


def window_initial(name: str, n: int, param) -> np.ndarray:
    """Create a window of given name (if the SciPy implementation exists)."""
    if name == 'chebwin' and chebwin is not None:
        w = chebwin(n, at=param)
    elif name == 'kaiser' and kaiser is not None:
        w = kaiser(n, beta=param)
    elif name == 'hamming' and hamming is not None:
        w = hamming(n, sym=False)
    elif name == 'hann' and hann is not None:
        w = hann(n, sym=False)
    elif name == 'blackman' and blackman is not None:
        w = blackman(n, sym=False)
    elif name == 'bartlett' and bartlett is not None:
        w = bartlett(n, sym=False)
    elif name == 'tukey' and tukey is not None:
        w = tukey(n, alpha=param)
    else:
        return None
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS & Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    Parameterisation: a = softmax(y), sum(a)=1.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)           # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z                               # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]              # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS cascade (smooth objective)
# ----------------------------------------------------------------------


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best vector seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # log‑space, respects positivity
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smooth surrogate
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_sched,
                  iterations: int,
                  deadline: float,
                  fft_len: int,
                  n: int,
                  lr: float = 0.10):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    The λ‑schedule is traversed linearly; each λ gets a fixed number of steps.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    total_lam = len(lam_sched)
    it_per_lam = max(1, iterations // total_lam)

    for lam in lam_sched:
        for _ in range(it_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            f, grad = objective_and_grad(y, n, fft_len, lam)
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            y = y - step
            it += 1

            # occasional evaluation
            if it % 20 == 0:
                a = np.exp(y - np.max(y))
                a = a / a.sum()
                sc = fast_score(a, fft_len)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"[Adam λ={lam:.1e}] fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirrored exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored SGD (exponentiated‑gradient) using the exact L∞‑subgradient.
    Runs until the deadline is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types.
    Temperature linearly cools from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25  # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # 'transfer'
                j = rng.integers(0, n)
            if i == j:
                continue

            # direction: move from larger entry to smaller
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue

            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate size balances granularity and speed
    # --------------------------------------------------------------
    n = 5000                     # a slightly larger size than the previous run
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # include the previous best if present (interpolated to the new size)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
            pool.append(a0)

    # classic windows (if SciPy available)
    if chebwin is not None:
        for att in (30, 50, 80, 100):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    if kaiser is not None:
        for beta in (4, 8, 12, 16, 20):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    # additional standard windows
    for win_name, param in [
        ('hamming', None), ('hann', None), ('blackman', None),
        ('bartlett', None), ('tukey', 0.4)
    ]:
        w = window_initial(win_name, n, param)
        if w is not None:
            pool.append(w)

    # random families
    for _ in range(8):
        pool.append(random_initial(n, rng))

    # parametric families
    for _ in range(5):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05        # small safety margin
    phase1_end = start_time + 0.35 * budget_s      # L‑BFGS aggressive phase
    phase2_end = start_time + 0.70 * budget_s      # Adam / smooth refinement
    # remaining time will be used for EG + anneal

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on a smoothed objective
    # --------------------------------------------------------------
    # a finer λ‑schedule, ending at 5·10⁻⁷
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5,
        1e-5, 5e-6, 2e-6, 1e-6, 5e-7
    ]
    maxiter_per_lam = 900   # a little more than the previous 600

    best_a = None
    best_score = float('inf')

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.35:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.012),
                                rng=rng)
        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best fast_score = {best_score:.9f}")

    if best_a is None:                     # fallback – uniform vector
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed surrogate
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        # keep the same λ‑schedule, allocate a modest number of iterations
        adam_iters = 8000
        best_a, best_score = adam_optimize(best_a,
                                            lam_schedule,
                                            adam_iters,
                                            phase2_end,
                                            fft_len,
                                            n,
                                            lr=0.12)
        print(f"[Adam] best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – long exponentiated‑gradient descent (exact max)
    # --------------------------------------------------------------
    if time.time() < deadline - 70.0:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            eta0=30.0
        )
        print(f"[exp‑grad] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – short simulated annealing for the last bits
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer and expose the result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
