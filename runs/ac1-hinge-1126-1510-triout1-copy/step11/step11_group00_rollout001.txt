def random_param_init(n, rng):
    mode = rng.choice(['gaussian', 'exponential', 'power', 'sinc', 'cosine_pow'])
    if mode == 'gaussian':
        sigma = rng.uniform(0.05*n, 0.3*n)
        i = np.arange(n)
        centre = (n-1)/2.0
        a = np.exp(-0.5 * ((i-centre)/sigma)**2)
    elif mode == 'exponential':
        decay = rng.uniform(0.001, 0.05)
        a = np.exp(-decay * np.arange(n))
        # optionally also mirror for symmetry:
        a = (a + a[::-1]) / 2
    elif mode == 'power':
        p = rng.uniform(0.2, 3.0)
        a = (np.arange(1,n+1)**(-p))
        a = (a + a[::-1]) / 2
    elif mode == 'sinc':
        # use normalized sinc function, width param
        width = rng.uniform(0.5, 5.0)   # cycles in length n
        i = np.arange(-n//2, n//2 + (n%2))

"""
Hybrid optimizer for the step‑function minimisation problem.

The algorithm proceeds in several stages:

1. **Diversified initial pool** – deterministic windows (Hann, Blackman,
   Gaussian, Kaiser, DPSS) together with many random parametric families
   (Gaussian, exponential, power‑law, sinc, cosine‑power).  The best
   candidate on the exact objective (fast_score) is kept.

2. **Coarse L‑BFGS refinement** – the current best is blended with a fresh
   random vector and optimised with L‑BFGS on a smoothed max‑convolution
   (log‑sum‑exp).  A schedule of decreasing smoothing parameters is used.
   This phase occupies ~35 % of the time budget.

3. **Exact Adam optimisation** – Adam on the exact max‑convolution
   (gradient obtained via FFT) is run for the bulk of the remaining time.

4. **Fine L‑BFGS** – after Adam a short L‑BFGS run with very small
   smoothing parameters finishes the optimisation.

5. **Mass‑transfer simulated annealing** – a long annealing phase with
   many random mass‑transfer moves explores the neighbourhood of the
   final vector and often yields a small additional improvement.

All operations are performed on the probability simplex (sum = 1); the
official `evaluate_sequence` rescales internally, so the returned list can
be used directly.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.signal import windows  # for DPSS windows


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    return e / s if s != 0 else np.full_like(y, 1.0 / y.size)


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective (sum = 1 ⇒ g(a)=2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum = 1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective and gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    return a / a.sum()


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def dpss_init(n: int, NW: float = 4.0) -> np.ndarray:
    """First DPSS (Slepian) sequence, normalised to sum = 1."""
    # windows.dpss returns an array of shape (Kmax, n) when Kmax>1.
    a = windows.dpss(n, NW, Kmax=1, sym=False).ravel()
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex (simple families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def random_param_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly sampled parametric families (Gaussian, exponential, power, sinc, cosine‑pow)."""
    mode = rng.choice(['gaussian', 'exponential', 'power', 'sinc', 'cosine_pow'])
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    if mode == 'gaussian':
        sigma = rng.uniform(0.05 * n, 0.30 * n)
        a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    elif mode == 'exponential':
        rate = rng.uniform(0.001, 0.04)
        a = np.exp(-rate * i)
        a = (a + a[::-1]) / 2.0  # make it roughly symmetric
    elif mode == 'power':
        p = rng.uniform(0.2, 4.0)
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = (a + a[::-1]) / 2.0
    elif mode == 'sinc':
        # width controls number of lobes; use a positive envelope
        width = rng.uniform(0.5, 4.0)
        x = (i - centre) / (n / width)
        a = np.sinc(x)  # sin(pi x)/(pi x)
        a = np.abs(a)
    else:   # cosine_pow
        p = rng.uniform(0.5, 3.0)
        # cos varies from –1 to 1; take absolute value then raise to p
        a = np.abs(np.cos(np.pi * (i - centre) / (n - 1))) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y so that a = softmax(y) stays on the simplex.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        # mask of points attaining (or numerically close to) the max
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient of max conv w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # gradient of the full objective (2·n·max_conv)
        grad_a = 2.0 * n * g_a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # decaying learning rate (a little more aggressive than the baseline)
        lr = 0.3 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to leave flat plateaux
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 250000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    total_moves = max_moves

    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        # transfer a fraction of mass from i to j
        delta = a[i] * rng.random() * 0.12  # a slightly smaller step than before
        if delta <= 0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (larger n gives more freedom but costs more time)
    # --------------------------------------------------------------
    n = 4096                     # keep a power‑of‑two size for FFT efficiency
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # --------------------------------------------------------------
    candidates = []

    # previous solution, if any
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.28))
    candidates.append(kaiser_init(n, beta=6.0))
    candidates.append(dpss_init(n, NW=3.5))
    candidates.append(dpss_init(n, NW=5.0))

    # randomised families (plain)
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # randomised parametric families
    for _ in range(6):
        candidates.append(random_param_init(n, rng))

    # --------------------------------------------------------------
    # Pick the best from the pool (fast_score == exact objective for sum=1)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    total_budget = budget_s
    # safety margin of half a second
    deadline = start_time + total_budget - 0.5

    # Phase 1 – Coarse L‑BFGS (≈35 % of time)
    phase1_end = start_time + 0.35 * total_budget
    lam_schedule_coarse = [0.3, 0.2, 0.1, 0.05, 0.02,
                           0.01, 0.005, 0.002, 0.001, 5e-4,
                           2e-4, 1e-4]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        rand_vec = random_param_init(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule_coarse,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑coarse] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Exact Adam optimisation (≈40 % of time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_score = adam_optimize(best_a,
                                           deadline,
                                           n,
                                           fft_len,
                                           rng)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – Fine L‑BFGS with very small smoothing (≈15 % of time)
    # --------------------------------------------------------------
    phase3_end = time.time() + 0.15 * total_budget
    lam_schedule_fine = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    maxiter_fine = 2000

    if time.time() < phase3_end:
        a_fine, sc_fine = lbfgs_continuation(best_a,
                                             lam_schedule_fine,
                                             maxiter_fine,
                                             phase3_end,
                                             fft_len, fft_len,
                                             n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[L‑BFGS‑fine] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 4 – Simulated annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print(f"[SA] start final annealing, remaining ≈{remaining:.1f}s")
        best_a, best_score = simulated_annealing(best_a,
                                                 deadline,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 max_moves=200000)
        print(f"[SA] final fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score for reporting
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
