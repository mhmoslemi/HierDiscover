"""
Optimiser for the step‑function evaluation problem.

The algorithm combines:
  * an extensive parametric sweep (exponential, power, logistic, sinusoid,
    quadratic, cosine‑bell etc.),
  * many random Dirichlet restarts optimised by a continuation L‑BFGS on a
    log‑sum‑exp smoothed max,
  * projected Adam on the true max‑convolution (sub‑gradient),
  * a rich simulated‑annealing phase with several move types,
  * a final ultra‑fine L‑BFGS polish.

All phases respect a global deadline (default 1000 s).  The best sequence
found before the deadline is returned.  The code works with only numpy /
scipy and does not perform any I/O.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the official evaluation (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (logits → probabilities)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a (linear, via FFT)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                # d smooth_max / d conv_k
    # back‑propagation through the convolution a*a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the part that contributes to a (indices n‑1 … 2n‑2)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax (y → a)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits – stay in the simplex
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)

        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best fast_score={best_sc:.6f}")

    return best_a, best_sc


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the true sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.4

    while time.time() < deadline:
        it += 1
        # true convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                 # any convex combination is a valid sub‑gradient

        # sub‑gradient of the max w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam step on the raw vector
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_cand = a - step
        a_cand = simplex_projection(a_cand)

        cand_sc = fast_score(a_cand, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, a_cand.copy()
        a = a_cand

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with square‑root cooling.
    Moves:
      – transfer a random fraction of mass i → j,
      – swap two indices,
      – redistribute among three positions,
      – smooth‑move (local averaging).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(4)
        a_new = a.copy()

        if move_type == 0:                 # transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:               # swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        elif move_type == 2:               # triple redistribution
            idx = rng.choice(n, size=3, replace=False)
            tot = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = tot * weights

        else:                              # smoothing move
            i = rng.integers(1, n - 1)
            # replace a[i] by a weighted average of its neighbours
            avg = 0.5 * (a_new[i - 1] + a_new[i + 1])
            # keep total sum invariant by adjusting neighbours
            delta = avg - a_new[i]
            a_new[i] = avg
            # distribute the opposite delta to neighbours proportionally
            a_new[i - 1] -= delta * 0.5
            a_new[i + 1] -= delta * 0.5
            # ensure feasibility
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, sigma: float, mu: float = None) -> np.ndarray:
    i = np.arange(n, dtype=float)
    if mu is None:
        mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_cosine_bell(n: int, width: float) -> np.ndarray:
    """Raised cosine on a window of given half‑width."""
    i = np.arange(n, dtype=float) - (n - 1) / 2.0
    a = np.cos(np.pi * i / width)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_sin_power(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=float) + 0.5
    a = np.sin(np.pi * i / n) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_quadratic(n: int, scale: float) -> np.ndarray:
    """Parabolic bump: 1 - ( (i - c) / scale )^2, clipped at 0."""
    i = np.arange(n, dtype=float)
    c = (n - 1) / 2.0
    a = 1.0 - ((i - c) / scale) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_double_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a

        if sc < best_sc:

            print(f"[grid] gauss sigma={sigma:.0f} → {best_sc:.6f}")

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a parametric family (used in cheap sweep)."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(300.0, 1800.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(300.0, n / 1.5)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "sin":
        p = rng.uniform(1.0, 10.0)
        a = np.sin(np.pi * (i + 0.5) / n) ** p
    elif family == "quad":
        scale = rng.uniform(0.2 * n, 0.5 * n)
        c = (n - 1) / 2.0
        a = 1.0 - ((i - c) / scale) ** 2
        a = np.maximum(a, 0.0)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    else:                                   # plain random Dirichlet
        a = rng.exponential(scale=1.0, size=n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Fast random exploration over many families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

        "bimodal",
        "logistic",
        "sin",

        "binary_sparse",
        "double_exp",

    while time.time() < deadline:

        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_sc:.6f} (family={fam})")

    return best_a, best_sc

    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 5e-07, 2e-07,
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # jitter the current best or start from a fresh random Dirichlet
        if rng.random() < 0.4:
            a0 = best_a * rng.normal(1.0, 0.025, size=n)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
        else:
            a0 = rng.exponential(scale=1.0, size=n)
            a0 /= a0.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a0,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = subgradient_adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] Adam sub‑gradient best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing with richer moves
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – ultra‑fine L‑BFGS run
    # ------------------------------------------------------------------
    if time.time() < deadline:
        tiny_sched = [1e-07, 5e-08, 2e-08, 1e-08]
        a_final, sc_final = lbfgs_continuation(
            best_a.copy(),
            tiny_sched,
            maxiter_per_lam=1500,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_sc:
            best_sc, best_a = sc_final, a_final.copy()
            print(f"[final‑lbfgs] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (the true scoring function)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation failed:", e)
        final_eval = float("inf")

    return final_seq
