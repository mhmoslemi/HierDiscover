"""
Hybrid continuation optimisation for the max‑auto‑convolution problem.

The search consists of four stages:

1.  A pool of diverse warm‑starts (old solution, random families,
    classic windows) is built and a quick cascade of L‑BFGS on a
    log‑sum‑exp smoothed surrogate drives each start into a good basin.
2.  A long accelerated mirror‑descent (exponentiated‑gradient) with
    momentum and an adaptive tolerance on the *exact* max‑convolution
    refines the best candidate from stage 1.  The step‑size decays as
    ηₜ ∝ 1/√t and a small amount of isotropic noise is injected when
    progress stalls.
3.  A final low‑λ L‑BFGS polish extracts the remaining improvement.
4.  A short simulated‑annealing hill‑climb explores the neighbourhood
    of the polished solution.

All stages respect a global 1000 s deadline.  The best sequence found
is stored in the global variable ``height_sequence_1`` and returned as a
plain list of floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when ``a`` sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a = simplex_normalise(a)
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    return simplex_normalise(noisy)


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns the smoothed max‑convolution value and its gradient w.r.t.
    the log‑variables y (so that a = softmax(y)).
    """
    # ----- softmax -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution (FFT) -----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = simplex_normalise(a)
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient of the hard max‑convolution
# ----------------------------------------------------------------------


def subgradient_max_conv(a: np.ndarray,
                         n: int,
                         fft_len: int,
                         rel_tol: float = 1e-9):
    """
    Compute a (normalized) sub‑gradient of the true max‑convolution.
    Returns gradient g (size n) and the current max value.
    """
    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * rel_tol + 1e-12

    mask = (conv >= max_val - eps).astype(np.float64)
    w_sum = mask.sum()
    if w_sum == 0.0:
        # fall back to the exact max index
        mask[np.argmax(conv)] = 1.0
        w_sum = 1.0
    mask /= w_sum                      # convex combination of active sites

    # gradient = 2 * conv(mask, rev(a))
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]
    return g, max_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) with momentum
# ----------------------------------------------------------------------


def exp_mirror_momentum_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 eta0: float = 30.0,
                                 beta: float = 0.9,
                                 rel_tol: float = 1e-9,
                                 max_iters: int = 20_000_000,
                                 noise_every: int = 2_000_000):
    """
    Accelerated exponentiated‑gradient on the true L∞‑objective.
    Uses a momentum term (EMA of sub‑gradients) and periodically injects
    a tiny isotropic noise when progress stalls.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)      # momentum buffer
    it = 0
    last_improve = 0

    while time.time() < deadline and it < max_iters:
        g, max_val = subgradient_max_conv(a, n, fft_len, rel_tol)

        # EMA of gradients
        m = beta * m + (1.0 - beta) * g

        # step size schedule (1/√t)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * m, -50, 50))
        a = a * exp_term
        a = simplex_normalise(a)

        # occasional cheap score check
        if (it & 31) == 0:          # every 32 iterations
            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it

        # stagnation detection – add a little noise
        if it - last_improve > noise_every:
            a = add_noise(a, sigma=0.001, rng=rng)
            last_improve = it

        it += 1

    return best_a, best_score

# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature decays linearly to near‑zero.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate‑large size gives a good trade‑off.
    # ------------------------------------------------------------------
    # Use a slightly larger dimension than the previous run; the
    # FFT length stays at 16384 (next power‑of‑two for 2·n‑1) for
    # n≈6000, keeping per‑iteration cost essentially unchanged.
    n = 6000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Re‑use the previous best (if any) – interpolate to new length.
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = simplex_normalise(a0)
            pool.append(a0)

    # 2) Randomly generated families
    for _ in range(14):
        pool.append(random_initial(n, rng))

    # 3) Classic windows (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w = simplex_normalise(w)
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w = simplex_normalise(w)
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.07 * budget_s       # quick L‑BFGS warm‑starts
    phase2_end = start_time + 0.78 * budget_s       # long mirror‑descent
    # remaining time → final polish + anneal

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 800

    best_a = None
    best_score = float('inf')
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional small jitter to increase diversity
        if rng.random() < 0.30:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.008), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑coarse] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long accelerated mirror‑descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_mirror_momentum_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=35.0,
            beta=0.92,
            rel_tol=1e-9,
            max_iters=15_000_000,
            noise_every=2_500_000,
        )
        print(f"[mirror‑descent] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final low‑λ L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [
            5e-5, 2.5e-5, 1.2e-5, 6e-6, 3e-6,
            1.5e-6, 7e-7, 4e-7, 2e-7
        ]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=1800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[L‑BFGS‑fine] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional short anneal (only if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
