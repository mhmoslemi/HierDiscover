def subgradient_optimize_line(a_start, deadline, n, fft_len, step_factor=1.0):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    while time.time() < deadline:
        # compute convolution and max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        idx = int(conv.argmax())
        # compute subgradient g: for i in 0..n-1, g_i = 2*a_{idx - i} if idx-i within [0,n-1] else 0
        g = np.zeros_like(a)
        # j = i, k = idx => contribution a_i * a_{idx-i}
        # So for each i, let j = idx - i
        # But contributions symmetrical: derivative wrt a_i = 2*a_{idx - i} if idx-i valid
        # Implementation: compute slice
        j_low = max(0, idx - (n-1))
        j_high = min(idx, n-1)
        # i runs from j_low to j_high
        # for i in that range, g_i = 2*a[idx - i]
        # we can assign g[j_low:j_high+1] = 2*a[idx - np.arange(j_low, j_high+1)]
        g[j_low:j_high+1] = 2.0 * a[idx - np.arange(j_low, j_high+1)]

        # projection to sum zero: d = -(g - mean(g))
        mean_g = np.mean(g)
        d = -(g - mean_g)
        # find max feasible step size
        # For indices where d_i < 0, step size limited by a_i / -d_i
        neg = d < 0
        if np.any(neg):
            max_alpha = np.min(a[neg] / (-d[neg]))
        else:
            max_alpha = np.inf
        # Choose step size as step_factor * max_alpha, but limit to some max bound
        alpha = step_factor * max_alpha
        # If step too small, break
        if alpha < 1e-12:
            # maybe break also after many attempts
            break
        # candidate
        a_new = a + alpha * d
        # due to floating error, maybe small negative
        a_new = np.maximum(a_new, 0.0)
        # re-normalize sum to 1 (should already be same sum, but rounding)
        s = a_new.sum()
        if s <= 0:
            break
        a_new /= s
        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

"""
Propose a low‑score step function.

The search consists of five phases:

1. **Initialization** – build a diverse pool of candidate sequences
   (spectral, window functions, power‑law, β‑distributions, etc.).
   The best candidate (by fast exact score) is kept.

2. **L‑BFGS polishing** – optimise a smooth approximation of the
   max‑convolution using a soft‑max parametrisation and a decreasing
   smoothing parameter λ.  Multiple random restarts are performed.

3. **Adam on the smooth objective** – a cheap first‑order optimiser
   (Adam) refines the best L‑BFGS solution further.

4. **Exact‑max sub‑gradient descent with line‑search** – the true
   max‑convolution gradient is computed; a feasible step size that
   respects the simplex constraints is taken (projected onto the
   simplex by an exact line search).

5. **Exploration & polishing** – simulated annealing and a greedy
   hill‑climb finish the search.

All phases respect a global deadline (default 1000 s).  The final
sequence is evaluated with the provided `evaluate_sequence` and
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (sum == 1 → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (∑a=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS/Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_param_init(n: int, p: float = 0.5) -> np.ndarray:
    """Generalised arcsine: a_i ∝ [x_i(1‑x_i)]^{-p}."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = (x * (1.0 - x)) ** (-p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Symmetric beta(α,α) on {1,…,n}."""
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    a = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lam_schedule):
    """
    Adam optimiser on the smooth max objective.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # inner loop per λ – a few hundred iterations are enough
        for _ in range(300):
            if time.time() >= deadline:
                break
            t += 1
            _, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam)
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = 0.5
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = softmax(y)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


def subgradient_line_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              step_factor: float = 0.95):
    """
    Exact‑max sub‑gradient descent with a feasible line‑search step.
    The direction d = -(g - mean(g)) has zero sum; we take the largest
    step α such that a + α d stays non‑negative, then project onto the
    simplex (renormalisation).  This yields much larger, safe steps.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # self‑convolution and index of the current maximum
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # sub‑gradient of conv[idx] w.r.t. a (2·a_{idx‑i} where defined)
        g = np.zeros_like(a)
        low = max(0, idx - (n - 1))
        high = min(idx, n - 1)
        # i runs from low … high
        sub_idx = np.arange(low, high + 1)
        g[sub_idx] = 2.0 * a[idx - sub_idx]

        # project to sum‑zero direction
        mean_g = g.mean()
        d = -(g - mean_g)

        # feasible step size (largest α keeping non‑negativity)
        neg = d < 0
        if np.any(neg):
            max_alpha = np.min(a[neg] / (-d[neg]))
        else:
            max_alpha = np.inf

        if max_alpha <= 0:
            break    # cannot move any further

        alpha = step_factor * max_alpha
        a_new = a + alpha * d

        # numerical cleanup
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0:
            break
        a_new /= s

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 120000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    total_moves = max_moves

    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        # move up to 15 % of the source mass
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use the same size as the previous best)
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build initial candidate pool (wide variety of families)
    # --------------------------------------------------------------
    candidates = []

    # If a previous construction is present, interpolate it.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))

    # New families
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families
    for _ in range(5):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(pow_decay_init(n, rng))

    # Beta and parametrised arcsine families
    for _ in range(4):
        alpha = rng.uniform(0.2, 2.0)
        candidates.append(beta_init(n, rng, alpha))
    for _ in range(3):
        p = rng.uniform(0.2, 0.7)
        candidates.append(arcsine_param_init(n, p))

    # Kerry windows with random beta
    for _ in range(2):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Pick the best of the pool (fast_score is exact for sum=1)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈30% of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector for diversity.
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smooth objective (≈20% time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.50 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            lam_schedule)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact‑max sub‑gradient descent with line search (≈20% time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.70 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_line_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            step_factor=0.95)
        print(f"[sub‑grad line] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (≈15% time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            init_temp=5e-4,
            final_temp=1e-7,
            max_moves=150000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=100000,
            delta_frac=0.08)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Evaluate the final result with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
