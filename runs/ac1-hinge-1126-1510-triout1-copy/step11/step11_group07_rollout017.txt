```
def exp_gradient_optimize_adagrad(a_start, deadline, n, fft_len, rng, eta0=30.0):
    a = a_start.copy()
    grad_sq = np.zeros_like(a) + 1e-12
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it=0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val*1e-12+1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        grad_conv = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a_rev, fft_len)))
        g = 2.0 * grad_conv[n-1:n-1+n]
        grad_sq += g**2
        eta = eta0 / np.sqrt(grad_sq + 1e-12)  # per-coordinate
        exp_term = np.exp(-eta * g)  # elementwise
        a *= exp_term
        a_sum = a.sum()
        if a_sum > 0:
            a /= a_sum
        else:
            a = best_a.copy()
        # maybe also clip and renormalize after each iteration
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        it += 1
    return best_a, best_score
```

```
def projected_subgradient_optimize_adaptive(a_start, deadline, n, fft_len, rng, eps_frac=0.001):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it=0
    step = 0.5
    # use diminishing step: step = step0 / sqrt(it+1) but also adjust if no improvement
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)
        grad_conv = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a[::-1], fft_len)))
        g = 2.0 * grad_conv[n-1:n-1+n]
        step = 0.5 / np.sqrt(it+1)   # base step adjustable based on experience
        # try a line search of step multipliers [1,0.5,0.25,0.125,...]
        for factor in [1.0, 0.5, 0.25, 0.125]:
            cand = a - step*factor*g
            cand = simplex_projection(cand)
            cand_score = fast_score(cand, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
            if cand_score < fast_score(a, fft_len):
                a = cand
                break
        else:
            # no improvement: random jitter
            a = a + rng.normal(scale=1e-4, size=n)
            a = np.clip(a, 0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score
```

```python
The solver:

    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

    """2·n·max(conv(a,a)) – the exact value returned by `evaluate_sequence`
    after the scale‑invariance has been taken into account."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    """Generate a random point on the simplex."""

def beta_initial(n: int, alpha: float) -> np.ndarray:
    """Beta(α,α) pdf discretised on the simplex.
    α=1 -> uniform, α=0.5 -> arcsine, α<1 -> heavy edges."""
    xs = (np.arange(n) + 0.5) / n     # 0 < xs < 1
    eps = 1e-12
    xs = np.clip(xs, eps, 1.0 - eps)

def exp_gradient_optimize_adagrad(a_start: np.ndarray,
                                  deadline: float,
                                  n: int,
                                  fft_len: int,
                                  rng: np.random.Generator,
                                  eta0: float = 30.0) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on max(conv)."""
    a = a_start.copy()
    grad_sq = np.full_like(a, 1e-12)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:

        a_rev = a[::-1]

        if s > 0.0:

        else:
            a = best_a.copy()

        if score < best_score:

            best_a = a.copy()

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

        improved = False

# Greedy hill‑climb that updates the convolution incrementally

def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb with cheap incremental convolution updates."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution, length 2n‑1

    while time.time() < deadline:
        # slowly shrink the permitted transfer fraction as we approach the deadline
        remaining = deadline - time.time()
        total = deadline - (deadline - remaining)  # just for readability
        frac = 0.02 + 0.18 * (remaining / (deadline - (deadline - remaining) + 1e-12))
        # However, the above expression is pointless; we simply use a linear schedule
        # based on elapsed time.
        elapsed = time.time() - (deadline - (deadline - start_score))
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / (deadline - start_score + 1e-12))

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (only the linear part)
        # the change in convolution caused by adding δ to dst and removing δ from src
        # is 2·δ·(a shifted‑difference vector)
        diff = np.zeros_like(conv)
        # contribution of the source index
        diff[src:src + n] -= best
        # contribution of the destination index
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score

        # from time to time recompute the exact convolution to flush accumulated rounding errors
        if np.random.rand() < 0.001:
            conv = np.convolve(best, best)

    return best, best_score

        # occasionally recompute the exact convolution to avoid drift
        if rng.random() < 0.001:
            conv = np.convolve(best, best)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 8000 offers a good balance between quality and speed
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1. Build several diverse starting distributions
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # arcsine (α = 0.5)
    init_candidates.append(arcsine_initial(n))

    # a small sweep of symmetric Beta(α,α) distributions around α=0.5
    for alpha in np.linspace(0.35, 0.65, 13):
        init_candidates.append(beta_initial(n, float(alpha)))

    # a few random simplex points
    for _ in range(3):
        init_candidates.append(random_initial(n, rng))

    # previous best (if the harness supplied one)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # interpolate onto the current size
                x_old = np.linspace(0.0, 1.0, old.size)
                x_new = np.linspace(0.0, 1.0, n)
                old = np.interp(x_new, x_old, old)
            old = np.maximum(old, 0.0)
            total = old.sum()
            if total == 0.0:
                old = np.full(n, 1.0 / n)
            else:
                old /= total
            init_candidates.append(old)

    # --------------------------------------------------------------
    # 2. Keep the best primitive candidate (cheap evaluation)
    # --------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    print(f"[init] best fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # 3. Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # tiny safety margin

    # --------------------------------------------------------------
    # 4. Exponentiated‑gradient (AdaGrad) – roughly 35 % of remaining time
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.35 * (deadline - time.time())
    best_a, best_score = exp_gradient_optimize_adagrad(
        best_a, phase1_deadline, n, fft_len, rng, eta0=30.0
    )
    print(f"[exp‑grad] fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # 5. Projected sub‑gradient – roughly 45 % of remaining time
    # --------------------------------------------------------------
    phase2_deadline = time.time() + 0.45 * (deadline - time.time())
    best_a, best_score = projected_subgradient_optimize(
        best_a, phase2_deadline, n, fft_len, rng,
        eps_frac=0.001, step0=0.5
    )
    print(f"[proj‑subgrad] fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # 6. Greedy hill‑climb using the remaining budget
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.8f}")

    # --------------------------------------------------------------
    # 7. Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
```
