"""
Propose a low‑score step function.

The optimiser works in four phases (total ≤ 1000 s)

* Phase 0 – initialise a diverse pool of simplex vectors and keep the
  best few.
* Phase 1 – smoothed‐max L‑BFGS (log‑sum‑exp surrogate).  A very
  aggressive λ‑schedule is used and the optimiser is run from many
  different starts.
* Phase 2 – exact Adam optimisation of the true max‑convolution.  The
  Adam step uses the exact sub‑gradient (the mask of the active
  constraints) and a tiny jitter is added whenever progress stalls.
* Phase 3 – a long simulated‑annealing / greedy hill‑climb that works
  directly on the simplex (mass‑transfer moves).  The move size is
  reduced with the temperature.  The final result is returned as a
  plain Python list (the official evaluator rescales internally).

All helpers are defined at module level; the code uses only ``numpy`` and
``scipy`` and therefore works on the two CPUs that the sandbox provides.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (sum = 1 → score = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact score for a probability vector (∑a = 1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(y: np.ndarray, n: int,
                        fft_len_conv: int, fft_len_g: int,
                        lam: float) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad) where
    f = 2·n·smooth_max( a * a )   and   a = softmax(y).
    `lam` is the smoothing temperature.
    """
    a = softmax(y)

    # full (linear) self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    # λ‑scaled smooth‑max
    shifted = conv_full - max_c
    # prevent overflow/underflow
    exp_shift = np.exp(shifted / (lam + 1e-12))
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                 # ∂smooth_max / ∂conv_i

    # back‑prop to a (conv = a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """FFT‑based randomised spectrum, projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target = 2 * n - 1
    # start from a flat spectrum
    S = np.fft.fft(np.random.rand(L))
    # random phase, keep magnitude ~ sqrt(|S|)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    A = sqrt_amp * np.exp(1j * phase / 2.0)
    a = np.real(np.fft.ifft(A))[:n]
    a[a < 0] = 0.0
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few stochastic families, all projected onto the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:edge] = heavy
        a[-edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = rng.uniform(0.5, 1.5) / n
    else:   # dirichlet‑like
        alpha = rng.uniform(0.5, 3.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def run_lbfgs(start_a: np.ndarray,
              lam_schedule,
              maxiter_per_lam: int,
              deadline: float,
              fft_len_conv: int,
              fft_len_g: int,
              n: int):
    """Run L‑BFGS for every λ in the schedule, starting from *start_a*."""
    y = np.log(start_a + 1e-12)

    best_a = start_a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=smooth_obj_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_optimize(start_a: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """Adam on the exact max‑convolution (gradient via active mask)."""
    y = np.log(start_a + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # exact max and its mask
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)   # active set

        # gradient of max conv w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # gradient of objective (2·n·max_c)
        grad_a = 2.0 * n * g_a
        # chain through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.15 / np.sqrt(t)       # a bit smaller than the original
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
    return best_a, best_sc


def annealing_hill_climb(start_a: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         init_temp: float = 3e-4,
                         final_temp: float = 5e-8,
                         max_moves: int = 200000):
    """Simulated annealing + greedy hill‑climb."""
    a = start_a.copy()
    best = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # propose a mass transfer (up to δ% of the donor)
        delta = a[i] * rng.random() * 0.12
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best = cand.copy()
        move += 1

    # final agressive greedy polishing
    it = 0
    while time.time() < deadline and it < 25000:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_sc

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine runs for at most *budget_s* seconds and returns the best
    list of floats found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good balance between resolution and speed
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool (all vectors are on the simplex)
    # --------------------------------------------------------------
    pool = []

    # deterministic families
    pool.append(spectral_init(n))
    pool.append(hann_init(n))
    pool.append(blackman_init(n))
    pool.append(gaussian_init(n))

    # random families
    for _ in range(6):
        pool.append(random_initial(n, rng))

    # keep a few elite members
    elites = []
    best_score = np.inf
    for v in pool:
        sc = fast_score(v, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_vec = v.copy()
        elites.append((v, sc))
    elites.sort(key=lambda x: x[1])
    elites = elites[:5]                         # keep top‑5

    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinements from several starts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_sched = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02,
                 0.01, 0.005, 0.002, 0.001,
                 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # pick a random elite, perturb it and run L‑BFGS
        base, _ = elites[rng.integers(len(elites))]
        alpha = rng.random() * 0.35
        rand_vec = random_initial(n, rng)
        start = (1 - alpha) * base + alpha * rand_vec
        start /= start.sum()
        a_opt, sc_opt = run_lbfgs(start,
                                  lam_sched,
                                  maxiter_per_lam,
                                  phase1_end,
                                  fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_vec = a_opt.copy()
            elites.append((a_opt, sc_opt))
            elites.sort(key=lambda x: x[1])
            elites = elites[:5]
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact Adam optimisation (longer)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        adam_deadline = deadline - 20.0   # keep 20 s for the final polish
        best_vec, best_score = adam_optimize(best_vec,
                                             adam_deadline,
                                             n,
                                             fft_len,
                                             rng)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing + greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Annealing] start final hill‑climb")
        best_vec, best_score = annealing_hill_climb(best_vec,
                                                    deadline,
                                                    n,
                                                    fft_len,
                                                    rng)
        print(f"[Annealing] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluator rescales internally)
    # --------------------------------------------------------------
    final_seq = best_vec.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
