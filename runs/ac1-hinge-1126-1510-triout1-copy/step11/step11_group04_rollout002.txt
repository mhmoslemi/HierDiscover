"""
step‑function optimiser – 1000 s budget (enhanced version)

Main ideas
----------
1.  Build a large, diverse pool of start vectors (including the previous best).
2.  Run an aggressive L‑BFGS cascade on a smoothed log‑sum‑exp surrogate.
    The λ‑schedule is longer and the maximum number of L‑BFGS iterations per
    λ is larger, so we spend more time finding a good basin.
3.  Refine the basin with a *tolerant* exponentiated‑gradient (mirror‑SGD)
    descent on the true L∞ objective.  The sub‑gradient uses a mask that
    contains **all** entries that are within a tiny relative tolerance of the
    maximum – this yields a smoother direction than the strict max‑mask.
4.  Perform an intensive local‑search phase:
      * small stochastic mass transfers (hill‑climbing)
      * occasional “peak‑spacing” rearrangements that spread the largest
        coefficients uniformly over the index range.
   The fast surrogate `fast_score` (2·n·max(conv)) is evaluated after each
   proposal, and only improving moves are kept.
5.  Finish with a very low‑λ L‑BFGS polish.
6.  The final sequence is evaluated with the official scorer and stored in
   the global variable ``height_sequence_1``.

All steps respect the 1000 s deadline; intermediate progress is printed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when the vector is normalized to sum=1:
        2·n·max(conv(a,a))
    The convolution is computed via FFT.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed maximum of the auto‑convolution.
    Returns the objective value and its gradient w.r.t. the log‑variables y
    (where a = softmax(y)).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # FFT convolution
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Exponentiated‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize_tolerant(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eta0: float = 12.0,
                                   max_iters: int = 10_000_000,
                                   eps_rel: float = 1e-12):
    """
    Mirror‑SGD (exponentiated‑gradient) on the exact max‑convolution.
    Uses a *tolerant* mask that includes all entries within `eps_rel`
    relative tolerance of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * eps_rel + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # cheap score update every 32 iterations
        if (it & 31) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        it += 1

    return best_a, best_score

def rearrange_peaks(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Take the largest ~5 % of the mass and spread those coefficients
    uniformly across the index range.  The remaining coefficients retain
    their relative order.
    """
    n = a.shape[0]
    k = max(1, int(0.05 * n))

    # indices of the k largest entries
    idx_desc = np.argsort(-a)
    top_idx = idx_desc[:k]
    top_vals = a[top_idx].copy()

    # remove the top mass from the vector
    leftover = a.copy()
    leftover[top_idx] = 0.0

    # new equally‑spaced positions
    spacing = n // k
    new_pos = (np.arange(k) * spacing) % n

    cand = np.zeros_like(a)
    # place the top values
    for p, v in zip(new_pos, top_vals):
        cand[p] = v

    # fill the remaining positions with the leftover mass (preserving order)
    remaining_pos = np.where(cand == 0)[0]
    leftover_vals = leftover[leftover > 0]
    # If there are more positions than leftover values we just leave zeros
    for p, v in zip(remaining_pos, leftover_vals):
        cand[p] = v

    return cand


def local_search_and_rearrange(a_start: np.ndarray,
                               cur_score: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               deadline: float):
    """
    Hill‑climbing with stochastic mass transfers plus occasional
    peak‑spacing rearrangements.  Only moves that strictly improve the fast
    surrogate are kept.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = cur_score

    max_frac = 0.25               # maximal fraction of a[src] that may be moved
    check_interval = 32

    it = 0
    while time.time() < deadline:
        # --- stochastic mass transfer ---
        # pick a source with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = rng.choice(src_candidates)
        dst = rng.integers(0, n)
        if src == dst:
            continue

        # always move from larger to smaller entry
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand /= cand.sum()

        if (it & (check_interval - 1)) == 0:
            cand_score = fast_score(cand, fft_len)
        else:
            # cheap approximate: reuse previous score; we only need the exact
            # evaluation when checking improvement, so compute only on acceptance.
            cand_score = fast_score(cand, fft_len)

        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            a = cand.copy()
            it += 1
            continue  # keep the improvement

        # --- occasional noisy jitter ---
        if rng.random() < 0.001:
            cand = add_noise(a, sigma=rng.uniform(0.001, 0.02), rng=rng)
            cand /= cand.sum()
            cand_score = fast_score(cand, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand.copy()
                it += 1
                continue

        # --- occasional peak‑spacing rearrangement ---
        if rng.random() < 0.005:
            cand = rearrange_peaks(a, rng)
            cand /= cand.sum()
            cand_score = fast_score(cand, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand.copy()
                it += 1
                continue

        it += 1

    return best_a, best_score

            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = old.copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
            pool.append(a0)

    # 2) Randomly generated families (more than before)
    for _ in range(15):
        pool.append(random_initial(n, rng))

    # 3) Classic window functions (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.25 * budget_s          # aggressive L‑BFGS
    phase2_end = start_time + 0.75 * budget_s          # long EG phase
    # remaining time → local search + final polish

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002,
        0.00001
    ]
    maxiter_per_lam = 1500

    best_a = None
    best_score = float('inf')
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # occasional jitter on the start vector
        if rng.random() < 0.3:
            a_start = add_noise(a_start,
                                sigma=rng.uniform(0.001, 0.02),
                                rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase 1 L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – tolerant exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize_tolerant(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=30.0,          # a bit larger initial step size
            max_iters=15_000_000,
            eps_rel=1e-3       # include near‑max entries in the mask
        )
        print(f"[Phase 2 EG] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – intensive local search + peak‑spacing rearrangements
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = local_search_and_rearrange(
            best_a,
            best_score,
            n,
            fft_len,
            rng,
            deadline
        )
        print(f"[Phase 3 local] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – final low‑λ L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_sched = [5e-6, 2.5e-6, 1.2e-6, 6e-7, 3e-7, 1e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_sched,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase 4 final‑L‑BFGS] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[FINAL] evaluated score = {final_eval:.9f}")

    # Store for possible warm‑starts in a future run
    globals()['height_sequence_1'] = final_seq
    return final_seq
