def fft_convolve(x, y):
    L = len(x) + len(y) - 1
    nfft = next_pow2(L)
    fx = np.fft.fft(x, nfft)
    fy = np.fft.fft(y, nfft)
    conv = np.fft.ifft(fx * fy)[:L].real
    return conv

def adam_optimize(a_start, deadline, n, rng, lr=0.1, beta_start=0.5, beta_factor=1.02, beta_max=80.0, eps=1e-8):

    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    beta = beta_start
    t = 0
    best_a = a.copy()
    best_score = fast_score(a, fft_len) # compute at start
    while time.time() < deadline:
        t += 1
        # compute conv and w
        conv = fft_convolve(a, a)
        max_conv = conv.max()
        # smoothing
        # subtract max for numerical stability
        temp = beta * (conv - max_conv)
        # clip maybe to avoid overflow?
        w = np.exp(temp - np.max(temp)) # Actually subtract max(temp) after scaling? Wait max(temp) = beta*(max_conv - max_conv) = 0. So w = exp(beta*(conv - max_conv)). This is stable as relative exponent <=0.
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient
        rev_a = a[::-1]
        grad_conv = fft_convolve(rev_a, w)
        # slice middle n entries
        grad = 2.0 * grad_conv[n-1:n-1+n]

        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        # update
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a_new = simplex_projection(a_new)

        # evaluate
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # maybe accept if score improves else maybe revert? Use always new a (as in Adam) but could also revert if too far.
        a = a_new

        # update beta
        beta = min(beta * beta_factor, beta_max)

        # optional: if stagnation for many iterations (no improvement for say 500), add jitter.
        # implement simple check: if last_improve_counter > some threshold, add small jitter to a.

    return best_a, best_score

"""
Low‑evaluation step‑function search.

This implementation improves the previous heuristic by:
  * Using a larger problem size (n≈8000) while keeping FFT sizes modest.
  * Employing a smoothed max objective (log‑sum‑exp) and a
    projected Adam optimiser with an annealing schedule on the
    smoothing parameter β.  This provides a smooth sub‑gradient and a
    well‑behaved adaptive step‑size.
  * Performing several independent restarts (uniform, arcsine,
    random, previous best) and keeping the best result across all
    runs.
  * Finishing with a short greedy hill‑climb that uses exact
    convolution updates for a final fine‑tune.
The whole search respects the 1000 s time budget and returns the best
list of non‑negative floats found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (output length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    nfft = next_pow2(L)
    X = np.fft.rfft(x, nfft)
    Y = np.fft.rfft(y, nfft)
    conv = np.fft.irfft(X * Y, n=nfft)[:L]
    return conv


def fast_score(a: np.ndarray, fft_len_self: int) -> float:
    """Cheap score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    # linear convolution using zero‑padded FFT (length >= 2·n‑1)
    X = np.fft.rfft(a, fft_len_self)
    conv = np.fft.irfft(X * X, n=fft_len_self)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth approximation of max(conv)
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  rng: np.random.Generator,
                  lr: float = 0.2,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  beta_init: float = 0.5,
                  beta_factor: float = 1.02,
                  beta_max: float = 80.0) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation of the smoothed max‑convolution.
    Uses log‑sum‑exp smoothing with a temperature β that is
    exponentially increased during the run.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # FFT lengths for the two convolutions we need
    fft_len_self = next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = next_pow2(3 * n - 2)          # for a_rev * w

    best_a = a.copy()
    # initial score (exact max of linear conv)
    X = np.fft.rfft(a, fft_len_self)
    conv_self = np.fft.irfft(X * X, n=fft_len_self)[: 2 * n - 1]
    best_score = float(2.0 * n * conv_self.max())

    it = 0
    no_improve_iters = 0
    beta = beta_init

    while time.time() < deadline:
        it += 1

        # ----- linear convolution a * a -----
        X = np.fft.rfft(a, fft_len_self)
        conv_self = np.fft.irfft(X * X, n=fft_len_self)[: 2 * n - 1]

        # ----- smooth max via log‑sum‑exp -----
        max_c = conv_self.max()
        # w_k ∝ exp(β (conv_k - max_c))
        w = np.exp(beta * (conv_self - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:          # numerical safety
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth max -----
        a_rev = a[::-1]
        X_rev = np.fft.rfft(a_rev, fft_len_grad)
        W = np.fft.rfft(w, fft_len_grad)
        conv_grad = np.fft.irfft(X_rev * W, n=fft_len_grad)[: 3 * n - 2]
        grad = 2.0 * conv_grad[n - 1:n - 1 + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = simplex_projection(a_new)

        # ----- evaluate (fast) -----
        X_new = np.fft.rfft(a_new, fft_len_self)
        conv_new = np.fft.irfft(X_new * X_new, n=fft_len_self)[: 2 * n - 1]
        score = float(2.0 * n * conv_new.max())

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve_iters = 0
        else:
            no_improve_iters += 1

        # occasional jitter to escape plateaux
        if no_improve_iters >= 500:
            jitter = rng.normal(scale=1e-4, size=n)
            a_new = a_new + jitter
            a_new = np.clip(a_new, 0.0, None)
            a_new = simplex_projection(a_new)
            no_improve_iters = 0

        a = a_new

        # raise β (make the smoothing sharper)
        beta = min(beta * beta_factor, beta_max)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            # occasional recomputation to avoid drift from rounding errors
            conv = np.convolve(best, best)

    return best, best_score

    for a in init_candidates:
        sc = fast_score(a, fft_len_self)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # --------------------------------------------------------------
    # 2. Adam optimisation on each primitive (≈ 45 % of budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    per_restart_budget = min(250.0, remaining * 0.3)

    for idx, cand in enumerate(init_candidates):
        if time.time() >= deadline - 10.0:
            break
        sub_deadline = time.time() + per_restart_budget
        cand_opt, cand_score = adam_optimize(
            cand, sub_deadline, n, rng,
            lr=0.2, beta_init=0.5, beta_factor=1.02, beta_max=80.0
        )
        print(f"[adam #{idx}] fast_score = {cand_score:.6f}")
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_opt.copy()

    # --------------------------------------------------------------
    # 3. Refine the current best for the remaining time (≈ 35 %)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = time.time() + (deadline - time.time()) * 0.6
        best_a, best_score = adam_optimize(
            best_a, sub_deadline, n, rng,
            lr=0.1, beta_init=5.0, beta_factor=1.01, beta_max=200.0
        )
        print(f"[adam‑refine] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Final greedy hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
