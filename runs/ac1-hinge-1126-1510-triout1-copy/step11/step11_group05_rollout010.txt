"""
Propose a low‑score step‑function sequence using a multi‑phase
search:

1. Build a diverse pool of initial probability vectors (uniform,
   Dirichlet with several concentrations, random‑mode, spectral,
   previous best).

2. Run many Adam restarts on a log‑sum‑exp smooth surrogate of the
   max‑convolution objective, with a long smoothing schedule and
   aggressive step counts.

3. Refine the current best with exponentiated‑gradient descent on the
   true (non‑smoothed) max‑convolution, using a decaying learning
   rate.

4. Apply projected sub‑gradient descent for further fine‑tuning.

5. Perform a long simulated‑annealing local search that moves tiny
   amounts of mass between indices, accepting uphill moves with a
   geometric temperature decay.

6. Mix the top‑K vectors found so far by a cheap golden‑section line
   search over convex combinations.

All vectors are kept on the probability simplex (∑a = 1, a ≥ 0).  The
best vector is stored globally as ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective = 2·n·max(conv) assuming ∑a = 1.
    Computed via FFT for speed.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # only the first 2*n-1 entries are relevant
    return float(2.0 * n * conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def dirichlet_initial(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet(α,…,α) vector and return it as a probability vector."""
    a = rng.dirichlet(np.full(n, alpha))
    # protection against numerical drift
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start from a square‑rooted flat spectrum (real‑valued)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate and its gradient (log‑sum‑exp)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score) found during the run.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          scale: float = 10.0,
                          max_iter: int = 5_000_000):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

def projected_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              init_eta: float = 0.5,
                              max_iter: int = 2_000_000):
    """
    Euclidean projected sub‑gradient descent on max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta = init_eta
    while time.time() < deadline and it < max_iter:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        a_step = a - eta * g
        a = proj_simplex(a_step)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
        eta = init_eta / np.sqrt(it + 1)
        if eta < 1e-8:
            break

    return best_a, best_score

def simulated_annealing_search(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               init_temp: float = 5e-3,
                               final_temp: float = 1e-9,
                               max_iter: int = 2_000_000):
    """
    Random mass‑transfer moves with a geometric temperature decay.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        # geometric temperature decay
        temp = max(final_temp, init_temp * (0.9995 ** it))

        donor = rng.integers(n)
        if a[donor] <= 0.0:
            it += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            it += 1
            continue

        eps = a[donor] * rng.uniform(0.001, 0.02)
        if eps <= 0.0:
            it += 1
            continue

        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps
        if a_new[donor] < 0.0:
            it += 1
            continue

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            # Metropolis acceptance
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score

        it += 1

    return best_a, best_score

def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each unordered pair of vectors perform a cheap golden‑section
    line search over t∈[0,1] to minimise the bound.
    Returns (best_a, best_score).
    """
    best_a = candidates[0].copy()
    best_score = scores[0]

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            lo, hi = 0.0, 1.0
            c = hi - invphi * (hi - lo)
            d = lo + invphi * (hi - lo)

            mix_c = (1.0 - c) * a + c * b
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, fft_len, n)

            mix_d = (1.0 - d) * a + d * b
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, fft_len, n)

            while time.time() < deadline and (hi - lo) > 1e-4:
                if fc < fd:
                    hi, fd = d, fc
                    d = c
                    c = hi - invphi * (hi - lo)
                    mix_c = (1.0 - c) * a + c * b
                    mix_c /= mix_c.sum()
                    fc = fast_score(mix_c, fft_len, n)
                else:
                    lo, fc = c, fd
                    c = d
                    d = lo + invphi * (hi - lo)
                    mix_d = (1.0 - d) * a + d * b
                    mix_d /= mix_d.sum()
                    fd = fast_score(mix_d, fft_len, n)

            tm = 0.5 * (lo + hi)
            mix_tm = (1.0 - tm) * a + tm * b
            mix_tm /= mix_tm.sum()
            ftm = fast_score(mix_tm, fft_len, n)

            if ftm < best_score:
                best_score = ftm
                best_a = mix_tm.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom to flatten the
    # convolution while keeping the computation still cheap.
    # ------------------------------------------------------------------
    n = 8192                     # power of two → efficient FFTs
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build an initial pool of diverse vectors
    # ------------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # previous global result, interpolated to the new size
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, prev)
            a_interp = np.maximum(a_interp, 0.0)
            a_interp = a_interp / a_interp.sum()
            candidates.append(a_interp)

    # Dirichlet draws with various concentrations
    for alpha in [0.3, 0.7, 1.5, 4.0]:
        candidates.append(dirichlet_initial(n, alpha, rng))

    # a few random‐mode vectors
    for _ in range(3):
        candidates.append(random_initial(n, rng))

    # spectral initialisation (deterministic)
    candidates.append(spectral_init(n))

    # ------------------------------------------------------------------
    # Evaluate the pool and keep the best
    # ------------------------------------------------------------------
    scores = [fast_score(c, fft_len_conv, n) for c in candidates]
    best_idx = int(np.argmin(scores))
    best_a = candidates[best_idx].copy()
    best_score = scores[best_idx]

    # keep a short list of the current top vectors (for later mixing)
    top_K = 12
    ordered = sorted(zip(scores, candidates), key=lambda x: x[0])
    top_vectors = [vec for _, vec in ordered[:top_K]]
    top_scores = [sc for sc, _ in ordered[:top_K]]

    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Adam restarts on smoothed surrogate
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.4, 0.3, 0.2, 0.15,
        0.10, 0.07, 0.04, 0.025, 0.015,
        0.010, 0.006, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 0.00005, 0.00002,
        0.00001
    ]
    steps_per_lam = 250
    num_restarts = 15

    for r in range(num_restarts):
        if time.time() >= deadline:
            break
        # Choose a start point: either the current best or a noisy top vector
        if r == 0:
            a0 = best_a.copy()
        else:
            base = top_vectors[r % len(top_vectors)]
            a0 = add_noise(base, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0, lam_schedule, steps_per_lam, deadline,
            fft_len_conv, fft_len_g, n)
        print(f"[Adam restart {r+1}/{num_restarts}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (true max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a, deadline, n, fft_len_conv, rng,
            scale=12.0, max_iter=5_000_000)
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after Exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = projected_subgrad_optimize(
            best_a, deadline, n, fft_len_conv,
            init_eta=0.6, max_iter=2_000_000)
        print(f"[Proj‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after proj‑subgrad: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = simulated_annealing_search(
            best_a, deadline, n, fft_len_conv, rng,
            init_temp=5e-3, final_temp=1e-9, max_iter=2_000_000)
        print(f"[Sim‑anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after anneal: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – convex‑combination (mixture) search among top vectors
    # ------------------------------------------------------------------
    # Assemble a small set of promising vectors (include the current best)
    mix_candidates = top_vectors[:5] + [best_a]
    mix_scores = [fast_score(v, fft_len_conv, n) for v in mix_candidates]

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            mix_candidates, mix_scores, n, fft_len_conv, deadline)
        print(f"[Mixture] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
