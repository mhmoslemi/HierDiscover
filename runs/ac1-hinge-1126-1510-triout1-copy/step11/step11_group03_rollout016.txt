def adam_optimize(a_start, deadline, n, fft_len, rng, eta0=1.0):
    a = a_start.copy()
    y = np.log(a + 1e-16)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    while time.time() < deadline:
        # compute gradient of max conv w.r.t a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        d = 2.0 * grad_conv[n-1:n-1+n]  # gradient wrt a

        # convert to gradient wrt y
        mean_d = np.dot(a, d)
        grad_y = a * (d - mean_d)

        # Adam step
        it += 1
        lr = eta0 * np.sqrt(1 - beta2**it) / (1 - beta1**it)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y**2)

        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # back to a via softmax
        e = np.exp(y - np.max(y))
        a = e / e.sum()

        # compute score, update best
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def root_sinc_initial(n, rng):
    """Generate an initial distribution based on sqrt of sinc in frequency domain."""
    fft_len = next_pow2(2 * n - 1)
    freq = np.fft.fftfreq(fft_len)  # cycles per unit
    # sinc with period 1: np.sinc(freq) returns sin(pi*freq)/(pi*freq)
    # Use absolute value to avoid complex.
    amp = np.sqrt(np.maximum(1e-12, np.abs(np.sinc(freq))))
    # Random phase for each frequency
    phase = rng.random(fft_len) * 2 * np.pi
    spec = amp * np.exp(1j * phase)
    # Enforce conjugate symmetry to get real sequence
    # For even length, indices 1..L/2-1 are symmetric; set spec[-i] = conj(spec[i]) 
    # We'll simply ensure real by taking real part after IFFT.
    time_seq = np.real(np.fft.ifft(spec))
    # Take first n entries
    a = time_seq[:n]
    # Ensure nonnegative
    a = np.maximum(a, 0.0)
    # Normalize to sum=1
    if a.sum() <= 0:
        # fallback uniform
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a

Key ideas added on top of the previous version

* A new “root‑sinc” parametric family – the inverse FFT of
  sqrt(|sinc|) with random phases – provides a very different shape
  for the initial pool and helps to escape local minima.
* A longer exploratory EG phase with many short restarts (phase 0)
  before the heavy L‑BFGS work.
* Adaptive Adam optimisation on the log‑softmax (y‑space) of the
  probabilities.  Adam converges faster than plain EG for the exact
  L∞‑subgradient.
* More aggressive λ‑schedules and higher iteration limits for the
  L‑BFGS cascades.
* The final annealing stage is unchanged but kept as a polishing
  step.

The function ``propose_candidate`` returns the best sequence found
before the 1000 s deadline and stores it in the global
``height_sequence_1`` for reuse by later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# optional windows – they may be missing on minimal installations
try:
    from scipy.signal import chebwin, kaiser
except Exception:  # pragma: no cover
    chebwin = None
    kaiser = None


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) for a already normalised to sum = 1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector from a diverse set of styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Symmetric Gaussian‑shaped density."""
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Cosine raised to a random power (non‑negative)."""
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass at the two edges, light uniform interior."""
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


def root_sinc_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Approximate the “sqrt‑sinc’’ distribution:
    draw a random phase spectrum whose magnitude is sqrt(|sinc|) and
    invert it.  The resulting sequence is shifted to be non‑negative
    and normalised.
    """
    fft_len = next_pow2(2 * n - 1)
    freq = np.fft.fftfreq(fft_len)               # cycles per sample
    # np.sinc = sin(pi*x)/(pi*x)
    amp = np.sqrt(np.maximum(1e-12, np.abs(np.sinc(freq))))
    phase = rng.random(fft_len) * 2.0 * np.pi
    spec = amp * np.exp(1j * phase)

    # force Hermitian symmetry for a real time‑domain signal
    spec[0] = np.real(spec[0]) + 0j
    if fft_len % 2 == 0:
        spec[fft_len // 2] = np.real(spec[fft_len // 2]) + 0j
    spec[1:fft_len // 2] = np.conj(spec[-1:fft_len // 2:-1])

    time_seq = np.real(np.fft.ifft(spec))
    a = time_seq[:n]
    a -= a.min()                # shift to be non‑negative
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y lives in ℝⁿ; a = softmax(y) belongs to the probability simplex.
    The regulariser λ is scaled by the current max value.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS problems over a decreasing λ‑schedule.
    Returns the best vector discovered and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact L∞ sub‑gradient descent (mirrored EG)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored SGD (exponentiated‑gradient) using the exact L∞‑subgradient.
    Runs until the deadline is reached.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax (adds momentum)
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  eta0: float = 0.5):
    """
    Adam on the log‑softmax (y‑space).  Uses the exact L∞ sub‑gradient.
    """
    a = a_start.copy()
    y = np.log(a + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # --- exact sub‑gradient of max(conv) w.r.t. a ---
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        d = 2.0 * grad_conv[n - 1: n - 1 + n]          # ∂max/∂a

        # chain rule to the log‑softmax variables
        mean_d = np.dot(a, d)
        grad_y = a * (d - mean_d)

        # Adam update in y‑space
        it += 1
        lr = eta0 * np.sqrt(1 - beta2 ** it) / (1 - beta1 ** it)

        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # back to probabilities via softmax
        e = np.exp(y - np.max(y))
        a = e / e.sum()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (enhanced)
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types:
    * mass transfer between two arbitrary positions
    * swapping two entries
    * local neighbour transfer (distance ≤ 5)
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25            # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]          # pure swap
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # 'transfer'
                j = rng.integers(0, n)
            if i == j:
                continue

            # move mass from larger entry
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue

            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a modest size gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # previous best (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0.0:
                a0 /= s
                pool.append(a0)

    # classic windows (if SciPy available)
    if chebwin is not None:
        for att in (30, 60):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w_sum = w.sum()
            if w_sum > 0:
                pool.append(w / w_sum)
    if kaiser is not None:
        for beta in (4, 8):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w_sum = w.sum()
            if w_sum > 0:
                pool.append(w / w_sum)

    # random families
    for _ in range(8):
        pool.append(random_initial(n, rng))

    # parametric families
    for _ in range(4):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(edge_heavy_initial(n, rng))
        pool.append(root_sinc_initial(n, rng))

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase0_end = start_time + 0.10 * budget_s       # many short EG runs
    phase1_end = phase0_end + 0.30 * budget_s       # aggressive L‑BFGS
    phase2_end = phase1_end + 0.30 * budget_s       # Adam optimisation
    phase3_end = phase2_end + 0.20 * budget_s       # fine L‑BFGS
    # remainder for final annealing

    # --------------------------------------------------------------
    # Phase 0 – many quick EG explorations
    # --------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    while time.time() < phase0_end:
        a_start = rng.choice(pool)
        a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.005), rng=rng)

        a_cand, sc_cand = exp_gradient_optimize(
            a_start, phase0_end, n, fft_len, rng, eta0=30.0
        )
        if sc_cand < best_score:
            best_score = sc_cand
            best_a = a_cand.copy()
            print(f"[Phase0] new best fast_score = {best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on smoothed objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.015, 0.008, 0.004,
        0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.2 and best_a is not None:
            a_start = 0.5 * a_start + 0.5 * best_a
            a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_cascade(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimiser (adds momentum)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=0.5
        )
        print(f"[Adam] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with ultra‑low λ
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        fine_lam_sched = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6,
                          1e-6, 5e-7, 2e-7, 1e-7]
        a_fine, sc_fine = lbfgs_cascade(
            best_a, fine_lam_sched, maxiter_per_lam=1500,
            deadline=phase3_end, fft_len=fft_len, n=n
        )
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – short simulated annealing for the last bits
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
