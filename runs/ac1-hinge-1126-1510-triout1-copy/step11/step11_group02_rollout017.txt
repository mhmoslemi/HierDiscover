"""
search_step_sequence.py

Search for a non‑negative step sequence that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works on the probability simplex (the sequence is
normalised to sum = 1, the scaling does not affect the score).

* Phase 0 – broad parametric sweep (exponential, power‑law, logistic,
  Beta‑U‑shapes and mixtures).  The best candidate from this sweep is
  taken as the starting point for the following phases.

* Phase 1 – many restarts of L‑BFGS on a smooth log‑sum‑exp
  approximation of the max‑convolution.  The smoothing parameter λ is
  reduced on a schedule that ends at 1e‑07, allowing the optimiser to
  approach the true nonsmooth objective.

* Phase 2 – projected Adam on the exact max‑convolution (sub‑gradient
  of the true max).  This phase is very robust and quickly removes
  remaining spikes.

* Phase 3 – a long simulated‑annealing run with several move types
  (mass transfer, swaps, three‑way redistributions).  The temperature
  follows a square‑root cooling schedule, which gives a finer search
  at the end of the budget.

* Phase 4 – final polishing by a short L‑BFGS run with an ultra‑small
  smoothing factor.

All phases respect a global deadline (the function is allowed 1000 s
in total).  The best sequence seen at any moment is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the official evaluation (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (logits → probabilities)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (using the reversed a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)                 # logits – stays on the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:        # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution (sub‑gradient of max).
    Very robust for long runs.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)            # first moment
    v = np.zeros_like(a)            # second moment
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.4

    while time.time() < deadline:
        it += 1
        # exact convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                     # any convex combination is a valid sub‑gradient

        # sub‑gradient of the max (ordered pairs)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update on the raw vectors (project later)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with square‑root cooling.
    Moves:
      – transfer a random fraction of mass from i → j,
      – swap the values of two indices,
      – redistribute a small fraction among three random positions.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            # transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            # triple redistribution
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """U‑shaped Beta distribution (α,β > 0)."""
    x = (np.arange(n, dtype=float) + 0.5) / n          # centre of each bin
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def generate_mix_beta(n: int, mix: float, alpha: float, beta: float) -> np.ndarray:
    """Convex mixture of uniform and a Beta shape."""
    uniform = np.full(n, 1.0 / n)
    beta_shape = generate_beta(n, alpha, beta)
    a = mix * uniform + (1.0 - mix) * beta_shape
    # already sums to 1 because both components sum to 1
    return a


def generate_inv_sqrt(n: int, eps: float = 1e-3) -> np.ndarray:
    """Symmetric 1/√(i·(n‑1‑i)) shape, regularised with eps."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + eps) * (n - 1 - i + eps))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_double_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a


def generate_random(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a parametric family."""
    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (np.arange(n, dtype=float) + 1.0) ** p
        if not inc:
            a = a[::-1]
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "beta":
        alpha = rng.uniform(0.1, 0.9)
        beta = rng.uniform(0.1, 0.9)
        a = generate_beta(n, alpha, beta)
    elif family == "mix_beta":
        mix = rng.uniform(0.0, 1.0)
        alpha = rng.uniform(0.1, 0.9)
        beta = rng.uniform(0.1, 0.9)
        a = generate_mix_beta(n, mix, alpha, beta)
    elif family == "inv_sqrt":
        eps = rng.uniform(1e-4, 1e-2)
        a = generate_inv_sqrt(n, eps)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = generate_double_exp(n, rate)
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = generate_logistic(n, mu, sigma)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        i = np.arange(n, dtype=float)
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "random":
        a = generate_random(n, rng)
    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Systematic / random sweeps
# ----------------------------------------------------------------------


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Grid over several families (including new Beta‑U shapes)."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential grid
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power‑law grid
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic grid
    mus = np.linspace(0.2 * n, 0.8 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # Beta‑U‑shape grid
    alphas = np.linspace(0.15, 0.85, 8)
    betas = np.linspace(0.15, 0.85, 8)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_beta(n, alpha, beta)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] beta α={alpha:.2f} β={beta:.2f} → score={best_sc:.6f}")

    # mixture (uniform + beta)
    mixes = np.linspace(0.0, 1.0, 5)
    for mix in mixes:
        for alpha in alphas:
            for beta in betas:
                if time.time() >= deadline:
                    return best_a, best_sc
                a = generate_mix_beta(n, mix, alpha, beta)
                sc = fast_score(a, fft_len, n)
                if sc < best_sc:
                    best_sc, best_a = sc, a.copy()
                    print(f"[grid] mix {mix:.2f} beta α={alpha:.2f} β={beta:.2f} → score={best_sc:.6f}")

    # inverse‑sqrt grid
    eps_vals = np.logspace(-4, -2, 6)
    for eps in eps_vals:
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_inv_sqrt(n, eps)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] inv_sqrt eps={eps:.1e} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Quick random exploration over an enlarged set of families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10  # 10 % of total budget

    families = [
        "power",
        "exp",
        "beta",
        "mix_beta",
        "inv_sqrt",
        "double_exp",
        "logistic",
        "gauss",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the official
    evaluation score.  Returns the best list of floats found before the
    deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we use a larger n than the previous runs; the lower
    # bound 2·n/(2·n‑1) ≈ 1.0 becomes tighter for larger n.
    # ------------------------------------------------------------------
    n = 8000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # ------------------------------------------------------------------
    # Initial candidate (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optionally start from a historic solution (if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – systematic grid + cheap random sweep
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time allocation for the remaining phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s   # L‑BFGS restarts
    phase2_end = start_time + 0.65 * budget_s   # Adam on true max
    phase3_end = start_time + 0.90 * budget_s   # Simulated annealing
    # remaining time → final polishing L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 5e-07, 2e-07,
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # jitter the best candidate or start from a fresh random family
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.025, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            fam = rng.choice(["beta", "mix_beta", "inv_sqrt", "exp", "power"])
            a_start = generate_family(fam, n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] Adam sub‑gradient best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing (rich move set)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing – short L‑BFGS run with ultra‑small λ
    # ------------------------------------------------------------------
    if time.time() < deadline:
        tiny_sched = [1e-07, 5e-08, 2e-08, 1e-08]
        a_final, sc_final = lbfgs_continuation(
            best_a,
            tiny_sched,
            maxiter_per_lam=1500,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_score:
            best_score, best_a = sc_final, a_final.copy()
            print(f"[final‑lbfgs] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (single call – may be slower)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
