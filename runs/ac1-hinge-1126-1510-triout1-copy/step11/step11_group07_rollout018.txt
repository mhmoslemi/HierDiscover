def uniform_initial(n):
    return np.full(n, 1.0/n)

def arcsine_initial(n):
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0/np.sqrt(idx*(n+0.5-idx))
    a /= a.sum()
    return a

def exponential_initial(n, r):
    # compute r**np.arange(n), normalize.
    powers = np.power(r, np.arange(n))
    total = powers.sum()
    return powers/total

    idx = np.arange(n, dtype=np.float64) + 0.5

    a = vals / vals.sum()
    return a

def random_initial(n, rng):
    a = rng.random(n)
    a /= a.sum()
    return a

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    n = 4000
    fft_len = next_pow2(2*n - 1)
    deadline = time.time() + budget_s - 0.1

    # Generate candidates list
    candidates = []

    candidates.append(uniform_initial(n))
    candidates.append(arcsine_initial(n))
    # exponential r grid
    for r in np.linspace(0.99, 0.9999, 200):
        cand = exponential_initial(n, r)
        candidates.append(cand)
    # power p grid
    for p in np.linspace(-1.5, -0.1, 30):
        cand = power_initial(n, p)
        candidates.append(cand)
    # random few
    for _ in range(5):
        candidates.append(random_initial(n, rng))

    best_a = None
    best_score = float('inf')
    for a in candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score}")

    # main optimization phase: long projected subgradient (repeat restarts)
    while time.time() < deadline - 10.0:

        phase_deadline = time.time() + min(20.0, deadline - time.time() - 10.0)
        a_opt, sc_opt = projected_subgradient_optimize(best_a, phase_deadline, n, fft_len, rng, eps_frac=0.001, step0=0.2)
        if sc_opt < best_score:

            best_a = a_opt.copy()

    if time.time() < deadline:

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)

"""
Propose a low‑evaluation step‑function sequence.

Algorithm (≈ 1000 s)
--------------------
1.  Build a diverse pool of deterministic candidates (uniform, arcsine,
    geometric, power‑law).  All candidates are normalised to sum = 1.
    The cheapest possible “fast_score” (2·n·max(conv)) is used to keep the
    best one.

2.  Refinement – a long projected sub‑gradient descent on the true
    max‑convolution.  The update uses the exact gradient of the active
    (nearly maximal) convolution entries, an adaptive step‑size
    ηₖ = η₀/√k and a back‑tracking line‑search that accepts only
    improvements.  The routine is called repeatedly on the current best
    vector until the time budget is exhausted.

3.  A final greedy hill‑climb (exact incremental convolution updates)
    removes any remaining tiny peaks.

4.  The resulting list is returned; the harness will also call
    ``evaluate_sequence`` for the official score.

All steps respect the 1000 s budget, stay on the probability simplex,
and use only numpy (no file/network I/O).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a)).
    The vector is assumed to be normalised (sum = 1) – this is exactly the
    value returned by ``evaluate_sequence`` after the clamping steps.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Deterministic initial shapes
# ----------------------------------------------------------------------


def uniform_initial(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def exponential_initial(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, normalised to sum = 1 (0 < r < 1)."""
    powers = np.power(r, np.arange(n, dtype=np.float64))
    total = powers.sum()
    a = powers / total
    return a


def power_initial(n: int, p: float) -> np.ndarray:
    """Symmetric power law a_i ∝ (i+½)*(n‑i‑½) raised to p (p may be < 0)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    vals = (idx * (n + 0.5 - idx)) ** p
    a = vals / vals.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (core of the search)
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.5) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set contains all convolution entries whose value is within
    `eps_frac` of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # current convolution (FFT based – O(n log n))
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # active mask – entries nearly attaining the maximum
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        # gradient of the active max‑convolution w.r.t. a
        a_rev = a[::-1]                     # reversed version of a
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad[n - 1: n - 1 + n]    # only the central n components

        # diminishing step size (√ decay)
        step = step0 / (np.sqrt(it + 1.0))

        # back‑tracking line‑search (up to 5 halvings)
        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)

            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()

            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5

        if not improved:
            # small stochastic jitter to escape possible stalls
            jitter = rng.normal(scale=1e-5, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        # gradually shrink the maximal moved fraction
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # occasional full recompute to avoid drift
        if accepted and accepted % 3000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain Python list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 works well for the given budget
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1. Build a pool of deterministic / random starting points
    # --------------------------------------------------------------
    candidates = []

    # a) uniform
    candidates.append(uniform_initial(n))

    # b) arcsine (classic “edge‑heavy” shape)
    candidates.append(arcsine_initial(n))

    # c) geometric families – scan a fine grid of r ∈ [0.99, 0.9999]
    for r in np.linspace(0.99, 0.9999, 250):
        candidates.append(exponential_initial(n, r))

    # d) power‑law families – symmetric shapes a_i ∝ (i·(n‑i))^p
    for p in np.linspace(-1.5, -0.05, 30):
        candidates.append(power_initial(n, p))

    # e) a handful of completely random vectors (for global exploration)
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # keep the best primitive (cheap evaluation)
    best_a = None
    best_score = float('inf')
    for a in candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # --------------------------------------------------------------
    # 3. Long projected sub‑gradient optimisation (≈ 80 % of remaining time)
    # --------------------------------------------------------------
    while time.time() < deadline - 20.0:   # leave a few seconds for hill‑climb
        phase_deadline = time.time() + min(15.0, deadline - time.time() - 5.0)
        best_a, best_score = projected_subgradient_optimize(
            best_a, phase_deadline, n, fft_len, rng,
            eps_frac=0.001, step0=0.6
        )
        print(f"[sub‑grad] fast_score = {best_score:.6f}")

        # occasional random jitter to escape flat plateaus
        if rng.random() < 0.2:
            jitter = rng.normal(scale=1e-4, size=n)
            best_a = best_a + jitter
            best_a = np.clip(best_a, 0.0, None)
            best_a /= best_a.sum()

    # --------------------------------------------------------------
    # 4. Final greedy hill‑climb (exact updates)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Official evaluation (the harness will also call it)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluates with clipping and scaling
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
