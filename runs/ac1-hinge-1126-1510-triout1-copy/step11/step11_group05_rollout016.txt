"""
Propose a low‑score non‑negative step‑function sequence.

The search works on a moderate problem size (n≈800) which is fast
enough to allow thousands of gradient and local‑search steps while
still providing enough degrees of freedom to beat the previous best
score (~1.506).  The pipeline consists of:

1. Build a diverse pool of initial vectors (uniform, random, exponential,
   spectral, noisy copy of the historic best).
2. Run several Adam restarts on a smooth log‑sum‑exp surrogate of the
   max‑convolution objective.
3. Refine the best vector with exponentiated‑gradient descent on the
   true (non‑smoothed) max‑convolution objective.
4. Perform a projected sub‑gradient descent for a few extra iterations.
5. Apply a linear‑programming refinement step: given the current a,
   solve the LP   maximise Σg   s.t.   (a∗g)≤M, g≥0, Σg≤UB,
   where M = max(a∗a).  Mix the resulting g back into a with a line
   search that never raises the max‑convolution.
6. Run a cheap simulated‑annealing local search.
7. Combine the top few candidates by a golden‑section line‑search over
   convex mixtures.

All vectors are kept on the probability simplex (Σa=1, a≥0) and the fast
FFT‑based convolution used in the inner loops is mathematically identical
to the official `evaluate_sequence` routine.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective = 2·n·max(a∗a) assuming Σa = 1.
    Used for quick evaluation during the search.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Random / spectral initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start from a square‑rooted flat spectrum (real‑valued)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          scale: float = 10.0):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgrad_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               init_eta: float = 0.5):
    """
    Euclidean projected sub‑gradient descent on max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta = init_eta
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        a_step = a - eta * g
        a = proj_simplex(a_step)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
        eta = init_eta / np.sqrt(it + 1)
        if eta < 1e-8:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing local search
# ----------------------------------------------------------------------


def simulated_annealing_search(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               init_temp: float = 1e-3,
                               final_temp: float = 1e-9,
                               max_iter: int = 5_000_000):
    """
    Random mass‑transfer moves with a temperature schedule.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        temp = max(final_temp, init_temp * (0.9996 ** it))

        donor = rng.integers(n)
        if a[donor] <= 0.0:
            it += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            it += 1
            continue

        eps = a[donor] * rng.uniform(0.001, 0.05)
        if eps <= 0.0:
            it += 1
            continue

        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps
        if a_new[donor] < 0.0:
            it += 1
            continue

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among top candidates
# ----------------------------------------------------------------------


def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    Golden‑section line search over convex mixtures of each unordered pair.
    Returns (best_a, best_score).
    """
    best_a = candidates[0].copy()
    best_score = scores[0]

    phi = (1.0 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            lo, hi = 0.0, 1.0
            c = hi - invphi * (hi - lo)
            d = lo + invphi * (hi - lo)

            mix_c = (1.0 - c) * a + c * b
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, fft_len, n)

            mix_d = (1.0 - d) * a + d * b
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, fft_len, n)

            while time.time() < deadline and (hi - lo) > 1e-4:
                if fc < fd:
                    hi, fd = d, fc
                    d = c
                    c = hi - invphi * (hi - lo)
                    mix_c = (1.0 - c) * a + c * b
                    mix_c /= mix_c.sum()
                    fc = fast_score(mix_c, fft_len, n)
                else:
                    lo, fc = c, fd
                    c = d
                    d = lo + invphi * (hi - lo)
                    mix_d = (1.0 - d) * a + d * b
                    mix_d /= mix_d.sum()
                    fd = fast_score(mix_d, fft_len, n)

            tm = 0.5 * (lo + hi)
            mix_tm = (1.0 - tm) * a + tm * b
            mix_tm /= mix_tm.sum()
            ftm = fast_score(mix_tm, fft_len, n)

            if ftm < best_score:
                best_score = ftm
                best_a = mix_tm.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (iteration of the “M‑decrease’’ scheme)
# ----------------------------------------------------------------------


def lp_refine(a_start: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int,
              ub_sum: float = 2.0,
              eps: float = 1e-12):
    """
    One‑step of the classic iteration:
        maximise Σg  s.t. a∗g ≤ M, g≥0, Σg ≤ ub_sum.
    The resulting g is mixed back into a with a back‑tracking line search
    that never increases the max‑convolution.
    Returns the best vector discovered during this refinement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    while time.time() < deadline:
        # ensure strict positivity to keep the LP bounded
        if (a < eps).any():
            a = np.maximum(a, eps)
            a = a / a.sum()

        # current max‑convolution bound
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # build the Toeplitz constraint matrix for a∗g ≤ M
        # each diagonal offset d (0 ≤ d < n) has constant value a[d]
        diag_data = [np.full(n, a[d]) for d in range(n)]
        offsets = np.arange(n)
        A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format='csr')
        # add the Σg ≤ ub_sum row
        A = sp.vstack([A, sp.csr_matrix(np.ones((1, n)))])
        b = np.concatenate([np.full(2 * n - 1, M), np.array([ub_sum])])

        # linear program: maximise Σg  → minimise –Σg
        c = -np.ones(n)
        bounds = [(0, None)] * n

        # solve – tighten the time limit to stay within the global deadline
        time_left = deadline - time.time()
        # high‑s performance: no explicit time limit for linprog,
        # but we break if we run out of budget.
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            # something went wrong – abort LP refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # cannot increase total mass without violating the bound → stop
            break

        # normalise g to Σg = 1  (the mixing step works on probability vectors)
        g_norm = g / sum_g

        # try a short line‑search for a safe mixing coefficient
        improved = False
        for t in [0.5, 0.25, 0.125, 0.0625, 0.03125]:
            new_a = (1.0 - t) * a + t * g_norm
            new_a = new_a / new_a.sum()
            new_score = fast_score(new_a, fft_len, n)

            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                improved = True

            # check that the max‑convolution does not rise
            fa_new = np.fft.fft(new_a, fft_len)
            conv_new = np.real(np.fft.ifft(fa_new * fa_new))[: 2 * n - 1]
            if conv_new.max() <= M + 1e-12:
                a = new_a
                break
        if not improved:
            # no progress – exit the refinement loop
            break

    return best_a, best_score

    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002, 0.00001
    ]
    steps_per_lam = 350

    for r in range(12):
        if time.time() >= deadline:
            break
        a0 = best_a.copy() if r == 0 else add_noise(best_a,
                                                    rng.uniform(0.001, 0.02), rng)
        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len,
            fft_len_g,
            n,
        )
        print(f"[Adam restart {r + 1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (true max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            scale=20.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after Exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = projected_subgrad_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            init_eta=0.7,
        )
        print(f"[Proj‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after proj‑subgrad: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = lp_refine(best_a, deadline, n, fft_len)
        print(f"[LP‑refine] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after LP refinement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = simulated_annealing_search(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            init_temp=1e-3,
            final_temp=1e-9,
            max_iter=6_000_000,
        )
        print(f"[Sim‑anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after anneal: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – Convex‑combination (mixture) search among top candidates
    # ------------------------------------------------------------------
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # keep the best K vectors
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    K = 5
    selected = ordered[:K]
    sel_scores, sel_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(sel_vectors),
            list(sel_scores),
            n,
            fft_len,
            deadline,
        )
        print(f"[Mixture] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for future runs
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
