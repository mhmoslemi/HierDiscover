def beta_init(n, alpha, beta_param):
    i = np.arange(n, dtype=float) + 0.5
    x = i / n
    a = x**(alpha-1) * (1-x)**(beta_param-1)
    a = np.nan_to_num(a, nan=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a.sum()
    return a

def random_block_init(n, rng):
    K = rng.integers(2,6)
    # choose K-1 cutpoints
    cuts = sorted(rng.choice(np.arange(1, n), size=K-1, replace=False))
    positions = [0] + cuts + [n]
    a = np.empty(n)
    total = 0
    for i in range(K):
        start, end = positions[i], positions[i+1]
        block_len = end-start
        weight = rng.random() + 0.1
        # assign uniform value for block: weight / block_len
        a[start:end] = weight / block_len
        total += weight
    a = a / a.sum()
    return a

def adam_optimize(a_start, deadline, n, fft_len, rng, lam_schedule, lr=0.1):
    y = np.log(a_start+1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    it=0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    schedule_len = len(lam_schedule)
    while time.time() < deadline:
        # determine current lambda based on progress
        lam = lam_schedule[min(it % schedule_len, schedule_len-1)]
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
        it += 1
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        y = y - lr * m_hat/(np.sqrt(v_hat)+eps)
        # optionally normalize as softmax automatically sum to 1
        a = softmax(y)
        # compute score periodically
        if it % 20 == 0:
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

"""
Propose a low‑score step sequence.

Algorithm
---------
1. Build a large pool of diverse initial candidates.

   • parametrised families (Kaiser, Gaussian, Beta, power‑increase)

5. Exact‑max sub‑gradient descent (projected onto the simplex) for a few

7. Multi‑stage greedy hill‑climb with decreasing step‑sizes.
8. Return the best sequence (list of floats).  All numbers are non‑negative,
   automatically normalised to sum = 1 (scaling does not affect the score).

The code respects the 1000 s budget, prints progress, and finally calls the

import time

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (sum=1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS/Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic, randomised, parametric families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def rising_triangle_init(n: int) -> np.ndarray:
    """Monotone increasing triangle (zero at left, max at right)."""
    a = np.arange(1, n + 1, dtype=float)
    a = a / a.sum()
    return a


def falling_triangle_init(n: int) -> np.ndarray:
    """Monotone decreasing triangle (max at left)."""
    a = np.arange(n, 0, -1, dtype=float)
    a = a / a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window (good sidelobe suppression)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window with a given beta."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay (heavier at the left side)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with a random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def power_increase_init(n: int, exp: float = 1.0) -> np.ndarray:
    """Increasing power law a_i ∝ (i+1)^exp."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** exp
    a = np.clip(a, 0.0, None)
    a = a / a.sum()
    return a


def beta_init(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """
    Discrete Beta distribution on {0,…,n‑1}.
    a_i ∝ x^{α‑1} (1‑x)^{β‑1},   x = (i+0.5)/n
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta_param - 1.0))
    a = np.nan_to_num(a, nan=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_block_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑constant blocks (mass moved in chunks)."""
    K = rng.integers(2, 6)  # number of blocks
    cuts = sorted(rng.choice(np.arange(1, n), size=K - 1, replace=False))
    positions = [0] + cuts + [n]
    a = np.empty(n)
    total_weight = 0.0
    for i in range(K):
        start, end = positions[i], positions[i + 1]
        block_len = max(1, end - start)
        # random block weight >0
        w = rng.random() + 0.1
        a[start:end] = w / block_len
        total_weight += w
    a = a / a.sum()
    return a


def smooth_variant(a: np.ndarray, rng: np.random.Generator, width: int = None) -> np.ndarray:
    """Convolve a with a small uniform kernel and renormalise."""
    n = a.size
    if width is None:
        width = rng.integers(3, 15)  # odd kernel size roughly
    if width % 2 == 0:
        width += 1
    kernel = np.ones(width, dtype=np.float64) / width
    # simple linear convolution, keep central part
    conv = np.convolve(a, kernel, mode='same')
    conv = np.maximum(conv, 0.0)
    conv /= conv.sum()
    return conv

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_schedule,
                  lr: float = 0.15,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8):
    """
    Adam optimiser on the smoothed max objective.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    it = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    # cyclical schedule – simply walk through lam_schedule repeatedly
    L = len(lam_schedule)

    while time.time() < deadline:
        lam = lam_schedule[it % L]
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
        it += 1

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 30 == 0:
            a = softmax(y)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a (2 * a[idx - j] where defined)
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    total_moves = max_moves

    while time.time() < deadline and move < total_moves:
        t_frac = move / total_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 50000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – keep it moderate (5000)
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an extensive pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous construction exists, interpolate it.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(rising_triangle_init(n))
    candidates.append(falling_triangle_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.45))
    candidates.append(gaussian_init(n, sigma_factor=0.48))
    candidates.append(gaussian_init(n, sigma_factor=0.5))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Tuning of Kaiser windows
    for beta in np.linspace(4.0, 12.0, 13):
        candidates.append(kaiser_init(n, beta))

    # Randomised families
    for _ in range(6):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(3):
        exp = rng.uniform(0.2, 2.5)
        candidates.append(power_increase_init(n, exp))
    for _ in range(4):
        candidates.append(random_block_init(n, rng))

    # Random Beta distributions (both increasing and decreasing)
    for _ in range(10):
        alpha = rng.uniform(0.2, 1.5)
        beta_param = rng.uniform(1.5, 5.0)   # decreasing
        candidates.append(beta_init(n, alpha, beta_param))
        # flipped version (increasing)
        candidates.append(beta_init(n, beta_param, alpha))

    # Apply a light smoothing to a subset of random candidates
    smoothed_candidates = []
    for a in candidates[:12]:
        smoothed = smooth_variant(a, rng, width=rng.integers(5, 13))
        smoothed_candidates.append(smoothed)
    candidates.extend(smoothed_candidates)

    # Evaluate pool and keep the best
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈40 % of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # blend the current best with a fresh random vector (helps escape local minima)
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4  # ≤0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on smoothed max (≈20 % of total time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        # a slowly descending λ‑schedule (more aggressive than in L‑BFGS)
        lam_sched_adam = [0.2, 0.12, 0.07, 0.04, 0.022, 0.012,
                          0.006, 0.003, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5]
        best_a, best_score = adam_optimize(best_a,
                                            phase2_end,
                                            n,
                                            fft_len,
                                            lam_sched_adam,
                                            lr=0.12)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact‑max sub‑gradient descent (≈15 % of total time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.75 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase3_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.35)
        print(f"[sub‑grad exact] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing (exploration)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 deadline,
                                                 n,
                                                 fft_len,
                                                 rng,
                                                 init_temp=5e-4,
                                                 final_temp=2e-7,
                                                 max_moves=150000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Multi‑stage hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # decreasing delta fractions give a coarse‑to‑fine search
        for delta in [0.06, 0.03, 0.015, 0.0075, 0.00375]:
            max_iter_stage = 30000
            best_a, best_score = hill_climb(best_a,
                                            best_score,
                                            fft_len,
                                            n,
                                            deadline,
                                            rng,
                                            max_iter=max_iter_stage,
                                            delta_frac=delta)
            print(f"[hill climb δ={delta:.4f}] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official score function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
