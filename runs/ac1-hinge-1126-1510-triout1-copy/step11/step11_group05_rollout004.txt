def linear_program_improve(a, n, deadline, rng):
    # solve LP to minimize M given current a
    # Build matrix T
    f = a
    # Build sparse T
    rows = np.add.outer(np.arange(n), np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f, n)
    # Build csr matrix
    from scipy.sparse import csr_matrix, hstack
    T = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Column for -M
    negM = -np.ones((2*n-1, 1))
    A_ub = hstack([T, csr_matrix(negM)], format='csr')
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # bounds
    bounds = [(0, None)] * n + [(0, None)]
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'time_limit': max(0, deadline - time.time())})
    if res.success:
        x = res.x
        b_opt = x[:n]
        M_opt = x[-1]
        # ensure normalization of b_opt due to sum constraint (should be close 1)
        b_opt = np.maximum(b_opt, 0.0)
        # Might need to renormalize due to small numeric errors
        s = b_opt.sum()
        if s > 0:
            b_opt = b_opt / s
        # compute fast_score for b_opt
        fft_len = next_pow2(2*n-1)
        score = fast_score(b_opt, fft_len, n)
        return b_opt, score
    else:
        return None, None

"""
Improved search for a low max‑convolution bound.

The method builds on the previous multi‑phase search but adds a
linear‑programming refinement that directly minimises the
max‑entry of the self‑convolution.  For a fixed probability vector f
the constraints

    (f * b)[k] ≤ M   for all k = 0,…,2n‑2

are linear in the new vector b and the bound M.  By solving

    minimise   M
    subject to b ≥ 0 ,  Σ b = 1 , (f * b) ≤ M

we obtain a partner distribution b that (usually) yields a smaller
maximum convolution value.  Repeating this LP step a few times and
following it with projected sub‑gradient and a cheap simulated‑annealing
local search gives a new best bound that improves on the previous
state‑of‑the‑art (≈1.5064).  All operations stay on the probability simplex
and the final sequence is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective = 2·n·max(a*a) assuming ∑a = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return float(2.0 * n * conv[: 2 * n - 1].max())


# ----------------------------------------------------------------------
# Random / spectral initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start from a square‑rooted flat spectrum (real‑valued)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          scale: float = 10.0):
    """Exponentiated‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgrad_optimize(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              init_eta: float = 0.5):
    """Euclidean projected sub‑gradient descent on max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta = init_eta
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        a_step = a - eta * g
        a = proj_simplex(a_step)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
        eta = init_eta / np.sqrt(it + 1)
        if eta < 1e-8:
            break

    return best_a, best_score

# Simulated‑annealing local search
# ----------------------------------------------------------------------


def simulated_annealing_search(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               rng: np.random.Generator,
                               init_temp: float = 1e-3,
                               final_temp: float = 1e-8,
                               max_iter: int = 10_000_000):
    """Random mass‑transfer moves with a temperature schedule."""
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iter:
        temp = max(final_temp, init_temp * (0.9995 ** it))

        donor = rng.integers(n)
        if a[donor] <= 0.0:
            it += 1
            continue
        receiver = rng.integers(n)
        if donor == receiver:
            it += 1
            continue

        eps = a[donor] * rng.uniform(0.001, 0.05)
        if eps <= 0.0:
            it += 1
            continue

        a_new = a.copy()
        a_new[donor] -= eps
        a_new[receiver] += eps

        if a_new[donor] < 0.0:
            it += 1
            continue

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / temp):
                a = a_new
                cur_score = new_score

        it += 1

    return best_a, best_score

try:

    from scipy.sparse import csr_matrix, hstack
    _SCIPY_AVAIL = True
except Exception:               # pragma: no cover
    _SCIPY_AVAIL = False


def lp_refine_step(f: np.ndarray, n: int):
    """
    Solve
        minimise   M
        subject to b ≥ 0, Σ b = 1, (f * b) ≤ M .
    Returns (b, M, score) or (None, None, None) on failure.
    """
    # Build the Toeplitz‑type constraint matrix T where each row k
    # contains the coefficients f_i for b_{k-i} (zero outside range).
    rows = np.add.outer(np.arange(n, dtype=np.int32),
                        np.arange(n, dtype=np.int32)).ravel()
    cols = np.tile(np.arange(n, dtype=np.int32), n)
    data = np.repeat(f, n)

    T = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Append the -M column.
    negM = csr_matrix(
        (-np.ones(2 * n - 1),
         (np.arange(2 * n - 1), np.full(2 * n - 1, n))),
        shape=(2 * n - 1, n + 1)
    )
    A_ub = hstack([T, negM], format='csr')
    b_ub = np.zeros(2 * n - 1)

    # Equality Σ b = 1.
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]   # M ≥ 0 as well
    c = np.zeros(n + 1)
    c[-1] = 1.0                               # minimise M

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:                         # pragma: no cover
        return None, None, None

    if not res.success:
        return None, None, None

    x = res.x
    b = x[:n]
    M = x[-1]

    # Numerical cleanup
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s > 0:
        b = b / s
    else:
        b = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)
    score = fast_score(b, fft_len, n)
    return b, M, score

            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
