"""
Improved step‑function optimiser (≈1000 s)

The optimiser works in three phases:

1. **Heavy EG‑search** – many short runs of mirrored exponentiated‑gradient
   descent starting from a very diverse pool (uniform, random families,
   Gaussian, cosine‑power, edge‑heavy, arcsine, bimodal).  Each run stops
   after a few seconds or when the global deadline is reached, keeping the
   best sequence seen.

2. **Fine L‑BFGS** – a cascade of L‑BFGS optimisations on the smooth
   log‑sum‑exp surrogate of the true max‑convolution, using a decreasing
   λ‑schedule that drives the surrogate arbitrarily close to the real
   max.

3. **Short anneal** – a Metropolis mass‑transfer annealer for the last
   few bits of improvement.

All vectors are kept on the probability simplex (sum = 1) so the
evaluation score reduces to `2·n·max(conv(a,a))`.  The code respects the
1000 s budget, prints intermediate scores and stores the final sequence
in the global variable `height_sequence_1`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    m_frac = rng.uniform(0.002, 0.02)
    m = max(1, int(n * m_frac))
    a = np.full(n, 1e-8)
    left_vals = rng.random(m) + 0.5
    right_vals = rng.random(m) + 0.5
    a[:m] = left_vals
    a[-m:] = right_vals
    a /= a.sum()
    return a


def arcsine_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Discretised arcsine law: a_i ∝ 1/√(x(1−x))."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    a = 1.0 / np.sqrt(x * (1.0 - x) + eps)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def bimodal_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Two Gaussian bumps at random locations."""
    pos1 = rng.integers(0, n)
    pos2 = rng.integers(0, n)
    width = rng.integers(max(1, n // 100), max(2, n // 20))
    sigma = width / 3.0
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - pos1) / sigma) ** 2)
    a += np.exp(-0.5 * ((idx - pos2) / sigma) ** 2)
    a = np.maximum(a, 1e-9)
    a /= a.sum()
    return a

# Smoothed log‑sum‑exp surrogate (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y lives in ℝⁿ; a = softmax(y) belongs to the probability simplex.
    The regulariser λ is scaled by the current max value.
    """
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)     # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirrored exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0,
                          max_iter: int = None):
    """
    Mirrored SGD (exponentiated‑gradient) using the exact L∞‑subgradient.
    Stops when the deadline is reached or when max_iter (if given) is hit.
    """
    a = a_start.copy()
    a /= a.sum()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        if max_iter is not None and it >= max_iter:
            break
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (enhanced)
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types:
    * mass transfer between two arbitrary positions
    * swapping two entries
    * local neighbour transfer (distance ≤ 5)
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25            # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:   # 'transfer'
                j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < a[j]:
                src, dst = j, i
            else:
                src, dst = i, j
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a tighter theoretical bound.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 0) previous construction (if present)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) classic deterministic families
    pool.append(np.full(n, 1.0 / n))                 # uniform
    pool.append(gaussian_initial(n, rng))
    pool.append(cosine_power_initial(n, rng))
    pool.append(edge_heavy_initial(n, rng))
    pool.append(arcsine_initial(n, rng))
    pool.append(bimodal_initial(n, rng))

    # 2) random families (several copies for diversity)
    for _ in range(8):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Initial best record
    # ------------------------------------------------------------------
    best_a = pool[0].copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[Init] best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Timing budget splits
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin
    phase1_end = start_time + 0.80 * budget_s       # heavy EG search

    # ------------------------------------------------------------------
    # Phase 1 – many short EG runs with random restarts
    # ------------------------------------------------------------------
    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        # occasionally inject a little noise
        if rng.random() < 0.15:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.02), rng=rng)

        # allocate a few seconds for this run (or whatever remains)
        remaining = phase1_end - time.time()
        if remaining < 5.0:
            break
        run_deadline = time.time() + min(10.0, remaining * 0.5)

        a_opt, sc_opt = exp_gradient_optimize(
            a_start,
            run_deadline,
            n,
            fft_len,
            rng,
            eta0=30.0
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[EG] new best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – fine L‑BFGS optimisation on a log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        lam_schedule = [
            0.4, 0.25, 0.15, 0.08, 0.04, 0.02,
            0.01, 0.005, 0.002, 0.001, 5e-4,
            2e-4, 1e-4, 5e-5, 2e-5, 1e-5,
            5e-6, 2e-6, 1e-6
        ]
        maxiter_per_lam = 1500
        a_opt, sc_opt = lbfgs_cascade(
            best_a,
            lam_schedule,
            maxiter_per_lam,
            deadline - 15.0,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS fine] new best fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – short simulated annealing for the final push
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(
            best_a,
            best_score,
            deadline,
            n,
            rng
        )
        print(f"[Anneal] fast_score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer and expose the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
