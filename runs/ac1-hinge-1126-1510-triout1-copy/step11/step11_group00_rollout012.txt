```
def tukey_init(n, alpha):
    i = np.arange(n)
    w = np.ones(n)
    if alpha <= 0:
        return np.full(n, 1.0 / n)
    if alpha >= 1:
        # Hann window shape
        return hann_init(n)
    period = (n - 1)
    edge = int(alpha * period / 2)
    # left taper
    if edge > 0:
        n_edge = np.arange(0, edge)
        w[:edge] = 0.5 * (1 + np.cos(np.pi * (2 * n_edge / (alpha * period) - 1)))
    # right taper
    if edge > 0:
        n_edge_rev = np.arange(0, edge)
        w[-edge:] = 0.5 * (1 + np.cos(np.pi * (2 * n_edge_rev / (alpha * period) - 1)))
    w /= w.sum()
    return w
```

def flattop_init(n):
    i = np.arange(n, dtype=np.float64)
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006947368
    w = (a0
         - a1 * np.cos(2*np.pi*i/(n-1))
         + a2 * np.cos(4*np.pi*i/(n-1))
         - a3 * np.cos(6*np.pi*i/(n-1))
         + a4 * np.cos(8*np.pi*i/(n-1)))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

"""
Propose a low‑score step function.

The algorithm builds a rich pool of window‑shaped candidates (Gaussian,
Kaiser, Tukey, Flattop, Hann, Blackman, triangular, spectral, random
Dirichlet‑like).  The best of those is taken as a starting point.

A multi‑phase optimisation then follows:

* Phase 1 – short L‑BFGS runs on a smoothed max‑convolution
  (log‑sum‑exp) with a continuation of smoothing parameters.
* Phase 2 – Adam optimiser on the exact max‑convolution (gradient
  obtained by FFT‑based back‑propagation).  A small Gaussian jitter is
  added regularly.
* Phase 3 – Simulated annealing with many mass‑transfer moves to escape
  possible plateaux.
* Phase 4 – strict greedy hill‑climb for final polishing.

All stages stop automatically when the overall time budget (default
1000 s) is exhausted.  The returned list is evaluated with the official
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast evaluation (sum=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.2, 5.0)
        a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalised to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def tukey_init(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey (tapered cosine) window."""
    if alpha <= 0:
        return np.full(n, 1.0 / n)
    if alpha >= 1:
        return hann_init(n)
    i = np.arange(n, dtype=np.float64)
    w = np.ones(n, dtype=np.float64)
    period = n - 1
    edge = int(alpha * period / 2.0)
    if edge > 0:
        # left taper
        n_edge = np.arange(edge, dtype=np.float64)
        w[:edge] = 0.5 * (1 + np.cos(np.pi * (2 * n_edge / (alpha * period) - 1)))
        # right taper (symmetrical)
        w[-edge:] = w[:edge][::-1]
    w /= w.sum()
    return w


def flattop_init(n: int) -> np.ndarray:
    """Standard 4‑term flat‑top window."""
    i = np.arange(n, dtype=np.float64)
    a0 = 0.21557895
    a1 = 0.41663158
    a2 = 0.277263158
    a3 = 0.083578947
    a4 = 0.006947368
    w = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1))
         - a3 * np.cos(6.0 * np.pi * i / (n - 1))
         + a4 * np.cos(8.0 * np.pi * i / (n - 1)))
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y so that a = softmax(y) stays on the simplex.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient of max conv w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # gradient of the objective (2·n·max_conv)
        grad_a = 2.0 * n * g_a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam bias‑corrected updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        # decaying learning rate – a bit more conservative than the
        # original 0.2/sqrt(t)
        lr = 0.12 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaux
        if t % 4000 == 0:
            y += rng.normal(scale=8e-5, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

    return best_a, best_score


def simulated_annealing_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 init_temp: float = 3e-4,
                                 final_temp: float = 1e-7,
                                 max_moves: int = 250000):
    """
    Simple Simulated Annealing with many mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # transfer a modest fraction of a[i] to a[j]
        delta = a[i] * rng.random() * 0.12
        if delta <= 0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 90000,
               delta_frac: float = 0.05):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))

        candidates.append(kaiser_init(n, beta))

    for alpha in np.linspace(0.0, 1.0, 11):
        candidates.append(tukey_init(n, alpha))

            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # --------------------------------------------------------------
    # Pick the best of the pool (fast_score is exact for sum=1)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS refinements (≈20 % of total time)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    # smoothing schedule – progresses down to a very small factor
    lam_schedule = [0.30, 0.20, 0.10, 0.05, 0.02, 0.01,
                    0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
                    5e-5, 2e-5, 1e-5]
    maxiter_per_lam = 800   # a bit more than the baseline

    while time.time() < phase1_deadline:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the exact max‑convolution (≈55 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_score = adam_optimize(best_a,
                                           deadline,
                                           n,
                                           fft_len,
                                           rng)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated Annealing for global exploration (≈20 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[SA] start simulated annealing")
        best_a, best_score = simulated_annealing_optimize(best_a,
                                                          deadline,
                                                          n,
                                                          fft_len,
                                                          rng,
                                                          init_temp=4e-4,
                                                          final_temp=5e-8,
                                                          max_moves=200000)
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (polish) (≈5 %)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 15.0)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        rng,
                                        max_iter=60000,
                                        delta_frac=0.04)
        print(f"[hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
