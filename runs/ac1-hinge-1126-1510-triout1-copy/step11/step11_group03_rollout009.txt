    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = np.exp(y - np.max(y))
    best_a = best_a / best_a.sum()
    best_score = fast_score(best_a, fft_len)
    t = 0
    while time.time() < deadline:
        t += 1
        # compute a
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        # symmetrize optionally
        a = (a + a[::-1]) / 2
        a = a / a.sum()
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # mask
        eps_conv = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_conv).astype(np.float64)
        # subgradient via FFT

        fm = np.fft.fft(mask, fft_len)

        avg = np.dot(a, g)
        grad_y = a * (g - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)

def objective_and_grad(y, n, fft_len, lam):
    # softmax
    m_max = np.max(y)
    e = np.exp(y - m_max)
    a = e / e.sum()
    # convolution
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c              
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient - compute w = exp_shift / Z (d smooth_max / d conv_i)
    w = exp_shift / Z
    # back-propagate
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y
```

    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    for lam in lam_sched:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam, 'gtol': 1e-12, 'disp': False})

"""
Optimiser for the non‑negative step‑function problem.

The algorithm proceeds in four stages

1. **Diverse random pool** – many Dirichlet / family samples,
   plus classic window functions (Chebyshev, Kaiser, Tukey, …).
   All vectors are normalised to the probability simplex.

2. **Aggressive L‑BFGS on a smoothed log‑sum‑exp surrogate**
   with a long λ‑schedule that ends at 1e‑5.
   The L‑BFGS runs are limited in time but use a fairly strict
   gradient tolerance.  After each L‑BFGS pass we enforce symmetry
   (a[i] ≈ a[n‑1‑i]) because the optimal solution is known to be
   symmetric.

3. **Log‑domain Adam on the exact ‖a∗a‖∞ sub‑gradient**
   (the true max‑convolution).  Working in log‑space guarantees
   positivity and the simplex constraint automatically.
   A standard Adam schedule (β₁=0.9, β₂=0.999) with a slowly
   decreasing step‑size provides a very fine‑grained descent.

4. **Short Metropolis‑type annealing** – local mass transfers,
   swaps and neighbourhood moves to polish the final answer.

The best sequence found is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) for a already normalised to sum = 1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a ≈ a[::-1] while preserving the sum."""
    a = (a + a[::-1]) / 2.0
    s = a.sum()
    if s == 0.0:
        return a
    return a / s


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector from a diverse set of styles."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(0.1 * n, 0.5 * n)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def cosine_power_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 3.0)
    centre = (n - 1) / 2.0
    cos_vals = np.cos(np.pi * (np.arange(n) - centre) / (n - 1))
    a = np.maximum(cos_vals, 0) ** p
    a /= a.sum()
    return a


def tukey_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    # Tukey window – if scipy is available we could use it, otherwise
    # build a simple analogue.
    alpha = rng.uniform(0.1, 0.8)
    t = np.arange(n) / (n - 1)
    w = np.ones_like(t)
    first = t < alpha / 2
    w[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1 - alpha / 2
    w[last] = 0.5 * (1 + np.cos(np.pi * (2 * (t[last] - 1) / alpha + 1)))
    a = np.maximum(w, 0.0)
    a /= a.sum()
    return a


def bartlett_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    a = 1.0 - np.abs(np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp) and its gradient.
    y lives in ℝⁿ; a = softmax(y) belongs to the probability simplex.
    The regulariser λ is scaled by the current max value.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)     # safe: exponent ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution entries
    w = exp_shift / Z                         # d smooth_max / d conv_i
    # back‑propagation through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]        # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """Run a cascade of L‑BFGS problems over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        a = symmetrize(a)
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lr0: float = 0.5):
    """
    Mirrored Adam optimiser (log‑domain) on the true
    L∞‑sub‑gradient of a∗a.
    """
    # initialise log‑variables from a symmetric version of a_start
    a_start = symmetrize(a_start)
    a_start = a_start / a_start.sum()
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = np.exp(y - np.max(y))
    best_a = best_a / best_a.sum()
    best_a = symmetrize(best_a)
    best_a = best_a / best_a.sum()
    best_score = fast_score(best_a, fft_len)

    t = 0
    while time.time() < deadline:
        t += 1

        # current a (always symmetric)
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        a = symmetrize(a)
        a = a / a.sum()

        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_conv = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_conv).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]      # size n

        # gradient w.r.t. log‑variables
        avg = np.dot(a, g)
        grad_y = a * (g - avg)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr0 / (np.sqrt(t) + 1.0)
        y = y - step * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate current symmetric a
        a_test = np.exp(y - np.max(y))
        a_test = a_test / a_test.sum()
        a_test = symmetrize(a_test)
        a_test = a_test / a_test.sum()
        score = fast_score(a_test, fft_len)

        if score < best_score:
            best_score = score
            best_a = a_test.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (enhanced)
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Simple Metropolis annealer with three move types:
    * mass transfer between two arbitrary indices,
    * swapping two entries,
    * local neighbour transfer (distance ≤ 5).
    Temperature is linearly cooled from 5 % of the current score to 0.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25            # max fraction of source mass moved

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]          # pure swap
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:  # 'transfer'
                j = rng.integers(0, n)
            if i == j:
                continue

            # move mass from larger to smaller entry
            src, dst = (i, j) if a[i] > a[j] else (j, i)
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue

            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        # keep feasibility (symmetry is optional – we keep it)
        cand = symmetrize(cand)
        cand = cand / cand.sum()

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # 0) Previous construction (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 1) Classic windows (if SciPy available)
    try:
        from scipy.signal import chebwin, kaiser, tukey, bartlett
    except Exception:
        chebwin = kaiser = tukey = bartlett = None

    if chebwin is not None:
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    if kaiser is not None:
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    if tukey is not None:
        for alpha in (0.2, 0.4, 0.6, 0.8):
            w = tukey(n, alpha=alpha)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    if bartlett is not None:
        w = bartlett(n)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        pool.append(w)

    # 2) Random families
    for _ in range(8):
        pool.append(random_initial(n, rng))

    # 3) Parametric families
    for _ in range(5):
        pool.append(gaussian_initial(n, rng))
        pool.append(cosine_power_initial(n, rng))
        pool.append(tukey_initial(n, rng))
        pool.append(bartlett_initial(n, rng))

    # 4) Dirichlet / Gamma draws (different concentration)
    for _ in range(6):
        alpha = rng.uniform(0.02, 5.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        pool.append(a)

    # --------------------------------------------------------------
    # Timing budget splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin
    phase1_end = start_time + 0.10 * budget_s       # quick random scan
    phase2_end = start_time + 0.45 * budget_s       # L‑BFGS aggressive phase
    phase3_end = start_time + 0.85 * budget_s       # Adam fine‑grained descent

    # --------------------------------------------------------------
    # Phase 1 – quick random exploration (fast_score)
    # --------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    while time.time() < phase1_end:
        a = rng.random(n)
        a /= a.sum()
        a = symmetrize(a)
        a = a / a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[Random] new best fast_score = {best_score:.9f}")

    # Include pool members as additional candidates
    for a0 in pool:
        a = symmetrize(a0)
        a = a / a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[Pool] new best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    lam_schedule = [

        0.0004, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5
    ]
    maxiter_per_lam = 800

    for restart in range(6):
        if time.time() >= phase2_end:
            break
        a_start = best_a.copy()
        # occasional random jitter to escape possible plateaus
        if rng.random() < 0.35:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.015), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(a_start,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase2_end,
                                      fft_len,
                                      n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam optimiser on the exact max‑convolution
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase3_end,
                                           n,
                                           fft_len,
                                           rng,
                                           lr0=0.6)
        print(f"[Adam] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – final annealing for last refinements
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(
            best_a,
            best_score,
            deadline,
            n,
            rng)
        print(f"[Anneal] fast_score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer and expose the result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # evaluate_sequence is provided
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
