"""
Optimise a non‑negative step‑function to minimise

    2·n·max(conv(a,a)) / (sum(a)²)

under the simplex constraint ∑a = 1, a_i ≥ 0.

The optimiser works in three stages:

1. **Diverse initialisation** – a mix of interpolated historic best,
   a spectral start, random draws and Beta‑shaped families.

2. **Adam on a log‑sum‑exp smooth approximation** – a long schedule of
   decreasing smoothing parameters and many optimiser steps.  The
   parameters are tuned for a much larger number of iterations than the
   reference implementation, allowing the solution to approach the true
   non‑smooth optimum.

3. **Exact sub‑gradient descent (exponentiated‑gradient)** – after the
   smooth phase we switch to the true max‑convolution objective, using
   an exponentiated‑gradient update on the probability simplex.  The
   step‑size decays with √t and occasional random perturbations help to
   escape plateaus.

A tiny golden‑section mixture step with the uniform distribution is
performed at the end, and the best vector is finally evaluated with the
provided `evaluate_sequence` routine.

The whole search is time‑boxed (default 1000 s) and adapts the amount
of work to the remaining budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective = 2·n·max(a*a) assuming ∑a = 1.
    Computes the convolution via FFT (real‑valued, but uses complex FFT for speed).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start from a square‑rooted flat spectrum (real‑valued)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def beta_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta‑family density on the discrete grid."""
    p = rng.uniform(0.2, 5.0)
    q = rng.uniform(0.2, 5.0)
    x = np.linspace(0.0, 1.0, n)
    # avoid singularities at the boundaries for p,q<1
    a = np.where((x == 0) | (x == 1),
                 0.0,
                 (x ** (p - 1)) * ((1 - x) ** (q - 1)))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed surrogate
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score) for the given time budget.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.07  # a slightly larger base learning rate for faster progress

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (exponentiated‑gradient) on the true max‑convolution
# ----------------------------------------------------------------------


def exact_exp_gradient_optimize(a_start: np.ndarray,
                                deadline: float,
                                n: int,
                                fft_len: int,
                                rng: np.random.Generator,
                                base_scale: float = 12.0):
    """
    Exponentiated‑gradient descent using the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    # The step size decays as scale / sqrt(t+1).  We also add occasional
    # random jitter to escape flat regions.
    while time.time() < deadline:
        # compute convolution (full)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max (sub‑gradient) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # exponentiated‑gradient update on the simplex
        eta = base_scale / np.sqrt(it + 1.0)
        # clip to avoid overflow
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum

        # occasional noise injection (every ~2000 iterations)
        if (it & 0x7FF) == 0:
            a = add_noise(a, rng.uniform(0.0005, 0.004), rng)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Simple golden‑section mixture with the uniform distribution
# ----------------------------------------------------------------------


def mix_with_uniform(a: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float):
    """
    Perform a cheap golden‑section line search over t∈[0,1] for the mixture
    (1−t)·a + t·u where u is uniform on the simplex.
    Returns the improved vector (or the original a if no improvement).
    """
    u = np.full_like(a, 1.0 / n)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    # initialise
    c = hi - invphi * (hi - lo)
    d = lo + invphi * (hi - lo)

    mix_c = (1.0 - c) * a + c * u
    mix_c /= mix_c.sum()
    fc = fast_score(mix_c, fft_len, n)

    mix_d = (1.0 - d) * a + d * u
    mix_d /= mix_d.sum()
    fd = fast_score(mix_d, fft_len, n)

    while time.time() < deadline and (hi - lo) > 1e-4:
        if fc < fd:
            hi, fd = d, fc
            d = c
            c = hi - invphi * (hi - lo)
            mix_c = (1.0 - c) * a + c * u
            mix_c /= mix_c.sum()
            fc = fast_score(mix_c, fft_len, n)
        else:
            lo, fc = c, fd
            c = d
            d = lo + invphi * (hi - lo)
            mix_d = (1.0 - d) * a + d * u
            mix_d /= mix_d.sum()
            fd = fast_score(mix_d, fft_len, n)

    tm = 0.5 * (lo + hi)
    mix_tm = (1.0 - tm) * a + tm * u
    mix_tm /= mix_tm.sum()
    ftm = fast_score(mix_tm, fft_len, n)

    if ftm < fast_score(a, fft_len, n):
        return mix_tm, ftm
    else:
        return a, fast_score(a, fft_len, n)
