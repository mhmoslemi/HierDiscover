    fft_len = next_pow2(2*n-1)

        # compute mask

            if nonzero.size == 0:

            dst = int(rng.integers(0, n))

"""
Improved step‑function optimizer.

The search proceeds in three stages:
1. A few short L‑BFGS bursts from random simplex points on a smoothed
   log‑sum‑exp approximation of the max‑convolution.  This yields a good
   starting distribution.
2. A long gradient‑guided hill‑climb that moves a fraction of the mass
   from the index with the largest sub‑gradient (most responsible for the
   current peak) to the index with the smallest sub‑gradient.  The update
   of the self‑convolution is performed *exactly* (including the quadratic
   δ² terms) and a simple simulated‑annealing acceptance rule allows
   occasional uphill moves to escape plateaus.
3. The final candidate is evaluated with the provided `evaluate_sequence`
   routine.

The algorithm works directly on the probability simplex, never lets
negative or overly large numbers appear, and adapts its step‑size and
temperature linearly with the remaining wall‑clock time.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) w.r.t. the logits y (a = softmax(y)).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. each conv entry
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1:n - 1 + n]  # derivative w.r.t. a (before softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact gradient‑guided hill‑climb with simulated annealing
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray, mask: np.ndarray,
                        n: int, fft_len: int) -> np.ndarray:
    """
    Sub‑gradient of max‑convolution at the current distribution a.
    mask is a binary array (length 2n‑1) with ones at the (near) maximal
    positions of conv(a,a).  Returns a vector of length n.
    """
    a_rev = a[::-1]
    g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                 np.fft.fft(a_rev, fft_len)))
    g = 2.0 * g_full[n - 1:n - 1 + n]
    return g


def hill_climb_exact(a_start: np.ndarray,
                     start_score: float,
                     deadline: float,
                     rng: np.random.Generator,
                     start_factor: float = 0.40,
                     end_factor: float = 0.001,
                     temp_start: float = 0.02,
                     temp_end: float = 1e-6,
                     recompute_interval: int = 3000) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb with *exact* convolution updates.
    Moves a fraction of mass from the index with the largest sub‑gradient
    to the index with the smallest sub‑gradient.  A simple SA rule
    (temperature decaying linearly) accepts occasional uphill moves.
    Returns the best vector found and its fast_score.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)                    # exact linear convolution
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    cur_max = conv.max()

    start_time = time.time()
    total_time = deadline - start_time
    accepted = 0
    steps = 0

    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) locate the (near‑)maximal positions of the convolution
        # --------------------------------------------------------------
        cur_max = conv.max()
        eps = cur_max * 1e-12 + 1e-12
        max_pos = np.where(conv >= cur_max - eps)[0]

        # --------------------------------------------------------------
        # 2) compute the sub‑gradient
        # --------------------------------------------------------------
        mask = np.zeros_like(conv)
        mask[max_pos] = 1.0
        g = compute_subgradient(a, mask, n, fft_len)

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # Degenerate case – fall back to a random move
        if src == dst or a[src] <= 0.0:
            nz = np.where(a > 1e-14)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz, p=a[nz] / a[nz].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # --------------------------------------------------------------
        # 3) step size and temperature (both decay linearly with time)
        # --------------------------------------------------------------
        rem_frac = max(0.0, (deadline - time.time()) / total_time)
        step_factor = end_factor + (start_factor - end_factor) * rem_frac
        delta = a[src] * step_factor
        if delta < 1e-15:
            continue
        delta = min(delta, a[src])

        temperature = temp_end + (temp_start - temp_end) * rem_frac

        # --------------------------------------------------------------
        # 4) apply an *exact* update to the convolution
        # --------------------------------------------------------------
        # copy of conv for tentative move
        conv_new = conv.copy()

        # linear part: -2*delta*a (src)  +2*delta*a (dst)
        conv_new[src:src + n] -= 2.0 * delta * a
        conv_new[dst:dst + n] += 2.0 * delta * a

        # quadratic δ² terms
        idx_2s = 2 * src
        idx_2d = 2 * dst
        idx_sd = src + dst
        if idx_2s < conv_new.size:
            conv_new[idx_2s] += delta * delta
        if idx_2d < conv_new.size:
            conv_new[idx_2d] += delta * delta
        if idx_sd < conv_new.size:
            conv_new[idx_sd] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # 5) acceptance test (downhill always, uphill with SA probability)
        # --------------------------------------------------------------
        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            # uphill – accept with probability exp(-Δ/temperature)
            delta_max = new_max - cur_max
            if rng.random() < math.exp(-delta_max / (temperature + 1e-15)):
                accept = True

        if accept:
            # commit the move
            conv = conv_new
            a[src] -= delta
            a[dst] += delta
            cur_score = new_score
            cur_max = new_max
            accepted += 1
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        # else: reject – state left unchanged

        steps += 1

        # --------------------------------------------------------------
        # 6) periodic re‑normalisation and exact recompute to avoid drift
        # --------------------------------------------------------------
        if accepted and accepted % recompute_interval == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s
            conv = np.convolve(a, a)

        # occasional progress dump (does not affect the harness)
        if steps % 20000 == 0:
            elapsed = time.time() - start_time
            print(f"[hill_climb] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f}")

    return best_a, best_score

    # ------------------------------------------------------------------
    # Initialise from a stored construction if available, otherwise uniform
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # linear interpolation to the required length
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS bursts from random starts
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-7, num=16)
    maxiter_per_lam = 300
    while time.time() < phase1_deadline:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_deadline, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact gradient‑guided hill‑climb with SA
    # ------------------------------------------------------------------
    if time.time() < deadline:
        start_sc = fast_score(best_a, fft_len)
        best_a, best_score = hill_climb_exact(
            best_a, start_sc, deadline, rng,
            start_factor=0.40, end_factor=0.001,
            temp_start=0.02, temp_end=1e-6,
            recompute_interval=3000
        )
        print(f"[hill_climb] final best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
