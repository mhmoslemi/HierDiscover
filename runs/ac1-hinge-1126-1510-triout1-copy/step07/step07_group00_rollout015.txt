"""
Propose a low‑evaluation step sequence.

The search works in three stages:

1. **Coarse optimisation (n≈1024).**  Many cheap L‑BFGS runs on a
   smoothed “log‑sum‑exp” surrogate of the max‑convolution are performed
   with random restarts.  The best coarse distribution is kept.

2. **Upscale to the target size (default n=4000).**  The coarse solution is
   linearly interpolated, normalised and mildly smoothed.

3. **Fine optimisation on the target size.**  The up‑scaled distribution
   is refined by a mixture of:
   * continuation L‑BFGS with a finer λ‑schedule,
   * projected sub‑gradient descent (with momentum),
   * Adam‑style adaptive gradient descent,
   * exponentiated‑gradient descent,
   * a long stochastic hill‑climb that moves mass between indices,
   * a deterministic “gradient‑guided” hill‑climb.

All phases respect the global 1000 s budget.  The best sequence found
(before the final call to `evaluate_sequence`) is returned.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    Σa = 1 (simplex).  The function works in O(n log n) time.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax parametrisation (guarantees positivity and Σa = 1)
    max_y = np.max(y)
    e_y = np.exp(y - max_y)
    a = e_y / e_y.sum()

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a * a) using FFT
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # The relevant part corresponds to indices (n‑1)…(2n‑2)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


# ----------------------------------------------------------------------
# Optimisation primitives (target‑size helpers)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            # should never happen, but stay safe
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ~ 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style adaptive gradient optimisation on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    base_lr = 0.5

    while time.time() < deadline:
        # convolution and mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of max convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam moments
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = simplex_projection(a - step)
        cand_score = fast_score(a_candidate, fft_len_conv, n)

        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # exponentiated‑gradient update
        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
        it += 1

    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that transfers a small amount of mass from the
    coordinate with the largest sub‑gradient (most harmful) to the one with the
    smallest sub‑gradient (most helpful).  Moves are kept only if they improve
    the objective.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # up to 10 % of the source mass is moved
        delta = best[i_hi] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the move
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def random_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_trials: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: repeatedly move a random fraction of mass from a
    randomly chosen source index to a random target index.  Only strictly
    improving moves are accepted.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    trials = 0

    while time.time() < deadline and trials < max_trials:
        i_src = rng.integers(0, n)
        if best[i_src] <= 1e-12:
            trials += 1
            continue
        i_dst = rng.integers(0, n)
        # move up to 15 % of the source mass
        delta = best[i_src] * rng.random() * 0.15
        if delta <= 0.0:
            trials += 1
            continue
        cand = best.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best_score, best = new_score, cand
        trials += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Global timing
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # tiny safety margin

    # ------------------------------------------------------------------
    # Problem sizes
    # ------------------------------------------------------------------
    n_target = 4000                    # final length
    fft_len_target = next_pow2(2 * n_target - 1)
    fft_len_g_target = next_pow2(3 * n_target - 2)

    # ------------------------------------------------------------------
    # Phase 0 – initialise with a uniformly random vector
    # ------------------------------------------------------------------
    best_a_target = np.full(n_target, 1.0 / n_target)
    best_score_target = fast_score(best_a_target, fft_len_target, n_target)
    print(f"[init‑target] uniform fast_score={best_score_target:.6f}")

    # ------------------------------------------------------------------
    # Optional: seed from a previously found sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n_target)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_target, n_target)
            if sc1 < best_score_target:
                best_a_target, best_score_target = a1, sc1
                print(f"[init‑prev] imported fast_score={best_score_target:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – coarse optimisation on a smaller problem (n≈1024)
    # ------------------------------------------------------------------
    n_small = 1024
    fft_len_small = next_pow2(2 * n_small - 1)
    fft_len_g_small = next_pow2(3 * n_small - 2)

    # initial uniform candidate on the small scale
    a_small = np.full(n_small, 1.0 / n_small)
    best_a_small = a_small.copy()
    best_score_small = fast_score(best_a_small, fft_len_small, n_small)

    # L‑BFGS schedule (same as before but a bit longer)
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001,
    ]
    maxiter_per_lam = 1500

    phase1_deadline = start_time + 0.30 * budget_s

    while time.time() < phase1_deadline:
        # random start (occasionally jitter around the current best)
        if rng.random() < 0.2:
            a_start = add_noise(best_a_small, sigma=rng.uniform(0.001, 0.01), rng=rng)
        else:
            a_start = random_initial(n_small, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_small,
            fft_len_g_small,
            n_small,
        )
        if sc_opt < best_score_small:
            best_score_small, best_a_small = sc_opt, a_opt.copy()
            print(f"[coarse‑lbfgs] new best fast_score={best_score_small:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to the target size and a brief polishing
    # ------------------------------------------------------------------
    x_small = np.linspace(0, 1, n_small)
    x_target = np.linspace(0, 1, n_target)
    a_upscaled = np.interp(x_target, x_small, best_a_small)
    a_upscaled = np.maximum(a_upscaled, 0.0)
    a_upscaled /= a_upscaled.sum()
    score_up = fast_score(a_upscaled, fft_len_target, n_target)
    if score_up < best_score_target:
        best_a_target, best_score_target = a_upscaled, score_up
        print(f"[upscale] fast_score={best_score_target:.6f}")

    # modest smoothing of the up‑scaled vector (helps later optimisation)
    a_upscaled = add_noise(best_a_target, sigma=1e-5, rng=rng)  # essentially a tiny jitter
    a_upscaled /= a_upscaled.sum()
    best_a_target = a_upscaled
    best_score_target = fast_score(best_a_target, fft_len_target, n_target)

    # ------------------------------------------------------------------
    # Phase 3 – fine optimisation on the target size
    # ------------------------------------------------------------------
    # split remaining budget
    remaining = deadline - time.time()
    phase3_end = time.time() + 0.55 * remaining
    phase3_mid = time.time() + 0.75 * remaining

    # 3a) L‑BFGS continuation (more aggressive λ schedule)
    lam_schedule_fine = lam_schedule + [5e-5, 2e-5, 1e-5, 5e-6, 2e-6]
    a_fine, sc_fine = lbfgs_continuation(
        best_a_target,
        lam_schedule_fine,
        maxiter_per_lam,
        phase3_end,
        fft_len_target,
        fft_len_g_target,
        n_target,
    )
    if sc_fine < best_score_target:
        best_score_target, best_a_target = sc_fine, a_fine.copy()
        print(f"[fine‑lbfgs] new best fast_score={best_score_target:.6f}")

    # 3b) Projected sub‑gradient descent with momentum
    if time.time() < phase3_mid:
        a_sub, sc_sub = subgradient_optimize_momentum(
            best_a_target, phase3_mid, fft_len_target, fft_len_g_target, n_target
        )
        if sc_sub < best_score_target:
            best_score_target, best_a_target = sc_sub, a_sub.copy()
            print(f"[sub‑grad‑mid] best fast_score={best_score_target:.6f}")

    # 3c) Adam optimiser (adaptive moments)
    if time.time() < phase3_end:
        a_adam, sc_adam = adam_optimize(
            best_a_target, phase3_end, fft_len_target, fft_len_g_target, n_target
        )
        if sc_adam < best_score_target:
            best_score_target, best_a_target = sc_adam, a_adam.copy()
            print(f"[adam] best fast_score={best_score_target:.6f}")

    # 3d) Exponentiated‑gradient descent
    if time.time() < phase3_end:
        a_exp, sc_exp = exp_gradient_optimize(
            best_a_target, phase3_end, fft_len_target, fft_len_g_target, n_target
        )
        if sc_exp < best_score_target:
            best_score_target, best_a_target = sc_exp, a_exp.copy()
            print(f"[exp‑grad] best fast_score={best_score_target:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – intensive stochastic + deterministic hill‑climbing
    # ------------------------------------------------------------------
    # allocate the rest of the time for hill‑climbing
    hill_deadline = deadline

    # deterministic gradient‑guided hill‑climb
    best_a_target, best_score_target = gradient_guided_hill_climb(
        best_a_target,
        best_score_target,
        fft_len_target,
        n_target,
        hill_deadline,
        max_iter=25000,
    )
    print(f"[grad‑hill] best fast_score={best_score_target:.6f}")

    # random hill‑climb (many cheap trials)
    best_a_target, best_score_target = random_hill_climb(
        best_a_target,
        best_score_target,
        fft_len_target,
        n_target,
        hill_deadline,
        max_trials=200000,
    )
    print(f"[random‑hill] best fast_score={best_score_target:.6f}")

    # final deterministic hill‑climb once more (may capture last improvements)
    best_a_target, best_score_target = gradient_guided_hill_climb(
        best_a_target,
        best_score_target,
        fft_len_target,
        n_target,
        hill_deadline,
        max_iter=30000,
    )
    print(f"[final‑grad‑hill] best fast_score={best_score_target:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a_target.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
