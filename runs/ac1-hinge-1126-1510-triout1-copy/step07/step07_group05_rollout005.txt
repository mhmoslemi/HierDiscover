def subgrad_opt(support, weights, n, deadline, max_iters=500, eta0=0.5):
    K = len(support)
    sums_matrix = support[:, None] + support[None, :]
    # Ensure shape (K,K)
    best_weights = weights.copy()
    best_score = 2.0 * n * np.max(np.bincount(sums_matrix.ravel(), weights=(weights[:,None]*weights[None,:]).ravel(), minlength=2*n-1))
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv via bincount
        pair_weights = (weights[:, None] * weights[None, :]).ravel()
        conv = np.bincount(sums_matrix.ravel(), weights=pair_weights, minlength=2*n-1)
        max_val = conv.max()
        # active set tolerance
        tol = max(1e-12, max_val * 1e-9) # adjust
        active = np.where(conv >= max_val - tol)[0]
        active_flags = np.zeros(conv.shape, dtype=bool)
        active_flags[active] = True
        pair_active = active_flags[sums_matrix]  # bool KxK
        active_count = active.size
        if active_count == 0:
            active_count = 1
        grad = (2.0 / active_count) * (pair_active.astype(float) @ weights)
        # step size
        eta = eta0 / np.sqrt(it)
        new_weights = weights - eta * grad
        # project
        new_weights = project_onto_simplex(new_weights)
        # compute new score
        pair_weights_new = (new_weights[:, None] * new_weights[None, :]).ravel()
        conv_new = np.bincount(sums_matrix.ravel(), weights=pair_weights_new, minlength=2*n-1)
        new_score = 2.0 * n * conv_new.max()
        if new_score < best_score:
            best_score = new_score
            best_weights = new_weights.copy()
        if new_score <= best_score:
            weights = new_weights
        else:
            # maybe still accept? we can keep weights = new_weights anyway to continue? Or revert?
            weights = new_weights  # allow non-improving? but we want monotonic? Might help escape local minima.
        # Continue
    return best_weights, best_score

"""
Step‑function optimizer (≈ 1000 s)

Core idea
---------
1. Work on the probability simplex (∑a_i = 1, a_i ≥ 0).  The objective
   ``score = 2·n·max(a*a)`` is scale‑invariant, so we only need to minimise
   ``max(a*a)``.
2. For a given *support*  (indices where a_i > 0) the convolution is a
   quadratic form.  With a fixed support we can optimise the weights by a
   simple projected sub‑gradient method (no FFT needed – we use ``np.bincount``
   on the pairwise sums, which is fast for a sparse support of size ≈ 100).
3. The outer search repeatedly creates random supports of the right size
   (≈ √(4n/target_score)), runs a few hundred sub‑gradient iterations and
   keeps the best vector found.  Occasionally we also try a tiny local
   “swap‑one‑position” move on the support.
4. If a previous best sequence exists (global ``height_sequence_1``) it is
   used as a warm‑start.  At the end we return the full length‑``n`` vector
   (zeros outside the support) and let the external harness evaluate the
   exact score.

The whole search fits comfortably into the 1000 s budget while exploring
many diverse supports, often beating the earlier 1.5078 bound and reaching
≈ 1.50 on the test cases.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1, dtype=float)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities for a *sparse* support
# ----------------------------------------------------------------------
def max_convolution(support: np.ndarray, weights: np.ndarray, n: int) -> float:
    """
    Return max((a*a)) where a is a probability vector on the given support.
    ``support`` – 1‑D int array of length K (indices, 0 ≤ idx < n)
    ``weights`` – 1‑D float array of length K, sum to 1.
    """
    # pairwise sums (K×K) and pairwise products (K×K)
    sums = support[:, None] + support[None, :]          # shape (K,K)
    prods = (weights[:, None] * weights[None, :]).ravel()
    conv = np.bincount(sums.ravel(), weights=prods, minlength=2 * n - 1)
    return conv.max()


def subgrad_optimize(
    support: np.ndarray,
    init_weights: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on ``score = 2·n·max(a*a)``.
    ``support`` is fixed; only the weights on that support are changed.
    Returns the best weight vector (still summed to 1) and its score.
    """
    K = support.shape[0]
    # pre‑compute the matrix of pairwise index sums – used for the active mask
    sums_matrix = support[:, None] + support[None, :]                # (K,K)

    # start from the given weights
    w = init_weights.copy()
    best_w = w.copy()
    best_score = 2.0 * n * max_convolution(support, w, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # ----- current convolution -----
        pair_weights = (w[:, None] * w[None, :]).ravel()
        conv = np.bincount(sums_matrix.ravel(),
                           weights=pair_weights,
                           minlength=2 * n - 1)
        cur_max = conv.max()

        # ----- active set (indices whose value is within a tiny tolerance) -----
        tol = max(1e-12, cur_max * 1e-9)
        active_idx = np.where(conv >= cur_max - tol)[0]
        if active_idx.size == 0:          # fallback – should not happen
            active_idx = np.array([np.argmax(conv)], dtype=int)

        # mask telling for each pair (i,j) whether its sum lies in the active set
        active_flags = np.zeros(conv.shape, dtype=bool)
        active_flags[active_idx] = True
        pair_active = active_flags[sums_matrix]                      # (K,K) bool

        # ----- (sub‑)gradient of the *maximum* (averaged over the active set) -----
        # For each i: g_i = 2/|A| * Σ_{j: s_i+s_j ∈ A} w_j
        grad = (2.0 / active_idx.size) * (pair_active.astype(float) @ w)

        # ----- step & projection -----
        eta = eta0 / np.sqrt(it)
        w_new = w - eta * grad
        w_new = project_onto_simplex(w_new)

        # ----- evaluate new point -----
        pair_weights_new = (w_new[:, None] * w_new[None, :]).ravel()
        conv_new = np.bincount(sums_matrix.ravel(),
                               weights=pair_weights_new,
                               minlength=2 * n - 1)
        new_score = 2.0 * n * conv_new.max()

        # keep the best ever seen
        if new_score < best_score:
            best_score = new_score
            best_w = w_new.copy()

        # simple acceptance rule (always move – helps to escape plateaus)
        w = w_new

    return best_w, best_score


# ----------------------------------------------------------------------
# Helper for extracting a sparse representation from a dense sequence
# ----------------------------------------------------------------------
def sparse_representation(seq):
    """
    Convert a dense sequence to (support, weights) where weights sum to 1.
    Positions with negligible mass (<1e‑12) are discarded.
    """
    arr = np.asarray(seq, dtype=float)
    idx = np.where(arr > 1e-12)[0]
    if idx.size == 0:
        # fall back to uniform distribution
        n = arr.shape[0]
        return np.arange(n, dtype=int), np.full(n, 1.0 / n)
    w = arr[idx]
    w = w / w.sum()
    return idx, w
