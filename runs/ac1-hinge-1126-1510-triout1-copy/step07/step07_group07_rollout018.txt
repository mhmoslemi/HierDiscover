"""
Propose a low‑evaluation step‑function.

The evaluator is
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

with non‑negative a_i and sum(a) > 0.  The objective is scale‑invariant,
so we work on the probability simplex (∑ a_i = 1) and minimise
    2 * n * max(conv(a, a)).

The search consists of three phases

1. **Rich initial pool** – analytic families (Beta, power‑law, exponential,
   sine‑power, semicircle, uniform, random mixtures).  Every prototype is
   normalised to sum = 1 and scored with a fast FFT‑based convolution.

2. **Continuation L‑BFGS** – the top few prototypes are refined by
   minimising a smoothed log‑sum‑exp surrogate of max(conv).  The smoothing
   parameter λ is decreased geometrically (≈ 30 continuation stages).  The
   unconstrained variables are the log‑softmax parameters; the simplex is
   guaranteed by the soft‑max.

3. **Projected sub‑gradient descent + aggressive hill‑climb** – after the
   L‑BFGS run the exact nonsmooth objective is tackled by sub‑gradient
   descent on the simplex (O(1/√t) step‑size).  Finally a short random /
   targeted hill‑climb polish is applied.

The whole pipeline respects the 1000 s budget, stores the final sequence
in the global ``height_sequence_1`` variable (so a later run can reuse the
length) and returns the best list of non‑negative floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize, minimize_scalar, differential_evolution, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n−1) using FFT.
    a is length n, fft_len ≥ 2·n−1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the evaluation function for a probability vector
    (∑ a_i = 1).  Returns 2·n·max(conv(a, a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate for max‑convolution (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective 2·n·smooth_max and its gradient w.r.t. the
    unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c            # makes the max become 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # gradient of conv w.r.t. a (correlation)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i  (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def beta_init(n: int, p: float) -> np.ndarray:
    """β(p,p) density on (0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def beta_general_init(n: int, p: float, q: float) -> np.ndarray:
    """β(p,q) density on (0,1)."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)  # ensure positivity
    return a / a.sum()


def binary_sparse_init(
    n: int, rng: np.random.Generator, density: float = 0.20
) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return a / a.sum()


def powerlaw_init(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-α)   (α > 0)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / (idx ** alpha)
    a = a / a.sum()
    return a


def exp_decrease_init(n: int, beta_: float) -> np.ndarray:
    """a_i ∝ exp(-β·i)."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-beta_ * idx)
    a = a / a.sum()
    return a


def sin_power_init(n: int, p: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^p  (zero at the boundaries)."""
    x = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * x) ** p
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised to sum = 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s

# Local search helpers (hill‑climbing / sub‑gradient)

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random greedy hill‑climb on the simplex.
    Moves a random fraction of mass from i → j.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from indices that take part
    in the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i = rng.integers(0, n)
        else:
            i = rng.choice(contributors)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    The step‑size decays as η₀/√t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        it += 1
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The function respects the supplied time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse length of a previous solution when reasonable.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 12000)
        else:
            n = 5000
    else:
        n = 5000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # ------------------------------------------------------------------
    # Build a broad pool of analytic candidates.
    # ------------------------------------------------------------------
    candidates = []

    # Uniform and semicircle (always good baselines)
    candidates.append(("uniform", uniform_init(n)))
    candidates.append(("semicircle", semicircle_init(n)))

    # β(p,p) – arcsine‑like shapes
    for p in np.linspace(0.51, 0.80, 15):
        candidates.append((f"beta_pp_{p:.3f}", beta_init(n, p)))

    # Full β(p,q) grid (asymmetric possibilities)
    p_vals = np.linspace(0.5, 1.0, 7)
    q_vals = np.linspace(0.5, 1.5, 7)
    for p in p_vals:
        for q in q_vals:
            candidates.append(
                (f"beta_{p:.2f}_{q:.2f}", beta_general_init(n, p, q))
            )

    # Power‑law tails
    for alpha in [0.5, 0.8, 1.0, 1.2, 1.5]:
        candidates.append((f"power_{alpha}", powerlaw_init(n, alpha)))

    # Exponential decrease
    for beta_ in [0.001, 0.002, 0.005, 0.010]:
        candidates.append((f"exp_{beta_:.3f}", exp_decrease_init(n, beta_)))

    # Sine‑power shapes (zero at the boundaries)
    for p in [1, 2, 4, 6]:
        candidates.append((f"sin_{p}", sin_power_init(n, p)))

    # Random Gaussian and sparse binary masks
    candidates.append(("random_gauss", random_gaussian_init(n, rng)))
    candidates.append(("binary_sparse", binary_sparse_init(n, rng, density=0.15)))

    # Single Gaussian bump
    candidates.append(("bump", gaussian_bump_init(n, rng)))

    # Simple mixtures of beta‑pp and uniform (different mixes)
    beta_ref = beta_init(n, 0.70)
    for w in [0.2, 0.4, 0.6, 0.8]:
        mixed = mixture_init(beta_ref, uniform_init(n), w)
        candidates.append((f"beta_mix_{w:.1f}", mixed))

    # If an historic sequence exists, add it (normalised)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            candidates.append(("previous", a))

    # ------------------------------------------------------------------
    # Fast screening – keep a handful of promising prototypes.
    # ------------------------------------------------------------------
    scored = []
    for name, a in candidates:
        a = a / a.sum()
        sc = fast_score(a, fft_len_conv, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:20s} fast score = {sc:.6f}")

    scored.sort(key=lambda x: x[0])
    top_k = 4                     # number of prototypes entering L‑BFGS
    top_candidates = scored[:top_k]

    best_a = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Continuation L‑BFGS on the smoothed log‑sum‑exp surrogate.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-11, num=36))
    maxiter_per_stage = 600
    lbfgs_deadline = start_time + 0.55 * budget_s
    print("[phase] L‑BFGS continuation started")

    for rank, (sc, name, a0) in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        print(f"[lbfgs] candidate #{rank+1} ({name}) start score = {sc:.6f}")
        y0 = np.log(a0 + 1e-12)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        print(f"[lbfgs] candidate #{rank+1} ({name}) final score = {score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[lbfgs] *** new global best = {best_score:.6f}")

        if time.time() > lbfgs_deadline:
            print("[phase] L‑BFGS time budget exhausted")
            break

    # Fallback – uniform if nothing succeeded (unlikely)
    if best_a is None:
        best_a = uniform_init(n)
        best_score = fast_score(best_a, fft_len_conv, n)

    # ------------------------------------------------------------------
    # Optional fine‑tuning of the best Beta parameterisation (2‑D search).
    # ------------------------------------------------------------------
    # As a cheap extra we try to improve a pure Beta(p,q) shape if it is the
    # current best.  The dimension is tiny, so a few hundred evaluations are
    # cheap.
    if best_a is not None:
        # Detect whether best_a resembles a Beta shape (heuristic)
        # We simply try a short Nelder‑Mead optimisation on (p,q) from the
        # current best prototype if its name contains “beta”.
        maybe_beta = any("beta" in name for _, name, _ in top_candidates)
        if maybe_beta:
            def beta_obj(params):
                p, q = params
                if p <= 0 or q <= 0:
                    return np.inf
                a = beta_general_init(n, p, q)
                return fast_score(a, fft_len_conv, n)

            # initialise at a known good point (p≈0.7,q≈0.7)
            x0 = np.array([0.70, 0.70])
            res = minimize(
                beta_obj,
                x0,
                method="Nelder-Mead",
                options={"maxiter": 300, "xatol": 1e-5, "fatol": 1e-5, "disp": False},
            )
            if res.success:
                a_refined = beta_general_init(n, res.x[0], res.x[1])
                score_refined = fast_score(a_refined, fft_len_conv, n)
                print(f"[beta‑refine] refined score = {score_refined:.6f}")
                if score_refined < best_score:
                    best_score = score_refined
                    best_a = a_refined.copy()
                    print("[beta‑refine] *** new global best after refinement ***")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (exact objective).
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 10.0, start_time + 0.85 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=600_000,
    )
    print(f"[subgrad] finished – best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive final polishing (random + targeted hill‑climbs).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=20000,
            max_move=0.12,
        )
        if time.time() < deadline:
            print("[final] targeted hill‑climb")
            best_a, best_score = targeted_hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=10000,
                max_move=0.10,
            )
        print("[final] polishing done")

    # ------------------------------------------------------------------
    # Final evaluation (using the provided evaluate_sequence).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)
        final_val = None

    # store for possible reuse in a subsequent run
    globals()["height_sequence_1"] = final_seq

    return final_seq
