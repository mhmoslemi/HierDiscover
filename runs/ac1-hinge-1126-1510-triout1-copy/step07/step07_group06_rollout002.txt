def annealed_hill_climb(a, current_score, fft_len, n, deadline):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    temperature = 0.1  # estimated scale of allowed increase
    # while time left:
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # pick delta proportion of source mass: random in (0, max_move)
        delta = best[i] * rng.random() * 0.2  # up to 20%
        if delta == 0: continue
        # propose
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        # recalc fast_score
        new_score = fast_score(new, fft_len, n)
        # Accept if lower or with prob exp(-(new_score-best_score)/temperature)
        if new_score < best_score:
            best = new
            best_score = new_score
            temperature *= 0.99  # cooling after improvement
        else:
            if rng.random() < np.exp(-(new_score - best_score)/temperature):
                best = new
                best_score = new_score
                temperature *= 0.995  # maybe slightly cooling also when accept?
            else:
                temperature *= 1.001  # maybe slightly warm up?

    return best, best_score

"""
Improved search for a low‑score non‑negative step function.

Algorithm overview
------------------
1. Choose a problem size `n`.  If a previous construction is available
   (`height_sequence_1`) we reuse its length (capped at 10000), otherwise
   we take `n = 8000`.

2. Initialise a probability vector `a` (sum = 1) from several heuristics
   (random, exponential, power‑law, edge‑heavy, arcsine, Sidon‑set,
   Dirichlet, U‑shape, uniform).  The current best solution is also used
   as a seed with a small Gaussian perturbation.

3. **Continuation Adam** – minimise a smooth log‑sum‑exp surrogate of
   `max(a⋆a)`.  The smoothing factor λ is decreased geometrically over
   many stages; for every λ we perform many Adam steps (gradient of the
   surrogate is computed analytically via FFT).  After each Adam run a
   short greedy hill‑climb is applied.

4. **Projected sub‑gradient descent** on the true objective
   `2·n·max(a⋆a)`.  The sub‑gradient is obtained by averaging the
   gradients of all almost‑active convolution entries, followed by a
   back‑tracking line‑search and Euclidean projection onto the simplex.

5. **Annealed hill‑climb** – a simulated‑annealing style local search
   that repeatedly moves a random fraction of mass from one index to
   another, accepting worsening moves with a temperature that decays
   over time.

6. The best vector found (already normalised) is returned; the harness
   evaluates the exact score via `evaluate_sequence`.

The search respects the 1000 s time budget by checking the wall‑clock at
every major step.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a logit vector (`y`) for a given heuristic `mode`."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a0 = 1.0 + p * (dist ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where

        f = 2·n·smooth_max(a*a)    (a = softmax(y))

    and grad_y is the gradient w.r.t. the unconstrained logits `y`.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) -----
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 20,
) -> np.ndarray:
    """
    Continuation Adam.  For each smoothing factor `lam_factor` we perform
    `steps_per_factor` Adam updates and finally return the logits.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)          # simple decaying schedule
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only moves that strictly improve the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses the average of the sub‑gradients of all (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # Determine entries that are essentially maximal
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # Gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):   # back‑track up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            # even the reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def annealed_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climb.  Randomly transfer a fraction of
    mass between two positions; worsening moves are sometimes accepted
    according to a temperature that decays over time.
    """
    best = a.copy()
    best_score = cur_score
    temperature = 0.04  # initial temperature (in objective units)
    min_temp = 1e-5
    cooling = 0.9995   # exponential cooling per iteration

    max_move_frac = 0.20  # up to 20 % of the source mass can be moved

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        # propose move
        new = best.copy()
        new[i] -= delta
        new[j] += delta

        new_score = fast_score(new, fft_len, n)

        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - best_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best = new
            best_score = new_score

        # cool down
        temperature = max(min_temp, temperature * cooling)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises `evaluate_sequence`.
    Returns a plain list of non‑negative floats (already normalised).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
            base_a = tmp / np.sum(tmp)          # use as initial probability vector
        else:
            n = 8000
            base_a = None
    else:
        n = 8000
        base_a = None

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate.
    # ------------------------------------------------------------------
    if base_a is not None and base_a.shape[0] == n:
        best_a = base_a.copy()
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam with continuation.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(10.0, 1e-10, num=120))
    adam_deadline = min(deadline, time.time() + (budget_s * 0.45))

    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "ushape", "uniform"
    ]

    iteration = 0
    while time.time() < adam_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            # never used – the list does not contain it
            y0 = best_y + rng.normal(scale=0.01, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = adam_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            steps_per_factor=25,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb after each Adam run (≈ 0.4 s)
        hc_deadline = min(adam_deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the true objective.
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, time.time() + (budget_s * 0.30))
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            subgrad_deadline,
            eta0=0.30,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Aggressive annealed hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[annealed HC] start")
        best_a, best_score = annealed_hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, rng
        )
        print(f"[annealed HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (the harness will also evaluate, but we do it once
    # to print the exact score).
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
