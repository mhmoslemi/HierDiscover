   a = a_start.copy()
   best = a.copy()

       moves += 1
       # linear cooling schedule: t = (deadline - time) / total_time

def simulated_annealing(a_start, cur_score_start, n, fft_len, deadline, max_moves):
    rng = np.random.default_rng()
    a = a_start.copy()
    cur_score = cur_score_start
    best_a = a.copy()
    best_score = cur_score
    # Temperature schedule: start T0 = cur_score * 0.1 maybe; T_end = cur_score * 1e-5
    T0 = cur_score * 0.05
    T_end = cur_score * 1e-6
    total = max_moves
    move_count = 0
    while time.time() < deadline and move_count < max_moves:
        move_count += 1
        # propose move
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # sample delta fraction
        delta = a[i] * rng.random() * 0.3
        # ensure delta > 0
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)

        # acceptance
        if new_score < cur_score:
            accept = True
        else:
            # compute temperature as linear cooling based on move count: T = T0 * (1 - move_count/total) + T_end * (move_count/total)
            t_frac = move_count / total
            T = T0 * (1 - t_frac) + T_end * t_frac
            # if T very small might cause overflow; ensure T > 0
            if T <= 0:
                T = 1e-12
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    return best_a, best_score

"""
Propose a step‑function (non‑negative heights) with a low
evaluate_sequence(score).  The algorithm combines three ideas:

1.  Many cheap random restarts that are refined by a continuation L‑BFGS
    optimisation of a smooth log‑sum‑exp surrogate of the max‑convolution.
2.  A projected sub‑gradient descent that directly works on the true
    max‑convolution objective.
3.  An aggressive simulated‑annealing phase that accepts occasional
    uphill moves, allowing the search to escape local minima.
   The whole pipeline respects the 1000 s time budget and returns the
   best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed to have length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of the max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) for the smoothed objective

        f = 2·n·smooth_max(a*a) ,   a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        proportion = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * proportion))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12  # allow up to 12 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] <= 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses an averaged sub‑gradient over all (almost) active convolution entries
    and a simple back‑tracking line search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                     # d(a_i a_j)/d a_i = 2 a_j

        # step size decreasing like 1/sqrt(it)
        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):                     # up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            # even the reduced step did not improve – keep the current a
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    cur_score_start: float,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive simulated‑annealing on the exact objective.
    A mass Δ is moved from a random source index to a random target.
    Moves that increase the objective are occasionally accepted according to
    a linear cooling schedule.
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    cur_score = cur_score_start
    best_a = a.copy()
    best_score = cur_score

    # Temperature schedule (linear cooling)
    T0 = cur_score * 0.05 + 1e-12
    T_end = cur_score * 1e-6 + 1e-12

    for move in range(1, max_moves + 1):
        if time.time() >= deadline:
            break

        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue

        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (up to 20 %) of the source mass
        delta = a[i] * rng.random() * 0.20
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)

        if new_score < cur_score:
            accept = True
        else:
            # linear interpolation of temperature
            t_frac = move / max_moves
            T = T0 * (1.0 - t_frac) + T_end * t_frac
            accept = rng.random() < np.exp(-(new_score - cur_score) / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            # use a larger size than the old sequence if possible
            n = n_default
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise the best candidate on the simplex.
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)   # uniform start
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(tmp, 0.0)
            s = best_a.sum()
            if s > 1e-12:
                best_a = best_a / s
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Parameters for the continuation L‑BFGS stage.
    # ------------------------------------------------------------------
    lam_schedule_main = list(np.geomspace(0.5, 1e-9, num=50))
    lam_schedule_refine = list(np.geomspace(0.5, 1e-12, num=80))

    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "best"
    ]

    # ------------------------------------------------------------------
    # Phase 1: many restarts with moderate L‑BFGS work.
    # ------------------------------------------------------------------
    while time.time() < deadline - 480.0:   # leave ~480 s for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_main,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb after each restart (≈ 1 s)
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2500
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 2: refined sub‑gradient descent.
    # ------------------------------------------------------------------
    if time.time() < deadline - 320.0:
        sub_deadline = deadline - 200.0
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.30,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: final high‑precision L‑BFGS (very low λ).
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        print("[refine L‑BFGS] start")
        y_opt = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_refine,
            maxiter_per_stage=1200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[refine L‑BFGS] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4: aggressive simulated annealing.
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print("[simulated annealing] start")
        cur_score = fast_score(best_a, fft_len_conv, n)
        best_a, best_score = simulated_annealing(
            best_a,
            cur_score,
            n,
            fft_len_conv,
            deadline - 30.0,
            max_moves=250_000,
        )
        print(f"[simulated annealing] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5: final hill‑climb (use whatever time remains).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
