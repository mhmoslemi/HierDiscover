"""
Propose a non‑negative step function (sequence) with a low evaluate_sequence score.

The algorithm works in three phases:

1. **Rich initialisation** – generate many heuristics (uniform,
   random Dirichlet, exponential with very small rates, power‑law,
   Hann/triangular windows, a tuned exponential, …) and keep the best
   candidate seen so far.

2. **Continuation Adam on a smooth log‑sum‑exp surrogate** – for each
   start point we run Adam for a long schedule of decreasing smoothing
   parameters.  The optimiser works on unrestricted logits `y`; the
   probability vector is obtained by `softmax(y)`.  This phase
   dramatically reduces the max‑convolution value.

3. **Projected sub‑gradient descent** – after the Adam stage we
   switch to the true non‑smooth objective.  The gradient is the
   averaged sub‑gradient of the `K` largest convolution entries
   (default `K=5`).  A simple back‑tracking line‑search on the step
   size is used; projected onto the probability simplex after every
   update.

A final aggressive hill‑climb moves small amounts of mass between
random coordinates while only accepting strictly improving moves.

The whole search respects a 1000 s wall‑clock budget: the Adam stage
gets ~45 % of the time, the sub‑gradient stage ~40 % and the final
hill‑climb the rest.  Intermediate scores are printed for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        score = 2·n·max(a*a) .
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the logits y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---------- smooth max (log‑sum‑exp) ----------
    shifted = conv - max_c          # max shifted to 0
    exp_shift = np.exp(shifted / lam)  # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient w.r.t. conv ----------
    w = exp_shift / Z               # ∂smooth_max/∂conv_i

    # ---------- back‑prop through convolution ----------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # indexes (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---------- chain rule through soft‑max ----------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 30,
) -> np.ndarray:
    """
    Continuation using Adam. For every smoothing factor `lam_factor`
    perform `steps_per_factor` Adam updates.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = smooth_obj_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)            # simple decay
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y

def init_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def init_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.gamma(0.4, 1.0, size=n)  # 0.4 gives a mildly sparse vector
    return a / a.sum()


def init_exp(n: int, rng: np.random.Generator, rate_low=1e-5, rate_high=0.02) -> np.ndarray:
    rate = rng.uniform(rate_low, rate_high)
    a = np.exp(-rate * np.arange(n, dtype=float))
    return a / a.sum()


def init_exp_optimal(n: int) -> np.ndarray:
    """
    Scan a geometric grid of rates (log‑space) and pick the one with the
    smallest fast_score.  The search is cheap (≈ 50 evaluations) and
    yields a good exponential start.
    """
    rates = np.geomspace(1e-5, 0.01, num=40)
    best_score = np.inf
    best_a = None
    for rate in rates:
        a = np.exp(-rate * np.arange(n, dtype=float))
        a /= a.sum()
        # rough score, cheap
        conv = conv_fft(a, next_pow2(2 * n - 1), n)
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a
    return best_a


def init_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.3, 2.0)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    return a / a.sum()


def init_hann(n: int) -> np.ndarray:
    """Classic Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=float)
    a = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def init_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mid = (n - 1) / 2.0
    a = 1.0 - np.abs(i - mid) / mid   # linear triangle, peak =1
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def init_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = (n - 1) / 2.0
    sigma = rng.uniform(n / 8.0, n / 3.0)
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a / a.sum()


def init_edge_heavy(n: int, rng: np.random.Generator) -> np.ndarray:
    """Large mass on the two ends, tiny mass in the centre."""
    m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
    a_edge = np.sqrt(n / 2.0) / n
    a_center = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, a_center)
    a[:m_edge] = a_edge
    a[-m_edge:] = a_edge
    return a / a.sum()


def init_sidon(n: int) -> np.ndarray:
    # greedy Sidon set – size ≈ √n
    chosen = []
    sums = set()
    for cand in range(n):
        conflict = False
        for c in chosen:
            if (cand + c) in sums or (2 * cand) in sums:
                conflict = True
                break
        if not conflict:
            chosen.append(cand)
            for c in chosen:
                sums.add(cand + c)
        if len(chosen) >= int(np.sqrt(n) * 1.5):
            break
    a = np.zeros(n)
    a[chosen] = 1.0 / len(chosen)
    return a / a.sum()


def init_ushape(n: int, rng: np.random.Generator) -> np.ndarray:
    """U‑shaped: large weights at the ends, small in the centre."""
    mid = (n - 1) / 2.0
    dist = np.abs(np.arange(n) - mid) / mid
    p = rng.uniform(0.5, 4.0)
    a = 1.0 + p * (dist ** 2)
    return a / a.sum()

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j,
    accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12   # up to 12 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
    top_k: int = 5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Uses the average of the gradients of the `top_k` largest entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)

        # pick the top‑k indices (ties are fine)
        idx = np.argpartition(conv, -top_k)[-top_k:]
        w = np.zeros_like(conv)
        w[idx] = 1.0 / top_k

        # gradient of weighted max w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        eta = eta0 / np.sqrt(it)

        # simple back‑tracking line search (up to 6 halvings)
        improved = False
        for _ in range(6):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-13:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            # still move (may escape plateaus)
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score

            max_iter=25000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
