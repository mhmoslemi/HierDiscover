def propose_candidate(seed=..., budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # choose n_small; maybe base on memory; set n_small = 2048 or 4096 (should be <8192 for speed)
    n_small = 4096  # ensure even.

    fft_len_small = next_pow2(2*n_small-1)

    # initial families
    init_candidates = []

    # Beta p across many values inclusive 0.4 to 2.0
    for p in np.linspace(0.4, 1.5, 12):
        a = beta_init(n_small, float(p))
        init_candidates.append((f'beta_{p:.3f}', a))

    # Uniform
    init_candidates.append(('uniform', uniform_init(n_small)))

    # Random Gaussian
    for _ in range(3):
        a = random_gaussian_init(n_small, rng)
        init_candidates.append(('randgaus', a))

    # binary sparse
    a = binary_sparse_init(n_small, rng, density=0.2)
    init_candidates.append(('binary_sparse', a))

    # possibly mixture of best beta and uniform
    # compute after we have best? We'll later.

    best_small = None
    best_score_small = np.inf
    best_y_small = None

    # Quick scan
    for name, a in init_candidates:
        # if time low break
        if time.time() > deadline: break
        sc = fast_score(a, fft_len_small, n_small)
        if sc < best_score_small:
            best_score_small, best_small = sc, a.copy()
            best_y_small = np.log(best_small + 1e-12)

    # L-BFGS continuation for top few candidates: maybe top K = 5
    # get sorted scores
    scored = sorted(((fast_score(a, fft_len_small, n_small), name, a) for name, a in init_candidates), key=lambda x:x[0])
    top_k = min(5, len(scored))
    lam_sched = list(np.geomspace(5.0, 1e-8, num=25))

    for rank in range(top_k):
        if time.time() > deadline: break
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(y0, n_small, fft_len_small, next_pow2(3*n_small-2), lam_sched, maxiter_per_stage=500)
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_small, n_small)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_small = a_opt.copy()
            best_y_small = y_opt.copy()
            print(f"[small stage] new best from {name} score = {best_score_small}")

    # Random restarts while time:
    while time.time() < deadline:
        # random gaussian init
        a0 = random_gaussian_init(n_small, rng)
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(y0, n_small, fft_len_small, next_pow2(3*n_small-2), lam_sched[:15], maxiter_per_stage=300)
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_small, n_small)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_small = a_opt.copy()
            best_y_small = y_opt.copy()
            print(f"[small restart] improved score = {best_score_small}")

        # safety break if many restarts
        if time.time() > deadline:
            break

    # Upsample to larger n
    n_large = 12000  # within budget, maybe 12000.
    if n_large < n_small:
        n_large = n_small
    # function upsample a
    a_large = upsample_shape(best_small, n_large)
    # Re-normalize sum to 1: it's already.
    fft_len_large = next_pow2(2*n_large - 1)
    # Fine tune L-BFGS on large n
    lam_sched_large = list(np.geomspace(5.0, 1e-10, num=30))
    y_large = np.log(a_large + 1e-12)
    # allocate some time for final L-BFGS
    # compute remaining time fraction
    remaining = deadline - time.time()
    # If remaining > 30 sec, run L-BFGS with moderate iterations.
    if remaining > 30:
        # decide iterations per stage based on remaining
        per_iter = max(300, int(remaining * 0.8 / len(lam_sched_large) / 2))
        # ensure at least 100
        per_iter = max(100, per_iter)
        y_opt_large = schedule_optimize(y_large, n_large, fft_len_large, next_pow2(3*n_large - 2), lam_sched_large, maxiter_per_stage=per_iter)
        a_opt_large = softmax(y_opt_large)
    else:
        a_opt_large = a_large

"""
Propose a low‑evaluation step‑function.

Algorithm outline
-----------------
1. Work on the probability simplex (∑a_i = 1).  The objective
   `evaluate_sequence` becomes 2·n·max(conv(a,a)) and is scale‑invariant.
2. Search first on a modest size (n_small ≈ 3 k).  This allows many
   independent L‑BFGS runs with a smoothed “log‑sum‑exp” surrogate of the
   max‑convolution.  A geometric schedule of smoothing parameters λ reduces
   from very smooth to almost exact.
3. Keep the best small‑scale shape, then interpolate it to a larger
   problem size n_large (≈ 8 k).  The interpolation preserves the shape
   while providing enough degrees of freedom for a finer optimisation.
4. Run a second L‑BFGS continuation on the large problem (full λ‑schedule
   and more iterations per stage) to obtain a high‑quality local optimum.
5. Finish with an aggressive random hill‑climb on the true objective
   (`fast_score`).  All steps respect a total runtime of ≤ budget_s
   seconds and the best sequence found is returned as a plain list of
   non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) via FFT.
    a has length n, fft_len ≥ 2*n-1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Evaluate the objective for a probability vector (∑a=1):
    2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate objective (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max(conv) and its gradient w.r.t.
    the unconstrained logits y (softmax parametrisation).

    lam = max(conv) * lam_factor  (a small smoothing constant).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z

    # correlation of a and w (derivative of conv w.r.t. a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Analytic initialisers
# ----------------------------------------------------------------------


def beta_init(n: int, p: float) -> np.ndarray:
    """Beta(p,p) density on (0,1); p<1 gives a U‑shape, p>1 a hump."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal style random start."""
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.15) -> np.ndarray:
    """Sparse binary mask with the given density."""
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def upsample_shape(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation of a_small (which sums to 1) onto a larger grid.
    The result is renormalised to sum to 1.
    """
    n_small = len(a_small)
    x_small = np.arange(n_small) / n_small
    x_large = np.arange(n_large) / n_large
    a_large = np.interp(x_large, x_small, a_small, left=0.0, right=0.0)
    s = a_large.sum()
    if s == 0.0:
        a_large = uniform_init(n_large)
        s = 1.0
    a_large = a_large / s
    return a_large


# ----------------------------------------------------------------------
# Simple hill‑climbing on the true objective
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Greedy random hill‑climb that conserves the simplex constraint."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a fraction of mass from i to j
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with minimal evaluation score.
    Returns a list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Phase 1 – cheap optimisation on a modest grid
    # ------------------------------------------------------------------
    n_small = 3072                     # fast enough for many restarts
    fft_len_small = next_pow2(2 * n_small - 1)
    fft_len_g_small = next_pow2(3 * n_small - 2)

    # generate a diverse pool of seeds
    seeds = []

    # Beta(p,p) for a dense range of p
    for p in np.linspace(0.40, 1.20, 9):
        seeds.append((f"beta_{p:.2f}", beta_init(n_small, float(p))))

    seeds.append(("uniform", uniform_init(n_small)))
    seeds.append(("rand_gauss", random_gaussian_init(n_small, rng)))
    seeds.append(("binary", binary_sparse_init(n_small, rng, density=0.18)))

    # ------------------------------------------------------------------
    # Quick scan – keep the best raw seed
    # ------------------------------------------------------------------
    best_small = None
    best_score_small = np.inf
    best_y_small = None

    for name, a in seeds:
        if time.time() > deadline:
            break
        sc = fast_score(a, fft_len_small, n_small)
        if sc < best_score_small:
            best_score_small = sc
            best_small = a.copy()
            best_y_small = np.log(best_small + 1e-12)
            print(f"[scan] new best raw seed {name:12s} score = {sc:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS continuation on the most promising candidates
    # ------------------------------------------------------------------
    scored = [(fast_score(a, fft_len_small, n_small), name, a) for name, a in seeds]
    scored.sort(key=lambda x: x[0])
    top_k = min(3, len(scored))

    lam_schedule_small = list(np.geomspace(5.0, 1e-9, num=25))
    maxiter_stage_small = 400

    for rank in range(top_k):
        if time.time() > deadline:
            break
        _, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(
            y0,
            n_small,
            fft_len_small,
            fft_len_g_small,
            lam_schedule_small,
            maxiter_per_stage=maxiter_stage_small,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_small, n_small)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_small = a_opt.copy()
            best_y_small = y_opt.copy()
            print(f"[l‑bfgs] improved from {name:12s} score = {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Random‑restart L‑BFGS while time permits (small problem)
    # ------------------------------------------------------------------
    restart_cnt = 0
    while time.time() < deadline - 30:  # keep ~30 s for the large‑scale phase
        restart_cnt += 1
        a0 = random_gaussian_init(n_small, rng)
        y0 = np.log(a0 + 1e-12)
        # a shorter schedule for speed
        lam_sched_r = lam_schedule_small[:12]
        y_opt = schedule_optimize(
            y0,
            n_small,
            fft_len_small,
            fft_len_g_small,
            lam_sched_r,
            maxiter_per_stage=250,
        )
        a_opt = softmax(y_opt)
        sc_opt = fast_score(a_opt, fft_len_small, n_small)
        if sc_opt < best_score_small:
            best_score_small = sc_opt
            best_small = a_opt.copy()
            best_y_small = y_opt.copy()
            print(f"[restart #{restart_cnt:03d}] improved score = {sc_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample to the target size and fine‑tune
    # ------------------------------------------------------------------
    n_large = 8000
    a_large = upsample_shape(best_small, n_large)
    fft_len_large = next_pow2(2 * n_large - 1)
    fft_len_g_large = next_pow2(3 * n_large - 2)

    lam_schedule_large = list(np.geomspace(5.0, 1e-9, num=30))
    # adapt iteration budget to remaining time
    remaining = deadline - time.time()
    if remaining > 100:
        maxiter_stage_large = 800
    elif remaining > 50:
        maxiter_stage_large = 500
    else:
        maxiter_stage_large = 300

    y_large = np.log(a_large + 1e-12)
    y_opt_large = schedule_optimize(
        y_large,
        n_large,
        fft_len_large,
        fft_len_g_large,
        lam_schedule_large,
        maxiter_per_stage=maxiter_stage_large,
    )
    a_opt_large = softmax(y_opt_large)
    best_score_large = fast_score(a_opt_large, fft_len_large, n_large)
    print(f"[final L‑BFGS] large‑scale score = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # Aggressive hill‑climb on the true objective (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt_large, best_score_large = hill_climb(
            a_opt_large,
            best_score_large,
            fft_len_large,
            n_large,
            deadline,
            max_iter=30000,
            max_move=0.25,
        )
        print(f"[final hill‑climb] improved score = {best_score_large:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain list)
    # ------------------------------------------------------------------
    final_seq = a_opt_large.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final evaluate] reported score = {final_val:.6f}")
    except Exception as e:
        print("[final evaluate] error:", e)

    return final_seq
