def adam_subgradient_optimize(a0, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    alpha = 0.05 # base step size
    it = 1
    while time.time() < deadline:
        # compute subgradient
        conv = full_conv(a, fft_len_conv)[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n-1):(n-1+n)]

        # Adam updates
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + epsilon)
        a_new = a - step * m_hat
        # project onto simplex
        a_new = simplex_projection(a_new)

        cur_score = fast_score(a_new, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score

def aggressive_hill_climb(a, current_score, fft_len_g, n, deadline, max_iter=200000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute gradient and max conv
        conv = full_conv(best, fft_len_g)[:2*n-1]  # use fft_len_g as longer
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n-1):(n-1+n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # attempt moving mass
        delta = best[i_hi]
        # test series of deltas
        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            d = delta * factor
            if d < 1e-12:
                break
            candidate = best.copy()
            candidate[i_hi] -= d
            candidate[i_lo] += d
            # ensure non-negative
            if candidate[i_hi] < -1e-12:
                continue
            # Projection not needed because we kept sum constant.
            cand_score = fast_score(candidate, fft_len_g, n)
            if cand_score < best_score - 1e-12:
                best = candidate
                best_score = cand_score
                improved = True
                break
        if not improved:
            # if no improvement, maybe try a random small move
            # small random transfer
            delta_small = best[i_hi] * rng.random() * 0.02
            if delta_small > 0:
                cand = best.copy()
                cand[i_hi] -= delta_small
                cand[i_lo] += delta_small
                cand_score = fast_score(cand, fft_len_g, n)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score
        it += 1
    return best, best_score

2.  Random‑restart L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution, with a decreasing temperature schedule.
3.  Projected Adam‑style stochastic sub‑gradient descent on the true
    max‑convolution.
4.  An aggressive deterministic hill‑climb that repeatedly moves a
    (large) fraction of mass from the coordinate with the highest
    sub‑gradient to the one with the lowest.  Only improving moves are
    kept.

All phases respect a global deadline; the best sequence found so far
is returned together with its true evaluation using the supplied
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)−1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  Returns the numerator of the
    original objective.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def adam_subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient descent on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.06  # base learning rate
    it = 1

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha / (np.sqrt(v_hat) + eps)
        a_new = a - step * m_hat
        a_new = simplex_projection(a_new)

        cur_score = fast_score(a_new, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
        it += 1

    return best_a, best_score


def aggressive_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a (large) fraction of mass from the
    coordinate with the largest sub‑gradient to the one with the smallest.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        conv = full_conv(best, fft_len_conv)[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # try moving a decreasing geometric series of fractions
        delta_total = best[i_hi]
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            d = delta_total * factor
            if d < 1e-14:
                break
            cand = best.copy()
            cand[i_hi] -= d
            cand[i_lo] += d
            # sum stays 1, non‑negativity preserved by construction
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score - 1e-12:
                best = cand
                best_score = cand_score
                improved = True
                break
        if not improved:
            # a small random nudge as a last resort
            d_small = best[i_hi] * rng.random() * 0.02
            if d_small > 1e-14:
                cand = best.copy()
                cand[i_hi] -= d_small
                cand[i_lo] += d_small
                cand_score = fast_score(cand, fft_len_conv, n)
                if cand_score < best_score:
                    best = cand
                    best_score = cand_score
        it += 1

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (larger n gives more degrees of freedom)
    # ------------------------------------------------------------------
    n = 8192  # 2ⁿ⁺¹, FFT length stays at 16384
    fft_len_conv = next_pow2(2 * n - 1)          # = 16384
    fft_len_g = next_pow2(3 * n - 2)            # = 32768

    # ------------------------------------------------------------------
    # Initial candidate (uniform or up‑sampled historic solution)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (power, exponential, cosine‑square)
    # ------------------------------------------------------------------
    # power‑law families (both increasing and decreasing)
    for p in np.linspace(-2.0, 3.0, 13):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[sweep] power {p:.2f} -> {best_score:.6f}")

        a = ((np.arange(n, 0, -1, dtype=float) ** p))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[sweep] rev‑power {p:.2f} -> {best_score:.6f}")

    # exponential families
    for lam in np.logspace(-4, -0.5, 9):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[sweep] exp {lam:.6f} -> {best_score:.6f}")

    # cosine‑square (zero at the ends, high in the middle)
    x = np.linspace(0, np.pi, n)
    a = np.sin(x) ** 2
    a /= a.sum()
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_a, best_score = a, sc
        print(f"[sweep] cos² -> {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # Phase split (roughly 35 % L‑BFGS, 35 % Adam sub‑gradient, 30 % hill‑climb)
    phase1_end = start_time + 0.35 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    # remaining time goes to phase 3

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005,
        0.0002, 0.0001, 5e-05, 2e-05, 1e-05,
    ]
    maxiter_per_lam = 600  # moderate number of inner L‑BFGS iterations

    while time.time() < phase1_end:
        # more frequent perturbations keep us from being stuck
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.02), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam‑style sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_subgradient_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam sub‑grad] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = aggressive_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
            max_iter=200000,
        )
        print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
