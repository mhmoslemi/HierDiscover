def directed_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000, step_max=0.1):
    for iter:
        compute conv = conv_fft(a)
        m = argmax(conv)
        # compute contributions c_i = a_i * a_{m-i} if valid else 0
        # compute "impact" of moving mass from i: derivative w.r.t a_i is 2*a_{m-i} (g_i)
        # So choose i with large g_i => large contribution.
        # Choose j with small g_j, maybe zero.
        # Use probability distribution based on g_i.
        g = np.zeros_like(a)
        counter = m - np.arange(n)
        valid = (counter>=0) & (counter<n)
        g[valid] = a[counter[valid]]
        # pick i with probability proportional to g_i (or g_i^2)
        probs_i = g / (g.sum() + 1e-12)
        i = rng.choice(n, p=probs_i)
        # pick j uniformly random among indices where g_j is low
        low_indices = np.where(g < np.percentile(g, 30))[0]
        if low_indices.size == 0:
            j = rng.integers(n)
        else:
            j = rng.choice(low_indices)
        # compute delta = a[i] * random * step_max ensuring a[i] not become negative

        a[i] -= delta
        a[j] += delta
        new_score = fast_score(a)
        if new_score < cur_score:
            cur_score = new_score
        else:
            # revert
            a[i] += delta
            a[j] -= delta

"""
Hybrid optimisation for the step‑function minimisation problem.

The routine works on the probability simplex (∑a = 1) because the
objective 2·n·max(conv) is scale‑invariant.  The search proceeds in four
phases:

1.  Multi‑restart L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution.  In addition to the classic initialisations a
    “beta” family (a_i ∝ x_i^{p‑1}(1‑x_i)^{p‑1}) and a simple “tent”
    shape are explored.  The continuation schedule is longer and the
    optimiser gets more iterations per stage than before.

2.  Projected sub‑gradient descent on the true (non‑smoothed) objective
    with a larger initial step‑size and a simple learning‑rate decay.

3.  A *directed* hill‑climb that preferentially moves mass from indices
    that contribute most to the current maximal convolution entry to
    indices that contribute least.  This uses the exact fast convolution
    (FFT) for each trial move.

4.  A final aggressive hill‑climb (as in the original code) to polish
    any remaining slack.

The routine never leaves the feasible set (non‑negative entries,
∑a = 1) and finally returns the best sequence found before the
time‑budget expires.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (∑a = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).
    """
    a = softmax(y)

    # convolution a*a (linear) via real FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes include random, exponential, power‑law, edge‑biased,
    Sidon‑set, binary mask, Gaussian bump, harmonic, arcsine,
    beta‑family and a simple tent shape.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy Sidon‑set construction
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums or (2 * candidate) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "beta":
        # Symmetric Beta(p,p) with random exponent p>0
        p = rng.uniform(0.1, 3.0)
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = (x ** (p - 1.0)) * ((1.0 - x) ** (p - 1.0)) + eps
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "tent":
        # Simple tent shape: rises to the centre then falls
        a0 = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
        a0 = a0.astype(float)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def directed_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Informed hill‑climb: pick a source index i proportional to its
    contribution to the current maximal convolution entry and a target
    index j with low contribution, then move a random fraction of mass.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current convolution and the index of the maximum entry
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (2*a_{m‑i} where defined)
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[counterpart[mask]]

        # choose a source index weighted by its gradient (large -> move away)
        prob_src = g / (g.sum() + 1e-12)
        i = rng.choice(n, p=prob_src)

        # choose a target index where the gradient is small
        # use complement of the probability mass to bias low‑gradient entries
        prob_tgt = (g.max() - g) + 1e-12
        prob_tgt /= prob_tgt.sum()
        j = rng.choice(n, p=prob_tgt)

        if i == j:
            continue

        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv) (i.e. minimise max(conv) under Σa = 1, a≥0).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine follows a 4‑phase scheme described
    above and returns the best sequence found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    default_n = 10000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            default_n = min(int(tmp.size), 15000)
    n = default_n

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Multi‑restart L‑BFGS (≈60 % of the time budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.9, 1e-9, num=55))
    iteration = 0
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "binary",
        "bump",
        "harmonic",
        "arcsine",
        "beta",
        "tent",
        "best",
    ]
    phase1_deadline = start_time + 0.60 * budget_s

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # short polishing hill‑climb
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
            max_move=0.15,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent (≈25 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 10.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.40,
        max_iters=800_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Directed hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[directed] start directed hill‑climb")
        best_a, best_score = directed_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=15_000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)
        print("[directed] finished")

    # ------------------------------------------------------------------
    # Phase 4 – Final aggressive hill‑climb (small finishing touch)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30_000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
