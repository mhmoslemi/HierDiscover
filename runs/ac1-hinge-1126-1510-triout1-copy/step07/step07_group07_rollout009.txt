"""
Improved search for a low‐evaluation step function.

Main ideas
-----------
* Try many analytic families (β(p,p) with a fine grid of p, semicircle,
  uniform, random Gaussian, sparse binary, single Gaussian bump and a
  few mixtures).  For each family we also test several problem sizes
  (n = 3000 … 8000) – the discretisation can affect the bound.
* Keep the K most promising seeds (K=4) and run a continuation L‑BFGS
  optimiser on a smoothed “log‑sum‑exp” surrogate of the max‑convolution.
  The smoothing factor λ is reduced geometrically from a large value
  (≈ 5 · max) down to 1e‑10 · max.  The L‑BFGS schedule is longer and
  uses more iterations per stage than the reference implementation.
* Perform a series of cheap random restarts (Gaussian initialisation)
  with a shortened continuation schedule to explore the neighbourhood.
* Follow up with a long projected sub‑gradient descent on the true
  objective (the max‑convolution).  The step‑size decays as 1/√t.
* Finish with aggressive hill‑climbing:
      – a plain random mass transfer,
      – a targeted move that removes mass from the indices that
        contribute most to the current maximal convolution value,
      – a final short‑run simulated‑annealing style climb.
* All phases respect a global 1000 s budget; the best sequence found so
  far is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    a is length n, fft_len >= 2*n-1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the evaluation function for a probability vector
    (∑ a_i = 1): 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c            # makes the max become 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # correlation of a and w (gradient of conv w.r.t. a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
    deadline: float = None,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    If a deadline is supplied we abort early.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # stop early if we are out of time
        if deadline is not None and time.time() > deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Analytic initialisation families
# ----------------------------------------------------------------------


def beta_init(n: int, p: float, q: float = None) -> np.ndarray:
    """β(p,q) density on (0,1) (q defaults to p)."""
    if q is None:
        q = p
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)  # ensure positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, renormalised to sum 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s

# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: move a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from the current
    max‑convolution contributors.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))                # shift that attains max
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contributions = best[idx[mask]] * best[counterpart[mask]]
        if contributions.size > 0:
            # pick a source index among the top‑10 % contributors
            thresh = np.percentile(contributions, 90)
            heavy = idx[mask][contributions >= thresh]
            i = rng.choice(heavy)
        else:
            i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    start_temp: float = 1e-2,
    end_temp: float = 1e-6,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """Very light simulated‑annealing style climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        t = start_temp * (end_temp / start_temp) ** (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.25
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - best_score) / max(t, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            if new_score < best_score:
                best_score = new_score
        else:
            # revert move
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Implements an extended pipeline that tries many
    analytic seeds, a long continuation L‑BFGS phase, random restarts,
    a long sub‑gradient descent, and aggressive final polishing.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Choose a set of problem sizes to test.
    # ------------------------------------------------------------------
    n_choices = [3000, 4000, 5000, 6000, 8000]

    # ------------------------------------------------------------------
    # Build a pool of analytic initialisers.
    # ------------------------------------------------------------------
    shape_generators = []

    # β(p,p) with a fine grid of p (including values <0.55)
    for p in [0.51, 0.52, 0.53, 0.54, 0.55, 0.60, 0.65, 0.70, 0.75, 0.80]:
        shape_generators.append(lambda n, p=p: beta_init(n, p))

    # asymmetric beta (p < q) – a few exotic options
    shape_generators.append(lambda n: beta_init(n, 0.45, 0.90))
    shape_generators.append(lambda n: beta_init(n, 0.90, 0.45))

    # classic families
    shape_generators.append(lambda n: semicircle_init(n))
    shape_generators.append(lambda n: uniform_init(n))

    # random families
    shape_generators.append(lambda n: random_gaussian_init(n, rng))
    shape_generators.append(lambda n: binary_sparse_init(n, rng, density=0.15))
    shape_generators.append(lambda n: gaussian_bump_init(n, rng))

    # mixtures of a reference beta with uniform
    beta_ref = lambda n: beta_init(n, 0.70)
    for w in [0.1, 0.3, 0.5, 0.7, 0.9]:
        shape_generators.append(
            lambda n, w=w: mixture_init(beta_ref(n), uniform_init(n), w)
        )

    # ------------------------------------------------------------------
    # Quick scan across all (shape, n) pairs – keep the most promising.
    # ------------------------------------------------------------------
    scored = []  # (score, n, vector, generator)
    for n in n_choices:
        fft_len = next_pow2(2 * n - 1)
        for gen in shape_generators:
            a = gen(n)
            sc = fast_score(a, fft_len, n)
            scored.append((sc, n, a, gen))

    scored.sort(key=lambda x: x[0])
    top_k = 4
    best_a = None
    best_score = np.inf
    best_n = None
    best_fft_len = None

    # ------------------------------------------------------------------
    # Continuation L‑BFGS on the top seeds.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-10, num=35))
    maxiter_per_stage = 800
    phase_deadline = start_time + 0.45 * budget_s

    print("[init] scanning complete – proceeding to L‑BFGS")
    for rank in range(min(top_k, len(scored))):
        sc, n, a0, gen = scored[rank]
        if time.time() > phase_deadline:
            print("[phase] L‑BFGS time budget exhausted")
            break

        fft_len_conv = next_pow2(2 * n - 1)
        fft_len_g = next_pow2(3 * n - 2)

        y0 = np.log(a0 + 1e-12)
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage,
            deadline=phase_deadline,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_n = n
            best_fft_len = fft_len_conv
            print(f"[l‑bfgs] new best (seed {rank}) n={n} score={best_score:.6f}")

    # If nothing succeeded (unlikely), fallback to uniform
    if best_a is None:
        best_n = n_choices[0]
        best_a = uniform_init(best_n)
        best_fft_len = next_pow2(2 * best_n - 1)
        best_score = fast_score(best_a, best_fft_len, best_n)

    # ------------------------------------------------------------------
    # Random‑restart continuation (exploratory).
    # ------------------------------------------------------------------
    restart_deadline = start_time + 0.65 * budget_s
    restart_cnt = 0
    while time.time() < restart_deadline:
        restart_cnt += 1
        n = best_n  # keep the same problem size for fairness
        fft_len_conv = best_fft_len
        fft_len_g = next_pow2(3 * n - 2)
        a0 = random_gaussian_init(n, rng)
        y0 = np.log(a0 + 1e-12)
        # a short schedule – only the first half of the full lam list
        short_lam = lam_schedule[:15]
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            short_lam,
            maxiter_per_stage=400,
            deadline=restart_deadline,
        )
        a_opt = softmax(y_opt)
        sc = fast_score(a_opt, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"[restart #{restart_cnt:03d}] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (majority of remaining budget).
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        best_fft_len,
        best_n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=900_000,
    )
    print(f"[subgrad] finished – best score so far = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive polishing: random hill‑climb → targeted → simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] random hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            best_fft_len,
            best_n,
            final_deadline,
            max_iter=30000,
            max_move=0.30,
        )
    if time.time() < deadline:
        print("[final] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            best_fft_len,
            best_n,
            deadline,
            max_iter=12000,
            max_move=0.25,
        )
    if time.time() < deadline:
        print("[final] simulated‑annealing polish")
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            best_fft_len,
            best_n,
            deadline,
            max_iter=25000,
        )
    # ------------------------------------------------------------------
    # Final evaluation (using the supplied evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)
        final_val = None

    return final_seq
