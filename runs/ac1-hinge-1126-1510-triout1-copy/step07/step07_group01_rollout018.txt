def lp_update(a, t):
    n = len(a)
    # Compute current M
    conv = full_conv(a, fft_len)   # get max
    M = conv.max()
    # Build constraint matrix A (size (2n-1, n)) as sparse
    rows = []
    cols = []
    data = []
    # For each column j (0..n-1):
    # rows = j .. j + n -1
    # data = a[0..n-1]
    # Loop j
    for j in range(n):
        rows.extend(range(j, j+n))
        cols.extend([j]*n)
        data.extend(a)
    A = scipy.sparse.csr_matrix((data,(rows,cols)), shape=(2*n-1, n))
    # Right-hand side ub = M * ones
    ub = np.full(2*n-1, M)
    # Linear program maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: return a
        return a
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a
    g = b / sum_b
    # convex combination
    a_new = (1 - t) * a + t * g
    # Ensure nonneg and sum=1 (should hold)
    a_new = np.clip(a_new, 0, None)
    s = a_new.sum()
    if s > 0:
        a_new = a_new / s
    return a_new
```

"""
Propose a low‑scoring step sequence.

Algorithm
---------
1. Initialise a few diverse candidates (uniform, spectral, random).
2. A short L‑BFGS run on a log‑sum‑exp smooth approximation of the max‑convolution.
3. Exponentiated‑gradient descent on the exact max‑convolution (hard max).
4. **Linear‑programming update** – for the current a compute
       max(a*a)=M,
   maximise Σ b_j  subject to  a*b ≤ M  (linear constraints) and b≥0.
   Scale b to have sum = 1 and replace a by a convex combination (1‑t)a + t b.
   This step can theoretically drive the objective towards the lower bound ≈1.
5. Final greedy hill‑climb.
All steps respect the time budget (default 1000 s) and keep the vector on the
probability simplex.  The LP stage is the main novelty over the previous
implementation.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed maximum of the self‑convolution using log‑sum‑exp."""
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS for each λ in lam_schedule."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """Greedy hill‑climb: move a small random fraction of the mass."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Initialise a via the square‑root of a uniform target in the Fourier domain."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    sqrt_amp = np.sqrt(np.abs(S))
    sqrt_phase = np.angle(S) / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Linear‑programming optimisation step
# ----------------------------------------------------------------------


def lp_update(a: np.ndarray, t: float, fft_len_conv: int) -> (np.ndarray, float):
    """
    Solve  max Σb  s.t.  a*b ≤ M  (component‑wise), b≥0,
    where M = max(a*a).  Return the convex combination
    a←(1‑t)a+tg (scaled to sum‑1) and its fast_score.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len_conv)
    M = conv.max()
    # Build *sparse* Toeplitz matrix for the linear constraints a*b ≤ M.
    # Column j contains a placed on rows j … j+n‑1.
    A = sp.diags([a] * n, offsets=np.arange(n),
                 shape=(2 * n - 1, n), format='csr')

    c = -np.ones(n)               # minimise –Σb  → maximise Σb
    bounds = [(0, None)] * n
    ub = np.full(2 * n - 1, M)

    # Solve the LP.
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return the unchanged vector
        return a, fast_score(a, fft_len_conv, n)

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return a, fast_score(a, fft_len_conv, n)

    g = b / sum_b                # scale to unit sum
    a_new = (1.0 - t) * a + t * g
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0.0:
        a_new = a_new / s
    new_score = fast_score(a_new, fft_len_conv, n)
    return a_new, new_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    max_n = 4000                # keep previous size for compatibility
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initial candidates
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    a_spec = spectral_init(n)

    candidates = [a0, a_spec,
                  random_initial(n, rng),
                  random_initial(n, rng)]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    phase1_deadline = start_time + 0.30 * budget_s   # L‑BFGS
    phase2_deadline = start_time + 0.60 * budget_s   # exp‑grad
    phase3_deadline = start_time + 0.85 * budget_s   # LP updates
    # remaining time -> final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS (light)
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 800    # shortened compared with the previous run

    while time.time() < phase1_deadline:
        a_start = random_initial(n, rng) if rng.random() < 0.5 else best_a
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_deadline,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   rng)
        print(f"[exp‑grad] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – linear‑programming update (the novel step)
    # --------------------------------------------------------------
    if time.time() < phase3_deadline:
        lp_deadline = phase3_deadline
        it = 0
        t = 0.5                     # initial mixing weight
        while time.time() < lp_deadline and it < 12:
            a_new, new_score = lp_update(best_a, t, fft_len_conv)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[LP] iter {it} new best score={best_score:.6f}")
                # if we made progress, keep a relatively aggressive step‑size
                t = min(0.9, t * 1.05)
            else:
                # shrink step‑size when no improvement
                t = max(0.05, t * 0.6)
            it += 1

    # --------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # --------------------------------------------------------------
    final_score = fast_score(best_a, fft_len_conv, n)
    best_a, final_score = hill_climb(best_a,
                                     final_score,
                                     fft_len_conv,
                                     n,
                                     deadline,
                                     max_iter=12000)
    print(f"[final hill‑climb] score={final_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
