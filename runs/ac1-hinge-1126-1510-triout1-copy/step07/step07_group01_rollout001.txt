"""
Non‑negative step‑function optimizer
-----------------------------------

The goal is to minimise

    2·n·max(a∗a) / (∑a)²

over non‑negative vectors a with arbitrary length.  The objective is
scale‑invariant, so we keep the vector on the probability simplex
(∑a = 1).  The optimiser combines three complementary ideas:

1. **Heavy L‑BFGS refinement** on a smooth log‑sum‑exp approximation
   of the max‑convolution.  A continuation on the smoothing parameter
   quickly drives the solution close to a stationary point.

2. **A long simulated‑annealing stage** that repeatedly moves a random
   fraction of mass between two coordinates.  The temperature schedule
   is very slow (geometric decay) and the move size adapts with the
   temperature, allowing the search to escape shallow basins.  The
   annealer is run many times from fresh random starts (Dirichlet‑like
   draws, noisy perturbations of the current best, and a spectral
   guess) – the best result of all runs is kept.

3. **A final aggressive greedy hill‑climb** (10 000 + iterations) that
   only accepts strictly improving moves.  Because the sequence is
   already near‑optimal, this final polishing can shave off the last
   few‑thousandths.

All stages respect the total 1000 s budget, use FFT‑based convolution
for O(n log n) evaluation, and keep the vector on the simplex via the
exponentiated‑gradient update (no explicit projection is needed).  The
procedure is deterministic given the seed but explores a diverse set of
starting points to avoid getting stuck in a local minimum.

The function `propose_candidate` returns the best list of floats found
before the deadline.  It also prints the evaluation score using the
provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is normalised (∑a = 1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)  # random concentration
        a = rng.gamma(alpha)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add small Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """A crude spectral guess (inverse FFT of sqrt‑amplitude)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each smoothing parameter in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] improved to {best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def anneal_optimize(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator):
    """
    Aggressive simulated annealing.
    Moves a random fraction of the mass from i to j.
    Temperature and move size decay very slowly to allow thorough search.
    """
    cur_a = a_start.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    best_a = cur_a.copy()
    best_score = cur_score

    # Parameters (tuned empirically)
    T = 0.15                     # initial temperature
    move_frac = 0.25             # initial maximum fraction of mass moved
    decay_T = 0.9995
    decay_move = 0.9997
    min_T = 5e-5
    min_move = 1e-4
    log_interval = 5000
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur_a[i] <= 0.0:
            continue
        # delta is a random fraction of a[i] bounded by move_frac
        delta = cur_a[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # No need to renormalise – total mass unchanged
        new_score = fast_score(cand, fft_len, n)

        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cand.copy()

        # Decay schedules
        if it % 1000 == 0:
            T = max(T * decay_T, min_T)
            move_frac = max(move_frac * decay_move, min_move)

        # occasional smoothing (local averaging) – can help escape plateaus
        if it % log_interval == 0 and it > 0:
            # simple 3‑point smoothing
            smooth = (np.roll(cur_a, 1) + cur_a + np.roll(cur_a, -1)) / 3.0
            smooth /= smooth.sum()  # re‑normalise
            smooth_score = fast_score(smooth, fft_len, n)
            if smooth_score < cur_score:
                cur_a, cur_score = smooth, smooth_score
                if smooth_score < best_score:
                    best_score = smooth_score
                    best_a = smooth.copy()
        it += 1
    return best_a, best_score


def hill_climb_optimize(a_start: np.ndarray,
                        n: int,
                        fft_len: int,
                        deadline: float,
                        max_iter: int = 20000):
    """
    Greedy hill‑climb that only accepts strict improvements.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move a small random fraction of mass from i to j
        delta = a[i] * rng.random() * 0.08
        if delta <= 0.0:
            it += 1
            continue

        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        it += 1
    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # deterministic RNG
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                       # length of the step function
    fft_len = next_pow2(2 * n - 1)  # length used for FFT‑based convolution

    # ------------------------------------------------------------------
    # Initialise from previous best if it exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Additional diverse starts
    # ------------------------------------------------------------------
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # Phase 1 : L‑BFGS continuation (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 800
    while time.time() < phase1_deadline:
        # pick a random start for diversity
        r = rng.random()
        if r < 0.33:
            a_start = random_initial(n, rng)
        elif r < 0.66:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.012), rng)
        else:
            a_start = spectral_init(n)

        a_opt, scr = lbfgs_continuation(a_start,
                                        lam_schedule,
                                        maxiter_per_lam,
                                        phase1_deadline,
                                        fft_len,
                                        fft_len,
                                        n)
        if scr < best_score:
            best_score = scr
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 : Exponentiated‑gradient descent (≈ 30 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.45 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 : Adam fine‑tuning (≈ 10 % of the budget)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.55 * budget_s
    if time.time() < phase3_deadline:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len,
                                           fft_len,
                                           phase3_deadline,
                                           lam_factor=0.0015,
                                           lr=0.12)
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 : Aggressive simulated annealing with many restarts
    # ------------------------------------------------------------------
    phase4_deadline = start_time + 0.90 * budget_s
    while time.time() < phase4_deadline:
        # allocate ~20 s per annealing run (or less if remaining time is small)
        sub_deadline = min(phase4_deadline, time.time() + 20.0)
        # start from a noisy perturbation of the current best or a fresh draw
        if rng.random() < 0.6:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.008), rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, scr = anneal_optimize(a_start,
                                     sub_deadline,
                                     n,
                                     fft_len,
                                     rng)
        if scr < best_score:
            best_score = scr
            best_a = a_opt.copy()
            print(f"[anneal‑restart] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 : Final greedy hill‑climb (use the remaining time)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    best_a, best_score = hill_climb_optimize(best_a,
                                             n,
                                             fft_len,
                                             hill_deadline,
                                             max_iter=25000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (using the official checker)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
