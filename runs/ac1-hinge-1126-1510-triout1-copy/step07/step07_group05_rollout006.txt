def shape_beta(n, alpha, beta):
    i = np.arange(n) + 1 # 1..n
    a = (i ** (alpha-1)) * ((n - i + 1) ** (beta-1))
    # avoid overflow: use log to compute if exponents large? But n 10000, exponents up to maybe 5, fine.
    sum_a = a.sum()
    if sum_a == 0:  # fallback
        return np.full(n, 1.0/n)
    return a / sum_a

def shape_exp(n, rate):
    i = np.arange(n)
    center = (n-1)/2
    a = np.exp(-rate * np.abs(i - center))
    return a / a.sum()

def shape_mixture(n, params):
    weight, alpha1,beta1,alpha2,beta2 = params
    a1 = shape_beta(n, alpha1, beta1)
    a2 = shape_beta(n, alpha2, beta2)
    a = weight * a1 + (1-weight) * a2
    return a / a.sum()

   improvements.
6. Return the final vector (list of floats).  The harness will call
   ``evaluate_sequence`` to obtain the true score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) for the smoothed objective

        f = 2·n·smooth_max(a*a) ,   a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max using log‑sum‑exp
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Shape generators for the cheap random search
# ----------------------------------------------------------------------


def shape_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density on {0,…,n‑1}: (i+1)^{α‑1}(n‑i)^{β‑1}."""
    i = np.arange(1, n + 1, dtype=float)          # 1 … n
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def shape_exp(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential centred in the middle."""
    i = np.arange(n, dtype=float)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    return a / a.sum()


def shape_mixture_beta(
    n: int,
    w: float,
    a1: float,
    b1: float,
    a2: float,
    b2: float,
) -> np.ndarray:
    """Convex mixture of two Beta‑type densities."""
    a = w * shape_beta(n, a1, b1) + (1.0 - w) * shape_beta(n, a2, b2)
    return a / a.sum()


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12  # up to 12 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def fine_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Very long hill‑climb with a decaying move size.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    move_size = 0.05  # start with 5 % moves

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # pick a fraction of the source mass
        delta = best[i] * rng.random() * move_size
        if delta == 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        # slowly shrink the step size (exponential decay)
        move_size = max(move_size * 0.9999, 1e-4)

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses an averaged sub‑gradient over all (almost) active convolution
    entries and a simple back‑tracking line search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                     # d(a_i a_j)/d a_i = 2 a_j

        # step size decreasing like 1/sqrt(it)
        eta = eta0 / np.sqrt(it)
        improvement = False
        for _ in range(8):                     # up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improvement = True
                break
            eta *= 0.5
        if not improvement:
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    Returns a plain ``list`` of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------- problem size -------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------- initial best candidate -------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------- cheap parametric random search -------------------
    # allocate about 20 % of the total budget to this stage
    param_end = min(deadline, time.time() + 0.20 * budget_s)
    while time.time() < param_end:
        mode = rng.choice(["beta", "exp", "mix"])
        if mode == "beta":
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            a = shape_beta(n, alpha, beta)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.03)
            a = shape_exp(n, rate)
        else:  # mixture
            w = rng.random()
            a1 = rng.uniform(0.2, 5.0)
            b1 = rng.uniform(0.2, 5.0)
            a2 = rng.uniform(0.2, 5.0)
            b2 = rng.uniform(0.2, 5.0)
            a = shape_mixture_beta(n, w, a1, b1, a2, b2)

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[shape] mode={mode} → fast score = {best_score:.6f}")

    # ------------------- continuation L‑BFGS -------------------
    # use a fairly aggressive schedule
    lam_schedule_main = list(np.geomspace(0.5, 5e-10, num=60))
    lam_schedule_refine = list(np.geomspace(0.5, 1e-12, num=90))

    print("[L‑BFGS] main phase start")
    y_opt = schedule_optimize(
        best_y,
        n,
        fft_len_conv,
        fft_len_g,
        lam_schedule_main,
        maxiter_per_stage=700,
    )
    a_opt = softmax(y_opt)
    score_opt = fast_score(a_opt, fft_len_conv, n)
    if score_opt < best_score:
        best_score = score_opt
        best_a = a_opt.copy()
        best_y = y_opt.copy()
        print(f"[L‑BFGS] improved fast score = {best_score:.6f}")

    # ------------------- sub‑gradient refinement -------------------
    if time.time() < deadline - 150.0:
        sub_deadline = deadline - 120.0
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.30,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------- final high‑precision L‑BFGS -------------------
    if time.time() < deadline - 70.0:
        print("[refine L‑BFGS] start")
        y_opt = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_refine,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[refine L‑BFGS] improved fast score = {best_score:.6f}")

    # ------------------- final aggressive hill‑climb -------------------
    if time.time() < deadline:
        print("[final fine HC] start")
        best_a, best_score = fine_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
        )
        print(f"[final fine HC] finished → fast score = {best_score:.6f}")

    # ------------------- official evaluation -------------------
    final_seq = best_a.tolist()
    # The harness will provide evaluate_sequence; we call it here for logging.
    final_val = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
