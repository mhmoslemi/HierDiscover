def lp_mix_step(a, cur_score, fft_len_conv, n, deadline):
    # compute M = max conv
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2*n - 1]

"""
Propose a low‑evaluation step sequence.

The search runs two independent workers (max 2 CPUs).  Each worker
performs a multi‑phase optimisation on a simplex‑normalised
distribution of length n (default 4000).  Phases

  1.  Scan a small library of analytic windows (Hann, Hamming, Blackman,
      Kaiser with several β) and keep the best.
  2.  Heavy L‑BFGS continuation on the smoothed‐max objective with a
      decreasing log‑sum‑exp smoothing schedule.
  3.  Projected sub‑gradient descent with momentum.
  4.  Exponentiated‑gradient descent on the true max‑convolution.
  5.  One linear‑programming mixing step (maximise the total mass of a
      sequence g under the constraint (f∗g)≤M, where M is the current
      max‑convolution of f).  The result is mixed back into the current
      candidate.
  6.  A short L‑BFGS refinement with very small smoothing values.
  7.  Gradient‑guided hill‑climb that moves mass from high‑gradient
      indices to low‑gradient ones, accepting only strictly improving
      moves.

The two workers use different random seeds; the main process evaluates
the returned candidates with the provided ``evaluate_sequence`` function
and returns the best one found before the global deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import i0

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective assuming a lies on the simplex
    (∑a = 1).  Returns 2·n·max(conv) where conv = a*a.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

# Window library (analytic families)
# ----------------------------------------------------------------------


def hann_window(n: int) -> np.ndarray:
    """Classic Hann window (zero at the ends)."""
    i = np.arange(n, dtype=float)
    w = 0.5 * (1 - np.cos(2.0 * np.pi * i / (n - 1)))
    return w


def hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w


def blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a0 = 0.42
    a1 = 0.5
    a2 = 0.08
    w = a0 - a1 * np.cos(2.0 * np.pi * i / (n - 1)) + a2 * np.cos(4.0 * np.pi * i / (n - 1))
    return w


def sine_window(n: int) -> np.ndarray:
    """Single‑sine (raised‑cosine) window."""
    i = np.arange(n, dtype=float)
    w = np.sin(np.pi * i / (n - 1))
    return w


def kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window with shape parameter beta."""
    i = np.arange(n, dtype=float)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    w = i0(beta * np.sqrt(1 - t * t)) / i0(beta)
    return w


def analytic_windows(n: int):
    """Yield a collection of (name, window_array) pairs."""
    yield "hann", hann_window(n)
    yield "hamming", hamming_window(n)
    yield "blackman", blackman_window(n)
    yield "sine", sine_window(n)
    # a handful of Kaiser windows with different betas
    for beta in np.linspace(0.5, 12.0, 14):
        yield f"kaiser_{beta:.2f}", kaiser_window(n, beta)


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (using correlation)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # Very rare; skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.95
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g

        step = 1.0 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true
    max‑convolution sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        if (it + 1) % 5000 == 0:
            if rng.random() < 0.3:
                sigma = rng.uniform(0.001, 0.008)
                a = a + rng.normal(0.0, sigma, size=n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def targeted_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb: move a small fraction of mass from indices
    with a large sub‑gradient to those with a small one, keeping only
    strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming mixing step
# ----------------------------------------------------------------------


def lp_mix_step(a: np.ndarray, cur_score: float,
                fft_len_conv: int, n: int, deadline: float):
    """
    Solve
        max   sum(g)
        s.t.  (a * g) ≤ M  (component‑wise)
              g ≥ 0
    where M = max_k (a*a)_k.
    Mix the solution back into a and keep the improvement if any.
    """
    # Current maximum of a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    M = conv.max()

    # Build the Toeplitz matrix T such that (a * g) = T @ g
    i = np.arange(2 * n - 1)[:, None]
    j = np.arange(n)[None, :]
    idx = i - j
    mask = (idx >= 0) & (idx < n)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    T[mask] = a[idx[mask]]

    # Linear program: maximise sum(g)  <=> minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    # Give the solver at most the remaining time
    time_left = max(0.0, deadline - time.time())
    opts = {"presolve": True}
    if time_left > 1.0:
        opts["time_limit"] = time_left

    try:
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method="highs", options=opts)
    except Exception as e:
        print("[lp_mix] linprog exception:", e)
        return a, cur_score

    if not res.success:
        # LP failed – just return the input
        return a, cur_score

    g = res.x
    if g.sum() == 0.0:
        return a, cur_score
    g = g / g.sum()

    # Mix with a modest coefficient and re‑project onto the simplex
    t = 0.5
    new_a = simplex_projection((1 - t) * a + t * g)
    new_score = fast_score(new_a, fft_len_conv, n)
    if new_score < cur_score:
        return new_a, new_score
    return a, cur_score


# ----------------------------------------------------------------------
# Worker – performs a full search on its own random seed
# ----------------------------------------------------------------------


def worker_search(seed: int, n: int, deadline: float) -> np.ndarray:
    rng = np.random.default_rng(seed)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 0.  Start from a small library of analytic windows
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)

    for name, w in analytic_windows(n):
        a = w.astype(np.float64)
        if np.all(a >= 0):
            a = a / a.sum()
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a
        # occasional print
        if rng.random() < 0.001:
            print(f"[seed {seed}] window {name} score {sc:.6f}")

    # ------------------------------------------------------------------
    # 1. Heavy L‑BFGS continuation (smoothed max)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.001, 0.0005, 0.0002,
    ]
    maxiter_per_lam = 800
    phase1_deadline = time.time() + 0.3 * (deadline - time.time())
    while time.time() < phase1_deadline:
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.018)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_deadline, fft_len_conv, fft_len_g, n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt
            print(f"[seed {seed}] L‑BFGS heavy new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    sub_deadline = time.time() + 0.2 * (deadline - time.time())
    if time.time() < sub_deadline:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, sub_deadline, fft_len_conv, fft_len_g, n)
        print(f"[seed {seed}] sub‑gradient score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Exponentiated‑gradient descent
    # ------------------------------------------------------------------
    eg_deadline = time.time() + 0.15 * (deadline - time.time())
    if time.time() < eg_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, eg_deadline, fft_len_conv, fft_len_g, n)
        print(f"[seed {seed}] exp‑gradient score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Linear‑programming mixing step
    # ------------------------------------------------------------------
    lp_deadline = time.time() + 0.05 * (deadline - time.time())
    if time.time() < lp_deadline:
        best_a, best_score = lp_mix_step(best_a, best_score,
                                         fft_len_conv, n, lp_deadline)
        print(f"[seed {seed}] LP‑mix score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Short L‑BFGS refinement with very small smoothing
    # ------------------------------------------------------------------
    fine_lam = [0.001, 0.0005, 0.0002, 0.0001, 5e-05]
    fine_maxiter = 300
    fine_deadline = time.time() + 0.1 * (deadline - time.time())
    if time.time() < fine_deadline:
        best_a, fine_score = lbfgs_continuation(
            best_a, fine_lam, fine_maxiter,
            fine_deadline, fft_len_conv, fft_len_g, n)
        if fine_score < best_score:
            best_score, best_a = fine_score, best_a
            print(f"[seed {seed}] L‑BFGS fine score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final targeted gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = targeted_hill_climb(
        best_a, best_score, fft_len_conv, fft_len_g, n,
        final_deadline, max_iter=80000)
    print(f"[seed {seed}] final hill‑climb score {best_score:.6f}")

    return best_a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the
    supplied evaluation function.  Two independent workers are launched;
    the best sequence among them (according to ``evaluate_sequence``) is
    returned.
    """
    import concurrent.futures

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # Problem size – 4000 worked well in the previous best run
    n = 4000

    # Split the remaining budget roughly equally between two workers
    worker_deadline = start_time + (budget_s - 0.2) / 2.0

    seeds = [seed, seed + 1]
    results = []

    with concurrent.futures.ProcessPoolExecutor(max_workers=2) as executor:
        futures = [executor.submit(worker_search, s, n, worker_deadline) for s in seeds]
        for fut in concurrent.futures.as_completed(futures):
            try:
                a = fut.result()
                results.append(a)
            except Exception as e:
                print("[main] worker raised exception:", e)

    # Choose the best candidate according to the *true* evaluation function
    best_seq = None
    best_val = float("inf")
    for a in results:
        seq = a.tolist()
        val = evaluate_sequence(seq)
        print(f"[main] candidate evaluated score = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq

    # Fallback: uniform distribution if everything failed (should never happen)
    if best_seq is None:
        best_seq = [1.0 / n] * n
        best_val = evaluate_sequence(best_seq)
        print("[main] fallback uniform score =", best_val)

    print(f"[main] final best score = {best_val:.6f}")
    return best_seq
