def build_convolution_matrix(a):
    n = len(a)
    rows = []
    cols = []
    data = []
    for j in range(n):
        # For each column j, rows t range from j to j+n-1 (where a index is t-j)

        cols.extend([j]*n)
        data.extend(a)
    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n))

    n = len(a)
    # Build matrix A
    col_repeat = np.repeat(np.arange(n), n)
    row_add = np.tile(np.arange(n), n)
    rows = col_repeat + row_add
    cols = col_repeat
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)

    if not res.success:
        return a, False
    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0 + 1e-12:  # no increase
        return a, False
    # Normalize g to sum 1
    g_norm = g / sum_g

    # compute conv matrices once
    conv_aa = conv_self(a, fft_len)
    conv_ag = conv_cross(a, g_norm, fft_len)
    conv_gg = conv_self(g_norm, fft_len)

    # ternary search for optimal alpha
    lo, hi = 0.0, 1.0
    best_alpha = 0.0
    best_val = np.max(conv_aa)  # current value
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        v1 = np.max((1-m1)**2 * conv_aa + 2*m1*(1-m1) * conv_ag + m1**2 * conv_gg)
        v2 = np.max((1-m2)**2 * conv_aa + 2*m2*(1-m2) * conv_ag + m2**2 * conv_gg)

            lo = m1
        else:

    new_conv = (1-alpha_opt)**2 * conv_aa + 2*alpha_opt*(1-alpha_opt) * conv_ag + alpha_opt**2 * conv_gg
    new_max = new_conv.max()
    if new_max < np.max(conv_aa) - 1e-12:

        # renormalize to sum=1 to avoid numeric drift
        new_a = new_a / new_a.sum()
        return new_a, True

"""
Improved step‑function search.

The algorithm builds on the previous multi‑phase pipeline and adds
a *convex‑programming improvement* that makes use of the exact linear
constraints (f * g) ≤ M, where M is the current maximal self‑convolution.
For the current distribution f we solve a linear program that maximises
the total mass of a new distribution g under these constraints.
After normalising g, the optimal 1‑D mixture (1‑α) f + α g is found by a
cheap ternary search on α, which strictly reduces the maximal
convolution.  Several LP–mix steps are performed while time permits,
followed by a long targeted gradient‑guided hill‑climb (the same
sub‑gradient as in the reference code, but with deterministic source /
target choice).  The final sequence is returned as a plain list of
floats.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def conv_cross(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*b via FFT (length 2·len(a)‑1, assumes len(a)=len(b))."""
    fa = np.fft.fft(a, fft_len)
    fb = np.fft.fft(b, fft_len)
    conv = np.real(np.fft.ifft(fa * fb))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised (∑a=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming improvement
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix A that computes a * g for any vector g.
    Row t, column j contains a[t‑j] if 0 ≤ t‑j < n, otherwise 0.
    The matrix has a constant value a[k] on diagonal offset k (0 ≤ k < n).
    """
    n = a.shape[0]
    # column index repeated n times, one block per column
    col_repeat = np.repeat(np.arange(n, dtype=np.int64), n)
    # row index is col + offset where offset runs 0..n‑1 inside each block
    row_add = np.tile(np.arange(n, dtype=np.int64), n)
    rows = col_repeat + row_add               # shape (n*n,)
    cols = col_repeat
    data = np.tile(a, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_improve(a: np.ndarray,
               M: float,
               fft_len: int,
               deadline: float) -> tuple[np.ndarray, bool]:
    """
    One LP‑based improvement step.
    Returns (new_a, improved).  The routine solves
        maximise   Σ g_j
        subject to a * g ≤ M   (pointwise)
                    g ≥ 0 .
    The returned distribution is the optimal mixture of the current a
    with the normalised LP solution g.
    """
    n = a.shape[0]

    # Build the constraint matrix a * g
    A = build_conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise –∑g  ⇒ maximise ∑g
    bounds = [(0.0, None)] * n

    # Solve the LP (HiGHS handles sparse matrices efficiently)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
    if not res.success:
        return a, False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 1.0 + 1e-12:          # no useful increase of total mass
        return a, False

    # Normalise the LP solution to unit sum
    g_norm = g / sum_g

    # Pre‑compute the three convolutions needed for the mixture
    conv_aa = full_conv(a, fft_len)
    conv_ag = conv_cross(a, g_norm, fft_len)
    conv_gg = full_conv(g_norm, fft_len)

    # Ternary search for the best mixture weight α∈[0,1]
    lo, hi = 0.0, 1.0
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        cand1 = (1 - m1) ** 2 * conv_aa + 2 * m1 * (1 - m1) * conv_ag + m1 ** 2 * conv_gg
        cand2 = (1 - m2) ** 2 * conv_aa + 2 * m2 * (1 - m2) * conv_ag + m2 ** 2 * conv_gg
        if cand1.max() > cand2.max():
            lo = m1
        else:
            hi = m2
    alpha = (lo + hi) / 2.0

    # Form the mixed distribution and verify improvement
    new_a = (1 - alpha) * a + alpha * g_norm
    new_a = new_a / new_a.sum()          # renormalise to avoid accumulated drift
    new_conv = (1 - alpha) ** 2 * conv_aa + 2 * alpha * (1 - alpha) * conv_ag + alpha ** 2 * conv_gg
    new_max = new_conv.max()
    if new_max < conv_aa.max() - 1e-12:
        return new_a, True
    return a, False


# ----------------------------------------------------------------------
# Targeted gradient‑guided hill‑climb (deterministic source/target)
# ----------------------------------------------------------------------


def targeted_hill_climb(a: np.ndarray,
                        current_score: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Deterministic gradient‑guided hill‑climb.
    In each iteration the index with the largest sub‑gradient (source)
    transfers a modest fraction of its mass to the index with the
    smallest sub‑gradient (target).  Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (size n)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # deterministic source / target
        i = int(np.argmax(g))
        j = int(np.argmin(g))

        if i == j:
            it += 1
            continue

        # move up to 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        # else discard the move
        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # --------------------------------------------------------------

    n = 4000

    if "height_sequence_1" in globals():

                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, a0)

    else:

    a0_sum = a0.sum()
    a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------

    start_time = time.time()

    if time.time() < deadline:

            fft_len_conv,
            fft_len_g,
            n,
            deadline,

        )

    # --------------------------------------------------------------
    # Final evaluation with the provided external function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    # The competition harness will call evaluate_sequence() itself,

        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)

    return final_seq
