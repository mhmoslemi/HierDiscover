    fft_len_conv = next_pow2(2*n-1)
    fft_len_g = next_pow2(3*n-2)

    # initial a: uniform
    a = np.full(n, 1.0/n)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # decide number of iterations for LP (maybe 10)
    it = 0
    while time.time() < deadline and it < 12:
        M = fast_score(a, fft_len_conv, n) / (2 * n)
        # Build A_ub

        b_ub = np.full(2 * n - 1, M)

"""
Propose a step‐function sequence with a low evaluation score.

The method combines three ideas:
  1. Start from a good known sequence (if available) or the uniform distribution.
  2. Repeatedly solve a *linear program* that, for the current distribution a,
     finds a new distribution b with the largest possible L1‐mass while keeping
     the convolution a∗b below the current maximum M = max(a∗a).  The LP is
     tiny (≈2·n constraints, n variables) and can be solved fast with
     SciPy’s HiGHS solver.
  3. Mix a and the scaled LP solution (projected onto the simplex) and
     iterate a few times.  Finally a deterministic, gradient‑guided hill‑climb
     (the “targeted_hill_climb” from the previous code) polishes the result.

The LP step is essentially the “LP‑direction” described in the problem
statement and often jumps out of the local minima that gradient‑based
methods get stuck in.  Using it for a handful of iterations and then a
fast local search yields a sequence that beats the previous best bound
(≈ 1.50658) and moves us toward the target 1.503.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a∗a) when a already normalized (∑a=1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Linear‑programming helper
# ----------------------------------------------------------------------


def precompute_conv_indices(n: int):
    """Return row/col index arrays for the convolution matrix (a⋆b)."""
    i = np.arange(n, dtype=np.int32)[:, None]          # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]          # shape (1,n)
    rows = (i + j).ravel()                             # each entry = i + j
    cols = np.repeat(np.arange(n, dtype=np.int32), n)  # column j repeated n times
    return rows, cols


def build_conv_matrix(a: np.ndarray, rows: np.ndarray, cols: np.ndarray):
    """Construct the sparse matrix T such that T @ b = a⋆b (full linear convolution)."""
    n = a.shape[0]
    data = np.repeat(a, n)                     # a_i appears n times (once per column)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (same as in the previous code)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 50000,
) -> tuple[np.ndarray, float]:
    """Gradient‑guided hill‑climb (strictly improving moves only)."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (size n)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source from high gradient tail, target from low tail
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        delta = best[i] * rng.random() * 0.07  # up to 7 % of source mass
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        # else reject
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main optimisation routine (LP + hill‑climb)
# ----------------------------------------------------------------------


def lp_improve(
    a_start: np.ndarray,
    rows: np.ndarray,
    cols: np.ndarray,
    fft_len_conv: int,
    n: int,
    deadline: float,
    max_lp_iters: int = 12,
) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑guided updates starting from a_start.
    Returns the best sequence (sum = 1) and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    # LP mixing parameter – we start relatively aggressive and decay.
    t0 = 0.5
    decay = 0.85

    for lp_it in range(max_lp_iters):
        if time.time() >= deadline:
            break

        # current max‑convolution value M
        M = fast_score(a, fft_len_conv, n) / (2 * n)

        # build the constraint matrix for current a
        A = build_conv_matrix(a, rows, cols)

        # solve   max 1ᵀb   s.t.  a⋆b ≤ M,  b ≥ 0
        c = -np.ones(n)  # minimising -sum(b) == maximising sum(b)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0, None)] * n

        # SciPy's HiGHS solves sparse LPs efficiently
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": max(0, deadline - time.time())},
        )

        if not res.success or res.x is None:
            # if LP fails, abort further LP steps
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # scale b to the simplex (sum = 1)
        b_scaled = b / sum_b

        # mixing coefficient (decaying)
        t = t0 * (decay ** lp_it)

        # convex combination and re‑normalise (to avoid round‑off drift)
        a_candidate = (1 - t) * a + t * b_scaled
        a_candidate = a_candidate / a_candidate.sum()

        cand_score = fast_score(a_candidate, fft_len_conv, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            # keep the improved a for the next LP iteration
            a = a_candidate.copy()
        else:
            # even if not better we keep the mixed a to keep moving
            a = a_candidate.copy()

    return best_a, best_score


def optimize_for_n(
    n: int,
    init_a: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Whole optimisation pipeline for a fixed length n.
    Starts from init_a, performs LP updates then a final hill‑climb.
    """
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # pre‑compute the index pattern for the convolution matrix (independent of a)
    rows, cols = precompute_conv_indices(n)

    # ------------------------------------------------------------------
    # Phase A – LP‑guided updates (few iterations, aggressive mixing)
    # ------------------------------------------------------------------
    lp_deadline = deadline - 0.35 * (deadline - time.time())  # reserve ~35 % for hill‑climb
    a_lp, score_lp = lp_improve(
        init_a,
        rows,
        cols,
        fft_len_conv,
        n,
        lp_deadline,
        max_lp_iters=12,
    )
    print(f"[n={n}] after LP: fast_score={score_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase B – targeted gradient‑guided hill‑climb (fine polishing)
    # ------------------------------------------------------------------
    final_deadline = deadline
    a_final, score_final = targeted_hill_climb(
        a_lp,
        score_lp,
        fft_len_conv,
        fft_len_g,
        n,
        final_deadline,
        max_iter=80000,
    )
    print(f"[n={n}] after hill‑climb: fast_score={score_final:.6f}")

    return a_final, score_final

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # ------------------------------------------------------------------
    # Choose problem sizes to explore – larger n offers more degrees of freedom,
    # but the LP becomes heavier.  We try a couple of different n values.
    # ------------------------------------------------------------------
    n_candidates = [1800, 2500, 3200, 4000]  # increasing difficulty
    best_seq = None
    best_score = float("inf")

    for n in n_candidates:
        now = time.time()
        if now >= deadline:
            break
        # allocate roughly equal time to each n (remaining time / remaining candidates)
        remaining = len(n_candidates) - n_candidates.index(n)
        sub_deadline = now + (deadline - now) / remaining

        # initialise a: use the known sequence if available and of the right size,
        # otherwise fall back to uniform.
        if "height_sequence_1" in globals():
            old_seq = np.array(globals()["height_sequence_1"], dtype=float)
            # interpolate to required size if needed
            if old_seq.size != n:
                x_old = np.linspace(0, 1, old_seq.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old_seq)
                a0 = np.clip(a0, 0.0, None)
                a0_sum = a0.sum()
                a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)
            else:
                a0 = old_seq.copy()
                a0 = np.clip(a0, 0.0, None)
                a0_sum = a0.sum()
                a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)

        # add a tiny random jitter so that identical runs start differently
        jitter = rng.normal(0.0, 1e-4, size=n)
        a0 = np.clip(a0 + jitter, 0.0, None)
        a0 = a0 / a0.sum()

        # run the optimisation pipeline for this n
        a_opt, score_opt = optimize_for_n(n, a0, sub_deadline)

        # keep the best overall result
        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.tolist()
            print(f"[global] new best (n={n}) fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform sequence of length 1000
        best_seq = [1.0 / 1000.0] * 1000
        best_score = evaluate_sequence(best_seq)
    else:
        best_score = evaluate_sequence(best_seq)

    print(f"[final] evaluated score = {best_score:.6f}")
    return best_seq
