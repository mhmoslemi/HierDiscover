def gradient_hill_climb(a, cur_score, fft_len, n, deadline, max_iters=2000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iters>0:
        max_iters -= 1
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))
        counterpart = k_max - np.arange(n)
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[counterpart[mask]]
        # select i weighted by contribution
        prob = contrib / contrib.sum()
        if prob.sum() == 0:
            i = rng.integers(0, n)
        else:
            i = rng.choice(n, p=prob)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.2
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score
```

"""
Hybrid optimizer for the step‑function minimisation problem.

The algorithm works on the probability simplex (∑a = 1) because the
evaluation objective 2·n·max(conv) is scale‑invariant.
It combines four complementary phases that fit comfortably into a
1000 s budget:

1.  Multi‑restart L‑BFGS on a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The starts include the arcsine law (known to be
    very good) together with a few random patterns.
2.  A steepest‑descent sub‑gradient step with a tiny line‑search that
    directly attacks the true maximal convolution entry.
3.  A fine‑grained L‑BFGS pass with an ultra‑low smoothing parameter.
4.  An aggressive hill‑climb (random moves plus a gradient‑biased
    variant) that exploits any remaining time.

All convolutions are performed with FFTs (numpy) and the simplex
projections are done with the Euclidean projection in O(n log n).
The routine returns the best sequence it has found before the time
budget expires.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a (linear)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂c/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 200,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        # Discrete arcsine law: a_i ∝ 1/√(x·(1‑x))
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Plain greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_step(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    One sub‑gradient descent step with a brute‑force line‑search over a few
    candidate step sizes.  The direction is the true sub‑gradient of
    max(conv) (i.e. 2·a_{k‑i} on the active index k).
    """
    conv = conv_fft(a, fft_len, n)
    k = int(np.argmax(conv))

    # sub‑gradient of the maximal entry w.r.t. a
    idx = np.arange(n)
    counterpart = k - idx
    mask = (counterpart >= 0) & (counterpart < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[counterpart[mask]]

    # try a handful of step sizes
    candidates = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    best_a = a
    best_score = cur_score
    for eta in candidates:
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new
    return best_a, best_score


def gradient_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb biased by the current maximal‑convolution entry: mass is
    preferentially moved away from indices that contribute heavily to the
    peak.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        # contribution of each coordinate to the maximal entry
        idx = np.arange(n)
        counterpart = k - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[counterpart[mask]]

        # pick a source index i with probability proportional to its contribution
        total = contrib.sum()
        if total > 0:
            prob = contrib / total
            i = rng.choice(n, p=prob)
        else:
            i = rng.integers(0, n)

        j = rng.integers(0, n)
        if i == j:
            continue

        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)
        restart += 1

    # ------------------------------------------------------------------
    # Phase 2 – Targeted sub‑gradient line‑search (≈30 % of the budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.80 * budget_s)
    print("[subgrad] starting targeted sub‑gradient steps")
    it = 0
    while time.time() < subgrad_deadline and it < 1500:
        best_a, best_score = subgradient_step(best_a, best_score, fft_len_conv, n)
        it += 1
        if it % 200 == 0:
            print(f"[subgrad] iteration {it}, score = {best_score:.6f}")
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 3 – Fine L‑BFGS refinement (very low smoothing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        print("[fine] final low‑smoothing L‑BFGS")
        fine_lam = list(np.geomspace(1e-4, 1e-12, num=25))
        y_fine = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam,
            maxiter_per_stage=400,
        )
        a_fine = softmax(y_fine)
        sc_fine = fast_score(a_fine, fft_len_conv, n)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine
            best_y = y_fine
            print(f"[fine] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] aggressive hill‑climb (random + gradient‑biased)")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=25000,
            max_move=0.30,
        )
        # a second pass that uses the gradient‑biased move selector
        best_a, best_score = gradient_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=15000,
            max_move=0.25,
        )
        print(f"[final] fast score after hill‑climb = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
