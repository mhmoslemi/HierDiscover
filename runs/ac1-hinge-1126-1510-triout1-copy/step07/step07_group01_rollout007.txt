"""
Propose a low‑score step‑function sequence.

The algorithm consists of three stages:

1. **Low‑dimensional search** – work on a modest length (n≈800)
   with many random restarts of a continuation‑L‑BFGS optimiser
   (smooth log‑sum‑exp approximation of the max‑convolution).
   A cheap but thorough search in low dimension finds a good
   shape of the distribution.

2. **Upsampling & refinement** – linearly interpolate the best
   low‑dimensional vector to the target length (n=4000).  The
   up‑sampled vector is then fine‑tuned on the exact objective
   using exponentiated‑gradient descent (which respects the simplex)
   and a short L‑BFGS run.

3. **Final greedy hill‑climb** – a cheap neighbourhood search that
   moves a small random fraction of mass between two coordinates
   and accepts only improving moves.  The best vector is finally
   evaluated with the provided `evaluate_sequence` function.

All phases respect the overall 1000 s budget and print intermediate
scores so that progress is visible.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Quickly compute 2·n·max(a*a), assuming a is normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max(a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract the part that corresponds to ∂/∂a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # very rare – just skip this λ.
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # exact convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule (standard 1/√t)
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s == 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a tiny random fraction of the mass from one
    coordinate to another, only accepting strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse probability vector on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the FFT
    of a uniform target (length 2·n‑1).  The result is projected onto the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """
    Linear interpolation of a_small onto a vector of length n_big.
    The result is projected onto the simplex.
    """
    x_small = np.linspace(0.0, 1.0, a_small.size)
    x_big = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    s = a_big.sum()
    if s == 0.0:
        a_big = np.full(n_big, 1.0 / n_big)
    else:
        a_big = a_big / s
    return a_big

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            a0 = np.interp(
                np.linspace(0, 1, n_big),
                np.linspace(0, 1, prev.size),
                prev)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n_big, 1.0 / n_big)
    else:
        a0 = np.full(n_big, 1.0 / n_big)

    # ------------------------------------------------------------------
    # Phase 1 – low‑dimensional intensive search (n≈800)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    n_small = 800
    fft_len_small = next_pow2(2 * n_small - 1)
    fft_len_g_small = next_pow2(3 * n_small - 2)

    # schedule for the smoothed max (more aggressive than the reference)
    lam_schedule_small = [
        0.6, 0.4, 0.28, 0.20, 0.14, 0.09,
        0.06, 0.04, 0.025, 0.015, 0.009, 0.005,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001
    ]
    maxiter_per_lam_small = 2000

    # keep the best low‑dimensional vector found
    best_small = None
    best_small_score = np.inf

    phase1_deadline = start_time + 0.38 * budget_s   # ~38 % of total time

    while time.time() < phase1_deadline:
        # diversified initial guess for the small problem
        if rng.random() < 0.5:
            a_start = random_initial(n_small, rng)
        else:
            a_start = spectral_init(n_small)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule_small,
            maxiter_per_lam_small,
            phase1_deadline,
            fft_len_small,
            fft_len_g_small,
            n_small)

        if score_opt < best_small_score:
            best_small_score = score_opt
            best_small = a_opt.copy()
            print(f"[low‑dim L‑BFGS] new best small‑scale score={best_small_score:.6f}")

    if best_small is None:
        # fallback – use a uniform vector
        best_small = np.full(n_small, 1.0 / n_small)

    # ------------------------------------------------------------------
    # Phase 2 – up‑sample to the target size and fine‑tune
    # ------------------------------------------------------------------
    a_big = upsample(best_small, n_big)

    # Mix with the global initial guess to increase diversity
    a_big = 0.6 * a_big + 0.4 * a0
    a_big /= a_big.sum()

    cur_score = fast_score(a_big, fft_len_big, n_big)
    print(f"[upsample] score after interpolation={cur_score:.6f}")

    # --- exponentiated‑gradient descent on the full size ---
    exp_deadline = start_time + 0.70 * budget_s   # up to 70 % of total budget
    a_big, cur_score = exp_gradient_optimize(
        a_big,
        exp_deadline,
        n_big,
        fft_len_big,
        fft_len_g_big)

    print(f"[exp‑grad (big)] score={cur_score:.6f}")

    # --- brief L‑BFGS fine‑tuning (starting from the current point) ---
    # reuse a slightly refined λ schedule
    lam_schedule_big = [0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
                        0.0015, 0.0007, 0.0004, 0.0002, 0.0001]
    maxiter_per_lam_big = 1500
    lbfgs_deadline = start_time + 0.82 * budget_s
    a_big, cur_score = lbfgs_continuation(
        a_big,
        lam_schedule_big,
        maxiter_per_lam_big,
        lbfgs_deadline,
        fft_len_big,
        fft_len_g_big,
        n_big)

    print(f"[L‑BFGS fine‑tune] score={cur_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = start_time + budget_s - 0.6   # leave a tiny buffer
    a_big, cur_score = hill_climb(
        a_big,
        cur_score,
        fft_len_big,
        n_big,
        final_deadline,
        max_iter=12000)

    print(f"[final hill‑climb] score={cur_score:.6f}")

    final_seq = a_big.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
