def subgradient_descent_momentum(a_start, fft_len, n, deadline, eta0=0.1, beta=0.9):
    a = a_start.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # gradient of max conv (subgradient)
        g = np.zeros_like(a)
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g[mask] = 2.0 * a[counterpart[mask]]
        # update momentum:
        eta = eta0 / np.sqrt(it)
        v = beta * v - eta * g
        a_new = a + v
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

def annealing_hill_climb(a, cur_score, fft_len, n, deadline, init_temp=0.01, final_temp=1e-5):
    best = a.copy()
    best_score = cur_score
    temp = init_temp
    rng = np.random.default_rng()
    while time.time() < deadline:
        # decaying temperature:
        remaining = deadline - time.time()
        # linearly decay:
        fraction = remaining / (deadline - start_time) # but we need start_time; maybe pass as param)
        temp = init_temp * fraction + final_temp * (1 - fraction)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # choose move size
        delta = best[i] * rng.random() * 0.1
        # attempt move
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        # ensure nonnegative:
        if new[i] < 0:
            continue
        new_score = fast_score(new, fft_len, n)
        dE = new_score - best_score
        if dE < 0 or rng.random() < np.exp(-dE / temp):
            best = new
            best_score = new_score
            if new_score < best_score:
                best = new
                best_score = new_score
    return best, best_score
```

"""
Hybrid search for a low‑max‑convolution step function.

The algorithm proceeds in three stages:

1.  A one‑parameter family of symmetric beta densities
    a_i ∝ (x_i·(1−x_i))^{α−1}  (α≈0.5 corresponds to the classic arcsine law)
    is scanned on a fairly large index set (default n≈15000).  The best
    α gives a strong baseline – the objective 2·n·max(conv) is already
    close to the theoretical limit.

2.  Projected sub‑gradient descent (with a √t diminishing step) is run
    on the true non‑smoothed objective.  The iteration stays on the
    probability simplex by Euclidean projection.

3.  A greedy hill‑climb finishes the optimisation: a random amount of
    mass is moved from one index to another, and the move is kept only
    if the fast FFT‑based objective improves.  The remaining time budget
    is spent on short random restarts followed by a few sub‑gradient
    steps and another hill‑climb, allowing the search to escape shallow
    local minima.

All heavy work (convolutions) uses NumPy’s FFT, which makes the search
feasible for n≈15000 within the 1000 s limit.  The final sequence is
returned as a plain list of floats; `evaluate_sequence` will be called
by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1):
        2·n·max(conv)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def beta_distribution(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric beta(α,α) density on the discrete grid of size n,
    returned as a probability vector (sum = 1).
    """
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    eps = 1e-15
    # a_i ∝ (x·(1−x))^{α−1}
    a = np.power(x * (1.0 - x) + eps, alpha - 1.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def subgradient_descent(a_start: np.ndarray,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        eta0: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(a*a)   (probability simplex, a ≥ 0).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))                     # index of maximal entry
        # sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000,
               max_move: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        new = best.copy()
        new[i] -= delta
        new[j] += delta
        if new[i] < 0.0:                     # safety – should not happen
            continue
        new_score = fast_score(new, fft_len, n)
        if new_score < best_score:
            best = new
            best_score = new_score
    return best, best_score

    best_a, best_score = subgradient_descent(best_a, fft_len, n, subgrad_deadline)
    print(f"[subgrad] best fast score after sub‑gradient = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3.  Aggressive hill‑climb on the remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=200_000,
                                        max_move=0.15)
        print(f"[hillclimb] best fast score after hill‑climb = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4.  Short random restarts with a few refinement steps
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:            # keep a few seconds for evaluation
        # perturb current best (log‑space to stay positive)
        sigma = 0.05
        perturb = rng.normal(scale=sigma, size=n)
        a_pert = best_a * np.exp(perturb)
        a_pert = project_onto_simplex(a_pert)

        # tiny sub‑gradient burst (≈2 s or less)
        sub_deadline = time.time() + min(2.0, deadline - time.time() - 1.0)
        a_pert, sc_pert = subgradient_descent(a_pert, fft_len, n, sub_deadline)
        # quick hill‑climb
        a_pert, sc_pert = hill_climb(a_pert,
                                     sc_pert,
                                     fft_len,
                                     n,
                                     deadline,
                                     max_iter=5_000,
                                     max_move=0.10)
        if sc_pert < best_score:
            best_score = sc_pert
            best_a = a_pert.copy()
            print(f"[restart] improved fast score to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5.  Final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)     # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
