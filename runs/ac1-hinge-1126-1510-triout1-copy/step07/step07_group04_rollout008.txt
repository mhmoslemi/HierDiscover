    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:

        conv = conv_fft(a, fft_len, n)

            best_a = a_new.copy()

"""
Hybrid optimizer for the step‑function minimisation problem.

The routine works on the probability simplex (∑a = 1) because the
objective 2·n·max(conv) is scale‑invariant.  The search proceeds in four
phases:

1.  A continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution, starting from the discrete arcsine law (the known
    near‑optimal continuous solution).

2.  Projected sub‑gradient descent on the true (non‑smoothed) objective,
    using the exact max‑convolution to obtain a sub‑gradient.

3.  A de‑convolution based refinement: we ask for a convolution that is
    perfectly flat (all entries equal to the average 1/(2n‑1)) and
    compute a de‑convolved vector by Wiener filtering; the result is
    blended with the current iterate.

4.  An aggressive hill‑climb that moves a random fraction of mass from
    one index to another while only accepting improvements.

All heavy work uses FFT‑based convolutions and the Euclidean projection
onto the simplex.  The algorithm adapts to the available time budget
(1000 s) and returns the best sequence it has found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)   (the denominator Σa² = 1 because the vector is on the simplex).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a (linear)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂c/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        max_idx = int(np.argmax(conv))

        # Sub‑gradient of the maximal entry:
        idx = np.arange(n)
        counterpart = max_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Diminishing step‑size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def deconvolution_step(a: np.ndarray, target: np.ndarray, fft_len: int, eps: float = 1e-8) -> np.ndarray:
    """
    Wiener‑type deconvolution solving a * x ≈ target (linear convolution).
    Returns the first n entries of the solution.
    """
    A = np.fft.fft(a, fft_len)
    C = np.fft.fft(target, fft_len)
    X = C * np.conj(A) / (np.abs(A) ** 2 + eps)
    x = np.real(np.fft.ifft(X))[: a.size]
    return x

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine follows the four‑phase scheme described
    above and returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    # Use the length of the supplied construction if sensible,
    # otherwise fall back to a fairly large n (≈12 000).
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 15000)
        else:
            n = 12000
    else:
        n = 12000

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial distribution (discrete arcsine law)
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    a0 = 1.0 / np.sqrt(x * (1.0 - x) + 1e-15)
    a0 /= a0.sum()
    # add tiny random jitter to avoid exact symmetry
    a0 += 0.0005 * rng.standard_normal(n)
    a0 = np.clip(a0, 0, None)
    a0 = project_onto_simplex(a0)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}  fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS on smoothed max
    # ------------------------------------------------------------------
    y_start = np.log(best_a + 1e-12)
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=35))
    y_opt = schedule_optimize(
        y_start,
        n,
        fft_len,
        fft_len,
        lam_schedule,
        maxiter_per_stage=500,
    )
    a_opt = softmax(y_opt)
    a_opt = project_onto_simplex(a_opt)
    cur_score = fast_score(a_opt, fft_len, n)
    if cur_score < best_score:
        best_score = cur_score
        best_a = a_opt.copy()
    print(f"[L‑BFGS] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.65 * budget_s)
    print("[subgrad] start")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len,
        n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=800_000,
    )
    print(f"[subgrad] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Deconvolution refinement towards a flat convolution
    # ------------------------------------------------------------------
    target_val = 1.0 / (2 * n - 1)          # the average of a∗a when Σa = 1
    target = np.full(2 * n - 1, target_val)
    deconv_deadline = min(deadline - 5.0, start_time + 0.85 * budget_s)
    print("[deconv] start")
    a = best_a.copy()
    while time.time() < deconv_deadline:
        # compute a de‑convolved vector that would give a perfectly flat convolution
        x = deconvolution_step(a, target, fft_len, eps=1e-8)
        x = np.maximum(x, 0.0)
        x = project_onto_simplex(x)

        # blend a little towards the deconvolved candidate
        alpha = 0.12
        a = (1.0 - alpha) * a + alpha * x
        a = project_onto_simplex(a)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    print(f"[deconv] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill‑climb] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=30_000,
            max_move=0.30,
        )
        print(f"[final hill‑climb] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
