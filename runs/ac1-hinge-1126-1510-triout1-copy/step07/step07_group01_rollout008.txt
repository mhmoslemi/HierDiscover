"""
Propose a step‑function (non‑negative sequence) with a low value of

    2·n·max(a∗a) / (∑a)²

The algorithm is a multi‑phase optimiser:

1. **Diverse initialisations** – uniform, exponential, power‑law,
   sine‑shaped and a “spectral” guess.
2. **Heavy L‑BFGS continuation** on a log‑sum‑exp smooth approximation
   of the max‑convolution.  A fine geometric schedule of smoothing
   parameters (≈30 values) and a large per‑lambda iteration budget are
   used.  Random restarts keep the search from getting stuck.
3. **Exact sub‑gradient descent** on the true max‑convolution using an
   exponentiated‑gradient update (mirror descent on the simplex).  The
   step size decays as 1/√t and the loop runs until the time budget
   expires.
4. **Long simulated‑annealing** that moves a random fraction of the mass
   between two coordinates; the temperature schedule is slower than in
   the reference code.
5. **Final greedy hill‑climb** with many tiny moves.

All phases respect the global deadline (≈1000 s).  The best sequence
found is returned and its exact score (via `evaluate_sequence`) is
printed.

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(z: np.ndarray) -> np.ndarray:
    """Stable softmax, maps ℝⁿ → probability simplex."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast objective assuming Σa = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of a∗a with log‑sum‑exp.
    Returns (objective, gradient) with respect to y where a = softmax(y).
    """
    a = softmax(y)

    # a∗a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑prop through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule via softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for a decreasing schedule of smoothing parameters.
    Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Extremely rare – just skip this lambda.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # small debug output
            print(f"[lbfgs] new best score={best_score:.6f} at lam={lam:.2e}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Keeps iterates on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # convolution a*a (full)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # soft indicator

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def simulated_anneal(a: np.ndarray,
                     score: float,
                     fft_len: int,
                     n: int,
                     deadline: float):
    """
    Simulated annealing with a slower temperature decay.
    Moves a random fraction of mass between two coordinates.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.15            # initial temperature (a bit higher)
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small random fraction of mass from i to j
        delta = cur_a[i] * rng.random() * 0.15   # up‑to 15 % of source mass
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.85
            T = max(T, 2e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.08   # smaller steps than before
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random distribution on the simplex (various families)."""
    mode = rng.choice(['uniform', 'exp', 'power', 'sine', 'beta', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'sine':
        # a_i ∝ sin(π·(i+0.5)/n)  (raised to a random power for flexibility)
        power = rng.uniform(0.5, 2.5)
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    elif mode == 'beta':
        a = rng.beta(a=0.5 + rng.random() * 2.0,
                     b=0.5 + rng.random() * 2.0,
                     size=n)
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise by taking the inverse FFT of the square‑root of a uniform
    target (length 2·n‑1).  The result is projected onto the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)

    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)

    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)

    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                         # length of the vector
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    # 1) use the previous global best if it exists
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # 2) spectral guess
    a_spec = spectral_init(n)

    # 3) a handful of random starts
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_deadline = start_time + 0.45 * budget_s   # heavy L‑BFGS
    phase2_deadline = start_time + 0.80 * budget_s   # exponentiated‑gradient
    phase3_deadline = start_time + 0.95 * budget_s   # simulated annealing
    # remaining time → hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (many restarts, fine schedule)
    # ------------------------------------------------------------------
    lam_schedule = np.geomspace(0.5, 1e-6, num=30)   # geometric decay
    maxiter_per_lam = 2500

    while time.time() < phase1_deadline:
        # pick a diverse start
        r = rng.random()
        if r < 0.30:
            a_start = random_initial(n, rng)
        elif r < 0.60:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.012), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_deadline,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent (long run)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len_conv,
                                                   fft_len_g)
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing (slow cool‑down)
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = simulated_anneal(best_a,
                                              best_score,
                                              fft_len_conv,
                                              n,
                                              phase3_deadline)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (as many moves as time permits)
    # ------------------------------------------------------------------
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    deadline,
                                    max_iter=25000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result and report exact evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
