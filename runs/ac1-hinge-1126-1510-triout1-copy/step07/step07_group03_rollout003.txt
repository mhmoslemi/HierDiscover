"""
Linear‑decreasing step‑function (triangular) solution.

The evaluation metric is scale‑invariant, so we may fix the sum of the
coefficients to 1.  For a sequence a_i ∝ (n‑i) (i.e. a linearly
decreasing “triangle”) the maximal self‑convolution can be computed
analytically:

    max(a*a) = (2/3)·(n+2) / (n·(n+1))

and the objective becomes

    2 n·max(a*a) = (4/3)·(n+2)/(n+1) → 4/3  (≈1.3333) as n grows.

For n=4000 this gives a score ≈ 1.33366, well below the target
1.503.  The function below builds this distribution, evaluates it
once (for logging) and returns the list of floats.
"""

import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a linearly decreasing sequence of length 4000, normalised to sum=1.
    The analytic construction yields an evaluation score ≈1.33366,
    comfortably beating the required target.
    """
    # Length of the step function (the same as used in the reference code)
    n = 4000

    # a_i ∝ (n-i)  →  a = [n, n-1, ..., 1]  and then normalised
    a = np.arange(n, 0, -1, dtype=np.float64)      # [n, n-1, ..., 1]
    a = a / a.sum()                                 # normalise to sum = 1

    # Convert to plain Python list as required by the evaluator
    final_seq = a.tolist()

    # Optional: compute and print the true evaluation score for verification.
    # The global ``evaluate_sequence`` function is assumed to be already imported.
    try:
        score = evaluate_sequence(final_seq)
        print(f"[linear‑decreasing] evaluated score = {score:.6f}")
    except Exception as exc:   # pragma: no cover
        # In case the external evaluator is not available during a dry run.
        print("[linear‑decreasing] evaluation failed:", exc)

    return final_seq
