"""
propose_candidate
-----------------
A fast, multi‑phase local search for the L∞–self‑convolution problem.

* Initialise from the previously best known sequence (global `height_sequence_1`)
  or from a uniform distribution.
* Run many restarts of a continuation Adam optimiser on a log‑sum‑exp
  smooth approximation of the max‑convolution.  The smoothing factor is
  decreased geometrically and we take many gradient steps per factor
  (≈ 2 × 10⁴ steps total) – this yields a much tighter approach than in the
  reference implementation.
* Refine the best candidate with a projected sub‑gradient method that works
  on the true non‑smooth objective.  The active set (all near‑maximal
  convolution entries) is used to build a cheap averaged sub‑gradient.
* Finish with an aggressive greedy hill‑climb that repeatedly moves a random
  fraction of probability mass between coordinates, accepting only improving
  moves.  Because the objective is scale‑invariant we stay on the probability
  simplex throughout.
* The whole procedure respects the 1000 s time budget; the last few seconds
  are reserved for the final exact evaluation.

The returned list contains non‑negative floats (already normalised to sum = 1),
so it can be handed directly to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (for FFT lengths)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed to have length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    v = v.astype(float, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a log‑probability vector (pre‑soft‑max) for a given heuristic."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy construction of a Sidon‑like set (size ≈ √n)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.2):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)

    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a0 = 1.0 + p * (dist ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max via log‑sum‑exp ----
    shifted = conv - max_c          # now max = 0
    exp_shift = np.exp(shifted / lam)    # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv ----
    w = exp_shift / Z                # ∂smooth_max/∂conv_i

    # ---- propagate through the convolution ----
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max ----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)    # (diag(a) - a aᵀ)·g_vec

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 80,
) -> np.ndarray:
    """
    Continuation Adam: for each smoothing factor λ (expressed as a multiplicative
    `lam_factor`), perform a fixed number of Adam updates.
    Returns the final logits y.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)          # gentle decay
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only moves that strictly improve the exact (fast) objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(a*a).
    Uses the average of the sub‑gradients of all (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set: entries whose value is within a tiny relative tolerance of the maximum
        tol = max_val * 1e-10 + 1e-14
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                     # sub‑gradient of max(conv)

        eta = eta0 / np.sqrt(it)

        # simple back‑tracking line search (max 8 halvings)
        improved = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            # even after back‑tracking we did not improve; still move a bit
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) with a low evaluation
    score.  The algorithm consists of:

      1. initialise from the global `height_sequence_1` if present,
         otherwise from a uniform distribution;
      2. run many continuation‑Adam optimisations on a smooth surrogate;
      3. refine the best candidate with a projected sub‑gradient method;
      4. finish with a long greedy hill‑climb;
      5. return the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 3.0   # small safety margin

    # ------------------------------------------------------------------
    # Load or initialise a starting distribution on the simplex.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4_000:
            n = min(int(tmp.size), 10_000)
            a0 = tmp[:n]                     # trim if necessary
        else:
            n = 5_000
            a0 = np.full(n, 1.0 / n)
    else:
        n = 5_000
        a0 = np.full(n, 1.0 / n)

    # Normalise to the probability simplex (sum = 1)
    best_a = a0 / np.sum(a0)
    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation Adam with a rich schedule.
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-7, num=120))   # very smooth → very sharp
    steps_per_factor = 80                                   # many steps per λ
    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "ushape", "uniform"
    ]

    adam_deadline = min(deadline, time.time() + budget_s * 0.45)

    iter_cnt = 0
    while time.time() < adam_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = adam_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            steps_per_factor=steps_per_factor,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iter_cnt:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # short hill‑climb after each Adam run (≈ 0.4 s)
        hc_deadline = min(adam_deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2500
        )
        best_y = np.log(best_a + 1e-12)

        iter_cnt += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true objective.
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, time.time() + budget_s * 0.35)
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            subgrad_deadline,
            eta0=0.30,
            max_iters=800_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final aggressive hill‑climb using all remaining time.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=120_000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
