def grad_guided_hill_climb(a, current_score, fft_len_conv, fft_len_g, deadline, max_iter=50000):
    n = len(a)
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and iter < max_iter:
        # compute conv & mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient:
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad = 2 * np.real(np.fft.ifft(fm * fa_rev))[(n-1):(n-1+n)]
        # source and dest selection:
        src = np.argmax(grad)
        dst = np.argmin(grad)
        if src == dst: continue
        delta = best[src] * rng.uniform(0.001, 0.02)   # up to 2% of src mass
        if delta <= 0: continue
        # apply move
        new = best.copy()
        new[src] -= delta
        new[dst] += delta
        # compute new score
        new_score = fast_score(new, fft_len_conv, n)
        if new_score < best_score:
            best = new
            best_score = new_score
            # optionally update gradient next loop
        else:
            # maybe try smaller delta or different dest
            # for simplicity reject
            pass
        iter += 1
    return best, best_score

"""
Propose a low‑score step‑function sequence.

The algorithm works in three phases:

1. **Warm‑start** – if a previous construction ``height_sequence_1`` is
   available it is interpolated to length 4000 and normalised.
   Otherwise a uniform distribution is used.

2. **Gradient‑guided hill‑climb** – repeatedly
   * compute the self‑convolution,
   * identify the maximal entries (the “active” constraints),
   * compute the sub‑gradient of the max‑convolution,
   * move a small fraction of probability mass from the index with the
     largest gradient (the most harmful) to the index with the smallest
     gradient (the least harmful),
   * accept the move only if the objective strictly improves.
   When the search plateaus a tiny Gaussian perturbation is added to
   escape the local minimum.

3. **Final evaluation** – the best sequence found is returned as a plain
   ``list[float]``.  The time budget (≈ 1000 s) is respected by checking the
   deadline in every loop.

All heavy work is performed with FFT‑based convolutions
(`fast_score` and the sub‑gradient computation) which are
≈ O(n log n) for *n* = 4000 and therefore cheap enough to allow
hundreds of thousands of refinement steps.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using FFT.
    Returns an array of length 2*len(a)-1 (the ``valid`` part of the
    circular convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """
    Fast evaluation of the objective:
        2 * n * max(a*a)
    where a is assumed to be a probability vector (sum=1).
    """
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise to a and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb
# ----------------------------------------------------------------------


def grad_guided_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    max_iter: int = 500_000,
    plateau_patience: int = 2_000,
) -> tuple[np.ndarray, float]:
    """
    Refine ``a_start`` by repeatedly moving a small fraction of probability
    mass from the index with the largest sub‑gradient to the one with the
    smallest.  Only strictly improving moves are kept.
    """
    a = a_start.copy()
    best_score = current_score
    best = a.copy()
    rng = np.random.default_rng()
    n = a.shape[0]

    iter_cnt = 0
    no_improve = 0

    while time.time() < deadline and iter_cnt < max_iter:
        # ---- convolution and active‑set mask --------------------------------
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ---- sub‑gradient of the max‑convolution ---------------------------
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # ---- choose source (largest gradient) and destination (smallest) ---
        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst:
            iter_cnt += 1
            continue

        # ---- try a handful of step sizes ----------------------------------
        improved = False
        for factor in (0.05, 0.02, 0.01, 0.005, 0.001):
            delta = best[src] * factor
            if delta <= 0.0:
                continue
            trial = best.copy()
            trial[src] -= delta
            trial[dst] += delta

            # keep the vector on the simplex (numerical safety)
            if trial[src] < 0.0:
                continue

            score_trial = fast_score(trial, fft_len_conv, n)
            if score_trial < best_score - 1e-14:  # strict improvement
                best = trial
                best_score = score_trial
                improved = True
                break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # ---- escape plateaus ------------------------------------------------
        if no_improve >= plateau_patience:
            sigma = rng.uniform(0.001, 0.010)
            best = add_noise(best, sigma, rng)
            best_score = fast_score(best, fft_len_conv, n)
            no_improve = 0

        iter_cnt += 1

    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (optional, used for a quick boost)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[np.ndarray, float]:
    """
    Classical exponentiated‑gradient update on the exact max‑convolution.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # step size schedule – decreasing like 1/√t, larger prefactor
        eta = 1.0 / np.sqrt(it + 1.0)

        # EG update (clipping the exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best = a.copy()
        it += 1

    return best, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size ---------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)          # for gradient (mask * rev(a))

    # ------------------------------------------------------------------
    # Warm‑start: use the previously saved construction if present

    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            if init_seq.size != n:

                x_old = np.linspace(0.0, 1.0, init_seq.size)
                x_new = np.linspace(0.0, 1.0, n)
                init_seq = np.interp(x_new, x_old, init_seq)
            init_seq = np.maximum(init_seq, 0.0)
            init_seq /= init_seq.sum()
        else:
            init_seq = np.full(n, 1.0 / n)
    else:
        init_seq = np.full(n, 1.0 / n)

    best_a = init_seq.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # ------------------------------------------------------------------
    # Stage 1 – short exponentiated‑gradient phase (quick improvement)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s

        final_deadline,
        max_iter=1_000_000,

    )
    print(f"[grad‑hill] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
