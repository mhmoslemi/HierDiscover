"""
Propose a candidate step‑function sequence with a low evaluation score.

The algorithm combines three ideas:

1. **Parameterized shape search** – we generate a few families of symmetric
   distributions (exponential, power‑law, edge‑flat) with a small number of
   free parameters, evaluate them quickly with an FFT‑based “fast_score”
   and keep the best.

2. **Continuous optimisation** – the best shape is refined with a
   continuation of L‑BFGS on a smooth log‑sum‑exp approximation of the
   max‑convolution, followed by an exact exponentiated‑gradient descent
   (mirror descent on the probability simplex) that uses the true
   sub‑gradient of the max‑convolution.

3. **Greedy polishing** – a final hill‑climb moves a random fraction of mass
   between coordinates and accepts only improving moves.

The whole procedure respects the 1000 s budget (with a small safety margin)
and works on a problem size n = 8192 (a power of two, which is FFT‑friendly).
All numbers are non‑negative and normalised to sum to 1, which makes the
objective independent of an overall scaling.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT helpers and fast objective
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of   2·n·max(a*a)   when a sums to 1.
    This is the numerator of ``evaluate_sequence`` (the denominator is 1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Shape generators (symmetric around the centre)
# ----------------------------------------------------------------------


def shape_exponential(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c * distance_to_nearest_end)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = np.exp(-c * d)
    a /= a.sum()
    return a


def shape_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (distance_to_nearest_end + 1)^(-p)."""
    d = np.arange(n)
    d = np.minimum(d, n - 1 - d)
    a = (d + 1) ** (-p)
    a /= a.sum()
    return a


def shape_edge_flat(n: int, frac: float, epsilon: float = 1e-6) -> np.ndarray:
    """
    Uniform mass on the left/right edges, tiny baseline epsilon elsewhere.
    ``frac`` is the fraction of positions taken from each side.
    """
    a = np.full(n, epsilon, dtype=float)
    m = max(1, int(n * frac))
    a[:m] = 1.0
    a[-m:] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Random initialisation on the simplex (used for L‑BFGS restarts)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of the max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient components that correspond to a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation on a schedule of smoothing parameters
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits for the soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (mirror descent on the simplex)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exact exponentiated‑gradient descent on the hard max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule – decreasing like 1/√t, with a modest prefactor
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 40000):
    """
    Move a random fraction of the mass from one coordinate to another.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10   # up to 10 % of source mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search proceeds in three stages:
      1. quick sweep over a few parametric families,
      2. a continuation of L‑BFGS on a smoothed objective,
      3. exact exponentiated‑gradient descent,
      4. greedy hill‑climb polishing.

    The best sequence found before the deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # -------------------- problem size --------------------
    n = 8192                       # power of two → fast FFTs
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # -------------------- initialise from a stored solution if any --------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # -------------------- time budget splits --------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # 5 % for cheap parametric sweep, 15 % for L‑BFGS,
    # 70 % for EG, rest for hill‑climb
    sweep_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.20 * budget_s
    phase2_end = start_time + 0.90 * budget_s

    # -------------------- 0. cheap parametric sweep --------------------
    # families: exponential, power‑law, edge‑flat
    # (few parameter choices – negligible time)
    param_best_score = best_score
    param_best_a = best_a.copy()

    # exponential family
    for c in np.logspace(-4, -0.3, 12):   # roughly 0.0001 … 0.5
        a = shape_exponential(n, c)
        sc = fast_score(a, fft_len_conv, n)
        if sc < param_best_score:
            param_best_score = sc
            param_best_a = a.copy()
    # power‑law family
    for p in np.linspace(0.2, 4.0, 12):
        a = shape_power(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < param_best_score:
            param_best_score = sc
            param_best_a = a.copy()
    # edge‑flat family
    for frac in np.linspace(0.001, 0.04, 10):
        a = shape_edge_flat(n, frac, epsilon=1e-7)
        sc = fast_score(a, fft_len_conv, n)
        if sc < param_best_score:
            param_best_score = sc
            param_best_a = a.copy()

    if param_best_score < best_score:
        best_score = param_best_score
        best_a = param_best_a.copy()
        print(f"[sweep] new best fast_score={best_score:.6f}")

    # -------------------- 1. L‑BFGS continuation (aggressive) --------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 800   # fewer iterations – larger n costs more

    while time.time() < phase1_end:
        # Occasionally perturb the current best, otherwise a fresh random start
        if best_score < np.inf and rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # -------------------- 2. Exponentiated‑gradient descent --------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # -------------------- 3. Greedy hill‑climb polishing --------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=80000,
    )
    print(f"[hill‑climb] fast_score={best_score:.6f}")

    # -------------------- final evaluation with the official scorer --------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
