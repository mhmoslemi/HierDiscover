"""
Propose a step‑function sequence with a low evaluation score.

The search consists of three stages:

1. **Exploratory parametric sweep** – we generate many candidate
   distributions from simple parametric families (symmetric Beta
   densities and two‑block edge‑heavy shapes).  Each candidate is
   evaluated with a fast FFT based routine.

2. **Adam‑style sub‑gradient optimisation** – starting from the best
   candidate we perform projected gradient descent on the simplex.
   The true sub‑gradient of the max‑convolution is used; the update
   is performed on the soft‑max logits with an Adam optimiser.
   This quickly sharpens the distribution and typically beats the
   previous L‑BFGS / hill‑climb pipeline.

3. **Targeted gradient‑guided hill‑climb** – a final deterministic
   local search that moves mass from indices with a large gradient
   to those with a small gradient, accepting only strictly improving
   moves.

The algorithm respects the 1000 s time budget, prints progress,
and finally calls ``evaluate_sequence`` on the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution with sum(a)=1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def beta_distribution(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Symmetric (or asymmetric) Beta density on the grid.
    Returns a probability vector (sum = 1).
    """
    x = (np.arange(n) + 0.5) / n  # mid‑point discretisation, avoids 0/1
    # Compute raw (unnormalised) density; safe for alpha,beta > 0.
    a = np.power(x, alpha - 1.0)
    b = np.power(1.0 - x, beta - 1.0)
    raw = a * b
    # Numerical safety – replace infinities / NaNs with 0.
    raw = np.nan_to_num(raw, nan=0.0, posinf=0.0, neginf=0.0)
    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return raw / s


def edge_distribution(edge_frac: float, ratio: float, n: int) -> np.ndarray:
    """
    Two‑block edge‑heavy distribution.
    ``edge_frac`` – fraction of points (per side) that form the edge block.
    ``ratio`` – weight of an edge point relative to a middle point.
    """
    L = int(edge_frac * n)
    if L == 0:
        return np.full(n, 1.0 / n)
    # unknown middle weight w_mid, edge weight w_edge = ratio * w_mid
    w_mid = 1.0 / (2 * L * ratio + (n - 2 * L))
    w_edge = ratio * w_mid
    a = np.full(n, w_mid)
    a[:L] = w_edge
    a[-L:] = w_edge
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient utilities (true, not smoothed)
# ----------------------------------------------------------------------


def max_convolution_subgrad(a: np.ndarray,
                            fft_len_conv: int,
                            fft_len_g: int) -> tuple[float, np.ndarray]:
    """
    Returns (max_val, g) where ``max_val`` is the maximal value of a*a,
    and ``g`` is the sub‑gradient of the *max* of the convolution
    (size n).  The gradient assumes the max is an average over all
    active indices (conv values within numerical tolerance of the max).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    # active set (values indistinguishably close to the max)
    eps = max_val * 1e-12 + 1e-12
    active = conv >= max_val - eps
    k = active.sum()
    if k == 0:
        # fallback – treat the sole max as active
        mask = np.zeros_like(conv)
        mask[conv.argmax()] = 1.0
    else:
        mask = np.zeros_like(conv)
        mask[active] = 1.0 / float(k)      # uniform convex combination

    # back‑propagation through convolution (size n)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[n - 1: n - 1 + n]     # ∂max/∂a_i (no outer factor)
    return max_val, g


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑softmax parameterisation)
# ----------------------------------------------------------------------


def adam_optimize(a0: np.ndarray,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int,
                  base_lr: float = 0.20,
                  max_iter: int = 20000) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the simplex using the true sub‑gradient of the
    max‑convolution.  Returns the best vector found and its fast_score.
    """
    # initialise logits
    y = np.log(a0 + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # probability vector from logits
        a = np.exp(y - np.max(y))
        a /= a.sum()

        # compute sub‑gradient of the max‑convolution
        max_val, g_raw = max_convolution_subgrad(a, fft_len_conv, fft_len_g)
        # gradient of the full objective f = 2*n*max_val
        g = (2.0 * n) * g_raw

        # chain rule through softmax
        grad_y = a * (g - np.dot(a, g))

        # Adam update (bias‑corrected)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y -= base_lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate current candidate
        a_cur = np.exp(y - np.max(y))
        a_cur /= a_cur.sum()
        cur_score = fast_score(a_cur, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_cur.copy()

        # occasional jitter to escape flat plateaus
        if it % 2000 == 0:
            if np.random.rand() < 0.15:
                sigma = np.random.uniform(0.001, 0.008)
                y += np.random.normal(0.0, sigma, size=n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (deterministic final polishing)
# ----------------------------------------------------------------------


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 50000) -> tuple[np.ndarray, float]:
    """
    Move a modest fraction of mass from indices with a large gradient to
    those with a small gradient; accept only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        max_val, g_raw = max_convolution_subgrad(best, fft_len_conv, fft_len_g)
        # scale to the objective
        g = (2.0 * n) * g_raw

        # thresholds for “high” and “low” gradient entries
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # move at most 7 % of the source mass
        delta = best[i] * rng.random() * 0.07
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        # else discard the move
        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient back‑propagation

    # --------------------------------------------------------------
    # Initialise with uniform distribution (simple baseline)
    # --------------------------------------------------------------
    a_best = np.full(n, 1.0 / n)
    best_score = fast_score(a_best, fft_len_conv, n)
    print(f"[init] uniform score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – exhaustive sweep over simple parametric families
    # --------------------------------------------------------------
    # 1a – symmetric Beta(p,p)
    print("[phase 1] scanning symmetric Beta families")
    for p in np.linspace(0.55, 2.0, 120):
        a = beta_distribution(p, p, n)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, a_best = sc, a.copy()
    # 1b – random asymmetric Beta
    for _ in range(250):
        alpha = rng.uniform(0.35, 3.0)
        beta = rng.uniform(0.35, 3.0)
        a = beta_distribution(alpha, beta, n)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, a_best = sc, a.copy()
    # 1c – edge‑heavy two‑block shapes
    print("[phase 1] scanning edge‑heavy shapes")
    for edge_frac in np.linspace(0.0, 0.07, 15):
        for ratio in np.linspace(1.0, 12.0, 15):
            a = edge_distribution(edge_frac, ratio, n)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, a_best = sc, a.copy()

    print(f"[phase 1] best score after parametric sweep = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam sub‑gradient optimisation (continuous refinement)
    # --------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.4   # keep a safety margin for final step
    a_best, best_score = adam_optimize(
        a_best,
        deadline,
        fft_len_conv,
        fft_len_g,
        n,
        base_lr=0.20,
        max_iter=20000,
    )
    print(f"[phase 2] after Adam optimisation score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Targeted gradient‑guided hill‑climb (final polishing)
    # --------------------------------------------------------------
    final_deadline = start + budget_s - 0.1
    a_best, best_score = targeted_hill_climb(
        a_best,
        best_score,
        fft_len_conv,
        fft_len_g,
        n,
        final_deadline,
        max_iter=60000,
    )
    print(f"[phase 3] final polished score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided function
    # --------------------------------------------------------------
    final_seq = a_best.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence -> {final_eval:.6f}")

    return final_seq
