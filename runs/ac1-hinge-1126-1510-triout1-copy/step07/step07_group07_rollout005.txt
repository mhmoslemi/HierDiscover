def adam_optimize(a_start, fft_len, n, deadline, lr=0.1, max_iters=1_000_000):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute gradient g (subgradient)
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

"""
Advanced search for a low‑evaluation step‑function.

The algorithm works in several stages:

1. **Problem size** – choose a modestly large `n` (≈ 8000–12000) and
   work on the probability simplex (∑a_i = 1).  The objective
   2·n·max(conv(a,a)) equals the value returned by `evaluate_sequence`.

2. **Rich initial families** – β‑distributions, semicircle,
   uniform, Gaussian, sparse binary masks and a few mixtures.
   The best few are refined with a continuation L‑BFGS optimizer that
   uses a smoothed log‑sum‑exp surrogate of the max‑convolution.

3. **Exploratory random restarts** – many short L‑BFGS runs
   from random Dirichlet‑type starts, followed by a few hundred
   projected Adam‑style sub‑gradient steps on the *exact* max‑convolution.

4. **Global simulated‑annealing polish** – multiplicative perturbations
   of the current best vector accepted with a temperature schedule.

5. **Aggressive hill‑climbing** – final random / targeted moves.

All phases respect a total 1000 s budget and the best sequence found
is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n-1) using FFT.
    a is length n, fft_len ≥ 2*n-1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a_i = 1):
    returns 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (a small smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c  # makes the max become 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w = exp_shift / Z

    # correlation of a and w (gradient of conv w.r.t. a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def beta_init(n: int, p: float) -> np.ndarray:
    """β(p,p) density on (0,1); p≈0.5 gives the classic arcsine."""
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, p - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)  # ensure positivity
    a = a / a.sum()
    return a


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, re‑normalised to sum 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: transfer a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from the current
    max‑convolution contributors.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def adam_subgradient(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    lr: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style sub‑gradient descent on the *exact* objective.
    The gradient is the sub‑gradient of max(conv) (i.e. g_i = 2·a_{m‑i}).
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)

    m1 = np.zeros_like(a)  # first moment
    m2 = np.zeros_like(a)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute sub‑gradient at current a
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam update
        m1 = beta1 * m1 + (1.0 - beta1) * g
        m2 = beta2 * m2 + (1.0 - beta2) * (g * g)
        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        step = lr * m1_hat / (np.sqrt(m2_hat) + eps)
        a = a - step
        a = project_onto_simplex(a)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Implements the multi‑stage pipeline described
    in the module docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Choose problem size (larger n gives more expressive power)
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            # use at most 12000 entries to keep FFT cheap
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # --------------------------------------------------------------
    # Build a diverse pool of initial shapes
    # --------------------------------------------------------------
    candidates = []

    # β(p,p) for several p values
    for p in [0.55, 0.60, 0.65, 0.70, 0.75, 0.80]:
        candidates.append((f"beta_{p:.2f}", beta_init(n, p)))

    # semicircle, uniform, random Gaussian, sparse binary mask, bump
    candidates.append(("semicircle", semicircle_init(n)))
    candidates.append(("uniform", uniform_init(n)))
    candidates.append(("random_gauss", random_gaussian_init(n, rng)))
    candidates.append(("binary_sparse", binary_sparse_init(n, rng, density=0.15)))
    candidates.append(("bump", gaussian_bump_init(n, rng)))

    # Mixtures of a good beta with uniform
    beta_ref = beta_init(n, 0.70)
    for w in [0.2, 0.4, 0.6, 0.8]:
        candidates.append((f"beta_mix_{w:.1f}", mixture_init(beta_ref, uniform_init(n), w)))

    # Historic sequence (if available)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp / tmp.sum()
            candidates.append(("previous", a))

    # --------------------------------------------------------------
    # Quick scan — keep a few promising initialisations
    # --------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len_conv, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:20s} fast score = {sc:.6f}")

    scored.sort()                     # low score first
    top_k = 4                         # number of candidates for full L‑BFGS
    best_a = None
    best_score = np.inf
    best_y = None

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS on the best few candidates
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(5.0, 1e-9, num=28))
    maxiter_per_stage = 600

    phase1_deadline = start_time + 0.45 * budget_s
    print("[phase1] L‑BFGS continuation started")
    for rank in range(min(top_k, len(scored))):
        sc, name, a0 = scored[rank]
        y0 = np.log(a0 + 1e-12)

        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[l‑bfgs] new best from {name:20s} score = {best_score:.6f}")

        # short hill‑climb on the current best (opportunistic)
        if time.time() < phase1_deadline:
            bc, bs = hill_climb(
                best_a, best_score, fft_len_conv, n,
                deadline=phase1_deadline, max_iter=2000
            )
            if bs < best_score:
                best_a, best_score = bc, bs
                best_y = np.log(best_a + 1e-12)
                print(f"[l‑bfgs] post‑hill‑climb improved score = {best_score:.6f}")

        if time.time() >= phase1_deadline:
            print("[phase1] time exhausted")
            break

    if best_a is None:
        # fallback (should never happen)
        best_a = uniform_init(n)
        best_score = fast_score(best_a, fft_len_conv, n)

    # --------------------------------------------------------------
    # Phase 2 – random restarts with short L‑BFGS + Adam fine‑tuning
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    restart_counter = 0
    short_lam = lam_schedule[:12]          # fewer smoothing steps for speed
    while time.time() < phase2_deadline:
        restart_counter += 1
        # Dirichlet‑style random start (concentration 0.7 gives moderate sparsity)
        a0 = rng.dirichlet(alpha=np.full(n, 0.7))
        y0 = np.log(a0 + 1e-12)

        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, short_lam, maxiter_per_stage=300
        )
        a_opt = softmax(y_opt)

        # Adam sub‑gradient refinement (time‑boxed)
        sub_deadline = min(phase2_deadline, time.time() + 8.0)  # ~8 s per restart
        a_opt, score_opt = adam_subgradient(
            a_opt, fft_len_conv, n, sub_deadline, lr=0.25, max_iters=200_000
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[restart] #{restart_counter:04d} improved score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing global polish
    # --------------------------------------------------------------
    phase3_deadline = start_time + 0.90 * budget_s
    cur = best_a.copy()
    cur_score = best_score
    temperature = 0.5 * cur_score  # initial temperature proportional to score
    sigma = 0.06                  # magnitude of multiplicative noise

    print("[anneal] started simulated annealing")
    while time.time() < phase3_deadline:
        # multiplicative perturbation
        a_new = cur * np.exp(rng.normal(scale=sigma, size=n))
        a_new = a_new / a_new.sum()
        new_score = fast_score(a_new, fft_len_conv, n)

        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / temperature):
            cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                best_y = np.log(best_a + 1e-12)
                print(f"[anneal] new best score = {best_score:.6f}")

        # exponential cooling (very slow)
        temperature *= 0.99997
        # gradually shrink noise as we approach deadline
        if time.time() > phase3_deadline - 60:
            sigma *= 0.995

    # --------------------------------------------------------------
    # Phase 4 – aggressive final hill‑climb polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] aggressive random hill‑climb")
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n,
            final_deadline, max_iter=15000, max_move=0.30
        )
        if time.time() < deadline:
            print("[final] targeted hill‑climb")
            best_a, best_score = targeted_hill_climb(
                best_a, best_score, fft_len_conv, n, deadline,
                max_iter=8000, max_move=0.20
            )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Final evaluation (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)
        final_val = None

    return final_seq
