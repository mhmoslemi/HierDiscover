"""
Propose a low‑score non‑negative step sequence.

Algorithm
---------
* Choose a moderate problem size (n≈5000) – large enough to give a
  flexible shape but still cheap to FFT.
* Initialise a diverse pool of candidates:
    – uniform, spectral “square‑root of uniform”, interpolation of a
      previously stored best, random Dirichlet‑like draws.
* Phase 1 (≈40 % of the time budget): many random‑restart L‑BFGS
  minimisation of a *smooth* surrogate of the max‑convolution.
  The surrogate uses a log‑sum‑exp (“soft‑max”) with a decreasing
  smoothing factor λ (implemented as a factor of the current max).
* Phase 2 (≈50 %): exponentiated‑gradient descent on the *exact*
  max‑convolution.  The step size follows ηₜ = η₀ / √t and we keep the
  iterate on the probability simplex via normalisation.
  Several restarts with small Gaussian perturbations of the current
  best are tried.
* Phase 3 (≈10 %): aggressive greedy hill‑climbing – repeatedly move a
  tiny random fraction of mass from one coordinate to another and keep
  strictly improving moves.
* The best vector found (always normalised to sum = 1) is returned.
  The evaluator `evaluate_sequence` is finally called to report the
  exact score.

All stages respect the global 1000 s deadline; the code constantly
keeps the current best sequence and prints intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is on the simplex (∑a=1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex helpers
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random distribution on the simplex (various families)."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the FFT
    of a uniform target (length 2·n‑1), then project onto the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    sqrt_amp = np.sqrt(np.abs(S))
    sqrt_phase = np.angle(S) / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    a /= a.sum() if a.sum() > 0 else 1.0
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # a*a (full convolution)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)  # ≈ max_c when lam→0
    f = 2.0 * n * smooth_max

    # derivative of smooth max w.r.t. each conv entry
    w = exp_shift / Z          # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       time_deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in lam_schedule, starting each stage from
    the result of the previous stage. Return the best a and its fast_score.
    """
    # initialise logits; add tiny epsilon to avoid log(0)
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= time_deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # very unlikely, but guard against numerical failures
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # debug
            print(f"[lbfgs] improved score={best_score:.6f} (lam={lam})")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the *exact* max‑convolution.
    The update a ← a·exp(−η·g) automatically stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # exact convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max­‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule – larger at the start, decays as 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip for safety)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # debug
            print(f"[exp‑grad] iter {it} new best={best_score:.6f}")

        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a tiny random fraction of mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07  # up to 7 % of the source mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # debug occasional
            if it % 2000 == 0:
                print(f"[hill‑climb] iter {it} best={best_score:.6f}")
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    # ------------------------------------------------------------------

    # Build a pool of diverse starters

    candidates.append(np.full(n, 1.0 / n))

        prev = np.array(globals()['height_sequence_1'], dtype=float)

            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)

            a_interp = np.maximum(a_interp, 0.0)
            a_interp /= a_interp.sum() if a_interp.sum() > 0 else 1.0

    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget partition
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    phase1_deadline = start_time + 0.40 * budget_s
    phase2_deadline = start_time + 0.90 * budget_s
    phase3_deadline = deadline

    # Phase 1 – many random‑restart L‑BFGS refinements
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 1500

    restart = 0
    while time.time() < phase1_deadline:
        restart += 1
        # diversify start points
        r = rng.random()
        if r < 0.3:
            a_start = random_initial(n, rng)
        elif r < 0.6:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.012), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_deadline,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] restart {restart} new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (multiple restarts)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        eg_start = best_a.copy()
        # First long run from the current best
        eg_deadline = phase2_deadline - 5.0   # keep a little slack
        eg_best_a, eg_best_score = exp_gradient_optimize(eg_start,
                                                         eg_deadline,
                                                         n,
                                                         fft_len_conv,
                                                         fft_len_g,
                                                         rng)
        if eg_best_score < best_score:
            best_score = eg_best_score
            best_a = eg_best_a.copy()
            print(f"[exp‑grad] primary run improved to {best_score:.6f}")

        # A few short noisy restarts (each ~10 % of remaining time)
        remaining = phase2_deadline - time.time()
        n_restarts = max(1, int(remaining // 60))  # roughly one per minute
        for k in range(n_restarts):
            sigma = rng.uniform(0.002, 0.008)
            a_noise = add_noise(best_a, sigma, rng)
            sub_dead = time.time() + min(40.0, remaining / (n_restarts - k + 1))
            a_opt, sc_opt = exp_gradient_optimize(a_noise, sub_dead, n,
                                                  fft_len_conv, fft_len_g, rng)
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"[exp‑grad] restart {k+1}/{n_restarts} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        phase3_deadline,
                                        max_iter=25000)
        print(f"[final hill‑climb] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence; also report the exact evaluation score.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
