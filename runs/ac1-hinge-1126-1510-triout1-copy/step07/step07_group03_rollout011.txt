def guided_simulated_annealing(a, score, fft_len, n, deadline):
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.2
    it = 0
    while time.time() < deadline:
        # compute gradient at cur_a
        # subgradient of max conv
        fa = np.fft.fft(cur_a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # subgradient
        a_rev = cur_a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n-1):(2*n-1)]

        # choose source/target indices based on gradient
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]
        if high_idx.size >0 and low_idx.size >0:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
        else:
            i, j = rng.integers(0, n, size=2)
        # delta
        delta = cur_a[i] * rng.random() * 0.1
        if delta <= 0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score)/T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 2000 == 0:
            T *= 0.9
            T = max(T, 1e-5)
    return best_a, best_score
```

"""
Multi‑scale gradient/LP search for the step‑function lower bound.

The algorithm proceeds in several stages:
  1. Optimize on a coarse grid (n≈1024) with many random L‑BFGS runs.
  2. Upsample to a larger grid (2048, 4096, 8192) and continue L‑BFGS, using the
     solution from the previous stage as a warm‑start.
  3. A short linear‑programming refinement that moves mass from the “over‑loaded”
     convolution positions to a feasible direction while maximising the total mass.
  4. Gradient‑guided simulated‑annealing + a final deterministic hill‑climb.
All steps respect the simplex constraint (non‑negative entries summing to 1),
so the raw sequence can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is on the simplex (∑a = 1):
        2·n·max(a * a)
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp (soft‑max) objective and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of 2·n·max(a*a) where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. the convolution entries
    w = exp_shift / Z                                 # d smooth_max / d conv_i

    # back‑propagation through the convolution (a↔a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]            # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage
    from the result of the previous stage.  Returns the best a and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits (softmax argument)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:
            # Very rare numerical failure – skip this stage
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑normalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:          # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def upsample_vector(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger size."""
    x_old = np.linspace(0.0, 1.0, a_small.size)
    x_new = np.linspace(0.0, 1.0, n_big)
    a_big = np.interp(x_new, x_old, a_small)
    a_big = np.maximum(a_big, 0.0)
    s = a_big.sum()
    if s <= 0.0:
        return np.full(n_big, 1.0 / n_big)
    return a_big / s

# Linear‑programming refinement (active‑set)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray,
              deadline: float,
              n: int,
              fft_len_conv: int):
    """
    One‑step LP refinement as described in the paper sketch.
    Uses only the currently active maximal‑convolution positions.
    Returns a (possibly) improved a and its fast_score.
    """
    # current convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    M = conv.max()
    eps = M * 1e-6 + 1e-12
    active = np.where(conv >= M - eps)[0]
    if active.size == 0:
        return a, fast_score(a, fft_len_conv, n)

    # Build the (small) constraint matrix A (|active| × n)
    m = active.size
    A = np.zeros((m, n))
    for row, k in enumerate(active):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # For j in [lo, hi] we have A[row, j] = a[k - j]
        idx = np.arange(lo, hi + 1, dtype=int)
        A[row, lo:hi + 1] = a[k - idx]

    # Linear program: maximise sum(b)  ⇔ minimise –∑b
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        return a, fast_score(a, fft_len_conv, n)

    b_opt = res.x
    s = b_opt.sum()
    if s <= 0.0:
        return a, fast_score(a, fft_len_conv, n)
    b_norm = b_opt / s

    # Mix the new vector with the old one (small step size)
    t = 0.3
    a_new = (1.0 - t) * a + t * b_norm
    new_score = fast_score(a_new, fft_len_conv, n)
    if new_score < fast_score(a, fft_len_conv, n):
        return a_new, new_score
    return a, fast_score(a, fft_len_conv, n)

def guided_anneal(a: np.ndarray,
                  score: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int,
                  deadline: float):
    """
    Simulated‑annealing that prefers moves from high‑gradient to low‑gradient
    positions, using the exact sub‑gradient of the max‑convolution.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.15          # initial temperature
    it = 0

    while time.time() < deadline:
        # --- sub‑gradient of the max convolution ---
        fa = np.fft.fft(cur_a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = cur_a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(2 * n - 1)]

        # --- choose source / target using percentiles of g ---
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]
        if high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
        else:
            i, j = rng.integers(0, n, size=2)

        if i == j:
            continue

        # transfer a modest fraction of the source mass
        delta = cur_a[i] * rng.random() * 0.09   # up to 9 %
        if delta <= 0.0:
            continue

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)

        # Metropolis acceptance
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 2500 == 0:
            T *= 0.85
            T = max(T, 3e-5)

    return best_a, best_score

# Deterministic hill‑climb (gradient‑guided, strictly improving)
# ----------------------------------------------------------------------


def targeted_hill_climb(a: np.ndarray,
                        current_score: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 80000):
    """
    Greedy hill‑climb that moves a small fraction of mass from a high‑gradient
    index to a low‑gradient one, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(2 * n - 1)]

        # choose source / target by percentile tails
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        delta = best[i] * rng.random() * 0.07   # up to 7 %
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1

    return best, best_score

def optimize_scale(n: int,
                   init_a: np.ndarray | None,
                   deadline: float,
                   rng: np.random.Generator):
    """
    Optimize a distribution on a grid of size n.
    If ``init_a`` is supplied it is up‑sampled and used as a warm‑start.
    """
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # initialise
    if init_a is not None:
        a0 = upsample_vector(init_a, n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[optimize_scale] n={n}, start_score={best_score:.6f}")

    lam_schedule = [
        0.70, 0.45, 0.30, 0.20, 0.12, 0.07,
        0.04, 0.025, 0.015, 0.008, 0.004,
        0.002, 0.001, 5e-04, 2e-04, 1e-04, 5e-05
    ]
    maxiter_per_lam = 700

    # main loop – keep trying random starts until the phase deadline
    while time.time() < deadline:
        # mixture: sometimes perturb the current best, otherwise fresh random start
        if best_score < np.inf and rng.random() < 0.18:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[scale {n}] new best L‑BFGS score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the provided
    ``evaluate_sequence`` objective.  The routine uses a multi‑scale
    L‑BFGS + LP refinement + gradient‑guided annealing + hill‑climb pipeline.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin for the final evaluation call
    deadline = start_time + budget_s - 0.2

    # ------------------------------------------------------------------
    # 1. Multi‑scale L‑BFGS
    # ------------------------------------------------------------------
    scales = [1024, 2048, 4096, 8192]          # increasing resolutions
    # allocate roughly 10 % / 20 % / 30 % / 30 % of the total budget
    fractions = [0.10, 0.20, 0.30, 0.30]
    cum = 0.0
    prev_a = None
    for n, frac in zip(scales, fractions):
        phase_end = start_time + (cum + frac) * budget_s - 0.2
        phase_end = min(phase_end, deadline)
        if time.time() >= phase_end:
            break
        best_a, best_score = optimize_scale(n, prev_a, phase_end, rng)
        print(f"[stage n={n}] best score after L‑BFGS = {best_score:.6f}")
        prev_a = best_a
        cum += frac

    # ------------------------------------------------------------------
    # 2. Linear‑programming refinement (active‑set)
    # ------------------------------------------------------------------
    final_n = prev_a.shape[0]
    fft_len_conv = next_pow2(2 * final_n - 1)
    fft_len_g = next_pow2(3 * final_n - 2)

    # Give the LP a few seconds (or whatever time is left) to improve the
    # current candidate.
    lp_deadline = min(deadline, time.time() + 30.0)
    best_a, best_score = lp_refine(prev_a, lp_deadline, final_n, fft_len_conv)
    print(f"[LP refine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Gradient‑guided simulated annealing
    # ------------------------------------------------------------------
    anneal_deadline = min(deadline, time.time() + 80.0)
    best_a, best_score = guided_anneal(best_a, best_score,
                                       fft_len_conv, fft_len_g,
                                       final_n, anneal_deadline)
    print(f"[guided anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Final deterministic hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = targeted_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        fft_len_g,
        final_n,
        final_deadline,
        max_iter=100000,
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation using the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
