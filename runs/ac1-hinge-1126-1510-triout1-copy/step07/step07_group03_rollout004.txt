    if s > 0:

"""
Search for a non‑negative step‑function (probability vector) that minimizes
    2·n·max(conv(a,a))  (the evaluation function is scale‑invariant).

The algorithm runs several independent phases for a few different lengths
(n≈2000‑8000).  For each length we:

1.  Perform a cheap systematic scan of simple parametric families (power,
    exponential, symmetric double‑exponential, edge‑heavy) to obtain a good
    starting point.
2.  Refine the best candidate with a fast Adam optimiser applied to a
    smooth approximation of the max‑convolution (log‑sum‑exp smoothing).
3.  Finish with a deterministic gradient‑guided hill‑climb that moves mass
    from indices with large sub‑gradient to those with small sub‑gradient.
4.  Keep the globally best sequence (any length) seen so far.

All operations are performed with FFT‑based convolutions (O(n log n)),
so even n≈8000 is tractable inside the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv(a,a)).
    Assumes a sums to 1 (simplex).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple families (used for the cheap scan)
# ----------------------------------------------------------------------


def power_family(n, p):
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def exponential_family(n, r):
    a = np.exp(-r * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def double_exponential_family(n, r):
    x = np.arange(n, dtype=float)
    a = np.exp(-r * x) + np.exp(-r * (n - 1 - x))
    a /= a.sum()
    return a


def edge_heavy_family(n, frac):
    m_edge = max(1, int(n * frac))
    a = np.full(n, 1.0 / n)
    heavy = np.sqrt(n / 2.0) / n
    a[:m_edge] = heavy
    a[-m_edge:] = heavy
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y). Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                # d smooth_max / d conv_i

    # back‑propagation through convolution (a*a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # the useful part corresponds to indices n‑1 … n‑1+n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the smoothed max‑convolution objective.
    Returns the best a (simplex) seen and its fast_score.
    """
    y = np.log(a_start + 1e-12)        # logits
    m = np.zeros_like(y)              # 1st moment
    v = np.zeros_like(y)              # 2nd moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # smoothing schedule – gradually tighten the approximation
    lam_sched = [(0, 0.5), (2000, 0.3), (5000, 0.2),
                 (10000, 0.1), (20000, 0.05), (40000, 0.02), (80000, 0.01)]

    while time.time() < deadline:
        # pick lam_factor according to current iteration
        lam = 0.5
        for thr, val in reversed(lam_sched):
            if it >= thr:
                lam = val
                break

        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # step size decays gently with sqrt(it)
        eta = 0.4 / (np.sqrt(it) + 1.0)
        y = y - eta * m_hat / (np.sqrt(v_hat) + eps)

        # occasional evaluation
        if it % 250 == 0:
            a = np.exp(y - np.max(y))
            a /= a.sum()
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        # safety cap
        if it >= 150000:
            break

    # final evaluation of the latest a
    a = np.exp(y - np.max(y))
    a /= a.sum()
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (final polishing)
# ----------------------------------------------------------------------


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 80000) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a modest fraction of mass from
    indices with a large sub‑gradient to those with a small sub‑gradient.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (size n)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # pick source from upper tail, target from lower tail
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # transfer up to 12 % of source mass (larger than before)
        delta = best[i] * rng.random() * 0.12
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1

    return best, best_score

# Helper utilities
# ----------------------------------------------------------------------


def upsample(a_old: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a_old (length m) to length new_n and renormalise."""
    old_n = a_old.size
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a_old)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        a_new = np.full(new_n, 1.0 / new_n)
    return a_new


def systematic_scan(n: int, fft_len: int) -> tuple[np.ndarray, float]:
    """
    Scan a few simple parametric families (power, exp, double‑exp, edge‑heavy)
    and return the best found vector and its fast_score.
    """
    best_a = None
    best_sc = np.inf

    # power law
    for p in np.linspace(0.1, 4.0, 40):
        a = power_family(n, p)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a

    # exponential
    for r in np.linspace(0.0005, 0.02, 40):
        a = exponential_family(n, r)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a

    # symmetric double exponential
    for r in np.linspace(0.0005, 0.02, 30):
        a = double_exponential_family(n, r)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a

    # edge‑heavy
    for frac in np.linspace(0.001, 0.02, 20):
        a = edge_heavy_family(n, frac)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a

    # a few random starts as a safety net
    rng = np.random.default_rng()
    for _ in range(200):
        a = rng.random(n)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin so we have a few hundred ms for final evaluation
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Lengths to explore (increasing order – we can up‑sample previous best)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]

    # global best across all lengths
    best_seq = None
    best_score = np.inf   # fast_score (lower is better)

    # if a previous construction exists, keep it for up‑sampling
    prev_seq = None
    if "height_sequence_1" in globals():
        tmp = np.array(globals()["height_sequence_1"], dtype=float)
        if tmp.ndim == 1 and tmp.size > 0:
            prev_seq = tmp / tmp.sum()

    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"[search] start length n={n}")

        fft_len_conv = next_pow2(2 * n - 1)
        fft_len_g = next_pow2(3 * n - 2)

        # --------------------------------------------------------------
        # initialise: either up‑sample previous best, or use systematic scan
        # --------------------------------------------------------------
        if prev_seq is not None:
            a0 = upsample(prev_seq, n)
        else:
            a0, _ = systematic_scan(n, fft_len_conv)

        # Quick systematic family scan to possibly improve the initial point
        a_scan, sc_scan = systematic_scan(n, fft_len_conv)
        if sc_scan < fast_score(a0, fft_len_conv, n):
            a0 = a_scan

        cur_score = fast_score(a0, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a0.copy()
            print(f"   [init] fast_score={cur_score:.6f}")

        # --------------------------------------------------------------
        # Phase A – Adam optimisation on smoothed objective
        # --------------------------------------------------------------
        phase_a_deadline = min(deadline,
                               time.time() + 0.30 * (deadline - start_time))
        a_opt, sc_opt = adam_optimize(a0, phase_a_deadline,
                                      fft_len_conv, fft_len_g, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_seq = a_opt.copy()
            print(f"   [adam] improved fast_score={sc_opt:.6f}")

        # --------------------------------------------------------------
        # Phase B – final gradient‑guided hill‑climb
        # --------------------------------------------------------------
        if time.time() < deadline:
            a_hc, sc_hc = targeted_hill_climb(best_seq, best_score,
                                              fft_len_conv, fft_len_g,
                                              n, deadline, max_iter=80000)
            if sc_hc < best_score:
                best_score = sc_hc
                best_seq = a_hc.copy()
                print(f"   [hill‑climb] improved fast_score={sc_hc:.6f}")

        # store for possible up‑sampling to larger n
        prev_seq = best_seq.copy()

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
