def generate_initial_a(mode, n, rng):
    if mode == "random":
        # Dirichlet(1)
        a = rng.random(n)
        a = np.abs(a)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a = np.exp(-rate * d)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n+1, dtype=float) ** (-p))
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        # use formula balancing sum=1
        a = np.full(n, 1.0 / n)
        a[:m_edge] = 3.0 / (2 * n)  # heuristically assign more mass at edges
        a[-m_edge:] = 3.0 / (2 * n)
        # ensure positivity; later normalize
    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    elif mode == "sidon":
        # greedy sidon set near sqrt(n) size
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2*cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.4):
                break
        a = np.zeros(n)
        a[chosen] = 1.0 / len(chosen)
    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a = rng.gamma(alpha, 1.0)
    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a = 1.0 + p * (dist ** 2)
    elif mode == "gaussian":
        mu = (n - 1) / 2.0
        sigma = n / (2.0 * rng.uniform(2.0, 8.0))  # random sigma between n/4 and n/16
        a = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)
    elif mode == "semicircle":
        x = (np.arange(n) - (n - 1) / 2.0) / ((n - 1) / 2.0)
        a = np.sqrt(np.maximum(0.0, 1 - x**2))
    elif mode == "tent":
        a = 1.0 - np.abs(np.arange(n) - (n - 1)/2.0) / ((n - 1)/2.0)
    elif mode == "uniform":
        a = np.full(n, 1.0)
    else:
        raise ValueError

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0)
    a = a / a.sum()
    return a
```

def smooth_obj_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    a = softmax(y)
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[:2*n-1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z
    # compute gradient via convolution with w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n-1):(n-1+n)]
    # gradient w.r.t softmax logits y
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    # objective
    f = 2.0 * n * (max_c + lam * np.log(Z))
    return f, grad_y
```

def adam_optimize(y0, n, fft_len_conv, fft_len_g, lam_factors, steps_per_factor=10):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for lam in lam_factors:
        for _ in range(steps_per_factor):
            f, grad = smooth_obj_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)  # can adjust
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return y
```

def subgradient_descent(a_start, n, fft_len_conv, fft_len_g, deadline, eta0=0.5, max_iters=200000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()
        # tolerance near maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0
        # compute gradient 2 * (a_rev * w conv)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n-1):(n-1+n)]
        grad = 2.0 * g_vec

        eta = eta0 / np.sqrt(it)
        # backtrack line search
        improved = False
        for _ in range(10):
            a_new = a - eta * grad
            # project onto simplex
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            # If not improved after backtrack, still move a little (optional)
            a = a - eta * grad
            a = project_onto_simplex(a)
    return best_a, best_score
```

def hill_climb(a_start, cur_score, fft_len, n, deadline, max_iter=5000):
    best = a_start.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.1  # up to 10%
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

--------
* Work on a probability vector (sum = 1); the evaluation is then
  `score = 2·n·max(conv)`.
* Multiple random starts (uniform, exponential, power‑law, Gaussian,
  semi‑circle, tent, edge‑heavy, etc.).
* Continuation Adam optimisation on a smooth log‑sum‑exp surrogate of the
  max‑convolution.  The smoothing factor λ is decreased geometrically.
* Projected sub‑gradient descent on the true max‑convolution objective,
  using the active (max‑tight) constraints to build the sub‑gradient.
* Short greedy hill‑climbs move a random fraction of mass between two
  coordinates and keep only improving moves.
* The whole budget (≈ 1000 s) is split into a phase of many restarts,
  a longer sub‑gradient refinement, and a final aggressive hill‑climb.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth surrogate of the max‑convolution (log‑sum‑exp)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the logits y.
    """
    a = softmax(y)

    # --- convolution a*a ---
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max (log‑sum‑exp) ---
    shifted = conv - max_c                 # max = 0
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient w.r.t. conv ---
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # --- back‑prop through convolution ---
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --- chain rule through soft‑max ---
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 10,
) -> np.ndarray:
    """
    Continuation Adam on the smooth surrogate.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            _, grad = smooth_obj_and_grad(
                y, n, fft_len_conv, fft_len_g, lam
            )
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    eta0: float = 0.5,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(a*a).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # indices whose value is almost maximal
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0                # give each active entry unit weight

        # gradient of (max) entry w.r.t. a: 2 * (a_rev * w)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        eta = eta0 / np.sqrt(it)

        # back‑tracking line search on the exact objective
        improved = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5

        if not improved:
            # even after back‑tracking no improvement – still step
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing on the exact objective
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j,
    accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_a(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a probability vector (sum = 1) for a given heuristic mode."""
    i_arr = np.arange(n, dtype=float)

    if mode == "random":
        a = rng.random(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * i_arr)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(i_arr, n - 1 - i_arr)
        a = np.exp(-rate * d)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a = (i_arr + 1) ** (-p)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a = np.full(n, 1.0 / n)
        a[:m_edge] = 3.0 / (2 * n)   # a bit more weight at the extremes
        a[-m_edge:] = 3.0 / (2 * n)

    elif mode == "arcsine":
        a = 1.0 / np.sqrt(i_arr + 1.0) + 1.0 / np.sqrt(n - i_arr)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.4):
                break
        a = np.zeros(n)
        if chosen:
            a[chosen] = 1.0 / len(chosen)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a = rng.gamma(alpha, 1.0)

    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(i_arr - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a = 1.0 + p * (dist ** 2)

    elif mode == "gaussian":
        mu = (n - 1) / 2.0
        sigma = n / (2.0 * rng.uniform(2.0, 8.0))
        a = np.exp(-0.5 * ((i_arr - mu) / sigma) ** 2)

    elif mode == "semicircle":
        x = (i_arr - (n - 1) / 2.0) / ((n - 1) / 2.0)
        a = np.sqrt(np.maximum(0.0, 1.0 - x ** 2))

    elif mode == "tent":
        a = 1.0 - np.abs(i_arr - (n - 1) / 2.0) / ((n - 1) / 2.0)

    elif mode == "uniform":
        a = np.full(n, 1.0)

    else:
        raise ValueError(f"unknown mode {mode}")

    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose problem size
    # --------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 10000)
        else:
            n = n_default
    else:
        n = n_default

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise the best candidate (probability vector)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Parameters for the continuation‑Adam stage
    # --------------------------------------------------------------
    lam_schedule = np.geomspace(5.0, 1e-9, num=120).tolist()
    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "ushape",
        "gaussian", "semicircle", "tent", "uniform", "best"
    ]

    # Allocate ~45 % of the budget for the Adam restarts
    adam_deadline = min(deadline, time.time() + budget_s * 0.45)
    iteration = 0

    while time.time() < adam_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            a0 = generate_initial_a(mode, n, rng)
            y0 = np.log(a0 + 1e-12)

        y_opt = adam_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            steps_per_factor=12,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[{iteration:04d}] mode={mode} improved → {best_score:.6f}")

        # short hill‑climb after each Adam run (≈ 0.4 s)
        hc_deadline = min(adam_deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Refine with projected sub‑gradient descent (≈ 30 % of budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline, time.time() + budget_s * 0.30)
    if time.time() < subgrad_deadline:
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,
            n,
            fft_len_conv,
            fft_len_g,
            subgrad_deadline,
            eta0=0.6,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final aggressive hill‑climb with remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[final hill] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
