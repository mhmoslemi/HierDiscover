"""
Hybrid search for a low‐evaluation step function.

Main ideas
-----------
* Work on the probability simplex (∑ a_i = 1) – the objective is
  scale invariant, so we can safely normalise the sequence.
* A dense parametric sweep over Beta(p,p) densities (including mixtures
  with the uniform distribution) discovers a very good starting shape.
* A continuation L‑BFGS optimiser (smooth log‑sum‑exp surrogate of the
  max‑convolution) refines the best candidates.
* Projected sub‑gradient descent on the true objective (max‑convolution)
  provides a fine‑grained polish.
* Finally a simulated‑annealing hill‑climber performs stochastic
  perturbations that can escape shallow local minima.
* All phases respect a global 1000 s budget; the best sequence found is
  returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a=1):
    2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed surrogate for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective 2·n·smooth_max(conv) and its gradient.
    y is an unconstrained vector; a = softmax(y) is the probability vector.
    lam = max(conv) * lam_factor  (smoothing constant).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # shift so max → 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t conv
    w = exp_shift / Z

    # gradient of conv w.r.t a (correlation of a and w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing (λ).
    Each stage starts from the result of the previous stage.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Analytic initialisers
# ----------------------------------------------------------------------


def beta_density(x: np.ndarray, p: float) -> np.ndarray:
    """Beta(p,p) density on (0,1) evaluated at points x (0 < x < 1)."""
    # use log to avoid overflow/underflow
    log_a = (p - 1.0) * (np.log(x) + np.log(1.0 - x))
    a = np.exp(log_a)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.size)
    return a / s


def semicircle_init(n: int) -> np.ndarray:
    """
    Discrete Wigner semicircle distribution on {0,…,n‑1}.
    a_i ∝ sqrt(1 - ((i - (n‑1)/2) / ((n‑1)/2))**2 )
    """
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    y = rng.standard_normal(n)
    a = np.exp(y)  # ensure positivity
    return a / a.sum()


def binary_sparse_init(n: int, rng: np.random.Generator, density: float = 0.20) -> np.ndarray:
    mask = rng.random(n) < density
    if not np.any(mask):
        mask[0] = True
    a = np.zeros(n)
    a[mask] = 1.0 / mask.sum()
    return a


def gaussian_bump_init(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = rng.integers(0, n)
    sigma = rng.uniform(0.05, 0.15) * n
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return a / a.sum()


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2, renormalised to sum 1."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: transfer a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """Hill‑climb that preferentially moves mass away from the current max‑conv contributors."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.002,
    decay: float = 0.995,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb.  Mass is moved between two
    randomly chosen coordinates; worsening moves may be accepted with a
    temperature that decays geometrically.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    temp = init_temp
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score
        accept = False
        if dE < 0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta
        temp = max(temp * decay, 1e-9)
    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  Implements a hybrid pipeline:
        1. Parameter sweep over Beta(p,p) (and mixtures with uniform)
        2. Continuation L‑BFGS refinement of the top few candidates
        3. Projected sub‑gradient descent
        4. Simulated‑annealing final polishing
    The function respects the global time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – adapt to any existing sequence
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] problem size n = {n}")

    # Precompute mid‑point coordinates for beta families
    x_mid = (np.arange(n, dtype=float) + 0.5) / n

    # ------------------------------------------------------------------
    # 1) Exhaustive grid search over Beta(p,p) and mixtures
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    best_desc = None

    # grid definition
    p_vals = np.linspace(0.45, 0.90, 46)   # fine grid
    w_vals = np.linspace(0.0, 1.0, 11)    # mixture weight with uniform

    # quick helper to evaluate and update the current best
    def consider(candidate: np.ndarray, desc: str):
        nonlocal best_a, best_score, best_desc
        sc = fast_score(candidate, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
            best_desc = desc
            print(f"[grid] new best {desc:30s} score = {sc:.6f}")

    # pure beta shapes
    for p in p_vals:
        a_beta = beta_density(x_mid, p)
        consider(a_beta, f"beta_{p:.3f}")

        # mixtures with uniform
        uniform = uniform_init(n)
        for w in w_vals:
            a_mix = mixture_init(a_beta, uniform, w)
            consider(a_mix, f"mix_{p:.3f}_{w:.2f}")

    # additional analytic families
    consider(semicircle_init(n), "semicircle")
    consider(uniform_init(n), "uniform")
    consider(random_gaussian_init(n, rng), "rand_gauss")
    consider(binary_sparse_init(n, rng, density=0.15), "binary")
    consider(gaussian_bump_init(n, rng), "bump")

    # historic sequence (if any)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a_prev = tmp / tmp.sum()
            consider(a_prev, "previous")

    print(f"[grid] best after sweep: {best_desc}  score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation L‑BFGS refinement (focus on the current best)
    # ------------------------------------------------------------------
    # If we have a handful of close runners we could also try a few others,
    # but the exhaustive sweep already gave us a strong candidate.
    y0 = np.log(best_a + 1e-12)

    # build a more gradual lam schedule
    lam_schedule = list(np.geomspace(5.0, 1e-10, num=35))
    maxiter_stage = 800

    # run L‑BFGS until the time budget for this phase expires
    lbg_deadline = start_time + 0.55 * budget_s
    print("[phase] L‑BFGS continuation started")
    y_opt = schedule_optimize(
        y0,
        n,
        fft_len_conv,
        fft_len_g,
        lam_schedule,
        maxiter_per_stage=maxiter_stage,
    )
    a_opt = softmax(y_opt)
    best_score = fast_score(a_opt, fft_len_conv, n)
    best_a = a_opt.copy()
    print(f"[l‑bfgs] refined best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent (majority of remaining budget)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.30,
        max_iters=600_000,
    )
    print(f"[subgrad] finished with score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing final polishing (tight remaining window)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline
        print("[anneal] simulated annealing start")
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            anneal_deadline,
            init_temp=0.002,
            decay=0.996,
            max_move=0.30,
        )
        print(f"[anneal] finished with score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final evaluation using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluate_sequence returned {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)
        final_val = None

    return final_seq
