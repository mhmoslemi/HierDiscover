def subgradient_max(a, fft_len):
    # compute conv via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)

"""
Propose a low‑evaluation step sequence.

Algorithm Overview
------------------
1. **Problem size** – we work with a fixed length `n = 4000`.  All
   candidates are normalized to sum = 1, because the objective
   `2·n·max(conv) / (∑a)²` is scale‑invariant.

2. **Initialisation** – start from a previously found construction
   (if the global variable `height_sequence_1` exists), otherwise from
   a uniform distribution.  A cheap sweep of power‑law families is run
   to obtain a decent baseline.

3. **Smooth optimisation (L‑BFGS)** – the non‑smooth “max‑convolution”
   is approximated by a log‑sum‑exp with a decreasing temperature
   (`λ`).  For each temperature we run a bounded‑memory L‑BFGS
   optimisation starting from a random restart or a jittered version
   of the current best.  The best sequence found so far is retained.

4. **Aggressive hill‑climbing** – we repeatedly compute a *sub‑gradient*
   of the true max‑convolution and transfer mass from the “most harmful”
   coordinates (largest sub‑gradient) to the “most beneficial” ones
   (smallest sub‑gradient).  The transfer amount is chosen by a tiny
   line‑search (full mass → half → quarter …) and the move is kept only
   if it strictly reduces the true objective.  When progress stalls a
   small Gaussian perturbation is added to escape local minima.

5. **Fine‑grained exponentiated‑gradient** – a short mirror‑descent
   phase refines the solution on the simplex.

6. **Result** – the best candidate is returned as a plain list of floats
   and evaluated with the supplied `evaluate_sequence` function.

All phases respect a global time budget (default 1000 s).  The code
records intermediate scores and prints a short progress trace.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1.  Returns the numerator `2·n·max(conv)`.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of the max‑convolution objective.
    The variables are the logits `y`; `a = softmax(y)` lives on the simplex.
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                # d smooth_max / d conv_i

    # back‑propagation through convolution (mask ← w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Sub‑gradient of the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_max(a: np.ndarray, fft_len_full: int) -> tuple[float, np.ndarray]:
    """
    Compute the true maximum of `a*a` and a sub‑gradient of the max.
    Returns (max_value, gradient_vector_of_length_n).
    """
    n = a.shape[0]
    # full convolution via FFT
    fa = np.fft.fft(a, fft_len_full)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)   # indicator of active shifts

    # sub‑gradient g_i = 2 * sum_{k} mask[k] * a_{k-i}
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_full)
    fa_rev = np.fft.fft(a_rev, fft_len_full)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

    return max_val, g

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each temperature in `lam_schedule`,
    starting each stage from the previous result.
    Returns (best_a, best_score) where best_score = fast_score(best_a).
    """
    y = np.log(a_start + 1e-12)               # logits on the simplex
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:       # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def aggressive_hill_climb(
    a0: np.ndarray,
    start_score: float,
    fft_len_full: int,
    n: int,
    deadline: float,
    max_iter: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves (possibly all) mass from the
    coordinate with the largest sub‑gradient to the one with the smallest.
    A tiny line‑search (full → 1/2 → 1/4 …) is performed; only strictly
    improving moves are kept.  When no improvement occurs for several
    iterations a small Gaussian jitter is applied to escape plateaus.
    """
    best = a0.copy()
    best_score = start_score
    it = 0
    no_improve = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iter:
        max_val, g = subgradient_max(best, fft_len_full)

        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max == i_min:
            break

        available = best[i_max]
        if available <= 0.0:
            # nothing to move – give up this pair
            break

        move_accepted = False
        # try a descending series of fractions of the available mass
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125]:
            delta = available * factor
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i_max] -= delta
            cand[i_min] += delta

            # ensure no negative entries due to rounding
            if cand[i_max] < -1e-12:
                continue

            cand_score = fast_score(cand, fft_len_full, n)
            if cand_score + 1e-12 < best_score:          # strict improvement
                best = cand
                best_score = cand_score
                move_accepted = True
                no_improve = 0
                break

        if not move_accepted:
            # try a second‑best destination (few lowest‑gradient indices)
            low_candidates = np.argsort(g)[:5]
            for i_low in low_candidates:
                if i_low == i_max:
                    continue
                delta = available * 0.5
                cand = best.copy()
                cand[i_max] -= delta
                cand[i_low] += delta
                cand_score = fast_score(cand, fft_len_full, n)
                if cand_score + 1e-12 < best_score:
                    best = cand
                    best_score = cand_score
                    move_accepted = True
                    no_improve = 0
                    break

        if not move_accepted:
            no_improve += 1

        # occasional jitter to escape stagnation
        if no_improve >= 400:
            sigma = rng.uniform(0.0005, 0.0020)
            best = add_noise(best, sigma=sigma, rng=rng)
            best_score = fast_score(best, fft_len_full, n)
            no_improve = 0

        it += 1

    return best, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_full: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the exact
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_full, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        _, g = subgradient_max(a, fft_len_full)

        eta = 0.30 / np.sqrt(it + 1.0)          # diminishing step size
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_score = fast_score(a, fft_len_full, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)          # for a*a
    fft_len_full = next_pow2(3 * n - 2)          # for mask‑a convolution

    # --------------------------------------------------------------
    # Initial candidate (use previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, hs)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)                # uniform start

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (power‑law families)
    # --------------------------------------------------------------
    for p in np.linspace(0.0, 3.0, 13):
        # increasing power‑law
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] power {p:.2f} -> {best_score:.6f}")

        # decreasing power‑law (reversed)
        a = (np.arange(n, 0, -1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] rev‑power {p:.2f} -> {best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # allocate roughly 55 % to L‑BFGS, 25 % to hill‑climb, 15 % to EG, 5 % spare
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = start_time + 0.95 * budget_s

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06,
    ]
    maxiter_per_lam = 800   # fewer iterations per λ to allow many restarts

    while time.time() < phase1_end:
        # Occasionally jitter around the current best, otherwise fresh start
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.012), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_full,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – aggressive hill‑climbing with line‑search
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = aggressive_hill_climb(
            best_a,
            best_score,
            fft_len_full,
            n,
            phase2_end,
            max_iter=25000,
        )
        print(f"[hill‑climb] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (fine refinement)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            fft_len_full,
            n,
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
