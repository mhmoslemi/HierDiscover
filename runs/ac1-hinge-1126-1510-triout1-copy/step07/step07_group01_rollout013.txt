def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimizes
    ``2·n·max(a∗a) / (∑a)²``.
    The algorithm combines many cheap random restarts of exponentiated‑gradient
    descent, a final L‑BFGS refinement and a short greedy hill‑climb.
    """

    import time, numpy as np
    from scipy.optimize import minimize

    # set rng
    rng = np.random.default_rng(seed)

    # problem size
    n = 4000   # baseline
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # initialise best from the previous global if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        # interpolate to current n
        if prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # other deterministic seeds
    candidates = [a0, spectral_init(n)]

    # evaluate initial candidates
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[initial] best_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # allocate 70% of time to random restarts + cheap refinement
    restart_deadline = start_time + 0.70 * budget_s

    # parameters for each restart
    min_run_time = 0.4   # seconds per restart (incl. gradient + hill)

    while time.time() < restart_deadline:

        mode = rng.choice(['uniform','random','exp','power','edge','bimodal','beta','cosine'])
        if mode == 'beta':
            # draw random alpha in (0.2, 2.0)
            alpha = rng.uniform(0.2, 2.0)
            # compute beta(alpha)=alpha (symmetric)
            x = (np.arange(n) + 0.5) / n
            a_start = np.power(x, alpha-1) * np.power(1-x, alpha-1)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        elif mode == 'cosine':
            a_start = np.sin(np.pi * (np.arange(n) + 0.5) / n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = random_initial(n, rng)

        run_deadline = min(time.time() + min_run_time, restart_deadline)
        a_tmp, score_tmp = exp_gradient_optimize(a_start, run_deadline,
                                                n, fft_len_conv, fft_len_g, rng)

        # quick hill‑climb to tidy up
        # give it a few thousand steps (speed‑controlled by time)
        hc_deadline = min(time.time() + 0.1, restart_deadline)
        a_tmp, score_tmp = hill_climb(a_tmp, score_tmp,
                                      fft_len_conv, n, hc_deadline,
                                      max_iter=3000)

        if score_tmp < best_score:
            best_score = score_tmp
            best_a = a_tmp.copy()

"""
Fast random‑restart optimisation for the “max‑convolution” objective.

We search a non‑negative vector a of length n (default 4000) that
minimises

    2·n·max(a∗a) / (∑a)² .

The optimisation consists of

* many cheap restarts of exponentiated‑gradient (mirror descent) from
  diverse initialisations (uniform, Beta‑shapes, cosine, random Dirichlet, …);
* a short “gradient‑informed” hill‑climb that moves probability mass from the
  indices that contribute most to the current maximal convolution entry;
* a final L‑BFGS refinement on a log‑sum‑exp smooth approximation of the
  max‑convolution;
* a last greedy hill‑climb to clean up any remaining slack.

All steps respect the simplex constraint ‖a‖₁=1 and a_i≥0.
The total wall‑time is limited to ``budget_s`` seconds (default 1000 s)."""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (returns length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a on the simplex (∑a=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (several families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta', 'cosine'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'beta':
        alpha = rng.uniform(0.2, 2.0)
        x = (np.arange(n) + 0.5) / n
        a = np.power(x, alpha - 1) * np.power(1 - x, alpha - 1)
    else:  # cosine
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def spectral_init(n: int) -> np.ndarray:
    """Approximate the solution of a*a = uniform via spectral square‑root."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (exact max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule (larger at the beginning)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Gradient‑informed hill‑climb
# ----------------------------------------------------------------------


def grad_hill_climb(a_start: np.ndarray,
                    start_score: float,
                    fft_len_conv: int,
                    fft_len_g: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer guided by the sub‑gradient of the max‑convolution.
    Moves a small fraction of probability from the index with the largest
    gradient to the index with the smallest gradient, accepting only
    improving moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score

    while time.time() < deadline:
        # compute sub‑gradient (same as in exponentiated‑gradient)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        i = int(np.argmax(g))          # most responsible index
        j = int(np.argmin(g))          # least responsible index
        if a[i] <= 1e-15:
            # nothing to move – pick a random pair as a fallback
            i, j = rng.integers(0, n, size=2)
            if i == j or a[i] <= 1e-15:
                continue

        # move a modest fraction of mass
        delta = a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = cand.copy()
            a = cand  # continue from the improved point
        # otherwise discard the move (keep current a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and L‑BFGS wrapper
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on a decreasing sequence of smoothing parameters.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # occasional numerical trouble – just skip this λ.
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[final‑exp‑grad] score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final gradient‑guided hill‑climb (exhaust remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = grad_hill_climb(best_a,
                                             best_score,
                                             fft_len_conv,
                                             fft_len_g,
                                             n,
                                             deadline,
                                             rng)
        print(f"[final‑grad‑hill] score {best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
