"""
Multi‑phase optimizer for the step‑function evaluation problem.

The algorithm proceeds through:
  1. A cheap power‑law sweep to obtain a decent starter.
  2. Random‑restart L‑BFGS on a log‑sum‑exp smooth approximation of the
     max‑convolution, with a decreasing smoothing factor.
  3. Adam (first‑moment/second‑moment) descent on the same smooth objective.
  4. Projected sub‑gradient descent with momentum on the true max‑convolution.
  5. Exponentiated‑gradient descent on the simplex.
  6. Deterministic gradient‑guided hill‑climb.
  7. A final stochastic pairwise‑swap local search.

All phases respect a global wall‑clock budget (default 1000 s).  The best
sequence found is returned and finally evaluated with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1.  O(n log n) using FFT.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 12000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a small fraction of mass from the
    coordinate with the largest sub‑gradient to the one with the smallest.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        delta = best[i_hi] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – we keep n=4000 (fast enough for many iterations)
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initial candidate (use previous best if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0.0, 1.0, hs.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, hs)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (power‑law families)
    # --------------------------------------------------------------
    for p in np.linspace(0.0, 3.2, 17):
        a = (np.arange(1, n + 1, dtype=float) ** p)
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] power {p:.2f} -> {best_score:.6f}")

        a_rev = (np.arange(n, 0, -1, dtype=float) ** p)
        a_rev /= a_rev.sum()
        sc = fast_score(a_rev, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_rev.copy()
            print(f"[sweep] rev‑power {p:.2f} -> {best_score:.6f}")

    # --------------------------------------------------------------
    # Time‑budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # tiny safety gap

    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.75 * budget_s   # Adam
    phase3_end = start_time + 0.90 * budget_s   # sub‑gradient
    phase4_end = start_time + 0.97 * budget_s   # exp‑gradient
    # remaining time → hill‑climb + stochastic swaps

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
    ]
    maxiter_per_lam = 1300

    while time.time() < phase1_end:
        if rng.random() < 0.2:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective
    # --------------------------------------------------------------
    y = np.log(best_a + 1e-12)
    m_adam = np.zeros_like(y)
    v_adam = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.07
    it_adam = 0

    while time.time() < phase2_end:
        # linearly decay the smoothing factor
        elapsed = time.time() - start_time
        total = phase2_end - start_time
        lam_factor = max(0.01, 0.30 * (1.0 - elapsed / total))

        _, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        it_adam += 1
        m_adam = beta1 * m_adam + (1 - beta1) * grad_y
        v_adam = beta2 * v_adam + (1 - beta2) * (grad_y ** 2)
        m_hat = m_adam / (1 - beta1 ** it_adam)
        v_hat = v_adam / (1 - beta2 ** it_adam)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase4_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – deterministic gradient‑guided hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = gradient_guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[hill‑climb] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – stochastic pairwise‑swap local search
    # --------------------------------------------------------------
    a_local = best_a.copy()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or a_local[i] <= 0.0:
            continue
        # move up to 10 % of the mass at i towards j
        delta = a_local[i] * rng.random() * 0.10
        a_local[i] -= delta
        a_local[j] += delta

        cur_score = fast_score(a_local, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_local.copy()
            print(f"[stochastic swap] new best fast_score={best_score:.6f}")
        else:
            # revert the move
            a_local[i] += delta
            a_local[j] -= delta

    # --------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
