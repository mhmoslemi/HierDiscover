"""
Propose a step‑function with a low evaluate_sequence score.

The algorithm builds on the previous L‑BFGS + sub‑gradient approach but
adds a few improvements:

* If a previous construction is available (height_sequence_1) it is
  up‑sampled to a larger problem size (≈ 2×, up to 12000 points) – a
  finer grid reduces the discretisation error and can lower the limit.
* The continuation L‑BFGS schedule uses many more iterations per
  smoothing stage and a finer λ‑schedule.
* A longer projected sub‑gradient phase with a larger step‑size is used.
* After the final refinement a more aggressive hill‑climb is performed.
* All steps respect the total wall‑clock budget (default 1000 s).

The code is completely self‑contained, uses only NumPy/SciPy and prints
progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) for the smoothed objective

        f = 2·n·smooth_max(a*a) ,   a = softmax(y)

    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max using log‑sum‑exp
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z                           # ∂smooth_max/∂conv_i

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)               # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        proportion = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * proportion))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Hill‑climbing helper
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12  # up to 12 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient on the true max‑convolution objective
# ----------------------------------------------------------------------


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.35,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective
    2·n·max(a*a).  Uses an averaged sub‑gradient over all (almost) active
    convolution entries and a simple back‑tracking line‑search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set: entries within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of the weighted max w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]

        grad = 2.0 * g_vec                     # derivative of a_i a_j sum

        # step size ~ 1/sqrt(it)
        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(8):                     # back‑track up to 8 halvings
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Upsampling helper (when a previous construction exists)
# ----------------------------------------------------------------------


def upsample_to_n(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly up‑sample a probability vector `old_seq` (sum = 1) to length `new_n`.
    The result is re‑normalised to sum = 1.
    """
    old_seq = np.asarray(old_seq, dtype=float)
    old_seq = np.maximum(old_seq, 0.0)
    s = old_seq.sum()
    if s == 0.0:
        old_seq = np.full_like(old_seq, 1.0 / old_seq.size)
        s = 1.0
    old_seq = old_seq / s

    m = old_seq.size
    # map both grids onto [0,1)
    x_old = np.linspace(0.0, 1.0, m, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    new_seq = np.interp(x_new, x_old, old_seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = new_seq / new_seq.sum()
    return new_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluate_sequence score.  Returns the best list found within the time
    budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose problem size, possibly up‑sampling a previous solution.
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        prev = np.asarray(init_seq, dtype=float)
        if prev.ndim == 1 and prev.size >= 4000:
            # aim for twice the size but never exceed 12000
            target_n = min(12000, max(8000, prev.size * 2))
            n = int(target_n)
            best_a = upsample_to_n(prev, n)
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Parameters for the continuation stages
    # --------------------------------------------------------------
    lam_schedule_main = list(np.geomspace(0.5, 1e-8, num=50))
    lam_schedule_refine = list(np.geomspace(0.5, 1e-14, num=80))

    modes = [
        "random", "exp", "exp_sym", "power", "edge",
        "arcsine", "sidon", "dirichlet", "best"
    ]

    # --------------------------------------------------------------
    # Phase 1 – many restarts with moderate L‑BFGS work.
    # --------------------------------------------------------------
    while time.time() < deadline - 250.0:   # reserve time for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_main,
            maxiter_per_stage=800,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart] mode={mode} → improved fast score = {best_score:.8f}")

        # short hill‑climb (≈ 1 s) to polish the result
        hc_deadline = min(deadline, time.time() + 1.0)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2500
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the true objective.
    # --------------------------------------------------------------
    if time.time() < deadline - 130.0:
        sub_deadline = deadline - 70.0
        print("[subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            sub_deadline,
            eta0=0.40,
            max_iters=200_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – final high‑precision L‑BFGS with an ultra‑fine λ‑schedule.
    # --------------------------------------------------------------
    if time.time() < deadline - 70.0:
        print("[refine L‑BFGS] start")
        y_opt = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_refine,
            maxiter_per_stage=1200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[refine L‑BFGS] improved fast score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 4 – aggressive final hill‑climb using any remaining time.
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30000,
        )
        print(f"[final HC] finished → fast score = {best_score:.8f}")

    # --------------------------------------------------------------
    # Official evaluation and return.
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
