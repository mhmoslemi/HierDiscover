def smooth_obj_and_grad(a, n, fft_len_conv, fft_len_g, lam_factor):
    conv = conv_fft(a, fft_len_conv, n)
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    # gradient wrt conv
    w = exp_shift / Z

    # compute gradient wrt a: derivative of sum_i w_i * conv_i = (w * a_rev) convolution? Actually conv = a * a, gradient is 2 * (rev_a * w) (as earlier)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    grad_a = 2.0 * conv_rev[(n - 1) : (n - 1 + n)]

    # final objective
    obj = 2.0 * n * smooth_max
    grad = 2.0 * n * grad_a  # because objective scaled by 2n from smooth_max
    return obj, grad

def projected_adam_a(a_start, n, fft_len_conv, fft_len_g, lam_factors, steps_per_factor):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for lam_factor in lam_factors:
        for _ in range(steps_per_factor):
            obj, grad = smooth_obj_and_grad_a(a, n, fft_len_conv, fft_len_g, lam_factor)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.5 / np.sqrt(t)  # adapt
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_onto_simplex(a)
    return a

    best_score = score

        # propose move: pick i,j random and delta = uniform*0.1* a[i]

            best_score = cand_score

                # track improvement

def projected_adam(a_start, n, fft_len_conv, fft_len_g, lam_factors, steps_per_factor):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    for lam_factor in lam_factors:
        for _ in range(steps_per_factor):
            t += 1
            obj, grad = smooth_obj_and_grad_a(a, n, fft_len_conv, fft_len_g, lam_factor)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.3 / np.sqrt(t)  # smaller
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_onto_simplex(a)
    return a

"""
Propose a non‑negative step‑function (sequence) with a low evaluate_sequence score.

Algorithm Overview
------------------
* Choose a problem size n (default 8000).  If a previously found sequence is
  available we reuse its length (capped at 10000) as a warm‑start.
* Several heuristic initialisations are generated (random, exponential,
  power‑law, edge‑heavy, arcsine, Sidon‑set, Dirichlet, U‑shape, uniform).
* For each start we run a **projected Adam** optimiser on a smooth
  log‑sum‑exp surrogate of the max‑convolution.  The smoothing factor λ is
  gradually reduced (continuation) so the surrogate approaches the true
  max.  The Adam iterates are projected onto the probability simplex after
  each update.
* After the Adam phase we refine the best candidate with a **multi‑active
  sub‑gradient descent** on the exact max‑convolution objective.
* A short **simulated‑annealing** phase performs random mass transfers
  between coordinates, accepting worsening moves with a temperature that
  decays linearly to zero.
* Finally an aggressive **hill‑climb** moves a random fraction of mass from
  one coordinate to another, accepting only improving moves.
* The resulting sequence is returned; the harness evaluates the exact score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Heuristic initialisers (return logits for softmax)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "ushape":
        mid = (n - 1) / 2.0
        dist = np.abs(np.arange(n) - mid) / mid
        p = rng.uniform(0.5, 4.0)
        a0 = 1.0 + p * (dist ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp surrogate) and its gradient w.r.t. a
# ----------------------------------------------------------------------


def smooth_obj_and_grad_a(
    a: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (obj, grad) for
        obj = 2·n·smooth_max(a*a)
    where smooth_max is a log‑sum‑exp approximation of max.
    grad is the gradient w.r.t. the probability vector a.
    """
    # convolution a*a
    conv = conv_fft(a, fft_len_conv, n)
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                     # max becomes 0
    exp_shift = np.exp(shifted / lam)          # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    obj = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # derivative of Σ w_i·conv_i w.r.t. a via convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    grad_conv = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient of the *full* objective (incl. the factor 2·n)
    grad = 4.0 * n * grad_conv
    return obj, grad


# ----------------------------------------------------------------------
# Projected Adam on the simplex
# ----------------------------------------------------------------------


def projected_adam(
    a_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    steps_per_factor: int = 30,
) -> np.ndarray:
    """
    Adam optimisation of the smoothed objective with projection onto the
    probability simplex after every update.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_factors:
        for _ in range(steps_per_factor):
            t += 1
            _, grad = smooth_obj_and_grad_a(a, n, fft_len_conv, fft_len_g, lam)

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = 0.3 / np.sqrt(t)          # decaying learning rate
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = project_onto_simplex(a)

    return a


# ----------------------------------------------------------------------
# Greedy hill‑climb (only accept improvements)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only moves that strictly improve the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Multi‑active projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    The sub‑gradient is an average over all (almost) active entries.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set: entries whose value is within a tiny tolerance of the maximum
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # gradient of weighted max w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec                     # derivative of Σ w_i·conv_i

        eta = eta0 / np.sqrt(it)
        improved = False
        for _ in range(8):                     # back‑tracking line‑search
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing on the fast objective
# ----------------------------------------------------------------------


def anneal_search(
    a_start: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Very simple annealing: move a random fraction of mass between two
    coordinates, accept worsening moves with a temperature that decays
    linearly to zero.
    """
    best = a_start.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start_time = time.time()
    total = deadline - start_time
    temperature_scale = best_score * 0.02  # initial temperature proportional to score

    while time.time() < deadline:
        # linear cooling
        fraction = max(0.0, (deadline - time.time()) / total)
        T = temperature_scale * fraction

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 15 % of the source mass
        delta = best[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(cand_score - best_score) / (T + 1e-12)):
                best = cand
                best_score = cand_score

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point for the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ------------------------------------------------------------------
    n_default = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
            base = np.maximum(tmp, 0.0)[:n]
            base = base / base.sum()
        else:
            n = n_default
            base = np.full(n, 1.0 / n)
    else:
        n = n_default
        base = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate.
    # ------------------------------------------------------------------
    best_a = base.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Continuation Adam phase.
    # ------------------------------------------------------------------
    lam_factors = list(np.geomspace(5.0, 1e-9, num=120))
    steps_per_factor = 30

    total_time = budget_s
    t0 = time.time()
    adam_deadline = t0 + total_time * 0.45
    iteration = 0
    while time.time() < adam_deadline:
        mode = rng.choice(
            [
                "random",
                "exp",
                "exp_sym",
                "power",
                "edge",
                "arcsine",
                "sidon",
                "dirichlet",
                "ushape",
                "uniform",
            ]
        )
        if mode == "random":
            a0 = rng.random(n)
        else:
            y0 = generate_initial_y(mode, n, rng)
            a0 = softmax(y0)

        # occasional mixing with the current best to stay in its basin
        if rng.random() < 0.15:
            mix = rng.random()
            a0 = mix * best_a + (1.0 - mix) * a0

        a0 = a0 / a0.sum()

        a_opt = projected_adam(
            a0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_factors,
            steps_per_factor,
        )
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Adam {iteration:04d}] mode={mode} → fast score = {best_score:.6f}")

        # short hill‑climb after each Adam run
        hc_deadline = min(adam_deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        iteration += 1

    # ------------------------------------------------------------------
    # Multi‑active sub‑gradient refinement.
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline, t0 + total_time * 0.80)
    if time.time() < subgrad_deadline:
        print("[Subgrad] start")
        best_a, best_score = multi_active_subgrad(
            best_a,
            fft_len_conv,
            fft_len_g,
            n,
            subgrad_deadline,
            eta0=0.5,
            max_iters=500_000,
        )
        print(f"[Subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated annealing phase.
    # ------------------------------------------------------------------
    anneal_deadline = min(deadline, t0 + total_time * 0.95)
    if time.time() < anneal_deadline:
        print("[Anneal] start")
        best_a, best_score = anneal_search(
            best_a, best_score, n, fft_len_conv, anneal_deadline
        )
        print(f"[Anneal] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=25000,
        )
        print(f"[Final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
