"""
Optimiser for the “step‑function” challenge.

The search proceeds in four timed phases:

1. **Heavy L‑BFGS** – many random restarts on a log‑sum‑exp smooth
   approximation of the max‑convolution.  The smoothing factor is
   driven down to 5·10⁻⁶.
2. **Adam continuation** – a smooth‑objective Adam optimiser that
   continues the annealing schedule with finer λ values, giving a
   more robust fine‑tuning than plain L‑BFGS.
3. **Exact exponentiated‑gradient descent with line‑search** – uses the
   true sub‑gradient of the hard max and chooses a step size from a
   small candidate set, always keeping the point on the probability
   simplex.
4. **Multi‑scale greedy hill‑climb** – repeatedly moves a random
   fraction of the mass between two coordinates, with the fraction
   slowly decreasing; only improving moves are kept.

All phases respect a global 1000 s deadline and keep the best
candidate seen.  The returned list is a non‑negative sequence that is
optimised for the official `evaluate_sequence` scorer.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the scoring term 2·n·max(a*a) for a probability vector.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation & noise utilities
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise to a probability vector and re‑project.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) used by L‑BFGS/Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    lam_factor controls the smoothing: λ = lam_factor * max(conv).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient components that correspond to a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation (heavy phase)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best probability vector,
    its fast_score, and the number of L‑BFGS calls performed.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp smooth objective.
    Returns the best probability vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    pow1 = 1.0
    pow2 = 1.0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            pow1 *= beta1
            pow2 *= beta2

            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - pow1)
            v_hat = v / (1.0 - pow2)

            # step size – a modest constant works well for the smooth problem
            step = 0.15
            y = y - step * m_hat / (np.sqrt(v_hat) + eps)

            if t % 30 == 0:
                a = softmax(y)
                cur_score = fast_score(a, fft_len_conv, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent with a small line‑search
# ----------------------------------------------------------------------


def exp_gradient_line_search(a_start: np.ndarray,
                             deadline: float,
                             n: int,
                             fft_len_conv: int,
                             fft_len_g: int,
                             rng: np.random.Generator):
    """
    Exponentiated‑gradient descent using the true sub‑gradient of the hard max.
    At each iteration a small line‑search over a few step‑size candidates is
    performed; only improving updates are kept.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # candidate step‑sizes (geometric progression)
    eta_candidates = [0.02, 0.05, 0.1, 0.2, 0.4, 0.8, 1.6]

    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # strict mask – only the exact maximal entries
        mask = (conv >= max_val - 1e-12).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        improved = False
        for eta in eta_candidates:
            exp_term = np.exp(np.clip(-eta * g, -50, 50))
            a_new = a * exp_term
            s = a_new.sum()
            if s <= 0.0:
                continue
            a_new /= s
            score = fast_score(a_new, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break  # accept first improving step
        if not improved:
            # no candidate improved – perturb a little to escape flat regions
            a = add_noise(a, sigma=0.001, rng=rng)
    return best_a, best_score


# ----------------------------------------------------------------------
# Multi‑scale greedy hill‑climb (polishing)
# ----------------------------------------------------------------------


def multi_hill_climb(a_start: np.ndarray,
                     current_score: float,
                     fft_len: int,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iter: int = 50000):
    """
    Greedy hill‑climb with three move scales.
    Each move transfers a random fraction of mass from i to j.
    Only strictly improving moves are kept.
    """
    best = a_start.copy()
    best_score = current_score

    # probabilities for the size of the move
    scale_probs = np.array([0.6, 0.3, 0.1])  # small / medium / large
    scales = np.array([0.05, 0.2, 0.6])     # max fraction of source mass

    it = 0
    while time.time() < deadline and it < max_iter:
        # pick move scale
        scale = rng.choice(scales, p=scale_probs)

        i = rng.integers(0, n)
        if best[i] <= 0.0:
            it += 1
            continue
        delta = best[i] * rng.random() * scale
        if delta <= 0.0:
            it += 1
            continue
        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        # occasional random re‑perturbation to avoid being stuck
        if it % 4000 == 0:
            best = add_noise(best, sigma=0.0005, rng=rng)
            best /= best.sum()
            tmp_score = fast_score(best, fft_len, n)
            if tmp_score < best_score:
                best_score = tmp_score

        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the global deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    n = 4000                                 # number of coefficients
    fft_len_conv = next_pow2(2 * n - 1)       # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for gradient‑related convolutions

    # start from a stored construction if one is available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget division
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # Phase boundaries (empirical)
    phase1_end = start_time + 0.45 * budget_s       # L‑BFGS heavy stage
    phase2_end = start_time + 0.78 * budget_s       # Adam fine‑tuning
    phase3_end = start_time + 0.96 * budget_s       # exact EG descent

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule_lbfgs = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002, 0.00001, 0.000005
    ]
    maxiter_per_lam = 1200  # a bit less than before to fit more restarts

    while time.time() < phase1_end:
        # Occasionally perturb the current best, otherwise fresh random start
        if best_score < np.inf and rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule_lbfgs,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam continuation on the smooth objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule_adam = [
            0.3, 0.2, 0.12, 0.07, 0.04,
            0.025, 0.015, 0.009, 0.005, 0.003,
            0.0018, 0.001, 0.0006, 0.0003,
            0.00015, 0.00008, 0.00004, 0.00002
        ]
        iters_per_lam = 120   # enough to see progress without overrunning time

        a_adam, score_adam = adam_optimize(
            best_a,
            lam_schedule_adam,
            iters_per_lam,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_adam < best_score:
            best_score = score_adam
            best_a = a_adam.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact exponentiated‑gradient descent with line search
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        a_eg, score_eg = exp_gradient_line_search(
            best_a,
            phase3_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        if score_eg < best_score:
            best_score = score_eg
            best_a = a_eg.copy()
            print(f"[Exp‑Grad] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – intensive multi‑scale hill‑climb polishing
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = multi_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        rng,
        max_iter=40000,
    )
    print(f"[Hill‑Climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
