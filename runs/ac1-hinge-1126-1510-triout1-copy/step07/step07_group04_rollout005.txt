def objective_and_grad(y, n, fft_len, lam):
    a = softmax(y)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
    max_c = conv.max()
    # compute smooth max via log-sum-exp with lam
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t conv entries

        idx = np.arange(n)
        counterpart = m - idx

def hill_climb_simulated(a, cur_score, fft_len, n, deadline):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T = 0.02

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        move = best[i] * rng.random() * 0.2
        a2 = best.copy()
        a2[i] -= move
        a2[j] += move
        new_score = fast_score(a2, fft_len, n)
        diff = new_score - best_score
        if diff < 0 or rng.random() < np.exp(-diff/T):
            best = a2
            best_score = new_score
        T *= 0.9999  # anneal
    return best, best_score

"""
Hybrid optimizer for the discrete step‑function minimisation problem.

The routine works on the probability simplex (∑a = 1) – the objective
2·n·max(conv)/(∑a)² is scale‑invariant – and seeks the distribution that
flattens the self‑convolution.  The search consists of four phases:

1.  A deep continuation L‑BFGS run on a log‑sum‑exp smooth approximation
    of the max‑convolution, starting from a high‑quality “arcsine”
    distribution and a few additional random seeds.

2.  Projected multiplicative sub‑gradient descent on the true (non‑smooth)
    objective; the update a_i ← a_i·exp(−η·g_i) is performed on the
    simplex and quickly eliminates remaining spikes.

3.  An aggressive simulated‑annealing hill‑climb that transfers a random
    fraction of mass between two coordinates, accepting worsening moves
    with a slowly‑cooling temperature to escape shallow local minima.

4.  Final evaluation with the provided ``evaluate_sequence`` routine.

The algorithm respects the 1000‑second wall‑time limit, prints intermediate
scores and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained logits y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # linear convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                      # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂c/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "ftol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "harmonic":
        a0 = 1.0 / (np.arange(1, n + 1, dtype=float))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        # Discrete arcsine law a_i ∝ 1/√(x·(1‑x))
        idx = np.arange(n, dtype=float) + 0.5
        x = idx / n
        eps = 1e-15
        a0 = 1.0 / np.sqrt(x * (1.0 - x) + eps)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Projected multiplicative sub‑gradient descent (true objective)
# ----------------------------------------------------------------------


def multiplicative_subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.3,
    max_iters: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative sub‑gradient descent on 2·n·max(conv).
    Update rule: a ← a·exp(−η·g) then project onto simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step‑size
        eta = eta0 / np.sqrt(it)

        # multiplicative update + simplex projection
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0:
            a = best_a.copy()
            continue
        a = a / s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def hill_climb_simulated(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.02,
    decay: float = 0.9999,
    max_move: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb with simulated‑annealing acceptance.
    """
    best = a.copy()
    best_score = cur_score
    T = init_temp
    rng = np.random.default_rng()
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)
        diff = new_score - best_score
        if diff < 0.0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            best = cand
            best_score = new_score
        T *= decay
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine follows the four‑phase scheme described
    above and returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    default_n = 15000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            default_n = min(int(tmp.size), 20000)   # respect an upper bound
    n = default_n

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 1 – deep continuation L‑BFGS (≈45 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.7, 1e-10, num=45))

    # start from a high‑quality arcsine distribution
    y0 = generate_initial_y("arcsine", n, rng)
    best_y = schedule_optimize(
        y0,
        n,
        fft_len_conv,
        fft_len_g,
        lam_schedule,
        maxiter_per_stage=800,
    )
    best_a = softmax(best_y)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[phase1] after deep L‑BFGS  fast score = {best_score:.6f}")

    # add a few cheap random restarts while time permits
    phase1_deadline = start_time + 0.45 * budget_s
    restart_modes = ["random", "exp", "power", "edge", "binary", "bump"]
    while time.time() < phase1_deadline:
        mode = rng.choice(restart_modes)
        y_start = generate_initial_y(mode, n, rng)
        y_opt = schedule_optimize(
            y_start,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,   # quicker restarts
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[restart] mode={mode:<7}  improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected multiplicative sub‑gradient descent (≈30 %)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 30.0, start_time + 0.80 * budget_s)
    print("[phase2] multiplicative sub‑gradient descent")
    best_a, best_score = multiplicative_subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.35,
        max_iters=1_200_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[phase2] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive simulated‑annealing hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase3] simulated‑annealing hill climb")
        best_a, best_score = hill_climb_simulated(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_temp=0.02,
            decay=0.99995,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[phase3] final fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
