def beta_general_init(n, p, q):
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p-1.0) * np.power(1 - x, q-1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / s
    return a
```

def gaussian_shape(n, sigma):
    centre = (n-1)/2.0
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a

def laplace_shape(n, scale):
    centre = (n-1)/2.0
    idx = np.arange(n, dtype=float)
    a = np.exp(- np.abs(idx - centre) / scale )
    a = a / a.sum()
    return a

"""
Improved search for a low‑evaluation step‑function.

The algorithm performs:
1.   A fast deterministic sweep over several analytic families
     (Beta(p,q), Gaussian, Laplace, mixtures) on the probability simplex.
2.   The best candidate is refined by a continuation L‑BFGS on a
     smooth “log‑sum‑exp” surrogate of the max‑convolution.
3.   A long projected sub‑gradient descent on the true non‑smooth
     objective further lowers the peak.
4.   Finally, aggressive small‑step hill‑climbs (random and
     targeted) polish the result until the time budget expires.

All steps respect a global 1000 s deadline and operate on the
probability simplex (∑a_i = 1) so that the evaluation function reduces to
`2·n·max(conv(a,a))`.  The code re‑uses the previous construction if
available and works with any sequence length up to about 12 k.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is length n, fft_len ≥ 2*n‑1.
    Returns the real part of the convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the evaluation function for a probability vector
    (∑a_i = 1): returns 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families of probability vectors
# ----------------------------------------------------------------------


def beta_general_init(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete Beta(p,q) density on (0,1); p,q>0.
    Returns a normalized probability vector of length n.
    """
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """
    Discrete Gaussian centred at the middle of the array.
    sigma is the standard deviation (in index units).
    """
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a = a / a.sum()
    return a


def laplace_shape(n: int, scale: float) -> np.ndarray:
    """
    Discrete Laplace (double‑exponential) centred at the middle.
    scale > 0 controls the decay rate.
    """
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    a = np.exp(-np.abs(idx - centre) / scale)
    a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n)


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """
    Convex mixture w·a1 + (1‑w)·a2, re‑normalised to sum 1.
    """
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    The smoothing constant λ = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                      # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv
    w_vec = exp_shift / Z

    # correlation of a and w (gradient of conv w.r.t. a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w_vec, fft_len_g)
    corr = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to ∂conv/∂a_i (factor 2 for symmetry)
    g_vec = 2.0 * corr[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Local optimisation helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """Random greedy hill‑climb: transfer a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb that preferentially moves mass away from the current
    max‑conv contributors.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        idx = np.arange(n)
        jdx = m - idx
        mask = (jdx >= 0) & (jdx < n)
        contributors = idx[mask]

        if contributors.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = rng.choice(contributors)
            j = rng.integers(0, n)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true (non‑smooth) objective.
    The sub‑gradient at the current maximal convolution entry m is
    g_i = 2·a_{m‑i} for valid i, otherwise 0.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

        lam_schedule = list(np.geomspace(5.0, 1e-9, num=35))
        maxiter_per_stage = 800
        y0 = np.log(best_a + 1e-12)
        y_opt = schedule_optimize(
            y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage
        )
        a_opt = softmax(y_opt)
        # enforce symmetry (the optimum is symmetric)
        a_opt = (a_opt + a_opt[::-1]) / 2.0
        a_opt = a_opt / a_opt.sum()
        best_a = a_opt
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[l‑bfgs] refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (long phase)
    # ------------------------------------------------------------------
    subgrad_deadline = min(deadline - 15.0, start_time + 0.90 * budget_s)
    print("[subgrad] projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a, fft_len_conv, n, subgrad_deadline, eta0=0.30, max_iters=1_500_000
    )
    print(f"[subgrad] finished, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive final polishing (small‑step hill‑climbs)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] random hill‑climb (tiny steps)")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=20000,
            max_move=0.05,   # very small moves for fine tuning
        )
        print(f"[final] after random hill‑climb score = {best_score:.6f}")

    if time.time() < deadline:
        print("[final] targeted hill‑climb")
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=10000,
            max_move=0.03,
        )
        print(f"[final] after targeted hill‑climb score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore – supplied by the platform
        print(f"[final] evaluate_sequence = {final_val:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation failed:", e)
        final_val = None

    return final_seq
