def projected_subgradient(a, fft_len_conv, fft_len_g, n, deadline):
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    iteration = 0
    eta = 0.5   # initial step
    while time.time() < deadline and iteration < max_iters:
        conv = conv_fft(best_a, fft_len_conv, n)
        max_val = conv.max()
        # identify active
        tol = 1e-12 # maybe smaller relative tolerance
        active = np.where(conv >= max_val - tol)[0]
        if len(active)==0:
            active = np.array([np.argmax(conv)])
        # subgradient g = 2 * avg_i conv_rev...
        w = np.zeros_like(conv)
        w[active] = 1.0/len(active)
        # compute gradient = 2*a_rev * w conv etc
        a_rev = best_a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        grad = 2.0 * conv_rev[(n-1):(n-1+n)]
        # line search: try decreasing step sizes until improvement
        improved = False
        for ls in range(10):
            step = eta / (1.0 + iteration*0.0001)   # maybe scaled per iteration
            new_a = best_a - step * grad
            # project onto simplex
            new_a = project_onto_simplex(new_a)
            new_score = fast_score(new_a, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = new_a
                improved = True
                break
            else:
                eta *= 0.5
        if not improved:
            # if no improvement even after shrink, reduce eta strongly
            eta *= 0.1

def build_constraint_matrix(f, active_indices):
    n = len(f)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active_indices):
        # i from max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            j = k - i
            # term f[i] * g[j], add coefficient f[i] to column j
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    # Build sparse matrix using scipy.sparse.csr_matrix
    A = csr_matrix((data, (rows, cols)), shape=(len(active_indices), n))
```

def improve_by_lp(f, n, fft_len, deadline):
    # assume f normalized sum=1
    # compute conv_f
    conv_f = conv_fft(f, fft_len, n)
    M = conv_f.max()
    best_f = f
    best_M = M
    # parameters
    alpha = 0.2  # mixing step
    while time.time() < deadline:
        # active indices within tolerance
        tol = 1e-10  # maybe relative tolerance based on M
        active = np.where(conv_f >= M - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv_f)])
        # build constraint matrix using current best_f
        A = build_constraint_matrix(best_f, active)
        # solve linear program
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=np.full(len(active), M), bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': min(30, deadline-time.time())})
        if res.status != 0:
            print("LP failed", res.message)
            break
        g_opt = np.maximum(res.x, 0)  # nonnegative
        sum_g = g_opt.sum()
        if sum_g <= 1 + 1e-12:
            break
        # normalize g
        g_norm = g_opt / sum_g
        # mix
        f_new = (1 - alpha) * best_f + alpha * g_norm
        # evaluate new max
        conv_new = conv_fft(f_new, fft_len, n)
        M_new = conv_new.max()
        if M_new < best_M - 1e-12:
            best_f = f_new
            best_M = M_new
            conv_f = conv_new
            # continue loop
        else:
            # no improvement even after mixing
            break
    return best_f, best_M

def propose_candidate(seed=42,budget_s=1000,**kwargs):
    start_time = time.time()
    deadline = start_time+budget_s-1 # safe margin

    # set n either from previous best or choose 5000.

    if isinstance(init_seq,(list,tuple,np.ndarray)):

    else:

    # Normalize to sum=1
    best_a = np.maximum(seq,0)
    s = best_a.sum()
    if s < 1e-8:
        best_a = np.full(n, 1.0/n)
    else:
        best_a = best_a / s

    # precompute FFT lengths

    fft_len_g = next_pow2(3*n-2)

    # Evaluate initial fast score
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast score {best_score}")

    # L-BFGS smoothing schedule (improved)

    # We'll use softmax-y representation.
    best_y = np.log(best_a+1e-12)

    # Run L-BFGS
    if time.time() < deadline - 300:
        y_opt = schedule_optimize(best_y, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage=800)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_a = a_opt
            best_y = y_opt
            best_score = score_opt
            print(f"[L-BFGS] improved fast score {best_score}")

    # Subgradient with more iterations
    if time.time() < deadline - 150:
        best_a, best_score = multi_active_subgrad(best_a, fft_len_conv, fft_len_g, n, deadline - 150, eta0=0.35, max_iters=400_000)
        best_y = np.log(best_a+1e-12)
        print(f"[subgrad] score {best_score}")

    # LP refinement
    if time.time() < deadline - 50:
        # define function to build constraint
        import scipy.sparse as sp
        def build_A(f, active):
            rows = []
            cols = []
            data = []
            n = len(f)
            for row_idx, k in enumerate(active):
                # i range
                i_start = max(0, k - (n-1))
                i_end = min(k, n-1)
                for i in range(i_start, i_end+1):
                    j = k - i
                    rows.append(row_idx)
                    cols.append(j)
                    data.append(f[i])
            if len(data)==0:
                return sp.csr_matrix((0, n))
            return sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))
        # iterative LP loop
        while time.time() < deadline - 5:
            conv = conv_fft(best_a, fft_len_conv, n)
            M = conv.max()
            # pick active indices where conv near max (within 1e-8)
            tol_act = max(1e-12, (M - conv.mean()) * 1e-8)
            active = np.where(conv >= M - tol_act)[0]
            if active.size == 0:
                active = np.array([np.argmax(conv)])
            # Build constraint matrix
            A = build_A(best_a, active)
            # Solve LP: maximize sum(g) subject to A@g <= M, g>=0

            c = -np.ones(n)

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,

                          options={'presolve': True, 'time_limit': min(10.0, deadline - time.time())})
            if res.success:
                g_opt = np.maximum(res.x, 0)
                total = g_opt.sum()
                if total <= 1.0 + 1e-9:
                    # cannot increase sum, break
                    break
                # normalize g to sum=1
                g_norm = g_opt / total
                # mix with step alpha

                a_new = (1 - alpha) * best_a + alpha * g_norm
                # evaluate new max
                conv_new = conv_fft(a_new, fft_len_conv, n)
                M_new = conv_new.max()
                if M_new < M - 1e-12:
                    best_a = a_new
                    best_score = fast_score(best_a, fft_len_conv, n)
                    best_y = np.log(best_a+1e-12)
                    print(f"[LP] improved max conv to {M_new:.12f}, fast score {best_score:.6f}")
                    continue
                else:
                    # maybe try smaller alpha
                    # attempt smaller alphas decreasing
                    improved = False
                    for a in [0.05,0.02,0.01]:
                        a_new = (1 - a) * best_a + a * g_norm
                        conv_new = conv_fft(a_new, fft_len_conv, n)
                        if conv_new.max() < M - 1e-12:
                            best_a = a_new
                            best_score = fast_score(best_a, fft_len_conv, n)
                            best_y = np.log(best_a+1e-12)
                            print(f"[LP] improved with alpha={a}")
                            improved = True
                            break
                    if not improved:
                        break
                # continue loop
            else:
                print("[LP] failed or infeasible:", res.message)
                break

    # final hill climb
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n, deadline, max_iter=60000)

    # final verification
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score {final_val}")
    return final_seq

"""
Optimise a non‑negative step‑function (a probability vector) to minimise

    2·n·max(a*a) / (sum(a)**2)

The algorithm is a hybrid of several proven techniques:

1.  Continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
    max‑convolution.  The smoothing parameter λ is decreased geometrically.
2.  Long projected sub‑gradient descent on the exact non‑smooth objective,
    using the averaged sub‑gradient of all (almost) active convolution entries.
3.  A linear‑programming refinement:
    given the current distribution f, maximise the total mass of a partner
    distribution g under the linear constraints (f*g)_k ≤ M, where
    M = max(f*f).  After normalising g we mix it into f (small step α).
    Repeating this reduces the maximal autocorrelation.
4.  A final polishing L‑BFGS with a very low smoothing factor.
5.  Aggressive hill‑climbing (random mass transfers) until the time budget
    expires.

The routine respects the 1000 s budget, works on the supplied
`height_sequence_1` if available, and returns a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Returns (f, grad_y) for the smoothed objective.

    f = 2·n·smooth_max(a*a) , a = softmax(y)
    λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                     # max = 0 now
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv
    w = exp_shift / Z

    # propagate through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # indices (n‑1)…(n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """Continuation L‑BFGS with a decreasing smoothing schedule."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        proportion = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * proportion))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        a0 = np.zeros(n)
        if chosen:
            a0[chosen] = 1.0 / len(chosen)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "dirichlet":
        alpha = np.full(n, 0.1)
        a0 = rng.gamma(alpha, 1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb: move a random fraction of mass from i to j."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.12

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def multi_active_subgrad(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true max‑convolution objective.
    Uses an averaged sub‑gradient over all (almost) active convolution entries
    and a simple back‑tracking line search.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len_conv, n)
        max_val = conv.max()

        # active set (within a tiny tolerance)
        tol = max(1e-12, (max_val - conv.mean()) * 1e-8)
        active = np.where(conv >= max_val - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        # weight vector for the max (averaged)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient of the (averaged) max entry w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fw = np.fft.rfft(w, fft_len_g)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
        g_vec = conv_rev[(n - 1) : (n - 1 + n)]
        grad = 2.0 * g_vec

        # step size decreasing like 1/sqrt(it)
        eta = eta0 / np.sqrt(it)

        # back‑tracking line‑search
        improved = False
        for _ in range(8):
            a_new = a - eta * grad
            a_new = project_onto_simplex(a_new)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            eta *= 0.5
        if not improved:
            # keep the current a but apply a reduced step
            a = a - eta * grad
            a = project_onto_simplex(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement utilities
# ----------------------------------------------------------------------


def build_constraint_matrix(f: np.ndarray, active: np.ndarray) -> sp.csr_matrix:
    """
    Build a sparse matrix A where (A @ g)[k] = (f * g)[k] for every k in `active`.
    Each row corresponds to one convolution index k.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i runs over indices of f that contribute to (f*g)_k
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(f[i])
    if not rows:
        return sp.csr_matrix((0, n))
    return sp.csr_matrix((data, (rows, cols)), shape=(len(active), n))


def lp_refinement(
    a: np.ndarray,
    fft_len_conv: int,
    n: int,
    deadline: float,
    max_outer: int = 12,
    mix_alpha: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑based refinement described in the docstring.
    Returns the improved vector and its fast score.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)

    while time.time() < deadline and max_outer > 0:
        max_outer -= 1
        conv = conv_fft(best, fft_len_conv, n)
        M = conv.max()

        # active indices (those essentially hitting the maximum)
        tol = max(1e-12, (M - conv.mean()) * 1e-8)
        active = np.where(conv >= M - tol)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        A = build_constraint_matrix(best, active)
        c = -np.ones(n)                     # minimise -sum(g)  → maximise sum(g)
        b_ub = np.full(active.shape[0], M)

        # time limit for the LP is kept modest (a few seconds)
        lp_time = max(0.0, min(8.0, deadline - time.time()))
        try:
            res = linprog(
                c,
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0, None)] * n,
                method="highs",
                options={"presolve": True, "time_limit": lp_time},
            )
        except Exception as e:
            print("[LP] exception:", e)
            break

        if not res.success:
            # infeasible or other failure – nothing more we can gain
            break

        g_opt = np.maximum(res.x, 0.0)
        total = g_opt.sum()
        # if we cannot increase total mass, stop
        if total <= 1.0 + 1e-9:
            break

        # normalise to a probability vector and mix
        g_norm = g_opt / total
        for alpha in (mix_alpha, mix_alpha * 0.5, mix_alpha * 0.2):
            candidate = (1.0 - alpha) * best + alpha * g_norm
            cand_score = fast_score(candidate, fft_len_conv, n)
            if cand_score < best_score - 1e-13:
                best = candidate
                best_score = cand_score
                print(f"[LP] improvement → fast score {best_score:.6f}")
                break
        else:
            # No improvement with any alpha – stop
            break
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low `evaluate_sequence` score.
    Returns a plain ``list`` of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # --------------------------------------------------------------
    # Choose problem size and initialise from the previous best if possible
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        # keep at most 10 000 entries, but not less than 4000 for a good search
        n = int(min(max(tmp.size, 4000), 10000))
        if tmp.size != n:
            if tmp.size > n:
                tmp = tmp[:n]
            else:
                # pad with zeros (they will stay zero after normalisation)
                tmp = np.concatenate([tmp, np.zeros(n - tmp.size)])
        a0 = np.maximum(tmp, 0.0)
        s = a0.sum()
        if s < 1e-8:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s
    else:
        n = 5000
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS on the smoothed surrogate
    # --------------------------------------------------------------
    lam_schedule_main = list(np.geomspace(0.5, 1e-12, num=70))

        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:

    # --------------------------------------------------------------

            fft_len_g,

        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline - 70.0:

            fft_len_conv,
            n,
            deadline - 70.0,
            max_outer=12,

        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Phase 4 – final very low‑λ L‑BFGS polishing

    if time.time() < deadline - 45.0:

        y_opt = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule_refine,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_a = a_opt
            best_y = y_opt
            best_score = score_opt
            print(f"[refine L‑BFGS] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – aggressive hill‑climbing until the deadline
    # --------------------------------------------------------------
    if time.time() < deadline:

        best_a, best_score = hill_climb(
            best_a,
            best_score,

            deadline,
            max_iter=40000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # --------------------------------------------------------------

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
