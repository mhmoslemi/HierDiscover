def adam_subgradient(a0, deadline, n, fft_len_conv, fft_len_g):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_score = fast_score(a)
    best_a = a.copy()
    while time.time() < deadline:
        t+=1
        # compute gradient g (subgradient of max conv)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12 * max_val).astype(np.float64)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[n-1: n-1+n]
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.1 / np.sqrt(t)   # can pick
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = simplex_projection(a)
        # evaluate
        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

def deterministic_targeted_hill_climb(a, score, fft_len_conv, fft_len_g, n, deadline):
    best = a.copy()
    best_score = score
    while time.time() < deadline:
        # compute gradient
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[n-1:n-1+n]

        i = np.argmax(g)

2. **Adam‑guided projected sub‑gradient descent** – after the L‑BFGS
   stage the true max‑convolution is used.  The sub‑gradient is
   obtained from the active constraints (indices attaining the
   maximum), Adam‑style momentum is applied and every step is
   projected onto the probability simplex.

3. **Deterministic gradient‑guided hill‑climb** – using the exact
   sub‑gradient, mass is moved *only* from the index with the largest
   gradient to the index with the smallest gradient.  The transferred
   fraction decays linearly from 0.5 to 0.005, guaranteeing fine
   adjustments near convergence.

4. **Final evaluation** – the best sequence is handed to the provided
   ``evaluate_sequence`` function.

All stages respect the overall 1000 s wall‑time budget and work on a
fixed length `n = 4000`.  The algorithm stays in the feasible region
(`a_i ≥ 0`, `∑ a_i = 1`) at every step, so the final call to
`evaluate_sequence` is guaranteed to return a finite score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a distribution with ∑a = 1:
        2·n·max(a*a)
    Uses FFT for speed.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed (log‑sum‑exp) objective and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (multi‑stage λ schedule)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in `lam_schedule`, starting each stage from
    the result of the previous stage.  Returns the best a found and its
    exact fast_score.
    """
    # work in the log‑space (softmax parametrisation)
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # very rare; ignore this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent on the true max‑conv
# ----------------------------------------------------------------------


def adam_subgradient(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the exact max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0
    rng = np.random.default_rng()

    # tolerance for the active mask (top 0.5 % of entries are considered)
    active_pct = 0.005

    while time.time() < deadline:
        t += 1
        # ----- convolution & max -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # ----- active mask (top percentile) -----
        thresh = np.partition(conv, -int(active_pct * conv.size))[-1]
        mask = (conv >= thresh).astype(np.float64)

        # ----- sub‑gradient of the max -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[n - 1 : n - 1 + n]   # size n

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate decays like 0.2 / sqrt(t)
        lr = 0.20 / np.sqrt(t)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = simplex_projection(a)

        # occasional jitter to escape tiny plateaus
        if (t % 4000) == 0 and rng.random() < 0.15:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print progress occasionally
            if (t % 2000) == 0:
                print(f"[adam] iter={t} best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic gradient‑guided hill‑climb (pairwise reallocation)
# ----------------------------------------------------------------------


def deterministic_targeted_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a fraction of mass from the index with the largest
    sub‑gradient to the index with the smallest sub‑gradient.
    The transferable fraction decays linearly from 0.5 to 0.005.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute convolution and sub‑gradient (active set = exact maxima)
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[n - 1 : n - 1 + n]

        i = int(np.argmax(g))
        j = int(np.argmin(g))

        if i == j or (g[i] - g[j]) < 1e-14:
            # gradients are (almost) equal – we are at a stationary point
            break

        # decay the step size from 0.5 to 0.005 over the allowed iterations
        frac = 0.5 - 0.495 * it / max_iter
        delta = best[i] * max(frac, 0.005)  # never smaller than 0.5 % of source

        if delta <= 0.0:
            break

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        # even if not improving we still count the iteration – the next
        # pair (or a smaller step) may lead to an improvement.
        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Random initialisers (diverse families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "spike"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:               # "spike"
        # a few large spikes plus a flat background
        spikes = rng.integers(3, 7)
        a = np.full(n, 1.0 / n)
        for _ in range(spikes):
            pos = rng.integers(0, n)
            mag = rng.uniform(20.0, 80.0) / n
            a[pos] = mag
        a = np.maximum(a, 0.0)
        a = a / a.sum()
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Fixed problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient convolution

    # ------------------------------------------------------------------
    # Starting point (use cached solution if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to the current length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # tiny safety margin

    # Phase schedule (percentages of the total budget)
    phase1_end = start_time + 0.22 * budget_s   # heavy L‑BFGS
    phase2_end = start_time + 0.55 * budget_s   # Adam sub‑gradient
    phase3_end = start_time + 0.90 * budget_s   # deterministic hill‑climb
    # remaining time → final evaluation

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS (very aggressive λ schedule)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.70, 0.50, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.001, 5e-04, 2e-04, 1e-04, 5e-05, 2e-05, 1e-05,
        5e-06,
    ]
    maxiter_per_lam = 700

    while time.time() < phase1_end:
        # occasionally perturb the current best to diversify
        if rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.012)
            a_start = best_a + rng.normal(0.0, sigma, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam‑style projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_subgradient(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic gradient‑guided hill‑climb (fine polishing)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iter=150_000,
        )
        print(f"[hill‑climb] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
