def subgradient_adam(a, fft_len, n, deadline, eta=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        # gradient: g_i = 2*a_{m_idx-i}
        g = np.zeros_like(a)
        i_vals = np.arange(n)
        j = m_idx - i_vals
        mask = (j >= 0) & (j < n)
        g[mask] = 2*a[j[mask]]
        # update moments
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_onto_simplex(a_new)
        # evaluate
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

"""
Optimization of a step‑function (non‑negative sequence) for the
evaluation function (2·n·max(conv))/sum².

Approach
--------
* Work on the probability simplex (∑ a_i = 1) – the objective
  is scale‑invariant, so we keep the sum fixed.
* Initialise from a rich family of analytic shapes.
  In particular we include the classical arcsine‐law
  (Beta(0.5,0.5)), which was missing in the previous run.
* Perform a long projected Adam / momentum sub‑gradient descent
  on the *true* non‑smooth objective.  The sub‑gradient at the
  maximal convolution index `m` is
        g_i = 2·a_{m‑i}   (if 0≤m‑i<n else 0).
  Adam’s adaptive steps together with a Euclidean projection onto
  the simplex give a fast, robust descent.
* When the time budget allows, a short aggressive hill‑climb
  (random mass transfers) is applied to polish the solution.
* The routine respects the global 1000 s budget; the best sequence
  found before the deadline is returned.

The code is self‑contained (only numpy, scipy and the provided
`evaluate_sequence` are used) and can be called as
`propose_candidate(seed=…, budget_s=1000)`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    a is length n, fft_len ≥ 2·n‑1.
    Returns the real part of the convolution (first 2·n‑1 entries).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector
    (∑ a_i = 1).  Returns 2·n·max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, ∑ x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def beta_init(n: int, p: float, q: float | None = None) -> np.ndarray:
    """
    Discrete Beta(p,q) density on (0,1).  If q is None then q = p.
    The result is normalised to sum 1.
    """
    if q is None:
        q = p
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def semicircle_init(n: int) -> np.ndarray:
    """Discrete Wigner semicircle distribution."""
    centre = (n - 1) / 2.0
    radius = centre
    idx = np.arange(n, dtype=float)
    dist = (idx - centre) / radius
    a = np.sqrt(np.clip(1.0 - dist * dist, 0.0, None))
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_gaussian_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal random start."""
    y = rng.standard_normal(n)
    a = np.exp(y)
    a = a / a.sum()
    return a


def mixture_init(a1: np.ndarray, a2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture w·a1 + (1‑w)·a2 (renormalised)."""
    a = w * a1 + (1.0 - w) * a2
    s = a.sum()
    if s == 0.0:
        return uniform_init(len(a1))
    return a / s


# ----------------------------------------------------------------------
# Optimisation helpers
# ----------------------------------------------------------------------


def momentum_subgrad(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    mu: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with classical momentum.
    Returns the best vector seen and its score.
    """
    a = a_start.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    idx = np.arange(n)
    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and max index
        conv = conv_fft(a, fft_len, n)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        m = int(np.argmax(conv))

        # sub‑gradient of max‑convolution
        j = m - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # decreasing step size
        eta = eta0 / np.sqrt(it)

        # momentum update and projection
        v = mu * v - eta * g
        a = a + v
        a = project_onto_simplex(a)

    # final evaluation
    final_score = fast_score(best_a, fft_len, n)
    return best_a, final_score


def adam_optimize(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the same non‑smooth objective.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    idx = np.arange(n)
    t = 0
    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        m_idx = int(np.argmax(conv))
        j = m_idx - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_onto_simplex(a)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: transfer a random fraction of mass
    from one index to another.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns a list of non‑negative floats (the heights).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 12000)
        else:
            n = 8000
    else:
        n = 8000

    fft_len = next_pow2(2 * n - 1)
    print(f"[init] problem size n = {n}")

    # --------------------------------------------------------------
    # Build a small pool of diverse initialisations
    # --------------------------------------------------------------
    candidates = []

    # classic Beta(p,p) family – crucially include p=0.5 (arcsine)
    for p in [0.45, 0.48, 0.50, 0.52, 0.55]:
        a = beta_init(n, p)
        candidates.append((f"beta_{p:.2f}", a))

    # some other shapes
    candidates.append(("semicircle", semicircle_init(n)))
    candidates.append(("uniform", uniform_init(n)))
    candidates.append(("random", random_gaussian_init(n, rng)))

    # mixture of beta and uniform (helps to escape tiny local minima)
    beta_ref = beta_init(n, 0.52)
    for w in [0.2, 0.5, 0.8]:
        mixed = mixture_init(beta_ref, uniform_init(n), w)
        candidates.append((f"mix_{w:.1f}", mixed))

    # reuse previous construction if it exists
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            a = tmp
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a = uniform_init(n)
            else:
                a = a / a.sum()
            candidates.append(("previous", a))

    # --------------------------------------------------------------
    # Evaluate all candidates quickly, keep the most promising ones
    # --------------------------------------------------------------
    scored = []
    for name, a in candidates:
        sc = fast_score(a, fft_len, n)
        scored.append((sc, name, a))
        print(f"[scan] {name:20s} fast score = {sc:.6f}")

    scored.sort()
    top_k = 4
    best_a = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Take the best initial candidate and run two rounds of optimisation:
    #   1) long projected Adam descent,
    #   2) a short hill‑climb polish.
    # --------------------------------------------------------------
    for rank in range(min(top_k, len(scored))):
        sc, name, a0 = scored[rank]
        # normalise to the simplex (should already be)
        a0 = a0 / a0.sum()
        # ------------------------------------------------------------------
        # Phase 1 – projected Adam (takes the bulk of the budget)
        # ------------------------------------------------------------------
        phase1_deadline = min(deadline - 30.0, start_time + 0.80 * budget_s)
        print(f"[phase1] Adam start from {name} (score {sc:.6f})")
        a_opt, sc_opt = adam_optimize(
            a0,
            fft_len,
            n,
            deadline=phase1_deadline,
            lr=0.07,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )
        print(f"[phase1] finished {name} – score = {sc_opt:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – short momentum sub‑gradient descent (fine tuning)
        # ------------------------------------------------------------------
        phase2_deadline = min(deadline - 10.0, start_time + 0.92 * budget_s)
        a_opt2, sc_opt2 = momentum_subgrad(
            a_opt,
            fft_len,
            n,
            deadline=phase2_deadline,
            eta0=0.5,
            mu=0.9,
        )
        print(f"[phase2] finished {name} – score = {sc_opt2:.6f}")

        # ------------------------------------------------------------------
        # Phase 3 – final random hill‑climb
        # ------------------------------------------------------------------
        if time.time() < deadline:
            a_opt3, sc_opt3 = hill_climb(
                a_opt2,
                sc_opt2,
                fft_len,
                n,
                deadline,
                max_iter=5000,
                max_move=0.30,
            )
        else:
            a_opt3, sc_opt3 = a_opt2, sc_opt2

        # keep the best across all seeds
        if sc_opt3 < best_score:
            best_score = sc_opt3
            best_a = a_opt3.copy()
            print(f"[best] new overall best from {name} → {best_score:.6f}")

        # early exit if we run out of time
        if time.time() >= deadline:
            break

    # fallback (should never happen)
    if best_a is None:
        best_a = uniform_init(n)
        best_score = fast_score(best_a, fft_len, n)

    # ------------------------------------------------------------------
    # Final polishing with a tiny targeted hill‑climb (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=10000,
            max_move=0.20,
        )
        print(f"[final] polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Convert to list, evaluate with the provided function and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluate_sequence returned {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation failed:", e)

    return final_seq
