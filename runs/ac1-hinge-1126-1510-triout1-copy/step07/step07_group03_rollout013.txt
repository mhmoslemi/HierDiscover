def compute_conv_and_grad(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2*n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    grad = 2.0 * conv_g[(n-1) : (n-1 + n)]
    return conv, max_val, grad

"""
Propose a low‑score step‑function sequence.

The search combines several complementary components:

1. **A fast L‑BFGS continuation** that works on a smooth
   log‑sum‑exp approximation of the max‑convolution.
2. **Projected sub‑gradient descent** (with momentum) for a long
   period – this reduces the true max‑convolution.
3. **Exponentiated‑gradient descent** on the simplex to keep the
   distribution well‑balanced.
4. **A targeted gradient‑guided hill‑climb** that moves mass from
   indices with a large sub‑gradient to those with a small one.
5. **A small‑scale linear‑programming refinement** (n≈2000) that
   solves the LP “maximise Σ g  subject to f∗g ≤ max(f∗f)”.
   The solution g is mixed back into the large‑scale vector.
6. **A final, long hill‑climb** (tiny moves) that polishes the result.

All phases respect the 1000 s wall‑time budget, keep the coefficients
non‑negative, normalise to a unit sum (the true objective is scale‑free)
and finally hand the best sequence to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a already lies on the simplex
    (∑a = 1).  Uses a single FFT‑based convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective & gradient (smooth max) – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y

def lbfgs_continuation(a_start: np.ndarray, lam_schedule,
                       maxiter_per_lam: int, deadline: float,
                       fft_len_conv: int, fft_len_g: int, n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)  # log‑space parameters

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as exc:
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray, deadline: float,
                                 fft_len_conv: int, fft_len_g: int,
                                 n: int):
    """Projected sub‑gradient descent with momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.95
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size 1/√it
        step = 1.0 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(a0: np.ndarray, deadline: float,
                         fft_len_conv: int, fft_len_g: int,
                         n: int):
    """Exponentiated‑gradient descent on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def targeted_hill_climb(a: np.ndarray, current_score: float,
                        fft_len_conv: int, fft_len_g: int,
                        n: int, deadline: float,
                        max_iter: int = 120000):
    """Gradient‑guided hill‑climb – only strictly improving moves."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        rev_f = np.fft.fft(rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(rev_f * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # percentile‑based source / target selection
        high_thr = np.percentile(g, 80)
        low_thr = np.percentile(g, 20)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i, j = rng.integers(0, n, size=2)
        else:
            i = int(rng.choice(high_idx))
            j = int(rng.choice(low_idx))

        if i == j:
            it += 1
            continue

        # move a modest fraction (≤ 3 %)
        delta = best[i] * rng.random() * 0.03
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        # else reject
        it += 1

    return best, best_score

def lp_refine_small(a_init: np.ndarray, n_small: int,
                    max_iters: int, deadline: float):
    """
    Iteratively improve a (size n_small) using a linear program:
        maximise Σ g    s.t.   a * g ≤ max(a * a),   g ≥ 0.
    Mixing a fraction of the LP solution back into a after each iteration.
    """
    a = a_init.copy()
    fft_len = next_pow2(2 * n_small - 1)
    best_score = fast_score(a, fft_len, n_small)
    best_a = a.copy()
    mix = 0.20
    it = 0
    offsets = np.arange(n_small)

    while time.time() < deadline and it < max_iters:
        # current max‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n_small - 1]
        M = conv.max()

        # build convolution matrix A (size (2n‑1) × n)
        # each column i contains the constant a[i] on diagonal offset i
        diag_data = np.tile(a[:, None], (1, n_small))   # shape (n_small, n_small)
        A = diags(diag_data, offsets, shape=(2 * n_small - 1, n_small), format='csc')

        # linear program: maximise Σ g  ↔ minimise −Σ g
        c = -np.ones(n_small)
        b = np.full(2 * n_small - 1, M)

        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n_small,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            break

        g = res.x
        s = g.sum()
        if s <= 0:
            break
        g = g / s                       # rescale to unit sum

        # mix the LP solution back into a
        a_new = (1.0 - mix) * a + mix * g
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n_small)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            mix = min(mix * 1.1, 0.8)   # gently increase step if it helped
        else:
            mix *= 0.6
            if mix < 1e-4:
                break

        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    max_n = 4000
    n = max_n

    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            # interpolate to required length
            x_old = np.linspace(0.0, 1.0, init_seq.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, init_seq)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)   # a bit larger for the gradient step

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split (percentages of total budget)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2     # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – light L‑BFGS continuation (30 % of time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
                    0.001, 0.0005, 0.0002]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len_conv, fft_len_g, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum (30 %)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (15 %)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.75 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted gradient‑guided hill‑climb (15 %)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.90 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = targeted_hill_climb(
            best_a, best_score,
            fft_len_conv, fft_len_g, n,
            phase4_end,
            max_iter=120000
        )
        print(f"[targeted‑hill] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – small‑scale LP refinement (remaining time before final)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Use a half‑size problem for the LP (faster to solve)
        n_small = max(500, min(2000, n // 2))
        x_big = np.linspace(0.0, 1.0, n)
        x_small = np.linspace(0.0, 1.0, n_small)
        a_small = np.interp(x_small, x_big, best_a)
        a_small = np.maximum(a_small, 0.0)
        a_small /= a_small.sum()

        lp_deadline = time.time() + min(30.0, deadline - time.time())
        a_small_opt, small_score = lp_refine_small(
            a_small, n_small, max_iters=12, deadline=lp_deadline
        )
        print(f"[LP‑refine] small n={n_small}, fast_score={small_score:.6f}")

        # Upsample back to the original size and finish with another hill‑climb
        a_up = np.interp(x_big, x_small, a_small_opt)
        a_up = np.maximum(a_up, 0.0)
        a_up /= a_up.sum()
        up_score = fast_score(a_up, fft_len_conv, n)
        if up_score < best_score:
            best_score = up_score
            best_a = a_up.copy()
            print(f"[LP‑up] new best fast_score={best_score:.6f}")

        # final polishing hill‑climb (few seconds)
        final_deadline = deadline
        best_a, best_score = targeted_hill_climb(
            best_a, best_score,
            fft_len_conv, fft_len_g, n,
            final_deadline,
            max_iter=50000
        )
        print(f"[final‑hill] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
