def targeted_descent(a, fft_len, n, deadline):
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    step_factor = 0.2
    last_improve_time = time.time()
    it = 0
    while time.time() < deadline:
        conv = np.fft.irfft(np.fft.rfft(best_a, fft_len) * np.fft.rfft(best_a, fft_len), fft_len)[:2*n-1]
        max_idx = int(conv.argmax())
        # compute gradient vector g as above
        g = np.zeros_like(best_a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        i_vec = np.arange(i_start, i_end+1)
        j_vec = max_idx - i_vec
        g[i_vec] = 2 * best_a[j_vec]
        # remove negative contributions? all non-negative.
        # choose source with high g and dest low g
        src = np.argmax(g * (best_a > 1e-12))
        dst = np.argmin(g)
        if src == dst:
            # choose next best dest
            sorted_dest = np.argsort(g)
            for cand in sorted_dest:
                if cand != src:
                    dst = cand
                    break
        # compute delta with diminishing step
        delta = step_factor * best_a[src]
        if delta < 1e-12:
            # reduce step factor
            step_factor *= 0.5
            if step_factor < 1e-4:
                step_factor = 0.2
            continue
        a_candidate = best_a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta
        # ensure nonnegative
        if a_candidate[src] < -1e-12:
            # adjust delta upward but we can just skip
            continue
        cand_score = fast_score(a_candidate, fft_len, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate
            # reset step factor maybe increase

            # optionally break early if improvement
        else:
            # try smaller delta
            step_factor *= 0.5
        it += 1
    return best_a, best_score

"""
Propose a non‑negative step function with a low evaluation score.

Algorithm Overview
------------------
1. **Initial pool** – generate a diverse set of candidate vectors using
   classic windows, spectral tricks and randomised families.
2. **Select the best start** according to the exact fast_score
   (2·n·max(self‑convolution) for a unit‑sum vector).
3. **Phase‑1 – L‑BFGS on a smoothed max objective** – a schedule of
   decreasing smoothing parameters (λ = factor·max(conv)) is followed.
   The optimizer works on the soft‑max logits to keep the vector on the
   simplex.
4. **Phase‑2 – Targeted max‑descent** – repeatedly locate the index
   where the self‑convolution attains its maximum, compute the exact
   gradient of that entry (g_i = 2·a_{k−i}), and transfer a small amount
   of mass from the position with the largest gradient to the one
   with the smallest.  A simple line‑search on the transferred amount
   is performed; moves are accepted only if the max‑convolution value
   decreases.
5. **Phase‑3 – Final greedy hill‑climb** – strictly improving random
   mass‑transfer moves to polish the solution.
6. **Evaluation** – the best vector is returned as a plain list and
   scored with the official ``evaluate_sequence`` routine.

All steps respect the global 1000 s time budget; the code is
self‑contained and uses only NumPy, SciPy and the built‑in random
generator.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Exact fast score (sum = 1 → objective = 2·n·max(conv))

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density (clipped)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def lbfgs_refine(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run L‑BFGS through the whole λ schedule once.
    Returns (best_a, best_score) obtained during the run.
    """
    y = np.log(a_start + 1e-12)  # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def compute_gradient_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Exact gradient of the self‑convolution entry at index max_idx.
    For each i, ∂(conv[max_idx])/∂a_i = 2·a_{max_idx-i} (if the index is inside range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    i = np.arange(i_start, i_end + 1)
    j = max_idx - i
    g[i] = 2.0 * a[j]
    return g


def targeted_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_no_improve: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy descent that always moves mass from the index with the
    largest contribution to the current peak of the self‑convolution
    to the index with the smallest contribution.
    """
    a = a_start.copy()
    best_score = fast_score(a, fft_len, n)

    # bookkeeping for stagnation
    it_without_improve = 0
    total_iters = 0

    while time.time() < deadline:
        # compute convolution and locate the peak
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # gradient of the peak entry
        g = compute_gradient_max(a, max_idx)

        # choose source (large gradient, non‑zero mass) and destination (small gradient)
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(g[src_candidates])]

        dst = int(np.argmin(g))
        if dst == src:
            # pick the second smallest if accidently equal
            sorted_idx = np.argsort(g)
            for cand in sorted_idx:
                if cand != src:
                    dst = cand
                    break

        # potential improvement amount (per unit transfer)
        grad_diff = g[src] - g[dst]
        if grad_diff <= 0:
            # no decreasing direction found – stop
            break

        # try a handful of transfer fractions and keep the best
        best_local_a = a.copy()
        best_local_score = best_score
        improved = False
        for frac in (0.20, 0.10, 0.05, 0.02):
            delta = frac * a[src]
            if delta <= 0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
            sc = fast_score(cand, fft_len, n)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = cand.copy()
                improved = True

        if improved:
            a = best_local_a
            best_score = best_local_score
            it_without_improve = 0
        else:
            # reduce the step size gradually – if we keep failing, we break after many attempts
            it_without_improve += 1
            if it_without_improve >= max_no_improve:
                break

        total_iters += 1

    return a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 300000,
    delta_frac: float = 0.09,
) -> tuple[np.ndarray, float]:
    """
    Strictly improving greedy hill‑climb (random mass transfers).
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – a few thousand coefficients give a good trade‑off.
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # If a previous solution exists, up‑sample it.
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:

            else:

            candidates.append(a0)

    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=9.0))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families
    for _ in range(7):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(4):
        beta = rng.uniform(5.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Choose the best start according to the exact fast_score.
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈40 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
        2e-06,
        1e-06,
        5e-07,
        2e-07,
        1e-07,
        5e-08,
        2e-08,
        1e-08,
        5e-09,
        2e-09,
        1e-09,
        5e-10,
        2e-10,
        1e-10,
    ]
    maxiter_per_lam = 5000

    best_a, best_score = lbfgs_refine(
        best_a, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
    )
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Targeted max‑descent (≈45 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_descent(
            best_a, fft_len, n, deadline, rng
        )
        print(f"[Targeted descent] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_iter=400000,
            delta_frac=0.07,
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
