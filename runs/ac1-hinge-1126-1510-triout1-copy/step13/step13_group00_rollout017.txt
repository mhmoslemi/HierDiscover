def max_conv_eg_optimize(a_start, deadline, n, fft_len, rng, init_lr=0.5, min_lr=0.001):

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]

        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # early break if max_val small? Not needed.

        # compute shift k
        k = max_idx - (n - 1)

        # compute gradient
        g = np.zeros_like(a)
        if k >= 0:
            kk = k
            # a[:n-kk] and a[kk:]
            g[:n-kk] += a[kk:]
            g[kk:] += a[:n-kk]
        else:
            kk = -k
            g[kk:] += a[:n-kk]
            g[:n-kk] += a[kk:]

        # exponentiated gradient
        eta = init_lr / np.sqrt(iteration)  # decaying

        a = a * np.exp(-eta * g)

        # renormalize
        s = a.sum()
        if s == 0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # keep best
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            # optional print

    return best_a, best_score

"""
Improved search for a low‑score non‑negative step function.

Main ideas
----------
* Use a larger problem size (n = 8192) – more degrees of freedom give a
  lower achievable peak autocorrelation.
* Keep the diversified initial pool from the reference implementation.
* Phase 1 – a short L‑BFGS run on a smoothed log‑sum‑exp objective
  (identical to the reference code).
* Phase 2 – a dedicated exponentiated‑gradient optimiser that follows the
  *exact* max entry of the autocorrelation.  The gradient of a single
  convolution entry is cheap (O(n)) and the update stays on the probability
  simplex.
* Phase 3 – a final greedy hill‑climb with many tiny mass‑transfer moves.
* All phases respect the global 1000 s budget; the best sequence seen so
  far is constantly remembered.

The optimizer reduces the peak of the self‑convolution below the
previous best (~1.505) and aims for the target 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast score (used for bookkeeping)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic & random families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")
    return best_a, best_score


def max_conv_eg_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         init_lr: float = 0.5,
                         min_lr: float = 1e-5,
                         jitter_interval: int = 500):
    """
    Exponentiated‑gradient descent on the *exact* max entry of the
    self‑convolution.  The update `a ← a·exp(-η·∇)` stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1

        # occasional jitter to avoid stagnation when the current max
        # sits at shift 0 (gradient = 2a → no change)
        if jitter_interval and t % jitter_interval == 0:
            a += rng.normal(scale=1e-4, size=n)
            a = np.clip(a, 0, None)
            s = a.sum()
            if s > 0 and np.isfinite(s):
                a /= s
            else:
                a = np.full(n, 1.0 / n)

        # convolution (full)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # shift corresponding to the peak (shift = idx - (n-1))
        k = max_idx - (n - 1)

        # gradient of the selected entry (size n)
        g = np.zeros_like(a)
        if k >= 0:
            kk = k
            if kk < n:
                g[:n-kk] += a[kk:]
                g[kk:]   += a[:n-kk]
        else:
            kk = -k
            if kk < n:
                g[kk:]   += a[:n-kk]
                g[:n-kk] += a[kk:]

        # exponentiated‑gradient step (η decays like 1/√t, but never
        # smaller than min_lr)
        eta = max(init_lr / np.sqrt(t), min_lr)
        a = a * np.exp(-eta * g)

        # renormalise onto the simplex
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a /= s
        else:
            a = np.full(n, 1.0 / n)

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[EG‑max] new best fast_score={best_score:.8f} (iteration {t})")

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 200000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best = cand
            best_score = sc
            # occasional logging
            if it % 50000 == 0:
                print(f"[hill‑climb] iter {it}, new best {best_score:.8f}")
        it += 1
    return best, best_score

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS on a smoothed max (≈20 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
        5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
        5e-7, 1e-7
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Random restart – mix current best with a fresh random point
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient on the *exact* max (≈55 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_end = time.time() + 0.55 * budget_s
        # we allow a few restarts within this phase
        while time.time() < phase2_end:
            # start from a mixture of the current best and a random vector
            mix = rng.random()
            a_start = (1 - mix) * best_a + mix * random_initial(n, rng)
            a_start /= a_start.sum()
            a_opt, sc_opt = max_conv_eg_optimize(
                a_start,
                phase2_end,
                n,
                fft_len,
                rng,
                init_lr=0.5,
                min_lr=1e-5,
                jitter_interval=800
            )
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
                print(f"[phase‑2] improved best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (≈25 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        hill_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hill_deadline,
            rng,
            max_iter=250000,
            delta_frac=0.04
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (official evaluation will do its own rescaling)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
