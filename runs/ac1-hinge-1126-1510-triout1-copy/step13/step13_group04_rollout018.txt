def lp_refine(a_start, deadline, n, max_constraints=300, t=0.5):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    iteration = 0
    while time.time() < deadline:
        # compute conv and M
        conv = np.convolve(a, a)
        M = conv.max()
        # build constraints for top constraints
        top_k = int(min(max_constraints, len(conv)))
        idx = np.argpartition(conv, -top_k)[-top_k:]
        # Build sparse matrix A
        rows = []
        cols = []
        data = []
        for rid, k in enumerate(idx):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end+1, dtype=int)
            coeffs = a[k - js]   # a_i
            rows.extend([rid] * len(js))
            cols.extend(js)
            data.extend(coeffs)
        A = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(len(idx), n))
        # Solve LP
        c = -np.ones(n) # maximize sum
        bounds = [(0, None)] * n
        ub = np.full(len(idx), M)
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b_opt = res.x
        if b_opt.sum() == 0:
            break
        g = b_opt / b_opt.sum()
        # mixing
        a = (1 - t) * a + t * g
        a = np.clip(a, 0, None)
        a = a / a.sum()
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        iteration += 1
        # optionally adapt t (reduce)
        if iteration % 20 == 0:
            t = max(0.1, t * 0.9)
    return best_a, best_score
```

"""
Search for a non‑negative step‑function (vector of heights) that minimises
the provided `evaluate_sequence` objective.
The algorithm mixes several ideas:

* A diverse pool of analytic / random initialisations.
* An iterative *linear‑programming refinement*: given a current shape f,
  we maximise the total mass of a new shape g under the linear constraints
  (f * g)_k ≤ M, where M = max(f * f).  The LP solution is then mixed
  with f; this systematically reduces the maximal self‑convolution.
* A short L‑BFGS polishing on the smooth log‑sum‑exp surrogate of the
  max‑convolution.
* A final greedy hill‑climb for strict improvement.

All phases respect the 1000 s time budget and return the best list of
floats found before the deadline.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact score (used for fast comparisons – sum of vector is assumed 1)
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisation families (all return a probability vector of length n)

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density (clipped)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------
def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS repeatedly over the λ‑schedule.
    Returns (best_a, best_score) observed during the run.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = scipy.optimize.minimize(fun=objective_and_grad,
                                          x0=y,
                                          args=(n, fft_len, fft_len, lam),
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': maxiter_per_lam,
                                                   'gtol': 1e-12,
                                                   'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 40000):
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 200000,
               delta_frac: float = 0.10):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Linear‑programming refinement (new phase)
# ----------------------------------------------------------------------
def lp_refine(a_start: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int,
              max_constraints: int = 120,
              mix_factor: float = 0.5):
    """
    Iteratively tighten the maximal self‑convolution using a simple LP.
    At each iteration we solve
        maximise   sum b_j
        subject to b ≥ 0,   (a * b)_k ≤ M   for the most violating k‑indices,
    where M = max(a * a).  The solution b is normalised and mixed with a.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    iteration = 0
    while time.time() < deadline:
        # current convolution and its maximum
        fa = np.fft.rfft(a, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv_full.max()

        # choose the most active constraints (largest entries)
        k_candidates = np.argpartition(conv_full, -max_constraints)[-max_constraints:]

        # build sparse constraint matrix A (rows = selected k's)
        rows = []
        cols = []
        data = []

        for rid, k in enumerate(k_candidates):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1)
            js = np.arange(j_start, j_end + 1, dtype=int)
            coeffs = a[k - js]                # f_i where i = k - j
            rows.extend([rid] * len(js))
            cols.extend(js.tolist())
            data.extend(coeffs.tolist())

        A = sp.coo_matrix((data, (rows, cols)), shape=(len(k_candidates), n))

        # linear program: maximize sum(b)  <=> minimize -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        ub = np.full(len(k_candidates), M)

        lp_res = scipy.optimize.linprog(c,
                                        A_ub=A,
                                        b_ub=ub,
                                        bounds=bounds,
                                        method='highs')
        if not lp_res.success:
            # if LP fails, abort the refinement loop
            print("[LP] linprog failed, status:", lp_res.message)
            break

        b_opt = lp_res.x
        if b_opt.sum() == 0.0:
            break
        g = b_opt / b_opt.sum()

        # line‑search on mixing factor
        t_cur = mix_factor
        improved = False
        while t_cur > 1e-3:
            a_cand = (1.0 - t_cur) * a + t_cur * g
            a_cand = np.clip(a_cand, 0.0, None)
            a_cand /= a_cand.sum()
            sc = fast_score(a_cand, fft_len, n)
            if sc < best_score:
                a = a_cand
                best_score = sc
                best_a = a.copy()
                improved = True
                break
            t_cur *= 0.5

        iteration += 1
        if not improved:
            # no improvement found – reduce mixing aggressiveness
            mix_factor = max(0.05, mix_factor * 0.9)
        if iteration % 10 == 0:
            print(f"[LP] iter={iteration}, best_score={best_score:.8f}")

        # safety stop if mixing factor becomes too small
        if mix_factor < 0.01:
            break
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – 5000 items give a good trade‑off between quality
    # and runtime.
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # Possibly up‑sample a previously found solution.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=9.0))

    # Additional families
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # Randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(5):
        candidates.append(pow_decay_init(n, rng))
    for _ in range(4):
        beta = rng.uniform(5.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Pick the best start according to the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – Linear‑programming refinement (≈30 % of budget)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.30 * budget_s
    if time.time() < phase0_end:
        best_a, best_score = lp_refine(best_a,
                                       phase0_end,
                                       n,
                                       fft_len,
                                       max_constraints=150,
                                       mix_factor=0.45)
        print(f"[LP refinement] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.60 * budget_s
    if time.time() < phase1_end:
        lam_schedule = [
            0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
            0.05, 0.03, 0.018, 0.01, 0.006,
            0.003, 0.0015, 0.0008, 0.0004, 0.0002,
            0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
            2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
            5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
            5e-10, 2e-10, 1e-10
        ]
        maxiter_per_lam = 2500
        best_a, best_score = lbfgs_refine(best_a,
                                          lam_schedule,
                                          maxiter_per_lam,
                                          phase1_end,
                                          fft_len,
                                          n)
        print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.05,
                                           max_iter=30000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final greedy hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=300000,
                                        delta_frac=0.12)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
