"""
Propose a low‑score step function.

Algorithm
---------
1. Build a diverse initial pool (deterministic windows, random Dirichlet‑like,
   beta‑shaped, previous solution if present). Keep the best on the exact
   fast_score = 2·n·max(conv) (sum(a)=1).

2. Phase 1 – L‑BFGS with a continuation on a smoothed max (log‑sum‑exp).
   Many random restarts are performed; each restart runs L‑BFGS for a
   geometric schedule of smoothing factors (lam‑factor).  The best vector
   is kept.

3. Phase 2 – Projected sub‑gradient descent on the *exact* max‑convolution.
   The sub‑gradient is obtained by an FFT‑based back‑propagation through the
   convolution and the current max‑entries are used as a mask.  After each
   step the iterate is projected onto the probability simplex.

4. Phase 3 – Greedy pairwise mass‑transfer hill‑climb for final polishing.
   Small amounts of mass are moved from one index to another; moves that
   improve the fast_score are kept.

5. The best sequence is returned; the official ``evaluate_sequence`` is
   called once for reporting only.

All phases respect a global deadline (default 1000 s).  The optimiser
works on vectors of length 4096 but can be adapted by changing ``n``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Fast exact score (sum(a)=1) – avoids the costly Python loop
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a∗a) for a on the simplex (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c                # ≤ 0
    exp_shift = np.exp(shifted / lam)          # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient w.r.t. a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (sqrt of uniform spectrum) initial guess."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a

def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shaped density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in ``lam_schedule`` (interpreted as a
    smoothing‐factor).  Starts each stage from the previous result.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits (ensures positivity)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")
    return best_a, best_sc

def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator):
    """
    Sub‑gradient descent on the exact max‑convolution with Euclidean projection
    onto the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    step0 = 0.8
    while time.time() < deadline:
        # convolution and mask of maximal entries
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-12).astype(np.float64)

        # gradient of max convolution wrt a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        # step size (diminishing)
        eta = step0 / np.sqrt(it + 1.0)
        a_new = a - eta * g
        a_new = simplex_projection(a_new)

        # occasional tiny jitter to shake out flat plateaux
        if it % 2500 == 0 and rng.random() < 0.12:
            a_new = a_new + 1e-4 * rng.normal(size=n)
            a_new = simplex_projection(a_new)

        a = a_new
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[ProjSub] it={it}, fast_score={best_sc:.9f}")
        it += 1
    return best_a, best_sc

def hill_climb_optimize(a_start: np.ndarray,
                        cur_sc: float,
                        n: int,
                        fft_len: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_iter: int = 200000,
                        delta_frac: float = 0.05):
    """
    Strictly improving greedy hill‑climb with tiny pairwise mass transfers.
    """
    a = a_start.copy()
    best_sc = cur_sc
    best = a.copy()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = a[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best = cand.copy()
            a = cand
            print(f"[HillClimb] it={it}, new best fast_score={best_sc:.9f}")
        else:
            # revert the move (keep the old a)
            it += 1
            continue
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # -------------------- problem size --------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # -------------------- build initial pool --------------------
    candidates = []

    # previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0:
                a0 = a0 / a0.sum()
                candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))

    # randomised families
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # beta‑shaped families
    for _ in range(6):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # keep the best initial candidate (exact fast_score)
    best_a = None
    best_sc = float('inf')
    for a in candidates:
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s <= 0:
            continue
        a = a / s
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    print(f"[init] best fast_score={best_sc:.9f}")

    # -------------------- time budget --------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinements (multiple random restarts)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s   # 55 % of total time
    lam_schedule = np.geomspace(0.5, 5e-6, num=40)   # smoothing‑factor schedule
    maxiter_per_lam = 300

    while time.time() < phase1_end:
        # create a blended start vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4                # up to 40 % new mass
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           fft_len,
                                           n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_end = time.time() + 0.25 * budget_s
        print("[ProjSub] start")
        best_a, best_sc = projected_subgradient_optimize(best_a,
                                                         sub_end,
                                                         n,
                                                         fft_len,
                                                         rng)
        print(f"[ProjSub] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Greedy hill‑climb for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + remaining
        print("[HillClimb] start")
        best_a, best_sc = hill_climb_optimize(best_a,
                                              best_sc,
                                              n,
                                              fft_len,
                                              hill_deadline,
                                              rng,
                                              max_iter=250000,
                                              delta_frac=0.04)
        print(f"[HillClimb] final fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.9f}")
    except Exception as e:   # pragma: no cover
        print("[final] evaluate_sequence error:", e)
    return final_seq
