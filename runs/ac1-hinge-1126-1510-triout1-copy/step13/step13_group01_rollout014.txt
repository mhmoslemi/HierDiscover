def propose_candidate(seed=42, budget_s=1000):
    import ...
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0
    best_a_global = None
    best_score_global = np.inf
    for n in [2048, 4096, 8192]:
        # compute N Fourier length
        fft_len = next_pow2(2*n-1)
        # generate candidates
        candidates = []
        # deterministic windows
        candidates.append(triangular_init(n))
        candidates.append(hann_init(n))
        candidates.append(blackman_init(n))
        candidates.append(gaussian_init(n, sigma_factor=0.25))
        # param grid for gaussian sigma
        for sigma_factor in [0.12,0.16,0.2,0.3]:
            candidates.append(gaussian_init(n, sigma_factor))
        # exponential init
        for rate in [0.005, 0.01, 0.02, 0.05]:
            i = np.arange(n)
            a = np.exp(-rate * i)
            a /= a.sum()
            candidates.append(a)
        # kaiser family param
        for beta in [4.0, 6.0, 8.0, 10.0]:
            candidates.append(kaiser_init(n,beta))
        # random families
        for _ in range(8):
            candidates.append(random_initial(n, rng))
        # if previous global exists, interpolate
        if 'height_sequence_1' in globals():
            prev = np.array(globals()['height_sequence_1'], dtype=float)
            if prev.shape[0] > 0:
                x_old = np.linspace(0,1,prev.shape[0])
                x_new = np.linspace(0,1,n)
                a0 = np.interp(x_new, x_old, prev)
                a0 = np.clip(a0, 0, None)
                a0 /= a0.sum()
                candidates.append(a0)
        # evaluate fast_score for each, keep best few (e.g., top 5)
        cand_scores = [(fast_score(c, fft_len, n), c) for c in candidates]
        cand_scores.sort(key=lambda x: x[0])
        top_candidates = [c for (_,c) in cand_scores[:5]]
        # for each top candidate, run L-BFGS continuation
        for a0 in top_candidates:
            if time.time() >= deadline: break
            # maybe run L-BFGS with schedule
            lam_schedule = [0.2,0.1,0.05,0.02,0.01,0.005,0.002,0.001,5e-4,2e-4,1e-4,5e-5,2e-5,1e-5]
            a_opt, sc_opt = lbfgs_continuation(a0, lam_schedule, maxiter_per_lam=2000,
                                               deadline=deadline, fft_len_conv=fft_len,
                                               fft_len_g=fft_len, n=n)
            if sc_opt < best_score_global:
                best_score_global = sc_opt
                best_a_global = a_opt.copy()
                print(f"[{n}] new global best fast_score={best_score_global:.12f}")
        # After L-BFGS, continue with Adam on best for that n
        if time.time() < deadline and best_a_global is not None:
            # Optimize with Adam topk
            best_a_global, best_score_global = adam_topk_optimize(best_a_global,
                                                                  deadline, n, fft_len,
                                                                  rng, start_top=0.2,
                                                                  end_top=0.001)
        # Adam smooth
        if time.time() < deadline:
            remaining = deadline - time.time()
            middle_deadline = time.time() + min(remaining*0.5, 60)  # allocate half of remaining
            best_a_global, best_score_global = adam_smooth_optimize(best_a_global,
                                                                    middle_deadline, n,
                                                                    fft_len, rng,
                                                                    lam_start=5e-2,
                                                                    lam_end=1e-8)
        # Simulated annealing
        if time.time() < deadline:
            remaining = deadline - time.time()
            sa_deadline = time.time() + min(remaining*0.5, 120)
            best_a_global, best_score_global = simulated_annealing(best_a_global,

                                                                   init_temp=1e-3,
                                                                   final_temp=1e-10,
                                                                   max_moves=800000)
        # Hill climb
        if time.time() < deadline:
            remaining = deadline - time.time()
            hill_deadline = time.time() + min(remaining, 30.0)
            best_a_global, best_score_global = hill_climb(best_a_global,
                                                          best_score_global,
                                                          fft_len, n,
                                                          hill_deadline,
                                                          rng,

    # final evaluation
    final_seq = best_a_global.tolist()
    final_eval = evaluate_sequence(final_seq)

"""
Propose a low‑autocorrelation step‑function.

The objective (used by `evaluate_sequence`) is            2·n·max(a∗a) / (∑a)².
Because the ratio is invariant under positive scaling we optimise
probability vectors (∑a = 1, a≥0).  The algorithm proceeds in several
phases:

1.  Build a diverse pool of initial candidates (window functions,
    spectral constructions, random draws, and an interpolation of the
    previous best if it exists).
2.  Pick the best few candidates using a cheap exact scorer
    (`fast_score` = 2·n·max(conv)).
3.  For each chosen start run a *continuation* of L‑BFGS on a smooth
    log‑sum‑exp approximation of the max‑convolution.  The smoothing
    factor is reduced through a long schedule, allowing the optimizer
    to slide into narrow minima.
4.  Refine the best vector with two Adam runs – a “top‑k” phase that
    gradually focuses on the true maximum and a pure smooth‑max phase
    with an exponentially decaying temperature.
5.  Run a long simulated‑annealing walk on the simplex with tiny
    mass‑transfer moves, accepting worsening steps with a slowly cooling
    temperature.
6.  Finish with a greedy hill‑climb that only accepts strictly
    improving moves.
7.  Return the best sequence found before the overall deadline.  The
    final score is evaluated with the provided `evaluate_sequence`.

The code is deliberately time‑aware: each phase checks the remaining
budget and aborts early if needed, guaranteeing a result well before the
1000 s limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax producing a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast scoring (objective for a probability vector)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(a∗a) for a probability vector (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (objective, gradient) for the log‑sum‑exp smooth max
    of the self‑convolution.  The variable y are logits; a = softmax(y)
    is a probability vector.  The smoothing parameter is
    lam = max(conv)·lam_factor.
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    # objective = 2·n·smooth_max
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral guess: inverse FFT of a flat spectrum, projected onto simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    pad = np.zeros(L)
    pad[:target_len] = uniform
    S = np.fft.fft(pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomised draws on the simplex (various shapes)."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'random'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS on a sequence of smoothing factors (lam_factor values).
    Returns the best probability vector discovered and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_topk_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       start_top: float = 0.15,
                       end_top: float = 0.001):
    """
    Adam optimiser with a shrinking top‑k mask that concentrates on the
    true maximum of the convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # shrink the top‑fraction linearly
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        top_frac = max(end_top, start_top * (1.0 - frac))
        k = max(1, int(np.ceil(top_frac * conv.size)))

        thresh = np.partition(conv, -k)[-k]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of masked max (w.r.t. a)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]
        grad_a = 2.0 * n * g_a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)          # decreasing learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if t % 4000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         lam_start: float = 0.05,
                         lam_end: float = 1e-6):
    """
    Adam optimiser on the log‑sum‑exp smooth maximum.
    λ decays exponentially from lam_start to lam_end.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0
    t = 0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    phase_start = time.time()
    total_allowed = deadline - phase_start

    while time.time() < deadline:
        t += 1
        elapsed = time.time() - phase_start
        frac = elapsed / total_allowed if total_allowed > 0 else 1.0
        # aggressive exponential decay
        lam = lam_end + (lam_start - lam_end) * (0.5 ** (10 * frac))

        a = softmax(y)

        # smoothed objective & gradient (lam is used as the λ‑factor)
        f, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam_factor=lam)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        lr = 0.25 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 3000 == 0:
            y += rng.normal(scale=5e-5, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 300000):
    """
    Simple SA on the simplex with tiny mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # transfer a random fraction of mass from i to j
        delta = a[i] * rng.random() * 0.22
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)

        if sc < best_sc:
            a = cand
            best_sc = sc
            best_a = cand.copy()
        else:
            prob = np.exp(-(sc - best_sc) / max(T, 1e-15))
            if rng.random() < prob:
                a = cand

        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 100000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb using tiny mass‑transfer moves.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best = cand
            best_sc = sc
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  The function respects the given time budget and returns the
    best list of heights found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_global_a = None
    best_global_sc = np.inf

    # Try a couple of problem sizes – larger n gives more freedom.
    # The runtime grows roughly linearly with n (FFT cost ⇒ O(n log n)).
    size_candidates = [2048, 4096, 8192]

    for n in size_candidates:
        if time.time() >= deadline:
            break

        fft_len = next_pow2(2 * n - 1)

        # ----------------------------------------------------------
        # Build a rich pool of initial vectors
        # ----------------------------------------------------------
        candidates = []

        # deterministic windows
        candidates.append(triangular_init(n))
        candidates.append(hann_init(n))
        candidates.append(blackman_init(n))
        candidates.append(gaussian_init(n, sigma_factor=0.25))
        candidates.append(spectral_init(n))

        # parameter sweeps for a few families
        for sigma in (0.12, 0.16, 0.20, 0.28):
            candidates.append(gaussian_init(n, sigma_factor=sigma))
        for beta in (4.0, 6.0, 8.0, 10.0):
            candidates.append(kaiser_init(n, beta))
        for rate in (0.005, 0.01, 0.02, 0.04):
            i = np.arange(n, dtype=np.float64)
            a = np.exp(-rate * i)
            a /= a.sum()
            candidates.append(a)

        # randomised families
        for _ in range(6):
            candidates.append(random_initial(n, rng))

        # interpolate the previous best if available
        if 'height_sequence_1' in globals():
            prev = np.array(globals()['height_sequence_1'], dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                x_old = np.linspace(0, 1, prev.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, prev)
                a0 = np.clip(a0, 0.0, None)
                s = a0.sum()
                if s > 0:
                    a0 /= s
                else:
                    a0 = np.full(n, 1.0 / n)
                candidates.append(a0)

        # ----------------------------------------------------------
        # Keep a few (e.g. 5) best seeds according to the cheap fast_score
        # ----------------------------------------------------------
        scored = [(fast_score(c, fft_len, n), c) for c in candidates]
        scored.sort(key=lambda tup: tup[0])
        top_seeds = [c for _, c in scored[:5]]

        # ----------------------------------------------------------
        # Phase 1 – L‑BFGS continuation on a smoothed max
        # ----------------------------------------------------------
        lam_schedule = [
            0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
            0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
            5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
            5e-7, 2e-7, 1e-7, 5e-8, 1e-8
        ]
        maxiter_per_lam = 2500    # a bit more than before

        for seed_a in top_seeds:
            if time.time() >= deadline:
                break
            a_opt, sc_opt = lbfgs_continuation(
                seed_a,
                lam_schedule,
                maxiter_per_lam,
                deadline,
                fft_len,
                fft_len,
                n
            )
            if sc_opt < best_global_sc:
                best_global_sc = sc_opt
                best_global_a = a_opt.copy()
                print(f"[{n}] L‑BFGS new best fast_score={best_global_sc:.12f}")

        # ----------------------------------------------------------
        # Phase 2 – Adam with a shrinking top‑k mask
        # ----------------------------------------------------------
        if best_global_a is not None and time.time() < deadline:
            best_global_a, best_global_sc = adam_topk_optimize(
                best_global_a,
                deadline,
                n,
                fft_len,
                rng,
                start_top=0.2,
                end_top=0.001
            )
            print(f"[{n}] after Adam‑topk fast_score={best_global_sc:.12f}")

        # ----------------------------------------------------------
        # Phase 3 – Adam on smooth‑max (very low λ)
        # ----------------------------------------------------------
        if best_global_a is not None and time.time() < deadline:
            # allocate roughly half of the remaining time to this phase
            remaining = deadline - time.time()
            phase_deadline = time.time() + min(remaining * 0.5, 60.0)
            best_global_a, best_global_sc = adam_smooth_optimize(
                best_global_a,
                phase_deadline,
                n,
                fft_len,
                rng,
                lam_start=5e-2,
                lam_end=1e-8
            )
            print(f"[{n}] after Adam‑smooth fast_score={best_global_sc:.12f}")

        # ----------------------------------------------------------
        # Phase 4 – Simulated annealing (many moves)
        # ----------------------------------------------------------
        if best_global_a is not None and time.time() < deadline:
            remaining = deadline - time.time()
            sa_deadline = time.time() + min(remaining * 0.4, 120.0)
            best_global_a, best_global_sc = simulated_annealing(
                best_global_a,
                sa_deadline,
                n,
                fft_len,
                rng,
                init_temp=1e-3,
                final_temp=1e-10,
                max_moves=600000
            )
            print(f"[{n}] after SA fast_score={best_global_sc:.12f}")

        # ----------------------------------------------------------
        # Phase 5 – final greedy hill‑climb
        # ----------------------------------------------------------
        if best_global_a is not None and time.time() < deadline:
            remaining = deadline - time.time()
            hill_deadline = time.time() + min(remaining, 30.0)
            best_global_a, best_global_sc = hill_climb(
                best_global_a,
                best_global_sc,
                fft_len,
                n,
                hill_deadline,
                rng,
                max_iter=200000,
                delta_frac=0.03
            )
            print(f"[{n}] after hill‑climb fast_score={best_global_sc:.12f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (evaluate_sequence rescales)
    # ------------------------------------------------------------------
    if best_global_a is None:
        # fallback – uniform distribution (never happens in practice)
        n = 4096
        best_global_a = np.full(n, 1.0 / n)

    final_seq = best_global_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.12f}")
    # expose the result for potential reuse by later runs
    globals()['height_sequence_1'] = final_seq
    return final_seq
