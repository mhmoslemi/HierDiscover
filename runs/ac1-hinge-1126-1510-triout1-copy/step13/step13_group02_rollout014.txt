"""
Propose a low‑score step function.

Idea
----
The objective is 2·n·max(conv(a,a)) / (sum(a)²), which is invariant under
scaling of *a*.  After fixing sum(a)=1 the problem reduces to
minimising the peak of the (linear) self‑convolution.
The theoretical lower bound is 2·n/(2·n‑1)≈1, so longer sequences give a
smaller possible score.

Algorithm
---------
* Use a longer sequence (default n=8192) – the lower bound is then ≈1.0.
* Initialise a diverse pool of candidates (spectral with random phase,
  Hann, Kaiser, random Dirichlet‑like, etc.) and keep the best.
* Phase 1 – smooth‑max L‑BFGS (log‑sum‑exp approximation) with a
  geometric λ‑schedule.  Each stage starts from the previous solution.
* Phase 2 – exponentiated‑gradient descent on the exact max‑convolution,
  using only the current peak (top‑fraction ≈ 0.001) as a mask.
* Phase 3 – sub‑gradient descent (exact max) with an EG‑style update.
* Phase 4 – Adam optimiser on the exact objective (logits → softmax).
* Phase 5 – short simulated‑annealing with pairwise mass transfers.
* Phase 6 – greedy hill‑climb (tiny pairwise transfers) for final polishing.
* All phases respect the global deadline (≈ budget_s seconds).

The code is written with top‑level helpers only, uses only NumPy /
SciPy and works with the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex (sum(a)=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
        f   = 2·n·smooth_max(conv)
        grad = ∂f/∂y.
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a (linear, zero‑padded)
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam)   # ∈ (0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution values
    w = exp_shift / Z    # (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisers (deterministic / random families)
# ----------------------------------------------------------------------


def spectral_random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Spectral init with random phases; magnitude = sqrt of uniform spectrum."""
    L = next_pow2(2 * n - 1)
    # uniform amplitude on frequency side
    amp = np.ones(L, dtype=np.float64)
    phase = rng.uniform(0.0, 2 * np.pi, size=L)
    # keep Hermitian symmetry for real IFFT
    phase[0] = 0.0
    phase[L // 2] = 0.0  # Nyquist frequency (if even)
    complex_spectrum = amp * np.exp(1j * phase)
    a_complex = np.fft.ifft(complex_spectrum).real[:n]
    a_complex[a_complex < 0] = 0.0
    s = a_complex.sum()
    if s <= 0:
        a_complex = np.full(n, 1.0 / n)
    else:
        a_complex = a_complex / s
    return a_complex


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_dirichlet_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample from a Dirichlet distribution with random concentration."""
    alpha = rng.uniform(0.2, 2.0, size=n)
    a = rng.gamma(alpha)
    a /= a.sum()
    return a


def random_edge_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mass concentrated in thin bands at both ends."""
    a = np.full(n, 1e-12)
    edge_frac = rng.uniform(0.005, 0.03)
    m_edge = max(1, int(n * edge_frac))
    heavy = np.sqrt(n / 2.0) / n
    a[:m_edge] = heavy
    a[-m_edge:] = heavy
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in ``lam_schedule`` starting each stage from the
    result of the previous stage.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.001):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Only the top ``top_fraction`` entries of the convolution receive a
    non‑zero weight (focuses updates on the current peak).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # mask for the current peak (or a tiny fraction around it)
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient wrt a (via reversed a)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)

        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        # Decaying step size η = 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        # EG update (keep on the simplex)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 250 == 0:
                print(f"[EG] it={it}, fast_score={best_sc:.8f}")
        it += 1
    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int):
    """
    Sub‑gradient descent on the exact max‑convolution.
    Uses an EG‑style multiplicative update.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        # mask of (almost) maximal entries (tolerance to avoid ties)
        tol = 1e-12
        mask = (conv >= max_c - tol).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]

        eta = 0.6 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 250 == 0:
                print(f"[SubGrad] it={it}, fast_score={best_sc:.8f}")
        it += 1
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int):
    """
    Adam optimisation on the exact max‑convolution objective.
    Operates on logits (a = softmax(y)) so that the simplex constraint
    is satisfied automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and its max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient wrt a of the max entry (sub‑gradient)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        grad_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # full objective gradient (2·n·max)
        grad_a = 2.0 * n * grad_a

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 4000 == 0:
            y += np.random.default_rng().normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.8f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 8e-4,
                        final_temp: float = 2e-7,
                        max_moves: int = 120000):
    """
    Simple SA with pairwise mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.12
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        accept = False
        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 60000,
               delta_frac: float = 0.05):
    """
    Greedy hill‑climb (strictly improving tiny pairwise transfers).
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size – larger n gives a smaller theoretical bound.
    # --------------------------------------------------------------
    n = 8192                     # 8192 gives a lower bound ≈ 1.00006
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it to the current n
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_random_init(n, rng))
    candidates.append(hann_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(5.0, 12.0)))
    # Random Dirichlet‑like families
    for _ in range(3):
        candidates.append(random_dirichlet_init(n, rng))
    # Edge‑heavy families
    for _ in range(2):
        candidates.append(random_edge_init(n, rng))

    # Keep the best candidate so far
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinements (aggressive λ‑schedule)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s   # 30 % of total time
    lam_schedule = np.geomspace(0.5, 5e-6, num=25)   # aggressive decay
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # Blend best with a fresh random start to diversify
        rand_vec = random_dirichlet_init(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.clip(a_start, 0, None)
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (focus on current peak)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        eg_deadline = time.time() + 0.15 * budget_s
        print("[EG] start")
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                eg_deadline,
                                                n,
                                                fft_len,
                                                top_fraction=0.001)
        print(f"[EG] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = time.time() + 0.15 * budget_s
        print("[SubGrad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                               sub_deadline,
                                               n,
                                               fft_len)
        print(f"[SubGrad] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam optimisation on the exact objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing fine‑tuning (short)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(15.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=8e-4,
                                              final_temp=2e-7,
                                              max_moves=150000)
        print(f"[SA] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 20.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=80000,
                                     delta_frac=0.04)
        print(f"[Hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
