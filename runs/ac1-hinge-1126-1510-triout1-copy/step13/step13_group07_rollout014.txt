"""
Propose a low‑score step‑function using a multi‑phase search.

Algorithm
---------
1️⃣  **Enriched initialization** – deterministic windows, random draws,
    and a dense grid of Beta‐distribution shapes (U‑shaped, bell‑shaped,
    skewed).  The best candidate by the exact fast score is kept.

2️⃣  **L‑BFGS polishing** – minimise a smoothed max (log‑sum‑exp) of the
    self‑convolution with a geometric λ‑schedule.  Several restarts from
    convex blends of the current best and fresh random vectors explore
    the landscape.

3️⃣  **Adam on the smoothed objective** – a classic Adam optimiser refines
    the solution obtained by L‑BFGS.

4️⃣  **Exponentiated‑gradient on the smoothed max** – a large‑scale EG
    optimiser (full log‑sum‑exp) moves mass rapidly while staying on the
    simplex.

5️⃣  **Exponentiated‑gradient on the true max (sub‑gradient descent)** –
    multiplicative updates driven by the exact max‑convolution gradient
    (g_i = 2·a_{k*‑i}) with a 1/√t step‑size schedule.

6️⃣  **Deterministic hill‑climb** – repeatedly moves a modest fraction of
    mass from a source that contributes most to the current peak to the
    index of minimal mass.  Convolution updates are cheap (FFT per move)
    and the best improvement is retained.

All phases respect a total 1000 s budget.  The final sequence is returned
as a plain Python list of floats (the official scorer rescales it anyway).
"""

import time
import numpy as np
from scipy.optimize import minimize

# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Fast exact score (probability vector → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on {0,…,n‑1} (α,β>0) projected onto simplex."""
    idx = np.arange(n, dtype=np.float64)
    x = idx / (n - 1) if n > 1 else np.zeros(1)
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.30):
    """Exponentiated‑gradient descent on a top‑fraction of the convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # mask top‑fraction entries
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient of smoothed max‑convolution (exact, not smoothed)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def eg_max_optimize(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator):
    """Exponentiated‑gradient descent on the true max (sub‑gradient)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution and the peak index
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient of max‑convolution
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_range = np.arange(i_start, i_end + 1)
        j_range = k - idx_range
        g[idx_range] = 2.0 * a[j_range]

        # diminishing step size η_t = η0 / sqrt(t)
        eta0 = 0.7
        eta = eta0 / np.sqrt(it)

        # multiplicative update + renormalisation (simplex)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_moves: int = 500000):
    """
    Greedy hill‑climb that always moves mass away from the positions
    contributing most to the current peak of the self‑convolution.
    Destination is the index of minimal mass (more deterministic than random).
    """
    best_a = a.copy()
    best_score = cur_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        # compute convolution
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # source selection (probability ∝ contribution to the peak)
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = idx - i_range
        contrib = best_a[i_range] * best_a[j_range]
        total = contrib.sum()
        if total == 0.0:
            src = rng.integers(0, n)
        else:
            probs = contrib / total
            src = rng.choice(i_range, p=probs)

        # deterministic destination = index of minimal mass (different from src)
        dest = np.argmin(best_a)
        if dest == src:
            # second smallest
            sorted_idx = np.argsort(best_a)
            dest = sorted_idx[1] if n > 1 else src

        # transfer a modest fraction
        max_delta = best_a[src] * 0.30
        delta = max_delta * rng.random()
        if delta <= 0.0:
            moves += 1
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dest] += delta
        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_score:
            best_score = sc_cand
            best_a = cand
        moves += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep n moderately large (5000 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an enriched pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2️⃣  Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # 3️⃣  Gaussian windows (grid over sigma)
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    # 4️⃣  Kaiser windows (grid over beta)
    for beta in np.linspace(4.0, 16.0, 13):
        candidates.append(kaiser_init(n, beta=beta))

    # 5️⃣  Power‑law decays (grid over exponent)
    for expo in np.linspace(0.15, 0.70, 12):
        candidates.append(power_decay_init(n, exponent=expo))

    # 6️⃣  Beta‑distribution family (grid over α,β)
    alpha_vals = np.linspace(0.2, 2.5, 7)
    beta_vals = np.linspace(0.2, 2.5, 7)
    for a_val in alpha_vals:
        for b_val in beta_vals:
            candidates.append(beta_init(n, alpha=a_val, beta=b_val))

    # 7️⃣  Randomised families (multiple draws)
    for _ in range(8):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Choose the best start according to the exact fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈25 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-9, num=60)   # finer λ‑grid
    maxiter_per_lam = 3000
    while time.time() < phase1_end:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30               # blend factor ≤ 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.40 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.06,
                                           max_iter=50000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient on top‑fraction (≈15 % of the budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.55 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase3_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.20)
        print(f"[Exp‑grad‑top] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exponentiated‑gradient on the true max (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.75 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = eg_max_optimize(best_a,
                                             phase4_end,
                                             n,
                                             fft_len,
                                             rng)
        print(f"[EG‑max] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Deterministic hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining_moves = 800000   # generous upper bound; loop will stop early by time
        best_a, best_score = targeted_hill_climb(best_a,
                                                 best_score,
                                                 fft_len,
                                                 n,
                                                 deadline,
                                                 rng,
                                                 max_moves=remaining_moves)
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
