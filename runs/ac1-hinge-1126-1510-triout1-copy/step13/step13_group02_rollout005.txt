def pgd_optimize(a_start, deadline, n, fft_len):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # compute gradient at max entries:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-12).astype(np.float64)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev*fm, fft_len)
        g = 2.0 * grad_rev[n-1:n-1+n]  # gradient wrt a of max conv (not multiplied by 2n)
        # gradient of objective = 2*n * g
        grad_obj = 2.0 * n * g

        # step size schedule: eta = eta0 / sqrt(it+1)
        eta0 = 0.5
        eta = eta0 / np.sqrt(it+1)

        # gradient descent step
        a_new = a - eta * grad_obj

        # projection onto simplex
        a_new = simplex_projection(a_new)

        # compute score
        sc = fast_score(a_new, fft_len, n)

        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_sc
```

"""
Propose a low‑score step function.

Algorithm (summary)
-------------------
* Initialise a diverse pool of candidates (window functions, random Dirichlet,
  beta‑shaped, sinc, etc.) and keep the best (exact fast_score = 2·n·max(conv)).
* Phase 1 – multi‑restart L‑BFGS on a log‑sum‑exp smoothed max‑convolution.
  A geometric λ‑schedule with many continuation steps refines the vector.
* Phase 2 – Projected Gradient Descent (PGD) on the exact max‑convolution.
  Uses the sub‑gradient of the max entry and Euclidean projection onto the
  simplex; many tiny steps polish the peak.
* Phase 3 – Short simulated‑annealing with pairwise mass transfers for
  escaping tiny plateaux.
* Phase 4 – Greedy hill‑climb (pairwise moves) for final polishing.
* The best sequence found is returned and scored with the provided
  `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact fast objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) where
       f    = 2·n·smooth_max(conv)
       grad = ∂f/∂y
    y are logits (a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-20

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c          # ≤ 0
    # clip exponent to avoid overflow/underflow
    exp_shift = np.exp(np.clip(shifted / lam, -700, 0))
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z + 1e-30)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t convolution values
    w = exp_shift / Z  # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient w.r.t a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def sinc_init(n: int, scale: float = 0.5) -> np.ndarray:
    """Sinc‑squared window (non‑negative, compact‑ish)."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    x = (i - centre) / (scale * n)
    a = np.sinc(x) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) shaped density on a discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i ** (alpha - 1.0)) * ((n - i) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random draws on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'sharp'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sharp':
        a = np.full(n, 1e-12)
        idx = rng.integers(0, n)
        a[idx] = 1.0
        a /= a.sum()
    else:   # dirichlet
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    a /= a.sum()
    return a

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for every λ in `lam_schedule`, starting each stage from the
    previous result. Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-15)          # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")
    return best_a, best_sc


def pgd_optimize(a_start: np.ndarray,
                 deadline: float,
                 n: int,
                 fft_len: int,
                 eta0: float = 0.5):
    """
    Projected gradient descent on the exact max‑convolution.
    Uses a sub‑gradient of the max entry, a diminishing step size,
    and Euclidean projection onto the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution and mask of maximal entries
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient wrt a of the summed maximal entries
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_rev = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]   # ∂max/∂a

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # gradient step and projection
        a = a - eta * (2.0 * n) * g
        a = simplex_projection(a)

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-8,
                        max_moves: int = 120000):
    """
    Simple SA with pairwise mass‑transfer moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        # transfer a small random fraction from i to j
        delta = a[i] * rng.random() * 0.15
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc or rng.random() < np.exp(-(sc_cand - best_sc) / max(T, 1e-12)):
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (power‑of‑two FFT size works best)
    # --------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an initial pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it to the current n
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))
    candidates.append(sinc_init(n, scale=0.3))
    candidates.append(sinc_init(n, scale=0.5))
    candidates.append(sinc_init(n, scale=0.7))

    # Randomised families
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # Beta‑shaped families
    for _ in range(6):
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        candidates.append(beta_init(n, alpha, beta))

    # --------------------------------------------------------------
    # Keep the best candidate so far (fast_score is exact for sum=1)
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinements (global search)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s   # 35 % of total time
    lam_schedule = np.geomspace(0.5, 1e-8, num=60)   # more gradual decay
    maxiter_per_lam = 1300

    while time.time() < phase1_end:
        # random convex combination of best and a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.25
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Gradient Descent on the exact max
    # ------------------------------------------------------------------
    if time.time() < deadline:
        pgd_deadline = time.time() + 0.20 * budget_s
        print("[PGD] start")
        best_a, best_sc = pgd_optimize(best_a, pgd_deadline, n, fft_len, eta0=0.7)
        print(f"[PGD] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning (short but many moves)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(25.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-8,
                                              max_moves=200000)
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Deterministic greedy hill‑climb for final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 30.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=60000,
                                     delta_frac=0.04)
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
