"""
Proposes a low‑score non‑negative step function.

Algorithm
---------
1. Build a diverse pool of candidate sequences (deterministic windows,
   random Dirichlet‑like, mixture‑Gaussian, sinc‑square, …).  Keep the
   best according to the exact objective 2·n·max(conv).

2. Phase 0 – aggressive L‑BFGS on a log‑sum‑exp smoothed max.  Several
   random restarts are performed; each restart runs a continuation of
   the smoothing parameter λ.  The best vector is retained.

3. Phase 1 – exact linear‑programming refinement.  For the current
   distribution f, solve

        max   Σ b_j
        s.t.  (f * b)_k ≤ M   ∀k ,   b_j ≥ 0

   where M = max(f*f).  The optimal b is normalised to a probability
   vector g and mixed with f.  A tiny 1‑D line‑search over the mixing
   coefficient is performed and the improvement is kept.  This step is
   repeated a few times (building a new convolution matrix each
   iteration).

4. Phase 2 – a final short L‑BFGS continuation to polish the result.

5. Phase 3 – deterministic greedy hill‑climb (pairwise tiny mass
   transfers) for the last polishing.

All phases respect the global 1000 s deadline.  The function returns
the best list of floats; the official ``evaluate_sequence`` rescales
internally, so the algorithm works on the probability simplex (sum = 1).

The implementation is self‑contained, uses only NumPy, SciPy and the
standard library, and runs on a single thread.
"""

import time
import numpy as np
from scipy import optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Fast exact objective for vectors on the simplex (sum == 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a⋆a) for a on the simplex (sum(a)=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the (full) autocorrelation of a."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective f = 2·n·smooth_max(conv) and its gradient wrt y.
    y are logits; a = softmax(y) lives on the simplex.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c                     # ≤0
    exp_shift = np.exp(shifted / lam)               # ∈(0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution entries
    w = exp_shift / Z                     # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (deterministic families)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral (square‑root of a flat spectrum)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def sinc_square_init(n: int) -> np.ndarray:
    """Squared sinc window (always non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2.0) / (n / 2.0)   # roughly ∈[-1,1]
    a = np.sinc(x) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Randomised families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector on the simplex using several heuristics."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'lognorm'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'dirichlet':
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    else:  # lognorm
        sigma = rng.uniform(0.3, 1.0)
        a = rng.lognormal(mean=0.0, sigma=sigma, size=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def mixture_gaussian_init(n: int, rng: np.random.Generator, K: int = 3) -> np.ndarray:
    """Mixture of K Gaussians with random means, sigmas and weights."""
    i = np.arange(n, dtype=np.float64)
    weights = rng.random(K)
    weights /= weights.sum()
    mus = rng.uniform(0, n - 1, size=K)
    sigmas = rng.uniform(0.05 * n, 0.25 * n, size=K)
    a = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def dirichlet_sparse_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse Dirichlet draw (small alphas)."""
    alpha = rng.uniform(0.01, 0.2, size=n)
    a = rng.gamma(alpha)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in ``lam_schedule``; each stage starts from the
    previous result.  Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits on the simplex
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = opt.minimize(fun=objective_and_grad,
                               x0=y,
                               args=(n, fft_len_conv, fft_len_g, lam),
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': maxiter_per_lam,
                                        'gtol': 1e-12,
                                        'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")
    return best_a, best_sc


def build_conv_matrix(a: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n convolution matrix C such that C @ b = a * b.
    """
    n = a.shape[0]
    offsets = -np.arange(n)                     # 0, -1, -2, …, -(n‑1)
    # each diagonal gets a constant value a[d]
    data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    C = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return C


def lp_refine(a: np.ndarray, M: float, n: int) -> np.ndarray:
    """
    Solve max Σ b_j  s.t.  (a * b)_k ≤ M  (∀k),  b ≥ 0.
    Returns the normalised b (sum = 1).  If the LP fails, returns a copy of a.
    """
    C = build_conv_matrix(a)
    c = -np.ones(n)                         # maximise Σ b_j  → minimise -Σ b_j
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    try:
        res = opt.linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True})
    except Exception as e:   # pragma: no cover
        print("[LP] linprog exception:", e)
        return a.copy()
    if not res.success:
        # fallback – keep current distribution
        # (occasionally linprog can report infeasibility due to numerical noise)
        print("[LP] infeasible or failed, keeping current a")
        return a.copy()
    b = res.x
    s = b.sum()
    if s <= 0.0:
        return a.copy()
    b /= s
    return b


def mix_line_search(a: np.ndarray, g: np.ndarray,
                    fft_len: int, n: int,
                    best_sc: float) -> (np.ndarray, float):
    """
    Simple 5‑point line search on t∈[0,1] for a convex combination
    a_t = (1‑t)·a + t·g.  Returns the best a_t and its fast_score.
    """
    # sample a few points (including the endpoints)
    t_vals = np.linspace(0.0, 1.0, 9)
    best_a = a
    cur_best = best_sc
    for t in t_vals:
        if t == 0.0:
            cand = a
        elif t == 1.0:
            cand = g
        else:
            cand = (1.0 - t) * a + t * g
        # re‑normalise to stay on the simplex
        cand_sum = cand.sum()
        if cand_sum <= 0.0:
            continue
        cand = cand / cand_sum
        sc = fast_score(cand, fft_len, n)
        if sc < cur_best:
            cur_best = sc
            best_a = cand
    return best_a, cur_best


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – choose a power‑of‑two length (larger = more freedom)
    # ------------------------------------------------------------------
    n = 4096                     # feel free to increase (memory grows ~n²)
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(6.0, 12.0)))
    candidates.append(sinc_square_init(n))

    # randomised families
    for _ in range(4):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(mixture_gaussian_init(n, rng, K=rng.integers(2, 5)))
    candidates.append(dirichlet_sparse_init(n, rng))

    # pick the best of the pool (exact fast_score)
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – aggressive L‑BFGS continuation (global searches)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.25 * budget_s   # ~25 % of total time
    lam_schedule = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1,
                    0.07, 0.05, 0.03, 0.02, 0.01,
                    6e-3, 4e-3, 2e-3, 1e-3, 5e-4,
                    2e-4, 1e-4, 5e-5]
    maxiter_per_lam = 1500

    while time.time() < phase0_end:
        # blend current best with a fresh random vector (occasionally a mixture)
        rand_vec = random_initial(n, rng)
        if rng.random() < 0.25:
            rand_vec = mixture_gaussian_init(n, rng, K=rng.integers(2, 4))
        alpha = rng.random() * 0.33
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase0_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 1 – Linear‑programming refinement (few iterations)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        lp_deadline = time.time() + 0.30 * (deadline - time.time())
        lp_iters = 0
        max_lp_iters = 6   # keep the total LP work modest
        while time.time() < lp_deadline and lp_iters < max_lp_iters:
            M = conv_max(best_a, fft_len)               # current peak
            b = lp_refine(best_a, M, n)                # solve LP
            # mix and line‑search
            best_a, best_sc = mix_line_search(best_a, b,
                                              fft_len, n, best_sc)
            print(f"[LP] iter={lp_iters}, best fast_score={best_sc:.9f}")
            lp_iters += 1
            # early exit if no noticeable improvement
            if lp_iters > 1 and abs(best_sc - M * 2 * n) < 1e-6:
                break

    # ------------------------------------------------------------------
    # Phase 2 – final short L‑BFGS polish (continuation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase2_end = deadline - 20.0   # reserve a few seconds for hill‑climb
        a_opt, sc_opt = lbfgs_continuation(best_a,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase2_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑final] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 20.0)
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=120000,
                                     delta_frac=0.04)
        print(f"[Hill‑climb] final fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the result – the official evaluator rescales internally.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)   # official score
        print(f"[final] evaluated score = {final_eval:.9f}")
    except Exception:
        # In the sandbox the helper may not be available yet.
        pass
    return final_seq
