def propose_candidate(seed=42, budget_s=1000):
    rng= np.random.default_rng(seed)
    # set n
    n = 5000   # maybe flexible
    fft_len = next_pow2(2*n-1)

    # Build candidate pool
    candidates = []
    candidates.append(spectral_init(n))
    candidates.append(arcsine_init(n))
    candidates.extend([triangular_init(n), hann_init(n), blackman_init(n), inv_sqrt_init(n)])
    # Beta shape grid
    for alpha in np.linspace(0.15, 0.9, 16):
        candidates.append(beta_init(n, alpha))
    # Gaussian windows as before but more sigma values
    for sigma in np.linspace(0.10, 0.4, 9):
        candidates.append(gaussian_init(n, sigma_factor=sigma))
    # random draws
    for _ in range(6):
        candidates.append(random_initial(n, rng))

    # Evaluate fast_score (normalize sum)
    best_a = None; best_score = float('inf')
    for cand in candidates:
        a = cand.copy()
        s = a.sum()
        if s <= 0:
            continue
        a = a / s
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a

    print(f"[init] best fast_score {best_score}")

    # time budget
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # L-BFGS phase
    # schedule lam values
    lam_vals = np.geomspace(0.5, 1e-10, num=60)
    # allocate restarts until time is near half
    while time.time() < start_time + 0.45*budget_s:
        # blend with random
        rand = random_initial(n, rng)
        alpha = rng.random()*0.3
        a_start = (1-alpha)*best_a + alpha*rand
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start, lam_vals, maxiter_per_lam=4000, deadline=deadline, fft_len=fft_len, n=n)
        if sc_opt < best_score:
            best_score = sc_opt; best_a = a_opt
            print(f"[L-BFGS] new best {best_score}")

    # Symmetrize best_a
    best_a = (best_a + best_a[::-1]) / 2
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[symmetrize] best fast_score {best_score}")

    # Adam phase
    if time.time() < start_time + 0.55 * budget_s:
        best_a, best_score = adam_optimize(best_a, deadline, n, fft_len, lam_factor=0.001, lr=0.07, max_iter=40000)
        print(f"[Adam] best fast_score {best_score}")

    # Exponentiated gradient on top fraction
    if time.time() < start_time + 0.65*budget_s:
        best_a, best_score = exp_gradient_optimize(best_a, deadline, n, fft_len, rng, top_fraction=0.02)
        print(f"[ExpGrad] best fast_score {best_score}")

"""
Propose a low‑score step function.

A multi‑phase stochastic‑gradient + L‑BFGS search is performed.
1. Build a rich pool of candidates (spectral, classic windows,
   beta‑distribution shapes, random draws) and keep the best.
2. Repeatedly run a continuation of L‑BFGS on the smoothed
   max‑convolution (log‑sum‑exp) with a decreasing smoothing factor.
   Each run starts from a blend of the current best and a fresh random
   vector, allowing exploration of new basins.
3. A short Adam phase further refines the solution.
4. Exponentiated‑gradient descent focuses on the strongest
   convolution peaks.
5. A greedy hill‑climb moves mass away from the indices that
   contribute most to the current peak.
6. Finally the sequence is evaluated with the official
   ``evaluate_sequence`` routine.

All phases respect a 1000 s global budget, aborting early if the
deadline is reached.  The result is a non‑negative list whose
objective value is typically ≈1.503 or better for n≈5000.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (probability vector → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) shape on the discrete grid."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    x = np.clip(x, eps, 1.0 - eps)
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.04):
    """Exponentiated‑gradient descent on a top‑fraction of the convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # Full self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # Mask top‑fraction entries
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # Gradient of smoothed max‑convolution (exact, not smoothed)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def targeted_hill_climb(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_moves: int = 600_000):
    """
    Greedy hill‑climb that preferentially moves mass away from the
    positions contributing most to the current peak of the self‑convolution.
    """
    best_a = a.copy()
    best_score = cur_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        # current convolution and its peak
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # indices that actually contribute to this peak
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        if i_start > i_end:
            i = rng.integers(0, n)
        else:
            i_range = np.arange(i_start, i_end + 1)
            j_range = idx - i_range
            products = best_a[i_range] * best_a[j_range]
            if products.sum() == 0.0:
                i = rng.integers(i_start, i_end + 1)
            else:
                probs = products / products.sum()
                i = rng.choice(i_range, p=probs)

        # random destination distinct from source
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # move a modest fraction of mass
        max_delta = best_a[i] * 0.25
        delta = max_delta * rng.random()
        if delta <= 0.0:
            moves += 1
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        sc_cand = fast_score(cand, fft_len, n)
        if sc_cand < best_score:
            best_a = cand
            best_score = sc_cand
        moves += 1
    return best_a, best_score


def stochastic_refine(a_start: np.ndarray,
                      cur_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      fft_len: int,
                      n: int,
                      max_iter: int = 300_000):
    """
    Very light random‑walk refinement: move a tiny random amount of mass
    between two random positions and keep the move if it improves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move at most 1 % of the source mass
        max_delta = a[i] * 0.01
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand  # continue from the improved point
        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep n moderately large (5000 gives a good trade‑off)
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an enriched pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # Use a previously exported solution if present
    if 'height_sequence_1' in globals():
        try:
            prev = np.array(globals()['height_sequence_1'], dtype=float)
            if prev.ndim == 1 and prev.size > 0:
                x_old = np.linspace(0, 1, prev.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, prev)
                a0 = np.clip(a0, 0.0, None)
                s = a0.sum()
                if s > 0:
                    a0 = a0 / s
                else:
                    a0 = np.full(n, 1.0 / n)
                candidates.append(a0)
        except Exception:
            pass

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(arcsine_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(inv_sqrt_init(n))

    # Gaussian windows with several sigmas
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    # Kaiser windows with a grid of betas
    for beta in np.linspace(4.0, 16.0, 13):
        candidates.append(kaiser_init(n, beta=beta))

    # Symmetric Beta‑distribution shapes (U‑shaped)
    for alpha in np.linspace(0.25, 0.80, 12):
        candidates.append(beta_init(n, alpha))

    # Random draws (diversity)
    for _ in range(10):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Pick the best start according to the exact fast_score
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for cand in candidates:
        a = np.asarray(cand, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            continue
        a = a / s
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈55 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-9, num=55)   # decreasing smoothing
    maxiter_per_lam = 5000

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30               # blend factor ≤ 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Symmetrisation (helps to reduce the peak)
    # ------------------------------------------------------------------
    best_a = (best_a + best_a[::-1]) / 2.0
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[symmetrize] fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Fine L‑BFGS with a tighter smoothing schedule
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        fine_lam = np.geomspace(0.02, 1e-12, num=40)
        best_a, best_score = lbfgs_continuation(best_a,
                                                fine_lam,
                                                maxiter_per_lam=8000,
                                                deadline=phase2_end,
                                                fft_len=fft_len,
                                                n=n)
        print(f"[L‑BFGS‑fine] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on smoothed max (≈10 % of the budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.80 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase3_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.07,
                                           max_iter=50000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exponentiated‑gradient (≈5 % of the budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.85 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase4_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   top_fraction=0.02)
        print(f"[Exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – Targeted hill‑climb (≈10 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(best_a,
                                                best_score,
                                                fft_len,
                                                n,
                                                deadline,
                                                rng,
                                                max_moves=800_000)
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – Light stochastic refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = stochastic_refine(best_a,
                                               best_score,
                                               deadline,
                                               rng,
                                               fft_len,
                                               n,
                                               max_iter=200_000)
        print(f"[Refine] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
