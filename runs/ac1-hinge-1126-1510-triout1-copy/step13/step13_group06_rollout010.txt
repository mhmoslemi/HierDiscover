```
def adam_max_optimize(a_start, deadline, n, fft_len, max_iters=20000, lr=0.07, beta1=0.9, beta2=0.999):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute conv
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        k = int(conv.argmax())
        # compute gradient w.r.t a of conv[k] (subgradient)
        shift = k - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g = np.zeros_like(best_a)
        g[mask] = 2 * best_a[shift[mask]]
        # chain rule to y
        mean_g = np.dot(best_a, g)
        grad_y = best_a * (g - mean_g)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_y
        v = beta2 * v + (1-beta2) * (grad_y**2)
        m_hat = m / (1-beta1**it)
        v_hat = v / (1-beta2**it)
        y -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # recompute a
        a = softmax(y)
        a = a / a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

The algorithm builds a large pool of diverse initial candidates (spectral,
window functions, random Dirichlet samples, etc.), picks the best,
then improves it by three stages:

1. Targeted greedy hill‑climbing that moves mass from indices that
   contribute most to the current maximal convolution entry.
2. Adam optimisation on the *exact* max‑convolution (sub‑gradient
   of the max) using momentum (Adam) on the soft‑max logits.
3. A final round of the same targeted hill‑climb for fine refinement.

All operations respect the constraints of ``evaluate_sequence`` (non‑negative,
finite, sum > 0).  The routine works on a probability simplex (sum = 1)
so the official scoring reduces to `2·n·max(conv)`.
"""
import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation for a probability vector (sum(a)=1)."""
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def dirichlet_init(n: int, concentration: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with common concentration parameter."""
    # Gamma(alpha,1) for each component
    g = rng.gamma(shape=concentration, scale=1.0, size=n)
    s = g.sum()
    if s <= 0:
        return np.full(n, 1.0 / n)
    return g / s


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def hill_climb_targeted(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_moves: int = 200_000,
                        delta_frac: float = 0.06,
                        top_frac: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that always moves mass from indices that
    contribute most to the current maximal convolution entry.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = start_score
    moves = 0
    while time.time() < deadline and moves < max_moves:
        # compute convolution and locate peak
        fa = rfft(best, fft_len)
        conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # contributions of each index to conv[k] = sum_i a_i a_{k-i}
        shift = k - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[shift[mask]]

        # pick source index from top‑fraction of contributions
        num_top = max(1, int(top_frac * n))
        top_idxs = np.argpartition(-contrib, num_top - 1)[:num_top]
        i = rng.choice(top_idxs)

        # pick a random destination index (different from i)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            moves += 1
            continue

        best[i] -= delta
        best[j] += delta

        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        moves += 1
    return best, best_score


def adam_max_optimize(a_start: np.ndarray,
                      deadline: float,
                      n: int,
                      fft_len: int,
                      max_iters: int = 30_000,
                      lr: float = 0.07,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact max‑convolution (sub‑gradient).
    Optimises the logits y such that a = softmax(y) stays on the simplex.
    """
    # initialise logits from the start probabilities
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    a = softmax(y)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        a = softmax(y)

        # convolution and location of the current maximum
        fa = rfft(a, fft_len)
        conv = irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # sub‑gradient of conv[k] w.r.t a: g_i = 2 * a_{k-i} (if in range)
        shift = k - np.arange(n)
        mask = (shift >= 0) & (shift < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[shift[mask]]

        # chain rule through softmax
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1

    return best_a, best_score

            best_score = sc_adam
            best_a = a_adam.copy()
            print(f"[Adam‑max] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final precise hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, sc_final = hill_climb_targeted(
            best_a, best_score,
            fft_len, n, deadline, rng,
            max_moves=300_000,
            delta_frac=0.04,
            top_frac=0.04
        )
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[final hill‑climb] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official scoring
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
