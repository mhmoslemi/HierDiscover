def targeted_transfer(a, n, fft_len, deadline, rng):
    a = a.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        idx = int(conv.argmax())
        # contributions vector
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = idx - i_vals
        contrib = a[i_vals] * a[j_vals]
        # probability distribution for source i
        total = contrib.sum()
        if total == 0:
            # choose random source
            src = rng.integers(i_start, i_end + 1)
        else:
            prob = contrib / total
            src = rng.choice(i_vals, p=prob)
        # choose destination j random not equal src
        dest = rng.integers(0, n)
        if dest == src:
            continue
        # transfer delta
        delta = a[src] * rng.random() * 0.05  # up to 5%
        # ensure dest does not exceed bounds (no upper bound)
        a[src] -= delta
        a[dest] += delta
        # enforce non-negativity
        if a[src] < 0:
            a[dest] += a[src]  # move back negative part
            a[src] = 0.0
        # compute score occasionally (to avoid many FFT each iteration)
        # maybe compute after each move.
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score

"""
Propose a low‑evaluation step function.

Algorithm
---------
1. Build a rich pool of candidates:
   - Existing solution (if any) interpolated to the working length.
   - Classical windows (hann, blackman, kaiser, …) and deterministic shapes.
   - Parametric “edge‑heavy’’ families a_i ∝ (i+1)^{-p}+(n-i)^{-p} and
     a_i ∝ ((i+1)(n-i))^{-p} scanned over many p values.
   - Randomised draws on the simplex.

2. Pick the best candidate according to the exact fast_score
   (2·n·max(self‑convolution) for a probability vector).

3. Refine with a long L‑BFGS run on the smooth log‑sum‑exp approximation
   of the max‑convolution.  The λ‑schedule is fine‑grained down to 1e‑10.

4. Perform several short L‑BFGS restarts from random mixtures of the
   current best and fresh random vectors – this helps escaping shallow
   basins.

5. Fine‑tune with Adam on the same smoothed objective (very small λ).

6. Final local search: repeatedly identify the current maximal
   convolution entry, pick a contributing index proportionally to its
   product a_i·a_{k‑i}, and transfer a small fraction of its mass to a
   random other index (always keeping the vector on the simplex and
   symmetrising it).  This directly attacks the exact max‑convolution.

7. Return the best vector found; the official ``evaluate_sequence``
   routine is called once at the end for the final score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (probability vectors → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t logits y
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.004, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # dirichlet
        concentration = rng.uniform(0.5, 3.0)
        a = rng.dirichlet(np.full(n, concentration))
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    """Inverse‑sqrt decay."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine density (clipped)."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def edge_power_init(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape a_i ∝ (i+1)^(-p) + (n-i)^(-p)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def inv_product_init(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape a_i ∝ ((i+1)*(n-i))^(-p)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** (-p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule once.
    Returns (best_a, best_score) obtained during the run.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 40000):
    """
    Adam optimiser on the smoothed max‑convolution.
    The smoothing factor λ = lam_factor·max(conv) is kept fixed and very small.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def targeted_transfer(a_start: np.ndarray,
                      n: int,
                      fft_len: int,
                      deadline: float,
                      rng: np.random.Generator,
                      delta_frac: float = 0.04,
                      symmetrize: bool = True):
    """
    Direct local search that attacks the exact maximal convolution entry.
    - Find the index k where the convolution is maximal.
    - Choose a contributing index i with probability proportional to a_i·a_{k‑i}.
    - Transfer a small fraction of a_i to a random destination.
    - Optionally enforce symmetry a = (a + a[::-1]) / 2 after each move.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        # compute full convolution once
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())
        max_val = conv[idx]

        # update best if improved
        cur_score = 2.0 * n * max_val  # exact objective (sum = 1)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # identify the range of i that actually participates in this shift
        i_low = max(0, idx - (n - 1))
        i_high = min(idx, n - 1)
        i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_vals = idx - i_vals
        contrib = a[i_vals] * a[j_vals]

        total = contrib.sum()
        if total <= 0:
            # fallback: pick a random source
            src = rng.integers(i_low, i_high + 1)
        else:
            prob = contrib / total
            src = rng.choice(i_vals, p=prob)

        # pick a destination distinct from src
        dest = rng.integers(0, n)
        if dest == src:
            continue

        # transfer a small fraction of the source mass
        delta = a[src] * rng.random() * delta_frac
        if delta <= 0:
            continue
        a[src] -= delta
        a[dest] += delta

        # maintain non‑negativity (numerical safety)
        if a[src] < 0:
            a[dest] += a[src]   # remove negative excess
            a[src] = 0.0

        if symmetrize:
            a = 0.5 * (a + a[::-1])  # keep a palindromic

        # renormalise occasionally to combat drift
        if rng.random() < 0.001:
            a = simplex_projection(a)

    # final check
    final_score = fast_score(a, fft_len, n)
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 5000 has proven effective and fast enough.
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build initial candidate pool
    # --------------------------------------------------------------
    candidates = []

    # 1) Interpolate a previously exported solution if available.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2) Classical deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=9.0))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # 3) Parametric edge‑heavy families (dense sweep over p)
    for p in np.linspace(0.05, 2.5, 48):
        candidates.append(edge_power_init(n, p))
        candidates.append(inv_product_init(n, p))

    # 4) Randomised families
    for _ in range(12):
        candidates.append(random_initial(n, rng))
    for _ in range(8):
        candidates.append(pow_decay_init(n, rng))

    # --------------------------------------------------------------
    # Pick the best start (exact fast_score)
    # --------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈35 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09,
        5e-10, 2e-10, 1e-10
    ]
    maxiter_per_lam = 4000

    best_a, best_score = lbfgs_refine(best_a,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 1b – short L‑BFGS restarts (exploration)
    # ------------------------------------------------------------------
    restarts = 6
    while time.time() < phase1_end and restarts > 0:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = simplex_projection(a_start)
        a_opt, sc_opt = lbfgs_refine(a_start,
                                     lam_schedule,
                                     maxiter_per_lam // 2,
                                     phase1_end,
                                     fft_len,
                                     n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best fast_score={best_score:.8f}")
        restarts -= 1

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.52 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.0005,
                                           lr=0.04,
                                           max_iter=80000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Targeted mass‑transfer local search (≈30 % of budget)
    # ------------------------------------------------------------------
    phase3_end = deadline
    if time.time() < phase3_end:
        best_a, best_score = targeted_transfer(best_a,
                                               n,
                                               fft_len,
                                               phase3_end,
                                               rng,
                                               delta_frac=0.04,
                                               symmetrize=True)
        print(f"[Targeted] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation with the official routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
