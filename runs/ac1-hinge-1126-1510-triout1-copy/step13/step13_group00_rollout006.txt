def frank_wolfe_optimize(a0, n, fft_len, num_iters, t0=0.5, t_decay=0.99):
    a = a0.copy()
    for it in range(num_iters):
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
        M = conv.max()
        # select top constraints
        idxs = np.argpartition(conv, -K)[-K:]  # largest K shifts
        # Build constraints
        # We'll use scipy.optimize.linprog with objective -sum(b) (since we want max sum)
        C = np.zeros((K, n))
        for row, k in enumerate(idxs):
            # For j in [0,n):
            # coefficient of b_j is a_{k-j} if 0 <= k-j < n else 0
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # i = k - j => j = k - i
            # i from lo to hi
            # for each i in [lo, hi], a_i coefficient for b_{k-i}
            # So we fill C[row, k - i] = a_i
            # vectorized:
            i_vals = np.arange(lo, hi+1)
            j_vals = k - i_vals
            C[row, j_vals] = a[i_vals]
        # solve LP
        # maximize sum(b) -> minimize -sum(b)
        res = linprog(c=-np.ones(n), A_ub=C, b_ub=np.full(K, M), bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        b = res.x
        if b.sum() == 0:
            break
        g = b / b.sum()
        # step size
        t = t0 * (t_decay**it)
        a = (1 - t) * a + t * g
        # optionally re-normalize sum=1
        a = a / a.sum()

        new_score = fast_score(a, fft_len, n)

"""
Propose a low‑score step‑function sequence.

Algorithm outline
-----------------
1. Build a diverse pool of initial candidates (deterministic windows and
   random Dirichlet draws) on the probability simplex (sum=1).

2. Run a Frank–Wolfe style improvement:
   * For the current vector a, compute the self‑convolution M = max(a*a).
   * Keep only the K largest convolution entries (the active constraints).
   * Solve a linear program
        maximise   Σ b_i
        subject   (a * b)[k] ≤ M  for the selected shifts k,
                   b_i ≥ 0,
     which finds a direction g = b/Σb that does not increase the current
     maximum.  A short line‑search over mixing coefficients t∈{0.1,…,1.0}
     updates a ← (1−t)a + t g.
   * Repeat while time permits (≈40 % of the total budget).

3. Refine the best vector with a smoothed L‑BFGS optimiser (log‑sum‑exp
   of the convolution) to push the max down further (≈30 % of budget).

4. Optional final hill‑climb (very cheap, ≈5 % of budget).

All phases respect the global 1000 s deadline.  The returned list of
floats is fed to the official ``evaluate_sequence`` which rescales
internally; the objective we minimise is the exact quantity
`2·n·max(conv)`, i.e. the evaluation score when the sum of the
coefficients equals √(2n).
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import coo_matrix

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'beta'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (used after the Frank‑Wolfe phase)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Frank–Wolfe style LP improvement
# ----------------------------------------------------------------------


def frank_wolfe_optimize(a_start: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_iters: int = 200,
                         K_active: int = 200):
    """
    Conditional‑gradient (Frank‑Wolfe) iteration that solves a linear
    program to obtain a feasible direction g which does not increase the
    current maximal convolution value M.  A tiny line‑search over a few
    mixing coefficients t is performed; the best improvement is kept.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # compute self‑convolution and current bound M
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # select K_active most critical shifts
        if K_active < conv.size:
            idxs = np.argpartition(conv, -K_active)[-K_active:]
        else:
            idxs = np.arange(conv.size)

        # build sparse constraint matrix C (size K_active × n)
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(idxs):
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_vals = np.arange(lo, hi + 1, dtype=int)
            j_vals = k - i_vals
            rows.extend([row_idx] * len(j_vals))
            cols.extend(j_vals.tolist())
            data.extend(a[i_vals].tolist())
        C = coo_matrix((data, (rows, cols)), shape=(len(idxs), n)).tocsr()

        # solve the LP: maximise Σ b_i   ⇔ minimise −Σ b_i
        res = linprog(c=-np.ones(n),
                      A_ub=C,
                      b_ub=np.full(len(idxs), M + 1e-12),
                      bounds=[(0, None)] * n,
                      method='highs')
        if not res.success:
            # LP failed – abort the FW loop
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break
        g = b / sum_b  # feasible direction, sum(g)=1, g≥0

        # line‑search over a small set of mixing coefficients
        improved = False
        for t in (0.1, 0.3, 0.5, 0.7, 0.9, 1.0):
            a_new = (1.0 - t) * a + t * g
            # slight renormalisation to guard against numerical drift
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s == 0:
                continue
            a_new = a_new / s
            sc = fast_score(a_new, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new  # accept this improvement and continue
                improved = True
                break
        if not improved:
            # no candidate improved – take a tiny step to keep moving
            a = (1.0 - 0.05) * a + 0.05 * g
            a = np.maximum(a, 0.0)
            a /= a.sum()

    return best_a, best_score

        # Phase 1 – Frank–Wolfe LP improvement (≈40 % of remaining budget)
        # --------------------------------------------------------------
        fw_deadline = time.time() + 0.40 * (deadline - time.time())
        best_a, best_score = frank_wolfe_optimize(
            best_a,
            n,
            fft_len,
            fw_deadline,
            rng,
            max_iters=200,
            K_active=300
        )
        print(f"[n={n}] after Frank‑Wolfe fast_score = {best_score:.8f}")

        # --------------------------------------------------------------
        # Phase 2 – Smoothed L‑BFGS refinement (≈30 % of remaining budget)
        # --------------------------------------------------------------
        if time.time() < deadline:
            lbfgs_deadline = time.time() + 0.30 * (deadline - time.time())
            # aggressive lam‑schedule (down to 1e‑7)
            lam_schedule = [
                0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,
                5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4,
                5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
                5e-7, 1e-7
            ]
            maxiter_per_lam = 800
            best_a, best_score = lbfgs_continuation(
                best_a,
                lam_schedule,
                maxiter_per_lam,
                lbfgs_deadline,
                fft_len,
                fft_len,
                n
            )
            print(f"[n={n}] after L‑BFGS fast_score = {best_score:.8f}")

        # --------------------------------------------------------------
        # Phase 3 – Very short greedy hill‑climb (≈5 % of remaining budget)
        # --------------------------------------------------------------
        if time.time() < deadline:
            hill_deadline = time.time() + 0.05 * (deadline - time.time())
            cur_a = best_a.copy()
            cur_score = best_score
            it = 0
            while time.time() < hill_deadline and it < 50000:
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    it += 1
                    continue
                delta = cur_a[i] * rng.random() * 0.03
                if delta <= 0:
                    it += 1
                    continue
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta
                s = cand.sum()
                if s == 0:
                    it += 1
                    continue
                cand = cand / s
                sc = fast_score(cand, fft_len, n)
                if sc < cur_score:
                    cur_a = cand
                    cur_score = sc
                it += 1
            best_a, best_score = cur_a, cur_score
            print(f"[n={n}] after hill‑climb fast_score = {best_score:.8f}")

        # --------------------------------------------------------------
        # Final evaluation with the official scorer.
        # --------------------------------------------------------------
        final_seq = best_a.tolist()
        final_eval = evaluate_sequence(final_seq)
        print(f"[n={n}] final evaluated score = {final_eval:.8f}")

        if final_eval < best_overall_score:
            best_overall_score = final_eval
            best_overall_seq = final_seq

    # Return the best sequence discovered.
    return best_overall_seq
