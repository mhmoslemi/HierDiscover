    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_val = conv.max()
        k = conv.argmax()
        # compute partner array: p = np.zeros(n)
        # For i in 0..n-1:
        #   idx = k - i
        #   p[i] = a[idx] if 0 <= idx < n else 0.0
        # compute product contributions: prod = a * p
        # find source i_max where prod is largest.
        prod = a * np.where((k - np.arange(n) >= 0) & (k - np.arange(n) < n), a[k - np.arange(n)], 0.0)
        i_src = np.argmax(prod)

        p_vals = np.where((k - np.arange(n) >= 0) & (k - np.arange(n) < n), a[k - np.arange(n)], 0.0)

        # optionally ensure i_src != i_dest.

        # Determine epsilon to move: maybe a fraction of a[i_src]

def targeted_hillclimb(a, deadline, n, fft_len, rng, max_iter=500000):
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute conv and find max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        max_val = conv.max()
        k = int(conv.argmax())
        # compute partner array p
        idx_arr = np.arange(n)
        partner_idx = k - idx_arr
        mask = (partner_idx >= 0) & (partner_idx < n)
        p = np.zeros_like(a)
        p[mask] = a[partner_idx[mask]]
        # product contributions
        prod = a * p
        # find source i_src where prod maximal
        i_src = int(np.argmax(prod))
        # find destination i_dest where p minimal (i.e., partner small)
        # choose among all indices where partner is defined; but if it's out of bounds partner=0 (mask False).
        # For out-of-bounds, p is 0.
        # So dest should be index where p is minimal (i.e., 0).
        # We'll pick any index where p==0 and a not too large maybe.
        zero_mask = p == 0
        if np.any(zero_mask):
            # pick dest with minimal a among zero partners to avoid moving all mass to one spot making peak elsewhere
            i_dest = int(np.argmin(np.where(zero_mask, a, np.inf)))
        else:
            i_dest = int(np.argmin(p))  # fallback
        # ensure i_src != i_dest
        if i_src == i_dest:
            # choose random other dest
            i_dest = (i_src + 1) % n
        # move epsilon
        eps_frac = min(0.1, rng.random() * 0.2)  # between 0 and 0.2
        eps = a[i_src] * eps_frac
        if eps <= 0:
            it += 1
            continue
        a_candidate = a.copy()
        a_candidate[i_src] -= eps
        a_candidate[i_dest] += eps
        # compute new score quickly
        sc_candidate = fast_score(a_candidate, fft_len, n)
        # objective is the max (fast_score). Accept only if better
        if sc_candidate < best_score:
            a = a_candidate
            best_score = sc_candidate
            best_a = a_candidate.copy()
        # else maybe try smaller epsilon
        else:
            # try reduce epsilon via binary search up to 5 attempts
            for _ in range(3):
                eps *= 0.5
                if eps < 1e-12:
                    break
                a_candidate = a.copy()
                a_candidate[i_src] -= eps
                a_candidate[i_dest] += eps
                sc_candidate = fast_score(a_candidate, fft_len, n)
                if sc_candidate < best_score:
                    a = a_candidate
                    best_score = sc_candidate
                    best_a = a_candidate.copy()
                    break
        it += 1
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) whose self‑convolution maximum
and therefore the evaluation score 2·n·max(a*a)/(sum a)² is as small as possible.

The search consists of several stages:

1.  Build a diverse pool of initial candidates (deterministic windows,
    random vectors, beta‑distributions with extreme parameters, etc.).
2.  Pick the best candidate (fast_score = 2·n·max(a*a) for sum a=1).
3.  Refine the candidate with a long L‑BFGS continuation on a smooth
    approximation of the max‑convolution (log‑sum‑exp smoothing) and a very
    fine λ‑schedule.
4.  Further improve with Adam optimisation on the same smooth objective.
5.  Perform a deterministic “targeted hill‑climb” that repeatedly moves a
    small amount of mass from the index participating most in the current
    maximal convolution entry to an index that does not participate,
    accepting only strictly improving moves.
6.  Return the best sequence found (as a plain list[float]).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast score (exact when sum(a)=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a probability vector."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (used by L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (objective, gradient) for a smooth approximation of max_i (a*a)_i
    where a = softmax(y).  The smoothing uses a log‑sum‑exp with a
    λ proportional to the current max value.
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient d smooth_max / d conv_i
    w = exp_shift / Z                     # shape (2n‑1,)

    # Back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # Extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initial families (diverse starting points)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'rand', 'exp', 'pow', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'rand':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'pow':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:edge] = heavy
        a[-edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS through the whole λ‑schedule."""
    y = np.log(a_start + 1e-12)      # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  max_iters: int = 25000,
                  lr: float = 0.07,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995):
    """Adam on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def targeted_hillclimb(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       max_iter: int = 600000):
    """
    Deterministic greedy hill‑climb that always moves mass from the index
    contributing most to the current maximal convolution entry towards an index
    that does not participate in that entry.  Only strictly improving moves
    are kept (with a tiny back‑off to try a smaller step if a large step fails).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current convolution and its worst index
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        k = int(conv.argmax())

        # partner values for this k (zero where undefined)
        idx = np.arange(n)
        partner_idx = k - idx
        valid = (partner_idx >= 0) & (partner_idx < n)
        partner = np.zeros_like(a)
        partner[valid] = a[partner_idx[valid]]

        # contribution of each coefficient to the maximal entry
        contrib = a * partner
        i_src = int(np.argmax(contrib))

        # destination: look for an index with zero partner (i.e. does not affect this entry)
        zero_mask = (partner == 0)
        if zero_mask.any():
            # among zero‑partner indices, pick the one with smallest current mass
            candidates = np.where(zero_mask)[0]
            i_dest = int(candidates[np.argmin(a[candidates])])
        else:
            # fall back to the minimal partner value
            i_dest = int(np.argmin(partner))

        if i_src == i_dest:
            i_dest = (i_src + 1) % n

        # try a modest step; we will halve it if the move is not improving
        eps = a[i_src] * rng.random() * 0.05  # up to 5 % of source weight
        if eps <= 0.0:
            continue

        # candidate move
        a_cand = a.copy()
        a_cand[i_src] -= eps
        a_cand[i_dest] += eps
        if a_cand[i_src] < 0.0:
            continue

        sc_cand = fast_score(a_cand, fft_len, n)
        if sc_cand < best_score:
            a = a_cand
            best_score = sc_cand
            best_a = a_cand.copy()
            continue

        # if not improving, try a smaller epsilon (up to three halvings)
        for _ in range(3):
            eps *= 0.5
            if eps < 1e-12:
                break
            a_cand = a.copy()
            a_cand[i_src] -= eps
            a_cand[i_dest] += eps
            if a_cand[i_src] < 0.0:
                continue
            sc_cand = fast_score(a_cand, fft_len, n)
            if sc_cand < best_score:
                a = a_cand
                best_score = sc_cand
                best_a = a_cand.copy()
                break

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # Problem size
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of candidates
    # ------------------------------------------------------------------
    candidates = []

    # reuse previous construction if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))
    candidates.append(kaiser_init(n, beta=rng.uniform(4.0, 12.0)))

    # random families
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # beta families – moderate parameters
    for _ in range(200):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))

    # beta families – extreme log‑uniform parameters
    for _ in range(100):
        log_alpha = rng.uniform(-5, 1)   # 10⁻⁵ … 10¹
        log_beta = rng.uniform(-5, 1)
        alpha = 10.0 ** log_alpha
        beta_param = 10.0 ** log_beta
        candidates.append(beta_init(n, alpha, beta_param))

    # raised‑cosine family
    for _ in range(120):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    # ------------------------------------------------------------------
    # Pick the best starting candidate
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS continuation
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09,
        1e-09, 5e-10, 2e-10, 1e-10, 5e-11,
        2e-11, 1e-11, 5e-12, 2e-12, 1e-12,
        5e-13, 2e-13, 1e-13, 5e-14, 2e-14,
        1e-14, 5e-15, 2e-15, 1e-15
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # diversify with a small random blend
        rand_vec = random_initial(n, rng)
        a_start = 0.85 * best_a + 0.15 * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (longer run)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.65 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, n, fft_len,
            max_iters=25000, lr=0.07, lam_init=0.30, lam_decay=0.9995
        )
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic targeted hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hillclimb(
            best_a, deadline, n, fft_len, rng, max_iter=600000
        )
        print(f"[targeted hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation (official scoring)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
