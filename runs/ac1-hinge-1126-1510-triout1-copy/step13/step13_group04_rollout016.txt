"""
Improved search for a low‑evaluation step function.

The algorithm follows three stages:
1. Build a diverse pool of initial candidates and keep the best
   (fast_score = 2·n·max(self‑convolution) for a probability vector).
2. Run a long L‑BFGS optimisation on a smoothed max‑convolution
   (log‑sum‑exp) with a decreasing smoothing schedule.
3. Perform a cheap, exact “max‑focused” hill‑climb that updates the
   convolution incrementally (no FFT per move).  Each move transfers a
   small amount of mass from the index that contributes most to the
   current maximal convolution entry to the index with smallest weight.
   The move is kept only if the global maximum strictly decreases.
The final vector is evaluated with the official ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact fast score (sum = 1 → objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Objective for a probability vector (exact, no smoothing)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Log‑sum‑exp smooth approximation of max(self‑conv)."""
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families (all return a probability vector of length n)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Simple random families on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS through the whole λ schedule once.
    Returns (best_a, best_score) obtained during the run.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            # In case something goes wrong with the optimiser
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑focused incremental hill‑climb (no FFT per move)
# ----------------------------------------------------------------------


def max_focused_descent(a: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Greedy hill‑climb that always reduces the global max of the
    self‑convolution.  Updates are performed incrementally without
    recomputing the full convolution.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)            # length 2·n‑1
    best_max = conv.max()
    moves = 0

    while time.time() < deadline:
        # index where the maximum occurs
        k = int(conv.argmax())

        # i‑range that participates in conv[k]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range

        # contribution of each i to the current maximal entry
        prod = a[i_range] * a[j_range]
        if prod.size == 0:
            break
        i_local = int(prod.argmax())
        i = int(i_range[i_local])

        # pick a destination index with minimal mass
        j = int(np.argmin(a))
        if i == j:
            # second smallest if they coincide
            sorted_idx = np.argsort(a)
            j = int(sorted_idx[0] if sorted_idx[0] != i else sorted_idx[1])

        if a[i] <= 0.0:
            continue

        # random fraction of the source mass (2 %‑20 %)
        frac = rng.random() * 0.18 + 0.02
        delta = a[i] * frac

        # keep copies of everything that will be overwritten
        slice_i = slice(i, i + n)
        slice_j = slice(j, j + n)
        orig_i = conv[slice_i].copy()
        orig_j = conv[slice_j].copy()
        orig_2i = conv[2 * i]
        orig_2j = conv[2 * j]
        orig_ij = conv[i + j]

        # apply exact update (see derivation in analysis)
        conv[slice_i] -= 2.0 * delta * a
        conv[slice_j] += 2.0 * delta * a
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        new_max = conv.max()
        if new_max < best_max - 1e-15:
            # accept the move
            a[i] -= delta
            a[j] += delta
            best_max = new_max
        else:
            # reject – restore everything
            conv[slice_i] = orig_i
            conv[slice_j] = orig_j
            conv[2 * i] = orig_2i
            conv[2 * j] = orig_2j
            conv[i + j] = orig_ij

        moves += 1

    return a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain ``list[float]``; the official ``evaluate_sequence`` will
    be called on the returned list.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – 5000 has proven both effective and fast.
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build initial candidate pool
    # --------------------------------------------------------------
    candidates = []

    # Use a previous solution if provided
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=9.0))

    # Randomised families
    for _ in range(6):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        candidates.append(kaiser_init(n, beta=rng.uniform(5.0, 12.0)))

    # Choose the best start according to the exact fast_score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.0015, 0.0008, 0.0004, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06
    ]
    maxiter_per_lam = 2500

    best_a, best_score = lbfgs_refine(best_a,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      fft_len,
                                      n)
    print(f"[L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Max‑focused incremental descent (use the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, max_conv = max_focused_descent(best_a, deadline, rng)
        best_score = 2.0 * n * max_conv
        print(f"[max‑focused descent] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
