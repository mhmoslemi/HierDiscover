"""
Propose a low‑score non‑negative step function.

Algorithm Overview
------------------
1. **Problem size** – use a relatively large power‑of‑two length (8192)
   that gives the optimiser many degrees of freedom while still being
   cheap enough for many FFT‑based evaluations.

2. **Initial pool** – a rich set of deterministic and stochastic families
   (window functions, power‑law, exponential, Beta‑distribution, random
   Dirichlet‑like draws, mixtures of Gaussians).  All vectors are projected
   onto the simplex (sum = 1) and the best one w.r.t. the exact objective
   `fast_score` is kept.

3. **Phase 1 – Smoothed L‑BFGS** – repeatedly run a continuation of
   L‑BFGS on a log‑sum‑exp smoothed version of the max‑convolution.
   Each run starts from the current best mixed with a fresh random
   vector, and a schedule of decreasing smoothing parameters drives the
   optimizer toward the true nonsmooth objective.

4. **Phase 2 – Exact sub‑gradient descent** – exponentiated‑gradient
   updates on the simplex using the true sub‑gradient of the max‑convolution.

5. **Phase 3 – Adam on the exact max** – Adam updates on the logits
   (softmax parametrisation) for faster convergence.

6. **Phase 4 – Simulated annealing** – cheap pairwise mass‑transfer moves
   with a geometric temperature schedule for fine‑tuning.

7. **Phase 5 – Greedy hill‑climb** – deterministic tiny transfers that
   accept only improving moves; the step size shrinks with the iteration
   count.

8. **Final polishing** – a short run of the sub‑gradient optimiser with a
   very small learning‑rate to clean up any remaining microscopic
   imperfections.

All phases respect the global 1000 s budget.  The best sequence found is
returned; the external `evaluate_sequence` will rescale it internally,
so the optimiser works on the probability simplex (sum = 1).

The implementation is self‑contained, uses only NumPy and SciPy and
contains no filesystem or network access.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the simplex (sum = 1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.full_like(a, 1.0 / a.size)
    return a / s


# ----------------------------------------------------------------------
# Exact fast objective for vectors on the simplex (sum == 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a⋆a) for a on the simplex (sum(a)=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective f = 2·n·smooth_max(conv) and its gradient w.r.t. y.
    y are logits; a = softmax(y) lives on the simplex.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c          # ≤0
    exp_shift = np.exp(shifted / lam)    # ∈(0,1]
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max wrt convolution entries
    w = exp_shift / Z                     # shape (2n‑1,)

    # back‑prop through convolution: d conv / d a = rev(a) * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]   # gradient wrt a (size n)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Spectral (square‑root of flat spectrum) – smoothly decaying shape."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    u = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = u
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    return normalize(a)


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return normalize(a)


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    return normalize(a)


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return normalize(a)


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    return normalize(a)

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random vector on the simplex using various heuristics."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge',
                       'bimodal', 'dirichlet', 'lognorm'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'dirichlet':
        alpha = rng.uniform(0.2, 2.0, size=n)
        a = rng.gamma(alpha)
    else:  # lognorm
        sigma = rng.uniform(0.3, 1.0)
        a = rng.lognormal(mean=0.0, sigma=sigma, size=n)
    return normalize(a)


def mixture_gaussian_init(n: int, rng: np.random.Generator, K: int = 3) -> np.ndarray:
    """Mixture of K Gaussians with random means, sigmas and weights."""
    i = np.arange(n, dtype=np.float64)
    weights = rng.random(K)
    weights /= weights.sum()
    mus = rng.uniform(0, n - 1, size=K)
    sigmas = rng.uniform(0.05 * n, 0.25 * n, size=K)
    a = np.zeros(n, dtype=np.float64)
    for w, mu, sigma in zip(weights, mus, sigmas):
        a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return normalize(a)


def dirichlet_sparse_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with small alphas → sparse vectors."""
    alpha = rng.uniform(0.01, 0.2, size=n)
    a = rng.gamma(alpha)
    return normalize(a)


def powerlaw_init(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (monotonically decreasing)."""
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    return normalize(a)


def exponential_init(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    return normalize(a)


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=float) + 0.5) / n  # mid‑point rule
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.maximum(a, 0.0)
    return normalize(a)

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in ``lam_schedule``; each stage starts from the
    previous result.  Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits on the simplex
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")
    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         eps: float = 1e-12):
    """
    Exact sub‑gradient descent on 2·n·max(conv).  Uses multiplicative
    updates on the simplex (exponentiated‑gradient style).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_c = conv.max()
        mask = (conv >= max_c - eps).astype(np.float64)   # tight constraints

        # gradient of max (sub‑gradient) w.r.t. a
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        # step size – classic 1/√t schedule
        eta = 1.0 / np.sqrt(it + 1.0)

        # EG update and re‑normalise
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0:
            a = best_a.copy()
        else:
            a /= s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[Subgrad] it={it}, best fast_score={best_sc:.9f}")

        it += 1
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective (2·n·max).
    Operates on logits y (a = softmax(y)) so that the simplex constraint
    holds automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution a*a and its max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-12).astype(np.float64)

        # sub‑gradient of the max entry
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a          # ∂ (2·n·max) / ∂ a

        # back‑prop through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)          # decaying learning rate
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter
        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()
            print(f"[Adam] iter={t}, best fast_score={best_sc:.9f}")

    return best_a, best_sc


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-7,
                        max_moves: int = 200000):
    """
    Simple pairwise mass‑transfer annealing.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    move = 0
    while time.time() < deadline and move < max_moves:
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            move += 1
            continue

        delta = a[i] * rng.random() * 0.16
        if delta <= 0:
            move += 1
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc_cand = fast_score(cand, fft_len, n)

        if sc_cand < best_sc:
            accept = True
        else:
            prob = np.exp(-(sc_cand - best_sc) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = cand
            if sc_cand < best_sc:
                best_sc = sc_cand
                best_a = cand.copy()
        move += 1

    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 100000,
               delta_frac: float = 0.04):
    """
    Strictly improving greedy hill‑climb with tiny pairwise transfers.
    """
    best = a.copy()
    best_sc = cur_sc
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

        0.5, 0.4, 0.3, 0.2, 0.15, 0.1,

        0.006, 0.004, 0.002, 0.001, 5e-4,
        2e-4, 1e-4, 5e-5

        # blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        if rng.random() < 0.25:
            rand_vec = mixture_gaussian_init(n, rng, K=rng.integers(2, 4))
        alpha = rng.random() * 0.35
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start = normalize(a_start)

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS‑global] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent (≈20 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = time.time() + 0.20 * budget_s
        print("[Subgrad] start")
        best_a, best_sc = subgradient_optimize(best_a,
                                                subgrad_deadline,
                                                n,
                                                fft_len,
                                                eps=1e-12)
        print(f"[Subgrad] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on the exact max‑convolution (≈25 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing fine‑tuning (≈15 % of time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sa_deadline = time.time() + min(20.0, deadline - time.time())
        print("[SA] start")
        best_a, best_sc = simulated_annealing(best_a,
                                              sa_deadline,
                                              n,
                                              fft_len,
                                              rng,
                                              init_temp=5e-4,
                                              final_temp=1e-7,
                                              max_moves=250000)
        print(f"[SA] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – Greedy hill‑climb polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 30.0)
        print("[Hill‑climb] start")
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=150000,
                                     delta_frac=0.04)
        print(f"[Hill‑climb] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Final ultra‑fine EG pass (tiny learning‑rate)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[Final‑EG] ultra‑fine tuning")
        best_a, best_sc = subgradient_optimize(best_a,
                                                final_deadline,
                                                n,
                                                fft_len,
                                                eps=1e-14)
        print(f"[Final‑EG] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Return the result – the official evaluator rescales internally.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
