def lp_step(a_cur, fft_len, n, active_frac):
    conv = compute_conv(a_cur, fft_len)
    M = conv.max()
    # choose active indices:
    num_active = max(5, int(active_frac * conv.size))
    top_idx = np.argpartition(conv, -num_active)[-num_active:]
    # ensure include max
    top_idx = np.unique(np.concatenate([top_idx, [conv.argmax()]]))
    # build LP
    num_constraints = top_idx.size
    A_ub = np.zeros((num_constraints, n+1))
    b_ub = np.zeros(num_constraints)
    for row_idx, k in enumerate(top_idx):
        start_i = max(0, k-(n-1))
        end_i = min(k, n-1)
        i_range = np.arange(start_i, end_i+1)
        A_ub[row_idx, start_i:end_i+1] = a_cur[k - i_range]
        A_ub[row_idx, -1] = -0.5
        b_ub[row_idx] = 0.5 * conv[k]
    # equality constraint sum =1
    A_eq = np.zeros((1, n+1))
    A_eq[0,:n] = 1.0
    b_eq = np.array([1.0])
    c = np.zeros(n+1)
    c[-1] = 1.0  # minimize M
    bounds = [(0, None)]*(n+1)
    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, None
    a_new = res.x[:n]
    M_new = res.x[-1]  # may be lower bound
    # Ensure non-negative and sum close to 1
    a_new[a_new < 0] = 0.0
    a_new /= a_new.sum()
    return a_new, M_new

"""
Iterative Linear‑Programming Refinement for Minimising the Max Self‑Convolution.

This solver works on the equivalent problem of normalising the step‑function
so that sum(a)=1.  The target objective is then 2·n·max(a*a).  We start from a
hand‑crafted set of initial vectors (uniform, spectral, Beta, etc.) and keep the
best one.  The core optimisation consists of repeatedly:

1. Computing the self‑convolution and identifying the indices that contribute
   most to the current maximum (a small “active set”, e.g. the top 1‑2 % of
   entries).

2. Building a linear programme that linearises each active quadratic constraint
   aᵀQₖa≤M around the current iterate and solves for a new candidate a and a
   new bound M.  The LP variables are the n probabilities aᵢ and the scalar
   M.  The constraints have the form

        Σᵢ a₀ₖ₋ᵢ·aᵢ – ½·M ≤ ½·(a₀*a₀)ₖ

   where a₀ is the current iterate.

3. Optionally mixing the LP solution with the current iterate (convex combination)
   to guarantee a decrease of the true (non‑linear) objective.

The loop stops when the time budget is exhausted or when no further improvement
is seen.  A final greedy hill‑climb (mass‑transfer moves) polishes the result.
The returned list of floats is a feasible step‑function and the official
`evaluate_sequence` is called at the end for the final score.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def compute_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Exact self‑convolution via FFT (full length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.size - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv) for a probability vector (sum(a)=1)."""
    conv = compute_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """A deterministic spectral construction (square‑root of flat spectrum)."""
    L = next_pow2(2 * n - 1)
    target = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    padded = np.zeros(L)
    padded[: 2 * n - 1] = target
    S = np.fft.fft(padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(amp)
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Randomly chosen parametric families."""
    mode = rng.choice(['uniform', 'rand', 'exp', 'pow', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'rand':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'pow':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:edge] = heavy
        a[-edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def beta_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Beta distribution with random shape parameters."""
    alpha = rng.exponential(scale=0.8) + 0.1
    beta = rng.exponential(scale=0.8) + 0.1
    i = np.arange(1, n + 1, dtype=float)
    a = (i / n) ** (alpha - 1) * (1 - i / n) ** (beta - 1)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def hann_init(n: int) -> np.ndarray:
    """Hann (raised cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑Programming refinement
# ----------------------------------------------------------------------


def lp_refine(a_cur: np.ndarray,
              conv_cur: np.ndarray,
              fft_len: int,
              n: int,
              active_frac: float = 0.02):
    """
    One LP refinement step based on a linearisation of the active
    quadratic constraints.  Returns (a_new, M_new) where M_new is the
    bound variable from the LP (not the true max).  If the LP fails,
    returns (None, None).
    """
    # active indices = top entries of conv_cur
    num_active = max(5, int(active_frac * conv_cur.size))
    top_idxs = np.argpartition(conv_cur, -num_active)[-num_active:]
    # definitely keep the global maximum index
    top_idxs = np.unique(np.concatenate([top_idxs, [int(conv_cur.argmax())]]))

    m = top_idxs.size
    # Build LP matrices
    A_ub = np.zeros((m, n + 1), dtype=np.float64)
    b_ub = np.zeros(m, dtype=np.float64)

    for row, k in enumerate(top_idxs):
        # contribution vector v_i = a_cur[k - i] where defined
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        # a_cur indexed at (k - i)
        A_ub[row, i_start:i_end + 1] = a_cur[k - i_range]
        # coefficient for M (variable at position n)
        A_ub[row, n] = -0.5
        # RHS = ½·conv_k
        b_ub[row] = 0.5 * conv_cur[k]

    # Equality: sum(a)=1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # Objective: minimise M (last variable)
    c = np.zeros(n + 1, dtype=np.float64)
    c[n] = 1.0

    bounds = [(0.0, None)] * (n + 1)   # a_i≥0, M≥0

    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})

    if not res.success:
        return None, None

    a_new = res.x[:n]
    # Numerical cleanup
    a_new[a_new < 0] = 0.0
    sum_a = a_new.sum()
    if sum_a <= 0:
        return None, None
    a_new /= sum_a   # re‑normalise
    M_new = res.x[n]
    return a_new, M_new


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (strictly improving)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.06):
    """Greedy hill‑climb on the exact fast_score."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42,

    rng = np.random.default_rng(seed)

    candidates = []

    # deterministic families

    candidates.append(hann_init(n))

    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Phase 1 – LP refinement (core of this solver)

    # Parameters controlling the LP active set
    active_frac = 0.03   # start relatively generous

        else:

            for alpha in (0.5, 0.3, 0.1):

                a_mix /= a_mix.sum()

    return final_seq
