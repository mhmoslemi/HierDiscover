        # compute conv via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
        max_idx = int(np.argmax(conv))

        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)

        i_vals = np.arange(i_start, i_end+1)

        idx = np.argmax(contributions)

"""
Improved search for the min‑max autocorrelation step‑function problem.

Main ideas
-----------
* Use several deterministic + random initialisations.
* Run a multi‑stage L‑BFGS continuation on the log‑sum‑exp smooth max.
  The λ‑schedule is longer (down to 1e‑7) and each λ receives a decent
  number of iterations.
* Perform a *targeted* hill‑climb that repeatedly attacks the lag where
  the convolution is maximal, moving a small fraction of mass away from
  the pair that contributes most to that lag.
* Occasionally prune to a small support, re‑normalise and run a short
  L‑BFGS refinement (helps to escape flat plateaus).
* All work respects the 1000 s time budget; the best sequence found is
  finally evaluated with the official `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Exact (fast) objective – used for comparison and final scoring
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum ≈ 1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Log‑sum‑exp smooth approximation of max convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                               # ∂smooth_max/∂conv_i

    # back‑prop through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Seed generated via inverse FFT of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'sinusoid'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sinusoid
        i = np.arange(n, dtype=np.float64)
        a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        a = a / a.sum()
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS continuation (multi‑λ)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)                     # logits

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            # any numerical trouble is ignored – we keep the current best
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Targeted hill‑climb that attacks the worst lag directly
# ----------------------------------------------------------------------


def targeted_hill_climb(a: np.ndarray,
                        n: int,
                        fft_len: int,
                        deadline: float,
                        rng: np.random.Generator,
                        step_frac_initial: float = 0.02):
    """
    Greedy refinement: locate the lag with maximal autocorrelation,
    pick the pair (i, j) contributing the largest product a_i*a_j,
    and move a small fraction of the larger mass to a random location.
    Repeats until time runs out.
    """
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    step_frac = step_frac_initial

    while time.time() < deadline:
        # current convolution
        fa = np.fft.rfft(best_a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # lag where the max occurs
        lag = int(np.argmax(conv))

        # indices that overlap for this lag
        i_start = max(0, lag - (n - 1))
        i_end = min(n - 1, lag)
        i_vals = np.arange(i_start, i_end + 1)
        j_vals = lag - i_vals

        # contribution of each overlapping pair
        contrib = best_a[i_vals] * best_a[j_vals]
        max_pair_idx = int(np.argmax(contrib))
        i_max = i_vals[max_pair_idx]
        j_max = j_vals[max_pair_idx]

        # choose the larger of the two as the donor
        src = i_max if best_a[i_max] >= best_a[j_max] else j_max
        dst = rng.integers(0, n)
        if dst == src:
            continue

        delta = best_a[src] * step_frac * rng.random()
        if delta <= 0:
            continue

        cand = best_a.copy()
        cand[src] -= delta
        cand[dst] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < best_sc - 1e-12:
            # accept improvement
            best_a = cand
            best_sc = sc
            # slowly shrink the step size to allow fine tuning
            step_frac = max(step_frac * 0.95, 1e-5)
        else:
            # occasional stochastic acceptance to escape plateaus
            if rng.random() < 1e-3:
                best_a = cand

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we use a safe power‑of‑two length 4096
    # ------------------------------------------------------------------
    n = 4096
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse initial pool (including possible previous result)
    # ------------------------------------------------------------------
    candidates = []

    if 'height_sequence_1' in globals():

        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # randomised families
    for _ in range(10):
        candidates.append(random_initial(n, rng))

    # pick the best start according to the cheap fast_score
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # L‑BFGS continuation – long λ schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01,

        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,

    ]
    maxiter_per_lam = 1500

        a_start /= a_start.sum()

                                           maxiter_per_lam,

                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Targeted hill‑climb refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[targeted hill‑climb] start")
        best_a, best_sc = targeted_hill_climb(best_a, n, fft_len,
                                             deadline, rng)
        print(f"[targeted] best fast_score={best_sc:.9f}")

    # ------------------------------------------------------------------
    # Optional support‑pruning + short L‑BFGS refinements
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:

        for frac in (0.02, 0.05, 0.1, 0.2):

            mask[idx] = 1.0

            total = a_pruned.sum()

    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.9f}")
    return final_seq
