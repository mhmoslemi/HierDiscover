"""
Propose a non‑negative step sequence with a low evaluate_sequence score.

The search combines many diverse initialisations, several gradient‑based
optimisers (L‑BFGS on a smooth log‑sum‑exp surrogate, Adam,
exponentiated‑gradient, projected sub‑gradient) and a long simulated‑annealing
phase.  After the stochastic phase a final L‑BFGS refinement and a greedy
hill‑climb are performed.  New “spiky’’ families are added to the pool of
candidates, and the final L‑BFGS run uses a finer λ‑schedule and more
iterations per λ, giving a higher‑precision polishing step.  The whole
procedure respects the constraints of evaluate_sequence (non‑negative,
finite, sum>0) and runs within the supplied time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact when sum(a)=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a vector on the simplex."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth approximation of 2·n·max(a*a) with gradient.
    The variable y are logits for the soft‑max; a = softmax(y).
    """
    a = softmax(y)

    # self‑convolution via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'rand', 'exp', 'pow', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'rand':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'pow':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:edge] = heavy
        a[-edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def spike_init(n: int, rng: np.random.Generator, k: int = 10) -> np.ndarray:
    """Create a sparse ‘spiky’ distribution: k random positions get larger mass."""
    a = np.full(n, 1e-12)
    idx = rng.choice(n, size=min(k, n), replace=False)
    # random weights in [0.5,2.0] before normalisation
    a[idx] = rng.uniform(0.5, 2.0, size=idx.size)
    a /= a.sum()
    return a


def bimodal_spike_init(n: int, rng: np.random.Generator, k: int = 5) -> np.ndarray:
    """Two symmetric clusters of spikes."""
    a = np.full(n, 1e-12)
    centre = n // 2
    width = max(1, n // 100)  # distance from centre
    left_mid = centre - width
    right_mid = centre + width
    left_idx = rng.choice(left_mid, size=min(k, left_mid), replace=False)
    right_idx = (right_mid + rng.choice(n - right_mid, size=min(k, n - right_mid), replace=False))
    a[left_idx] = rng.uniform(0.5, 2.0, size=left_idx.size)
    a[right_idx] = rng.uniform(0.5, 2.0, size=right_idx.size)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """Run L‑BFGS through a whole λ‑schedule."""
    y = np.log(a_start + 1e-12)      # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:   # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  max_iters: int = 15000,
                  lr: float = 0.07,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9993):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          top_fraction: float = 0.02):
    """Exponentiated‑gradient on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.30):
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv[idx] w.r.t. a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # zero‑sum projection (stay on simplex)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_temp: float = 5e-4,
                        final_temp: float = 1e-9,
                        max_moves: int = 300000):
    """Simple SA with many mass‑transfer moves."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for move in range(max_moves):
        if time.time() >= deadline:
            break
        t_frac = move / max_moves
        T = init_temp * (final_temp / init_temp) ** t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction of mass from i to j
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        if sc_candidate < best_score or rng.random() < np.exp(-(sc_candidate - best_score) / max(T, 1e-12)):
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.06):
    """Greedy hill‑climb (strictly improving)."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list[float] found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of candidates (including some spiky families)
    # --------------------------------------------------------------
    candidates = []

    # reuse a previously stored solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # linear interpolation to new size
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # random families
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    for _ in range(4):
        candidates.append(kaiser_init(n, beta=rng.uniform(4.0, 12.0)))

    # beta / raised‑cosine systematic search
    for _ in range(200):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))
    for _ in range(100):
        p = rng.uniform(0.5, 3.0)
        candidates.append(raised_cosine_init(n, p))

    # new spiky families
    for _ in range(6):
        candidates.append(spike_init(n, rng, k=rng.integers(5, 15)))
        candidates.append(bimodal_spike_init(n, rng, k=rng.integers(3, 10)))

    # --------------------------------------------------------------
    # Pick the best starting candidate
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensified L‑BFGS (≈30 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    # Very fine λ‑schedule extending to 1e‑12
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09,
        1e-09, 5e-10, 2e-10, 1e-10, 5e-11,
        2e-11, 1e-11, 5e-12, 2e-12, 1e-12
    ]
    maxiter_per_lam = 3000

    while time.time() < phase1_end:
        # diversify with a small random blend
        rand_vec = random_initial(n, rng)
        a_start = 0.80 * best_a + 0.20 * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam (≈15 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, n, fft_len,
            max_iters=20000, lr=0.07, lam_init=0.30, lam_decay=0.9992
        )
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient (≈10 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.55 * budget_s
    if time.time() < phase3_end:
        for tf in (0.02, 0.01, 0.006, 0.003, 0.001):
            best_a, best_score = exp_gradient_optimize(
                best_a, phase3_end, n, fft_len, top_fraction=tf
            )
            print(f"[exp‑grad tf={tf}] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient exact (≈10 % of budget)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.65 * budget_s
    if time.time() < phase4_end:
        best_a, best_score = subgradient_exact_optimize(
            best_a, phase4_end, n, fft_len, step0=0.30
        )
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 5 – heavy simulated annealing (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, deadline, n, fft_len, rng,
            init_temp=5e-4, final_temp=1e-9, max_moves=300000
        )
        print(f"[SA] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 6 – final L‑BFGS refinement (≈10 % of budget)
    # ------------------------------------------------------------------
    phase6_end = deadline - 5.0  # leave a few seconds for hill‑climb
    if time.time() < phase6_end:
        # use a tighter λ‑schedule and more iterations per λ
        fine_lam = [1e-08, 5e-09, 2e-09, 1e-09, 5e-10, 2e-10, 1e-10,
                    5e-11, 2e-11, 1e-11, 5e-12, 2e-12, 1e-12]
        a_opt, sc_opt = lbfgs_continuation(
            best_a, fine_lam, maxiter_per_lam=4000,
            deadline=phase6_end, fft_len_conv=fft_len, fft_len_g=fft_len, n=n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[final L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 7 – final greedy hill‑climb (≈5 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len, n, deadline, rng,
            max_iter=300000, delta_frac=0.06
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation (official scoring)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
