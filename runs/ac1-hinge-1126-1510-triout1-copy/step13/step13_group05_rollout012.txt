```python
def generate_sequence_from_params(params, n):
    K = len(params)//3
    w = np.exp(params[0:K])   # positive
    mu = 1/(1+np.exp(-params[K:2*K]))  # sigmoid to [0,1]
    sigma = np.exp(params[2*K:3*K])   # positive; cap to max 0.5
    i = np.arange(n) / (n-1)
    a = np.zeros(n)
    for k in range(K):
        a += w[k] * np.exp(-0.5 * ((i - mu[k]) / sigma[k])**2)
    sum_a = a.sum()
    if sum_a <= 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / sum_a
    return a
```

"""
Propose a low‑score step sequence.

Algorithm (≈1000 s budget)

1.  Initialise a pool of candidates (previous best, spectral, random).
2.  Global optimisation in a low‑dimensional “mixture‑of‑Gaussians” space
    using SciPy’s differential evolution.  The mixture parameters are
    transformed to a non‑negative sequence on the probability simplex.
3.  Refine the best vector with a continuation of L‑BFGS on a
    smoothed max‑convolution (soft‑max) objective.
4.  Further polish with Adam on the same smooth objective.
5.  Finish with a fine‑grained exponentiated‑gradient / hill‑climb on the
    exact max‑convolution.
6.  Return the best list; store it in the global ``height_sequence_1``
    for possible warm‑starts on the next run.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (exact when the vector is normalised to sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(a*a) for a probability vector a (sum = 1).
    This equals the value returned by ``evaluate_sequence`` after normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Arguments:
        y               – logits (a = softmax(y))
        n               – length of the sequence
        fft_len_conv    – FFT length for the convolution
        fft_len_g       – FFT length for the back‑propagation step
        lam_factor      – factor to set the smoothing scale
    Returns:
        (objective, gradient) wrt. y
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initial families (used for the very first candidate)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in ``lam_schedule``, starting each stage
    from the previous result.  Returns the best sequence seen and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 5000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.05):
    """
    Adam optimisation of the smoothed max‑convolution (same objective
    as used by L‑BFGS).  Works on logits y = log(a) to stay on the simplex.
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 200000,
                    delta_start: float = 0.05,
                    delta_end: float = 0.003):
    """
    Fine‑grained hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Low‑dimensional mixture model + Differential Evolution
# ----------------------------------------------------------------------


def generate_sequence_from_params(params: np.ndarray, n: int) -> np.ndarray:
    """
    Mixture of K Gaussian bumps on [0,1] defined by the flat parameter vector:
        params = [log_w[0..K-1], mu[0..K-1], sigma[0..K-1]]
    Returns a non‑negative probability vector of length n.
    """
    K = len(params) // 3
    log_w = params[0:K]
    mu = params[K:2 * K]          # already in [0,1] because of bounds
    sigma = params[2 * K:3 * K]   # positive because of bounds

    w = np.exp(log_w)             # ensure positivity
    xs = np.linspace(0.0, 1.0, n)
    a = np.zeros(n, dtype=np.float64)

    for wk, mk, sk in zip(w, mu, sigma):
        # Gaussian profile, truncated at the interval edges
        a += wk * np.exp(-0.5 * ((xs - mk) / sk) ** 2)

    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise best candidate from the global cache or simple families
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to the required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            best_a = a0
            best_score = fast_score(best_a, fft_len, n)

    # a couple of deterministic starters for safety
    for starter in (spectral_init(n), random_initial(n, rng)):
        sc = fast_score(starter, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = starter.copy()

    print(f"[init] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Global search in a low‑dimensional mixture model (DE)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    de_end = start_time + 0.20 * budget_s
    if time.time() < de_end:
        K = 6                      # number of Gaussian components
        dim = 3 * K
        bounds = [(-10, 10)] * K + [(0.0, 1.0)] * K + [(0.02, 0.5)] * K

        def de_obj(params):
            a = generate_sequence_from_params(params, n)
            return fast_score(a, fft_len, n)

        # callback to abort when we run out of time
        def de_stop(xk, convergence):
            return time.time() >= de_end

        de_res = differential_evolution(de_obj,
                                         bounds,
                                         strategy='best1bin',
                                         maxiter=150,
                                         popsize=5,
                                         polish=False,
                                         init='latinhypercube',
                                         seed=int(rng.integers(1 << 30)),
                                         callback=de_stop,
                                         disp=False)

        a_de = generate_sequence_from_params(de_res.x, n)
        sc_de = fast_score(a_de, fft_len, n)
        print(f"[DE] best fast_score={sc_de:.8f}")

        if sc_de < best_score:
            best_score = sc_de
            best_a = a_de.copy()

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈30 % of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        # blend the current best with a fresh random draw to keep diversity
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈20 % of total time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           max_iters=8000,
                                           lam_init=0.30,
                                           lam_decay=0.9993,
                                           lr=0.07)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine‑grained hill‑climb until the deadline
    # ------------------------------------------------------------------
    while time.time() < deadline:
        cur_sc = fast_score(best_a, fft_len, n)
        best_a, best_score = hill_climb_fine(best_a,
                                             cur_sc,
                                             fft_len,
                                             n,
                                             deadline,
                                             rng,
                                             max_iter=250000,
                                             delta_start=0.04,
                                             delta_end=0.003)
        print(f"[hill‑climb fine] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final evaluation (official scoring)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose the best sequence for possible warm‑starts in the next run
    globals()['height_sequence_1'] = final_seq
    return final_seq
