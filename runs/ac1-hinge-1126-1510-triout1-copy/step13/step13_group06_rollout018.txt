def frank_wolfe_optimize(a_start, deadline, n, fft_len):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    iter = 0
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2*n - 1]
        k = int(conv.argmax())
        # compute gradient g_i = 2 * a_{k - i} where defined else 0
        # We need to construct vector g of length n: for i where j = k - i within [0,n-1]
        g = np.zeros_like(a)
        # compute indices
        j_low = max(0, k - (n - 1))
        j_high = min(k, n - 1)
        # for i in [j_low, j_high], g[i] = 2 * a[k - i]
        # Let's compute:
        i_range = np.arange(j_low, j_high + 1)
        g[i_range] = 2.0 * a[k - i_range]
        # linear subproblem: min <g, v> over simplex => pick i with minimal g_i
        i_star = np.argmin(g)  # if multiple choose any
        # step size
        gamma = 2.0 / (iter + 2.0)  # typical FW step size
        a = (1 - gamma) * a
        a[i_star] += gamma  # adding gamma mass to that index
        # projection automatically satisfied (weights sum remains 1)
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        iter += 1
    return best, best_score

"""
Propose a low‑value step sequence for the given evaluation function.

The algorithm proceeds in several phases within the allotted time budget:

1. **Initial population** – a diverse set of candidate vectors generated from
   classic windows (Hann, Blackman, Gaussian, …) plus many random beta‑
   distributions.  The best of these (by the fast exact objective
   `fast_score`) becomes the starting point.

2. **Intensified L‑BFGS** – a continuation over a fine λ‑schedule on the
   smoothed log‑sum‑exp approximation of the max‑convolution.  The vector is
   represented via logits for a soft‑max, guaranteeing non‑negativity and
   unit L¹‑norm.

3. **Adam optimisation** – further refinement of the same smooth objective
   with an adaptive learning‑rate schedule and slowly decreasing λ.

4. **Frank‑Wolfe minimisation** – direct minimisation of the *exact* max
   convolution.  At each iteration the shift with maximal convolution is
   located, a sub‑gradient is formed (g_i = 2·a_{k‑i}) and a classic Frank‑Wolfe
   step moves mass toward the index with the smallest gradient component.
   This rapidly pushes mass away from the most critical overlap.

5. **Greedy hill‑climb** – a final local search that transfers a small
   fraction of mass between two random positions only when the exact score
   improves.

All phases respect the constraints of `evaluate_sequence` (non‑negative,
finite numbers) and keep the vector on the probability simplex, so the final
vector can be fed directly to `evaluate_sequence`.  The routine returns the
best sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (sum(a) = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a probability vector."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth approximation of max_i (a*a)_i where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Various quick random distributions."""
    mode = rng.choice(['uniform', 'rand', 'exp', 'pow', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'rand':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'pow':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge = max(1, int(n * rng.uniform(0.006, 0.03)))
        heavy = np.sqrt(n / 2.0) / n
        a = np.full(n, 1.0 / n)
        a[:edge] = heavy
        a[-edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def inv_sqrt_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_init(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def random_beta_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Log‑uniform sampling of α,β over a wide range."""
    log_a = rng.uniform(-4.0, 2.0)      # α ∈ [10⁻⁴, 10²]
    log_b = rng.uniform(-4.0, 2.0)
    alpha = 10.0 ** log_a
    beta = 10.0 ** log_b
    return beta_init(n, alpha, beta)


def raised_cosine_init(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS through the whole λ‑schedule."""
    y = np.log(a_start + 1e-12)      # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:   # pragma: no cover
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  max_iters: int = 20000,
                  lr: float = 0.07,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9993):
    """Adam on the smoothed max‑convolution."""
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


def frank_wolfe_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int):
    """
    Frank‑Wolfe minimisation of the exact max‑convolution.
    Each iteration computes the current maximal shift, forms a sub‑gradient,
    then moves mass towards the index with the smallest gradient component.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    it = 0

    while time.time() < deadline:
        # convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k = int(conv.argmax())                     # shift with maximal value

        # sub‑gradient g_i = 2·a_{k-i} where defined
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k - idx]

        i_star = int(np.argmin(g))                # best vertex of simplex
        gamma = 2.0 / (it + 2.0)                   # classic FW step size

        a = (1.0 - gamma) * a
        a[i_star] += gamma                         # stay on simplex

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best = a.copy()

        it += 1

    return best, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 300000,
               delta_frac: float = 0.06):
    """Greedy hill‑climb (strictly improving)."""
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list[float] (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 5000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of candidates
    # --------------------------------------------------------------
    candidates = []

    # previous best (if available) – interpolate to current length
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # deterministic families
    candidates.append(spectral_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(inv_sqrt_init(n))
    candidates.append(arcsine_init(n))

    # random families
    for _ in range(12):
        candidates.append(random_initial(n, rng))
    for _ in range(6):
        candidates.append(raised_cosine_init(n, rng.uniform(0.5, 3.0)))
    for _ in range(300):
        alpha = rng.exponential(scale=0.8) + 0.1
        beta_param = rng.exponential(scale=0.8) + 0.1
        candidates.append(beta_init(n, alpha, beta_param))
    for _ in range(300):
        candidates.append(random_beta_init(n, rng))

    # --------------------------------------------------------------
    # Pick the best starting candidate
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensified L‑BFGS (≈35 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09,
        1e-09, 5e-10, 2e-10, 1e-10, 5e-11, 2e-11, 1e-11, 5e-12, 2e-12, 1e-12
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # diversify with a small random blend
        rand_vec = random_initial(n, rng)
        a_start = 0.85 * best_a + 0.15 * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, fft_len, n
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam (≈25 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, n, fft_len,
            max_iters=20000, lr=0.07,
            lam_init=0.30, lam_decay=0.9993
        )
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Frank‑Wolfe (≈20 % of budget)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.80 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = frank_wolfe_optimize(
            best_a, phase3_end, n, fft_len
        )
        print(f"[Frank‑Wolfe] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy Hill‑Climb (≈15 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_sc = fast_score(best_a, fft_len, n)
        best_a, best_score = hill_climb(
            best_a, cur_sc, fft_len, n, deadline, rng,
            max_iter=600000, delta_frac=0.06
        )
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Final evaluation (official scoring)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
