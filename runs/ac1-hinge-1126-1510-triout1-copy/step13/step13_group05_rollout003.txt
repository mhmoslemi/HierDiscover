"""
Propose a low‑score step function.

The algorithm builds a diverse pool of initial windows (uniform,
triangular, Hann, Hamming, Blackman, Nuttall, Kaiser, Tukey,
Chebyshev, DPSS, several random families).  The best of these is
selected, then refined by a sequence of continuous optimisers:

1. L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution.
2. Adam on the same smooth objective (longer run).
3. Projected sub‑gradient descent on the exact max‑convolution.
4. A very long fine‑grained hill‑climb (random mass transfers).

All optimisation steps keep the vector on the probability simplex,
so the final evaluation is invariant to the scaling performed by
`evaluate_sequence`.  The routine respects the 1000 s budget and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# optional imports – the algorithm works without them
try:
    from scipy.signal import windows as win
    _has_scipy_windows = True
except Exception:  # pragma: no cover
    _has_scipy_windows = False

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (probability vector => same as evaluate_sequence)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(conv) for a probability vector a (sum=1).
    This equals the value returned by `evaluate_sequence` after normalisation.
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisation families (all return a probability vector)

def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def hamming_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def nuttall_init(n: int) -> np.ndarray:
    # coefficients from the classic Nuttall window
    c0, c1, c2, c3 = 0.355768, 0.487396, 0.144232, 0.012604
    i = np.arange(n, dtype=np.float64)
    a = (c0 - c1 * np.cos(2 * np.pi * i / (n - 1))
         + c2 * np.cos(4 * np.pi * i / (n - 1))
         - c3 * np.cos(6 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def tukey_init(n: int, alpha: float = 0.5) -> np.ndarray:
    # scipy's tukey window (if available) otherwise fall back to Hann
    if _has_scipy_windows:
        a = win.tukey(n, alpha)
    else:
        # simple linear taper: same shape as Hann when alpha=1
        a = np.ones(n, dtype=np.float64)
        taper_len = int(np.ceil(alpha * n / 2.0))
        if taper_len > 0:
            t = np.linspace(0, np.pi / 2, taper_len)
            win_part = (1 - np.cos(2 * t)) / 2.0
            a[:taper_len] = win_part
            a[-taper_len:] = win_part[::-1]
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def chebwin_init(n: int, attn: float = 100.0) -> np.ndarray:
    # Chebyshev window (if scipy is present), otherwise uniform fallback
    if _has_scipy_windows:
        a = win.chebwin(n, at=attn)
    else:
        a = np.full(n, 1.0 / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def dpss_init(n: int, nw: float = 2.5) -> np.ndarray:
    if _has_scipy_windows:
        a = win.dpss(n, nw, Kmax=1, norm='manual')[0]
    else:
        a = np.full(n, 1.0 / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def pow_decay_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Power‑law decay with a random exponent."""
    exp = rng.uniform(0.1, 0.7)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-exp)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits for soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_iters: int = 8000,
                  lam_init: float = 0.30,
                  lam_decay: float = 0.9995,
                  lr: float = 0.07):
    """
    Adam optimisation of the smoothed max‑convolution.
    Operates on logits y = log(a). Adaptive step size helps escape plateaus.
    """
    y = np.log(a_start + 1e-12)

    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lam = lam_init

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 1
    while time.time() < deadline and it <= max_iters:
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        lam = max(lam * lam_decay, 1e-12)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        it += 1
    return best_a, best_score


def subgradient_exact_optimize(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               step0: float = 0.25):
    """
    Projected sub‑gradient descent using the exact max entry
    of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        idx = int(conv.argmax())

        # gradient of conv_idx w.r.t. a is a shifted copy of a
        g = np.zeros_like(a)
        j_low = max(0, idx - (n - 1))
        j_high = min(idx, n - 1)
        sub = a[idx - np.arange(j_low, j_high + 1)]
        g[j_low:j_high + 1] = 2.0 * sub

        # projection onto simplex after step
        mean_g = np.mean(g)          # because objective includes sum constraint
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def hill_climb_fine(a: np.ndarray,
                    cur_score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    rng: np.random.Generator,
                    max_iter: int = 300000,
                    delta_start: float = 0.04,
                    delta_end: float = 0.003):
    """
    Fine‑grained hill‑climb with a linearly decreasing step size.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        delta_frac = delta_start - (delta_start - delta_end) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer a random fraction of mass from i to j
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) with a low
    evaluation score.  Returns the best list found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (feel free to adjust – larger n gives more freedom)
    # --------------------------------------------------------------
    n = 5000                       # length of the candidate vector
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse initial candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous solution exists, interpolate it to the current size
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(uniform_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(hamming_init(n))
    candidates.append(blackman_init(n))
    candidates.append(nuttall_init(n))
    candidates.append(gaussian_init(n))
    candidates.append(kaiser_init(n, beta=5.0))
    candidates.append(kaiser_init(n, beta=10.0))
    candidates.append(tukey_init(n, alpha=0.5))
    candidates.append(tukey_init(n, alpha=0.2))
    if _has_scipy_windows:
        candidates.append(chebwin_init(n, attn=80))
        candidates.append(chebwin_init(n, attn=120))
        candidates.append(dpss_init(n, nw=2.5))
        candidates.append(dpss_init(n, nw=3.0))

    # Random families (moderate number – can be enlarged)
    for _ in range(12):
        candidates.append(random_initial(n, rng))
    for _ in range(8):
        candidates.append(pow_decay_init(n, rng))

    # Systematic sweep over Kaiser beta and Tukey alpha
    for _ in range(60):
        beta = rng.uniform(0.0, 12.0)
        candidates.append(kaiser_init(n, beta))
    for _ in range(30):
        alpha = rng.uniform(0.0, 1.0)
        candidates.append(tukey_init(n, alpha))

    # Sweep over Chebyshev attenuation if available
    if _has_scipy_windows:
        for _ in range(30):
            attn = rng.uniform(30.0, 170.0)
            candidates.append(chebwin_init(n, attn))

    # --------------------------------------------------------------
    # Select the best of the initial pool (fast evaluation)
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf
    for cand in candidates:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] {len(candidates)} candidates, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time‑budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈25 % of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # random convex combination of the current best and a fresh random guess
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.4          # blend factor ≤ 0.4
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈20 % of total time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           max_iters=12000,
                                           lam_init=0.30,
                                           lam_decay=0.9993,
                                           lr=0.07)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact sub‑gradient descent (≈20 % of total time)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.65 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_exact_optimize(best_a,
                                                        phase3_end,
                                                        n,
                                                        fft_len,
                                                        step0=0.25)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – long fine‑grained hill‑climb (remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        a_new, sc_new = hill_climb_fine(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=350000,
                                        delta_start=0.04,
                                        delta_end=0.003)
        if sc_new < best_score - 1e-9:   # tiny tolerance to avoid spam
            best_score = sc_new
            best_a = a_new.copy()
            print(f"[hill‑climb fine] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official scoring function
    print(f"[final] evaluated score = {final_eval:.8f}")

    # expose for possible future runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
