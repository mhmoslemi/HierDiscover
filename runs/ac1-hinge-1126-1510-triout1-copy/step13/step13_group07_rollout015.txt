def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start = time.time()
    # set n based on remaining time: try large n = 12000 if time permits else 8000
    n = 12000
    fft_len = next_pow2(2*n - 1)

    # Build candidate vectors:
    candidates = []
    # existing previous solution if any (resampled to n)
    # families: spectral_init, etc.
    # loop over beta-param families
    for alpha in np.linspace(0.2, 2.0, 9):
        for beta in np.linspace(0.2, 2.0, 9):
            candidates.append(beta_init(n, alpha, beta))
    # Additional param families: Gaussian, Kaiser, power law etc
    for sigma in np.linspace(0.1, 0.4, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))
    for beta_para in np.linspace(4.0, 16.0, 7):
        candidates.append(kaiser_init(n, beta=beta_para))
    # Random draws
    for _ in range(8):
        candidates.append(random_initial(n, rng))
    # Evaluate fast_score and keep best
    best_seq = candidates[0]; best_score = fast_score(best_seq, fft_len, n)
    for a in candidates[1:]:
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc; best_seq = a
    print(f"[init] best fast_score={best_score}")

    # Phase 1: L-BFGS with schedule
    deadline = start + budget_s-0.5
    phase1_end = start + 0.4 * budget_s
    lam_schedule = np.geomspace(0.4, 1e-9, num=30)
    while time.time() < phase1_end:
        # blend with random vector
        rand_vec = random_initial(n, rng)
        a_start = 0.8*best_seq + 0.2*rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start, lam_schedule,
                                           maxiter_per_lam=2500,
                                           deadline=phase1_end,
                                           fft_len=fft_len,
                                           n=n)
        if sc_opt < best_score:
            best_score = sc_opt; best_seq = a_opt
            print(f"[L-BFGS] new best score {best_score}")

    # Phase 2: Adam
    phase2_end = start + 0.6 * budget_s
    if time.time() < phase2_end:
        best_seq, best_score = adam_optimize(best_seq,
                                             phase2_end,
                                             n,
                                             fft_len,
                                             lam_factor=5e-4,
                                             lr=0.07,
                                             max_iter=80000)
        print(f"[Adam] best fast_score={best_score}")

    # Phase 3: targeted hill-climb with more moves
    if time.time() < deadline:
        best_seq, best_score = targeted_hill_climb(best_seq,
                                                   best_score,
                                                   fft_len,
                                                   n,
                                                   deadline,
                                                   rng,
                                                   max_moves=1200000)
        print(f"[hill] final fast_score={best_score}")

    # final evaluation
    final_seq = best_seq.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval}")

    a = a.copy()
    conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
    best_score = float('inf')
    best_a = a.copy()
    best_conv = conv.copy()
    moves = 0
    while time.time() < deadline and moves < max_moves:
        idx = int(conv.argmax())
        # compute range of i's
        i_start = max(0, idx - (n-1))
        i_end = min(idx, n-1)
        i_range = np.arange(i_start, i_end+1)
        j_range = idx - i_range
        # find source i with largest product a_i * a_j
        products = a[i_range] * a[j_range]
        # choose source i randomly weighted by product maybe
        # Use max for simplicity
        max_idx = np.argmax(products)
        i = i_range[max_idx]
        # random dest
        dest = rng.integers(0, n)
        if dest == i:

        # delta limited to fraction of a[i]
        max_delta = a[i] * 0.3
        if max_delta <= 0:

        # shift addition of dest
        conv[dest:dest+n] += 2*delta * a

"""
Propose a low‑score step function.

The search works in three stages:

1.  Build a rich pool of initial candidates (deterministic windows,
    parametric families such as Beta‑PDF, Gaussian, Kaiser, plus random
    draws).  The best candidate is selected by the exact fast score.

2.  Refine the best candidate with two gradient‑based phases:
    * L‑BFGS on a smooth approximation of the max‑convolution.
    * Adam on the same smooth objective.
   Both phases respect a time budget and keep the current best.

3.  A large‑scale greedy hill‑climb that updates the self‑convolution
    incrementally (no FFT per move).  At each step the current peak
    index is identified, mass is taken from a contributing position
    and moved to a random location; the convolution is updated by
    cheap linear operations.  The move size shrinks with the amount of
    mass at the source.  The loop runs as long as time permits.

The final sequence is returned as a plain list of floats; its official
score is printed before returning.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact score (probability vector → 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a vector with Σa = 1."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (logits → softmax)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Initialisation families

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def arcsine_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def beta_sym_init(n: int, s: float) -> np.ndarray:
    """
    Symmetric Beta distribution on [0,1] with shape parameter s (>0).
    For s<1 the density is U‑shaped (heavy at the edges).
    """
    x = (np.arange(n) + 0.5) / n
    # avoid log(0) – clamp x away from 0/1
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    a = np.exp((s - 1) * (np.log(x) + np.log(1.0 - x)))
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def edge_power_init(n: int, p: float) -> np.ndarray:
    """
    U‑shaped power law: a_i ∝ min(i+1, n-i)^{-p}.
    Larger p concentrates mass near the endpoints.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0
    d = np.minimum(idx, n - idx + 1.0)
    a = d ** (-p)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """Run a sequence of L‑BFGS optimisations over a decreasing λ schedule."""
    eps = 1e-12
    y = np.log(np.maximum(a_start, eps))          # logits for soft‑max
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factor: float = 0.001,
                  lr: float = 0.08,
                  max_iter: int = 20000):
    """Adam optimiser on the smoothed max‑convolution."""
    y = np.log(np.maximum(a_start, 1e-12))
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        a = softmax(y)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def incremental_hill_climb(a_start: np.ndarray,
                           n: int,
                           fft_len: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_moves: int = 800000,
                           recompute_interval: int = 50000):
    """
    Greedy hill‑climb with an *incremental* convolution update.
    Mass is moved from a location that contributes most to the current
    peak of the self‑convolution to a random destination.
    The convolution is updated by cheap vector operations, allowing
    many more moves than the FFT‑based version.
    """
    a = a_start.copy()
    # initial convolution
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()
    moves = 0

    while time.time() < deadline and moves < max_moves:
        # current peak index
        idx = int(conv.argmax())

        # indices (i,j) that contribute to conv[idx]  (i+j = idx)
        i_start = max(0, idx - (n - 1))
        i_end = min(idx, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        j_range = idx - i_range

        # pick the pair with maximal product a_i * a_j
        products = a[i_range] * a[j_range]
        if products.max() <= 0.0:
            # fallback: random source in the contributing range
            i = rng.integers(i_start, i_end + 1)
        else:
            src_idx = np.argmax(products)
            i = i_range[src_idx]

        # random destination (different from source)
        dest = rng.integers(0, n)
        if dest == i:
            continue

        # step size – a fraction of the source mass
        max_delta = a[i] * 0.25
        if max_delta <= 0.0:
            continue
        delta = max_delta * rng.random()
        if delta <= 0.0:
            continue

        # --- incremental convolution update ---------------------------------
        # store a copy of current a for the update (avoid extra allocation)
        a_old = a

        # removal of mass from i
        conv[i:i + n] -= 2.0 * delta * a_old
        # addition of mass at dest
        conv[dest:dest + n] += 2.0 * delta * a_old
        # quadratic terms from Δa * Δa
        conv[2 * i] += delta * delta
        conv[2 * dest] += delta * delta
        conv[i + dest] -= 2.0 * delta * delta
        # ---------------------------------------------------------------------

        # apply the move to a
        a[i] -= delta
        a[dest] += delta

        # occasional renormalisation to combat floating‑point drift
        if moves % recompute_interval == 0:
            a /= a.sum()
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # current score
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        moves += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderately large n gives a good trade‑off
    # ------------------------------------------------------------------
    n = 8000                      # larger than the previous 5000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # Resample a previously known solution if present
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(arcsine_init(n))

    # Gaussian windows
    for sigma in np.linspace(0.12, 0.35, 7):
        candidates.append(gaussian_init(n, sigma_factor=sigma))

    # Kaiser windows
    for beta in np.linspace(4.0, 16.0, 13):
        candidates.append(kaiser_init(n, beta=beta))

    # Symmetric Beta PDFs (U‑shaped for s<1)
    for s in np.linspace(0.20, 0.80, 13):
        candidates.append(beta_sym_init(n, s))

    # Edge‑power family
    for p in np.linspace(0.2, 1.5, 12):
        candidates.append(edge_power_init(n, p))

    # Random draws
    for _ in range(12):
        candidates.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Choose the best start according to the exact fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time‑budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (≈40 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-9, num=35)
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.30          # ≤ 0.30
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (≈20 % of the budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           lam_factor=0.001,
                                           lr=0.07,
                                           max_iter=60000)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Incremental hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        max_moves = 1200000
        best_a, best_score = incremental_hill_climb(best_a,
                                                    n,
                                                    fft_len,
                                                    deadline,
                                                    rng,
                                                    max_moves=max_moves,
                                                    recompute_interval=50000)
        print(f"[Hill‑climb] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
