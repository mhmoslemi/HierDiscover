        conv_full = np.fft.irfft(fa * fw, n=fft_len)[: n + L - 1]   # length n+L-1
        conv_rev = np.fft.irfft(fa * np.fft.rfft(w_rev, n=fft_len), n=fft_len)[: n + L - 1]
        # compute gradient g_i
        # term1: conv_rev[L-1 - i] i from 0..n-1
        # conv_rev indexes need offset: conv_rev size = n+L-1; L-1-i is index from L-1 down to L-n
        idx1 = L - 1 - np.arange(n)   # shape n
        term1 = conv_rev[idx1]
        term2 = conv_full[:n]
        g = 2.0 * (term1 + term2)
        grad_a = 2.0 * n * g   # factor from F_T derivative
        # compute gradient w.r.t x
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        eta = 0.01  # base learning rate; adapt might need tuning
        x -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # keep best if improved
        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # decay temperature
        T = max(T * 0.99, 1e-8)
    return best_a, best_score

"""
Propose a step‑function (non‑negative heights) with a low evaluation bound.

Algorithm
---------
0. Initialise a pool of candidate sequences.
   * Random families (power‑law, exponential, two‑sided, beta).
   * Random sparse 0/1 masks with equal non‑zero weight.
   * A fallback to the previously discovered best sequence
     (available as the global ``height_sequence_1``).

1. Broad stochastic search – keep the best ``TOP_K`` candidates
   (by fast_score) while the allotted time permits.

2. Refine each elite candidate with a smooth‑max Adam optimiser.
   The objective is a temperature‑smoothed version of
   ``max(conv(a,a))``; the gradient is obtained efficiently by
   two FFT‑based convolutions.  The temperature is annealed so the
   optimiser first sees a smooth landscape and then focuses on the
   true maximum.  The optimiser works on the log‑it representation
   (softmax) which automatically respects the simplex constraints.

3. Return the best sequence found (as a plain Python list).  The final
   sequence is verified with the supplied ``evaluate_sequence`` function.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective: 2·n·max(conv(seq,seq))/sum(seq)^2."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Shape generators (same as the previous implementation)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf: a_i ∝ (i+1)^(α−1)·(n−i)^(β−1)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sparse_mask(n: int, frac: float) -> np.ndarray:
    """
    Random 0/1 mask with approximately ``frac`` proportion of ones.
    The non‑zero entries are given equal weight so that the whole
    vector sums to 1.
    """
    if not (0.0 < frac < 1.0):
        raise ValueError
    mask = np.random.rand(n) < frac
    m = mask.sum()
    if m == 0:
        # fallback to uniform
        return np.full(n, 1.0 / n)
    seq = np.where(mask, 1.0 / m, 0.0)
    return seq.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed max of the convolution
# ----------------------------------------------------------------------


def refine_adam(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eta0: float = 0.05,
    T0: float = 0.05,
    T_decay: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Refine ``seq`` (already normalised to sum=1) using an Adam optimiser
    on a temperature‑smoothed version of ``max(conv)``.
    Returns the best sequence seen and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]

    # Initialise logits for a softmax parametrisation
    x = np.log(a + 1e-15)

    m = np.zeros_like(x)  # first–moment estimate
    v = np.zeros_like(x)  # second–moment estimate

    L = 2 * n - 1               # length of a*a
    fft_len = next_pow2(2 * n)  # enough for all convolutions

    best_a = a.copy()
    best_sc = fast_score(a)

    T = max(T0, 1e-8)      # temperature
    it = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        it += 1

        # ----- 1) obtain probability vector from logits -----
        max_x = np.max(x)
        exp_x = np.exp(x - max_x)          # stability
        a = exp_x / exp_x.sum()

        # ----- 2) convolution a*a (full) -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:L]

        # ----- 3) soft‑max weights over the convolution -----
        M = conv.max()
        # temperature‑controlled softmax (avoid overflow)
        w_exp = np.exp((conv - M) / T)
        w = w_exp / w_exp.sum()            # shape (L,)

        # ----- 4) compute gradient of the smoothed max -----
        # w_rev is the flipped weight vector
        w_rev = w[::-1]

        # a * w  (full convolution) – we only need the first n entries
        fw = np.fft.rfft(w, n=fft_len)
        conv_full = np.fft.irfft(fa * fw, n=fft_len)[: n + L - 1]

        # a * w_rev
        fw_rev = np.fft.rfft(w_rev, n=fft_len)
        conv_rev = np.fft.irfft(fa * fw_rev, n=fft_len)[: n + L - 1]

        # term2_i = Σ_k w_k a_{i‑k}  → conv_full[i]
        term2 = conv_full[:n]

        # term1_i = Σ_k w_k a_{k‑i}  → conv_rev[L‑1‑i]
        idx = L - 1 - np.arange(n)
        term1 = conv_rev[idx]

        g = 2.0 * (term1 + term2)               # ∂/∂a_i  Σ w_k conv_k
        grad_a = 2.0 * n * g                    # gradient of F_T = 2n·log‑sum‑exp

        # ----- 5) map gradient to logits (softmax Jacobian) -----
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # ----- 6) Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        eta = eta0 / (np.sqrt(it) + 1.0)        # simple decay on top of Adam
        x -= eta * m_hat / (np.sqrt(v_hat) + 1e-12)

        # ----- 7) keep the best sequence -----
        cur_sc = fast_score(a)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- 8) temperature annealing -----
        T = max(T * T_decay, 1e-10)

        # occasional random perturbation if stuck
        if no_improve >= 4000:
            sigma = 0.02 / np.sqrt(n)
            x += sigma * np.random.randn(n)
            no_improve = 0

    # final exact score for the best candidate
    best_sc = fast_score(best_a)
    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 0) Initialise with a previously known good construction if present
    # --------------------------------------------------------------
    try:

    except Exception:
        # fallback – a modest two‑sided power law
        init_len = 8000
        init_seq = generate_two_sided_power_law(init_len, 0.55, 0.55)

    # normalise to a probability vector
    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0.0:
        init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
    else:

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # --------------------------------------------------------------
