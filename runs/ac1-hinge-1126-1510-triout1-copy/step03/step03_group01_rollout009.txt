def solve_lp_max_sum(f, M):
    n = len(f)
    rows = 2*n-1
    # Build A_ub as sparse csr
    data = []
    col_ind = []
    row_ptr = [0]

    for k in range(rows):
        start_j = max(0, k - (n-1))
        end_j = min(n-1, k)
        # For each j in [start_j, end_j], coefficient = f_{k-j}
        for j in range(start_j, end_j+1):
            coeff = f[k-j]
            data.append(coeff)

        row_ptr.append(len(data))

    A_ub = sp.csr_matrix((data, col_ind, row_ptr), shape=(rows, n))

    g = res.x
    return g

"""
Improved step‑function search.

The algorithm consists of three phases:

1.  Global exploration – many random restarts on the probability simplex.
    Each restart performs a continuation L‑BFGS optimisation on a smooth‑max
    surrogate of the convolution peak and a short greedy hill‑climb.

2.  Linear‑program mixing – after the time budget for cheap restarts is
    exhausted, a sparse LP is built that maximises the total mass of a
    partner sequence `g` under the constraint   f ∗ g ≤ M   (where *M* is the
    current peak of f∗f).  Mixing a small amount of `g` into `f` strictly
    decreases the peak and therefore the final score.

3.  Final polishing – another hill‑climb and a short fine‑grained L‑BFGS
    run with a very small smoothing factor.

All numbers are kept non‑negative and renormalised to sum = 1, so the
evaluation is invariant under scaling.  The routine respects the 1000 s
budget and returns the best sequence found.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# --------------------------------------------------------------
#  Basic utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
#  Smooth‑max objective for L‑BFGS
# --------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the convolution and its gradient w.r.t. the unconstrained
    vector y (soft‑max parametrisation of the probability vector).
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    # this is a convolution of a (reversed) with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# --------------------------------------------------------------
#  Greedy hill‑climb
# --------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple greedy hill‑climb moving a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# --------------------------------------------------------------
#  Continuation L‑BFGS
# --------------------------------------------------------------


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = opt.minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# --------------------------------------------------------------
#  Initialisation strategies
# --------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# --------------------------------------------------------------
#  Linear‑program (LP) utilities
# --------------------------------------------------------------


_lpcache = {}  # cache of constraint structure per n


def _build_lp_structure(n: int):
    """Create the CSR pattern for the convolution constraints f * g ≤ M."""
    rows = 2 * n - 1
    row_ptr = np.empty(rows + 1, dtype=np.int32)
    col_idx = []
    f_idx = []
    nnz = 0
    row_ptr[0] = 0
    for k in range(rows):
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        for j in range(start_j, end_j + 1):
            col_idx.append(j)
            f_idx.append(k - j)          # coefficient comes from f_{k‑j}
            nnz += 1
        row_ptr[k + 1] = nnz
    col_idx = np.array(col_idx, dtype=np.int32)
    f_idx = np.array(f_idx, dtype=np.int32)
    return row_ptr, col_idx, f_idx


def solve_lp_max_sum(f: np.ndarray, M: float, deadline: float):
    """
    Solve   max Σ g_i   s.t.   f * g ≤ M,   g ≥ 0.
    Returns the optimal g (numpy array) or None on failure.
    """
    n = len(f)
    if n not in _lpcache:
        _lpcache[n] = _build_lp_structure(n)

    row_ptr, col_idx, f_idx = _lpcache[n]

    # Build data array for the sparse matrix (coefficients = f_i)
    data = f[f_idx]
    A_ub = sp.csr_matrix((data, col_idx, row_ptr), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=float)

    c = -np.ones(n)                     # maximise sum => minimise -sum
    bounds = [(0, None)] * n

    # Linprog with the HiGHS interior‑point solver (fast for large sparse problems)
    res = opt.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
    )
    if not res.success:
        return None
    return res.x


# --------------------------------------------------------------
#  Main entry point
# --------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine performs a large number of random restarts using a
    continuation L‑BFGS on a smoothed max‑convolution objective,
    occasional greedy hill‑climbs, and finishes with a sparse LP‑mixing
    phase that pushes the peak of the self‑convolution downward.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0    # small safety margin

    # ----------------------------------------------------------
    # problem size – a larger n gives a finer search space
    # ----------------------------------------------------------
    n = 5000
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 1000:
                n = max(n, int(tmp.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ----------------------------------------------------------
    # initialise with a uniform distribution (baseline)
    # ----------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # sum = 1
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # ----------------------------------------------------------
    # hyper‑parameters for the global search
    # ----------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 0.001, num=16))
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    iteration = 0

    # ----------------------------------------------------------
    # Phase 1 – many cheap restarts (until ~80 % of the budget)
    # ----------------------------------------------------------
    while time.time() < deadline - 150.0:            # keep ~150 s for LP phase
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # aggressive local hill‑climb (short, time‑bounded)
        hc_deadline = min(deadline - 10.0, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=3000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ----------------------------------------------------------
    # Phase 2 – LP based mixing (spreads mass while respecting current peak)
    # ----------------------------------------------------------
    if time.time() < deadline - 30.0:   # ensure enough time for polishing later
        lp_iter = 0
        while time.time() < deadline - 30.0 and lp_iter < 6:
            M = conv_fft(best_a, fft_len_conv, n).max()
            # solve the LP
            g_raw = solve_lp_max_sum(best_a, M, deadline)
            if g_raw is None:
                print("[lp] LP failed, terminating LP phase")
                break
            sum_g = g_raw.sum()
            if sum_g == 0.0:
                break
            g_norm = g_raw / sum_g               # now Σ g = 1  and f·g ≤ M
            # small convex mixing
            t = 0.15
            candidate = (1.0 - t) * best_a + t * g_norm
            cand_score = fast_score(candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_a = candidate
                best_score = cand_score
                best_y = np.log(best_a + 1e-12)
                print(f"[lp] iter={lp_iter} improved fast score = {best_score:.6f}")
                lp_iter += 1
            else:
                # if no improvement, shrink the step size and try once more
                t = 0.05
                candidate = (1.0 - t) * best_a + t * g_norm
                cand_score = fast_score(candidate, fft_len_conv, n)
                if cand_score < best_score:
                    best_a = candidate
                    best_score = cand_score
                    best_y = np.log(best_a + 1e-12)
                    print(f"[lp] small step improved fast score = {best_score:.6f}")
                lp_iter += 1

    # ----------------------------------------------------------
    # Phase 3 – final polishing (hill‑climb + fine L‑BFGS)
    # ----------------------------------------------------------
    if time.time() < deadline:
        # deep hill‑climb with remaining time
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # optional very fine L‑BFGS run (if a few seconds remain)
    if time.time() < deadline - 5.0:
        fine_lam_schedule = list(np.geomspace(0.3, 5e-5, num=12))
        best_y = schedule_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            fine_lam_schedule,
            maxiter_per_stage=300,
        )
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[fine] final fast score = {best_score:.6f}")

    # ----------------------------------------------------------
    # final evaluation and return
    # ----------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    # expose the sequence for possible reuse by subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
