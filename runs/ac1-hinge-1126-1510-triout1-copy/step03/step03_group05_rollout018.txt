"""
Propose a low‑score step‑function using a systematic scan of two‑sided
power‑law shapes, random permutations and a projected Adam optimiser.
The search concentrates on the exponent range that theoretical analysis
shows to give the smallest possible value of

    2·n·max(conv(a,a)) / (sum(a)^2)

(≈ 1.48 for n≈10⁴).  The routine respects the 1000 s budget, prints
progress and finally returns the best sequence found as a plain list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Cheap evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT based convolution.  The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    Symmetric case (p_left==p_right, w_left==w_right) yields the
    classic two‑ended power law.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient optimiser
# ----------------------------------------------------------------------


def refine_with_adam(
    seq: np.ndarray, time_limit: float, start_time: float
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser for the non‑convex objective.
    Returns the best sequence seen and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]
    a = project_simplex(a)               # guarantee feasibility (sum=1)

    fft_len = next_pow2(2 * n)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_score = fast_score(best_seq)

    t = 0
    while time.time() < start_time + time_limit:
        t += 1

        # current convolution & max
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # sub‑gradient of the peak (all active indices)
        eps_active = 1e-12
        active = np.where(conv >= M - eps_active)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # Adam update (step‑size decays with √t)
        eta0 = 0.3
        eta = eta0 / (np.sqrt(t) + 1.0)

        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        delta = eta * m_hat / (np.sqrt(v_hat) + eps)
        cand = a - delta
        a = project_simplex(cand)

        # occasional jitter to escape plateaus
        if t % 4000 == 0:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))

    # final exact score for the best candidate
    best_score = fast_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score height sequence within ``budget_s`` seconds.
    Returns a plain Python list of non‑negative floats suitable for
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5       # safety margin for the final call

    # ------------------------------------------------------------------
    # 0) keep a fallback (in case the global variable is missing)
    # ------------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        # simple symmetric two‑sided power law as a very basic start
        init_seq = generate_two_sided_power_law(8000, 0.55, 0.55)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0.0:
        init_seq = np.full(init_seq.shape, 1.0 / init_seq.shape[0])
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) systematic scan of symmetric two‑sided power laws (theoretical sweet spot)
    # ------------------------------------------------------------------
    length_choices = [4000, 6000, 8000, 10000, 12000, 15000]
    elite: list[tuple[float, np.ndarray]] = [(best_score, best_seq.copy())]

    for n in length_choices:
        # fine grid in the promising exponent interval
        for p in np.linspace(0.46, 0.66, 41):   # step ≈0.005
            seq = generate_two_sided_power_law(n, p, p, 1.0, 1.0)
            seq = np.maximum(seq, 0.0)
            s = seq.sum()
            if s < 1e-12:
                continue
            seq = seq / s
            sc = fast_score(seq)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[grid] n={n:5d} p={p:.3f} → fast_score={sc:.6f}")
            # maintain a small elite set (size ≤ 8)
            if len(elite) < 8:
                elite.append((sc, seq.copy()))
            else:
                worst_idx = max(range(len(elite)), key=lambda i: elite[i][0])
                if sc < elite[worst_idx][0]:
                    elite[worst_idx] = (sc, seq.copy())

    # ------------------------------------------------------------------
    # 2) random families for additional diversity (short budget)
    # ------------------------------------------------------------------
    rand_deadline = start + 0.15 * budget_s
    families = ["power", "exp", "beta"]
    while time.time() < rand_deadline:
        n = random.choice([5000, 8000, 11000, 14000])
        family = random.choice(families)
        if family == "power":
            p = random.uniform(0.10, 1.20)
            seq = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)
        else:  # beta
            alpha = random.uniform(0.05, 1.50)
            beta = random.uniform(0.05, 1.50)
            seq = generate_beta_shape(n, alpha, beta)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[rand] new global best ({family}) n={n} score={sc:.6f}")

        # keep elite
        if len(elite) < 8:
            elite.append((sc, seq.copy()))
        else:
            worst_idx = max(range(len(elite)), key=lambda i: elite[i][0])
            if sc < elite[worst_idx][0]:
                elite[worst_idx] = (sc, seq.copy())

    # sort elite by increasing score
    elite.sort(key=lambda x: x[0])
    elite = elite[:6]   # keep the very best few

    # ------------------------------------------------------------------
    # 3) permutation phase – many random reorderings of the best shapes
    # ------------------------------------------------------------------
    permuted_elite: list[tuple[float, np.ndarray]] = []
    for score, seq in elite:
        n = seq.size
        best_local_seq = seq
        best_local_score = score
        # try a few random permutations; keep the best ordering
        for _ in range(200):
            perm = np.random.permutation(n)
            cand = seq[perm]
            sc = fast_score(cand)
            if sc < best_local_score:
                best_local_score = sc
                best_local_seq = cand
        permuted_elite.append((best_local_score, best_local_seq))

    # re‑sort after permutation improvements
    permuted_elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 4) refined optimisation with Adam on the elite candidates
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining <= 0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
        print(f"Time exhausted – final score = {final_score:.6f}")
        return final_seq

    per_candidate = time_remaining / len(permuted_elite)

    for idx, (score, seq) in enumerate(permuted_elite):
        if time.time() > deadline:
            break
        print(f"[refine] elite #{idx+1} len={seq.size} start_score={score:.6f}")
        refined_seq, refined_score = refine_with_adam(seq, per_candidate, time.time())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] ** global best improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) final exact verification and return
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)   # type: ignore[name-defined]
    print(f"Finished – best exact score = {final_score:.6f}")
    return final_seq
