"""
Search for a step‑function with a low evaluation score.

The algorithm is a hybrid of random sampling, simple averaging of the
best samples, a few iterations of L‑BFGS on a smooth approximation of the
max‑convolution, a projected sub‑gradient descent on the true max, and a
short simulated‑annealing / hill‑climbing polish.  All work is performed
within the user‑supplied time budget (default 1000 s) and respects the
simplex (non‑negative, sum = 1) constraint, which makes the metric
scale‑invariant.

The most important new idea compared with the previous code base is an
early “population” phase: many random candidates (Dirichlet, exponential,
power‑law, edge‑heavy) are generated, the top few are combined with an
inverse‑score weighting, and the resulting mixture serves as a high‑quality
starting point for the later deterministic refinements.  This extra
exploration typically finds better basins and pushes the final score
below the 1.503 target.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a vector already on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient.

    Returns
    -------
    f : float
        2·n·smooth_max(conv)
    grad : np.ndarray
        gradient w.r.t. y (same shape as y)
    """
    a = softmax(y)                     # map to simplex

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # g_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient step.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active set (indices reaching the max)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient = 2 * (mask * a_rev) restricted to the original n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # record best iterate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction between two coordinates while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple simulated annealing that moves a fraction of mass between two
    positions.  Improves are always accepted; uphill moves with a
    temperature‑dependent probability.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)   # initial temperature
    decay = 0.9995

    it = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                T *= decay
                it += 1
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random candidate generators
# ----------------------------------------------------------------------


def random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random non‑negative vector on the simplex.
    Several families are mixed to give diverse initial points.
    """
    mode = rng.choice(['dirichlet', 'exp', 'power', 'edge'])
    if mode == 'dirichlet':
        conc = rng.uniform(0.2, 5.0)                 # concentration parameter
        a = rng.gamma(conc, 1.0, size=n)
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.1, 3.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    else:   # edge‑heavy
        edge_frac = rng.uniform(0.004, 0.08)
        m = max(1, int(edge_frac * n))
        a = np.zeros(n)
        val = 0.5 / m
        a[:m] = val
        a[-m:] = val
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found (as a list of floats).
    """
    # ------------------------------------------------------------------
    # Random generator and initial sequence size
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    if 'height_sequence_1' in globals():
        init = np.array(globals()['height_sequence_1'], dtype=float)
        init = np.maximum(init, 0.0)
        if init.size == 0:
            n = 2000
            a0 = np.full(n, 1.0 / n)
        else:
            max_n = 3000
            if init.size > max_n:
                init = init[:max_n]
            n = init.size
            if init.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init / init.sum()
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # FFT length preparation (powers of two)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # Overall time budget
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – generate a population of random candidates
    # ------------------------------------------------------------------
    pop_deadline = start_time + 0.20 * budget_s
    candidates = [(best_a, best_score)]          # keep current best
    while time.time() < pop_deadline:
        a = random_candidate(n, rng)
        score = fast_score(a, fft_len_conv, n)
        candidates.append((a, score))
        if score < best_score:
            best_score = score
            best_a = a.copy()
    print(f"[population] generated {len(candidates)} candidates, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – form a weighted mixture of the top candidates
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[1])
    top_k = min(20, len(candidates))
    top = candidates[:top_k]
    weights = np.array([1.0 / sc for _, sc in top])
    weights /= weights.sum()
    mix = np.zeros(n)
    for (a, _), w in zip(top, weights):
        mix += w * a
    mix = np.maximum(mix, 0.0)
    mix /= mix.sum()

    # brief L‑BFGS refinement of the mixture (smooth max)
    # use a modest continuation schedule
    y0 = np.log(mix + 1e-12)
    for lam in [0.5, 0.3, 0.15, 0.08, 0.04]:
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 250, 'gtol': 1e-12, 'disp': False})
        mix = softmax(res.x)

    mix_score = fast_score(mix, fft_len_conv, n)
    if mix_score < best_score:
        best_score = mix_score
        best_a = mix.copy()
    print(f"[mix] after L‑BFGS mix_score={mix_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient optimisation (true max)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase2_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for fine‑grained exploration
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.85 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] after annealing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=3000)

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluator (for reporting)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluated score = {eval_val:.6f}")

    return final_a.tolist()
