"""
Propose a candidate step function with low objective.
Algorithm:
1. Randomly sample a few families, keep the best.
2. Starting from the best (or arcsine), run many iterations of a
   sub‑gradient descent on the max‑convolution.  The sub‑gradient is
   computed from *all* active convolution indices (those attaining the
   current maximum) rather than a single peak.  The update is
   multiplicative (keeps non‑negativity) followed by a projection onto
   the probability simplex.
3. The best sequence seen is returned and finally verified with the
   official `evaluate_sequence` function.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper / family generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """Generate a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """Generate a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


# ----------------------------------------------------------------------
# Fast evaluation (uses numpy.convolve, sufficient for moderate n)
# ----------------------------------------------------------------------


def fast_score(seq: np.ndarray) -> float:
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_conv = float(np.max(conv))
    return float(2.0 * n * max_conv / (s * s))

        best_seq = init.copy()

        print(f"[init] using provided sequence, n={best_seq.size}, score={best_score:.6f}")
    except Exception:
        # fallback: arcsine‑like with moderate length
        n0 = 2000
        best_seq = generate_arcsine_like(n0)
        best_seq = np.maximum(best_seq, 0.0)
        best_seq /= best_seq.sum()
        best_score = fast_score(best_seq)
        print(f"[init] fallback arcsine, n={n0}, score={best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------

    np.random.seed(random_seed)

            i = np.arange(n, dtype=np.float64)
            cand = np.exp(-0.5 * ((i - mid) / sigma) ** 2)

        else:

    # --------------------------------------------------------------

    jitter_scale = 0.005       # relative jitter magnitude

        # ----- convolution and active set -----
        conv_full = np.convolve(a, a)          # length 2n‑1
        max_val = conv_full.max()
        # tolerance to capture numerically equal maxima
        tol = max_val * 1e-12 + 1e-14
        mask = (conv_full >= max_val - tol).astype(np.float64)   # 1.0 where active

        # ----- sub‑gradient from the whole active set -----
        # compute g_i = 2 * Σ_{k active} a_{k-i}
        # This can be expressed via a convolution with the reversed mask.
        mask_rev = mask[::-1]
        conv_rev = np.convolve(a, mask_rev)   # length 3n‑2
        idx = 2 * n - 2 - np.arange(n)        # indices that give the required sums
        g = 2.0 * conv_rev[idx]

        # ----- multiplicative update -----
        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * g)

        # ----- projection onto the simplex (simple normalisation) -----
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional jitter to avoid stagnation -----
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a = a / max(a.sum(), 1e-12)

        # ----- bookkeeping -----
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.1f}s] iter {iteration} new best score={cur_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best official score = {final_score:.6f}")
    return final_seq
