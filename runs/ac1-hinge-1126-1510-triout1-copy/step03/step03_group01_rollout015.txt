Simulated annealing:

```
def simulated_annealing(a, cur_score, fft_len, n, deadline):
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    T0 = 0.01
    while time.time() < deadline:
        # compute temperature decreasing linearly
        t = (deadline - time.time()) / (deadline - start_time)
        T = T0 * t
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i==j:
            continue
        delta = best_a[i] * rng.random() * 0.1  # up to 10% of mass at i
        if delta == 0:
            continue
        new_a = best_a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # ensure sum remains 1
        # compute score
        new_score = fast_score(new_a, fft_len, n)
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / T):
            best_a = new_a
            best_score = new_score
    return best_a, best_score
```

def project_simplex(v):
    # from literature: Sort descending.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑evaluation step sequence.

The search works in three phases:

1️⃣  Random restarts with a soft‑max parametrisation and a continuation
    L‑BFGS optimiser.  The smoothing factor (lam) is decreased
    geometrically (15‑30 stages).  After each optimisation a fast greedy
    hill‑climb refines the result.

2️⃣  Fine tuning of the current best candidate by a short sub‑gradient
    descent on the *exact* max‑convolution (projected onto the simplex)
    and a final high‑precision L‑BFGS run with a very small smoothing
    parameter.

3️⃣  A cheap simulated‑annealing polishing that repeatedly moves a
    random fraction of mass from an index that contributes heavily to
    the current maximal convolution to an index with a small
    contribution.  The annealing temperature is lowered linearly until
    the time budget expires.

All helper functions are top‑level, the algorithm respects the
non‑negative and normalisation constraints and returns a plain list of
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Objective for smooth‑max L‑BFGS optimisation
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t.
    the unconstrained vector y.

    lam = max(conv) * lam_factor   (lam_factor is the smoothing schedule).
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z   # d smooth_max / d conv_i   (weights)

    # compute g_i = Σ_j a_j * w_{i+j}
    # this is a convolution of a (reversed) with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the central n entries (indices n‑1 … 2n‑2)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Generation of starting points (pre‑soft‑max space)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # heavier mass at the two ends (1‑3 % of indices)
        m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set (pairwise sums distinct) – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if cand + c in sums:
                    conflict = True
                    break
            if not conflict and 2 * cand not in sums:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode!r}")
    return y0


# ----------------------------------------------------------------------
# Continuation optimiser (multiple lam stages)
# ----------------------------------------------------------------------


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Greedy hill‑climb (random moves)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: repeatedly move a random fraction of mass
    from i to j (keeping the sum = 1).  Keep a move only if the fast
    score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 200,
    step_factor: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on 2·n·max(a*a).  The step direction
    is a sub‑gradient of the (non‑smooth) max‑convolution projected onto
    the probability simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute full convolution and locate a maximal index
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # sub‑gradient g_i = 2 * a_{k_max - i} (0 if out of range)
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[partner[mask]]

        # direction that keeps the sum = 0  (projection onto the simplex)
        mean_g = g.mean()
        d = -g + mean_g

        # maximal step without violating non‑negativity
        neg = d < 0
        if np.any(neg):
            alpha_max = np.min(best[neg] / -d[neg])
        else:
            alpha_max = np.inf

        # take a fraction of the maximal feasible step
        alpha = step_factor * alpha_max
        if not np.isfinite(alpha) or alpha <= 0:
            break

        new = best + alpha * d
        # re‑normalise (tiny numerical drift)
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0:
            break
        new /= s

        new_score = fast_score(new, fft_len, n)
        if new_score < best_score:
            best = new
            best_score = new_score
            # keep the same step_factor
        else:
            # if no improvement, slowly shrink the step size
            step_factor *= 0.9

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing polishing
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Low‑temperature simulated annealing that moves a random fraction of
    mass from an index contributing heavily to the current maximal
    convolution to a random other index.  Acceptances follow the usual
    Metropolis rule with a linearly decreasing temperature.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    T0 = 0.01  # initial temperature

    while time.time() < deadline:
        # linear cooling
        remaining = max(deadline - time.time(), 0.0)
        frac = remaining / (deadline - start)
        T = T0 * frac

        # pick source index weighted by its contribution to the peak
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))
        partner = np.full(n, -1, dtype=int)
        idx = np.arange(n)
        p = k_max - idx
        mask = (p >= 0) & (p < n)
        partner[mask] = p[mask]
        contrib = np.zeros(n)
        contrib[mask] = best[idx[mask]] * best[p[mask]]
        total = contrib.sum()
        if total == 0:
            # fallback to uniform random pick
            i = rng.integers(0, n)
        else:
            probs = contrib / total
            i = rng.choice(n, p=probs)

        # destination index (uniform, but different from source)
        j = rng.integers(0, n)
        if j == i:
            continue

        # move a random fraction (up to 5 %) of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue

        new = best.copy()
        new[i] -= delta
        new[j] += delta

        new_score = fast_score(new, fft_len, n)
        if new_score < best_score:
            best = new
            best_score = new_score
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(new_score - best_score) / (T + 1e-20)):
                best = new
                best_score = new_score

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns a list of non‑negative floats (sum > 0) ready for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a known construction if available, otherwise
    # start with a moderately large vector.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 5000:
            n = max(8000, int(tmp.size))
            best_a = tmp / tmp.sum()
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)           # uniform start, sum = 1

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)           # soft‑max pre‑image

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # FFT lengths (powers of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = fft_len_conv   # sufficient for the gradient convolution

    # ------------------------------------------------------------------
    # Random‑restart continuation optimisation
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 0.001, num=20))   # coarser than the
                                                            # previous run
    modes = ["random", "exp", "power", "edge", "sidon"]
    iteration = 0

    # keep a few seconds for the final polishing phases
    reserve = 12.0

    while time.time() < deadline - reserve:
        mode = rng.choice(modes)
        y0 = generate_initial_y(mode, n, rng)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=200,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- greedy hill‑climb -----
        hc_deadline = min(deadline - reserve, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- sub‑gradient refinement (a few cheap steps) -----
        sd_deadline = min(deadline - reserve, time.time() + 0.4)
        best_a, best_score = subgradient_descent(
            best_a,
            best_score,
            fft_len_conv,
            n,
            sd_deadline,
            max_iter=150,
            step_factor=0.6,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Fine‑grained L‑BFGS with a very small smoothing factor
    # ------------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-6, num=30))
    for lam in fine_lam_schedule:
        if time.time() >= deadline - reserve:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 600, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[fine] lam={lam:.2e} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy polishing
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline - 4.0,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Simulated‑annealing polishing (uses the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
