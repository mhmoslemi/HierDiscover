Search for a non‑negative step function (a list of heights) that minimises

The optimiser consists of four stages:

1. **Parametric sweep** – generate a large variety of simple analytic
   families (power‑law, exponential, double‑exponential, Beta,
   arcsine‑like).  Each candidate is projected onto the probability simplex
   and the best one (smallest max‑convolution) is kept.

2. **Adam on a smoothed max** – replace the hard ``max`` by a soft‑max
   ``log‑sum‑exp(α·conv)/α``.  The gradient is obtained via FFTs.
   Adam updates are performed on the simplex, while the smoothing
   parameter ``α`` is increased from a modest value to a large value
   (≈800).  This brings the solution close to a good basin.

3. **Sub‑gradient descent on the true max** – once ``α`` is large we
   switch to the genuine sub‑gradient of ``max(conv)`` (uniform weights
   on the active set).  A diminishing step‐size ``η/√t`` is used for a
   long run.

4. **Random local moves** – cheap mass‑transfer moves and occasional
   swaps are tried; any improvement is kept.  This helps to escape tiny
   plateaus that gradient steps may miss.

All heavy lifting (convolution, correlation) uses FFTs (``O(L log L)``)
with ``L`` the next power of two ≥ ``2·n‑1``.  The routine respects the
supplied time budget (default 1000 s) and returns the best sequence
found, scaled to the normalisation required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)‑1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of <weights, a*a> with respect to a.
    Implements 2·correlation(weights, a) using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase      # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # should never happen
        w[conv.argmax()] = 1.0
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The function respects the given time budget (seconds) and returns a
    plain Python list of floats suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # 0. Choose a problem size (n)
    # ------------------------------------------------------------------
    try:
        # if a previous construction exists, keep its length unless it is very short
        init_len = len(height_sequence_1)   # type: ignore[name-defined]
        n = max(init_len, 15000)
    except Exception:
        n = 15000

    # ------------------------------------------------------------------
    # 1. FFT helpers (same for the whole run)
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())       # smallest power of two ≥ 2n‑1
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # ------------------------------------------------------------------
    # 2. Keep the best sequence discovered so far
    # ------------------------------------------------------------------
    best_seq = None
    best_max = float("inf")                   # best `max(conv)` seen so far

    # ------------------------------------------------------------------
    # Helper that evaluates a candidate and updates the global best.
    # ------------------------------------------------------------------
    def _maybe_update(cand: np.ndarray, tag: str = "") -> None:
        nonlocal best_seq, best_max
        cand = cand.astype(np.float64)
        cand /= cand.sum()
        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"New best {tag:12s} max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # ------------------------------------------------------------------
    # 3. Parametric sweep over simple families
    # ------------------------------------------------------------------
    print(f"--- Phase 1: parametric sweep (n={n}) ---")
    xs = np.arange(n, dtype=np.float64)

    # power‑law family
    for p in np.linspace(0.0, 3.0, 61):
        if time.time() > deadline:
            break
        base = (xs + 1) ** p
        _maybe_update(base / base.sum(), f"power p={p:.2f}")

    # exponential family
    for lam in np.linspace(0.0, 6.0, 61):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        _maybe_update(exp_shape / exp_shape.sum(), f"exp  λ={lam:.2f}")

    # double‑exponential (U‑shaped) family
    for lam in np.linspace(0.0, 6.0, 31):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        right = left[::-1]
        u_shape = left + right
        _maybe_update(u_shape / u_shape.sum(), f"U‑shape λ={lam:.2f}")

    # Beta‑density family
    alphas = [0.5, 1.0, 2.0, 5.0]
    betas = [0.5, 1.0, 2.0, 5.0]
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta_shape = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
            if np.all(beta_shape > 0):
                _maybe_update(beta_shape / beta_shape.sum(),
                              f"Beta a={a} b={b}")

    # arcsine‑like shape (classic “U‑profile”)
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    _maybe_update(arcsine / arcsine.sum(), "arcsine")

    # ------------------------------------------------------------------
    # If sweep produced nothing (should not happen) fall back to uniform
    # ------------------------------------------------------------------
    if best_seq is None:
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        _maybe_update(uni, "uniform")

    # ------------------------------------------------------------------
    # 4. Adam descent on the smoothed max
    # ------------------------------------------------------------------
    print("--- Phase 2: Adam descent with soft‑max ---")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12                       # base learning rate
    max_adam_iters = 180000
    it = 0
    alpha_start, alpha_end = 4.0, 800.0

    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, a.copy()
            print(f"Adam it {it:6d} improved max={cur_max:.8e}  score={2*n*cur_max:.6f}")

        # smoothing parameter growing linearly
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smoothed objective
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update on the simplex (Euclidean projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        a = a_new

    # ------------------------------------------------------------------
    # 5. Sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    print("--- Phase 3: Sub‑gradient descent on true max ---")
    a = best_seq.copy()
    sub_it = 0
    max_sub_iters = 180000
    while time.time() < deadline and sub_it < max_sub_iters:
        sub_it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, a.copy()
            print(f"Sub‑grad it {sub_it:6d} improved max={cur_max:.8e}  score={2*n*cur_max:.6f}")

        w = _submax_weights(conv)                 # uniform on active set
        grad = _gradient_from_weights(a, w, phase, L)

        step_size = 0.5 / np.sqrt(sub_it)        # classic diminishing step
        a_new = a - step_size * grad
        a_new = _project_simplex(a_new)

        a = a_new

    # ------------------------------------------------------------------
    # 6. Random permutations + short local refinement
    # ------------------------------------------------------------------
    print("--- Phase 4: Permutation+short Adam refinements ---")
    perm_trials = 25
    for p in range(perm_trials):
        if time.time() >= deadline:
            break
        perm = np.random.permutation(n)
        a_perm = best_seq[perm].copy()

        # a few cheap Adam steps on the permuted vector
        m = np.zeros_like(a_perm)
        v = np.zeros_like(a_perm)
        inner_iters = 800
        for inner in range(inner_iters):
            conv = _compute_convolution(a_perm, L)
            cur_max = conv.max()
            if cur_max < best_max:
                best_max, best_seq = cur_max, a_perm.copy()
                print(f"Perm {p:2d} inner it {inner:4d} improved max={cur_max:.8e}  score={2*n*cur_max:.6f}")

            # use a fairly large α to approximate the true max
            w = _softmax_weights(conv, 300.0)
            grad = _gradient_from_weights(a_perm, w, phase, L)

            # simple SGD step (no momentum) – large learning rate works well for short runs
            step = 0.04 * grad
            a_perm = a_perm - step
            a_perm = _project_simplex(a_perm)

    # ------------------------------------------------------------------
    # 7. Cheap random hill‑climbing moves
    # ------------------------------------------------------------------
    print("--- Phase 5: Random local mass transfers & swaps ---")
    a_current = best_seq.copy()
    while time.time() < deadline:
        # mass transfer move
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        delta = np.random.rand() * 0.05 * a_current[i]   # up to 5 % of source
        if delta <= 0.0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:               # sanity (should never happen)
            continue

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            a_current = a_new
            print(f"Transfer improved max={max_new:.8e}  score={2*n*max_new:.6f}")

        # occasional swap move (helps escape flat plateaus)
        if np.random.rand() < 0.015:
            k = np.random.randint(0, n)
            a_swap = a_current.copy()
            a_swap[i], a_swap[k] = a_swap[k], a_swap[i]
            conv_swap = _compute_convolution(a_swap, L)
            max_swap = conv_swap.max()
            if max_swap < best_max:
                best_max, best_seq = max_swap, a_swap.copy()
                a_current = a_swap
                print(f"Swap improved max={max_swap:.8e}  score={2*n*max_swap:.6f}")

    # ------------------------------------------------------------------
    # 8. Final scaling to the normalisation used by ``evaluate_sequence``
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)   # for logging only
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) "
          f"after {it} Adam iters, {sub_it} sub‑grad iters ---")
    return final_seq
