def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w


def _gradient_from_weights(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <w, a*a>  w.r.t. a.
    Implements  2·correlation(w, a)  using FFTs.
    """
    n = a.shape[0]

    # FFT of the weight vector (length 2n‑1, zero‑padded to L)
    fft_w = np.fft.fft(w, n=L)

    # FFT of the reversed a (zero‑padded to L)
    rev_a = np.zeros(L, dtype=np.float64)
    rev_a[:n] = a[::-1]
    fft_rev_a = np.fft.fft(rev_a, n=L)

    # Linear convolution w * rev_a  (the first  n  entries after index n‑1
    # form the desired correlation)
    corr_full = np.real(np.fft.ifft(fft_w * np.conj(fft_rev_a)))
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad

    if init_seq is not None:
        _try_candidate(init_seq)

    # --------------------------------------------------  Phase 1 – parametric sweep
    print(f"--- Phase 1: parametric sweep (n={n}) ---")
    xs = np.arange(n, dtype=np.float64)

    # exponential family (lam ∈ [0,10])
    for lam in np.linspace(0.0, 10.0, 81):
        if time.time() > deadline:
            break
        cand = np.exp(-lam * xs)
        cand /= cand.sum()
        _try_candidate(cand)
        _try_candidate(cand[::-1])

    # power‑law family (p ∈ [0,3])
    for p in np.linspace(0.0, 3.0, 61):
        if time.time() > deadline:
            break
        cand = (xs + 1) ** p
        cand /= cand.sum()
        _try_candidate(cand)
        _try_candidate(cand[::-1])

    # double‑exponential “U‑shape” (lam ∈ [0,10])
    for lam in np.linspace(0.0, 10.0, 41):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        right = left[::-1]
        cand = left + right
        cand /= cand.sum()
        _try_candidate(cand)

    # discrete Beta shapes
    betas = [(a, b) for a in [0.2, 0.5, 1.0, 2.0, 5.0] for b in [0.2, 0.5, 1.0, 2.0, 5.0]]
    x_mid = (xs + 0.5) / n
    for a, b in betas:
        if time.time() > deadline:
            break
        beta_shape = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
        if np.all(beta_shape > 0):
            cand = beta_shape / beta_shape.sum()
            _try_candidate(cand)
            _try_candidate(cand[::-1])

    # arcsine‑like shape
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    arcsine /= arcsine.sum()
    _try_candidate(arcsine)
    _try_candidate(arcsine[::-1])

    # random Dirichlet‑like seeds (skewed towards the edges)
    for _ in range(30):
        if time.time() > deadline:
            break
        beta_pow = np.random.uniform(0.1, 3.0)
        rand_vec = np.random.rand(n) ** beta_pow
        rand_vec /= rand_vec.sum()
        if np.random.rand() < 0.5:
            rand_vec = rand_vec[::-1]
        _try_candidate(rand_vec)

    # Fallback if sweep produced nothing
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _compute_convolution(best_seq, L).max()

    # --------------------------------------------------  Phase 2 – Adam on smoothed max
    print("--- Phase 2: Adam descent with smoothed max ---")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.85, 0.995
    eps = 1e-8
    lr0 = 0.20
    max_adam_iters = 50000
    it = 0
    alpha_start, alpha_end = 2.0, 500.0

    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, L)
        # soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smoothed objective
        grad = _gradient_from_weights(a, w, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max = max_new
            best_seq = a_new.copy()
            print(f"Adam it {it:5d} improved max={max_new:.9e}  score={2 * n * max_new:.6f}")

        a = a_new

    # --------------------------------------------------  Phase 3 – sub‑gradient on true max
    print("--- Phase 3: sub‑gradient descent on true max ---")
    a = best_seq.copy()
    sub_iters = 40000
    for sub_it in range(1, sub_iters + 1):
        if time.time() > deadline:
            break
        conv = _compute_convolution(a, L)
        w = _submax_weights(conv)
        grad = _gradient_from_weights(a, w, L)

        step = 0.5 / np.sqrt(sub_it)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max = max_new
            best_seq = a_new.copy()
            print(f"Sub‑grad it {sub_it:5d} improved max={max_new:.9e}  score={2 * n * max_new:.6f}")

        a = a_new

    # --------------------------------------------------  Phase 4 – simulated annealing moves
    print("--- Phase 4: random annealing moves ---")
    a_cur = best_seq.copy()
    cur_max = best_max
    # temperature linearly decays from 10 % of current max to 0.1 % of it
    temp_start = 0.10 * cur_max
    temp_end = 0.001 * cur_max

    while time.time() < deadline:
        # linear temperature schedule based on elapsed proportion
        elapsed = time.time() - start
        frac = min(1.0, elapsed / (budget_s * 0.9))
        temp = temp_start * (1 - frac) + temp_end * frac

        # propose a tiny mass transfer
        i = np.random.choice(n, p=a_cur)                 # source favouring larger mass
        j = np.random.randint(0, n)
        if i == j:
            continue
        # transfer up to a few percent of the source mass, scaled by temperature
        delta = np.random.rand() * min(0.02 * a_cur[i], temp)
        if delta <= 0:
            continue

        a_new = a_cur.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:  # numeric safety
            continue

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()

        # Metropolis acceptance
        if max_new < cur_max or np.random.rand() < math.exp(-(max_new - cur_max) / temp):
            a_cur = a_new
            cur_max = max_new
            if max_new < best_max:
                best_max = max_new
                best_seq = a_new.copy()
                print(f"Anneal move improved max={max_new:.9e}  score={2 * n * max_new:.6f}")

    # --------------------------------------------------  Final scaling
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()
    # Logging final score (optional – does not affect the returned value)
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception:
        final_score = float('nan')
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) "
          f"after {time.time() - start:.2f}s ---")
    return final_seq
