        beta = rng.uniform(0.0005, 0.005)  # small values

        # normalize
    elif mode == "random":
        a = rng.random(n)
    else: # best

"""
Hybrid L‑BFGS + deterministic influence‑based hill climbing.

The search works directly on the probability simplex (∑ a_i = 1, a_i ≥ 0).
* L‑BFGS minimizes a smooth approximation of max⁡(a⋆a) using a
  continuation schedule (λ decreasing geometrically).  Several random
  restarts with simple families (exponential, power‑law, edge‑biased,
  fully random) explore the space.
* After each L‑BFGS run a fast “influence” hill‑climb moves mass from
  indices that contribute most to the current convolution peak to those
  that contribute least.  The move size is adapted automatically.
* The best sequence seen is refined with a final deep L‑BFGS pass and a
  polishing influence‑climb before the time budget expires.

All numbers are non‑negative; the returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of a with itself (full length 2·len(a)‑1) using FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * a.size - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a⋆a) for a probability vector (∑a = 1).
    This equals the value returned by ``evaluate_sequence`` (denominator = 1).
    """
    conv = conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (the log‑soft‑max coordinates).

    Parameters
    ----------
    y : ndarray
        Unconstrained parameters; a = softmax(y).
    n : int
        Length of the probability vector.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for convolution and the auxiliary
        convolution needed for the gradient.
    lam_factor : float
        Multiplicative factor for the smoothing width λ = max_c·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 200,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list (geometric schedule) that
    gradually reduces the smoothing of the max‑operator.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def influence_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    step_factor: float = 0.20,
    max_iter: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves probability mass from the index
    that contributes most to the current convolution peak to the index
    that contributes least.  The move size is a fraction of the source
    mass and is adaptively reduced if a step fails.
    """
    best = a.copy()
    best_score = cur_score
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full auto‑convolution and identify the maximal value(s)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_b = conv.max()
        K = np.where(conv >= max_b - 1e-12)[0]   # indices attaining the max

        # compute influence(i) = max_{k∈K} a_{k-i}
        influence = np.zeros(n)
        for k in K:
            start_i = max(0, k - (n - 1))
            end_i = min(k, n - 1)
            i_range = np.arange(start_i, end_i + 1, dtype=int)
            j_vals = best[k - i_range]               # a_{k-i}
            influence[start_i : end_i + 1] = np.maximum(
                influence[start_i : end_i + 1], j_vals
            )

        src = np.argmax(influence)          # source index (large influence)
        tgt = np.argmin(influence)          # target index (small influence)

        if src == tgt or best[src] <= 0.0:
            # fall back to a tiny random move
            src = np.random.randint(0, n)
            tgt = np.random.randint(0, n)
            if src == tgt or best[src] <= 0.0:
                continue

        delta = step_factor * best[src]     # move a fraction of the source mass
        if delta <= 0.0:
            continue

        candidate = best.copy()
        candidate[src] -= delta
        candidate[tgt] += delta

        new_score = fast_score(candidate, fft_len, n)
        if new_score < best_score:
            best = candidate
            best_score = new_score
            # slight increase of step size after a success
            step_factor = min(0.5, step_factor * 1.02)
        else:
            # reject and shrink step size
            step_factor = max(0.01, step_factor * 0.70)

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low

    budget (seconds) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # problem size – inherit from a previous construction if available
    # --------------------------------------------------------------
    n = 6000
    try:

            arr = np.asarray(prev, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                n = max(n, int(arr.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise best sequence (use previous result when present)

    if globals().get("height_sequence_1") is not None:
        best_a = np.asarray(globals()["height_sequence_1"], dtype=float)
        if best_a.size < n:

            best_a = np.concatenate([best_a, pad])
        else:
            best_a = best_a[:n]
        best_a = np.maximum(0.0, best_a)
        best_a = best_a / best_a.sum()
    else:
        # edge‑biased start (≈80 % mass on the ends)
        m_edge = max(1, int(n * 0.03))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        best_a = np.full(n, a_center)
        best_a[:m_edge] = a_edge
        best_a[-m_edge:] = a_edge
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 0.001, num=30)   # continuation schedule
    maxiter_stage = 200                              # L‑BFGS iterations per stage
    iteration = 0

    while time.time() < deadline - 5.0:
        mode = rng.choice(["exponential", "power", "edge", "random", "best"])

        if mode == "exponential":
            beta = rng.uniform(5e-4, 4e-3)
            dist = np.minimum(np.arange(n), n - 1 - np.arange(n))

            a_edge_val = rng.uniform(5.0, 15.0) / n
            a_center_val = 1.0 / n
            a0 = np.full(n, a_center_val)
            a0[:m_edge] = a_edge_val
            a0[-m_edge:] = a_edge_val
        elif mode == "random":
            a0 = rng.random(n)
        else:  # perturb the current best
            a0 = best_a + rng.normal(scale=0.01, size=n)

        a0 = np.maximum(0.0, a0)
        a0 = a0 / a0.sum()
        y0 = np.log(a0 + 1e-12)

        # ----- L‑BFGS with continuation -----
        y_opt = schedule_optimize(

            deadline,
