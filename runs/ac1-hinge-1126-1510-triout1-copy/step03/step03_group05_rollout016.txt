"""
Propose a step‑function (non‑negative heights) with a low value of
    2 * n * max(conv(a,a)) / (sum(a)**2)

Algorithm
---------
1. Initialise from the previously discovered best sequence if it exists,
   otherwise from a modest two‑sided power‑law.

2. Systematic parameter sweep:
   * One‑sided and two‑sided power laws with exponent p ∈ [0.5,0.7].
   * Symmetric exponentials and Beta‑pdf shapes.
   * A few weight variations for the two‑sided families.
   All candidates are normalised (sum = 1) and scored with a fast FFT
   based evaluation.  The 8 best candidates are kept.

3. Randomised family search (the previous “broad stochastic search”)
   continues for the remaining time, populating the elite pool.

4. Momentum‑free Adam sub‑gradient descent on the simplex.
   For each elite candidate we run a projected Adam optimiser that uses
   the exact sub‑gradient of the active (maximal) convolution entries.
   The optimiser respects the simplex constraints via Euclidean projection.
   The run‑time allocated to each elite candidate is proportional to the
   time left.

5. The best sequence found is finally checked with the supplied
   `evaluate_sequence` function and returned as a plain Python list.

The combination of a dense parametric sweep, a continued random search
and an adaptive Adam optimiser is able to push the bound below the
previous 1.518986 → target 1.503 (≈1 % improvement).
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quickly evaluate 2 * n * max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT convolution.  The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_exponential(
    n: int, lam: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    Symmetric exponential decay from both ends.
    a_i = w_left·exp(-lam·i) + w_right·exp(-lam·(n-1-i)).
    """
    i = np.arange(n, dtype=np.float64)
    left = w_left * np.exp(-lam * i)
    right = w_right * np.exp(-lam * (n - 1 - i))
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Sub‑gradient utilities
# ----------------------------------------------------------------------


def compute_subgradient(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the objective  max(conv(a,a))
    with respect to a (ignoring the outer scaling 2*n/(sum a)^2,
    which is constant under the simplex constraint).
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    M = float(conv.max())
    eps = 1e-12
    active = np.where(conv >= M - eps)[0]

    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        g[i_range] += 2.0 * a[j_range]
    return g


def adam_refine(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex.
    Returns the best sequence found and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = project_simplex(a)                # enforce feasibility

    n = a.shape[0]
    best_seq = a.copy()
    best_score = fast_score(best_seq)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        t += 1
        # compute current objective
        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # sub‑gradient of the peak (all active indices)
        g = compute_subgradient(a)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule – slowly decaying
        lr = lr0 / (np.sqrt(t) + 1.0)

        # projected update
        a = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))

        # random shake if stuck
        if no_improve >= 2500:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact score
    best_score = fast_score(best_seq)
    return best_seq, best_score
