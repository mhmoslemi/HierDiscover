"""
LP‑based iterative refinement for the step‑function bound.

The evaluation metric
    E(a) = 2·n·max(a*a) / (sum a)²
is homogeneous, therefore we keep the sequence on the probability simplex
(sum = 1).  Starting from a known good sequence (or the uniform guess) we
repeat the following steps until the time budget expires:

1.  Given the current sequence f and its maximal self‑convolution M,
    solve the linear program

        maximise   1·g  (i.e. sum(g))
        subject to  f ∗ g ≤ M   (component‑wise)
                    g ≥ 0.

    The constraint matrix is the Toeplitz convolution matrix of f; it
    is built once (row/col pattern) and the data vector is f tiled n
    times.  `scipy.optimize.linprog` with the HiGHS solver solves the LP
    quickly even for n≈2000.

2.  The LP returns a vector g₀ with a larger total mass.
    Normalise it to the simplex   g = g₀ / sum(g₀).

3.  Mix the current iterate with g:
        f₊ = (1−t)·f + t·g
    trying t = 1, 0.5, 0.25, 0.125, 0.0625.
    Accept the first mixture that strictly reduces M
    (the maximal convolution, evaluated by FFT).

4.  Occasionally (every few LP steps) perform a cheap greedy hill‑climb
    that transfers a few percent of mass between two coordinates while
    keeping the sum fixed.

5.  When the time budget is near its end, run a longer hill‑climb
    to polish the final sequence.

The routine prints intermediate scores and returns the best list of
non‑negative heights it found.  It respects the 1000 s budget and never
returns an empty or invalid sequence.
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.sparse


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def maxconv_fft(a: np.ndarray, fft_len: int) -> float:
    """Maximum value of the linear convolution a*a (full length)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return float(conv[: 2 * a.shape[0] - 1].max())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the evaluation metric (sum(a) is assumed 1)."""
    return 2.0 * n * maxconv_fft(a, fft_len)


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy local search: move a small random fraction of mass from one
    coordinate to another, keeping the sum = 1.  Accept only improving
    moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

    Search for a non‑negative step sequence with a low evaluation score.
    Returns the sequence as a list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            n = init.size
            s = init.sum()
            f = init / s if s > 0 else np.full(n, 1.0 / n)
    else:

    fft_len = next_pow2(2 * n - 1)
    best_f = f.copy()
    best_M = maxconv_fft(best_f, fft_len)          # current max convolution
    best_score = 2.0 * n * best_M
    print(f"[init] n={n}, maxconv={best_M:.6e}, score={best_score:.6f}")
