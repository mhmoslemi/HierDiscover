                best_local_seq = a_new.copy()
                cur_score = 2 * n * max_new

                    global_best_score = cur_score
                    global_best_seq = a_new.copy()
                    global_best_n = n
                    print(f"Hill climb improved to global best score={cur_score:.6f}")
                a_current = a_new

    final_score = evaluate_sequence(final_seq)
    print(f"--- Finished: best overall score = {final_score:.6f} using n={global_best_n} ---")
    return final_seq

"""
Propose a step‑function (non‑negative vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm works in several stages and tries a few different
sequence lengths.  For each length we

1. generate a bunch of simple parametric candidates (power‑law,
   exponential, U‑shaped, Beta, arcsine, triangular, sinusoid) and
   keep the best one (lowest max–convolution);
2. run Adam on a smoothed max (soft‑max of the convolution);
3. switch to a true sub‑gradient descent on the max;
4. finish with a cheap random hill‑climbing that occasionally mixes
   a fresh random Dirichlet vector.

All updates stay on the probability simplex (L1‑norm = 1).  The
best sequence obtained over the explored lengths is finally scaled
to the normalisation required by ``evaluate_sequence`` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights approximating max(conv)."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def active_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set where conv is (almost) maximal."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # should not happen
        w[conv.argmax()] = 1.0
    return w


def gradient_from_weights(a: np.ndarray, w: np.ndarray,
                         phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of <w, a*a> w.r.t. a.
    Implements 2·correlation(w, a) using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    # FFT of reversed a (zero‑padded) = conj(fft_a) * phase
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(w, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]

        phase = np.exp(2j * np.pi * np.arange(L) / L)

        # 1) Parametric sweep – generate a quick pool of candidates
        xs = np.arange(n, dtype=np.float64)

        _update_candidate(sine)

        # fallback to uniform if nothing produced
        if best_local_seq is None:
            best_local_seq = np.full(n, 1.0 / n, dtype=np.float64)
            best_local_max = compute_convolution(best_local_seq, L).max()

        # also try the global construction if length matches
        if init_global is not None and init_global.size == n:
            _update_candidate(init_global)

        # --------------------------------------------------------------
        # Record the initial candidate
        cur_score = 2 * n * best_local_max
        if cur_score < global_best_score:
            global_best_score = cur_score
            global_best_seq = best_local_seq.copy()
            global_best_n = n
            print(f"  initial best for n={n}: score={cur_score:.6f}")

        # --------------------------------------------------------------
        # 2) Adam descent on the smoothed max
        a = best_local_seq.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        lr0 = 0.10
        max_adam_iters = 120000
        it = 0
        alpha_start, alpha_end = 4.0, 200.0

        while time.time() < deadline and it < max_adam_iters:
            it += 1
            # --- convolution ---
            fft_a = np.fft.fft(a, n=L)
            conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
            cur_max = conv.max()

            # --- soft‑max weights ---
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
            w = softmax_weights(conv, alpha)

            # --- gradient ---
            grad = gradient_from_weights(a, w, phase, L)

            # --- Adam update on the simplex ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step
            a_new = project_simplex(a_new)

            # evaluate the new point
            fft_a_new = np.fft.fft(a_new, n=L)
            conv_new = np.real(np.fft.ifft(fft_a_new * fft_a_new))[: 2 * n - 1]
            max_new = conv_new.max()

            if max_new < best_local_max:
                best_local_max = max_new
                best_local_seq = a_new.copy()
                cur_score = 2 * n * max_new
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_seq = a_new.copy()
                    global_best_n = n
                    print(f"  Adam it {it:5d}: new global best score={cur_score:.6f}")

            a = a_new

        # --------------------------------------------------------------
        # 3) Sub‑gradient descent on the true max
        a = best_local_seq.copy()
        sub_it = 0
        max_sub_iters = 120000
        while time.time() < deadline and sub_it < max_sub_iters:
            sub_it += 1
            fft_a = np.fft.fft(a, n=L)
            conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
            cur_max = conv.max()

            w = active_weights(conv)               # uniform on active set
            grad = gradient_from_weights(a, w, phase, L)

            step = 0.5 / np.sqrt(sub_it)           # diminishing step
            a_new = a - step * grad
            a_new = project_simplex(a_new)

            fft_a_new = np.fft.fft(a_new, n=L)
            conv_new = np.real(np.fft.ifft(fft_a_new * fft_a_new))[: 2 * n - 1]
            max_new = conv_new.max()

            if max_new < best_local_max:
                best_local_max = max_new
                best_local_seq = a_new.copy()
                cur_score = 2 * n * max_new
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_seq = a_new.copy()
                    global_best_n = n
                    print(f"  Sub‑grad it {sub_it:5d}: new global best score={cur_score:.6f}")

            a = a_new

        # --------------------------------------------------------------
        # 4) Cheap random hill‑climbing (with occasional random mixing)
        a_cur = best_local_seq.copy()
        hill_counter = 0
        while time.time() < deadline:
            hill_counter += 1
            # occasional random mixing to escape plateaus
            if hill_counter % 2000 == 0:
                rnd = np.random.gamma(1.0, 1.0, size=n)
                rnd /= rnd.sum()
                sigma = 0.02   # mixing weight
                a_new = (1.0 - sigma) * a_cur + sigma * rnd
                a_new = project_simplex(a_new)
            else:
                # choose a source index proportionally to its mass
                i = np.random.choice(n, p=a_cur)
                j = np.random.randint(0, n)
                if i == j:
                    continue
                # move a small fraction of mass from i to j
                delta = np.random.rand() * 0.05 * a_cur[i]   # up to 5 %
                if delta <= 0:
                    continue
                a_new = a_cur.copy()
                a_new[i] -= delta
                a_new[j] += delta
                if a_new[i] < 0:
                    continue   # safety

            fft_a_new = np.fft.fft(a_new, n=L)
            conv_new = np.real(np.fft.ifft(fft_a_new * fft_a_new))[: 2 * n - 1]
            max_new = conv_new.max()

            if max_new < best_local_max:
                best_local_max = max_new
                best_local_seq = a_new.copy()
                cur_score = 2 * n * max_new
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best_seq = a_new.copy()
                    global_best_n = n
                    print(f"  Hill‑climb improved to global best score={cur_score:.6f}")
                a_cur = a_new
            else:
                a_cur = a_new

        # ------------------------------------------------------------------
        # End of optimisation for this length
        print(f"Finished length {n}: best score for this n = {2*n*best_local_max:.6f}")

    # ----------------------------------------------------------------------
    # If nothing was found (should never happen) fall back to a uniform vector
    if global_best_seq is None:
        n = 5000
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_n = n

    # Final scaling to the norm required by evaluate_sequence
    scaled_seq = (global_best_seq * math.sqrt(2.0 * global_best_n)).tolist()
    final_score = evaluate_sequence(scaled_seq)  # just for logging
    print(f"\n--- Finished overall: best score = {final_score:.6f} "
          f"with n = {global_best_n} ---")
    return scaled_seq
