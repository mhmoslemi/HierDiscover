"""
Propose a low‑score step‑function sequence.

Approach
--------
* Work on the probability simplex (∑a_i = 1) and represent a_i = softmax(y_i).
* Minimise a smooth approximation of the infinity‑norm of the self‑convolution
  using a continuation (log‑sum‑exp) schedule and L‑BFGS.
* Perform many random restarts with diverse initialisations (random,
  exponential, power‑law, edge‑heavy, Sidon‑like, and perturbations of the
  current best).
* After every L‑BFGS run apply a *targeted* hill‑climb that preferentially
  moves mass away from the indices that currently dominate the convolution
  maximum.
* When the time budget is nearly exhausted, run a fine‑grained L‑BFGS
  continuation followed by a short sub‑gradient descent (using the exact
  max‑convolution index) and a final generic hill‑climb.
* The best sequence found is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector (sum=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    The smoothing parameter is lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z  # d smooth_max / d conv_i

    # g_vec = derivative of smooth_max w.r.t a (vector of length n)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Simple greedy Sidon set (pairwise sums distinct)
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums:
                    conflict = True
                    break
                if (candidate * 2) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    rho = np.where(u > cssv / ind)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Normalise to eliminate tiny numerical drift
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def hill_climb_targeted(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially moves mass away from the indices
    contributing to the current maximum of the convolution.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute full convolution and the index of its maximum
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each i to conv[k_max] (product a[i] * a[k_max-i])
        i_range = np.arange(n)
        j = k_max - i_range
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[i_range[mask]] * best[j[mask]]
        total_contrib = contrib.sum()
        if total_contrib == 0:
            # fall back to random hill‑climb
            i, j = rng.integers(0, n, size=2)
        else:
            # pick i with probability proportional to its contribution
            probs = contrib / total_contrib
            i = rng.choice(n, p=probs)

            # pick a destination j uniformly (different from i)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

        # move a fraction of the mass at i towards j
        delta = best[i] * rng.random() * 0.08  # up to 8 % of a[i]
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert move
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    step0: float = 0.3,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on the exact max‑convolution objective.
    Uses the sub‑gradient at the current maximising index.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # compute convolution and its maximiser
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient: g_i = 2 * a_{k-i} (if index valid, else 0)
        idx = np.arange(n)
        j = k - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[j[mask]]

        # step size schedule
        step = step0 / np.sqrt(it)

        # gradient step followed by projection onto the simplex
        a_candidate = project_simplex(best - step * g)

        cand_score = fast_score(a_candidate, fft_len, n)
        if cand_score < best_score:
            best = a_candidate
            best_score = cand_score
        else:
            # occasional random perturbation to escape shallow plateaus
            if rng.random() < 0.004:
                pert = rng.normal(scale=0.02, size=n)
                a_candidate = project_simplex(best + pert)
                cand_score = fast_score(a_candidate, fft_len, n)
                if cand_score < best_score:
                    best = a_candidate
                    best_score = cand_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)

    # leave a small safety margin for the final evaluation
    deadline = time.time() + budget_s - 5.0

    # --------------------------------------------------------------
    # problem size – try a larger n than the previous baseline
    # --------------------------------------------------------------
    n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        try:
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 2000:
                n = max(n, int(tmp.size))
        except Exception:
            pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise with the uniform distribution (baseline)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # sum = 1
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)            # representation in soft‑max space
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    # continuation schedule – geometric decay of the smoothing factor
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=30))
    maxiter_per_stage = 500                     # more iterations per stage

    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]

    # -------------------------- main stochastic search --------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            # perturb current best on the log‑space (softmax logits)
            y0 = best_y + rng.normal(scale=0.06, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- targeted hill‑climb (short burst) -----
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb_targeted(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final refinement phase (more delicate)
    # --------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-7, num=40))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # ----- sub‑gradient descent on the exact max -----
    if time.time() < deadline:
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            deadline,
            step0=0.4,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[sub‑grad] fast score = {best_score:.6f}")

    # ----- final generic hill‑climb with remaining time -----
    if time.time() < deadline:
        best_a, best_score = hill_climb_targeted(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
