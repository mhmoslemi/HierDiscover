"""
Propose a low‑score step‑function.

The search consists of three stages

1️⃣  Random family exploration – generate a few hundred candidates from
    several analytic families (power‑law, exponential, two‑sided power‑law,
    arcsine, cosine, Gaussian).  The best five (by a fast FFT‑based score)
    are kept as seeds.

2️⃣  Smooth‑max Adam optimisation – each seed is refined with a
    gradient‑based method.  The variables are the log‑weights `v` of a
    soft‑max parametrisation `a_i = exp(v_i)/∑exp(v)`, which guarantees
    non‑negativity and the normalisation ∑a_i=1 automatically.

    The objective is the *soft‑max* of the self‑convolution
        L(v) = (1/γ)·log Σ_k exp(γ·(a*a)_k)
    which approaches the true peak `max_k (a*a)_k` as γ grows.
    For a given `γ` the gradient can be written analytically:

        w_k = exp(γ·c_k) / Σ_j exp(γ·c_j)          (c = a*a)
        g_i = 2· Σ_k w_k·a_{k-i}  –  w_{2i}·a_i

    The first term is a correlation of `w` with `a`; it is evaluated
    efficiently with FFTs.  The gradient w.r.t. the log‑weights is
    `∇_v L = a ∘ (g – ⟨a,g⟩)`.  Adam (β₁=0.9, β₂=0.999) is used
    with a slowly decaying learning‑rate and a linear schedule
    `γ(t)` from 5 → 40.

    The optimisation runs until the global time budget (≈1000 s) is
    exhausted, periodically storing the best sequence seen.

3️⃣  Final verification – the best sequence found is evaluated with the
    official `evaluate_sequence` function and returned as a plain list.

All operations are vectorised NumPy calls; the total runtime fits comfortably
within the 1000 s limit on a single CPU.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using FFT‑based convolution.  `seq` is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    L = next_pow2(2 * n)
    Fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
    max_conv = float(conv.real.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, len(v) + 1)) > 0)[0]
    if len(rho) == 0:
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of analytic sequences (used for the random initial pool)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^{-1/2}, i.e. a_i ≈ 1/√i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann (cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian window of width `sigma = sigma_frac * n`."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)

def adam_optimize(init_a: np.ndarray,
                  time_budget: float,
                  global_start: float) -> np.ndarray:
    """
    Refine `init_a` with Adam on the smooth‑max of the self‑convolution.
    Returns the best sequence (normalised to sum = 1) seen during the run.
    """
    n = init_a.shape[0]
    # FFT length large enough for linear correlation (a length n, w length 2n‑1)
    L = next_pow2(3 * n)

    # initialise log‑weights
    v = np.log(np.maximum(init_a, 1e-12))
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                     # base learning rate
    iteration = 0

    # smooth‑max schedule
    gamma_start = 5.0
    gamma_end = 40.0
    gamma_iters = 20000            # linear ramp length

    best_local = init_a.copy()
    best_local_score = fast_score(best_local)

    # main optimisation loop
    while time.time() < global_start + time_budget - 0.5:
        iteration += 1

        # ----- soft‑max to obtain a (∑a = 1, a ≥ 0) -----
        max_v = np.max(v)
        a_exp = np.exp(v - max_v)
        a = a_exp / a_exp.sum()

        # ----- self‑convolution (linear) -----
        Fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]

        # ----- smooth‑max weights -----
        gamma = gamma_start + (gamma_end - gamma_start) * min(iteration / gamma_iters, 1.0)
        # stabilise the soft‑max
        exp_arg = gamma * conv_full - np.max(gamma * conv_full)
        w_raw = np.exp(exp_arg)
        w = w_raw / w_raw.sum()                    # shape (2n‑1,)

        # ----- correlation a ⋆ w (linear, via FFT) -----
        Fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(Fa) * Fw, n=L)[:n]   # length n, lags 0…n‑1

        # ----- sub‑gradient of the smooth‑max -----
        idx2 = 2 * np.arange(n)                 # indices 0,2,4,…,2n‑2  (always in range)
        correction = w[idx2] * a
        grad_a = 2.0 * corr_full - correction   # shape (n,)

        # ----- chain rule through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_v = a * (grad_a - avg)             # shape (n,)

        # ----- Adam parameter updates -----
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v2_hat = v2 / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + iteration * 1e-4)     # gentle decay
        v -= lr * m_hat / (np.sqrt(v2_hat) + eps)

        # ----- occasional scoring -----
        if iteration % 500 == 0:
            cur_score = fast_score(a)
            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = a.copy()
                print(f"[adam] it {iteration:6d}  score {cur_score:.6f}")

    return best_local

    except Exception:
        n_init = 8000
        init = generate_arcsine_like(n_init)

    init = np.maximum(init, 0.0)
    if init.sum() < 1e-12:
        init = np.full(n_init, 1.0 / n_init, dtype=np.float64)
    else:
        init = init / init.sum()

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Random family exploration (collect a small elite set)
    # ------------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = ["power", "exp", "twosided", "arcsine", "cosine", "gaussian"]
    elite = [(best_seq, best_score)]          # list of (seq,score)
    max_elite = 5

    # allocate ~250 s for the random pool, leave the rest for refinement
    random_deadline = deadline - 250.0
    while time.time() < random_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.30, 0.80)
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_cosine_window(n)

        elif family == "gaussian":
            sigma_frac = random.uniform(0.07, 0.30)
            cand = generate_gaussian_window(n, sigma_frac)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[rand] new global best ({family}) n={n} score={sc:.6f}")

        # maintain elite list
        if len(elite) < max_elite:
            elite.append((cand, sc))
            elite.sort(key=lambda x: x[1])
        elif sc < elite[-1][1]:
            elite[-1] = (cand, sc)
            elite.sort(key=lambda x: x[1])

    # ------------------------------------------------------------------
    # 2️⃣  Adam‑based refinement of the elite candidates
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and len(elite) > 0:
        per_seed = remaining / len(elite)

        for idx, (seed_seq, seed_score) in enumerate(elite):
            print(f"[adam] start refinement of seed {idx+1}/{len(elite)} (len={seed_seq.size})")
            refined = adam_optimize(seed_seq, per_seed, start)
            refined_score = fast_score(refined)
            if refined_score < best_score:
                best_score = refined_score
                best_seq = refined.copy()
                print(f"[adam] new global best after seed {idx+1} score={refined_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"[final] best reported score = {final_score:.6f} (length={len(final_seq)})")
    return final_seq
