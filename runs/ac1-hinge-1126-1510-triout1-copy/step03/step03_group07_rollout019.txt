"""
Propose a step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where ``conv`` is the linear convolution of a non‑negative vector with itself.
The returned list is scaled so that ``sum(sequence) = sqrt(2·n)``,
exactly as required by ``evaluate_sequence``.

The algorithm consists of the following stages (all respecting the supplied
time budget):

1. **Initialisation**
   – If a previous construction is available via the global variable
     ``height_sequence_1`` it is used; otherwise a modest length (≈6000) is
     chosen.
   – A diverse pool of analytic families (uniform, arcsine, power‑law,
     exponential, logistic, sine‑squared and a Fourier‑based “square‑root of
     uniform” construction) is generated, normalised to the probability simplex
     and the best one (smallest max‑convolution) is kept.
   – A few random Dirichlet seeds are also evaluated.

2. **Smoothed‑max Adam optimisation**
   – The hard ``max`` is replaced by a soft‑max with a temperature parameter
     ``α``.  Gradient of the smooth objective is obtained analytically via FFT
     (correlation) and Adam updates are performed on the simplex.
   – The smoothing parameter is increased from ``α≈2`` to ``α≈400`` so that the
     iterate approaches a true local optimum of the original max‑objective.

3. **Sub‑gradient descent on the true max**
   – Once the smooth phase has finished the active set (indices whose
     convolution value is within numerical tolerance of the current maximum)
     is formed and a sub‑gradient (uniform weight on the active set) is used.
   – A diminishing step size ``η_t = 0.6 / sqrt(t)`` is applied together with a
     projection onto the simplex.

4. **Random‑restart polishing**
   – Several cheap restarts are performed: the current best vector is
     perturbed slightly and the Adam+sub‑gradient phases are rerun with a
     reduced iteration budget.  This helps to escape shallow plateaus.

5. **Simulated‑annealing hill‑climbing**
   – Tiny mass‑transfer moves (up to 5 % of the source entry) are repeatedly
     tried.  Improvements are always accepted; occasional uphill moves are
     accepted with a Boltzmann probability to encourage exploration.

All heavy lifting (convolution, gradients) uses ``np.fft`` (complexity
``O(L log L)`` per iteration).  The routine monitors the wall‑clock and stops
short of the supplied budget, returning the best sequence found.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(
    a: np.ndarray, weights: np.ndarray, phase: np.ndarray, L: int
) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase  # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w


def _score_of(a: np.ndarray, n: int, L: int) -> float:
    """Return the evaluation score (2·n·max(conv(a,a))) for an L1‑normalised a."""
    conv = _compute_convolution(a, L)
    return float(2.0 * n * conv.max())


def _optimize_from(
    a_start: np.ndarray,
    n: int,
    L: int,
    phase: np.ndarray,
    deadline: float,
    best_seq_ref: list,
    best_max_ref: list,
    max_adam_iters: int = 80000,
    max_sub_iters: int = 80000,
):
    """
    Adam optimisation on a smoothed max followed by sub‑gradient descent.
    ``best_seq_ref`` and ``best_max_ref`` are mutable one‑element containers
    that store the globally best sequence and its convolution maximum.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12
    it = 0
    alpha_start, alpha_end = 2.0, 400.0

    # --------------------------   Adam on smooth max   --------------------
    while time.time() < deadline and it < max_adam_iters:
        it += 1
        conv = _compute_convolution(a, L)

        # soft‑max weighting
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_adam_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth objective
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # check improvement of the *hard* max
        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max_ref[0]:
            best_max_ref[0] = max_new
            best_seq_ref[0] = a_new.copy()
            print(
                f"  Adam it {it:6d} improved max={max_new:.8e}  "
                f"global score={2.0 * n * max_new:.6f}"
            )
        a = a_new

    # ---------------------   Sub‑gradient on true max   --------------------
    sub_it = 0
    while time.time() < deadline and sub_it < max_sub_iters:
        sub_it += 1
        conv = _compute_convolution(a, L)
        w = _submax_weights(conv)
        grad = _gradient_from_weights(a, w, phase, L)

        step_size = 0.6 / np.sqrt(sub_it)
        a_new = a - step_size * grad
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        if max_new < best_max_ref[0]:
            best_max_ref[0] = max_new
            best_seq_ref[0] = a_new.copy()
            print(
                f"  Sub‑grad it {sub_it:6d} improved max={max_new:.8e}  "
                f"global score={2.0 * n * max_new:.6f}"
            )
        a = a_new

    return a

def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget (seconds) and returns a plain
    Python list of floats (scaled to sqrt(2·n) as required by the evaluator).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # safety margin

    # ---- Determine the length ------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # may be defined globally
        n = len(init_seq)
    except Exception:
        n = 6000  # a reasonable default if no previous construction is present

    # ---- FFT helper ---------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())           # smallest power of two ≥ 2n‑1
    phase = np.exp(2j * np.pi * np.arange(L) / L)  # used in gradient computation

    # ---- Containers for the best solution ------------------------------------
    best_seq_ref = [None]   # mutable holder for the best probability vector
    best_max_ref = [np.inf]  # holds the minimum max(conv) seen so far

    # ----------------------------------------------------------------------
    # 0.  Global candidate (if any)
    # ----------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        cand = np.array(height_sequence_1, dtype=np.float64)
        cand = cand / cand.sum()
        sc = _score_of(cand, n, L)
        best_seq_ref[0] = cand.copy()
        best_max_ref[0] = sc / (2.0 * n)
        print(f"Initial global candidate: score={sc:.6f}")

    # ----------------------------------------------------------------------
    # 1.  Build a modest pool of analytic families
    # ----------------------------------------------------------------------
    xs = np.arange(n, dtype=np.float64)
    cand_list = []

    # uniform
    cand_list.append(np.full(n, 1.0 / n, dtype=np.float64))

    # arcsine‑like (classic “U” profile)
    x_mid = (xs + 0.5) / n
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    cand_list.append(arcsine / arcsine.sum())

    # power‑law families (both directions)
    for p in [0.0, 0.5, 1.0, 1.5, 2.0]:
        base = (xs + 1) ** p
        cand_list.append(base / base.sum())
        cand_list.append(base[::-1] / base[::-1].sum())

    # exponential families (both directions)
    for lam in [0.0, 0.5, 1.0, 2.0, 3.0, 5.0]:
        exp_shape = np.exp(-lam * xs)
        cand_list.append(exp_shape / exp_shape.sum())
        cand_list.append(exp_shape[::-1] / exp_shape[::-1].sum())

    # logistic‑type (sigmoid) profiles
    for sigma in [n / 10.0, n / 6.0, n / 4.0]:
        logistic = 1.0 / (1.0 + np.exp((xs - n / 2.0) / sigma))
        cand_list.append(logistic / logistic.sum())

    # sine‑squared profile
    sine_sq = np.sin(np.pi * (xs + 0.5) / n) ** 2
    cand_list.append(sine_sq / sine_sq.sum())

    # Fourier‑based “square‑root of a uniform” construction
    try:
        # target: constant vector of length 2n‑1 (the desired convolution)
        b = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)
        b_pad = np.zeros(L, dtype=np.float64)
        b_pad[: 2 * n - 1] = b
        fft_b = np.fft.fft(b_pad)
        sqrt_fft = np.sqrt(fft_b)        # principal square‑root (complex)
        a_tmp = np.real(np.fft.ifft(sqrt_fft))
        a_tmp = a_tmp[:n]
        a_tmp = np.maximum(a_tmp, 0.0)   # discard tiny negatives
        if a_tmp.sum() > 0:
            cand_list.append(a_tmp / a_tmp.sum())
    except Exception:
        pass  # ignore if something goes wrong – it is only an additional seed

    # ----------------------------------------------------------------------
    # Evaluate all deterministic candidates
    # ----------------------------------------------------------------------
    for cand in cand_list:
        if time.time() > deadline:
            break
        sc = _score_of(cand, n, L)
        if sc < 2.0 * n * best_max_ref[0]:
            best_seq_ref[0] = cand.copy()
            best_max_ref[0] = sc / (2.0 * n)
            print(f"  initial pool improved: score={sc:.6f}")

    # ----------------------------------------------------------------------
    # 2.  Random Dirichlet seeds (cheap cheap)
    # ----------------------------------------------------------------------
    rnd_seeds = 0
    while time.time() < deadline - 30 and rnd_seeds < 80:
        rnd = np.random.rand(n)
        rnd = rnd / rnd.sum()
        sc = _score_of(rnd, n, L)
        if sc < 2.0 * n * best_max_ref[0]:
            best_seq_ref[0] = rnd.copy()
            best_max_ref[0] = sc / (2.0 * n)
            print(f"  random seed {rnd_seeds:02d} improved: score={sc:.6f}")
        rnd_seeds += 1

    # ----------------------------------------------------------------------
    # 3.  Core optimisation from the current best (Adam → sub‑grad)
    # ----------------------------------------------------------------------
    _optimize_from(
        best_seq_ref[0],
        n,
        L,
        phase,
        deadline,
        best_seq_ref,
        best_max_ref,
        max_adam_iters=120000,
        max_sub_iters=120000,
    )

    # ----------------------------------------------------------------------
    # 4.  Several cheap restarts (perturb + re‑optimise)
    # ----------------------------------------------------------------------
    while time.time() < deadline - 15:
        # small random perturbation of the best vector
        perturb = np.random.rand(n) * 0.02
        start_vec = best_seq_ref[0] + perturb
        start_vec = _project_simplex(start_vec)
        _optimize_from(
            start_vec,
            n,
            L,
            phase,
            deadline,
            best_seq_ref,
            best_max_ref,
            max_adam_iters=30000,
            max_sub_iters=30000,
        )
        if time.time() > deadline - 5:
            break

    # ----------------------------------------------------------------------
    # 5.  Simulated‑annealing hill‑climbing (tiny mass transfers)
    # ----------------------------------------------------------------------
    a_current = best_seq_ref[0].copy()
    temperature = 1e-3
    decay = 0.99995

    while time.time() < deadline:
        temperature *= decay
        # choose a source proportionally to its mass
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # transfer up to 5 % of the source entry
        delta = np.random.rand() * 0.05 * a_current[i]
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue

        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        score_new = 2.0 * n * max_new

        if score_new < 2.0 * n * best_max_ref[0]:
            best_max_ref[0] = max_new
            best_seq_ref[0] = a_new.copy()
            a_current = a_new
            print(f"  hill‑climb improved score={score_new:.6f}")
        else:
            # occasional uphill acceptance (simulated annealing)
            if np.random.rand() < np.exp(-(score_new - 2.0 * n * best_max_ref[0]) / max(temperature, 1e-12)):
                a_current = a_new

    # ----------------------------------------------------------------------
    # 6.  Final scaling to the required L1‑norm sqrt(2·n)
    # ----------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq_ref[0] * target_sum).tolist()
    final_val = evaluate_sequence(final_seq)  # for logging only
    print(
        f"--- Finished: final score = {final_val:.6f} (n={n}) "
        f"elapsed {time.time() - start:.1f}s ---"
    )
    return final_seq
