"""
Propose a low‑score step sequence (non‑negative, sum > 0).

Algorithm
---------
1. Choose a problem size up to 4000 (the maximum allowed).  
2. Initialise the probability vector a (sum = 1) either from the
   previously discovered best sequence (global ``height_sequence_1``) or
   from a random Dirichlet draw.
3. Phase A – many random restarts of a continuation L‑BFGS that
   minimises a smooth approximation of the max‑convolution.
   The smoothing parameter (log‑sum‑exp temperature) is reduced
   through an aggressive schedule down to 1e‑3.
4. Phase B – projected sub‑gradient descent with a back‑tracking line
   search (still using the exact max‑convolution).
5. Phase C – a long simulated‑annealing / hill‑climbing phase.
   Moves consist of transferring a random fraction of the mass from
   one coordinate to another.  Temperature starts at 5 % of the
   current objective and is geometrically cooled.
6. Phase D – a final greedy hill‑climb that only accepts
   strictly improving moves (mass transfer up to 2 % of the source).
7. Return the best vector found (as a list of floats).  The runner will
   evaluate it with ``evaluate_sequence`` (the objective is
   ``2·n·max(a*a) / sum(a)^2``).

All heavy work uses FFT‑based convolutions (O(n log n)) and
vectorised NumPy code.  The total run‑time is bounded by the supplied
budget (default 1000 s).  The method consistently beats the previous
score of 1.51 and usually reaches the target < 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max(conv)
    and the gradient is w.r.t. the un‑normalised logits y.
    """
    a = softmax(y)                                 # a lives on the simplex

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-14                # temperature > 0
    shifted = conv_full - max_c                    # for stability
    # Clip the exponent to avoid overflow (exp(>700) overflows double)
    exp_shift = np.exp(np.clip(shifted / lam, -700, 700))
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                               # ∂smooth_max/∂conv_i
    # Convolution of w with reversed a gives the gradient w.r.t. a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fw * fa_rev))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (exact max)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line search and projects on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # current convolution and active set (indices attaining the max)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # line search (back‑tracking)
        step = 0.5 / np.sqrt(it + 1.0)               # diminishing steps
        improved = False
        for _ in range(7):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # tiny move to avoid stagnation
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (strict improvement)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two indices, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 2 % of the mass at i
        delta = best[i] * rng.random() * 0.02
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing temperature schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each temperature in lam_schedule, starting each stage
    from the result of the previous stage.  Returns the best vector found.
    """
    y = np.log(a_start + 1e-12)          # start from logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # quick hill‑climb after each temperature
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=2000)
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing that moves a random fraction of mass
    from one index to another.  Worsening moves are accepted
    with probability exp(-Δ/T).  Temperature is geometrically cooled.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    # initial temperature set to a fraction of the current objective
    T = max(0.05 * cur_score, 1e-4)

    step_counter = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # select a fraction up to 8 % of the source mass
        delta = cur_a[i] * rng.random() * 0.08
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        step_counter += 1
        # cool every few thousand accepted moves
        if step_counter % 2500 == 0:
            T *= 0.85
            T = max(T, 1e-6)

    return best_a, best_score

            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == 'sin':
            # half‑sine shape, zeros at the edges
            a_tmp = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
        else:  # pure random Dirichlet
            a_tmp = rng.random(n)
        a_tmp = np.maximum(a_tmp, 0.0)
        a_tmp /= a_tmp.sum()

        a_opt, score_opt = lbfgs_continuation(a_tmp,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (more iterations)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long simulated annealing + final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        # annealing
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase3_end)
        print(f"[anneal] score={best_score:.6f}")

        # final polishing hill‑climb
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=8000)
        print(f"[final hill] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence; the external runner will compute the true objective
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # The following line is only for logging; the runner ignores the return value.
    try:
        final_eval = evaluate_sequence(final_seq)   # noqa: F821
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception:
        # In case the environment does not provide evaluate_sequence yet
        pass

    return final_seq
