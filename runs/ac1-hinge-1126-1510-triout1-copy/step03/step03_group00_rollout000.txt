"""
Propose a low‑score non‑negative step sequence.

The algorithm uses several complementary optimisation phases within a
1000 s time budget:

1. **Continuation L‑BFGS** on a log‑sum‑exp smooth approximation of the
   max‑convolution.  A decreasing smoothing schedule (lam) gradually
   approaches the true max.  Each L‑BFGS run starts from a randomly
   generated distribution; many restarts are performed while time
   permits.

2. **Projected sub‑gradient descent** (Euclidean projection onto the
   probability simplex) with a simple back‑tracking line search.

3. **Exponentiated‑gradient descent** – an alternative simplex‑preserving
   update that often refines the solution after sub‑gradient steps.

4. **Simulated annealing** that moves a fraction of mass between two
   coordinates, accepting occasional worsening moves.

5. **Greedy hill‑climb** (mass‑transfer) as a final polishing step.

All convolutions are performed via FFT for O(n log n) speed.  The
sequence length is chosen up to 5000 (default 3000‑4000) to give the
optimiser enough freedom while remaining fast.  The best sequence
found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns smooth max objective (2·n·smooth_max(conv)) and its gradient
    w.r.t. the logits y (where a = softmax(y)).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12               # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                                 # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (many random restarts)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lambda in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            # occasional numerical failures; just skip this lam
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
        # small hill‑climb after each lambda
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (Euclidean projection)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Direct minimisation of 2·n·max(a*a) via projected sub‑gradient descent.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(6):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # still move a tiny amount to stay away from a stall
            a = simplex_projection(a - step * g)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (KL‑projection)
# ----------------------------------------------------------------------
def exponentiated_gradient(a0: np.ndarray,
                          deadline: float,
                          fft_len_conv: int,
                          fft_len_g: int,
                          n: int):
    """
    Mirror‑descent on the simplex using KL divergence (a ← a·exp(−ηg)).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution and active mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size schedule (decreasing)
        eta = 0.5 / np.sqrt(it + 1.0)

        # exponentiated update, safely clipped
        exp_arg = -eta * g
        exp_arg = np.clip(exp_arg, -50, 50)
        a_new = a * np.exp(exp_arg)
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            # fallback to uniform if underflow occurs
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= a_new_sum

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy hill‑climb: repeatedly transfer a random fraction of mass
    between two coordinates while keeping the total sum fixed.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep move
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature schedule.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02                     # initial temperature

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # up to 12 % of a[i]
        if delta <= 0.0:
            continue

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.90           # geometric cooling
            T = max(T, 5e-5)

    return best_a, best_score

        else:  # gauss
            sigma = rng.uniform(0.15, 0.35) * n
            centre = n / 2
            a_tmp = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)

        a_tmp = np.maximum(a_tmp, 0)
        if a_tmp.sum() == 0:
            a_tmp = np.full(n, 1.0 / n)
        else:
            a_tmp /= a_tmp.sum()

        a_opt, score_opt = lbfgs_continuation(a_tmp,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient + exponentiated‑gradient refinements
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑grad] score={best_score:.6f}")

        eg_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_score = exponentiated_gradient(best_a,
                                                    eg_deadline,
                                                    fft_len_conv,
                                                    fft_len_g,
                                                    n)
        print(f"[exp‑grad] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           deadline)
        print(f"[anneal] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    # --------------------------------------------------------------
    # Return the sequence after final external evaluation (for logging)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
