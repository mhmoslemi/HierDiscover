"""
search_step_function.py

A fresh implementation of the “step‑function” optimiser.

The goal is to minimise

    E(a) = 2·n·max(a*a) / (sum(a)**2)

over non‑negative vectors a of length n.  The functional is homogeneous, so we
restrict the search to the probability simplex (∑a = 1, a ≥ 0).  The optimiser
combines three complementary ingredients:

1. **Heavy‑weight random restarts of L‑BFGS** – a smooth (log‑sum‑exp)
   surrogate of the max‑convolution is minimised from many diverse
   initialisations (power‑law, exponential, edge‑heavy, previous best).
2. **Projected sub‑gradient descent** – direct minimisation of the true
   non‑smooth objective with a diminishing step‑size and exact Euclidean
   projection onto the simplex (the O(n) algorithm of Wang & Carreira‑Perpiñán).
3. **Non‑local simulated annealing** – a final exploration phase that moves a
   random fraction of mass between two coordinates, accepting uphill moves
   with a temperature schedule.

All convolution operations are carried out by FFT (O(n log n)).  The routine
obeys a user supplied time budget (default 1000 s) and prints progress
information.  The best sequence found is returned as a plain Python list.
"""

import time
import math

# Utility functions (no closures, no lambdas)

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of   2·n·max(a*a)   for a distribution already
    normalised to unit sum.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
        { x ≥ 0,  Σ x = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all elements are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) – used by L‑BFGS
# ------------------------------------------------------------

def smooth_obj_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int,
                       fft_len_grad: int,
                       lam_factor: float = 0.30):
    """
    Returns (f, grad) for the smooth surrogate

        f(y) = 2·n·smooth_max( conv( softmax(y) ) ),

    where `smooth_max` is a log‑sum‑exp approximation with smoothing
    parameter λ = lam_factor·max(conv) .
    """
    a = softmax(y)                     # map to simplex

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth max (log‑sum‑exp) -----
    m = conv_full.max()
    lam = max(m * lam_factor, 1e-12)   # ensure positivity
    shifted = conv_full - m            # improve numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = m + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. a (see derivation in the original paper) -----
    w = exp_shift / Z                 # ∂smooth_max / ∂conv_i
    # g_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_grad)
    fa_rev = np.fft.fft(a_rev, fft_len_grad)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # convolution and active set (indices attaining the max)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_grad)
        fa_rev = np.fft.fft(a_rev, fft_len_grad)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        subg = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # diminishing step size, Euclidean projection back onto simplex
        step = step0 / np.sqrt(it + 1.0)
        a = simplex_projection(a - step * subg)

        # keep the best iterate seen so far
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        it += 1

    return best, best_score


# ------------------------------------------------------------
# Greedy hill‑climb (mass transfer between two coordinates)
# ------------------------------------------------------------

def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Simple deterministic hill‑climb that repeatedly shifts a small
    fraction of mass from one coordinate to another while keeping the sum
    fixed.  Only improving moves are accepted.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ------------------------------------------------------------
# Simulated annealing (non‑local search)
# ------------------------------------------------------------

def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Non‑local simulated annealing that moves a random fraction of mass
    from one index to another.  Uphill moves are accepted according to a
    temperature schedule that cools geometrically.
    """
    cur = a_start.copy()
    cur_score = score_start
    best = cur.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)      # initial temperature
    decay = 0.9993                       # cooling factor

    it = 0
    while time.time() < deadline:
        # pick a source with non‑negligible mass
        i = rng.integers(0, n)
        if cur[i] < 1e-12:
            continue

        # distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction of the source mass (up to 25 %)
        max_delta = cur[i] * 0.25
        delta = rng.random() * max_delta
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            # Metropolis rule
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # reject – revert move
                cur[i] += delta
                cur[j] -= delta

        # best ever?
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()

        # temperature decay
        T *= decay
        # occasional reheating to keep the walk alive
        if it % 4000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best, best_score


# ------------------------------------------------------------
# Main entry point required by the judge
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **_ignore) -> list[float]:
    """
    Search for a non‑negative step‑height sequence with a low evaluation
    score.  The routine respects the supplied time budget and returns the
    best sequence found as a plain Python list.
    """
    # ----- deterministic RNG -----
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    # ----- initialise ----------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        n = init_seq.size if init_seq.size > 0 else 2000
        # ensure a reasonable size (too large → slow LP / FFT)
        max_n = 3000
        if n > max_n:
            init_seq = init_seq[:max_n]
            n = max_n
        a0 = init_seq / init_seq.sum() if init_seq.sum() > 0 else np.full(n, 1.0 / n)
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # ----- FFT infrastructure -------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_grad = next_pow2(3 * n - 2)   # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------
    # Phase 1 – many random L‑BFGS runs on the smoothed max
    # ------------------------------------------------------------
    phase1_end = start + 0.45 * budget_s
    lam_schedule = [0.5, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01]
    it = 0
    while time.time() < phase1_end:
        lam = lam_schedule[it % len(lam_schedule)]

        # pick a random start distribution
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0004, 0.015)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:      # edge‑heavy distribution
            edge_frac = rng.uniform(0.007, 0.035)
            m_edge = max(1, int(n * edge_frac))
            a_tmp = np.full(n, 1.0 / n)
            heavy = np.sqrt(n / 2.0) / n
            a_tmp[:m_edge] = heavy
            a_tmp[-m_edge:] = heavy
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # L‑BFGS on smoothed objective
        res = minimize(fun=smooth_obj_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_grad, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        sc_opt = fast_score(a_opt, fft_len_conv, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS {it}] lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # quick hill‑climb to polish the L‑BFGS result
        deadline_hc = min(phase1_end, time.time() + 0.30)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        deadline_hc,
                                        max_iter=800)
        it += 1

    # ------------------------------------------------------------
    # Phase 2 – exhaustive projected sub‑gradient descent
    # ------------------------------------------------------------
    phase2_end = start + 0.75 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase2_end,
                                                  fft_len_conv,
                                                  fft_len_grad,
                                                  n)
        print(f"[sub‑gradient] refined score={best_score:.6f}")

    # ------------------------------------------------------------
    # Phase 3 – simulated annealing for final fine‑tuning
    # ------------------------------------------------------------
    phase3_end = deadline
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] final score={best_score:.6f}")

    # ------------------------------------------------------------
    # Phase 4 – final greedy polishing
    # ------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=2500)

    # sanity check with the provided evaluator (may raise if not imported)
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence returns {eval_val:.6f}")

    return final_a.tolist()
