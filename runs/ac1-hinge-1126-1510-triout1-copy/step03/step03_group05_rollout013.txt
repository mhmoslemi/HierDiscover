"""
Improved search for a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of three phases:

1. **Diverse random‑shape exploration** – we generate many candidates from
   a variety of families (power‑law, exponential, two‑sided power‑law,
   Beta‑pdf, cosine‑modulated and triangular shapes) of several lengths.
   All candidates are normalised to sum = 1 (the objective is homogeneous
   in the scale) and the best few are kept.

2. **Adaptive Adam‑style sub‑gradient refinement** – each elite candidate
   is improved by a projected Adam optimiser on the simplex.
   The sub‑gradient is the sum of the contributions of *all* active
   convolution entries (those attaining the current maximum).  After each
   step we project back onto the probability simplex.

3. **Lightweight simulated‑annealing polishing** – the best sequence after
   the Adam phase is further polished with a simple Metropolis move that
   transfers a random fraction of mass from one index to another while
   preserving the simplex constraints.

All heavy work (convolutions) is performed with an FFT, which makes a
single evaluation O(n log n).  The routine respects the 1000 s time budget,
printing progress information and finally returning the best sequence
as a plain Python list.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2 * n * max(conv(seq, seq)) / (sum(seq) ** 2) using FFT.
    The sequence is assumed non‑negative and already normalised (sum=1),
    so the denominator equals 1 and the formula reduces to 2 * n * max_conv.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (all return raw, un‑normalised arrays)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf (up to a multiplicative constant):
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_cosine_modulated(n: int, amp: float) -> np.ndarray:
    """
    Cosine‑modulated shape:
        a_i = (1-amp) + amp·cos(2π i / n)
    The amplitude is limited to ≤0.5 to keep all entries non‑negative.
    """
    i = np.arange(n, dtype=np.float64)
    base = 1.0 - amp
    seq = base + amp * np.cos(2.0 * np.pi * i / n)
    return seq


def generate_triangular(n: int) -> np.ndarray:
    """
    Simple symmetric triangular shape (linearly increasing then decreasing).
    """
    half = n // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        right = np.arange(half, 0, -1, dtype=np.float64)
    else:
        right = np.arange(half + 1, 0, -1, dtype=np.float64)
    return np.concatenate([left, right])


# ----------------------------------------------------------------------
# Adam‑style sub‑gradient optimiser on the simplex
# ----------------------------------------------------------------------


def subgradient(seq: np.ndarray, eps: float = 1e-12) -> np.ndarray:
    """
    Compute a sub‑gradient of the objective 2·n·max(conv(seq,seq))
    with respect to `seq` (sum(seq)=1).  The returned vector has the same
    shape as `seq`.
    """
    n = seq.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    M = float(conv.max())
    active = np.where(conv >= M - eps)[0]

    g = np.zeros_like(seq)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        g[i_range] += 2.0 * seq[j_range]
    return g


def refine_adam(seq: np.ndarray, time_limit: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent.
    Returns the best sequence found and its fast_score.
    """
    a = project_simplex(seq.copy())
    n = a.shape[0]
    best_seq = a.copy()
    best_score = fast_score(best_seq)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    while time.time() < start_time + time_limit:
        it += 1

        # sub‑gradient of the current max‑convolution entry
        g = subgradient(a)

        # Adam updates (biased but good enough for our purpose)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        # step‑size schedule – slowly decaying
        lr = 0.5 / (math.sqrt(it) + 1.0)

        step = lr * m / (np.sqrt(v) + eps)
        candidate = a - step
        a = project_simplex(candidate)

        cur_score = fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()

        # occasional random jitter to escape plateaus
        if it % 3000 == 0:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))

    best_score = fast_score(best_seq)
    return best_seq, best_score

def refine_annealing(seq: np.ndarray, time_limit: float, start_time: float) -> tuple[np.ndarray, float]:
    """
    Metropolis‑style local search that preserves the simplex.
    A random fraction of mass is moved from one index to another.
    Accepts worse moves with a temperature that decays exponentially.
    """
    a = seq.copy()
    n = a.shape[0]
    cur_score = fast_score(a)
    best_seq = a.copy()
    best_score = cur_score

    # Initialise temperature as a small fraction of the current objective
    temp = max(0.01 * cur_score, 1e-4)
    decay = 0.9995

    while time.time() < start_time + time_limit:
        # pick two distinct indices
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        # transfer a random fraction (up to 30%) of a[i] to j
        frac = random.random() * 0.3
        delta = a[i] * frac

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = fast_score(new_a)

        # Metropolis acceptance
        if new_score < cur_score or random.random() < math.exp((cur_score - new_score) / max(temp, 1e-12)):
            a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_a.copy()

        temp *= decay
        if temp < 1e-6:
            temp = max(0.01 * cur_score, 1e-6)   # occasional reheating

    best_score = fast_score(best_seq)
    return best_seq, best_score

            best_score = sc
            best_seq = seq.copy()
            print(f"[rand] new global best ({fam}) n={n} score={sc:.6f}")

        # maintain elite pool
        if len(elite) < elite_size:
            elite.append((sc, seq.copy()))
        else:
            worst_idx = max(range(len(elite)), key=lambda i: elite[i][0])
            if sc < elite[worst_idx][0]:
                elite[worst_idx] = (sc, seq.copy())

    # sort elite by score (ascending)
    elite.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2) Adam‑style refinement on the elite candidates
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left <= 0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Time exhausted after random phase. Final score = {final_score:.6f}")
        return final_seq

    per_candidate = time_left * 0.65 / len(elite)    # 65 % of remaining time for Adam
    for idx, (sc, seq) in enumerate(elite):
        if time.time() >= deadline:
            break
        print(f"[adam] elite #{idx+1} start (score={sc:.6f}, n={seq.size})")
        refined_seq, refined_score = refine_adam(seq, per_candidate, time.time())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[adam] improved global best → {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Final simulated‑annealing polish (uses the remaining time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 0.1:
        print(f"[anneal] polishing best candidate (n={best_seq.size}) with {remaining:.1f}s left")
        best_seq, best_score = refine_annealing(best_seq, remaining, time.time())
        print(f"[anneal] post‑polish best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final verification with the provided evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")

    return final_seq
