"""
Propose a low‐evaluation step sequence.

Algorithm
---------

* **Initialisation** – Build a short arc‑sine shaped distribution (heavy at the
  ends, light in the middle) and normalise it to the probability simplex.
  If a previous best sequence is present (global variable `height_sequence_1`)
  it is used instead.

* **Smooth optimisation** – Optimise the smoothed max of the auto‑convolution
  using Adam on the soft‑max parametrisation `a = softmax(y)`.  The smoothing
  parameter (relative to the current max) is decreased on a schedule
  `λ ∈ {0.5,0.3,0.2,0.1,0.07,0.04,0.02,0.01,0.005}`.  The Adam loop runs as
  long as time permits, keeping the best feasible vector (according to the
  true max‑convolution).

* **Sub‑gradient refinement** – A classic diminishing‑step sub‑gradient
  method on the exact `max(a*a)` is applied to the best vector found so far.

* **Final greedy hill‑climb** – A cheap local search that repeatedly moves a
  small fraction of mass between two coordinates while preserving the total
  sum.  Only improving moves are kept; this polishes the solution.

All convolutions are performed with FFTs (O(n log n)) and the simplex projection
uses the exact Euclidean algorithm.  The routine respects a 1000 s budget and
prints progress information.  The returned list contains non‑negative numbers
ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained parameters; a = softmax(y).
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths pre‑computed for the convolution and gradient steps.
    lam_factor : float
        Relative smoothing factor (λ = max_c * lam_factor).

    Returns
    -------
    f : float
        2·n·smooth_max(conv)
    grad_y : np.ndarray
        Gradient of f w.r.t. y.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c           # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central n entries correspond to the gradient w.r.t. a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smooth objective
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  lam_schedule: list[float],
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int) -> tuple[np.ndarray, float]:
    """
    Run Adam on the smoothed max objective.

    Returns the best vector a (on the simplex) seen during the run and its
    true fast_score.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    alpha = 0.2          # base learning rate
    t = 0

    # initialise best
    a_best = softmax(y)
    best_score = fast_score(a_best, fft_len_conv, n)

    lam_idx = 0
    lam = lam_schedule[lam_idx]

    while time.time() < deadline:
        # possibly move to a tighter smoothing factor
        if t % 2000 == 0 and t > 0:
            lam_idx = min(lam_idx + 1, len(lam_schedule) - 1)
            lam = lam_schedule[lam_idx]

        # compute objective and gradient
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=lam)
        t += 1

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # periodic evaluation of the true objective
        if t % 100 == 0:
            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                a_best = a.copy()
                print(f"[Adam] iter {t:5d}, lam={lam:.3f}, score={best_score:.6f}")

        # early exit if we are very close to the deadline
        if time.time() > deadline - 0.5:
            break

    return a_best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep best
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb.  Moves a random fraction of one coordinate
    to another while preserving the total sum.  Accepts only improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05  # up to 5 % of mass at i
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # improvement kept
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score

    # Return a plain Python list as required by the evaluation harness
    return best_a.tolist()
