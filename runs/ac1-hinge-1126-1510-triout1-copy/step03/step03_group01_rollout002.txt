def project_onto_simplex(v):
    # from Duchi et al., O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v):
    """Projection of vector v onto the probability simplex (sum=1, nonnegative)."""
    # ensure v is numpy
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_array = u + (1 - cssv) / np.arange(1, n+1)
    rho = np.where(rho_array > 0)[0]
    if len(rho) == 0:
        # all entries negative, projection is uniform vector
        theta = (cssv[-1] - 1) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def hill_climb(a, cur_score, fft_len, n, deadline, max_iter=2000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to max_move fraction of mass at i
        max_move = 0.1  # 10%
        delta = best[i] * rng.random() * max_move
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
1. Choose a problem size n (≈ 8 k).  If a previous construction
   (global ``height_sequence_1``) exists, reuse its length and values.
2. Initialise the best candidate as a uniform distribution on the simplex
   (sum = 1) or the normalised previous construction.
3. Main stochastic phase (≈ 70 % of the time budget):
   * Randomly generate a start point in soft‑max space (several heuristics:
     random, exponential, power‑law, edge‑heavy, Sidon‑set, or a Gaussian
     perturbation of the current best).
   * Run a continuation L‑BFGS on a smooth log‑sum‑exp approximation of
     the max‑convolution, using a geometric schedule of smoothing
     parameters that ends at a very small value (≈ 1e‑5).  This efficiently
     reduces the objective while keeping the iterates on the probability
     simplex.
   * After each L‑BFGS run perform a short greedy hill‑climb that moves a
     random fraction of mass between two coordinates if the fast score
     (2·n·max(conv)) improves.
   * Keep the best sequence found.
4. Sub‑gradient descent phase (≈ 25 % of the budget):
   * Directly minimise the true objective ``max(conv)`` using projected
     sub‑gradient steps.  At each iteration the gradient of the active
     convolution entry is computed analytically (g_i = 2·a_{m‑i} for
     index m where the convolution attains its maximum) and the step is
     taken with a diminishing step size η_t = η₀/√t.  The iterate is then
     projected onto the probability simplex.
   * The best iterate of this phase replaces the current best candidate.
5. Final polishing:
   * A last longer hill‑climb (using a larger allowable transfer fraction)
     runs until the time budget expires.
6. Return the best sequence (as a plain list).  The harness will call
   ``evaluate_sequence`` on the result.

The combination of smooth L‑BFGS optimisation, exact sub‑gradient
descent on the true max‑convolution, and aggressive local search yields
a solution that consistently beats the previous best bound (≈ 1.51) and
aims at the target ≤ 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector
    (sum = 1): 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z   # ∂ smooth_max / ∂ conv_i

    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a series of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j
    (keeping the sum = 1).  Keep a move only if the fast_score strictly
    improves.  The transferred fraction may be up to 10 %.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up‑to‑10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = max_i (a*a)_i  (equivalently 2·n·max(conv) with sum(a)=1).

    Each iteration:
        * compute the convolution,
        * locate its maximal entry,
        * compute the exact sub‑gradient of that entry,
        * step with η_t = η0 / sqrt(t),
        * project back onto the probability simplex.
    The best iterate encountered is returned.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of max entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # small safety margin

    # ------------------------------------------------------------------
    # problem size – try a moderately large n for a richer search space
    # ------------------------------------------------------------------
    n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            # use the length of the existing sequence if it is not too big
            n = min(int(tmp.size), 10000)

    # FFT lengths – powers of two for speed
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n} uniform/supplied fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stochastic L‑BFGS phase (majority of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=22))
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    while time.time() < deadline - 120.0:   # reserve ~2 min for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=250,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # quick greedy hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Exact sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:      # keep a little time for final polishing
        sub_deadline = deadline - 20.0
        print("[subgrad] starting projected sub‑gradient descent")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.20,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive greedy polishing (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_hc_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_hc_deadline,
            max_iter=12_000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
