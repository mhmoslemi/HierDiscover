"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

by a combination of:

1.  A fairly exhaustive parametric sweep (power‑law, exponential,
    double‑exponential, Beta‑densities, arcsine) on a moderately large
    length (≈10 000).  All candidates are projected onto the simplex.

2.  Adam optimisation on a smooth approximation of the max (log‑sum‑exp)
    with an adaptive smoothing schedule and a relatively large learning rate.
    The parameters are kept on the simplex by Euclidean projection.

3.  A sub‑gradient descent on the true ‑∞‑norm using the active set of
    maximal convolution entries.

4.  Guided hill‑climbing moves that transfer mass from indices with a large
    gradient (i.e. that increase the max) to indices with a small gradient,
    plus a small amount of simulated‑annealing style random moves.

The routine respects a user supplied time budget (default 1000 s) and
returns the best sequence found, scaled to the `sqrt(2 n)` normalisation
required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)‑1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase          # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # extremely unlikely
        w[conv.argmax()] = 1.0
    return w


def _score_from_conv(max_conv: float, n: int) -> float:
    """Score = 2*n*max(conv)  (the value returned by evaluate_sequence after scaling)."""
    return 2.0 * n * max_conv


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function with low evaluation score.
    The function respects the given time budget (seconds) and returns a
    plain Python list of floats ready for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------------
    # 1.  Choose a length.  If a previous construction exists, keep its
    #     length as a lower bound but try a slightly larger vector.
    # --------------------------------------------------------------
    try:
        # height_sequence_1 may be defined in the global namespace
        init_vec = np.array(height_sequence_1, dtype=np.float64)
        base_n = len(init_vec)
    except Exception:
        base_n = 5600

    # try a longer length – larger n gives a finer discretisation.
    n = min(12000, max(base_n, 10000))
    print(f"[setup] Using length n = {n}")

    # FFT length (next power of two ≥ 2n‑1)
    L = 1 << ((2 * n - 1).bit_length())
    # Phase factor needed for the reversed FFT in gradient computation
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # --------------------------------------------------------------
    # Helper to compute the score of a *candidate* (which is assumed to sum to 1)
    # --------------------------------------------------------------
    def _score_of_vec(v: np.ndarray) -> float:
        conv = _compute_convolution(v, L)
        return _score_from_conv(conv.max(), n)

    # --------------------------------------------------------------
    # 2.  Initialise the best candidate with a few deterministic guesses.
    # --------------------------------------------------------------
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)     # uniform
    best_score = _score_of_vec(best_seq)
    print(f"[init] uniform score = {best_score:.8f}")

    # arcsine (U‑shaped) – classic good start
    xs = np.arange(n, dtype=np.float64)
    x_mid = (xs + 0.5) / n
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    arcsine = arcsine / arcsine.sum()
    sc = _score_of_vec(arcsine)
    if sc < best_score:
        best_seq, best_score = arcsine, sc
        print(f"[init] arcsine score = {best_score:.8f}")

    # --------------------------------------------------------------
    # 3.  Parametric sweep (simple families).  All candidates are normalised
    #     to sum = 1 before scoring.
    # --------------------------------------------------------------
    print("[sweep] starting parametric families")

    # Power‑law family  a_i ∝ (i+1)^p  (and its mirror)
    for p in np.linspace(0.0, 2.0, 41):
        if time.time() > deadline:
            break
        base = (xs + 1.0) ** p
        for cand in (base, base[::-1]):
            cand = cand / cand.sum()
            sc = _score_of_vec(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[sweep] power p={p:.2f} → score={best_score:.8f}")

    # Exponential decay a_i ∝ exp(-λ i)  (and mirrored)
    for lam in np.linspace(0.0, 5.0, 41):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        for cand in (exp_shape, exp_shape[::-1]):
            cand = cand / cand.sum()
            sc = _score_of_vec(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[sweep] exp λ={lam:.2f} → score={best_score:.8f}")

    # Double‑exponential (U‑shape)  a_i ∝ exp(-λ i)+exp(-λ (n‑1‑i))
    for lam in np.linspace(0.0, 5.0, 21):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        right = left[::-1]
        u_shape = left + right
        cand = u_shape / u_shape.sum()
        sc = _score_of_vec(cand)
        if sc < best_score:
            best_seq, best_score = cand, sc
            print(f"[sweep] double‑exp λ={lam:.2f} → score={best_score:.8f}")

    # Beta‑like densities  a_i ∝ x^{α‑1} (1‑x)^{β‑1}
    alphas = np.linspace(0.2, 3.0, 15)
    betas  = np.linspace(0.2, 3.0, 15)
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta_shape = (x_mid ** (a - 1.0)) * ((1.0 - x_mid) ** (b - 1.0))
            if np.any(beta_shape <= 0):
                continue
            cand = beta_shape / beta_shape.sum()
            sc = _score_of_vec(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[sweep] Beta a={a:.2f} b={b:.2f} → score={best_score:.8f}")

    # --------------------------------------------------------------
    # 4.  Mixture of the current best with the arcsine shape – cheap 1‑D
    #     optimisation that sometimes reduces the peak.
    # --------------------------------------------------------------
    if time.time() < deadline:
        for w in np.linspace(0.0, 1.0, 21):
            cand = w * best_seq + (1.0 - w) * arcsine
            cand = cand / cand.sum()
            sc = _score_of_vec(cand)
            if sc < best_score:
                best_seq, best_score = cand, sc
                print(f"[mix] weight {w:.2f} → score={best_score:.8f}")

    # --------------------------------------------------------------
    # 5.  Adam optimisation on the smooth approximation of max(conv).
    # --------------------------------------------------------------
    print("[adam] starting Adam on smoothed max")
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.20                     # a bit larger than before
    max_adam_iters = 60000
    alpha_start, alpha_end = 4.0, 250.0   # keep smoothing longer

    adam_it = 0
    while time.time() < deadline and adam_it < max_adam_iters:
        adam_it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        # soft‑max weights (smooth max)
        alpha = alpha_start + (alpha_end - alpha_start) * (adam_it / max_adam_iters)
        w = _softmax_weights(conv, alpha)

        # gradient of the smooth objective
        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** adam_it)
        v_hat = v / (1.0 - beta2 ** adam_it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluation (fast version)
        conv_new = _compute_convolution(a_new, L)
        new_score = _score_from_conv(conv_new.max(), n)

        if new_score < best_score:
            best_seq, best_score = a_new.copy(), new_score
            print(f"[adam] it {adam_it:5d} improved → score={best_score:.8f}")

        a = a_new

    # --------------------------------------------------------------
    # 6.  Sub‑gradient descent on the true max (active‑set uniform weights).
    # --------------------------------------------------------------
    print("[subgrad] starting sub‑gradient descent")
    a = best_seq.copy()
    sub_it = 0
    max_sub_it = 80000
    while time.time() < deadline and sub_it < max_sub_it:
        sub_it += 1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()

        w = _submax_weights(conv)                     # uniform over active set
        grad = _gradient_from_weights(a, w, phase, L)

        step_size = 0.5 / np.sqrt(sub_it)             # diminishing
        a_new = a - step_size * grad
        a_new = _project_simplex(a_new)

        conv_new = _compute_convolution(a_new, L)
        new_score = _score_from_conv(conv_new.max(), n)

        if new_score < best_score:
            best_seq, best_score = a_new.copy(), new_score
            print(f"[subgrad] it {sub_it:5d} improved → score={best_score:.8f}")

        a = a_new

    # --------------------------------------------------------------
    # 7.  Guided hill‑climbing (mass transfer from high‑gradient to
    #     low‑gradient indices) + a thin simulated‑annealing tail.
    # --------------------------------------------------------------
    print("[hill] starting guided hill climbing")
    a_current = best_seq.copy()
    temperature = 0.01               # initial temperature for occasional uphill moves
    temp_decay = 0.9999

    while time.time() < deadline:
        # compute gradient w.r.t. current max (active set)
        conv = _compute_convolution(a_current, L)
        w = _submax_weights(conv)
        grad = _gradient_from_weights(a_current, w, phase, L)

        # indices with largest positive gradient (donors) and most negative
        # (receivers).  We move a small fraction of mass from donor→receiver.
        donor_idx = np.argmax(grad)        # worst contributor
        receiver_idx = np.argmin(grad)     # best contributor
        if donor_idx == receiver_idx:
            # fall back to a random move if gradient is flat
            donor_idx = np.random.choice(n, p=a_current)
            receiver_idx = np.random.randint(0, n)

        delta = 0.02 * a_current[donor_idx]   # up to 2 % of donor mass
        if delta <= 0:
            continue

        a_new = a_current.copy()
        a_new[donor_idx] -= delta
        a_new[receiver_idx] += delta
        if a_new[donor_idx] < 0:
            continue

        # evaluate
        conv_new = _compute_convolution(a_new, L)
        new_score = _score_from_conv(conv_new.max(), n)

        # acceptance rule (occasionally allow uphill moves)
        if new_score < best_score or np.random.rand() < math.exp(-(new_score - best_score) / temperature):
            a_current = a_new
            if new_score < best_score:
                best_seq, best_score = a_new.copy(), new_score
                print(f"[hill] improved → score={best_score:.8f}")

        temperature *= temp_decay   # cool down slowly

    # --------------------------------------------------------------
    # 8.  Final scaling to the normalisation required by evaluate_sequence.
    # --------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()
    # One final call to evaluate_sequence for logging (no impact on the result)
    try:
        final_score = evaluate_sequence(final_seq)
    except Exception:
        final_score = best_score
    elapsed = time.time() - start_time
    print(f"[end] best_score = {final_score:.8f} (n={n}) after {elapsed:.1f}s")
    return final_seq
