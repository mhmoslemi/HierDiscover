"""
Propose a step sequence with a low evaluation score.

Algorithm
---------
1. Initialise the sequence.
   * Use the previously discovered `height_sequence_1` (truncated to
     `max_n` = 1200) if it exists, otherwise start from a uniform distribution.
   * Normalise to a probability simplex (sum = 1).

2. LP‑based improvement (the core of the algorithm).
   * For the current sequence `f` compute `M = max(f*f)`.
   * Solve the linear programme
          maximise   1ᵀg
          subject to  f * g ≤ M   (pointwise convolution)
                     g ≥ 0.
     The constraint matrix is a Toeplitz matrix that represents the linear
     convolution `f * g`.  Because the constraints are linear in `g`, the LP can
     increase the total mass of `g` while keeping the convolution bounded by `M`.
   * Normalise the LP solution `g` to a probability vector `ĝ = g / sum(g)` and
     mix it with the current `f`:
                f_new = (1‑t)·f + t·ĝ
     for several mixing coefficients `t ∈ {0.30,0.20,0.10,0.05,0.02}`.
     The best improving mixture (if any) replaces `f`.

   The loop repeats until the time budget is exhausted (or no improvement is
   observed for a few consecutive LP calls).

3. Refine the best sequence with a projected sub‑gradient optimiser
   (same routine that was used in the previous search).  This phase
   finishes the optimisation with a fast local search.

4. Return the best‑found sequence as a plain Python list; `evaluate_sequence`
   will be called outside of this function to obtain the final score.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns only the needed part (real values).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (copy‑paste from the previous solution)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based improvement
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) == f * g (full linear convolution).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Direct double loop – fine for n up to ~1200
    for i in range(n):
        coeff = f[i]
        if coeff == 0.0:
            continue
        # coefficient f[i] appears in row k = i + j for every column j
        for j in range(n):
            A[i + j, j] = coeff
    return A


def lp_improve(f: np.ndarray,
               fft_len: int,
               n: int,
               deadline: float,
               max_no_improve: int = 12):
    """
    Perform a series of linear‑programming steps.
    Returns the improved sequence and its score.
    """
    best_f = f.copy()
    best_score = fast_score(best_f, fft_len, n)
    no_improve = 0

    while time.time() < deadline and no_improve < max_no_improve:
        # current bound M
        conv = full_conv(best_f, fft_len)
        M = conv.max()

        # LP: maximise sum(g)  subject to  A·g ≤ M,  g ≥ 0
        A = build_conv_matrix(best_f)
        b_ub = np.full(A.shape[0], M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  -> minimise -sum(g)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if not res.success:
            # LP failed – exit early
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break

        # Normalise the LP solution to a probability vector
        g_norm = g / sum_g

        # Try several mixing coefficients; pick the best improvement
        improved = False
        for t in (0.30, 0.20, 0.10, 0.05, 0.02):
            candidate = (1.0 - t) * best_f + t * g_norm
            cand_score = fast_score(candidate, fft_len, n)
            if cand_score < best_score - 1e-9:      # strict improvement
                best_f = candidate
                best_score = cand_score
                improved = True
                # reset no‑improve counter
                no_improve = 0
                break
        if not improved:
            no_improve += 1

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    """
    rng = np.random.default_rng(seed + 12345)
