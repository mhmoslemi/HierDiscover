    best = a.copy()
    conv = conv_fft(best, fft_len, n)
    best_max = conv.max()
    best_idx = conv.argmax()
    i_array = np.arange(n)
    rng = np.random.default_rng()
    step_fraction = 0.1  # initial fraction to move from source
    topk = 15
    no_improve_counter = 0
    while time.time() < deadline:
        # recalc conv
        conv = conv_fft(best, fft_len, n)
        max_val = conv.max()
        max_idx = conv.argmax()
        if max_val < best_max - 1e-12:
            best_max = max_val
            # Reset no improve
            no_improve_counter = 0
        else:
            no_improve_counter += 1
        # compute contributions vector for this max_idx
        j = max_idx - i_array
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[valid] = best[i_array[valid]] * best[j[valid]]
        # source: topk indices with highest contrib
        src_candidates = np.argpartition(-contrib, topk)[:topk]

            tgt = rng.choice(tgt_candidates)
            if src == tgt:
                continue

            delta = delta_frac * best[src]
            if delta <= 0:

"""
Guided sub‑gradient hill‑climb for the max auto‑convolution problem.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2)
which is homogeneous: scaling a does not change the score.
Thus we fix sum(a)=1 and minimise  max(conv(a,a)).
For a given maximum index k* the contribution of each coordinate i is
    c_i = a_i * a_{k*‑i}   (zero outside the support).
Moving mass from coordinates with large c_i to those with tiny c_j
strictly reduces the current maximal convolution value.
The algorithm repeatedly:
* finds the current maximal convolution index,
* builds the contribution vector c,
* selects a source (large c) and a target (small c),
* transfers a fraction of the source mass to the target,
* accepts the move only if the maximal convolution value decreases.
A simple schedule for the transferred fraction (step size) is used,
and occasional random re‑normalisations help escape shallow plateaus.
If a previous construction is present (`height_sequence_1`) it is used as
the starting point; otherwise a uniform distribution is taken.
The routine works on the probability simplex (sum = 1) and finally
returns a list of non‑negative floats directly usable by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    The result is a real array of length 2·n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def max_convolution(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return max(conv(a,a))."""
    return conv_fft(a, fft_len, n).max()


# ----------------------------------------------------------------------
# Guided hill‑climb
# ----------------------------------------------------------------------


def guided_hill_climb(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    step_init: float = 0.10,
    topk: int = 20,
    max_no_improve: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient descent on the L∞‑norm of the auto‑convolution.

    Parameters
    ----------
    a : np.ndarray
        Starting probability vector (sum = 1, all non‑negative).
    fft_len, n : int
        FFT length and vector length.
    deadline : float
        Unix time after which the routine must stop.
    step_init : float
        Initial fraction of source mass that may be moved.
    topk : int
        Number of candidate source / target indices examined.
    max_no_improve : int
        After this many unsuccessful iterations a small random
        perturbation is applied to escape plateaus.

    Returns
    -------
    best_a : np.ndarray
        Improved probability vector.
    best_max : float
        Corresponding max(conv(best_a,best_a)).
    """
    best = a.copy()
    best_max = max_convolution(best, fft_len, n)
    i_array = np.arange(n)
    rng = np.random.default_rng()
    step = step_init
    no_improve = 0

    while time.time() < deadline:
        # recompute max convolution and its index
        conv = conv_fft(best, fft_len, n)
        cur_max = conv.max()
        k = conv.argmax()
        # keep track of improvement
        if cur_max < best_max - 1e-14:
            best_max = cur_max
            no_improve = 0
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # compute contributions c_i = a_i * a_{k-i}
        # ------------------------------------------------------------------
        j = k - i_array                     # partner index
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=float)
        contrib[mask] = best[i_array[mask]] * best[j[mask]]

        # select candidates
        src_candidates = np.argpartition(-contrib, topk)[:topk]  # largest c_i
        tgt_candidates = np.argpartition(contrib, topk)[:topk]   # smallest c_i

        improved = False
        # try a handful of random source/target pairs
        for _ in range(5):
            src = rng.choice(src_candidates)
            tgt = rng.choice(tgt_candidates)
            if src == tgt:
                continue
            # fractional move – randomised a little to avoid deterministic cycles
            delta_frac = rng.uniform(step * 0.2, step)
            delta = delta_frac * best[src]
            if delta <= 0.0:
                continue
            candidate = best.copy()
            candidate[src] -= delta
            candidate[tgt] += delta
            # safeguard against negative rounding errors
            if candidate[src] < -1e-15:
                continue
            cand_max = max_convolution(candidate, fft_len, n)
            if cand_max + 1e-12 < best_max:
                best = candidate
                best_max = cand_max
                improved = True
                # slightly increase step after a success
                step = min(step * 1.05, 0.30)
                break

        if not improved:
            # gradually shrink step size when we are struggling
            step = max(step * 0.90, 0.005)

        # ------------------------------------------------------------------
        # occasional random perturbation to jump out of flat regions
        # ------------------------------------------------------------------
        if no_improve >= max_no_improve:
            # add a small zero‑mean Gaussian noise, re‑normalise
            noise = rng.normal(scale=step, size=n)
            pert = best + noise
            pert = np.clip(pert, 0.0, None)
            if pert.sum() == 0.0:
                pert = np.full(n, 1.0 / n)
            else:
                pert /= pert.sum()
            best = pert
            best_max = max_convolution(best, fft_len, n)
            no_improve = 0
            step = step_init

    return best, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with low evaluation score.

    The routine works on the probability simplex (Σ a_i = 1) and uses a
    guided sub‑gradient hill‑climb that moves mass from coordinates that
    contribute most to the current maximal auto‑convolution value towards
    coordinates that contribute the least.  If a previous construction
    (``height_sequence_1``) is available it is used as the starting point;
    otherwise a uniform distribution is taken.  The algorithm respects the
    time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0          # leave a safety margin

    # ------------------------------------------------------------------
    # initialise a – either from a saved construction or uniform
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        a0 = np.asarray(prev, dtype=float)
        a0 = np.maximum(a0, 0.0)                     # safety
    else:
        # a modest size that still gives enough freedom
        n = 5000
        a0 = np.full(n, 1.0 / n)

    # normalise to a probability vector (sum = 1)
    a0_sum = a0.sum()
    if a0_sum == 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / a0_sum

    n = a0.size
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initial score (for logging)
    # ------------------------------------------------------------------
    init_max = max_convolution(a0, fft_len, n)
    init_score = 2.0 * n * init_max
    print(f"[init] n={n}, initial fast score = {init_score:.6f}")

    # ------------------------------------------------------------------
    # guided hill‑climb – main optimisation phase
    # ------------------------------------------------------------------
    best_a, best_max = guided_hill_climb(
        a0,
        fft_len,
        n,
        deadline,
        step_init=0.10,
        topk=25,
        max_no_improve=300,
    )

    # ------------------------------------------------------------------
    # final normalisation & evaluation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    final_score = evaluate_sequence(best_a.tolist())
    print(f"[final] evaluated score = {final_score:.6f}")

    # expose the sequence for possible reuse by subsequent runs
    globals()["height_sequence_1"] = best_a.tolist()

    return best_a.tolist()
