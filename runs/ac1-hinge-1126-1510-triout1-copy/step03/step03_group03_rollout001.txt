propose_candidate
-----------------
Search for a step‑function (non‑negative heights) that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of three stages:

1) **Initialisation** – take the best known sequence (global
   ``height_sequence_1``) if it exists, otherwise fall back to an
   arcsine‑like (≈1/√i) distribution.  A short random sweep over a few
   parametric families (power‑law, exponential, two‑sided, Gaussian)
   is performed to obtain a good starting point.

2) **Soft‑max gradient descent** – the exact objective is scale‑invariant,
   so we fix ``sum(a)=1`` and minimise ``max(conv)``.
   The non‑smooth max is replaced by a soft‑max
   ``(1/α)·log Σ exp(α·c_k)`` with a temperature ``α`` that is
   increased gradually.  The gradient of this smooth surrogate is
   ``4·n·(p ⋆ a)`` where ``p`` are the soft‑max weights and ``⋆`` denotes
   (linear) correlation.  Correlation is evaluated efficiently with FFTs.
   An exponentiated‑gradient update
        a_i ← a_i·exp(‑η·(p⋆a)_i)
   keeps the variables non‑negative; a normalisation step projects back
   onto the probability simplex.  Learning‑rate ``η`` decays slowly and
   occasional jitter (small random perturbation) helps to escape plateaus.

3) **Final verification** – the best sequence found is returned as a
   plain Python list and its true score is obtained from the provided
   ``evaluate_sequence`` function.

The whole routine respects a 1000 s wall‑clock budget, prints progress,
and works with any sequence length up to a few‑tens of thousands.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast (FFT based) evaluation of
        2·n·max(conv(seq,seq)) / (sum(seq)**2)
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σx_i = 1, x_i ≥ 0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i).  Unnormalised as 1/√i.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a

    corr_full = np.fft.irfft(np.conj(Fp) * Fa, n=L)
    return corr_full[: len(a)]  # keep only the first |a| entries

            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- occasional jitter to break symmetry / plateaus -----
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s <= 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

        # ----- periodic evaluation and bookkeeping -----
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:.1f}s] iter {iteration}, new best fast_score={cur_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. best reported score = {final_score:.6f}")
    return final_seq
