"""
Propose a step‑function (non‑negative vector with arbitrary length) that
minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

(the same quantity that ``evaluate_sequence`` returns).

The algorithm consists of three stages:

1. **Broad stochastic search** – generate many candidates from several
   parametric families (power‑law, exponential, two‑sided power‑law,
   beta‑pdf, sigmoid, mixtures).  Each candidate is normalised to
   ``sum = 1`` and scored with a fast FFT‑based evaluator.  The best
   few (≤ 5) are kept as an elite pool.

2. **Smoothed‑gradient refinement** – each elite candidate is refined by
   projected Adam optimisation on the probability simplex.  The
   non‑smooth “max‑convolution’’ objective is replaced by a
   temperature‑controlled soft‑max
   `F(a)= (1/α)·log∑exp(α·conv(a,a))`.  The gradient of *F* can be
   computed with two FFTs per iteration, after which the new iterate is
   projected onto the simplex.  The temperature α is increased
   gradually (from 2 to 40) so that the optimisation first sees a
   smooth landscape and then concentrates on the true maximum.

3. **Fine‑grained local search** – a short hill‑climbing phase that
   performs tiny mass transfers between two random coordinates.  Any
   improvement is kept.

The search respects the 1000 s budget – roughly 30 % of the time is used
for the random stage, 65 % for gradient refinement and the final 5 %
for local polishing.  All helper routines are top‑level (no closures)
and the code works with the provided ``evaluate_sequence`` function.
"""

import time

# Utility functions

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Evaluate 2·n·max(conv(seq,seq)) / (sum(seq)**2) quickly using FFT.
    The sequence is assumed to be non‑negative and summed to 1,
    therefore the denominator equals 1 and the returned value is simply
    2·n·max(conv).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^{α-1}·(n-i)^{β-1}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sigmoid(n: int, centre: float, scale: float) -> np.ndarray:
    """
    a_i = 1 / (1 + exp((i-centre)/scale)).
    """
    i = np.arange(n, dtype=np.float64)
    return 1.0 / (1.0 + np.exp((i - centre) / scale))


def generate_mixture_power(n: int, p1: float, p2: float, w: float) -> np.ndarray:
    """
    Convex mixture of two power‑law tails:
        a_i = w·(i+1)^{-p1} + (1-w)·(i+1)^{-p2}.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    return w * idx ** (-p1) + (1.0 - w) * idx ** (-p2)

# Smoothed‑gradient (Adam) refinement
# ----------------------------------------------------------------------


def refine_adam(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    alpha_start: float = 2.0,
    alpha_end: float = 40.0,
    eta0: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation of the soft‑max surrogate of the peak
    convolution.  The iterate is kept on the simplex.
    Returns the best sequence seen and its exact  ``fast_score``.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = project_simplex(a)               # ensure feasibility
    n = a.shape[0]

    # pre‑allocate FFT buffers (length that can hold a (n) and w (2n‑1) together)
    fft_len = next_pow2(3 * n)            # enough for (2n‑1) + n
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = fast_score(best_seq)

    it = 0
    while time.time() < start_time + time_limit:
        it += 1

        # --------- convolution of a with itself ----------
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

        # current exact score (since sum(a)=1)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.copy()

        # --------- soft‑max weights ----------
        # schedule α linearly in log‑space
        alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, it / 2000.0)
        # stabilize exponentials
        shift = conv.max()
        exps = np.exp(alpha * (conv - shift))
        w = exps / exps.sum()                     # sum_k w_k = 1

        # --------- gradient = 2·∑_k w_k·a_{k-i} ----------
        # This is a convolution of w (size 2n‑1) with reversed a.
        a_rev = a[::-1]
        # pad both to fft_len
        W = np.fft.rfft(w, n=fft_len)
        A_rev = np.fft.rfft(a_rev, n=fft_len)
        grad_full = np.fft.irfft(W * A_rev, n=fft_len)   # length fft_len
        # The relevant slice is indices (n‑1 … n‑1 + n‑1)
        grad = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # --------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta0 / (np.sqrt(v_hat) + eps)
        a_candidate = a - step * m_hat

        # --------- projection onto the simplex ----------
        a = project_simplex(a_candidate)

        # occasional random jitter to shake the optimiser
        if it % 3000 == 0:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))

    # final exact evaluation of the best sequence found
    best_val = fast_score(best_seq)
    return best_seq, best_val

def local_hill_climb(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    step_scale: float = 0.005,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climber that repeatedly moves a tiny amount of
    mass from one coordinate to another.  The move is kept only if it
    improves the exact score.
    """
    a = seq.copy()
    n = a.shape[0]
    best_val = fast_score(a)

    while time.time() < start_time + time_limit:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # propose moving a small ε from i to j
        eps = step_scale * np.random.rand() * a[i]
        if eps <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # re‑normalise (numerical drift)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            continue
        a_new /= s
        val = fast_score(a_new)
        if val < best_val:
            best_val = val
            a = a_new
    return a, best_val
