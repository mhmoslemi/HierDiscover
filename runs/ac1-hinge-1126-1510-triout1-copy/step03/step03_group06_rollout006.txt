def fast_convolve(x, y):
    n = len(x)
    m = len(y)
    conv_len = n + m - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:conv_len]
    return conv

def softmax_weights(conv, beta):
    maxc = conv.max()
    # compute exponentials
    exp_vals = np.exp(beta * (conv - maxc))
    sum_exp = exp_vals.sum()
    return exp_vals / sum_exp

def local_refine(a0, steps, deadline):
    a = a0.copy()
    n = a.shape[0]
    best = a.copy()
    best_score = 2.0 * n * fast_convolve(a, a).max()
    temperature = 0.05
    for s in range(steps):
        if time.time() >= deadline:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # move from i to j
        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)
        if a[src] <= 0: continue
        delta = np.random.uniform(0.0, 0.05) * a[src]
        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        # ensures sum = 1
        # compute score
        new_score = 2.0 * n * fast_convolve(new, new).max()
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
        temperature *= 0.9995
    return best, best_score

"""
Projected gradient descent with entropy‑regularised soft‑max on the simplex.
The algorithm works on a few problem sizes, repeatedly refines the
distribution, up‑samples a good solution to larger lengths and finishes
with a cheap stochastic local search.  All operations stay on the
probability simplex (∑a_i = 1, a_i ≥ 0) so the evaluation score
2·n·max(conv(a,a)) is invariant under scaling.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast linear convolution via FFT (real‑valued)
# ----------------------------------------------------------------------
def fast_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the linear convolution of two real vectors using an FFT of a
    power‑of‑two length (fast for the sizes we use, up to a few 10⁴).
    """
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:n]
    return conv


# ----------------------------------------------------------------------
# Upsample a sequence to a new length by linear interpolation
# ----------------------------------------------------------------------
def upsample_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Interpolate a distribution to a new length while preserving positivity."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    sm = new.sum()
    if sm == 0.0:
        new = np.full(new_n, 1.0 / new_n)
    else:
        new = new / sm
    return new


# ----------------------------------------------------------------------
# Smoothed (entropy‑regularised) projected gradient descent
# ----------------------------------------------------------------------
def smooth_descent(
    a0: np.ndarray,
    n: int,
    max_iters: int,
    step0: float,
    beta0: float,
    beta_max: float,
    beta_inc: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) with a on the simplex using a soft‑max
    approximation of the max.  Returns the best sequence encountered.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_score = float("inf")
    beta = beta0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --------------------------------------------------------------
        # current convolution and score
        # --------------------------------------------------------------
        conv = fast_convolve(a, a)                # length 2n‑1
        max_conv = conv.max()
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # occasional progress output
            print(f"[smooth n={n} it={it}] new best {score:.6f}")

        # --------------------------------------------------------------
        # soft‑max weights for the active peak(s)
        # --------------------------------------------------------------
        beta = min(beta_max, beta + beta_inc)
        # stable soft‑max
        exp_vals = np.exp(beta * (conv - max_conv))
        p = exp_vals / exp_vals.sum()

        # --------------------------------------------------------------
        # gradient of the soft‑max objective (2·Σ p_k·a_{k‑i})
        # --------------------------------------------------------------
        # grad_i = 2· Σ_k p_k·a_{k‑i}
        # this is exactly the (cross‑correlation of p with a) multiplied by 2
        a_rev = a[::-1]
        conv_rev = fast_convolve(p, a_rev)       # length 3n‑2
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1]   # slice of length n

        # --------------------------------------------------------------
        # projected gradient step
        # --------------------------------------------------------------
        step = step0 / np.sqrt(it)
        a = a - step * grad
        a = project_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light random walk refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic refinement: repeatedly move a small amount of mass
    between two random coordinates, accepting improvements or,
    with decreasing temperature, occasional uphill moves.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * fast_convolve(a, a).max()
    temperature = 0.05

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of the transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        if a[src] <= 0.0:
            continue

        # move up to 5 % of the source mass
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        # (sum stays 1 by construction)
        new = np.clip(new, 0.0, None)

        new_score = 2.0 * n * fast_convolve(new, new).max()

        # Metropolis acceptance
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature *= 0.9995
        temperature = max(temperature, 1e-6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    The routine mixes entropy‑regularised projected gradient descent,
    up‑sampling of promising solutions to larger lengths and a concluding
    stochastic refinement.  The best sequence observed before the time
    budget expires is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Length schedule – start modest, then grow.
    # --------------------------------------------------------------
    base_len = 2000
    length_candidates = [base_len, base_len * 2, base_len * 4, base_len * 6]
    length_candidates = [min(20000, n) for n in length_candidates]  # cap to keep FFT cheap
    length_candidates = sorted(set(length_candidates))

    # --------------------------------------------------------------
    # Global best across all sizes
    # --------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")
    best_by_len = {}   # keep the best sequence for each length (useful for up‑sampling)

    # ------------------------------------------------------------------
    # Hyper‑parameters (mostly conservative – they adapt automatically)
    # ------------------------------------------------------------------
    step0 = 0.30          # base step size for gradient updates
    beta0 = 0.5           # start entropy regularisation
    beta_max = 12.0       # final strength of the soft‑max
    # we will increase beta linearly; the increment depends on the size of the run
    max_iters_per_len = 12000   # upper bound; actual number is limited by remaining time

    # ------------------------------------------------------------------
    # Main search loop over the chosen lengths (ascending)
    # ------------------------------------------------------------------
    for n in length_candidates:
        # ----------------------------------------------------------
        # Determine how many iterations we can still afford
        # ----------------------------------------------------------
        time_left = deadline - time.time()
        if time_left < 2.0:
            break
        # Empirical cost: ~0.0015 s per iteration for n≈8000 (FFT + overhead)
        est_iters = int(time_left / 0.0018)
        it_budget = min(max_iters_per_len, max(est_iters, 1))

        # ----------------------------------------------------------
        # Prepare a good starting point for this length
        # ----------------------------------------------------------
        if n in best_by_len:
            a0 = best_by_len[n].copy()
            # add a tiny random perturbation to avoid exact stagnation
            a0 += 0.02 * np.random.randn(n)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        elif global_best_seq is not None and len(global_best_seq) == n:
            a0 = np.array(global_best_seq, copy=True)
            a0 += 0.02 * np.random.randn(n)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
        elif global_best_seq is not None and len(global_best_seq) < n:
            # up‑sample the current global best to the new size
            a0 = upsample_seq(np.array(global_best_seq), n)
        else:
            # randomised power‑law initialisation (the most productive so far)
            p = np.random.uniform(0.45, 0.80)
            idx = np.arange(1, n + 1, dtype=np.float64)
            a0 = idx ** (-p)
            a0 = a0 / a0.sum()

        # ----------------------------------------------------------
        # Run the smoothed projected gradient descent
        # ----------------------------------------------------------
        beta_inc = (beta_max - beta0) / max(it_budget, 1)
        seq, score = smooth_descent(
            a0,
            n,
            it_budget,
            step0,
            beta0,
            beta_max,
            beta_inc,
            deadline,
        )

        # ----------------------------------------------------------
        # Optional cheap stochastic refinement
        # ----------------------------------------------------------
        time_rem = deadline - time.time()
        if time_rem > 0.6:
            refine_steps = int(min(3000, time_rem / 0.0009))
            seq_ref, score_ref = local_refine(seq, refine_steps, deadline)
            if score_ref < score:
                seq, score = seq_ref, score_ref

        # ----------------------------------------------------------
        # Record the best found for this length
        # ----------------------------------------------------------
        if score < global_best_score:
            global_best_score = score
            global_best_seq = seq.copy()
            print(f"[len={n}] NEW GLOBAL BEST = {score:.6f}")

        # keep the best for possible up‑sampling later
        best_by_len[n] = seq.copy()

        # ------------------------------------------------------------------
        # Early exit if we have already beaten the target bound (1.503)
        # ------------------------------------------------------------------
        if global_best_score < 1.503:
            print("Target achieved – terminating early.")
            break

    # ----------------------------------------------------------------------
    # Fallback in the unlikely case that nothing succeeded
    # ----------------------------------------------------------------------
    if global_best_seq is None:
        n = base_len
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-0.55)
        a = a / a.sum()
        global_best_seq = a

    # Convert the numpy array to a plain Python list as required by the harness
    return [float(x) for x in global_best_seq.tolist()]
