def refine_sequence_improved(seq, time_limit, start_time, eta0=0.5, eps=1e-12, momentum=0.9, use_exponential=False):
   a = seq.copy()
   a = np.maximum(a, 0)
   if a.sum()==0: a = np.full_like(a, 1/a.size)
   else: a = a / a.sum()
   n = a.size
   N = next_pow2(3*n)  # zero pad for correlation.

   best_seq = a.copy()
   best_score = fast_score(best_seq)  # compute objective
   # Precompute gamma for gradient: not needed.

   # Initialize momentum vector if using
   v = np.zeros_like(a)
   it=0
   no_improve=0

   while time.time() < start_time + time_limit:
       it+=1
       # FFT of a
       fa = np.fft.rfft(a, n=N)
       # conv = a convolve a via IRFFT of product
       conv = np.fft.irfft(fa * fa, n=N)[:2*n-1]
       M = conv.max()
       cur_score = 2.0 * n * M
       if cur_score < best_score:
           best_score = cur_score
           best_seq = a.copy()
           no_improve=0
       else:
           no_improve+=1

       # active mask
       active = conv >= M - eps
       d = active.astype(np.float64)   # length 2n-1

       # compute gradient via correlation
       fd = np.fft.rfft(d, n=N)
       # correlation: ifft(fd * conj(fa))
       corr = np.fft.irfft(fd * np.conj(fa), n=N)[:3*n-2]   # linear correlation length
       # gradient g_i = 2 * corr[n-1 + i]
       g = 2.0 * corr[n-1: n-1+n]

       # momentum update
       eta = eta0 / (np.sqrt(it) + 1.0)
       v = momentum * v + (1.0 - momentum) * g
       if use_exponential:
           # exponentiated gradient
           # compute log update: subtract eta * v
           update = np.exp(-eta * v)
           a = a * update
           a_sum = a.sum()
           if a_sum == 0:
               a = np.full_like(a, 1.0 / n)
           else:
               a = a / a_sum
       else:
           # Euclidean projected gradient
           cand = a - eta * v
           # project onto simplex
           # using our project_simplex function
           a = project_simplex(cand)

       # occasional random perturbation
       if no_improve >= 2000:
           sigma = 0.02 / np.sqrt(n)
           a = project_simplex(a + sigma * np.random.randn(n))
           no_improve = 0

   # After loop compute final exact score via fast_score
   best_score = fast_score(best_seq)
   return best_seq, best_score

"""
Search for a step‑function that minimizes the supplied ``evaluate_sequence``.
The method consists of two phases:

1. **Diverse random‑shape exploration** – many candidate vectors are sampled
   from several parametrised families (power‑law, exponential,
   two‑sided power‑law, Beta, sinusoidal and squared‑sinc).  Each candidate
   is normalised to sum = 1 and scored with a cheap FFT‑based evaluator.
   The best ``top_k`` candidates are retained.

2. **FFT‑accelerated exponentiated‑gradient refinement** – every elite
   candidate is improved by projected sub‑gradient descent on the simplex.
   The sub‑gradient of the objective
   ``2·n·max(conv(a,a))`` is obtained from *all* active convolution entries.
   Correlations needed for the gradient are computed via FFT, which makes
   thousands of iterations cheap.  The update uses an exponentiated‑gradient
   (multiplicative) step with Nesterov‑type momentum and occasional Gaussian
   perturbations to escape plateaus.  The refinement runs until the time
   budget allocated to that candidate is exhausted.

The whole search respects the 1000 s time limit and returns the best sequence
found (as a plain Python list of non‑negative floats).  The algorithm is
fully deterministic given a seed.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quick evaluation of 2·n·max(conv(seq, seq)) for a sequence that is
    already normalised to sum = 1.  This is exactly the value returned by
    ``evaluate_sequence`` up to the normalisation factor.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_sine_shape(n: int, p: float = 1.0) -> np.ndarray:
    """a_i ∝ sin(π·(i+1)/(n+1))^p, symmetric, non‑negative."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return np.sin(np.pi * i / (n + 1)) ** p


def generate_sinc_squared_shape(n: int, scale: float) -> np.ndarray:
    """
    a_i ∝ sinc((i - (n-1)/2) / scale)^2.
    ``scale`` controls the width; larger scale ⇒ flatter shape.
    """
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64) - centre
    # np.sinc implements sin(pi*x)/(pi*x)
    shape = np.sinc(i / scale) ** 2
    # enforce non‑negativity (already non‑negative) and avoid a zero sum
    shape = np.maximum(shape, 0.0)
    return shape


# ----------------------------------------------------------------------
# Gradient refinement (FFT‑based, exponentiated update)
# ----------------------------------------------------------------------


def refine_sequence(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    eta0: float = 0.5,
    momentum: float = 0.9,
    eps: float = 1e-12,
    use_exponential: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Perform many‑iteration projected (exponentiated) sub‑gradient descent on
    the simplex, using FFT to obtain the sub‑gradient of the peak
    convolution value.
    Returns the best sequence found and its exact fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    # guarantee feasibility
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / a.sum()

    n = a.size
    conv_len = next_pow2(2 * n)        # length for a*a convolution
    corr_len = next_pow2(3 * n)        # length for correlation a*d

    best_seq = a.copy()
    best_score = fast_score(best_seq)

    # momentum buffer (Euclidean for exponential updates as well)
    v = np.zeros_like(a)

    it = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        it += 1

        # ----- compute convolution and current objective -----
        fa = np.fft.rfft(a, n=conv_len)
        conv = np.fft.irfft(fa * fa, n=conv_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = float(2.0 * n * M)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- active‑set mask -----
        active = conv >= M - eps
        d = active.astype(np.float64)  # binary mask, length 2n‑1

        # ----- gradient via FFT correlation -----
        fd = np.fft.rfft(d, n=corr_len)
        # correlation of d with a (linear, not circular)
        corr = np.fft.irfft(fd * np.conj(fa), n=corr_len)[: 3 * n - 2]
        # gradient: g_i = 2 * corr[n-1 + i]
        g = 2.0 * corr[n - 1 : n - 1 + n]

        # ----- momentum update -----
        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * g

        # ----- multiplicative (exponentiated) vs additive update -----
        if use_exponential:
            # exponentiated‑gradient step (always stays on simplex)
            factor = np.exp(-eta * v)
            a = a * factor
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s
        else:
            # Euclidean projected gradient descent
            cand = a - eta * v
            a = project_simplex(cand)

        # ----- occasional random perturbation to leave plateaus -----
        if no_improve >= 2500:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact score for the best candidate discovered during refinement
    best_score = fast_score(best_seq)
    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score height sequence within ``budget_s`` seconds.
    The returned list can be fed directly to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # 0) initialise from a previously discovered sequence, if available
    # --------------------------------------------------------------
    try:

        if init_seq.ndim != 1 or init_seq.size == 0:

        init_seq = generate_two_sided_power_law(init_len, 0.55, 0.55, 1.0, 1.0)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0.0:
        init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Broad random‑shape search (≈ 40 % of the total time)
    # --------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 15000, 20000]
    families = [
        "power",
        "exp",
        "twosided",
        "beta",
        "sine",
        "sinc",
    ]

    # elite pool (keep the best few candidates)
    top_k: list[tuple[float, np.ndarray]] = [(best_score, best_seq.copy())]

    rand_deadline = start + 0.4 * budget_s
    while time.time() < rand_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.05, 2.0)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(1e-5, 5e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            p_l = random.uniform(0.05, 2.0)
            p_r = random.uniform(0.05, 2.0)
            w_l = random.uniform(0.1, 5.0)
            w_r = random.uniform(0.1, 5.0)
            seq = generate_two_sided_power_law(n, p_l, p_r, w_l, w_r)

        elif family == "beta":
            alpha = random.uniform(0.05, 1.8)
            beta = random.uniform(0.05, 1.8)
            seq = generate_beta_shape(n, alpha, beta)

        elif family == "sine":
            p = random.uniform(0.5, 3.0)
            seq = generate_sine_shape(n, p)

        else:  # "sinc"
            # scale determines width; choose between n/8 and n*2
            scale = random.uniform(n / 8.0, n * 2.0)
            seq = generate_sinc_squared_shape(n, scale)

        # normalise and sanitise
        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total < 1e-12:
            continue
        seq = seq / total

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(
                f"[{time.time() - start:.1f}s] new global best ({family}) n={n} "
                f"score={sc:.6f}"
            )

        # manage elite pool (size ≤ 15)
        if len(top_k) < 15:
            top_k.append((sc, seq.copy()))
        else:
            # replace worst if this one is better
            worst_idx = max(range(len(top_k)), key=lambda i: top_k[i][0])
            if sc < top_k[worst_idx][0]:
                top_k[worst_idx] = (sc, seq.copy())

    # sort elite candidates by increasing score
    top_k.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 2) Momentum sub‑gradient refinement on elites (remaining time)
    # --------------------------------------------------------------
    time_remaining = deadline - time.time()

        print(f"Time exhausted. Final score = {final_score:.6f}")

        if refined_score < best_score:

    # --------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
