def upscale_sequence(seq, new_len):
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0, 1, old_len)
    x_new = np.linspace(0, 1, new_len)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new, 0)  # ensure nonnegative
    # renormalize sum
    if seq_new.sum() == 0:
        seq_new = np.full(new_len, 1.0 / new_len)
    else:
        seq_new = seq_new / seq_new.sum()
    return seq_new

   descent (the `refine_sequence` routine from the previous solution) for
   the remaining time budget.
4. Return the final sequence; a final call to `evaluate_sequence` verifies
   the bound.

The whole search respects the 1000 s wall‑clock limit and prints progress.
"""

import time
import random
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quick evaluation of 2 * n * max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT convolution.  The sequence is assumed non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Truncated Gaussian centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    return np.exp(-0.5 * ((idx - mu) / sigma) ** 2)


def generate_sine_shape(n: int, freq: float) -> np.ndarray:
    """a_i = sin(π * (i+1)/(n+1))^freq, zero at the ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    base = np.sin(np.pi * i / (n + 1))
    return base ** freq


def generate_random_spikes(n: int, k: int) -> np.ndarray:
    """k random spikes (positive values) placed on a zero background."""
    a = np.zeros(n, dtype=np.float64)
    indices = np.random.choice(n, size=k, replace=False)
    a[indices] = np.random.rand(k) + 0.1  # avoid exact zeros
    return a


# ----------------------------------------------------------------------
# Momentum sub‑gradient refinement (same as earlier)
# ----------------------------------------------------------------------


def refine_sequence(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    momentum: float = 0.9,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected (momentum) sub‑gradient descent on the simplex.
    Returns the best sequence found and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]
    a = project_simplex(a)

    fft_len = next_pow2(2 * n)

    best_seq = a.copy()
    # initial objective (sum(a)=1 ⇒ score = 2*n*max(conv))
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * float(conv.max())

    v = np.zeros_like(a)
    it = 0
    no_improve = 0
    eps = 1e-12

    while time.time() < start_time + time_limit:
        it += 1

        # evaluate current a
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # active set (indices that are (almost) maximal)
        active = np.where(conv >= M - eps)[0]
        g = np.zeros_like(a)
        for k_idx in active:
            i_start = max(0, k_idx - (n - 1))
            i_end = min(n - 1, k_idx)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k_idx - i_range
            g[i_range] += 2.0 * a[j_range]

        eta = eta0 / (np.sqrt(it) + 1.0)
        v = momentum * v + (1.0 - momentum) * g
        cand = a - eta * v
        a = project_simplex(cand)

        if no_improve >= 3000:
            sigma = 0.03 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    best_score = fast_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑program based alternating refinement (core new ingredient)
# ----------------------------------------------------------------------


def lp_refine(
    seq: np.ndarray,
    max_iter: int,
    time_limit: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Alternating LP refinement:
        * Given current f, compute M = max(conv(f,f)).
        * Solve   max Σ b_i   s.t. conv(f,b) ≤ M , b ≥ 0.
        * Normalise b to unit sum and move a small step towards it.
    Returns the best sequence discovered and its (fast) score.
    """
    f = np.asarray(seq, dtype=np.float64).copy()
    n = f.shape[0]

    # initial best
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(f, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * float(conv.max())
    best_seq = f.copy()
    print(f"[LP] start length={n}, score={best_score:.6f}")

    for it in range(1, max_iter + 1):
        if time.time() > start_time + time_limit:
            break

        # recompute M with current f
        fa = np.fft.rfft(f, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())

        # build Toeplitz matrix for convolution f * b
        col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(col, row)  # shape (2n-1, n)

        # Linear program: maximise Σ b_i  ⇔ minimise -Σ b_i
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            print(f"[LP] iteration {it}: linprog failed – stopping.")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 0.0:
            print(f"[LP] iteration {it}: sum_b zero – stopping.")
            break

        # normalise the LP solution to unit ℓ₁‑norm
        g = b_opt / sum_b

        # step size – decreasing
        t = 0.5 / (1.0 + it)
        f = (1.0 - t) * f + t * g
        f = project_simplex(f)

        # evaluate new score
        fa = np.fft.rfft(f, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[LP] iter {it:3d} improved score={cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Upscaling helper
# ----------------------------------------------------------------------


def upscale_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger length."""
    old_len = seq.shape[0]
    if new_len == old_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len)
    x_new = np.linspace(0.0, 1.0, new_len)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s < 1e-12:
        up = np.full(new_len, 1.0 / new_len)
    else:
        up = up / s
    return up


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a low‑score height sequence within the given time budget.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # 0) obtain a tiny fallback sequence (uniform) for safety
    # ------------------------------------------------------------------
    fallback_len = 1500
    fallback_seq = np.full(fallback_len, 1.0 / fallback_len, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Broad random search (short length, cheap)
    # ------------------------------------------------------------------
    small_len = 2000
    families = [
        "power",
        "exp",
        "twosided",
        "beta",
        "gauss",
        "sine",
        "spikes",
    ]

    best_seq = fallback_seq
    best_score = fast_score(best_seq)

    rand_end = start + 0.25 * budget_s
    while time.time() < rand_end:
        fam = random.choice(families)

        if fam == "power":
            p = random.uniform(0.08, 1.25)
            cand = generate_power_law(small_len, p)
        elif fam == "exp":
            lam = random.uniform(1e-4, 5e-2)
            cand = generate_exponential(small_len, lam)
        elif fam == "twosided":
            pl = random.uniform(0.08, 1.25)
            pr = random.uniform(0.08, 1.25)
            wl = random.uniform(0.2, 5.0)
            wr = random.uniform(0.2, 5.0)
            cand = generate_two_sided_power_law(small_len, pl, pr, wl, wr)
        elif fam == "beta":
            a = random.uniform(0.04, 1.6)
            b = random.uniform(0.04, 1.6)
            cand = generate_beta_shape(small_len, a, b)
        elif fam == "gauss":
            sigma = random.uniform(0.1 * small_len, 0.4 * small_len)
            cand = generate_gaussian(small_len, sigma)
        elif fam == "sine":
            freq = random.uniform(1.0, 3.0)
            cand = generate_sine_shape(small_len, freq)
        else:  # spikes
            k = random.randint(3, 12)
            cand = generate_random_spikes(small_len, k)

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand /= s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start:.1f}s] new global best (rand {fam}) "
                f"score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2) Alternating LP refinement on the short sequence
    # ------------------------------------------------------------------
    time_for_lp = 0.45 * budget_s
    lp_seq, lp_score = lp_refine(
        best_seq,
        max_iter=120,
        time_limit=time_for_lp,
        start_time=time.time(),
    )
    if lp_score < best_score:
        best_score = lp_score
        best_seq = lp_seq
        print(f"[LP] best after refinement: score={lp_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Upscale and final momentum refinement (remaining budget)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0.0:
        final = best_seq.tolist()
        print(f"Time exhausted after LP. Final fast_score={best_score:.6f}")
        return final

    # target sizes to try – each step roughly doubles work, we split time equally
    target_lengths = [4000, 8000, 10000]
    for target_n in target_lengths:
        if time.time() > deadline:
            break
        # interpolate current best to target_n
        cand = upscale_sequence(best_seq, target_n)
        # allocate a share of the remaining budget
        now = time.time()
        remaining = deadline - now
        if remaining <= 0:
            break
        per_budget = remaining / (len(target_lengths) - target_lengths.index(target_n))
        print(f"[ upscale ] length {target_n}, budget {per_budget:.1f}s")
        refined, refined_score = refine_sequence(cand, per_budget, time.time())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"[refine] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final verification with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
