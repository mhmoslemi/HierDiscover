    for it in range(max_iters):

def project_simplex(v):
    # ensures sum(v)=1, v>=0
    # based on Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def targeted_hill_climb(a, fft_len, n, deadline, max_iter=2000):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    while time.time() < deadline and max_iter>0:
        max_iter -= 1
        # compute conv and find peak
        conv = conv_fft(best, fft_len, n)  # linear conv length 2n-1
        peak_idx = int(np.argmax(conv))
        # find i in [0,n-1] that maximizes a[i] * a[peak_idx-i]
        i_low = max(0, peak_idx - (n-1))
        i_high = min(n-1, peak_idx)
        # possible i values
        slice_i = np.arange(i_low, i_high+1)
        products = best[slice_i] * best[peak_idx - slice_i]
        i_star = slice_i[np.argmax(products)]
        # choose j random that is far (|j - i_star| > n/10 maybe)
        j = rng.integers(0, n)

        else:
            # revert
            best[i_star] += delta
            best[j] -= delta

"""
Search for a non‑negative step sequence that minimizes the evaluation
function 2·n·max(conv(a,a)) / (sum(a))**2.

The algorithm combines three complementary ideas:

1. **Smooth‑max L‑BFGS** – the max‑convolution is approximated by a
   log‑sum‑exp (smooth max) and optimized with L‑BFGS on the
   unconstrained soft‑max parametrisation of a probability vector.

2. **Targeted greedy hill‑climb** – after each L‑BFGS run the current
   convolution peak is identified and a small amount of mass is moved
   from the most responsible index to a random faraway index.  The move
   is kept only if the fast (FFT‑based) score improves.

3. **Final refinement** – a finer L‑BFGS schedule (very small smoothing)
   and a long targeted hill‑climb polish the best solution found.

Many random restarts (different initial shapes) are performed until the
time budget expires.  The best distribution is returned as a plain list
of non‑negative floats (the sum is >0, therefore the inside `evaluate_sequence`
function will compute the true score).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
#  Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    a is assumed length n and zero‑padded to fft_len (a power of two).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max version of 2·n·max(convolution) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).
    """
    a = softmax(y)                     # probability vector, sum = 1

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth max -----
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12     # smoothing parameter
    shifted = conv - max_c                # now maximal entry = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max (w.r.t. conv) -----
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # convolution of a (reversed) with w gives ∂smooth_max/∂a_i
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a succession of L‑BFGS optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
#  Initialisation helpers (different shapes on the simplex)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a pre‑softmax vector y for a given initialisation mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # simple greedy Sidon‑like set (≈ √n elements)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if cand + c in sums:
                    conflict = True
                    break
            if not conflict and (cand * 2) not in sums:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "cluster":
        # two Gaussian‑like clusters placed roughly at 1/4 and 3/4 of the interval
        centers = [int(n * 0.25), int(n * 0.75)]
        sigma = n * 0.07
        x = np.arange(n)
        a0 = np.exp(-((x - centers[0]) ** 2) / (2 * sigma ** 2))
        a0 += np.exp(-((x - centers[1]) ** 2) / (2 * sigma ** 2))
        a0 += 1e-12                # avoid exact zeros before the log
        a0 /= a0.sum()
        y0 = np.log(a0)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
#  Greedy hill‑climbing that directly attacks the current convolution peak
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly moves a fraction of the mass from the
    index that contributes most to the current convolution maximum to a
    random far‑away index.  The move is kept only if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current convolution and its peak
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        # indices i that appear in the sum i + (k‑i) = k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        idx = np.arange(i_low, i_high + 1)
        prod = best[idx] * best[k - idx]

        # most responsible index (largest product)
        i_star = idx[np.argmax(prod)]

        # pick a destination far enough from i_star
        j = rng.integers(0, n)
        if j == i_star:
            continue
        if abs(j - i_star) < n // 10:
            # enforce some distance to obtain a noticeable effect
            continue

        # move a modest fraction of the mass
        delta = best[i_star] * rng.random() * 0.1
        if delta <= 0.0:
            continue

        best[i_star] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert the move
            best[i_star] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
#  Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).

    The function respects the supplied time budget (default 1000 s).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.5   # small safety margin

    # ------------------------------------------------------------------
    # problem size – trade‑off between granularity and speed
    # ------------------------------------------------------------------
    n = 5000
    # If a previously discovered construction exists and is long enough,
    # start from at least that length.
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 1000:
                n = max(n, int(tmp.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise with the uniform distribution (baseline)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)      # probability vector, sum = 1
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parametrisation
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # geometric decay of the smoothing factor used in L‑BFGS
    lam_schedule = list(np.geomspace(0.5, 0.001, num=14))

    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "cluster", "best"]

    # ------------------------------------------------------------------
    # Main random‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)

        if mode == "best":
            # Gaussian perturbation of the current best
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ----- smooth‑max L‑BFGS phase -----
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- directed hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final refinement – finer smoothing schedule and longer L‑BFGS runs
    # ------------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-5, num=22))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy polishing (use whatever time is left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
