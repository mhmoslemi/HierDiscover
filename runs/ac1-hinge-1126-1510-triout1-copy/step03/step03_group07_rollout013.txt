"""
Propose a step‑function (non‑negative vector on the simplex) that
minimises the evaluation score

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm consists of four phases

1. **Rich parametric sweep** – many families (power, exponential,
   double‑exponential, beta, arcsine, cosine, triangular, plus a few
   random Dirichlet draws).  The best candidate (lowest max‑convolution)
   is kept.

2. **Linear‑program refinement** – for the current vector `f` we fix the
   current worst convolution value `M = max(conv(f,f))`.  We then solve a
   small LP that *maximises* the ℓ₁‑mass of a new vector `b` under the
   *linear* constraints `conv(f,b) ≤ M`.  The obtained `b` is rescaled
   to the required ℓ₁‑norm and mixed with `f` (step size `t`).  Repeating
   this mixing strictly reduces `M`.  A tiny active‑set cutting‑plane
   loop is used so that the LP never becomes large.

3. **Adam on a smoothed max** – we replace the hard max by a
   soft‑max `log‑sum‑exp(α·conv)/α`.  The gradient is a correlation that
   is computed with FFTs and we perform Adam updates on the simplex.
   The smoothing parameter `α` is slowly increased to make the
   optimisation focus on the true maximum.

4. **Sub‑gradient + cheap hill‑climbing** – after the smooth phase we
   switch to true sub‑gradients (uniform weights on the active set) and
   finally perform many tiny mass‑transfer moves between random
   coordinates.  The best vector found during the whole run is returned,
   finally scaled to the normalisation `sum = sqrt(2·n)` required by
   `evaluate_sequence`.

The whole routine respects a user supplied time budget (default 1000 s)
and aggressively logs progress.  All heavy work uses FFTs (`O(L log L)`)
and the LP uses only a handful of constraints, so the method comfortably
fits into the allowed time.
"""

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _fft_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _fft_convolution_two(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of two vectors a and b (real part)."""
    fft_a = np.fft.fft(a, n=L)
    fft_b = np.fft.fft(b, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_b))
    return conv_full[: len(a) + len(b) - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    # reversed a (zero‑padded) in the frequency domain:
    fft_a_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:
        w[conv.argmax()] = 1.0
    return w


# ----------------------------------------------------------------------
def _build_A_matrix(f: np.ndarray, idxs: list[int]) -> np.ndarray:
    """Build the (len(idxs) × len(f)) matrix representing conv(f,·) on the given lags."""
    n = f.shape[0]
    r = len(idxs)
    A = np.zeros((r, n), dtype=np.float64)
    for row, k in enumerate(idxs):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=int)
        j_vals = k - i_vals
        A[row, j_vals] = f[i_vals]
    return A


def _lp_improve(f: np.ndarray, M: float, sum_target: float,
                L: int, max_lp_iters: int = 30) -> np.ndarray | None:
    """
    Perform one LP‑based refinement step.
    Returns the improved vector `g` (already normalised to sum = sum_target)
    or None if the LP cannot increase the ℓ₁‑mass.
    """
    n = f.shape[0]
    # initial active set: points where conv(f,f) attains the current max
    conv_ff = _fft_convolution(f, L)
    tol = M * 1e-12 + 1e-15
    active = list(np.where(conv_ff >= M - tol)[0])
    if not active:
        active = [int(conv_ff.argmax())]

    # cutting‑plane loop – keep adding the most violated constraints
    for _ in range(8):                     # at most a few augmentations
        A = _build_A_matrix(f, active)
        b_ub = np.full(len(active), M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)     # maximise sum(b)  → minimise -sum(b)

        res = optimize.linprog(c, A_ub=A, b_ub=b_ub,
                               bounds=[(0, None)] * n,
                               method='highs',
                               options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            return None
        b = res.x

        # verify all constraints (including those not in the active set)
        conv_fb = _fft_convolution_two(f, b, L)
        viol = np.where(conv_fb > M + 1e-12)[0]
        if viol.size == 0:           # all constraints satisfied – optimal b found
            break
        # add the most violated lags (up to 5)
        viol_vals = conv_fb[viol] - M
        add = viol[np.argsort(-viol_vals)[:5]]
        for k in add:
            if k not in active:
                active.append(int(k))

    # after the loop we have a feasible b that maximises sum(b) subject to conv≤M
    sum_b = b.sum()
    if sum_b <= sum_target * (1.0 + 1e-12):
        # cannot increase total mass while respecting the bound → stop
        return None

    # scale to the required ℓ₁‑norm
    scale = sum_target / sum_b
    g = b * scale
    return g
