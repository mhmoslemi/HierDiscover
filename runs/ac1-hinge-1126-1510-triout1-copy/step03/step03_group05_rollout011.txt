def gradient_smooth_max(a, beta):
    # a is numpy array sum 1.
    n = len(a)
    fft_len = next_pow2(3*n) # Ensure length enough for conv with w_rev length 2n-1
    # compute convolution a*a using FFT
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2*n - 1]  # full conv
    # compute softmax weights
    max_val = conv.max()
    # subtract max to avoid overflow
    exps = np.exp(beta * (conv - max_val))
    w = exps / exps.sum()
    # compute gradient: g[j] = 2 * sum_i a_i w[i + j]
    # w length = 2n-1
    w_rev = w[::-1]
    # convolution a * w_rev
    # compute via FFT with same length maybe should be >= n + (2n-1) -1 = 3n-2
    L = next_pow2(3*n)
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa * fw, n=L)[: 3*n - 2]
    # gradient
    idx_start = 2*n - 2  # index of central element in conv_rev corresponding to lag 0
    # for j in 0..n-1, gradient = 2 * conv_rev[idx_start - j]
    g = 2.0 * conv_rev[idx_start - np.arange(n)]

"""
Proposes a step‑function sequence using smooth‑max gradient descent.
The algorithm keeps the sequence on the probability simplex and
optimises the quantity 2·n·max(conv(a,a)) (the exact objective of
evaluate_sequence).  A soft‑max approximation of the max is used so
that a differentiable gradient is available; the gradient is computed
by FFT‑based correlation in O(n log n).  Several random initial
shapes and a quick “U‑shaped” two‑sided power‑law start are tried,
each is refined for a share of the remaining time.  The best
candidate found is returned as a plain list.
"""

import time
import random
import numpy as np

# Helper utilities
def next_pow2(x: int) -> int:
    p = 1
    while p < x:
        p <<= 1
    return p

def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) for a non‑negative sequence.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exact 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w

# Shape generators – used only for random initialisation
def generate_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)

def generate_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)

def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right

def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))

# Gradient of the smooth max of the auto‑convolution
def grad_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Returns the Euclidean gradient of
        smooth_max_beta (conv(a,a))
    where smooth_max_beta(x) = log( sum(exp(beta * x)) ) / beta.
    """
    n = a.shape[0]
    # Full convolution a*a
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    # soft‑max weights (numerically stable)
    max_val = conv.max()
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # gradient g_j = 2 * Σ_i a_i * w_{i+j}
    # Compute convolution of a with reversed w.
    w_rev = w[::-1]
    L = next_pow2(3 * n)
    fa = np.fft.rfft(a, n=L)
    fw = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa * fw, n=L)[: 3 * n - 2]
    idx_center = 2 * n - 2          # corresponds to lag 0
    g = 2.0 * conv_rev[idx_center - np.arange(n)]
    return g

# One optimisation run – projected (momentum) gradient descent
def optimise_one_start(
    init: np.ndarray,
    time_budget: float,
    start_time: float,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
    eta0: float = 0.5,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Refines a single initial sequence using smooth‑max gradient descent.
    Returns the best sequence seen and its exact fast_score.
    """
    a = project_simplex(init.copy())
    n = a.shape[0]
    best_seq = a.copy()
    # initial exact score
    best_score = fast_score(best_seq)
    vel = np.zeros_like(a)

    # schedule parameters
    total_iters_est = 1e9  # placeholder, we stop on time

    it = 0
    no_improve = 0
    while time.time() < start_time + time_budget:
        it += 1
        # linear interpolation for beta
        elapsed = time.time() - start_time
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / time_budget)

        # gradient of smooth max
        g = grad_smooth_max(a, beta)

        # momentum update
        lr = eta0 / (np.sqrt(it) + 1.0)
        vel = momentum * vel + (1.0 - momentum) * g
        cand = a - lr * vel
        a = project_simplex(cand)

        # exact score (cheap: we already have convolution from gradient computation)
        # recompute for exactness every few iterations to keep cost low
        if it % 5 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # occasional random kick to escape plateaus
        if no_improve >= 2000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact evaluation
    final_score = fast_score(best_seq)
    return best_seq, final_score

"""
Proposes a step‑function sequence using smooth‑max gradient descent.

The target of `evaluate_sequence` is
    2·n·max(conv(a,a)) / (sum a)² .
Because the expression is homogeneous we normalise every candidate so that
∑aᵢ = 1 and minimise the simpler quantity 2·n·max(conv(a,a)).
A differentiable approximation of the “max’’ is obtained with a soft‑max,
the gradient of which can be written as a correlation between the current
vector and a set of soft‑max weights.  Both the convolution and the
correlation are evaluated by FFT in O(n log n) time, allowing many
iterations even for n≈1.5·10⁴.

The algorithm proceeds as follows

1. Choose a size n (8 k – 16 k) and build a small pool of initial shapes.
   If a historic good sequence `height_sequence_1` is available it is
   interpolated to the chosen length.  Additional two‑sided power‑law
   and random family members are added.

2. Each pool member is refined for an equal share of the remaining time.
   The refinement uses projected (momentum) gradient descent on the
   probability simplex.  The smooth‑max temperature β is increased
   linearly from 1 to 200, while the learning rate decays as 1/√t.
   A periodic random “kick’’ prevents stagnation.

3. The best sequence found is finally checked with the supplied
   `evaluate_sequence` (the slow but exact evaluator) and returned as a
   plain Python list of non‑negative floats.

All helper functions are defined at top level; the only public entry point
required by the harness is `propose_candidate`.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) for a non‑negative sequence.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Families of random initial shapes
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha‑1)·(n‑i)^(beta‑1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Gradient of the smooth‑max of the auto‑convolution
# ----------------------------------------------------------------------


def grad_smooth_max(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of   smooth_max_beta(conv(a,a))   where

        smooth_max_beta(x) = log(∑ exp(beta·x_i)) / beta.

    Returns a vector of length len(a).  All arithmetic is performed in
    double precision using FFTs for O(n log n) cost.
    """
    n = a.shape[0]

    # ---- 1) convolution a*a (full length 2n‑1) ----
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    # ---- 2) soft‑max weights (stable) ----
    max_val = conv.max()
    w = np.exp(beta * (conv - max_val))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.shape[0])
    else:
        w /= w_sum

    # ---- 3) gradient g_j = 2·∑_i a_i·w_{i+j} ----
    # Correlation of a with w = convolution of a and reversed w.
    w_rev = w[::-1]
    L = next_pow2(3 * n)                  # ≥ n + (2n‑1) – 1
    fa_big = np.fft.rfft(a, n=L)
    fw_rev = np.fft.rfft(w_rev, n=L)
    conv_rev = np.fft.irfft(fa_big * fw_rev, n=L)[: 3 * n - 2]

    idx_center = 2 * n - 2                    # index in conv_rev that corresponds to lag 0
    g = 2.0 * conv_rev[idx_center - np.arange(n)]
    return g


# ----------------------------------------------------------------------
# Single optimisation run (projected momentum gradient descent)
# ----------------------------------------------------------------------


def optimise_one_start(
    init: np.ndarray,
    time_budget: float,
    start_time: float,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
    eta0: float = 0.5,
    momentum: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Refines a candidate sequence for at most ``time_budget`` seconds.
    Returns the best sequence seen and its exact fast_score.
    """
    a = project_simplex(init.copy())
    n = a.shape[0]

    best_seq = a.copy()
    best_score = fast_score(best_seq)

    vel = np.zeros_like(a)
    it = 0
    no_improve = 0

    while time.time() < start_time + time_budget:
        it += 1
        # linear schedule for the soft‑max temperature
        elapsed = time.time() - start_time
        beta = beta_start + (beta_end - beta_start) * min(1.0, elapsed / time_budget)

        # gradient of the smooth max
        g = grad_smooth_max(a, beta)

        # Nesterov‑style momentum and diminishing step‑size
        lr = eta0 / (np.sqrt(it) + 1.0)
        vel = momentum * vel + (1.0 - momentum) * g
        cand = a - lr * vel
        a = project_simplex(cand)

        # exact objective every few iterations (cheap because we already have the FFTs)
        if it % 5 == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # occasional random perturbation when stuck
        if no_improve >= 2000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # one final exact evaluation of the best candidate
    final_score = fast_score(best_seq)
    return best_seq, final_score
