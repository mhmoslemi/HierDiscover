"""
A fast multi‑phase optimizer for the max‑convolution norm.
The algorithm works on a probability simplex (sum‑to‑1, non‑negative).

Phase 1 – many random restarts of a continuation L‑BFGS on a
log‑sum‑exp smooth approximation of the max‑convolution.  The
smoothing parameter λ is decreased in a schedule, each stage using
a relatively long L‑BFGS run (max 300 iterations per λ).  A diverse
set of initial distributions (uniform, exponential, power‑law,
edge‑heavy, sine‑shaped, Gaussian) is tried.

Phase 2 – projected sub‑gradient descent with a simple back‑tracking
line‑search (more aggressive step‑size decay than the reference).

Phase 3 – simulated annealing that repeatedly moves a random fraction
of mass between two coordinates; the temperature starts higher
(0.08) and cools slowly, accepting uphill moves with the usual
Metropolis probability.

Finally a greedy hill‑climb (mass‑transfer moves) is applied until the
time budget expires.  The best sequence found (as a plain Python list)
is returned; an external ``evaluate_sequence`` call is used only once
at the very end to report the true score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast 2·n·max(a*a) assuming sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def smooth_obj_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (objective, gradient) where the objective is the
    log‑sum‑exp smoothed max of the full convolution a*a,
    with a = softmax(y).  The gradient is ∂obj/∂y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c            # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt conv entries
    w = exp_shift / Z                       # ∂smooth_max/∂conv_i

    # gradient wrt a: convolution of w with reversed a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (multiple λ values)
# ----------------------------------------------------------------------
def lbfgS_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result.  Returns the best a (sum=1) and its
    fast_score seen so far.
    """
    y = np.log(a_start + 1e-12)          # start in ℝⁿ
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=smooth_obj_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best score={best_sc:.6f} (λ={lam})")

        # a brief hill‑climb after each λ can help escape plateaus
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_sc = hill_climb(best_a, best_sc,
                                     fft_len_conv, n,
                                     hc_deadline,
                                     max_iter=800)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with back‑tracking line search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a back‑tracking line‑search and Euclidean simplex projection.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)
    it = 0
    step0 = 0.5

    while time.time() < deadline:
        # current convolution and mask for active constraints
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size schedule (decreasing)
        step = step0 / np.sqrt(it + 1.0)

        improved = False
        for _ in range(5):
            cand = simplex_projection(a - step * g)
            cand_sc = fast_score(cand, fft_len_conv, n)
            if cand_sc < best_sc:
                best_sc, best_a = cand_sc, cand.copy()
                a = cand
                improved = True
                break
            step *= 0.5

        if not improved:
            # still move a tiny bit to avoid stagnation
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000):
    """
    Greedy hill‑climb that moves a small random fraction of mass
    from one index to another while keeping the sum fixed.
    Only strictly improving moves are accepted.
    """
    best = a.copy()
    best_sc = cur_sc
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 10 % of the mass at i (larger than original 5 %)
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_sc = fast_score(best, fft_len, n)
        if new_sc < best_sc:
            best_sc = new_sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Simulated annealing with adaptive temperature
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  cur_sc: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated‑annealing that moves a random fraction of mass between
    two coordinates.  A higher initial temperature and a slower cooling
    schedule are used compared to the reference implementation.
    """
    cur_a = a.copy()
    cur_sc = cur_sc
    best_a = a.copy()
    best_sc = cur_sc
    rng = np.random.default_rng()
    T = 0.08               # start a bit hotter
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 15 % transfer of mass from i to j
        delta = cur_a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_sc = fast_score(cand, fft_len, n)
        if new_sc < cur_sc:
            cur_a, cur_sc = cand, new_sc
            if new_sc < best_sc:
                best_a, best_sc = cand.copy(), new_sc
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_sc - cur_sc) / T)
            if rng.random() < prob:
                cur_a, cur_sc = cand, new_sc

        it += 1
        # slower cooling
        if it % 4000 == 0:
            T *= 0.94
            T = max(T, 5e-6)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Helper to produce diverse initial distributions
# ----------------------------------------------------------------------
def random_initial_distribution(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector of length n."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'sine', 'gauss'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(5e-4, 2e-2)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.008, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        # give a small bump at both ends
        a[:m_edge] = np.sqrt(n) / (2 * n)
        a[-m_edge:] = np.sqrt(n) / (2 * n)
    elif mode == 'sine':
        # half‑sine, strictly positive
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    else:  # 'gauss'
        sigma = rng.uniform(0.12, 0.25) * n
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (larger n gives a tighter theoretical lower bound)
    # ------------------------------------------------------------------
    max_n = 4000                 # feel free to adjust; 4000 fits comfortably
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n]
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient‑related convolutions

    best_a = a0.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a safety margin
    phase1_end = start_time + 0.55 * budget_s   # ~55 % of the budget
    phase2_end = start_time + 0.80 * budget_s   # next ~25 %
    phase3_end = deadline                        # remaining time

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    # a finer λ schedule that reaches very small smoothing parameters
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.07, 0.04,
                    0.02, 0.01, 0.005, 0.002, 0.001]
    maxiter_per_lam = 300   # a bit longer than the original 250

    while time.time() < phase1_end:
        a_start = random_initial_distribution(n, rng)
        a_opt, sc_opt = lbfgS_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len_conv,
                                           fft_len_g,
                                           n)
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (more aggressive)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_sc = subgradient_optimize(best_a,
                                                sg_deadline,
                                                fft_len_conv,
                                                fft_len_g,
                                                n)
        print(f"[sub‑gradient] score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing + final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = random_anneal(best_a,
                                        best_sc,
                                        fft_len_conv,
                                        n,
                                        phase3_end)
        print(f"[anneal] score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_sc = hill_climb(best_a,
                                 best_sc,
                                 fft_len_conv,
                                 n,
                                 final_deadline,
                                 max_iter=8000)

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
