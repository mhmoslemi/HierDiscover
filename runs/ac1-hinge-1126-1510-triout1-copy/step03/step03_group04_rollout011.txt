def lp_step(f, M, epsilon=1e-12):

    n = len(f)

    # Build Toeplitz matrix A shape (2n-1, n). Use loops to fill.

    K = 2*n - 1

    A = np.zeros((K, n), dtype=np.float64)

    for k in range(K):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # indices i in [lo, hi], j = k - i
        # f_j = f[k-i]
        A[k, lo:hi+1] = f[k - np.arange(lo, hi+1)]

    # Linear program: maximize sum(b), i.e. minimize -sum(b)
    c = -np.ones(n)

    # Upper bound M for each conv constraint: A @ b <= M
    bounds = [(0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=np.full(K, M), bounds=bounds, method='highs', options={'presolve': True, 'time_limit': remaining_time})

    return res

"""
Propose a low‑value step sequence.

The script combines three complementary ideas:

1. **Stochastic L‑BFGS on a smooth approximation** – many random
   restarts on the log‑sum‑exp (smooth‑max) of the auto‑convolution.
   The softmax parametrisation guarantees non‑negativity and the simplex
   constraint.

2. **Linear‑programming “active‑set” improvement** – given a current
   vector `f` (∑f=1) we maximise `∑b` under the linear constraints
   `f * b ≤ M` where `M = max(f*f)`.  By normalising the LP solution
   (`g = b/∑b`) we obtain a new feasible vector with a strictly smaller
   max‑convolution (the objective is scale‑invariant).  Repeating this
   step quickly drives the score down.

3. **Fine‑grained polishing** – a few rounds of projected sub‑gradient
   descent, simulated annealing and a final greedy hill‑climb.

All phases respect the global time limit (default 1000 s).  The best
sequence found is returned as a plain list of floats ready for
`evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a via FFT (full length 2·len(a)‑1).
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector `a`
    (∑a = 1).  Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, ∑x=1}.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.
    Returns (value, gradient) where the gradient has shape (n,).
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # convolution of a_rev with w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimiser (true max)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) with a simple sub‑gradient method.
    Each iterate is projected back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: 2 * (mask ▷ a_rev)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb.
    Randomly moves a small fraction of one coordinate to another,
    keeping the simplex constraint.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple SA that moves a fraction of mass between two positions.
    Improves the best known solution while exploring uphill moves.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta
        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming improvement step
# ----------------------------------------------------------------------


def lp_improve(f: np.ndarray,
               best_f: np.ndarray,
               best_score: float,
               fft_len_conv: int,
               deadline: float):
    """
    One‑direction LP improvement:
    maximise Σb subject to  f * b ≤ M  (where M = max(f*f)).
    Normalise the result and accept it if it reduces the objective.
    """
    n = f.shape[0]
    while time.time() < deadline:
        # Current maximal convolution value
        conv = full_conv(f, fft_len_conv)
        M = conv.max()

        # Build Toeplitz constraint matrix A (shape (2n‑1, n))
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for i in range(n):
            A[i:i + n, i] = f

        # Linear programme: maximise sum(b)  ⇒ minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs')
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-8:    # no further gain possible
            break

        g = b / sum_b
        score_g = fast_score(g, fft_len_conv, n)

        if score_g < best_score - 1e-12:
            best_score = score_g
            best_f = g.copy()
            print(f"[LP] improvement: score={best_score:.6f}")
        # continue from the new point
        f = g
    return best_f, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops a bit early
        (≈0.5 s) to guarantee a return.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Initialise sequence (use previous construction if available)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size == 0:
            n = 2000
            a0 = np.full(n, 1.0 / n)
        else:
            max_n = 3000
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            if init_seq.sum() <= 0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    # Ensure strict positivity (needed for the LP step)
    epsilon = 1e-9
    a0 = np.maximum(a0, epsilon)
    a0 /= a0.sum()

    # FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient related convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}  fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS runs on the smoothed max
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    lam_schedule = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]

    iter_cnt = 0
    while time.time() < phase1_end:
        lam = lam_schedule[iter_cnt % len(lam_schedule)]

        # random start for the soft‑max variables
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'prev'])
        if mode == 'prev':
            y0 = np.log(best_a + 1e-12)
        elif mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(5e-4, 2e-2)
            a_tmp = np.exp(-rate * np.arange(n))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.2, 2.5)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        else:   # edge‑heavy heuristic
            edge_frac = rng.uniform(0.008, 0.04)
            m_edge = max(1, int(n * edge_frac))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a_tmp = np.full(n, a_center)
            a_tmp[:m_edge] = a_edge
            a_tmp[-m_edge:] = a_edge
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)

        # L‑BFGS on smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'gtol': 1e-12,
                                'disp': False})
        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS {iter_cnt:04d}] lam={lam:.3f} mode={mode} score={best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_dead = min(phase1_end, time.time() + 0.25)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_dead,
                                        max_iter=1200)
        iter_cnt += 1

    # ------------------------------------------------------------------
    # Phase 2 – LP based improvement (the core novel step)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.65 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = lp_improve(best_a, best_a, best_score,
                                        fft_len_conv,
                                        phase2_end)
        print(f"[LP] after phase 2 best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient refinement (longer)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.85 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase3_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] refined score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing (non‑local)
    # ------------------------------------------------------------------
    phase4_end = deadline
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase4_end,
                                                 rng)
        print(f"[SA] after annealing score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=5000)

    # ------------------------------------------------------------------
    # Final evaluation (for logging only)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluated score = {eval_val:.6f}")

    return final_a.tolist()
