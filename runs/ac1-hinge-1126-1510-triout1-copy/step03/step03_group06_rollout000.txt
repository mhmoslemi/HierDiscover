"""
Adaptive projected sub‑gradient descent with line‑search and annealed
local refinement.

The objective to minimise is

    score(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

which is homogeneous, therefore we keep the vector on the probability
simplex (a_i ≥ 0, Σa_i = 1).  The algorithm works as follows:

* Choose a few problem sizes (n ≈ 4k … 32k).  Larger n gives a lower
  theoretical limit (≈1) because the uniform self‑convolution has
  height 1/(2n‑1).

* For each size run many independent restarts.
  – initialise a with a uniform vector, a random Dirichlet sample,
    or a power‑law a_i ∝ i^{‑p}.
  – run an adaptive projected sub‑gradient method:
        – compute the full convolution by FFT,
        – find the index k where the convolution attains its maximum,
        – the sub‑gradient of the active peak is
              grad_i = 2·a_{k‑i}  (only for i that contribute).
        – perform a simple back‑tracking line search (step, step/2,
          step/4) and keep the best feasible move,
        – project the result back onto the simplex.
  – after the descent perform a cheap simulated‑annealing style local
    refinement that repeatedly shifts a small amount of mass between
    random coordinates.  Larger moves (up to 20 % of the source mass)
    and a slowly cooling temperature help to escape shallow minima.

* Keep the globally best sequence seen over all lengths and restarts.
  The routine stops just before the time budget expires and returns the
  best sequence as a plain list of Python floats.

The whole search fits comfortably in the 1000 s budget on a single
CPU core.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, using power‑of‑two length)
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the full linear convolution a*a and its maximal entry.
    """
    n = a.shape[0]
    # next power of two that can contain the full convolution (2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper that returns only the maximal convolution entry."""
    return compute_conv_and_max(a)[1]


# ----------------------------------------------------------------------
# Adaptive projected sub‑gradient descent with simple line search
# ----------------------------------------------------------------------
def subgradient_descent_adaptive(
    a0: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the simplex with back‑tracking line
    search.  Returns the best sequence seen and its score (with Σa = 1).
    """
    a = a0.copy()
    a = a / a.sum()                     # enforce simplex
    n = a.shape[0]

    best_seq = a.copy()
    best_score = float("inf")
    step = 1.0                           # initial step size

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # convolution and current score
        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[subgrad n={n} it={iteration}] new best={score:.6f}")

        # sub‑gradient of the active peak
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # try a few step sizes (back‑tracking)
        improved = False
        for factor in (1.0, 0.5, 0.25):
            step_try = step * factor
            a_candidate = project_simplex(a - step_try * grad)

            # evaluate candidate
            cand_score = 2.0 * n * compute_max_conv(a_candidate)

            if cand_score < score:
                a = a_candidate
                step = step_try * 1.1          # modestly increase after success
                improved = True
                break

        if not improved:
            step *= 0.5                         # shrink step if no improvement

        # safety stop – very tiny step size will not move the point any more
        if step < 1e-9:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Stochastic local refinement (simulated annealing style)
# ----------------------------------------------------------------------
def local_refine(
    a0: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap random walk that moves a (potentially sizable) amount of mass
    between random coordinates, accepting improvements or Metropolis‑type
    uphill moves.  Returns the best sequence seen.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.05                     # initial temperature
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # decide direction of transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # amount to move – up to 20 % of the source mass
        delta = np.random.uniform(0.0, 0.20) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s                                 # re‑normalise to simplex

        new_score = 2.0 * n * compute_max_conv(new)

        # Accept if better, otherwise with Metropolis probability
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # cooling
        temperature *= 0.9995
        temperature = max(temperature, 1e-6)

    return best_seq, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (height sequence) with a low
    evaluation score.  The routine runs several restarts of an adaptive
    projected sub‑gradient optimiser on a handful of problem sizes,
    followed by a short annealing refinement.  The best sequence ever
    observed is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem sizes to try – larger sizes have a lower theoretical limit.
    # The list is kept modest to respect the time budget.
    # ------------------------------------------------------------------
    length_candidates = [4000, 8000, 16000, 32000]
    # Remove sizes that would be obviously too large for the remaining budget
    max_len = 50000
    length_candidates = [n for n in length_candidates if n <= max_len]

    # ------------------------------------------------------------------
    # Global best across all sizes / restarts
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Main search loop
    # ------------------------------------------------------------------
    for n in length_candidates:
        # Loop restarts for this length until time runs out
        while time.time() < deadline:
            # ---------- initialise a start point ----------
            r = np.random.rand()
            if r < 0.15:
                # uniform start
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            elif r < 0.55:
                # Dirichlet (random simplex) start
                a0 = np.random.dirichlet(np.ones(n))
            elif r < 0.85:
                # power‑law start
                p = np.random.uniform(0.45, 0.80)
                idx = np.arange(1, n + 1, dtype=np.float64)
                a0 = idx ** (-p)
                a0 /= a0.sum()
            else:
                # perturb current global best if sizes match
                if global_best_seq is not None and len(global_best_seq) == n:
                    a0 = np.array(global_best_seq, copy=True)
                    a0 += 0.02 * np.random.randn(n)
                    a0 = np.maximum(a0, 0.0)
                    s = a0.sum()
                    if s == 0.0:
                        a0 = np.full(n, 1.0 / n)
                    else:
                        a0 /= s
                else:
                    # fallback – exponential decay
                    lam = np.random.uniform(0.0005, 0.005)
                    idx = np.arange(n, dtype=np.float64)
                    a0 = np.exp(-lam * idx)
                    a0 /= a0.sum()

            # Estimate remaining time and decide on a generous iteration budget
            time_left = deadline - time.time()
            if time_left < 1.0:
                break

            # Run the adaptive sub‑gradient optimiser
            sub_deadline = time.time() + min(0.6 * time_left, 30.0)  # at most ~30 s per run
            seq, score = subgradient_descent_adaptive(a0, sub_deadline)

            # Local refinement if time permits
            time_now = time.time()
            if deadline - time_now > 0.5:
                refine_deadline = deadline
                seq, score = local_refine(seq, refine_deadline)

            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best_seq = seq.copy()
                print(
                    f"[len={n}] new GLOBAL best score={score:.6f}"
                )
                # Early exit if we have already crossed the target threshold
                if global_best_score < 1.503:
                    deadline = time.time()  # force outer loops to stop
                    break

        # break outer length loop if time is exhausted or target reached
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a sensible default.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-0.55)
        a /= a.sum()
        global_best_seq = a

    # Convert to plain Python list of floats
    return [float(x) for x in global_best_seq.tolist()]
