```
def _initial_candidates(n):
    xs = np.arange(n, dtype=np.float64)
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # power law p in [0,2] step maybe 0.2
    for p in np.linspace(0.0, 2.0, 21):
        base = (xs+1)**p
        candidates.append(base / base.sum())
        candidates.append(base[::-1] / base.sum())
    # exponential
    for lam in np.linspace(0.0,5.0,21):
        exp_shape = np.exp(-lam*xs)
        candidates.append(exp_shape / exp_shape.sum())
        candidates.append(exp_shape[::-1] / exp_shape.sum())
    # double expo
    for lam in np.linspace(0.0,5.0,11):
        left = np.exp(-lam*xs)
        u = left + left[::-1]
        candidates.append(u / u.sum())
    # beta families with small parameters
    alphas_betas = [0.2, 0.5, 1.0, 2.0, 5.0]
    x_mid = (xs + 0.5)/n
    for a in alphas_betas:
        for b in alphas_betas:
            beta_shape = (x_mid**(a-1))*((1-x_mid)**(b-1))
            if np.all(np.isfinite(beta_shape)):
                candidates.append(beta_shape / beta_shape.sum())
                candidates.append(beta_shape[::-1] / beta_shape.sum())
    # arcsine
    arcsine = 1.0/np.sqrt(x_mid*(1-x_mid) + 1e-30)
    candidates.append(arcsine/arcsine.sum())
    candidates.append(arcsine[::-1]/arcsine.sum())
    # random Dirichlet (few)
    for _ in range(5):
        rand_vec = np.random.rand(n)
        candidates.append(rand_vec / rand_vec.sum())
    # also random Gamma(0.5) to produce spiky
    for _ in range(5):
        gamma = np.random.gamma(0.5, 1.0, size=n)
        candidates.append(gamma / gamma.sum())
    return candidates
```

```
def _adam_optimize(a, L, phase, deadline, max_iters=20000):
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1,beta2 = 0.9,0.999
    eps = 1e-8
    lr0 = 0.2
    best_max = _compute_convolution(a, L).max()
    best_a = a.copy()
    it = 0
    alpha_start, alpha_end = 4.0, 300.0
    while time.time() < deadline and it < max_iters:
        it +=1
        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        if cur_max < best_max - 1e-12:
            best_max, best_a = cur_max, a.copy()
        # compute softmax weights
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)
        grad = _gradient_from_weights(a, w, phase, L)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m/(1-beta1**it)
        v_hat = v/(1-beta2**it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)
        a = a_new
    return best_a, best_max
```

```
def _hill_climb(a, L, deadline, max_moves=None):
    n = len(a)
    best_max = _compute_convolution(a, L).max()
    best_a = a.copy()
    moves = 0
    while time.time() < deadline:
```

The goal is to minimise  

\[
\text{score}=2n\frac{\max (a*a)}{(\sum a)^2},
\]

which – after normalising the step‑function to \(\sum a = 1\) – is simply  
\(\text{score}=2n\;\max (a*a)\).  
Thus we have to find a non‑negative vector `a` on the simplex that makes the
largest entry of its auto‑convolution as small as possible.

The new search routine combines several ideas:

1. **Broad initialisation** – a large set of analytic families (power law,
   exponential, double‑exponential, Beta, arcsine) together with a few random
   Dirichlet / Gamma draws.  
   The best of these is kept as the starting point.

2. **Adam on a smooth approximation** – the hard max is replaced by a
   soft‑max with a temperature that is increased from 4 up to 300.
   Adam with Euclidean projection onto the simplex quickly reaches a good
   basin.

3. **Projected sub‑gradient descent** – once the temperature is large the true
   sub‑gradient of the max‑convolution is used (uniform weight on the active
   set).  A diminishing stepsize \(c/\sqrt{t}\) drives the solution further
   down.

4. **Intensive random hill‑climbing** – tiny mass transfers between random
   coordinates are accepted only if they improve the objective; this helps
   escape tiny plateaus.

5. **LP‑refinement (Menkes‑Ruderman step)** – for the current iterate `f`
   we solve the linear programme  

   \[
   \begin{aligned}
   \max\ &\sum_j g_j\\
   \text{s.t. } &g_j\ge 0,\;
   (f*g)_k\le M_f\;\forall k,
   \end{aligned}
   \]

   where \(M_f=\max(f*f)\).  
   The optimal `g` is renormalised to unit‑sum and mixed with `f`
   (\(f\leftarrow(1-t)f+t\,g\)).  Repeating this step (with shrinking
   mixing factor `t`) provably never increases the maximal convolution and
   typically yields a noticeable improvement.

6. **Multi‑size search** – the whole pipeline is executed for several
   sequence lengths (including the length of the previously found best
   construction).  The best result over all lengths is returned.

All heavy work (convolutions, gradients) is performed with FFTs
\(O(n\log n)\).  The routine respects the supplied time budget (default
1000 s) and prints progress and intermediate scores.

```python
"""
Propose a step‑function that minimises
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------

1.  For a given length `n` generate a large pool of candidate shapes
    (power‑law, exponential, double‑exponential, Beta, arcsine,
    several random Dirichlet/Gamma draws).  Keep the one with the smallest
    max‑convolution.

2.  Adam on a smooth max (soft‑max with a temperature that is raised from
    4 to 300).  After each step the vector is projected onto the simplex.

3.  Projected sub‑gradient descent on the true max‑convolution.
    The sub‑gradient is the average correlation over the active set.

4.  Random hill‑climbing that moves a small fraction of mass from one
    coordinate to another whenever it improves the objective.

5.  LP‑refinement (Menkes‑Ruderman step):
    maximise Σ g_j subject to g≥0 and ‖f∗g‖_∞ ≤ M_f.
    The solution `g` is normalised and mixed with `f`.  Repeating the step
    (with a shrinking mixing factor) never increases the maximal
    auto‑convolution and often reduces it further.

6.  Run the whole pipeline for several lengths (including the length of the
    supplied `height_sequence_1`).  Return the best sequence found,
    scaled to the normalisation used by `evaluate_sequence`.

All steps respect the overall time budget, use FFTs for speed and project
onto the simplex with the Euclidean projector.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase          # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    n = a.shape[0]
    # correlation length is 2n-1, we need the central n entries
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    # tolerance for active set (relative + absolute)
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # should never happen
        w[conv.argmax()] = 1.0
    return w


# ----------------------------------------------------------------------
# Initial candidate generation (analytic families + random draws)
# ----------------------------------------------------------------------


def _initial_candidates(n: int):
    """Return a list of candidate vectors (already normalised to sum=1)."""
    xs = np.arange(n, dtype=np.float64)
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # power‑law family
    for p in np.linspace(0.0, 2.0, 21):
        base = (xs + 1) ** p
        cand = base / base.sum()
        candidates.append(cand)
        candidates.append(cand[::-1])

    # exponential family
    for lam in np.linspace(0.0, 5.0, 21):
        e = np.exp(-lam * xs)
        cand = e / e.sum()
        candidates.append(cand)
        candidates.append(cand[::-1])

    # double‑exponential (U‑shape)
    for lam in np.linspace(0.0, 5.0, 11):
        left = np.exp(-lam * xs)
        u = left + left[::-1]
        cand = u / u.sum()
        candidates.append(cand)

    # Beta‑density family (including very spiky shapes)
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    betas = alphas
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            beta = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
            if np.all(beta > 0) and np.isfinite(beta).all():
                cand = beta / beta.sum()
                candidates.append(cand)
                candidates.append(cand[::-1])

    # arcsine‑like shape
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    cand = arcsine / arcsine.sum()
    candidates.append(cand)
    candidates.append(cand[::-1])

    # a few random Dirichlet / Gamma draws (spiky possibilities)
    for _ in range(5):
        rand = np.random.rand(n)
        candidates.append(rand / rand.sum())
    for _ in range(5):
        gamma = np.random.gamma(0.5, 1.0, size=n)
        candidates.append(gamma / gamma.sum())

    return candidates


# ----------------------------------------------------------------------
# Optimisation phases for a fixed length `n`
# ----------------------------------------------------------------------


def _adam_optimize(a: np.ndarray, L: int, phase: np.ndarray,
                   deadline: float, max_iters: int = 20000) -> tuple:
    """Adam on the smooth max‑convolution."""
    n = a.shape[0]
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.20                     # base learning rate
    alpha_start, alpha_end = 4.0, 300.0

    best_a = a.copy()
    best_max = _compute_convolution(a, L).max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        if cur_max < best_max - 1e-14:
            best_max, best_a = cur_max, a.copy()

        # temperature schedule
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = _softmax_weights(conv, alpha)

        grad = _gradient_from_weights(a, w, phase, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_simplex(a - step)

    return best_a, best_max


def _subgrad_optimize(a: np.ndarray, L: int, phase: np.ndarray,
                      deadline: float, max_iters: int = 40000) -> tuple:
    """Projected sub‑gradient descent on the true max‑convolution."""
    n = a.shape[0]
    best_a = a.copy()
    best_max = _compute_convolution(a, L).max()

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _compute_convolution(a, L)
        cur_max = conv.max()
        if cur_max < best_max - 1e-14:
            best_max, best_a = cur_max, a.copy()

        w = _submax_weights(conv)
        grad = _gradient_from_weights(a, w, phase, L)

        # diminishing step size (the constant 0.9 works well in practice)
        step_size = 0.9 / np.sqrt(it)
        a = _project_simplex(a - step_size * grad)

    return best_a, best_max


def _hill_climb(a: np.ndarray, L: int, deadline: float,
                max_moves: int = None) -> tuple:
    """Cheap random local moves (mass transfers)."""
    n = a.shape[0]
    best_a = a.copy()
    best_max = _compute_convolution(a, L).max()
    moves = 0

    while time.time() < deadline:
        i = np.random.choice(n, p=a)           # source proportional to mass
        j = np.random.randint(0, n)            # destination (any index)
        if i == j:
            continue
        # transfer up to 5 % of the source mass
        delta = np.random.rand() * 0.05 * a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue

        cur_max = _compute_convolution(a_new, L).max()
        if cur_max < best_max - 1e-14:
            best_max, best_a = cur_max, a_new.copy()
            a = a_new
        moves += 1
        if max_moves is not None and moves >= max_moves:
            break

    return best_a, best_max


def _lp_refine(a: np.ndarray, L: int, deadline: float,
               max_iters: int = 15, t_init: float = 0.5) -> tuple:
    """
    Menkes‑Ruderman refinement:
        maximise Σ g_j   s.t.   g≥0,   (a * g)_k ≤ M_a .
    The optimal g is renormalised to unit‑sum and mixed with `a`.
    """
    try:
        import cvxpy as cp
    except Exception:
        # cvxpy missing – just skip this phase
        return a, _compute_convolution(a, L).max()

    n = a.shape[0]
    best_a = a.copy()
    best_max = _compute_convolution(a, L).max()
    t = t_init

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        M = best_max
        g = cp.Variable(n)
        f_const = cp.Constant(best_a)
        conv_fg = cp.conv(f_const, g)                 # length 2n‑1
        constraints = [g >= 0, conv_fg <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        # ECOS is a fast LP solver for moderate sizes
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                   abstol=1e-9, reltol=1e-9)

        if prob.status not in ("optimal", "optimal_inaccurate"):
            break
        g_sol = np.array(g.value).flatten()
        if g_sol.ndim != 1 or g_sol.sum() <= 1e-12:
            break

        g_norm = g_sol / g_sol.sum()
        a_new = (1.0 - t) * best_a + t * g_norm
        a_new = _project_simplex(a_new)

        new_max = _compute_convolution(a_new, L).max()
        if new_max < best_max - 1e-14:
            best_max, best_a = new_max, a_new.copy()
            # keep the same mixing factor
        else:
            # if no improvement, shrink the mixing factor
            t *= 0.5
            if t < 1e-4:
                break

    return best_a, best_max


# ----------------------------------------------------------------------
# Search for a single length `n`
# ----------------------------------------------------------------------


def _search_for_length(n: int, deadline: float):
    """Run the whole pipeline for a given length `n`."""
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # ----- 0.  Parametric sweep + random draws -----------------------
    candidates = _initial_candidates(n)
    best_seq = None
    best_max = float("inf")
    for cand in candidates:
        if time.time() >= deadline:
            break
        cur_max = _compute_convolution(cand, L).max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()

    # Fallback to uniform if nothing better was found (unlikely)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _compute_convolution(best_seq, L).max()

    # ----- 1.  Adam optimisation ----------------------------------
    a, max_a = _adam_optimize(best_seq, L, phase, deadline, max_iters=25000)
    if max_a < best_max:
        best_max, best_seq = max_a, a.copy()

    # ----- 2.  Sub‑gradient descent -------------------------------
    a, max_a = _subgrad_optimize(best_seq, L, phase, deadline,
                                 max_iters=50000)
    if max_a < best_max:
        best_max, best_seq = max_a, a.copy()

    # ----- 3.  Random hill‑climbing -------------------------------
    a, max_a = _hill_climb(best_seq, L, deadline)
    if max_a < best_max:
        best_max, best_seq = max_a, a.copy()

    # ----- 4.  LP‑refinement (Menkes‑Ruderman step) ---------------
    a, max_a = _lp_refine(best_seq, L, deadline, max_iters=20, t_init=0.5)
    if max_a < best_max:
        best_max, best_seq = max_a, a.copy()

    return best_seq, best_max

def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The function respects the given time budget (seconds) and returns a
    plain Python list of floats (scaled to the normalisation required by
    `evaluate_sequence`).

    The routine explores several sequence lengths (including the length of
    the previously supplied `height_sequence_1`), runs a broad family sweep,
    Adam + sub‑gradient descent, random local moves and finally a short
    linear‑programming refinement.  The best sequence over all lengths is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin for final scaling

    # --------------------------------------------------------------
    # Choose a set of lengths to explore
    # --------------------------------------------------------------
    lengths = []

    # include the length of the previously discovered construction (if any)
    try:
        n0 = len(height_sequence_1)                # may be defined globally
        if n0 > 0:
            lengths.append(n0)
            # also try a slightly smaller / larger neighbour
            if n0 > 2000:
                lengths.append(max(2000, n0 - 500))
                lengths.append(n0 + 500)
    except Exception:
        pass

    # a few hand‑picked lengths (ensuring enough room for the FFT)
    lengths.extend([2000, 3000, 4000, 6000, 8000, 10000])

    # keep them unique and sorted
    lengths = sorted(set([n for n in lengths if n > 0]))

    # --------------------------------------------------------------
    # Global best trackers
    # --------------------------------------------------------------
    best_overall_seq = None
    best_overall_score = float("inf")
    best_overall_n = None

    # --------------------------------------------------------------
    # Loop over candidate lengths
    # --------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Exploring length n = {n} ===")
        seq, max_conv = _search_for_length(n, deadline)
        score = 2.0 * n * max_conv               # equals evaluate_sequence after scaling
        print(f"  Length {n}: best max‑conv = {max_conv:.8e} → score = {score:.6f}")
        if score < best_overall_score:
            best_overall_score = score
            best_overall_seq = seq.copy()
            best_overall_n = n

    # --------------------------------------------------------------
    # Final scaling to the normalisation used by evaluate_sequence
    # --------------------------------------------------------------
    if best_overall_seq is None:
        # extremely unlikely – fallback to uniform of size 2000
        best_overall_n = 2000
        best_overall_seq = np.full(best_overall_n, 1.0 / best_overall_n)
        best_overall_score = 2.0 * best_overall_n * (1.0 / best_overall_n)  # = 2

    scaling = math.sqrt(2.0 * best_overall_n)
    final_seq = (np.array(best_overall_seq) * scaling).tolist()

    # Log the final score using the provided evaluator (for confirmation)
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"\n--- Finished: best score = {final_score:.6f} (n = {best_overall_n}) ---")
    except Exception:
        # evaluator may not be available in the local test environment
        print(f"\n--- Finished (no evaluator): best theoretical score = {best_overall_score:.6f} (n = {best_overall_n}) ---")

    return final_seq
```
