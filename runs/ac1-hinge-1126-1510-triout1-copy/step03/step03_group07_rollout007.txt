"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation builds on the previous state‑of‑the‑art routine
but adds a few extra tricks:

* a wider parametric sweep (power‑law, exponential, double‑exponential,
  Beta‑densities and the classic arcsine shape);
* a longer Adam phase with a bigger final smoothing parameter (α≈400);
* a more aggressive sub‑gradient phase;
* a simulated‑annealing hill‑climber that can accept occasional
  uphill moves (this often escapes shallow plateaus);
* time‑aware loops that automatically stop before the 1000 s budget.

All vectors are kept on the probability simplex (∑a = 1) during the
search; the final scaling to √(2 n) is done only at return time.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    # convolution with reversed a: multiply by conjugated FFT and the phase term
    fft_rev = np.conj(fft_a) * phase
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_rev))
    # extract the n entries that correspond to the linear convolution gradient
    return 2.0 * corr[n - 1: n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # fallback – should never happen
        w[conv.argmax()] = 1.0
    return w


def _max_conv(a: np.ndarray, L: int) -> float:
    """Convenient wrapper that returns the maximum entry of a*a."""
    return _compute_convolution(a, L).max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    The function respects the given time budget (seconds) and returns a
    plain Python list of floats (already scaled to the normalisation
    required by `evaluate_sequence`).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for final scaling

    # ------------------------------------------------------------------
    # 1. Determine the length and initialise from the provided global if possible
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # may be defined externally
        n = init.shape[0]
        init = init / init.sum()
        best_seq = init.copy()
        best_max = _max_conv(best_seq, 1 << ((2 * n - 1).bit_length()))
        print(f"[init] Loaded global sequence (n={n}), max={best_max:.8e}")
    except Exception:
        n = 8000
        best_seq = None
        best_max = float("inf")
        print(f"[init] No global sequence – using length n={n}")

    # ------------------------------------------------------------------
    # FFT helpers (fixed throughout the run)
    L = 1 << ((2 * n - 1).bit_length())       # smallest power of two ≥ 2n‑1
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # ------------------------------------------------------------------
    # 2. Broad parametric sweep (cheap families) – keep the best few candidates
    print("[sweep] Starting parametric sweep")
    xs = np.arange(n, dtype=np.float64)

    # store several top candidates (up to 5) to allow restarts later
    top_candidates = []   # list of (max_val, ndarray)

    def _maybe_update(cand):
        nonlocal best_max, best_seq, top_candidates
        cand = cand / cand.sum()
        cur_max = _max_conv(cand, L)
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"  ** new global best  max={cur_max:.8e}  score={2*n*cur_max:.6f}")
        # keep a short list of the best few
        if len(top_candidates) < 5:
            top_candidates.append((cur_max, cand.copy()))
            top_candidates.sort(key=lambda x: x[0])
        else:
            if cur_max < top_candidates[-1][0]:
                top_candidates[-1] = (cur_max, cand.copy())
                top_candidates.sort(key=lambda x: x[0])

    # power‑law family
    for p in np.linspace(0.0, 5.0, 61):
        if time.time() > deadline: break
        base = (xs + 1) ** p
        for cand in (base, base[::-1]):
            _maybe_update(cand)

    # exponential family
    for lam in np.linspace(0.0, 8.0, 81):
        if time.time() > deadline: break
        exp_shape = np.exp(-lam * xs)
        for cand in (exp_shape, exp_shape[::-1]):
            _maybe_update(cand)

    # double‑exponential (U‑shaped) family
    for lam in np.linspace(0.0, 8.0, 41):
        if time.time() > deadline: break
        left = np.exp(-lam * xs)
        u_shape = left + left[::-1]
        _maybe_update(u_shape)

    # Beta‑density family (discrete Beta(a,b) shape)
    alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    betas  = [0.2, 0.5, 1.0, 2.0, 5.0]
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            if time.time() > deadline: break
            beta_shape = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
            if np.any(beta_shape <= 0):
                continue
            for cand in (beta_shape, beta_shape[::-1]):
                _maybe_update(cand)

    # arcsine‑like shape (the classic “U‑profile”)
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    for cand in (arcsine, arcsine[::-1]):
        _maybe_update(cand)

    # fall back to uniform if nothing was found (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _max_conv(best_seq, L)
        print("[fallback] Using uniform vector")

    # ------------------------------------------------------------------
    # 3. Local optimisation phases (Adam → sub‑gradient → simulated annealing)
    # we will run the sequence of phases on each of the top candidates,
    # always keeping the overall best.

    def _adam_phase(a_start, max_iters, alpha_start, alpha_end, lr):
        """Projected Adam on the smooth‑max objective."""
        a = a_start.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break
            conv = _compute_convolution(a, L)
            cur_max = conv.max()
            # schedule smoothing
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
            w = _softmax_weights(conv, alpha)
            grad = _gradient_from_weights(a, w, phase, L)
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a_new = _project_simplex(a_new)
            # evaluate new point
            new_max = _max_conv(a_new, L)
            if new_max < best_max:
                nonlocal best_max, best_seq
                best_max, best_seq = new_max, a_new.copy()
                print(f"[adam] it {it:5d} improved max={new_max:.8e}  score={2*n*new_max:.6f}")
            a = a_new
        return a

    def _subgrad_phase(a_start, max_iters):
        """Sub‑gradient descent on the true max."""
        a = a_start.copy()
        for it in range(1, max_iters + 1):
            if time.time() > deadline:
                break
            conv = _compute_convolution(a, L)
            cur_max = conv.max()
            w = _submax_weights(conv)
            grad = _gradient_from_weights(a, w, phase, L)
            step = 0.5 / np.sqrt(it)          # diminishing step size
            a_new = a - step * grad
            a_new = _project_simplex(a_new)
            new_max = _max_conv(a_new, L)
            if new_max < best_max:
                nonlocal best_max, best_seq
                best_max, best_seq = new_max, a_new.copy()
                print(f"[sub]  it {it:5d} improved max={new_max:.8e}  score={2*n*new_max:.6f}")
            a = a_new
        return a

    def _anneal_phase(a_start):
        """Simulated‑annealing hill climbing (accepts occasional uphill moves)."""
        a = a_start.copy()
        cur_max = _max_conv(a, L)

        # start temperature as a modest fraction of current max
        temp = cur_max * 0.2
        cooling = 0.99995         # exponential cooling factor

        # rough schedule for maximal step size (fraction of donor mass)
        max_frac = 0.12

        while time.time() < deadline:
            # pick donor proportional to mass, target uniformly
            i = np.random.choice(n, p=a)
            j = np.random.randint(0, n)
            if i == j:
                continue
            # transfer up to `max_frac` of donor mass
            delta = np.random.rand() * max_frac * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # numerical safety – never let an entry become negative
            if a_new[i] < 0:
                continue
            new_max = _max_conv(a_new, L)

            accept = False
            if new_max < cur_max:
                accept = True
            else:
                # metropolis criterion
                prob = math.exp(-(new_max - cur_max) / max(temp, 1e-15))
                if np.random.rand() < prob:
                    accept = True

            if accept:
                a = a_new
                cur_max = new_max
                # keep the globally best sequence
                if new_max < best_max:
                    best_max, best_seq = new_max, a.copy()
                    print(f"[anneal] improved max={new_max:.8e}  score={2*n*new_max:.6f}")

            # cool down slowly
            temp *= cooling
            # occasionally shrink the maximal transfer fraction
            if temp < best_max * 1e-4:
                max_frac = max(0.02, max_frac * 0.95)

    # ------------------------------------------------------------------
    # Run the optimisation on the best candidates collected during the sweep.
    # Each candidate receives a short Adam→sub‑grad→anneal pass.
    # The overall best (global) sequence is remembered in `best_seq`.
    for idx, (_, cand) in enumerate(top_candidates):
        if time.time() > deadline:
            break
        print(f"\n[restart] Optimising candidate #{idx+1}/{len(top_candidates)}")
        # Adam phase – a few ten‑thousands of iterations (time‑aware)
        cand = _adam_phase(cand,
                           max_iters=30000,
                           alpha_start=2.0,
                           alpha_end=400.0,
                           lr=0.07)
        # Sub‑gradient phase
        cand = _subgrad_phase(cand, max_iters=30000)
        # Annealing phase – consume whatever time is left for this candidate
        _anneal_phase(cand)

    # ------------------------------------------------------------------
    # Final scaling to the format expected by `evaluate_sequence`.
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()
    # optional logging of the final score (does not affect the competition)
    try:
        final_score = evaluate_sequence(final_seq)   # type: ignore
        print(f"\n[finished] best score = {final_score:.6f} (n={n})")
    except Exception:
        pass
    return final_seq
