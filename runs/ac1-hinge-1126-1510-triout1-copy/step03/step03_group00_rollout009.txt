"""
Search for a non‑negative step sequence that minimises the
evaluation score defined by `evaluate_sequence`.  The algorithm runs
in three timed phases:

1. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  Several diverse random
   starts (uniform, exponential, power‑law and an arcsine‑shaped
   guess) are used; after every smoothing level a short hill‑climb
   cleans up the result.

2. **Projected sub‑gradient descent** – directly minimise the true
   non‑smooth objective `2·n·max(a*a)/(sum(a)**2)` while staying on
   the probability simplex.

3. **Simulated annealing / hill‑climb** – random mass transfers
   between coordinates are accepted with a decreasing temperature,
   followed by a final greedy hill‑climb.

All phases use FFT‑based convolutions for `O(n log n)` speed and
respect the constraints `a_i ≥ 0, sum(a) = 1`.  The best sequence
found before the time budget (default 1000 s) is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a distribution already
    normalised to sum = 1:
        score = 2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.  Returns
        f = 2·n·smooth_max(conv)
    and the gradient w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                     # stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                                 # d smooth_max / d conv_k
    # derivative of conv w.r.t. a (using reversal trick)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # relevant slice corresponds to the original indices 0 … n‑1
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (non‑smooth max)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent with a
    simple back‑tracking line search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size (diminishing)
        step = 0.5 / np.sqrt(it + 1.0)

        improved = False
        for _ in range(5):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)  # tiny move to escape plateaus

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS with a decreasing λ schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in `lam_schedule`, starting each stage from
    the result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)                     # start in soft‑max space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            # loss of time on occasional numerical failures – ignore
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()

        # a short hill‑climb after each λ can squeeze out a few percent
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline, max_iter=800)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02                     # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T = max(T * 0.92, 1e-5)   # geometric cooling

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    max_n = 4000                     # fits comfortably inside the time budget
    n = max_n

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    # ------------------------------------------------------------------
    # Initial distribution (simple uniform)
    # ------------------------------------------------------------------
    a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        # ---- generate a diverse start distribution --------------------
        mode = rng.choice(['uniform', 'exp', 'power', 'arcsine', 'cosine', 'random'])
        if mode == 'uniform':
            a_tmp = np.full(n, 1.0 / n)
        elif mode == 'random':
            a_tmp = rng.random(n)
        elif mode == 'exp':
            rate = rng.uniform(0.0004, 0.015)
            a_tmp = np.exp(-rate * np.arange(n))
        elif mode == 'power':
            p = rng.uniform(0.2, 2.2)
            a_tmp = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == 'arcsine':
            # a_i ∝ 1 / sqrt( (i+1)*(n-i) )
            idx = np.arange(n, dtype=float)
            a_tmp = 1.0 / np.sqrt((idx + 1) * (n - idx))
        else:   # cosine shaped, non‑negative, peaked in the centre
            theta = np.pi * (np.arange(n) - (n - 1) / 2) / n
            a_tmp = np.maximum(np.cos(theta), 0.0)
        a_tmp /= a_tmp.sum()                     # project onto simplex

        # ---- run continuation L‑BFGS ---------------------------------
        a_opt, score_opt = lbfgs_continuation(a_tmp,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with simple line‑search
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.3)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing + final hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase3_end)
        print(f"[anneal] score={best_score:.6f}")

    # final greedy polishing (as much time as left)
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # the external runner will have `evaluate_sequence` available
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
