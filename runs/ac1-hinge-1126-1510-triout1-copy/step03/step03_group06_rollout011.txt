def run_adam_opt(a0, deadline):
    a = a0.copy()
    n = len(a)
    L = 1 << ((3*n - 2).bit_length())
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    lr0 = 0.1
    T = 0.2
    T_decay = 0.9995
    min_T = 1e-5
    t = 0
    best_score = np.inf
    best_seq = a.copy()
    while time.time() < deadline:
        t += 1
        # compute A
        A = np.fft.rfft(a, L)
        # conv and max
        conv = np.fft.irfft(A * A, L)[:2*n-1]
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # compute softmax weights
        # Note: Use T for smoothing; ensure T > 0
        exp_arg = (conv - max_b) / T
        # To avoid overflow, but exp_arg <= 0
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum
        # Compute gradient via correlation
        W = np.fft.rfft(w, L)
        corr_full = np.fft.irfft(A * np.conj(W), L)[: n + (2*n-1) - 1]
        shift = (2*n-1) - 1
        grad = 2.0 * corr_full[shift:shift + n]
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr0 / (np.sqrt(v_hat) + epsilon)
        # optional step-size decay with sqrt(it)
        a = a - step
        # projection onto simplex
        a = project_simplex(a)
        # update temperature and lr if desired
        T = max(min_T, T * T_decay)
        # optionally lr decay
        lr0 *= 0.9999
        # early break if score very low
        if best_score < 1.45:
            break
        # Check iteration count limit
        if t >= 50000: break
    return best_seq, best_score

"""
Hybrid Adam‑based smooth‑max minimisation on the probability simplex.

The objective  2·n·max(conv(a,a)) / (sum(a)^2)  is homogeneous, so the
search is carried out on the simplex Σ a_i = 1 , a_i ≥ 0.
A temperature‑controlled log‑sum‑exp (smooth max) is minimised by a
projected Adam optimiser.  The gradient of the smooth max is a
cross‑correlation between the current coefficient vector and the
soft‑max weights of its self‑convolution – this can be evaluated
efficiently with a single FFT of length ≈ 3 n.  Several random
restarts and a cheap stochastic refinement are performed until the
time budget expires.  The best sequence found (as a plain Python list)
is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimiser – Adam on the smooth max of the self‑convolution
# ----------------------------------------------------------------------
def adam_smooth_max(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 0.09,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    T0: float = 0.30,
    T_decay: float = 0.9996,
    min_T: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    minimise 2·n·max(conv(a,a)) on the simplex using a smooth‑max loss.

    Returns the best sequence (as a NumPy array) and its corresponding
    evaluation score (i.e. 2·n·max(conv)).
    """
    a = a0.copy()
    n = a.shape[0]

    # length for FFT that can contain a full correlation of length 3·n‑2
    fft_len = 1 << ((3 * n - 2).bit_length())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature for the smooth‑max
    T = T0

    best_seq = a.copy()
    # initialise best score (exact max, not the smooth surrogate)
    conv = np.fft.irfft(
        np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len
    )[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()

    t_iter = 0
    no_improve = 0
    while time.time() < deadline and t_iter < max_iters:
        t_iter += 1

        # ---------- FFT of the current vector ----------
        A = np.fft.rfft(a, fft_len)

        # ---------- self‑convolution & exact max ----------
        conv_full = np.fft.irfft(A * A, fft_len)[: 2 * n - 1]
        max_b = conv_full.max()
        score = 2.0 * n * max_b

        # store best observed point
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            no_improve = 0
            print(f"[adam it {t_iter}] new best score={score:.6f}")
        else:
            no_improve += 1

        # ---------- smooth‑max weights (softmax over conv) ----------
        # avoid overflow/underflow: shift by the maximum value
        shifted = (conv_full - max_b) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # ---------- gradient of the smooth‑max ----------
        # grad_i = 2 * Σ_j w_{i+j} * a_j   (cross‑correlation)
        W = np.fft.rfft(w, fft_len)
        corr_full = np.fft.irfft(A * np.conj(W), fft_len)[: n + (2 * n - 1) - 1]
        shift = (2 * n - 1) - 1                     # = 2·n‑2
        grad = 2.0 * corr_full[shift : shift + n]   # shape (n,)

        # ---------- Adam update (projected) ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # keep the vector on the simplex
        a = project_simplex(a)

        # ---------- temperature & learning‑rate decay ----------
        T = max(min_T, T * T_decay)
        lr0 *= 0.9999

        # stop early if we have stalled
        if no_improve >= 3000:
            break

        # occasional safety break
        if t_iter % 5000 == 0 and time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic refinement (moves a tiny amount of mass between two
# coordinates, accepting improvements or occasional uphill moves).
# ----------------------------------------------------------------------
def stochastic_refine(a: np.ndarray, steps: int, deadline: float) -> np.ndarray:
    """
    Run a cheap random walk that moves a small fraction of mass between
    two random coordinates, accepting improvements (or Metropolis moves).
    """
    n = a.shape[0]
    best = a.copy()
    best_score = 2.0 * n * np.max(np.convolve(best, best))

    temperature = 0.02
    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # pick direction (source→dest)
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        delta = np.random.uniform(0.0, 0.05) * best[src]
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[src] -= delta
        cand[dst] += delta
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum == 0.0:
            continue
        cand /= cand_sum

        cand_score = 2.0 * n * np.max(np.convolve(cand, cand))
        if cand_score < best_score or np.random.rand() < np.exp((best_score - cand_score) / temperature):
            best = cand
            if cand_score < best_score:
                best_score = cand_score

        temperature = max(temperature * 0.9999, 1e-6)

    return best

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence with a low evaluation score.
    Several random restarts are performed; each restart runs the Adam
    optimiser on a smooth approximation of the max‑convolution, followed
    by a brief stochastic refinement.  The best sequence ever observed is
    returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose a few candidate lengths (the problem size).  4000 works
    # well with the available time budget; we also try a smaller
    # and a larger size to diversify the search.
    # --------------------------------------------------------------
    length_candidates = [2000, 4000, 8000]

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Main loop over lengths – keep restarting until we run out of time.
    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.time() >= deadline:
            break

        # allocate a modest amount of time to each length (fair share)
        # but allow early exit if the global optimum is already very good.
        while time.time() < deadline:
            # ---- initialise a fresh starting point -----------------
            r = np.random.rand()
            if r < 0.2:
                # uniform start
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
            elif r < 0.7:
                # power‑law start, exponent drawn uniformly
                p = np.random.uniform(0.45, 0.80)
                idx = np.arange(1, n + 1, dtype=np.float64)
                a0 = idx ** (-p)
                a0 /= a0.sum()
            else:
                # perturb the current global best (if same size)
                if global_best_seq is not None and len(global_best_seq) == n:
                    a0 = np.array(global_best_seq, copy=True)
                    a0 += 0.02 * np.random.randn(n)
                    a0 = np.maximum(a0, 0.0)
                    s = a0.sum()
                    a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
                else:
                    # fallback power‑law
                    p = np.random.uniform(0.45, 0.70)
                    idx = np.arange(1, n + 1, dtype=np.float64)
                    a0 = idx ** (-p)
                    a0 /= a0.sum()

            # ---------------------------------------------------------
            # Run the Adam optimiser on the smooth max.
            # ---------------------------------------------------------
            seq_opt, score_opt = adam_smooth_max(a0, deadline)

            # ---------------------------------------------------------
            # Short stochastic refinement (if time allows)
            # ---------------------------------------------------------
            time_left = deadline - time.time()
            if time_left > 1.0:
                steps = int(min(8000, time_left / 0.0007))
                seq_ref = stochastic_refine(seq_opt, steps, deadline)
                # compute exact score after refinement
                conv = np.convolve(seq_ref, seq_ref)
                score_ref = 2.0 * n * conv.max()
                if score_ref < score_opt:
                    seq_opt, score_opt = seq_ref, score_ref

            # ---------------------------------------------------------
            # Update the global best
            # ---------------------------------------------------------
            if score_opt < global_best_score:
                global_best_score = score_opt
                global_best_seq = seq_opt.copy()
                print(f"[len={n}] new global best score={score_opt:.6f}")

                # target reached – stop early
                if global_best_score < 1.503:
                    deadline = time.time()   # force termination of outer loops
                    break

            # optional early termination for this length if no progress
            # is observed for a while
            if time.time() >= deadline:
                break

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_seq = a

    # Return as a plain Python list of floats
    return [float(x) for x in global_best_seq.tolist()]
