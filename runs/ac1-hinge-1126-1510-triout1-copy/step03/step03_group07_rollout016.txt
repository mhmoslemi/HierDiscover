"""
Propose a non‑negative step function that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2),

i.e. the maximum entry of the self‑convolution of a probability vector.
The algorithm consists of four stages:

1.  A quick parametric sweep (power‑law, exponential, Beta, arcsine …)
    to obtain a decent starting point.
2.  Continuation with L‑BFGS on a smoothed objective
    `log‑sum‑exp(α·conv)/α`.  The smoothing parameter α is increased
    progressively (4 → 800).  The variables are the logits of the
    simplex (a = softmax(u)), so positivity and the sum‑to‑one
    constraint are built‑in.
3.  Sub‑gradient descent on the true max (α→∞) using the active set
    of lags where the convolution is maximal.
4.  Aggressive random hill‑climbing (mass transfers up to 10 % of the
    source entry) to polish any remaining tiny peaks.

All heavy work (convolution, gradients) is performed with FFTs
(O(L log L) per evaluation, L≈2 n).  The routine respects the
provided time budget (default 1000 s) and returns the best sequence
found, scaled to the normalisation expected by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Global variables that are filled inside ``propose_candidate``.
GLOBAL_N = None          # length of the sequence
GLOBAL_L = None          # FFT length (power of two ≥ 2·n‑1)
GLOBAL_PHASE = None      # e^{2πi·k/L} used for reversed FFT
CURRENT_ALPHA = None     # smoothing parameter for the L‑BFGS stage
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a * a (real part) truncated to length 2·len(a)‑1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, w: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of   Σ_k w_k·(a*a)_k   w.r.t. a.
    Implements   2·correlation(w, a)   using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase      # FFT of reversed a
    fft_w = np.fft.fft(w, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    # centre the correlation so that it aligns with the original indices
    return 2.0 * corr[n - 1: n - 1 + n]


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set (indices attaining the max)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # safety fallback
        w[conv.argmax()] = 1.0
    return w


# ----------------------------------------------------------------------
def _smooth_obj_and_grad(u: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Evaluate the smoothed objective `logsumexp(alpha*conv)/alpha`
    and its gradient w.r.t. the logits u (where a = softmax(u)).
    This function is used by ``scipy.optimize.minimize`` with ``jac=True``.
    """
    global GLOBAL_N, GLOBAL_L, GLOBAL_PHASE, CURRENT_ALPHA

    # a lives on the simplex by construction
    a = np.exp(u)
    a_sum = a.sum()
    if a_sum == 0.0:          # avoid division by zero
        a = np.full_like(a, 1.0 / GLOBAL_N)
    else:
        a = a / a_sum

    # FFT of a and its self‑convolution
    fft_a = np.fft.fft(a, n=GLOBAL_L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * GLOBAL_N - 1]

    # ----- smoothed max -----
    scaled = CURRENT_ALPHA * conv
    # log‑sum‑exp in a numerically stable way
    lse = logsumexp(scaled)
    s = lse / CURRENT_ALPHA                     # smoothed max value

    # soft‑max weights (derivative of the smooth max)
    max_scaled = scaled.max()
    exp_shifted = np.exp(scaled - max_scaled)
    w = exp_shifted / exp_shifted.sum()

    # gradient w.r.t. a (2·correlation)
    fft_a_rev = np.conj(fft_a) * GLOBAL_PHASE
    fft_w = np.fft.fft(w, n=GLOBAL_L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    corr = corr[GLOBAL_N - 1: GLOBAL_N - 1 + GLOBAL_N]
    grad_a = 2.0 * corr                         # ∂s/∂a

    # chain rule to logits u
    dot = np.dot(grad_a, a)                     # Σ_i a_i ∂s/∂a_i
    grad_u = a * (grad_a - dot)

    # The optimizer works on the smoothed max alone (the factor 2·n is
    # irrelevant for the location of the optimum).
    return s, grad_u

    The function respects the given time budget (seconds) and returns a
    plain Python list of floats (scaled to √(2 n) as required by
    ``evaluate_sequence``).
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin for final post‑proc

    # ------------------------------------------------------------------
    # Determine the length n and an initial candidate
    # ------------------------------------------------------------------
    global GLOBAL_N, GLOBAL_L, GLOBAL_PHASE, CURRENT_ALPHA

    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # may be defined globally
        n = len(init_seq)
        init_seq = init_seq / init_seq.sum()          # normalise to probability simplex
    except Exception:
        n = 8000                                        # fallback length
        init_seq = None

    GLOBAL_N = n
    GLOBAL_L = 1 << ((2 * n - 1).bit_length())         # next power of two ≥ 2n‑1
    GLOBAL_PHASE = np.exp(2j * np.pi * np.arange(GLOBAL_L) / GLOBAL_L)

    # ------------------------------------------------------------------
    # Phase 0 – a quick parametric sweep to obtain a reasonable start
    # ------------------------------------------------------------------
    best_seq = init_seq if init_seq is not None else np.full(n, 1.0 / n)
    best_max = _compute_convolution(best_seq, GLOBAL_L).max()
    print(f"Initial candidate (n={n})  max={best_max:.8e}  score={2*n*best_max:.6f}")

    xs = np.arange(n, dtype=np.float64)

    # --- power‑law ----------------------------------------------------
    for p in np.linspace(0.0, 2.0, 41):
        if time.time() > deadline:
            break
        base = (xs + 1) ** p
        for cand in (base, base[::-1]):
            cand = cand / cand.sum()
            cur_max = _compute_convolution(cand, GLOBAL_L).max()
            if cur_max < best_max:
                best_max, best_seq = cur_max, cand.copy()
                print(f"  new best (power p={p:.2f})  max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # --- exponential --------------------------------------------------
    for lam in np.linspace(0.0, 5.0, 41):
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * xs)
        for cand in (exp_shape, exp_shape[::-1]):
            cand = cand / cand.sum()
            cur_max = _compute_convolution(cand, GLOBAL_L).max()
            if cur_max < best_max:
                best_max, best_seq = cur_max, cand.copy()
                print(f"  new best (exp λ={lam:.2f})   max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # --- double‑exponential (U‑shape) ---------------------------------
    for lam in np.linspace(0.0, 5.0, 21):
        if time.time() > deadline:
            break
        left = np.exp(-lam * xs)
        u_shape = left + left[::-1]
        cand = u_shape / u_shape.sum()
        cur_max = _compute_convolution(cand, GLOBAL_L).max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"  new best (U‑shape λ={lam:.2f}) max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # --- Beta densities ------------------------------------------------
    alphas = [0.5, 1.0, 2.0, 5.0]
    betas = [0.5, 1.0, 2.0, 5.0]
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:
            if time.time() > deadline:
                break
            beta_shape = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
            if np.all(beta_shape > 0):
                for cand in (beta_shape, beta_shape[::-1]):
                    cand = cand / cand.sum()
                    cur_max = _compute_convolution(cand, GLOBAL_L).max()
                    if cur_max < best_max:
                        best_max, best_seq = cur_max, cand.copy()
                        print(f"  new best (Beta a={a} b={b}) max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # --- arcsine‑like ---------------------------------------------------
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    for cand in (arcsine, arcsine[::-1]):
        cand = cand / cand.sum()
        cur_max = _compute_convolution(cand, GLOBAL_L).max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, cand.copy()
            print(f"  new best (arcsine)        max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smoothed max (continuation in α)
    # ------------------------------------------------------------------
    print("--- Phase 1: L‑BFGS on smoothed max ---")
    a = best_seq.copy()
    u = np.log(a + 1e-20)                # logits, guarantees positivity

    # progression of the smoothing parameter
    alpha_schedule = [4.0, 10.0, 30.0, 100.0, 300.0, 800.0]

    for alpha in alpha_schedule:
        if time.time() > deadline:
            break
        CURRENT_ALPHA = alpha
        # L‑BFGS – we limit the inner optimizer to a modest number of iterations
        res = minimize(_smooth_obj_and_grad,
                       u,
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 200, 'disp': False})
        u = res.x
        a = np.exp(u)
        a = a / a.sum()
        conv = _compute_convolution(a, GLOBAL_L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max, best_seq = cur_max, a.copy()
            print(f"  L‑BFGS α={alpha:4g} improved max={cur_max:.8e}  score={2*n*cur_max:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient descent on the true max
    # ------------------------------------------------------------------
    print("--- Phase 2: sub‑gradient descent on true max ---")
    a = best_seq.copy()
    sub_max_iters = 60000
    sub_it = 0
    while time.time() < deadline and sub_it < sub_max_iters:
        sub_it += 1
        conv = _compute_convolution(a, GLOBAL_L)
        cur_max = conv.max()
        w = _submax_weights(conv)            # uniform over active indices
        grad = _gradient_from_weights(a, w, GLOBAL_PHASE, GLOBAL_L)
        step = 0.6 / np.sqrt(sub_it)          # diminishing step size
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        conv_new = _compute_convolution(a_new, GLOBAL_L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            print(f"  Sub‑grad it {sub_it:5d} improved max={max_new:.8e}  score={2*n*max_new:.6f}")
        a = a_new

    # ------------------------------------------------------------------
    # Phase 3 – cheap random hill‑climbing (larger moves)
    # ------------------------------------------------------------------
    print("--- Phase 3: random hill‑climbing ---")
    a_current = best_seq.copy()
    while time.time() < deadline:
        # pick a source index proportional to its mass (large entries are sampled more often)
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # move up to 10 % of the source mass
        delta = np.random.rand() * 0.10 * a_current[i]
        if delta <= 0.0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0.0:               # numerical safety
            continue
        conv_new = _compute_convolution(a_new, GLOBAL_L)
        max_new = conv_new.max()
        if max_new < best_max:
            best_max, best_seq = max_new, a_new.copy()
            a_current = a_new
            print(f"  move improved max={max_new:.8e}  score={2*n*max_new:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the normalisation expected by ``evaluate_sequence``
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_seq * target_sum).tolist()

    # Logging of the final score (does not affect the returned value)
    final_score = evaluate_sequence(final_seq)
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) "
          f"after {sub_it} sub‑grad iters ---")
    return final_seq
