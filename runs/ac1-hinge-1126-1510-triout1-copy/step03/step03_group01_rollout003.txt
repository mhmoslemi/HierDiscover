def simplex_projection(v):
    # v: np.ndarray
    # returns w
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

        # multiply by factor 2*n for objective? Not needed for step scaling; but include 2*n maybe for magnitude
        grad_a = 2.0 * n * grad_a  # now gradient of 2*n*max(wrt a)
        # gradient w.r.t y
        mean_g = np.dot(a, grad_a)
        dy = a * (grad_a - mean_g)
        # Adam update
        m = beta1 * m + (1 - beta1) * dy
        v = beta2 * v + (1 - beta2) * (dy ** 2)

        v_hat = v / (1 - beta2 ** t)
        # schedule learning rate decreasing over time
        lr_t = lr / np.sqrt(t)  # simple schedule
        y -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
    return y

The goal is to find a non‑negative step sequence `a = (a₀,…,a_{n‑1})` that minimizes  

\[
\text{score}(a)=\frac{2n\; \max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear convolution of `a` with itself.
Because the score is invariant under a global scaling of `a`, we may restrict the
search to the probability simplex `\sum a = 1`.  
The problem is non‑convex (the maximum of a quadratic form) and a direct
linear‑programming formulation is impossible.

The new search combines three complementary ideas:

1. **Exact sub‑gradient of the max‑convolution** –  
   For a current distribution `a` let `s* = argmax (a*a)`.  
   The gradient of `max(a*a)` w.r.t. `a_i` is `2·a_{s*-i}` (zero when the index is
   out of range).  This provides a cheap, exact sub‑gradient that we can use in
   a first‑order method.

2. **Adam on the soft‑max parametrisation** –  
   We keep the simplex constraint by writing `a = softmax(y)`.  
   Using the exact sub‑gradient we compute the gradient w.r.t. `y` and apply the
   Adam optimiser.  Adam’s adaptive step‑size and momentum make it much more
   robust than plain projected gradient descent.

3. **Hybrid global search** –  
   *Random restarts* (uniform, exponential, power‑law, edge‑heavy, Sidon‑like)
   give diverse starting points.  Each restart runs the Adam optimiser for a
   few thousand iterations (still fast thanks to FFT‑based convolutions).  
   The best candidate is refined afterwards by  
   * a continuation L‑BFGS optimiser (smooth‑max objective, as in the
     reference code) and  
   * a greedy hill‑climber that moves a tiny fraction of mass between two
     indices if it strictly improves the true (unsmoothed) score.

The whole procedure is time‑budget aware: it stops well before the supplied
`budget_s` (default 1000 s), falls back to the best solution found so far, and
finally evaluates the returned sequence with the supplied `evaluate_sequence`
function.

The code below follows the required API – it defines a top‑level
`propose_candidate` function and only uses standard scientific Python libraries.

"""
Hybrid Adam / L‑BFGS optimiser for the self‑convolution max problem.

*  a is forced onto the probability simplex via a soft‑max parametrisation.
*  Exact sub‑gradient of max(a*a) (derived analytically) drives an Adam optimiser.
*  Several random restarts explore the landscape.
*  The best solution is polished with a continuation L‑BFGS (smooth‑max)
   and a final greedy hill‑climb.
*  All heavy work (convolutions) uses FFTs, so the routine is fast enough
   to run many thousands of iterations within the 1000 s time budget.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the first (2·n‑1) real coefficients.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        score = 2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a start vector y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # put a modest extra weight on the first/last 2 % of the indices
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon‑like set (pairwise sums distinct) – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (cand * 2) not in sums:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode!r}")
    return y0


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    This is the exact same routine used by the reference code – it enables a
    continuation L‑BFGS refinement after the Adam phase.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                           # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a small random fraction of mass from i → j.
    Keep a move only if the true (unsmoothed) fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05   # ≤ 5 % of mass at i
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len: int,
    max_iter: int,
    deadline: float,
) -> np.ndarray:
    """
    Adam optimiser on the soft‑max parametrisation using the *exact* sub‑gradient
    of max(a*a).  No smoothing is employed – we directly minimise the true
    objective 2·n·max(a*a).
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.12   # empirically stable for our scale

    for t in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        a = softmax(y)                     # probability vector (sum = 1)

        # convolution a*a
        conv = conv_fft(a, fft_len, n)
        s_max = int(conv.argmax())

        # exact gradient of max(conv) w.r.t. a
        grad_a = np.zeros_like(a)
        i_start = max(0, s_max - (n - 1))
        i_end = min(n - 1, s_max)
        idx = np.arange(i_start, i_end + 1)
        j = s_max - idx
        # derivative of max(a*a) is 2·a_j (see analysis)
        grad_a[idx] = 2.0 * a[j]

        # incorporate the outer factor 2·n from the objective
        grad_a *= 2.0 * n

        # gradient w.r.t. the unconstrained y (soft‑max chain rule)
        mean_g = np.dot(a, grad_a)
        dy = a * (grad_a - mean_g)

        # Adam update with a simple decay of the learning rate
        lr_t = base_lr / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * dy
        v = beta2 * v + (1.0 - beta2) * (dy ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y -= lr_t * m_hat / (np.sqrt(v_hat) + eps)

    return y

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    The routine:
      * picks a problem size (default n = 5000, possibly enlarged by a
        previously discovered construction),
      * performs many Adam‑based random restarts,
      * refines the best candidate with a continuation L‑BFGS schedule,
      * finishes with a short greedy hill‑climb,
      * returns the final list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # keep a small safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 5000
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            arr = np.asarray(init_seq, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                n = max(n, int(arr.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise baseline (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # probability vector
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # hyper‑parameters for the Adam restarts
    # --------------------------------------------------------------
    max_iters_per_restart = 2000
    restart_modes = ["random", "exp", "power", "edge", "sidon"]
    iteration = 0

    # --------------------------------------------------------------
    # main random‑restart loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:   # keep a few seconds for the final phase
        mode = rng.choice(restart_modes + ["best"])
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # Adam optimisation (exact sub‑gradient)
        y_opt = adam_optimize(y0, n, fft_len_conv, max_iters_per_restart, deadline)
        a_opt = softmax(y_opt)
        cur_score = fast_score(a_opt, fft_len_conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode}  fast score = {best_score:.6f}")

        # short aggressive hill‑climb on the current best
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=4000
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final refinement with continuation L‑BFGS (smooth‑max)
    # --------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-4, num=18))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.5g}  fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final greedy polishing (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=8000
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
