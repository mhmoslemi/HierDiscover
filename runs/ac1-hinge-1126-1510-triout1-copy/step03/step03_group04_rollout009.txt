"""
Propose a low‑score step sequence.

The algorithm consists of four stages, all respecting the 1000 s budget.

1. **Continuation L‑BFGS** – minimise a smooth approximation of the
   max‑convolution using a decreasing schedule of smoothing factors.
   The variables are Lie‑group logits `y` (soft‑max) so the simplex
   constraint is handled automatically.

2. **Projected sub‑gradient refinement** – minimise the true max
   directly.  A diminishing step size together with an exact Euclidean
   projection onto the probability simplex yields a reliable descent.

3. **Penalty‑gradient descent** – enforce the bound
   `conv_k ≤ M` by a quadratic penalty on the violations.
   The penalty gradient is computed by a single FFT‑based convolution,
   followed by a simplex projection.  The bound `M` is slowly decreased
   whenever the constraints are satisfied.

4. **Simulated annealing + greedy hill‑climb** – non‑local moves
   (mass transfer between two positions) escape any remaining local
   minima; a final aggressive hill‑climb fine‑tunes the answer.

All convolutions are performed by FFT (complexity `O(n log n)`); the
code works for any `n` (default 3000) and automatically falls back to
the previously discovered sequence if `height_sequence_1` is present.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (`y` → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    Returns the real part (the convolution is real‑valued).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # keep only the needed part
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a vector that already
    satisfies Σa = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, Σx = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f   : float                 – 2·n·smooth_max(conv)
    grad: np.ndarray (shape n) – gradient w.r.t. y
    """
    # map to simplex
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    # g_vec_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation of the true max
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step is projected back onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active (binding) indices
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: 2·(mask * a_rev) restricted to n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best observed iterate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Penalty‑gradient descent on max‑convolution constraint
# ----------------------------------------------------------------------


def penalty_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Minimise the bound `max(conv) ≤ M` via a quadratic penalty on the
    violations `v = max(conv - M, 0)`.  The bound M is reduced whenever
    all violations are zero.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    # initialise the bound from the current point
    cur_conv = full_conv(a, fft_len_conv)
    M = cur_conv.max()

    it = 0
    while time.time() < deadline:
        # current convolution
        conv = full_conv(a, fft_len_conv)
        # violation vector (only positive part)
        v = np.maximum(conv - M, 0.0)
        max_violation = v.max()

        if max_violation < 1e-12:
            # all constraints satisfied – try to tighten M
            M *= 0.995
            # continue without a gradient step
            continue

        # gradient of the quadratic penalty L = Σ v_k²
        a_rev = a[::-1]
        fm = np.fft.fft(v, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        # ∂L/∂a_i = 4 * (v * a_rev)_i
        grad = 4.0 * conv_g[(n - 1) : (n - 1 + n)]

        # simple line‑search step: try a few decreasing step sizes
        step = 1.0 / (np.linalg.norm(grad) + 1e-12)
        improved = False
        for _ in range(5):
            a_candidate = simplex_projection(a - step * grad)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                a = a_candidate
                best_score = cand_score
                best_a = a.copy()
                improved = True
                break
            step *= 0.5

        if not improved:
            # if none of the tried steps improved, just take a small move
            a = simplex_projection(a - 0.01 * grad)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction (≤5 %) of the mass from one position to another.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
):
    """
    Simple simulated annealing that moves a fraction of mass between
    two positions.  Improves on the hill‑climb by occasionally
    accepting uphill moves.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9997
    it = 0

    while time.time() < deadline:
        # pick a source with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue

        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a random fraction of the source mass
        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        # Metropolis acceptance
        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                cur_a[i] += delta
                cur_a[j] -= delta

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0:
            # occasional reheating
            T = max(best_score * 0.02, 1e-8)

        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the supplied time budget and returns the best
    sequence found (as a list of floats).  Progress is printed to stdout.
    """
    # --------------------------------------------------------------
    # Random generator and initial sequence
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    # use previously discovered sequence if it exists, otherwise start uniform
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        init_seq = np.maximum(init_seq, 0.0)
        if init_seq.size == 0:
            n = 3000
            a0 = np.full(n, 1.0 / n)
        else:
            max_n = 3500
            if init_seq.size > max_n:
                init_seq = init_seq[:max_n]
            n = init_seq.shape[0]
            if init_seq.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = init_seq / init_seq.sum()
    else:
        n = 3000                     # a size large enough to improve the bound
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # FFT length preparation (powers of two)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)    # for a*a
    fft_len_g = next_pow2(3 * n - 2)       # for gradient‑related convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a safety margin

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS on a smoothed max
    # --------------------------------------------------------------
    # schedule of smoothing factors (λ = factor·max(conv) )
    smooth_factors = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    for lam in smooth_factors:
        if time.time() >= deadline:
            break

        # initialise logit vector from the currently best distribution
        y0 = np.log(best_a + 1e-12)

        res = minimize(
            fun=objective_and_grad,
            x0=y0,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 400, "ftol": 1e-12, "gtol": 1e-12, "disp": False},
        )
        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] lam={lam:.4f} score={best_score:.6f}")

        # short hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.25)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=1200
        )

    # --------------------------------------------------------------
    # Phase 2 – longer sub‑gradient refinement
    # --------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] refined score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – penalty‑gradient descent on the max‑constraint
    # --------------------------------------------------------------
    phase3_end = start_time + 0.85 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = penalty_descent(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[penalty‑descent] refined score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing for non‑local exploration
    # --------------------------------------------------------------
    phase4_end = deadline
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, phase4_end, rng
        )
        print(f"[SA] final score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final intensive greedy hill‑climb polish
    # --------------------------------------------------------------
    final_a, final_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_iter=4000,
    )

    # --------------------------------------------------------------
    # Final evaluation (using the provided evaluator) for reporting
    # --------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float("nan")
    print(f"[final] evaluated score = {eval_val:.6f}")

    return final_a.tolist()
