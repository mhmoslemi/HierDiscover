    best sequence seen (according to the fast_score) is stored.

The routine runs for the allotted time (≈1000 s) and returns the best list
of heights found.  It uses only NumPy and the provided `evaluate_sequence`
function.
"""

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv(seq,seq))/(sum(seq)²).
    The sequence is assumed to be non‑negative and normalised to sum=1,
    therefore the denominator is 1.
    """
    n = seq.shape[0]
    # FFT based convolution (full length 2n‑1)
    fft_len = next_pow2(2 * n)
    Fa = np.fft.rfft(seq, n=fft_len)
    conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv_full.max())
    return 2.0 * n * max_conv


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators (same families as the reference implementation)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Coefficients of (1‑z)^{-1/2}: a_i = a_{i‑1}·(2i‑1)/(2i)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann window (raised cosine)."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the window."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


# ----------------------------------------------------------------------
# Smoothed max and its gradient
# ----------------------------------------------------------------------


def smooth_max_and_grad(
    a: np.ndarray,
    lam: float,
    fft_len: int,
) -> tuple[float, np.ndarray, np.ndarray]:
    """
    Given a (sum=1), compute

        S = (1/lam)·log Σ_k exp(lam·conv_k)   (smooth approximation of max)
        g = ∂S/∂a  (gradient, shape same as a)

    Both are obtained with FFTs in O(n log n) time.
    Returns (S, g, conv_full) where conv_full is the full self‑convolution.
    """
    n = a.shape[0]

    # ---- self‑convolution a∗a (full length 2n‑1) ----
    Fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]

    # ---- soft‑max weights w_k = exp(lam·conv_k) / Σ exp(lam·conv) ----
    max_c = np.max(conv_full)
    exp_shift = np.exp(lam * (conv_full - max_c))
    sum_exp = np.sum(exp_shift)
    w = exp_shift / sum_exp

    # ---- gradient g = 2·(w ⋆ a)  (cross‑correlation) ----
    a_rev = a[::-1]                         # reversed a, length n
    w_fft = np.fft.rfft(w, n=fft_len)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    conv_rev_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)[: (3 * n - 2)]

    # indices (n‑1) … (n‑1 + n‑1) give the desired n components
    g = 2.0 * conv_rev_full[(n - 1) : (n - 1 + n)]

    # ---- smooth max value (for possible monitoring) ----
    smooth_max = max_c + (1.0 / lam) * np.log(sum_exp)

    return smooth_max, g, conv_full


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Find a non‑negative height sequence that minimises
    2·n·max(conv(a,a))/(sum(a)²).

    Returns the best sequence found as a plain Python list of floats.
    """
    # ----- initialisation -------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ----- start from a known construction if available ------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:

    except Exception:

        init = generate_arcsine_like(8000)

    init = np.maximum(init, 0.0)
    if init.sum() == 0.0:
        init = np.full_like(init, 1.0 / init.size)
    else:
        init = init / init.sum()

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ----- 1️⃣  Randomised family search (≈ 200 s) -----------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = [
        "power",
        "exp",
        "twosided",
        "arcsine",
        "cosine",
        "gaussian",
    ]

    # time reserved for the refinement phase
    reserve_refine = 350.0

    while time.time() < deadline - reserve_refine:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.30, 0.80)
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_cosine_window(n)

        elif family == "gaussian":
            sigma_frac = random.uniform(0.07, 0.30)
            cand = generate_gaussian_window(n, sigma_frac)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start_time:.1f}s] new random best "
                f"({family}) n={n} score={sc:.6f}"
            )

    # ----- 2️⃣  Smoothed gradient descent (the main work) -----------------
    a = best_seq.copy()
    n = a.shape[0]

    # FFT length sufficient for both convolutions (max of 2n‑1 and 3n‑2)
    fft_len = next_pow2(3 * n)

    # hyper‑parameters
    eta0 = 5.0            # initial multiplicative step size factor
    tau = 5000.0          # decay constant for η
    eval_interval = 2000  # how often we evaluate the fast_score
    jitter_interval = 5000
    jitter_scale = 5e-4   # relative amplitude of jitter
    lam_start = 3.0       # start temperature (smooth)
    lam_end = 80.0        # final temperature (sharp)

    iteration = 0
    last_eval = 0

    while time.time() < deadline - 1.0:
        iteration += 1

        # linear schedule for the temperature λ
        elapsed = time.time() - start_time
        total = deadline - start_time
        lam = lam_start + (lam_end - lam_start) * min(1.0, elapsed / total)

        # compute smooth max and its full gradient
        _, grad, conv_full = smooth_max_and_grad(a, lam, fft_len)

        # multiplicative (exponentiated) gradient step
        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * grad)

        # simplex projection (simple renormalisation)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # occasional jitter to escape plateaus
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = np.clip(a + noise, 0.0, None)
            s = a.sum()
            if s <= 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

        # periodic performance check
        if iteration % eval_interval == 0:
            cur_score = 2.0 * n * np.max(conv_full)  # fast_score when sum=1
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start_time:.1f}s] iter {iteration}, "
                    f"new best score={cur_score:.6f}"
                )

    # ----- 3️⃣  Final verification with the official evaluator -------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")

    return final_seq
