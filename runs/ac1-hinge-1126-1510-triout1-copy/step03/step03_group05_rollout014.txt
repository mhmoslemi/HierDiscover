def adam_refine(seq, time_limit, start_time):
    a = np.array(seq, dtype=np.float64).copy()
    a = project_simplex(a)
    n = len(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps = 1e-8
    t = 0
    best_seq = a.copy()
    best_score = fast_score(a)
    while time.time() < start_time + time_limit:
        t+=1
        # compute conv and M
        fft_len = next_pow2(2*n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
        M = conv.max()
        cur_score = 2.0*n*M / (a.sum()**2)  # but a.sum()=1.
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # compute subgradient for active indices
        eps0 = 1e-12
        active = np.where(conv >= M - eps0)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k-(n-1))
            i_end = min(n-1, k)
            i_range = np.arange(i_start, i_end+1)
            j_range = k - i_range
            # contribution factor is a[j] + a[i]? Actually derivative of a_i*a_{k-i} w.r.t a_i when i != j is a_j (plus symmetrical). Our earlier subgradient used g[i_range] += 2*a[j]. Should we use that? Yes.
            g[i_range] += 2.0 * a[j_range]
        # bias for endpoint? conv at 0 gives a_0^2; derivative w.r.t a_0 is 2*a_0, and we counted that because for k=0, i_range=0, j_range=0 ==> g[0]+=2*a[0]; fine.
        # Now Adam update
        lr = 0.1 / np.sqrt(t) # or constant
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)
    best_score = fast_score(best_seq)

"""
Propose a step‑function (non‑negative sequence) with a low
evaluate_sequence() bound.

Algorithm
---------
1. **Exploratory sampling** – randomly generate candidates from a
   collection of parametric families (power‑law, exponential,
   two‑sided power‑law, Beta‑pdf, Dirichlet).  Lengths ranging from
   500 to 20000 are tried.  Each candidate is normalised to sum = 1
   and scored with a cheap FFT based “fast_score”.  The best few
   candidates are kept in an elite pool.

2. **Projected Adam refinement** – each elite candidate is refined
   by a projected Adam optimiser that minimises the max‑convolution
   value.  The gradient is the sub‑gradient of the active
   convolution entries (all indices where the convolution attains its
   current maximum).  After each Adam step the iterate is projected
   back onto the probability simplex.

3. **Verification** – the best sequence found is finally evaluated
   with the provided `evaluate_sequence` function.

The whole search is limited to the supplied time budget (≈ 1000 s).
Printing statements report progress and intermediate bounds.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2 n max(conv(seq,seq)) / (sum(seq)^2)
    using an FFT‑based convolution.  The sequence may be un‑normalised.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(‑lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{‑p_left} + w_right·(n‑i)^{‑p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^{alpha‑1}·(n‑i)^{beta‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Draw a Dirichlet‑distributed vector (concentration > 0)."""
    if concentration <= 0:
        concentration = 1.0
    sample = np.random.gamma(concentration, 1.0, size=n)
    return sample


def generate_linear_decrease(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)."""
    return (n - np.arange(n, dtype=np.float64))


# ----------------------------------------------------------------------
# Adam optimiser for the max‑convolution objective
# ----------------------------------------------------------------------


def adam_refine(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    base_lr: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the simplex.
    Returns the best (lowest) sequence seen and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    a = project_simplex(a)
    n = a.shape[0]
    fft_len = next_pow2(2 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_score = fast_score(a)

    it = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        it += 1

        # ---- current convolution and objective ----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_score = 2.0 * n * M  # sum(a)=1 after projection
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ---- sub‑gradient of the active max entries ----
        eps_active = 1e-12
        active = np.where(conv >= M - eps_active)[0]
        g = np.zeros_like(a)

        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            # each ordered pair contributes a[j] to the derivative w.r.t a[i]
            g[i_range] += 2.0 * a[j_range]

        # ---- Adam update ----
        lr = base_lr / (np.sqrt(it) + 1.0)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = project_simplex(a)

        # ---- occasional random jitter to escape plateaus ----
        if no_improve >= 3000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    best_score = fast_score(best_seq)
    return best_seq, best_score

            seq = generate_dirichlet(n, conc)

        else:  # "lin_dec"
            seq = generate_linear_decrease(n)

        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total < 1e-12:
            continue
        seq = seq / total

        sc = fast_score(seq)

        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(
                f"[{time.time()-start:.1f}s] new global best ({family}) n={n} score={sc:.6f}"
            )

        # maintain elite pool
        if len(elite) < elite_size:
            elite.append((sc, seq.copy()))
        else:
            worst_idx = max(range(len(elite)), key=lambda i: elite[i][0])
            if sc < elite[worst_idx][0]:
                elite[worst_idx] = (sc, seq.copy())

    elite.sort(key=lambda x: x[0])
    print(f"[elite] retained {len(elite)} candidates; best elite score={elite[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement on elite candidates (remaining budget)
    # ------------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining > 5 and elite:
        per_candidate = time_remaining / len(elite)
        for idx, (sc, seq) in enumerate(elite):
            if time.time() >= deadline:
                break
            print(
                f"[refine] elite #{idx+1} length={seq.size} start_score={sc:.6f}"
            )
            refined_seq, refined_sc = adam_refine(seq, per_candidate, time.time())
            if refined_sc < best_score:
                best_score = refined_sc
                best_seq = refined_seq.copy()
                print(f"[refine] improved global best → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification using the supplied evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")

    return final_seq
