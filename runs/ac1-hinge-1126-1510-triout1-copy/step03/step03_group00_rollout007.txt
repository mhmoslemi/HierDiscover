"""
Improved search for low max‑convolution step sequence.

The algorithm uses:
1. Random restarts with continuation L‑BFGS on a smooth max (log‑sum‑exp),
   using a finer temperature schedule.
2. Projected momentum sub‑gradient descent.
3. Simulated‑annealing style random mass transfer with geometric cooling.
4. A multi‑delta greedy hill‑climb for final polishing.

All steps respect the probability simplex (non‑negative, sum = 1) and run
within the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ---------- Utility functions ----------

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Map ℝⁿ to probability simplex via softmax (stable)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Return 2·n·max(a*a) (a must already sum to 1)."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * np.max(conv))

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ---------- Smooth max objective (log‑sum‑exp) ----------

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth approximation of max(conv) using log‑sum‑exp.
    Returns f and ∇_y f.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c      # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                 # ∂smooth_max/∂conv_i
    # gradient of conv = a * a w.r.t. a (via convolution with reversed a)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part that corresponds to derivative w.r.t. each a_i
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# ---------- Continuation L‑BFGS ----------

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each temperature in lam_schedule.
    Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # L‑BFGS on smoothed objective
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-10,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[lbfgs] improved score={best_score:.6f} (lam={lam})")
        # short hill‑climb after each temperature
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)
    return best_a, best_score

# ---------- Momentum sub‑gradient descent ----------

def subgradient_momentum_optimize(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)

    velocity = np.zeros_like(a)
    beta = 0.85          # momentum coefficient
    it = 0

    while time.time() < deadline:
        # compute convolution and active mask (entries close to max)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * g
        step = 0.5 / np.sqrt(it + 1.0)

        a_candidate = simplex_projection(a - step * velocity)
        cand_sc = fast_score(a_candidate, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_sc

# ---------- Greedy hill‑climb with multi‑delta ----------

def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 3000):
    """
    Greedy hill‑climb that moves a fraction of mass between two indices.
    Tries several candidate fractions each iteration.
    """
    best = a.copy()
    best_sc = current_score
    rng = np.random.default_rng()
    # candidate fractions of the source mass
    fractions = [0.02, 0.05, 0.1, 0.2]

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] <= 0.0:
            it += 1
            continue

        src = best[i]
        improved = False
        best_local_sc = best_sc
        best_local = None

        for f in fractions:
            delta = src * f
            if delta <= 0.0 or delta > src:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = fast_score(cand, fft_len, n)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = cand
                improved = True
                # we keep moving to the best fraction found
        if improved:
            best = best_local
            best_sc = best_local_sc
        it += 1

    return best, best_sc

# ---------- Simulated annealing ----------

def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simple annealing that swaps a random fraction of mass between two
    coordinates, accepting worsening moves with a temperature that
    decays geometrically.
    """
    cur_a = a.copy()
    cur_sc = score
    best_a = a.copy()
    best_sc = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_a[i] <= 0.0:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < cur_sc:
            cur_a, cur_sc = cand, sc
            if sc < best_sc:
                best_a, best_sc = cand.copy(), sc
        else:
            prob = np.exp(-(sc - cur_sc) / T)
            if rng.random() < prob:
                cur_a, cur_sc = cand, sc

        it += 1
        if it % 5000 == 0:
            T *= 0.92
            T = max(T, 1e-5)

    return best_a, best_sc

# ---------- Random start generation ----------

def generate_random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector using several heuristics.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'two_block', 'triangular', 'hann'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.008, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, a_center)
        a[:m_edge] = a_edge
        a[-m_edge:] = a_edge
    elif mode == 'two_block':
        block_frac = rng.uniform(0.04, 0.15)
        L = max(1, int(block_frac * n))
        block_weight = rng.uniform(0.4, 0.8)
        a = np.zeros(n)
        a[:L] = block_weight / L
        a[-L:] = (1 - block_weight) / L
    elif mode == 'triangular':
        i = np.arange(n, dtype=float)
        a = np.minimum(i + 1, n - i)
        a = a.astype(float)
    elif mode == 'hann':
        i = np.arange(n, dtype=float)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

"""
Improved search for a low‑max‑convolution step sequence.

The algorithm consists of four stages:
1.  Continuation L‑BFGS on a smoothed max (log‑sum‑exp) with a finer
    temperature schedule and many random restarts.
2.  Projected momentum sub‑gradient descent.
3.  Simulated‑annealing style random mass transfers with geometric cooling.
4.  A multi‑delta greedy hill‑climb for final polishing.

All updates respect the probability simplex (non‑negative, sum = 1) and
the whole procedure respects the 1000 s budget.  The result is returned
as a list of floats; the harness will subsequently call
`evaluate_sequence` on it.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Return 2·n·max(a*a) (a already normalised to sum = 1)."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * np.max(conv))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) with respect to y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c          # stability shift
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z                    # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
):
    """
    Run L‑BFGS for each temperature in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-10, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[lbfgs] improved score={best_score:.6f} (lam={lam})")

        # tiny hill‑climb after each temperature
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=1200,
        )
    return best_a, best_score


# ----------------------------------------------------------------------
# Momentum sub‑gradient descent
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
):
    """
    Projected sub‑gradient descent with exponential momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)

    velocity = np.zeros_like(a)
    beta = 0.85
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        velocity = beta * velocity + (1.0 - beta) * g
        step = 0.5 / np.sqrt(it + 1.0)

        a_candidate = simplex_projection(a - step * velocity)
        cand_sc = fast_score(a_candidate, fft_len_conv, n)

        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = a_candidate.copy()

        a = a_candidate
        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Greedy hill‑climb with several candidate deltas
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
):
    """
    Greedy improvement by moving a fraction of mass between two indices.
    Multiple candidate fractions are tried each iteration.
    """
    best = a.copy()
    best_sc = current_score
    rng = np.random.default_rng()
    fractions = [0.02, 0.05, 0.1, 0.2]

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] <= 0.0:
            it += 1
            continue

        src = best[i]
        improved = False
        best_local_sc = best_sc
        best_local = None

        for f in fractions:
            delta = src * f
            if delta <= 0.0 or delta > src:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            sc = fast_score(cand, fft_len, n)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local = cand
                improved = True

        if improved:
            best = best_local
            best_sc = best_local_sc

        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Simulated annealing / random perturbation
# ----------------------------------------------------------------------


def random_anneal(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
):
    """
    Random mass transfers with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_sc = score
    best_a = a.copy()
    best_sc = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_a[i] <= 0.0:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        sc = fast_score(cand, fft_len, n)
        if sc < cur_sc:
            cur_a, cur_sc = cand, sc
            if sc < best_sc:
                best_a, best_sc = cand.copy(), sc
        else:
            prob = np.exp(-(sc - cur_sc) / T)
            if rng.random() < prob:
                cur_a, cur_sc = cand, sc

        it += 1
        if it % 5000 == 0:
            T *= 0.92
            T = max(T, 1e-5)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Random start generation (various heuristics)
# ----------------------------------------------------------------------


def generate_random_start(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector using several predefined patterns."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "two_block",
            "triangular",
            "hann",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.008, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, a_center)
        a[:m_edge] = a_edge
        a[-m_edge:] = a_edge
    elif mode == "two_block":
        block_frac = rng.uniform(0.04, 0.15)
        L = max(1, int(block_frac * n))
        left_weight = rng.uniform(0.4, 0.8)
        a = np.zeros(n)
        a[:L] = left_weight / L
        a[-L:] = (1.0 - left_weight) / L
    elif mode == "triangular":
        i = np.arange(n, dtype=float)
        a = np.minimum(i + 1, n - i)
    elif mode == "hann":
        i = np.arange(n, dtype=float)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a
