def fft_convolve(x, y):
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n-1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:n]
    return conv

def smooth_max_and_gradient(a, alpha):
    n = a.shape[0]
    # compute convolution a*a
    conv = compute_full_conv(a)   # shape 2n-1
    # stable log-sum-exp
    cmax = np.max(conv)
    exp_terms = np.exp(alpha * (conv - cmax))
    sum_exp = np.sum(exp_terms)
    smooth = cmax + np.log(sum_exp) / alpha
    w = exp_terms / sum_exp   # shape 2n-1

    # compute gradient: 2 * conv(w, a_rev)[n-1:n-1+n]
    a_rev = a[::-1]
    # convolution of w and a_rev
    grad_conv = compute_fft_convolve(w, a_rev)  # length len(w)+len(a_rev)-1 = (2n-1)+n-1 = 3n-2
    # extract slice
    start = n - 1
    grad = 2.0 * grad_conv[start:start + n]
    return smooth, grad

def compute_fft_convolve(x, y):
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())   # smallest power of two >= n
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:n]
    return conv

def smooth_descent(a0, steps, alpha_schedule, step0, deadline):
    a = a0.copy()
    n = a.shape[0]
    a = a / a.sum()
    best_seq = a.copy()
    best_score = float('inf')
    it = 0
    # optional Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for alpha in alpha_schedule:
        # Determine steps per alpha portion: allocate roughly steps // len(alpha_schedule)
        steps_alpha = max(1, steps // len(alpha_schedule))
        for _ in range(steps_alpha):
            if time.time() >= deadline:
                return best_seq, best_score
            it += 1
            # compute smooth and gradient
            smooth, grad = smooth_max_and_gradient(a, alpha)
            # scale gradient for actual objective: gradient of score = 2*n * gradient of smooth
            grad_full = 2.0 * n * grad
            # Adam update
            m = beta1 * m + (1 - beta1) * grad_full
            v = beta2 * v + (1 - beta2) * (grad_full ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            lr = step0 / (np.sqrt(v_hat) + eps)    # typical Adam adapt step
            # step
            a = a - lr * m_hat
            # projection onto simplex
            a = project_simplex(a)

            # evaluate actual score
            max_b = np.max(compute_full_conv(a))
            score = 2.0 * n * max_b
            if score < best_score:
                best_score = score
                best_seq = a.copy()
    return best_seq, best_score

def compute_fft_convolve(x, y):
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:n]
    return conv

def subgradient_descent(a0, max_iters, step0, deadline):
    a = a0.copy()
    n = a.shape[0]
    a = a / a.sum()
    best_seq = a.copy()
    best_score = float('inf')
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # compute subgradient for active peak
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]
        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)
    return best_seq, best_score

"""
Hybrid optimizer for the step‑function evaluation problem.

The algorithm searches several sequence lengths.  For each length it
performs many independent restarts.  A restart consists of:

1.  Initialise a non‑negative vector on the probability simplex
    (uniform, random power‑law or a perturbed best solution).
2.  Smooth‑max gradient descent: minimise an
    `log‑sum‑exp` approximation of `max(conv(a,a))` using the exact
    gradient (two FFT‑based convolutions per iteration).  The smoothing
    parameter `α` is increased gradually (continuation method) to focus
    the search on the true maximum.
3.  Projected sub‑gradient descent on the true `max` (active‑peak
    gradient).  This fine‑tunes the solution while keeping it on the
    simplex.
4.  A cheap stochastic local refinement (simulated‑annealing style)
    that moves a small amount of mass between two coordinates and
    accepts improving or occasionally uphill moves.

All phases respect a global deadline (the provided time budget).  The
best sequence seen across all lengths and restarts is returned as a
plain list of floats.  The implementation uses only `numpy` and
standard library modules, runs on one or two CPUs and respects the
constraints of the evaluation harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT based linear convolution (real inputs)
# ----------------------------------------------------------------------
def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a power‑of‑two FFT."""
    n = len(x) + len(y) - 1
    fft_len = 1 << ((n - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, fft_len)[:n]
    return conv


def compute_full_conv(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    return fft_convolve(a, a)


def compute_max_conv(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    conv = compute_full_conv(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) and its exact gradient
# ----------------------------------------------------------------------
def smooth_max_and_gradient(a: np.ndarray, alpha: float):
    """
    Returns a smooth approximation of max(conv(a,a)) and the gradient
    of that approximation w.r.t. a.
    """
    n = a.shape[0]

    # full convolution and its entries
    c = compute_full_conv(a)                     # length 2n‑1

    # stable log‑sum‑exp
    cmax = np.max(c)
    exp_terms = np.exp(alpha * (c - cmax))
    sum_exp = np.sum(exp_terms)
    smooth = cmax + np.log(sum_exp) / alpha

    # softmax weights (sum to 1)
    w = exp_terms / sum_exp                      # length 2n‑1

    # gradient = 2 * (w * a_rev)_{n‑1 … n‑1+n‑1}
    a_rev = a[::-1]
    conv_w_rev = fft_convolve(w, a_rev)         # length 3n‑2
    start = n - 1
    grad = 2.0 * conv_w_rev[start:start + n]    # length n
    return smooth, grad


def smooth_descent(a0: np.ndarray,
                   total_steps: int,
                   alpha_schedule,
                   step0: float,
                   deadline: float):
    """
    Gradient descent on the smooth max approximation.
    Returns the best sequence and its true evaluation score.
    """
    a = a0.copy()
    a = a / a.sum()               # enforce Σa = 1
    n = a.shape[0]

    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    it = 0
    # allocate roughly equal steps to each α
    steps_per_alpha = max(1, total_steps // max(1, len(alpha_schedule)))

    for alpha in alpha_schedule:
        for _ in range(steps_per_alpha):
            if time.time() >= deadline:
                return best_seq, best_score
            it += 1
            smooth, grad = smooth_max_and_gradient(a, alpha)

            # gradient of the true objective 2·n·smooth
            grad_obj = 2.0 * n * grad
            step = step0 / np.sqrt(it)
            a = project_simplex(a - step * grad_obj)

            # true score
            cur_score = 2.0 * n * compute_max_conv(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max
# ----------------------------------------------------------------------
def subgradient_descent(a0: np.ndarray,
                        max_iters: int,
                        step0: float,
                        deadline: float):
    """
    Classic projected sub‑gradient method that follows the gradient of
    the currently active peak of the self‑convolution.
    """
    a = a0.copy()
    n = a.shape[0]
    a = a / a.sum()                       # Σa = 1

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, max_b = compute_full_conv(a), None
        # compute max only once (the function above returns the full conv)
        max_b = float(conv.max())

        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # occasional progress report (kept lightweight)
            if it % 1000 == 0:
                print(f"[subgrad n={n} it={it}] score={score:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient of the *active* peak
        # --------------------------------------------------------------
        k = int(np.argmax(conv))           # index where the peak occurs
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        # grad_i = 2 * a_{k-i}
        grad[idx] = 2.0 * a[k - idx]

        # diminishing step size
        step = step0 / np.sqrt(it)

        a = project_simplex(a - step * grad)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (SA style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float):
    """
    Cheap random walk that moves a tiny amount of mass between random
    coordinates, accepting improvements (or occasional uphill moves).
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02  # initial acceptance temperature

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of mass transfer
        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)

        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        # Metropolis acceptance
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        # cooling
        temperature *= 0.9999
        temperature = max(temperature, 1e-6)

    return best_seq, best_score
