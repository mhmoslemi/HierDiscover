"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that
minimises the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)**2)

(the denominator is scale‑invariant, so we keep sum(a)=1).

The algorithm is a hybrid of three ideas:

1. **Continuation L‑BFGS** on a smooth approximation of the max‑operator.
   A geometric schedule of smoothing parameters λ drives the solution
   towards the true max.  Several random restarts (uniform,
   exponential, power‑law, edge‑biased and perturbations of the current
   best) explore the space.

2. **Aggressive hill‑climb** – a greedy local search that repeatedly
   transfers a fraction of the mass from one coordinate to another and
   keeps the move only if the fast‑score (2·n·max(conv)) improves.

3. **Projected sub‑gradient descent** on the exact non‑smooth objective.
   The sub‑gradient of the peak autocorrelation is easy to compute:
   for the index `k` where the convolution attains its maximum,
   ∂c_k/∂a_i = a_{k‑i} + a_{i‑k} (zero when the index is out of range).
   A step along the negative sub‑gradient is taken, then the vector is
   projected back onto the probability simplex.  The step size follows a
   1/√t schedule.

All phases respect a global time budget (default 1000 s).  The best
sequence found before the deadline is returned.

The implementation works with any `n` (default 5000) and uses only
NumPy/SciPy – no filesystem or network access.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when sum(a)=1:
        2·n·max(conv(a,a))
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x >= 0, sum(x)=1}.
    """
    if np.all(v <= 0):
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a at the index where the
    convolution attains its maximum.
    """
    n = a.shape[0]
    i = np.arange(n)
    # first term a_{max_idx-i}
    idx1 = max_idx - i
    mask1 = (idx1 >= 0) & (idx1 < n)
    g = np.zeros_like(a)
    g[mask1] += a[idx1[mask1]]

    # second term a_{i-max_idx}
    idx2 = i - max_idx
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] += a[idx2[mask2]]
    return g


def projected_subgradient_descent(
    a0: np.ndarray,
    fft_len: int,
    n: int,
    steps: int,
    eta0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Returns the best sequence encountered and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for t in range(steps):
        if time.time() >= deadline:
            break

        conv = conv_fft(a, fft_len, n)
        max_idx = int(np.argmax(conv))

        g = subgradient_max(a, max_idx)
        # Remove the mean to keep the step on the simplex (sum‑preserving)
        g_centered = g - g.mean()

        eta = eta0 / np.sqrt(t + 1)
        a_new = a - eta * g_centered
        a_new = project_onto_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly moves a fraction of the mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.08  # up to 8 % of the source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑softmax coordinates y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z  # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS using a decreasing schedule of λ (lam_factors).
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def random_initial_distribution(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a probability vector according to ``mode``.
    """
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "random":
        a = rng.random(n)
        a /= a.sum()

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()

    elif mode == "edge":
        # put a higher constant height on both ends, lower constant in the centre
        edge_frac = rng.uniform(0.01, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a_center = 1.0 / np.sqrt(2.0) / n
        a_edge = np.sqrt(n / 2.0) / n
        a = np.full(n, a_center)
        a[:m_edge] = a_edge
        a[-m_edge:] = a_edge
        a /= a.sum()

    else:
        # fallback – tiny random perturbation of the uniform vector
        a = np.full(n, 1.0 / n)
        a += rng.normal(scale=0.001, size=n)
        a = np.clip(a, 0, None)
        a /= a.sum()

    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # problem dimension
    # ------------------------------------------------------------------
    n = 5000
    # If a previous solution is available, try to beat its length.
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                n = max(n, int(arr.size))
        except Exception:
            pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initial best (uniform) – also respect an existing solution if any
    # ------------------------------------------------------------------
    if prev is not None:
        try:
            a0 = np.asarray(prev, dtype=float)
            a0 = np.clip(a0, 0, None)
            if a0.sum() < 1e-12:
                a0 = np.full_like(a0, 1.0 / a0.size)
            else:
                a0 = a0 / a0.sum()
            if a0.shape[0] != n:
                # resize by simple interpolation if sizes differ
                a0 = np.interp(
                    np.linspace(0, a0.shape[0] - 1, n),
                    np.arange(a0.shape[0]),
                    a0,
                )
                a0 = np.clip(a0, 0, None)
                a0 /= a0.sum()
        except Exception:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 0.001, num=22)   # smoother continuation
    maxiter_stage = 350                               # L‑BFGS per λ

    iteration = 0

    # ------------------------------------------------------------------
    # multi‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # choose a stochastic start
        mode = rng.choice(["uniform", "random", "exp", "power", "edge", "best"])
        if mode == "best":
            a_start = best_a + rng.normal(scale=0.03, size=n)
            a_start = np.clip(a_start, 0, None)
            a_start /= a_start.sum()
        else:
            a_start = random_initial_distribution(mode, n, rng)

        y0 = np.log(a_start + 1e-12)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- aggressive hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- projected sub‑gradient refinement -----
        sg_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = projected_subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            steps=2500,
            eta0=0.20,
            deadline=sg_deadline,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final deep refinement (L‑BFGS + sub‑gradient + hill‑climb)
    # ------------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # one last round of sub‑gradient descent
    if time.time() < deadline:
        sg_deadline = deadline
        best_a, best_score = projected_subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            steps=6000,
            eta0=0.15,
            deadline=sg_deadline,
        )
        best_y = np.log(best_a + 1e-12)

    # final hill‑climb polishing (use any remaining time)
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
