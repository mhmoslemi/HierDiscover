def golden_section_search(f, lo, hi, max_iter=30):
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2
    a = hi - invphi * (hi - lo)
    b = lo + invphi * (hi - lo)
    fa = f(a)
    fb = f(b)
    for _ in range(max_iter):
        if fa < fb:
            hi = b
            b = a
            fb = fa
            a = hi - invphi * (hi - lo)
            fa = f(a)
        else:
            lo = a
            a = b
            fa = fb
            b = lo + invphi * (hi - lo)
            fb = f(b)
    return (lo + hi) / 2.0, f((lo+hi)/2)

"""
Improved search for a low‑evaluation step‑function.

Idea
----
1.  The most effective simple families are (i) an exponential decay
    a_i ∝ exp(−λ·i) (or its mirror) and (ii) a two‑level “edge” shape.
    Both families are described by one or two scalar parameters,
    therefore we can optimise them with a cheap 1‑D golden‑section search.
2.  The best exponential candidate is taken as the starting point for a
    continuation L‑BFGS optimisation on the probability simplex (soft‑max
    parametrisation).  The smooth‑max approximation already used in the
    reference code is retained, but we run a longer continuation schedule
    and a single, focused restart (no random restarts – the exponential
    start is already very good).
3.  Finally a short greedy hill‑climb refines the solution by moving a
    small fraction of mass between random coordinates whenever the fast
    score improves.

The whole routine respects the 1000 s time budget, prints a few progress
messages and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (FFT‑based convolution)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    a is assumed to have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    This equals the value returned by ``evaluate_sequence`` (the denominator
    ``sum(a)**2`` is 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (the log‑soft‑max coordinates).

    Returns
    -------
    f : float
        Smooth approximation of 2·n·max(a*a).
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list (geometric schedule) that
    gradually reduces the smoothing of the max‑operator.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple greedy hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly moves a fraction of the mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10  # up to 10 % of the source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Exponential family search (1‑D golden‑section)
# ----------------------------------------------------------------------


def exponential_initial(n: int, rate: float) -> np.ndarray:
    """Return a normalised exponential decay vector a_i ∝ exp(-rate·i)."""
    a = np.exp(-rate * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def golden_section_minimize(func, lo: float, hi: float, iters: int = 30):
    """Standard golden‑section search on a unimodal 1‑D function."""
    phi = (1 + np.sqrt(5)) / 2.0
    invphi = (np.sqrt(5) - 1) / 2.0

    a = hi - invphi * (hi - lo)
    b = lo + invphi * (hi - lo)
    fa = func(a)
    fb = func(b)

    for _ in range(iters):
        if fa < fb:
            hi = b
            b = a
            fb = fa
            a = hi - invphi * (hi - lo)
            fa = func(a)
        else:
            lo = a
            a = b
            fa = fb
            b = lo + invphi * (hi - lo)
            fb = func(b)

    x_opt = (lo + hi) / 2.0
    f_opt = func(x_opt)
    return x_opt, f_opt


def find_best_exponential(n: int, fft_len: int, deadline: float):
    """
    Search the rate λ of an exponential decay a_i ∝ exp(-λ·i) that yields the
    smallest fast_score.  The search is a cheap golden‑section optimisation.
    """
    # fast_score needs the FFT length to be pre‑computed
    def score_of_rate(rate):
        a = exponential_initial(n, rate)
        return fast_score(a, fft_len, n)

    # reasonable bounds: 0 (uniform) to 0.02 (very steep)
    lo, hi = 0.0, 0.02
    # first coarse scan to avoid wasting time if the budget is tiny
    if time.time() > deadline:
        return exponential_initial(n, 0.0)  # fallback to uniform

    best_rate, best_val = golden_section_minimize(score_of_rate, lo, hi, iters=30)
    print(f"[exp‑search] best rate = {best_rate:.6f}, score = {best_val:.6f}")
    return exponential_initial(n, best_rate)


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    The method:
        * a quick 1‑D optimisation of an exponential decay,
        * a continuation L‑BFGS refinement,
        * a short greedy hill‑climb.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # problem size – reuse a previous long construction if available
    # --------------------------------------------------------------
    n = 5000
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, list) and len(prev) > n:
            n = len(prev)
    except Exception:
        pass

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1.  Exponential‑family pre‑search
    # --------------------------------------------------------------
    a0 = find_best_exponential(n, fft_len, deadline)
    score0 = fast_score(a0, fft_len, n)
    print(f"[init] exponential start fast score = {score0:.6f}")

    # --------------------------------------------------------------
    # 2.  Continuation L‑BFGS (single focused run)
    # --------------------------------------------------------------
    y0 = np.log(a0 + 1e-12)

    # a moderately long, fine‑grained continuation schedule
    lam_factors = np.geomspace(0.5, 0.001, num=30)
    maxiter_stage = 400

    y_opt = schedule_optimize(
        y_start=y0,
        n=n,
        fft_len_conv=fft_len,
        fft_len_g=fft_len_g,
        lam_factors=lam_factors,
        maxiter_per_stage=maxiter_stage,
    )
    a_opt = softmax(y_opt)
    score_opt = fast_score(a_opt, fft_len, n)
    print(f"[lbfgs] after continuation fast score = {score_opt:.6f}")

    # --------------------------------------------------------------
    # 3.  Final greedy hill‑climb (uses any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, score_opt = hill_climb(
            a_opt,
            score_opt,
            fft_len,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[hill‑climb] final fast score = {score_opt:.6f}")

    # --------------------------------------------------------------
    # 4.  Official evaluation and return
    # --------------------------------------------------------------
    final_seq = a_opt.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence for possible reuse in later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
