Propose a step‑function (non‑negative heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Broad random sampling** – generate several thousand candidates from
   various parametric families (power‑law, exponential, two‑sided power‑law,
   beta‑like, cosine/Hann, Gaussian).  The best candidate according to a
   fast FFT‑based score is kept.

2. **Multi‑peak exponentiated‑gradient refinement** – starting from the best
   candidate we repeatedly:
      * compute the self‑convolution by FFT,
      * pick the top‑K (default 3) indices whose values are within a small
        tolerance of the current maximum,
      * form a sub‑gradient by adding the contributions of all active
        indices,
      * perform a multiplicative update  a ← a·exp(−η·g)  (η decays with
        the iteration count) and renormalise onto the simplex,
      * occasionally add a tiny random jitter to escape plateaus.

   peak leads to a noticeably lower final value.

3. **Verification** – the best sequence found is returned as a plain list;

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast (FFT based) evaluation of
          2·n·max(conv(seq,seq)) / (sum(seq)**2)

    ``seq`` is assumed to be non‑negative and 1‑normalised (sum==1);
    the function returns ``inf`` for degenerate inputs.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    # convolution via FFT (real‑to‑real)
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Discrete version of (1‑z)^{-1/2}."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_cosine_hann(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))  # length‑n Hann


def generate_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian window centred in the middle; ``sigma_frac`` ∈ (0,0.5)."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_like(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density sampled on a uniform grid."""
    x = (np.arange(1, n + 1, dtype=np.float64) - 0.5) / n
    # avoid under‑flow for extreme α
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    return a
