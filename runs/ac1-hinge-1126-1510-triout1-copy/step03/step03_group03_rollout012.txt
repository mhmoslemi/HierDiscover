"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the ordinary linear convolution.
The algorithm works in two phases inside the given time budget
(≈ 1000 s).

1. **Broad random sampling** – a few thousand candidates are drawn from
   several parametrised families (power‑law, exponential, two‑sided
   power‑law, arcsine, Hann, Gaussian).  The best candidate (lowest
   fast score) is kept as a starting point.

2. **Smooth‑max exponentiated gradient descent** – starting from the
   best candidate we iteratively flatten the self‑convolution.
   In each iteration we
   * compute the full convolution `c = a*a` with FFT,
   * build a soft‑max weight vector `w` over the entries of `c`
     (the temperature β grows from 0.5 to 200, turning the smooth
     approximation into the true max),
   * obtain the gradient of the smooth max:
       g_i = 2·∑_j a_j·w_{i+j}   (cross‑correlation of `a` and `w`),
     computed again with FFT,
   * perform an *exponentiated* (multiplicative) update
       a_i ← a_i·exp(−η·g_i)      (η decays with the iteration count),
   * renormalise `a` onto the probability simplex.

   Periodically (every few thousand steps) the current sequence is
   evaluated with a cheap FFT‑based score; if it improves, it becomes
   the new best.  Small random jitter is added occasionally to escape
   plateaus.

The final best sequence is returned as a plain Python list; a final
call to the official `evaluate_sequence` function reports the true
score.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using FFT.  The sequence must be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    size = next_pow2(2 * n)
    Fa = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(Fa * Fa, n=size)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate sequences (used in the random exploration)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Discrete version of the arcsine law a_i ∝ 1/√i.
    a_0 = 1, a_i = a_{i-1}·(2i-1)/(2i).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_cosine_hann(n: int) -> np.ndarray:
    """Hann window (cosine shape)."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian(n: int, sigma_frac: float) -> np.ndarray:
    """Truncated Gaussian centred in the middle."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a low‑score height sequence under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    # --------------------------------------------------------------
    # Setup
    # --------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 0️⃣  Determine a working length (larger lengths give a finer bound)
    # --------------------------------------------------------------
    # We pick a length that makes the FFT size a power of two and still
    # finishes comfortably inside the time budget.
    n = 12000                                 # feel free to adjust
    print(f"[info] chosen length n = {n}")

    # --------------------------------------------------------------
    # 1️⃣  Broad random exploration (short phase – ~150 s)
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    families = [
        "power", "exp", "twosided", "arcsine",
        "cosine", "gaussian"
    ]

    # First try the historic best if it exists
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        # If the historic sequence is of a different length, resize it
        if init.size != n:
            init = generate_arcsine_like(n)
    except Exception:
        init = generate_arcsine_like(n)

    init = np.maximum(init, 0.0)
    if init.sum() == 0:
        init = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        init = init / init.sum()

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] fast_score = {best_score:.6f}")

    # Random families – a few thousand samples
    exploratory_deadline = start + 150.0
    while time.time() < exploratory_deadline:
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.30, 0.80)
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_cosine_hann(n)

        elif family == "gaussian":
            sigma_frac = random.uniform(0.07, 0.30)
            cand = generate_gaussian(n, sigma_frac)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:.1f}s] new random best ({family}) score={sc:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑max exponentiated gradient descent
    # --------------------------------------------------------------
    a = best_seq.copy()
    L = next_pow2(3 * n)          # size for all FFTs (covers 2n‑1 + n‑1)

    # Hyper‑parameters (tuned empirically)
    beta_start = 0.5
    beta_end = 200.0
    beta = beta_start
    max_iters = 300000           # will be cut by time limit
    eta0 = 1.0
    tau = 8000.0                 # decay scale for the step size
    eval_interval = 2000
    jitter_interval = 5000
    jitter_scale = 0.002

    # Linear schedule for β
    beta_inc = (beta_end - beta_start) / max_iters

    iteration = 0
    while time.time() < deadline - 2.0 and iteration < max_iters:
        iteration += 1

        # ------------------- convolution & max -------------------
        Fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(Fa * Fa, n=L)[: 2 * n - 1]
        max_conv = float(conv_full.max())

        # ------------------- soft‑max weights --------------------
        # w_k ∝ exp(β·(conv_k - max_conv))  →  sum w = 1
        # this is numerically safe because the exponent is ≤ 0
        exp_term = np.exp(beta * (conv_full - max_conv))
        w = exp_term / exp_term.sum()

        # ------------------- gradient (cross‑correlation) -------
        Fw = np.fft.rfft(w, n=L)
        cross = np.fft.irfft(Fa * np.conj(Fw), n=L)[:n]   # g_i / 2
        g = 2.0 * cross

        # ------------------- multiplicative (exponentiated) update
        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * g)

        # ------------------- re‑normalise onto simplex ----------
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ------------------- occasional jitter -------------------
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum()

        # ------------------- book‑keeping & best update ----------
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start
                print(
                    f"[{elapsed:.1f}s] iter {iteration:,} – new best score={cur_score:.6f}"
                )

        # ------------------- temperature schedule ---------------
        beta = min(beta_end, beta + beta_inc)

    # --------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_seq.tolist()
    # The official evaluator will clip negatives / huge values anyway,
    # but we keep a clean sequence.
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"[finished] best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
