def lp_step(f, M):
    # solve max sum(g) subject to conv(f,g) <= M, g>=0.
    # f size n, M scalar.
    n = len(f)
    num_k = 2*n - 1
    # Build sparse matrix A shape (num_k, n) where A[k,i] = f[i] if 0 <= k-i < n else 0
    rows = []
    cols = []
    data = []
    for i in range(n):
        # for k from i to i + n - 1 (since k-i index j from 0..n-1)
        row_start = i
        row_end = i + n
        # this may be large loops; but we can vectorize using range.
        # Instead, fill using loops outer on i.
        rows.extend(range(row_start, row_end))
        cols.extend([i] * n)
        data.extend([f[i]] * n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(num_k,n))
    # constraints: A @ g <= M
    # Solve LP using linprog with method 'highs' and specifying A_ub, b_ub.
    c = -np.ones(n)  # minimize negative sum = maximize sum
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(num_k, M), bounds=bounds,
                                 method='highs', options={'presolve': True, 'time_limit': 10})
    g = res.x
    return g, res.success
```

"""
Step‑function optimizer using a linear‑programming refinement loop.

The algorithm works with a probability vector a (sum = 1).  Its
objective is to minimise 2·n·max(a*a) where * denotes linear
convolution.  The strategy is:

1. Initialise a with a random (or edge‑biased) distribution.
2. Repeatedly:
   • Compute M = max(a*a).
   • Solve a linear programme:
       maximise   Σ g_i
       subject to  (a·g)_k ≤ M   for all k = 0 … 2n‑2
                   g_i ≥ 0 .
     The constraints are linear in g because (a·g) is a convolution.
   • Normalise the LP solution to Σ g_i = 1 and move a a small step
     towards g (convex combination).  The step size t is modest so
     M never increases – this reproduces the “active‑set” method
     described in the statement.
   • A short greedy hill‑climb (mass transfers) is applied to polish
     the current a.
3. Keep the best sequence found over many random restarts until the
   time budget expires.  Finally the best vector is returned; the
   harness will call ``evaluate_sequence`` on it.

The LP formulation dramatically reduces the maximum convolution
value, often achieving a fast‑score well below the previous 1.512.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# --------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·n‑1) using FFT.
    a must be a 1‑D real array of length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1).
    This equals the value returned by ``evaluate_sequence``.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the (2·n‑1) × n matrix A such that
        (f * g) = A @ g
    for any vector g of length n.
    The entry A[k, j] equals f_{k‑j} when 0≤k‑j<n, otherwise 0.
    """
    n = f.shape[0]
    # i – index in f, j – index in g (the variable)
    i = np.arange(n, dtype=np.int32)[:, None]          # shape (n,1)
    j = np.arange(n, dtype=np.int32)[None, :]          # shape (1,n)

    rows = (i + j).ravel()                             # k = i + j
    cols = np.tile(j.ravel(), n)                       # repeat 0..n-1 for each i
    data = np.repeat(f, n)                             # f_i is repeated n times

    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly moves a fraction of mass from one index to another and
    keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    ``evaluate_sequence`` score.  The method uses a linear‑programming
    refinement loop combined with random restarts and a final hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Problem size – a value of a few thousand keeps the LP tractable.
    n = 1500
    fft_len = next_pow2(2 * n - 1)

    # Baseline: uniform distribution
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast score = {best_score:.6f}")

    # Some hyper‑parameters
    t_step = 0.30                # convex‑combination step size
    inner_lp_iters = 12          # LP refinements per restart
    max_restarts = 10_000        # effectively limited by time

    restart_id = 0

    while time.time() < deadline - 2.0 and restart_id < max_restarts:
        restart_id += 1

        # ---------- initialise a ----------
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            a0 = np.abs(rng.standard_normal(n))
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif mode == "edge":
            # concentrate a little at the two edges
            edge_len = max(1, int(n * rng.uniform(0.005, 0.02)))
            a0 = np.full(n, 1.0 / n)
            a0[:edge_len] = a0[-edge_len:] = np.sqrt(n) / n
        else:  # perturb the current best
            if best_a is None:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = best_a.copy()
                a0 += rng.normal(scale=0.02, size=n)
                a0 = np.clip(a0, 0, None)

        # normalise to sum = 1 (the evaluation function works with any scale)
        a0 = a0 + 1e-14
        a0 /= a0.sum()
        a = a0.copy()

        # ---------- LP refinement loop ----------
        for it in range(inner_lp_iters):
            # current max convolution
            conv = conv_fft(a, fft_len, n)
            M = conv.max()

            # build the linear constraints (f * g)_k ≤ M
            A = build_conv_matrix(a)                     # shape (2n‑1, n)
            b_ub = np.full(2 * n - 1, M, dtype=float)
            c = -np.ones(n, dtype=float)                 # maximise Σ g_i  → minimise -Σg_i
            bounds = [(0, None)] * n

            # solve the LP (highs is fast and accepts sparse matrices)
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method="highs", options={"presolve": True, "time_limit": 5})

            if not res.success:
                break

            g = res.x
            if g.sum() == 0.0:
                break

            # rescale g to keep Σ g_i = 1  (the same normalisation as a)
            g = g / g.sum()

            # convex combination step (guaranteed not to increase the max)
            a = (1.0 - t_step) * a + t_step * g
            a = a / a.sum()          # numerical safety

        # ---------- local greedy hill‑climb ----------
        cur_score = fast_score(a, fft_len, n)
        hc_deadline = min(deadline, time.time() + 0.8)
        a, cur_score = hill_climb(a, cur_score, fft_len, n,
                                  hc_deadline, max_iter=6000)

        # ---------- update the global best ----------
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{restart_id}] mode={mode} improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, n, final_deadline,
                                        max_iter=12000)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
