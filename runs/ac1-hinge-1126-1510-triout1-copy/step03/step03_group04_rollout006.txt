def exp_gradient_optimize(a0, deadline, n, fft_len_conv, fft_len_g):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    eta0 = 0.5
    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # active set
        # compute subgradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1:n-1+n]
        # step size
        eta = eta0 / np.sqrt(it + 1.0)
        # exponential update
        a = a * np.exp(-eta * g)
        # renorm
        sum_a = a.sum()
        if sum_a == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        # evaluate score
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def random_candidate(n, rng):
    mode = rng.choice(['uniform','exp','power','beta','edge','dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0/n)
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 3.0)
        a = np.arange(1, n+1) ** (-p)
        a /= a.sum()
    elif mode == 'beta':
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        x = (np.arange(1, n+1) - 0.5)/n
        a = (np.power(x, alpha-1) * np.power(1-x, beta-1))
        a = np.maximum(a, 1e-30)
        a /= a.sum()
    elif mode == 'edge':
        edge_frac = rng.uniform(0.001, 0.05)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0/n)
        # assign extra mass to edges, preserve sum
        # we design: total edge mass = 0.5 * sum? Not necessary.
        # Instead: set a[:m_edge] = a[-m_edge:] = 2 * 1.0/n  (more mass). ensure normalize.
        a[:m_edge] = a[-m_edge:] = 2.0/n
        a /= a.sum()
    else:  # dirichlet
        alpha_vec = np.ones(n) * rng.uniform(0.1, 5.0) # concentration
        a = rng.dirichlet(alpha_vec)
    return a

"""
Propose a low‑score step sequence.

Strategy
--------
1. **Exploratory scan** – Try a few different sequence lengths (≈500‑2500)
   and a variety of parametric families (uniform, exponential,
   power‑law, beta, edge‑heavy, random Dirichlet).  Keep the best
   candidate (sum = 1) as a starting point.

2. **Continuation L‑BFGS** – Repeatedly minimise a smooth approximation
   of the max‑convolution (log‑sum‑exp) with L‑BFGS.  The smoothing
   factor λ is decreased according to a schedule (0.5 → 0.001).  After
   each L‑BFGS run a short greedy hill‑climb is applied.

3. **Exponentiated‑gradient refinement** – Directly minimise the true
   objective using a multiplicative‑update (mirror‑descent) step on the
   simplex, with a diminishing step‑size.

4. **Final greedy polish** – A fast hill‑climb that repeatedly moves a
   small fraction of mass between two coordinates while preserving the
   sum.

All FFTs use powers‑of‑two lengths for speed, and the algorithm respects
the 1000 s time budget.  The returned list is ready for `evaluate_sequence`.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    y = np.asarray(y, dtype=np.float64)
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random sequence generators (all return a probability vector)
# ----------------------------------------------------------------------


def gen_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def gen_exponential(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n))
    return a / a.sum()


def gen_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    return a / a.sum()


def gen_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(1, n + 1) - 0.5) / n
    a = (np.power(x, alpha - 1.0) *
         np.power(1.0 - x, beta - 1.0))
    a = np.maximum(a, 1e-30)
    return a / a.sum()


def gen_edge_heavy(n: int, edge_frac: float) -> np.ndarray:
    m_edge = max(1, int(n * edge_frac))
    a = np.full(n, 1.0 / n)
    # give a little extra weight to the edges
    extra = 2.0 / n
    a[:m_edge] = extra
    a[-m_edge:] = extra
    a /= a.sum()
    return a


def gen_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    concentration = rng.uniform(0.2, 3.0)
    alphas = np.full(n, concentration)
    return rng.dirichlet(alphas)


def random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random parametric family and draw a candidate."""
    mode = rng.choice(['uniform', 'exp', 'power', 'beta',
                       'edge', 'dirichlet'])
    if mode == 'uniform':
        return gen_uniform(n)
    if mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        return gen_exponential(n, rate)
    if mode == 'power':
        p = rng.uniform(0.2, 3.0)
        return gen_powerlaw(n, p)
    if mode == 'beta':
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        return gen_beta(n, alpha, beta)
    if mode == 'edge':
        edge_frac = rng.uniform(0.001, 0.05)
        return gen_edge_heavy(n, edge_frac)
    # dirichlet
    return gen_dirichlet(n, rng)


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns
    -------
    f : float      – 2·n·smooth_max(conv)
    grad : np.ndarray – gradient w.r.t. y (shape n)
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                         # d(smooth_max)/d conv_i
    # g_vec_i = Σ_j a_j * w_{i+j}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimisation (mirror descent on the simplex)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a0: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len_conv: int,
                         fft_len_g: int) -> tuple[np.ndarray, float]:
    """
    Multiplicative‑update (mirror descent) minimisation of the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    eta0 = 0.4                                # initial learning rate
    it = 0

    while time.time() < deadline:
        # ----- convolution & active set -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the max -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # ----- mirror‑descent step -----
        eta = eta0 / np.sqrt(it + 1.0)
        update = np.exp(np.clip(-eta * g, -50.0, 50.0))
        a = a * update
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ----- record best -----
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42,

    """

    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed + 12345)

    else:
        init_len = 2000
    # candidate lengths, clipped to a reasonable window

    cand_ns = sorted({max(500, min(3000, int(init_len * f))) for f in factors})

    # ------------------------------------------------------------------
    # Exploratory scan (≈ 5 % of total budget)
    # ------------------------------------------------------------------
    start_time = time.time()
    total_deadline = start_time + budget_s - 0.5   # tiny safety margin
    scan_deadline = start_time + min(30.0,
                                    0.05 * (total_deadline - start_time))

    best_a = None
    best_score = float('inf')
    best_n = None

    print("[scan] exploring candidate lengths:", cand_ns)

    for n in cand_ns:
        if time.time() >= scan_deadline:
            break
        fft_len_conv = next_pow2(2 * n - 1)
        fft_len_g = next_pow2(3 * n - 2)

        # try a handful of random candidates for this n
        for _ in range(25):
            a = random_candidate(n, rng)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
        print(f"[scan] n={n:4d}  best_score_sofar={best_score:.6f}")

    # Fallback – if nothing was generated (should not happen)
    if best_a is None:
        best_n = 2000
        best_a = gen_uniform(best_n)

    # ------------------------------------------------------------------
    # Prepare FFT lengths for the chosen n
    # ------------------------------------------------------------------
    n = best_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] chosen n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------

    remaining = total_deadline - time.time()

    while time.time() < phase1_end:
