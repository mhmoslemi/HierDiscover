def propose_candidate(seed=42, budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    # choose n based on existing sequence or set default
    n = 8000
    init_seq = globals().get('height_sequence_1')
    if isinstance(init_seq, (list, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim ==1 and tmp.size >= 1000:
            n = max(n, int(tmp.size))
    # power of two length for FFT
    fft_len = 1 << ((2 * n - 1).bit_length())
    # initial distribution: exponential decay
    a = np.exp(-rng.uniform(0.001, 0.02) * np.arange(n))
    a /= a.sum()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} init score {best_score}")

    iteration = 0
    while time.time() < deadline:
        # pick method: subgradient EG or random restart
        if iteration % 10 == 0 and iteration>0:
            # random restart perturb
            a = best_a + rng.normal(scale=0.05, size=n)
            a = np.maximum(a, 0)
            a /= a.sum()
        # Perform a batch of exponentiated gradient steps
        for step in range(200):
            conv = conv_fft(a, fft_len, n)
            idx_max = np.argmax(conv)
            # gradient = reversed a shifted by idx_max
            g = np.zeros_like(a)
            i_start = max(0, idx_max - (n - 1))
            i_end = min(n - 1, idx_max)
            # slice
            idx_i = np.arange(i_start, i_end + 1)
            g[idx_i] = a[idx_max - idx_i]
            # step size schedule
            eta = 0.5 / np.sqrt(iteration * 200 + step + 1)
            a = a * np.exp(-eta * g)
            # renormalize to sum=1
            sum_a = a.sum()
            if sum_a == 0:
                a = np.full(n, 1.0/n)
            else:
                a /= sum_a
        # compute fast_score
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{iteration}] Improved score {best_score:.6f}")
        # occasional hillclimb for refinement
        if iteration % 5 == 0:
            # hill climb with more aggressive moves
            best_a, best_score = hill_climb(best_a, best_score, fft_len, n,
                                            min(deadline, time.time() + 0.5),
                                            max_iter=5000)
        iteration += 1
    # final L-BFGS from best (optional)
    best_y = np.log(best_a + 1e-12)
    lam_schedule = list(np.geomspace(0.3, 1e-4, num=20))
    for lam in lam_schedule:
        if time.time() >= deadline: break
        res = minimize(fun=objective_and_grad,
                       x0=best_y,
                       args=(n, fft_len, fft_len, lam),
                       method='L-BFGS-B',

        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final L-BFGS] lam={lam:.5g} score {best_score:.6f}")
    # final hill climb with remaining time
    if time.time() < deadline:

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)             # already exists

    return final_seq

"""
Improved step‑function search.

The algorithm follows the blueprint of the previous solver but
adds a more aggressive optimisation schedule:

* a longer geometric lam‑factor schedule (down to 1e‑5) and more
  L‑BFGS iterations per stage;
* a simple exponentiated‑gradient (EG) sub‑gradient loop that
  directly attacks the currently maximal convolution entry;
* more intensive hill‑climbing (larger moves, more iterations);
* larger problem size (n≈8000) – a longer vector gives the optimizer
  more freedom to flatten the auto‑convolution.

All helper functions are top‑level; the entry point is
``propose_candidate`` which returns a list of non‑negative floats
ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t.
    the unconstrained vector y (soft‑max parametrisation).

    Parameters
    ----------
    y : ndarray
        Unconstrained parameters.
    n : int
        Length of the probability vector.
    fft_len_conv, fft_len_g : int
        FFT lengths for the forward and gradient convolutions.
    lam_factor : float
        Multiplier that determines the smoothing parameter:
        lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # ----- a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth‑max of the convolution -----
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth‑max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}
    # Convolution of a (reversed) with w.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous stage.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set (pairwise sums distinct) – size ≈ sqrt(n)
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction (up to 10 %) of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score
    (the V function defined in the harness).  The routine runs for
    ``budget_s`` seconds (default 1000) and returns the best list it
    found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    # a larger n gives more freedom; 8000 works comfortably within the time budget
    n = 8000
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 1000:
                n = max(n, int(tmp.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # start from a uniform distribution (baseline)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # sum = 1
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    # aggressive lam‑factor schedule (smooth‑max becomes sharper)
    lam_schedule = list(np.geomspace(0.6, 1e-5, num=28))
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    iteration = 0

    # --------------------------------------------------------------
    # main search loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ----- continuation L‑BFGS (smooth approximation) -----
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,            # a bit more than the previous code
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- exponentiated‑gradient sub‑gradient loop -----
        # a few quick EG steps to polish the current best
        a = best_a.copy()
        for _ in range(30):
            conv = conv_fft(a, fft_len_conv, n)
            s_max = int(np.argmax(conv))
            # gradient = a shifted and reversed according to s_max
            g = np.zeros_like(a)
            i_start = max(0, s_max - (n - 1))
            i_end = min(n - 1, s_max)
            # slice indices
            idx = np.arange(i_start, i_end + 1)
            g[idx] = a[s_max - idx]
            # step size decays with the iteration number
            eta = 0.5 / np.sqrt(iteration * 30 + _ + 1)
            a = a * np.exp(-eta * g)
            s = a.sum()
            if s == 0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s
        eg_score = fast_score(a, fft_len_conv, n)
        if eg_score < best_score:
            best_score = eg_score
            best_a = a.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] EG refinement improved fast score = {best_score:.6f}")

        # ----- aggressive hill‑climb -----
        # give it a short burst of greedy moves
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final refinement: very small lam‑factor and more iterations
    # --------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-6, num=22))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final greedy polishing (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
