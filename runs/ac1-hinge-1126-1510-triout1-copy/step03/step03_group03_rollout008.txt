def refine_sequence(a_initial, time_budget):
   a = a_initial.copy()
   n = len(a)
   fft_len = next_pow2(2*n)
   best = a.copy()
   best_score = fast_score(a)
   e = 0.001   # tolerance for active indices
   eta0 = 2.0 # base step
   start = time.time()
   iteration = 0
   while time.time() < start + time_budget:
      # compute conv
      fa = np.fft.rfft(a, n=fft_len)
      conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
      M = conv.max()
      # determine active set
      tol = 1e-6  # absolute tolerance
      active = np.where(conv >= M - tol)[0]  # might be many; but keep threshold relative too
      # compute gradient as sum over active
      g = np.zeros_like(a)
      for k in active:
          i_start = max(0, k - (n - 1))
          i_end = min(n - 1, k)
          i_range = np.arange(i_start, i_end + 1)
          j_range = k - i_range
          g[i_range] += 2.0 * a[j_range]
      # step size
      iteration += 1
      eta = eta0 / np.sqrt(iteration)  # decays
      # multiplicative update
      a_new = a * np.exp(-eta * g)
      # renormalize
      s = a_new.sum()
      if s == 0:
          a_new = np.full_like(a, 1.0 / n)
      else:
          a_new = a_new / s
      # evaluate
      cur_score = fast_score(a_new)
      if cur_score < best_score - 1e-9:
          best_score = cur_score
          best = a_new.copy()
          print(f"[{time.time()-start:.1f}s] iter {iteration}, improved score {best_score:.6f}")
      # accept always? maybe using cooling:
      if cur_score <= best_score + 1e-6:  # if not too bad
          a = a_new
      else:
          # maybe keep old a
          pass
      # occasional jitter
      if iteration % 2000 == 0:
          noise = np.random.randn(n) * 0.005 * a
          a = np.clip(a + noise, 0, None)
          a = a / a.sum()
   return best

def refine_seq(a, max_time, start_time):
    n = a.shape[0]
    fft_len = next_pow2(2*n)
    best = a.copy()
    best_score = fast_score(a)
    iteration = 0
    eta0 = 5.0
    tau = 2000.0
    jitter_interval = 3000
    jitter_scale = 0.005
    while time.time() < start_time + max_time:
        iteration += 1
        # compute full conv
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2*n - 1]
        M = conv_full.max()
        # Determine active set: all indices where conv >= M - eps
        # relative tolerance
        eps = max(1e-12, M * 1e-6)  # may be small
        active_idx = np.where(conv_full >= M - eps)[0]
        # Compute subgradient sum over active
        g = np.zeros_like(a)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
            j_range = k - i_range
            # add contribution
            np.add.at(g, i_range, 2.0 * a[j_range])
        # step size
        eta = eta0 / (1.0 + iteration / tau)
        # multiplicative update
        a_new = a * np.exp(-eta * g)
        # Projection to simplex (renormalization)
        s = a_new.sum()
        if s <= 0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new = a_new / s
        # jitter occasionally
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a_new
            a_new = np.clip(a_new + noise, 0.0, None)
            s = a_new.sum()
            if s <= 0:
                a_new = np.full_like(a, 1.0 / n)
            else:
                a_new = a_new / s
        # monitor improvement periodically
        if iteration % 500 == 0:
            cur_score = fast_score(a_new)
            if cur_score < best_score:
                best_score = cur_score
                best = a_new.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:.1f}s] iter {iteration}, new best {best_score:.6f}")
        a = a_new

The algorithm works in three stages:

   * compute the self‑convolution,
   * collect the *active* indices where the convolution is (almost) maximal,
   * build the sub‑gradient   g_i = Σ_{k∈active} 2·a_{k‑i},
   * perform a multiplicative update a_i ← a_i·exp(‑η·g_i) and renormalise
     onto the probability simplex.
   The step size η decays as η₀/(1+t/τ).  Small stochastic jitter is added
   periodically to escape plateaus.  The best sequence seen during refinement
   is stored.

3. **Final verification** – the returned list is evaluated with the
   official `evaluate_sequence` function and printed.

The whole routine respects the 1000 s budget, never returns an empty or
negative list, and works on any length up to a few × 10⁴ (FFT‑based
convolutions are O(n log n)).  It also re‑uses a previously discovered
construction if the global variable `height_sequence_1` is present.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT‑based convolution.  The sequence is assumed to be
    non‑negative and already normalised to sum=1 (the normalisation
    cancels out of the ratio, but we keep it for numerical stability).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    Fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    {x | x_i ≥ 0, Σ x_i = 1 }.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families used during the random search
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """
    Coefficients of (1‑z)^{-1/2}: a_0 = 1,
    a_i = a_{i‑1}·(2i‑1)/(2i).  Behaviour ∼ 1/√i gives a very flat
    self‑convolution.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann (cosine) window, non‑negative."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the window."""
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_like(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) density on {0,…,n‑1} (mid‑point rule).
    α,β > 0.  For α,β < 1 the distribution diverges at the ends and
    typically yields a flatter self‑convolution.
    """
    # mid‑point x in (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    seq = np.maximum(seq, 0.0)
    return seq


# ----------------------------------------------------------------------
# Active‑set exponentiated‑gradient refinement
# ----------------------------------------------------------------------


def refine_active_subgradient(
    a0: np.ndarray,
    time_budget: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Refine ``a0`` (already normalised to sum=1) by minimizing the peak
    of its self‑convolution using an exponentiated‑gradient scheme with
    an active set of binding inequalities.

    Parameters
    ----------
    a0 : np.ndarray
        Initial non‑negative vector (sum == 1).
    time_budget : float
        Number of seconds allocated for this refinement.
    start_time : float
        Wall‑clock time when the whole optimisation started (used only
        for printing).

    Returns
    -------
    best_seq : np.ndarray
        The best sequence seen during the refinement.
    best_score : float
        Its fast_score value.
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = next_pow2(2 * n)

    best_seq = a.copy()
    best_score = fast_score(a)

    iteration = 0
    eta0 = 5.0          # initial step‑size factor
    tau = 2500.0        # decay rate for η
    jitter_interval = 3000
    jitter_scale = 0.004
    eval_interval = 800

    deadline = start_time + time_budget

    while time.time() < deadline:
        iteration += 1

        # ---------- self‑convolution ----------
        Fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        M = conv_full.max()

        # ---------- active set (indices where conv is essentially maximal) ----------
        # We take a very tight tolerance to keep the active set small.
        eps = max(1e-12, M * 1e-9)
        active_idx = np.where(conv_full >= M - eps)[0]

        # ---------- sub‑gradient (sum over the active indices) ----------
        g = np.zeros_like(a)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int32)
            j_range = k - i_range
            # contribution 2·a_{j}
            np.add.at(g, i_range, 2.0 * a[j_range])

        # ---------- multiplicative update ----------
        eta = eta0 / (1.0 + iteration / tau)
        a = a * np.exp(-eta * g)

        # ---------- normalise onto the simplex ----------
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ---------- occasional jitter ----------
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = np.clip(a + noise, 0.0, None)
            s = a.sum()
            if s <= 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

        # ---------- periodic evaluation ----------
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:.1f}s] iter {iteration}, new best {cur_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list:
    """
    Search for a low‑score height sequence under a fixed time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5        # tiny safety margin

    # --------------------------------------------------------------
    # 0️⃣  Initialise with a known construction (if available)
    # --------------------------------------------------------------
    init_candidates = []

    try:
        seq0 = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if seq0.ndim == 1 and seq0.size > 0:
            seq0 = np.maximum(seq0, 0.0)
            if seq0.sum() > 0:
                seq0 = seq0 / seq0.sum()
                init_candidates.append(seq0)
    except Exception:
        pass

    # fallback – a moderate‑size arcsine‑like sequence
    fallback = generate_arcsine_like(8000)
    fallback = fallback / fallback.sum()
    init_candidates.append(fallback)

    # --------------------------------------------------------------
    # 1️⃣  Broad random exploration (≈ 45 % of the total budget)
    # --------------------------------------------------------------
    random_time = budget_s * 0.45
    random_deadline = start + random_time

    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = [
        "power",
        "exp",
        "twosided",
        "arcsine",
        "cosine",
        "gaussian",
        "beta",
    ]

    best_seq = None
    best_score = float("inf")

    # also consider the initial seeds
    for cand in init_candidates:
        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[init] score={sc:.6f}, length={cand.size}")

    while time.time() < random_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.2, 5.0)
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-5, 5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.2, 5.0)
            pr = random.uniform(0.2, 5.0)
            wl = random.uniform(0.2, 5.0)
            wr = random.uniform(0.2, 5.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_cosine_window(n)

        elif family == "gaussian":
            sigma = random.uniform(0.05, 0.25) * n
            cand = generate_gaussian_window(n, sigma)

        elif family == "beta":
            alpha = random.uniform(0.1, 2.0)
            beta = random.uniform(0.1, 2.0)
            cand = generate_beta_like(n, alpha, beta)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            elapsed = time.time() - start
            print(
                f"[{elapsed:.1f}s] new random best ({family}) n={n} score={sc:.6f}"
            )

    # --------------------------------------------------------------
    # 2️⃣  Refinement with active‑set exponentiated gradient
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        print(
            f"Starting refinement (remaining budget ≈ {remaining:.1f}s), best score so far {best_score:.6f}"
        )
        refined_seq, refined_score = refine_active_subgradient(
            best_seq, remaining, start
        )
        if refined_score < best_score:
            best_seq = refined_seq.copy()
            best_score = refined_score
            print(
                f"Refinement finished, improved score to {best_score:.6f}"
            )
        else:
            print(
                f"Refinement finished, best score unchanged ({best_score:.6f})"
            )
    else:
        print("Not enough time left for refinement – returning current best.")

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Final score reported by evaluate_sequence = {final_score:.6f}")

    return final_seq
