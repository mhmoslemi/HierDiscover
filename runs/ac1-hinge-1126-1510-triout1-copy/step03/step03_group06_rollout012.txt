"""
Hybrid optimisation:
  1. Smooth Adam‑based descent on a soft‑max approximation of the
     self‑convolution peak (log‑sum‑exp smoothing).  The smoothing
     parameter λ is increased from a low to a moderate value, allowing
     the optimizer to see a well‑behaved objective and to escape shallow
     local minima.
  2. A short projected sub‑gradient phase that directly attacks the
     active peak of the exact convolution.
  3. Fast stochastic “mass‑transfer” local refinement (simulated‑annealing
     style).

The routine works on a number of candidate lengths (≈2000 – 8000),
restarts with various initial shapes (uniform, symmetric power‑law,
sinusoidal), and keeps the best sequence seen before the time budget
expires.  All vectors are kept on the probability simplex
(a_i ≥ 0, Σ a_i = 1) so that the evaluation score equals
   2·n·max(conv(a,a))
and is invariant to scaling.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def compute_full_conv(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1) using FFT."""
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv


def compute_max_conv(a: np.ndarray) -> float:
    """Return the maximal entry of the linear self‑convolution."""
    conv = compute_full_conv(a)
    return float(conv.max())


# ----------------------------------------------------------------------
# Gradient of a soft‑max (log‑sum‑exp) surrogate of the peak
# ----------------------------------------------------------------------
def gradient_smooth(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Gradient of  s(a)= (1/λ) log Σ_i exp(λ·conv_i)  w.r.t. a.
    p is the soft‑max vector (Σ p_i = 1, p_i ≥ 0).
    The formula is  g_j = 2 * Σ_i p_i * a_{i‑j},
    which we obtain via one FFT‑based convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    L = 1 << ((len(p) + n - 1).bit_length())
    A = np.fft.rfft(a_rev, n=L)
    P = np.fft.rfft(p, n=L)
    conv = np.fft.irfft(A * P, n=L)                     # length L
    start = n - 1
    g = 2.0 * conv[start:start + n]                     # extract g_0…g_{n‑1}
    return g


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def smooth_adam_opt(
    a0: np.ndarray,
    max_iters: int,
    lr: float,
    lambda0: float,
    lambda_max: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth surrogate.
    λ grows linearly from λ₀ to λ_max during the run.
    Returns the best sequence (on the simplex) and its true score.
    """
    a = a0.copy()
    a = a / a.sum()                     # enforce Σ a = 1
    n = a.shape[0]

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- current convolution & soft‑max weights -----
        conv = compute_full_conv(a)                     # length 2·n‑1
        lam = lambda0 + (lambda_max - lambda0) * (it - 1) / max_iters
        # stable soft‑max
        shift = conv - conv.max()
        w = np.exp(lam * shift)
        p = w / w.sum()

        # ----- Adam step on the smooth surrogate -----
        grad = gradient_smooth(a, p)                    # shape (n,)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ----- back to the simplex -----
        a = project_simplex(a)

        # ----- occasional true‑score evaluation -----
        if it % 50 == 0 or it == max_iters:
            conv_max = conv.max()
            score = 2.0 * n * conv_max / (a.sum() ** 2)   # a.sum()==1
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[smooth n={n} it={it}] new best={score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact peak (as in the baseline)
# ----------------------------------------------------------------------
def subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent on the true objective
    f(a)=2·n·max(conv(a,a)).  The step size decays as step0/√t.
    """
    a = a0.copy()
    n = a.shape[0]
    a = a / a.sum()                       # enforce Σa = 1

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, max_b = compute_full_conv(a), None      # compute full conv
        conv = compute_full_conv(a)
        max_b = conv.max()
        score = 2.0 * n * max_b                     # Σa = 1

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[subgrad n={n} it={it}] new best={score:.6f}")

        # ----- sub‑gradient of the active peak -----
        k = int(np.argmax(conv))                     # index of the peak
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]                 # ∂conv_k/∂a_i = 2 a_{k-i}

        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """Cheap random walk that moves a small amount of mass between two coordinates."""
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of transfer
        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)

        # up to 5 % of source mass
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_score
