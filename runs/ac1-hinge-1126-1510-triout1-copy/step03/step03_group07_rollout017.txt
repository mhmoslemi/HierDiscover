"""
Propose a step‑function that (approximately) flattens the self‑convolution
distribution.  The optimisation is performed in three stages:

1.  **Rich parametric sweep** – a large collection of analytic windows
    (power‑law, exponential, Beta, arcsine, triangular, Hann, Hamming,
    Blackman and a few Gaussian / Kaiser windows) is normalised to the
    simplex and the best candidate (smallest max‑convolution) is kept.

2.  **Adam on the log‑itself (exponential‑gradient)** – the variables are
    stored as logits `z` and the distribution is `a = softmax(z)`.  The
    objective is `max(conv(a,a))`.  A smooth approximation
    `softmax_weights(conv, α)` is used at first (α grows from 2 to 300);
    the gradient w.r.t. the logits is obtained from the sub‑gradient of
    the smoothed max via FFT‑based correlation.  Adam updates are applied
    on `z`.  The loop runs as long as time permits (up to a few ×10⁵
    iterations).

3.  **Simulated‑annealing local moves** – when the gradient phase ends,
    tiny mass transfers between random coordinates are attempted.  Moves
    that improve the score are always accepted, otherwise they are accepted
    with probability `exp(-(Δscore)/T)` where the temperature T cools
    linearly to a very small value.

The final sequence is scaled to the required ℓ₁‑norm `√(2n)` and returned.
All operations respect the 1000 s budget and use only NumPy (SciPy is
optional for the Kaiser window).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution a*a (real part) truncated to length 2·len(a)-1."""
    fft_a = np.fft.fft(a, n=L)
    conv_full = np.real(np.fft.ifft(fft_a * fft_a))
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient_from_weights(a: np.ndarray, weights: np.ndarray,
                          phase: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of  <weights, a*a>  w.r.t. a.
    Implements  2·correlation(weights, a)  using FFTs.
    """
    n = a.shape[0]
    fft_a = np.fft.fft(a, n=L)
    fft_a_rev = np.conj(fft_a) * phase          # FFT of reversed a (zero‑padded)
    fft_w = np.fft.fft(weights, n=L)
    corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
    return 2.0 * corr[n - 1 : n - 1 + n]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights for the smoothed max."""
    scaled = alpha * conv
    shift = scaled.max()
    exp_scaled = np.exp(scaled - shift)
    return exp_scaled / exp_scaled.sum()


def _submax_weights(conv: np.ndarray) -> np.ndarray:
    """Uniform weights on the active set of max(conv)."""
    max_val = conv.max()
    tol = max_val * 1e-12 + 1e-15
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv, dtype=np.float64)
    if active.size:
        w[active] = 1.0 / active.size
    else:   # very unlikely
        w[conv.argmax()] = 1.0
    return w

    """Search for a non‑negative step function with a low evaluation score."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1. Choose a working length.
    # ------------------------------------------------------------------
    default_n = 10000
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # may exist
        n0 = len(init_seq)
        if 2000 <= n0 <= 20000:
            n = n0
            init_seq = init_seq / init_seq.sum()
        else:
            n = default_n
            init_seq = None
    except Exception:
        n = default_n
        init_seq = None

    # ------------------------------------------------------------------
    # 2. FFT helpers (common for all phases).
    # ------------------------------------------------------------------
    L = 1 << ((2 * n - 1).bit_length())       # smallest power of two ≥ 2n‑1
    phase = np.exp(2j * np.pi * np.arange(L) / L)

    # ------------------------------------------------------------------
    # 3. Keep the best candidate discovered so far.
    # ------------------------------------------------------------------
    best_a = None
    best_max = np.inf                         # max(conv) for the L1‑normalised a

    def _consider(cand: np.ndarray, tag: str = ""):
        """Normalise cand, evaluate and possibly replace the current best."""
        nonlocal best_a, best_max
        if cand.ndim != 1:
            return
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            return
        cand = cand / cand.sum()
        conv = _compute_convolution(cand, L)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = cand.copy()
            print(f"New best {tag:>12}: max={cur_max:.8e}  score={2*n*cur_max:.6f}")
        return

    # ------------------------------------------------------------------
    # 4. Parametric sweep (rich family of analytic windows).
    # ------------------------------------------------------------------
    print(f"--- Phase 0: parametric sweep (n={n}) ---")
    xs = np.arange(n, dtype=np.float64)

    # power‑law (including reversed)
    for p in np.linspace(0.0, 3.0, 61):
        if time.time() > deadline:
            break
        base = (xs + 1) ** p
        _consider(base, f"power p={p:.2f}")
        _consider(base[::-1], f"rev‑p p={p:.2f}")

    # exponential (including reversed)
    for lam in np.linspace(0.0, 6.0, 61):
        if time.time() > deadline:

        exp_shape = np.exp(-lam * xs)

        if time.time() > deadline:

        left = np.exp(-lam * xs)
        right = left[::-1]
        u_shape = left + right
        _consider(u_shape, f"U‑shape λ={lam:.2f}")

    # discrete Beta(a,b) (and its mirror)

    betas = [0.5, 1.0, 2.0, 5.0]
    x_mid = (xs + 0.5) / n
    for a in alphas:
        for b in betas:

            beta_shape = (x_mid ** (a - 1)) * ((1.0 - x_mid) ** (b - 1))
            if np.all(beta_shape > 0):

                _consider(beta_shape[::-1], f"rev‑Beta a={a} b={b}")

    # arcsine‑like shape
    arcsine = 1.0 / np.sqrt(x_mid * (1.0 - x_mid) + 1e-30)
    _consider(arcsine, "arcsine")
    _consider(arcsine[::-1], "rev‑arcsine")

    # triangular (zero at ends, peak in centre)
    tri = (xs + 1) * (n - xs)
    _consider(tri, "triangular")
    _consider(tri[::-1], "rev‑triangular")

    # Hann window (raised cosine, zero at ends)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * xs / (n - 1))
    _consider(hann, "hann")
    _consider(hann[::-1], "rev‑hann")

    # Hamming window
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * xs / (n - 1))
    _consider(hamming, "hamming")
    _consider(hamming[::-1], "rev‑hamming")

    # Blackman window
    blackman = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * xs / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * xs / (n - 1))
    )
    _consider(blackman, "blackman")
    _consider(blackman[::-1], "rev‑blackman")

    # Gaussian bumps centred in the middle (different widths)
    centre = (n - 1) / 2.0
    for sigma_frac in [0.1, 0.15, 0.2, 0.25]:
        if time.time() > deadline:
            break
        sigma = sigma_frac * n
        gau = np.exp(-0.5 * ((xs - centre) / sigma) ** 2)
        _consider(gau, f"gauss σ={sigma_frac:.2f}")

    # Optional: Kaiser window (requires scipy, but we keep it guarded)
    try:
        from scipy.special import i0

        for beta in [2.0, 5.0, 8.0, 12.0]:
            if time.time() > deadline:
                break
            t = (2.0 * xs) / (n - 1) - 1.0
            kaiser = i0(beta * np.sqrt(1.0 - t ** 2)) / i0(beta)
            _consider(kaiser, f"kaiser β={beta:.1f}")
            _consider(kaiser[::-1], f"rev‑kaiser β={beta:.1f}")
    except Exception:
        pass  # scipy not available – just skip

    # Fallback if nothing survived (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _compute_convolution(best_a, L).max()

    # ------------------------------------------------------------------
    # 5. Adam optimisation on logits (smooth‑max → true max).
    # ------------------------------------------------------------------
    print("--- Phase 1: Adam on logits with smoothed max ---")
    # initialise logits from the best candidate
    eps = 1e-12
    z = np.log(best_a + eps)      # logits
    m = np.zeros_like(z)
    v = np.zeros_like(z)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr0 = 0.20                    # base learning rate for logits

    it = 0
    alpha_start, alpha_end = 2.0, 300.0

    # allow up to ~200 000 iterations or until time runs out
    max_iters = 200_000

    while time.time() < deadline and it < max_iters:
        it += 1

        # softmax → probability vector a (always sums to 1)
        exp_z = np.exp(z - np.max(z))   # shift for numerical stability
        a = exp_z / exp_z.sum()

        # convolution and smooth‑max weights
        conv = _compute_convolution(a, L)

        # update best solution if we improved
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_a = a.copy()
            print(
                f"  Adam it {it:6d} improved max={cur_max:.8e} "
                f"score={2*n*cur_max:.6f}"
            )

        # schedule for smoothing parameter α
        frac = it / max_iters
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # weights for the smoothed objective
        w = _softmax_weights(conv, alpha)

        # gradient w.r.t. a
        grad_a = _gradient_from_weights(a, w, phase, L)

        # transform gradient to logits space (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps_adam)

        z -= step

    # ------------------------------------------------------------------
    # 6. Simulated‑annealing local random moves (tiny mass transfers).
    # ------------------------------------------------------------------
    print("--- Phase 2: Simulated‑annealing local moves ---")
    a_current = best_a.copy()
    cur_max = best_max
    move_cnt = 0
    # temperature linearly decreasing from a fraction of the current max
    temp_start = cur_max * 0.2 + 1e-6
    temp_end = 1e-7
    # we will perform as many moves as time permits
    while time.time() < deadline:
        move_cnt += 1
        # pick source index with probability proportional to its mass
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # transfer up to 3 % of the source mass
        delta = np.random.rand() * 0.03 * a_current[i]
        if delta <= 0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # numerical safety
        if a_new[i] < 0:
            continue
        conv_new = _compute_convolution(a_new, L)
        max_new = conv_new.max()
        # temperature schedule (linear)
        elapsed = time.time() - start
        frac = elapsed / budget_s
        temp = temp_start * (1.0 - frac) + temp_end * frac
        # accept if improvement or with Metropolis probability
        accept = False
        if max_new < cur_max:
            accept = True
        else:
            if np.random.rand() < math.exp(-(max_new - cur_max) / max(temp, 1e-12)):
                accept = True
        if accept:
            a_current = a_new
            cur_max = max_new
            if max_new < best_max:
                best_max = max_new
                best_a = a_new.copy()
                print(
                    f"  move #{move_cnt:7d} improved max={max_new:.8e} "
                    f"score={2*n*max_new:.6f}"
                )

    # ------------------------------------------------------------------
    # 7. Final scaling to the required ℓ₁‑norm and return.
    # ------------------------------------------------------------------
    target_sum = math.sqrt(2.0 * n)
    final_seq = (best_a * target_sum).tolist()
    # simple logging of the final score (uses the provided evaluate_sequence)
    try:
        final_score = evaluate_sequence(final_seq)
        print(
            f"--- Finished: best score = {final_score:.6f} (n={n}) "
            f"after {it} Adam iters, {move_cnt} local moves ---"
        )
    except Exception:
        # evaluate_sequence may not be present in a dry‑run environment
        pass

    return final_seq
