def subgradient_descent(a, fft_len, n, deadline, max_iters=20000, init_step=0.1):
    best = a.copy()
    best_val = fast_score(best, fft_len, n)
    step = init_step
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute convolution
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))
        max_val = conv[k]
        # compute gradient
        g = np.zeros_like(best)
        # compute indices where k - j in range
        # we can compute slice:
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # for j from low..high
        idx_range = np.arange(low, high + 1)
        g[idx_range] = 2.0 * best[k - idx_range]
        # project gradient to simplex tangent (zero sum)
        g -= np.mean(g)
        # step size as decreasing
        step = init_step / (1 + it * 0.001)  # gradually decrease
        a_new = best - step * g
        # enforce nonnegativity
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = np.full(n, 1.0 / n)
        new_val = fast_score(a_new, fft_len, n)
        if new_val < best_val - 1e-12:
            best = a_new
            best_val = new_val
            # maybe increase step?
        # else reject
        it += 1
    return best, best_val

"""
Improved search for a low‑evaluation step sequence.

The algorithm builds on the previous L‑BFGS + hill‑climb approach
but adds several refinements:

* A deeper continuation schedule (more smoothing stages and more
  L‑BFGS iterations per stage) to obtain a tighter stationary point.
* A deterministic projected sub‑gradient descent that directly
  minimises the true ∞‑norm of the self‑convolution (instead of the
  smooth‑max surrogate).  The gradient of the peak convolution entry
  can be computed in O(n) time, allowing thousands of cheap updates.
* A more aggressive random‑restart strategy and a final long
  L‑BFGS run with an ultra‑fine smoothing schedule.
* All computations are performed on the probability simplex
  (∑a_i = 1); the evaluation score is invariant under scaling, so
  normalising to a simplex is safe.

The search runs until the supplied time budget (≈1000 s) expires,
printing intermediate scores.  The best sequence found is returned
as a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Quick evaluation of 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max version of 2·n·max(a*a) and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained vector (inputs to soft‑max).
    n : int
        Length of the step sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for convolution and gradient.
    lam_factor : float
        Smoothing factor as a fraction of the current maximum
        convolution value.

    Returns
    -------
    f : float
        Smoothed objective value.
    grad_y : np.ndarray
        Gradient of ``f`` w.r.t. ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth‑max -----
    w = exp_shift / Z  # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}  (convolution of a (reversed) with w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations with decreasing smoothing.
    Each stage uses the result of the previous stage as its start point.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial‑point generators (various heuristic shapes)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a pre‑soft‑max vector y for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        # a bit more mass near the two ends of the interval
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set – roughly √n distinct sums
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (cand * 2 not in sums):
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError("unknown mode")
    return y0


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iters: int = 20000,
    init_step: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Deterministic sub‑gradient descent on 2·n·max(a*a).

    The gradient of the current maximal convolution entry k is:
        g_j = 2·a_{k‑j}   for indices where 0 ≤ k‑j < n, else 0.
    After a gradient step we project onto the probability simplex.
    """
    best = a.copy()
    best_val = fast_score(best, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute full convolution (FFT)
        conv = conv_fft(best, fft_len, n)
        k = int(conv.argmax())
        # gradient of the max entry
        g = np.zeros_like(best)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        g[idx] = 2.0 * best[k - idx]

        # remove the component that changes the L1‑norm (project onto tangent)
        g -= g.mean()

        # simple diminishing step size
        step = init_step / (1.0 + it * 0.0005)
        cand = best - step * g

        # projection onto the simplex (non‑negative, sum = 1)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0:
            cand /= s
        else:
            cand = np.full(n, 1.0 / n)

        cand_val = fast_score(cand, fft_len, n)
        if cand_val < best_val - 1e-12:
            best = cand
            best_val = cand_val
        it += 1
    return best, best_val


# ----------------------------------------------------------------------
# Greedy hill‑climb (kept from the previous code – useful as a final polish)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb that moves a small fraction of mass between
    two random positions if it improves the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point (the function invoked by the harness)
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (sum > 0).
    """
    rng = np.random.default_rng(seed)
    # a little safety margin so that the harness does not kill us prematurely
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # problem size – choose a reasonably large n (more DOF → better optimum)
    # ------------------------------------------------------------------
    n = 8000
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 500:
                n = max(n, int(tmp.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise with the best known construction (if any) or uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_arr = np.asarray(globals()["height_sequence_1"], dtype=float)
        if init_arr.ndim == 1:
            best_a = init_arr.copy()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    # Normalise to a probability vector (scaling does not change the score)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a /= best_a.sum()

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)  # representation in soft‑max space
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # a relatively fine continuation schedule for the L‑BFGS phase
    lam_schedule = list(np.geomspace(0.8, 0.001, num=20))
    maxiter_stage = 500  # more iterations per stage than before

    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]

    # ------------------------------------------------------------------
    # Main search loop (random restarts + L‑BFGS + sub‑gradient fine‑tune)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ----- continuation L‑BFGS (smooth‑max) -----
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        # ----- deterministic sub‑gradient descent (real max) -----
        sub_deadline = min(deadline, time.time() + 0.6)
        a_opt, score_opt = subgradient_descent(
            a_opt,
            fft_len_conv,
            n,
            sub_deadline,
            max_iters=5000,
            init_step=0.12,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- short greedy hill‑climb for polishing -----
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final refinement: an ultra‑fine continuation schedule + longer sub‑grad
    # ------------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.4, 5e-5, num=30))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.5g} fast score = {best_score:.6f}")

    # One more round of sub‑gradient descent (more iterations, tiny step)
    sub_deadline = deadline
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        sub_deadline,
        max_iters=15000,
        init_step=0.08,
    )
    best_y = np.log(best_a + 1e-12)

    # Final greedy polish if there is any time left
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation (the harness will also call evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
