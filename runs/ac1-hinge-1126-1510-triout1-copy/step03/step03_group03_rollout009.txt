"""
Propose a candidate step‑function (non‑negative heights) that tries to
minimise 2 n·max(conv(a,a)) / (sum(a)**2).

The algorithm works in two phases:

1️⃣ **Broad random search** – a few thousand candidates are drawn from a
   variety of analytic families (power‑law, exponential, two‑sided,
   arcsine‑like, Hann‑window and Gaussian windows) with several lengths.
   The best candidate (lowest fast_score) is kept.

2️⃣ **Smoothed‑max mirror descent** – starting from the best candidate we
   perform a projected mirror‑descent on the simplex.  The objective
   (the maximum of the self‑convolution) is approximated by the
   smooth log‑sum‑exp:  

        F(a) = (1/λ)·log∑_k exp(λ·(a*a)_k).

   For a given λ the gradient is

        ∂F/∂a_i = 2·∑_k p_k·a_{k‑i},   where
        p_k = exp(λ·(a*a)_k) / ∑_j exp(λ·(a*a)_j).

   The gradient is obtained efficiently by an FFT‑based correlation.
   An exponential (multiplicative) update
        a_i ← a_i·exp(−η·∂F/∂a_i)
   keeps the variables non‑negative; a final normalisation projects onto
   the probability simplex.  The temperature λ is increased slowly so
   that the smooth approximation approaches the true max, while the step
   size η decays.  Periodically we evaluate the true fast_score and keep
   the best sequence seen.

The whole routine respects the 1000 s budget and returns a plain Python
list ready for `evaluate_sequence`.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using FFT based convolution.  The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x | x_i ≥ 0, Σx_i = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sequence generators (families)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Discrete approximation of (1‑z)^{-1/2} – behaves like 1/√i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_hann_window(n: int) -> np.ndarray:
    """Hann (cosine‑squared) window."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Discrete Gaussian centred in the array, sigma = sigma_frac·n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_sym_power_law(n: int, p: float) -> np.ndarray:
    """Symmetric two‑sided power‑law: a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (-p)
    right = (n - i + 1) ** (-p)
    return left + right


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a low‑score height sequence under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    safety = 0.5                 # leave a small safety margin for the caller
    deadline = start + budget_s - safety

    # ------------------------------------------------------------------
    # 0️⃣  Load previous construction if available, otherwise start from
    #     an arcsine‑like sequence of moderate length.
    # ------------------------------------------------------------------
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:
            raise ValueError
        n_init = init.size
    except Exception:
        n_init = 8000
        init = generate_arcsine_like(n_init)

    init = np.maximum(init, 0.0)
    if init.sum() == 0:
        init = np.full(n_init, 1.0 / n_init, dtype=np.float64)
    else:
        init = init / init.sum()

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Broad random exploration (≈20 % of the total budget)
    # ------------------------------------------------------------------
    random_phase_end = start + 0.20 * budget_s
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = [
        "power",
        "exp",
        "twosided",
        "arcsine",
        "cosine",
        "gaussian",
        "sym_power",
    ]

    while time.time() < random_phase_end:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.30, 0.80)
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(5e-4, 2.5e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.30, 0.80)
            pr = random.uniform(0.30, 0.80)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_hann_window(n)

        elif family == "gaussian":
            sigma_frac = random.uniform(0.07, 0.30)
            cand = generate_gaussian_window(n, sigma_frac)

        elif family == "sym_power":
            p = random.uniform(0.30, 0.80)
            cand = generate_sym_power_law(n, p)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start:.1f}s] new random best ({family}) n={n} score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed‑max mirror‑descent refinement (remaining budget)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # FFT length sufficient for both the 2‑n convolution and the
    # correlation (a has length n, weight has length 2n‑1).
    nfft = next_pow2(3 * n)

    iteration = 0
    # step‑size schedule
    eta0 = 0.8
    eta_decay = 5000.0
    # temperature (λ) schedule
    lam0 = 0.5
    lam_max = 5000.0
    lam_ramp = 20000.0   # number of iterations over which λ grows

    eval_interval = 500   # evaluate the true fast_score this often
    jitter_interval = 5000
    jitter_scale = 0.005

    while time.time() < deadline:
        iteration += 1

        # ----- 1) self‑convolution (full) -----
        Fa = np.fft.rfft(a, n=nfft)
        conv_full = np.fft.irfft(Fa * Fa, n=nfft)[: 2 * n - 1]

        # ----- 2) build smoothed max weights -----
        lam = lam0 + (lam_max - lam0) * min(1.0, iteration / lam_ramp)
        max_conv = conv_full.max()
        # Use log‑sum‑exp trick for numerical stability
        exp_arg = lam * (conv_full - max_conv)
        # Clip to avoid overflow on the positive side (should never happen)
        exp_arg = np.clip(exp_arg, -700, 700)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 3) gradient = 2·(a ⋆ w) -----
        Fw = np.fft.rfft(w, n=nfft)
        corr_full = np.fft.irfft(Fa * np.conj(Fw), n=nfft)[:n]
        g = 2.0 * corr_full.real

        # ----- 4) multiplicative (exponential) mirror update -----
        eta = eta0 / (1.0 + iteration / eta_decay)
        # Clamp the exponent to avoid overflow/underflow
        update_factor = np.exp(np.clip(-eta * g, -700, 700))
        a = a * update_factor

        # ----- 5) projection onto the simplex -----
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- 6) occasional jitter to escape flat regions -----
        if iteration % jitter_interval == 0:
            noise = np.random.randn(n) * jitter_scale * a
            a = a + noise
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum <= 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

        # ----- 7) periodic evaluation and bookkeeping -----
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start:.1f}s] iter {iteration:6d}, new best score={cur_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    # The evaluation function is assumed to be already imported.
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
