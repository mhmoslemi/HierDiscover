def exp_gradient_descent(a0, max_iters, lr0, beta0, beta_inc, deadline):
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_score = float('inf')
    lr = lr0
    beta = beta0
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # compute convolution and max
        conv = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(a, L), n=2*n-1)[:2*n-1]
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # smoothing
        # compute weighted softmax over conv
        w = np.exp(beta * (conv - conv.max()))  # subtract max for stability
        w /= w.sum()
        # gradient via correlation
        # Use FFT for correlation
        Fa = np.fft.rfft(a, L)
        Fw = np.fft.rfft(w, L)
        corr = np.fft.irfft(np.conj(Fw) * Fa, L)[:2*n-1]
        grad = 2.0 * corr[(n-1)::-1][:n]  # reversed segment length n
        # multiplicative update
        a = a * np.exp(-lr * grad)
        # renormalize
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= s
        # adapt lr, beta
        lr = lr0 / np.sqrt(t)
        beta = min(beta0 + beta_inc * t, 100.0)  # saturate
    return best_a, best_score
```

def exp_gradient_descent(a0, max_iters, lr0, beta0, beta_inc, deadline, L):
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = float('inf')
    lr = lr0
    beta = beta0
    for t in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # compute FFT of a
        A = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(A * A, n=L)[:2*n-1]
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()
        # compute softmax weights
        # To improve numerical stability, subtract max conv
        exp_vals = np.exp(beta * (conv - conv.max()))
        w = exp_vals / exp_vals.sum()
        # compute gradient via correlation
        Fw = np.fft.rfft(w, n=L)
        corr = np.fft.irfft(np.conj(Fw) * A, n=L)[:2*n-1]
        grad = 2.0 * corr[(n-1)::-1][:n]  # reversed segment length n
        # multiplicative update
        # Clip gradient to avoid overflow in exp
        # optionally scale gradient if too large
        upd = np.exp(-lr * grad)

"""Exponentiated‑gradient optimisation with soft‑max smoothing.

The algorithm works on the probability simplex (a_i ≥ 0, Σ a_i = 1) and
minimises the objective

    score = 2 * n * max(conv(a, a))

by repeatedly
* computing the self‑convolution,
* forming a soft‑max approximation of the max (temperature β),
* taking a multiplicative (exponentiated‑gradient) step on the simplex,
* gradually decreasing the step size and increasing β.

After the gradient phase, a cheap simulated‑annealing walk is applied to
polish the solution.  Several problem sizes and random restarts are tried
until the time budget expires.  The best sequence found (or the previously
known one, if it is better) is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear convolution via FFT (full length 2n‑1)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, L: int) -> np.ndarray:
    """Return linear convolution a*a (length 2*len(a)-1) using FFT length L."""
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: gradient of the soft‑max smoothed max‑convolution.
# ----------------------------------------------------------------------
def smoothed_gradient(a: np.ndarray, conv: np.ndarray, beta: float, L: int) -> np.ndarray:
    """
    Given a, its self‑convolution `conv` (length 2n‑1) and a soft‑max temperature β,
    return the gradient of  (1/β)·log Σ exp(β·conv)  w.r.t. a.
    The result has shape (n,).
    """
    n = a.size
    # soft‑max weights (numerically stable)
    shifted = conv - conv.max()
    w = np.exp(beta * shifted)
    w /= w.sum()

    # cross‑correlation w ⋆ a = Σ_k w_k a_{k+i}
    # correlation can be obtained via FFT: corr = ifft(conj(Fw) * Fa)
    Fa = np.fft.rfft(a, n=L)
    Fw = np.fft.rfft(w, n=L)
    corr = np.fft.irfft(np.conj(Fw) * Fa, n=L)[: 2 * n - 1]

    # grad_i = 2 * Σ_k w_k a_{k-i} = 2 * corr[-i]
    # In the linear correlation array, index (n‑1‑i) corresponds to shift -i
    grad = 2.0 * corr[(n - 1) :: -1][:n]
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the simplex with annealed β.
# ----------------------------------------------------------------------
def exp_gradient_descent(
    a0: np.ndarray,
    max_iters: int,
    lr0: float,
    beta0: float,
    beta_inc: float,
    deadline: float,
    L: int,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponentiated) gradient descent on the simplex.
    Returns the best sequence seen and its score (2·n·max(conv)).
    """
    a = a0.copy()
    n = a.size
    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # self‑convolution and current score
        conv = full_convolution(a, L)
        max_b = conv.max()
        score = 2.0 * n * max_b
        if score < best_score:
            best_score = score
            best_seq = a.copy()

        # annealing of β
        beta = min(beta0 + beta_inc * it, 50.0)

        # gradient of the smoothed max
        grad = smoothed_gradient(a, conv, beta, L)

        # multiplicative update, then renormalise to the simplex
        # (clip the exponent to avoid overflow)
        step = lr0 / np.sqrt(it)
        upd = np.exp(-step * grad)
        a = a * upd
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic refinement (as in the reference solution)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * full_convolution(a, 1 << ((2 * n - 1).bit_length())).max()
    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        src, dst = (i, j) if np.random.rand() < 0.5 else (j, i)
        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * full_convolution(new, 1 << ((2 * n - 1).bit_length())).max()
        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_score
