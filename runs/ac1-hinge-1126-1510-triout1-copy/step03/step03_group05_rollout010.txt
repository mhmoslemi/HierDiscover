def compute_conv(a):
    n = a.shape[0]
    L = next_pow2(2*n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]
    return conv

def compute_grad(a, w):
    n = a.shape[0]
    L = next_pow2(3*n - 2)  # length for conv w*a
    fw = np.fft.rfft(w, n=L)
    fa = np.fft.rfft(a, n=L)
    grad_conv = np.fft.irfft(fw * fa, n=L)[:n]  # first n entries correspond to g
    return 2 * grad_conv

def smooth_descent(seq, time_limit, start_time, beta_init=0.5, beta_factor=1.2, eta0=0.1, momentum=0.9):
    a = seq.copy()
    n = a.size
    # projection onto simplex
    a = project_simplex(a)
    # variables for momentum
    v = np.zeros_like(a)
    beta = beta_init
    best_seq = a.copy()
    best_obj = fast_score(a)
    it = 0
    # compute conv for first iteration
    while time.time() < start_time + time_limit:
        it += 1
        # compute convolution
        conv = compute_conv(a)  # length 2n-1
        # smooth max weights
        # exponentiate with stabilization: subtract max conv for numeric stability
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))  # avoid overflow
        w = exp_vals / exp_vals.sum()
        # compute gradient
        grad = compute_grad(a, w)
        # update with Adam or momentum
        eta = eta0 / (np.sqrt(it) + 1)
        v = momentum * v + (1 - momentum) * grad
        cand = a - eta * v
        a = project_simplex(cand)
        # evaluate
        cur_obj = fast_score(a)
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_seq = a.copy()
        # schedule beta
        if it % 200 == 0:
            beta *= beta_factor
            # maybe cap beta
            if beta > 200:
                beta = 200
    return best_seq, best_obj

"""
Propose a step‑function using a *smooth‑max* continuation
and projected (Adam‑style) gradient descent on the probability simplex.

Algorithm
---------
1. **Broad random search** – generate a few hundred candidates from
   simple families (power‑law, exponential, two‑sided power‑law,
   discretised Beta).  Keep the five best candidates (including the
   optional historic `height_sequence_1`).

2. **Smooth‑max descent** – for each elite candidate run a constrained
   optimisation that minimises an *approximation* of the ∞‑norm
   `max(conv(a,a))`.  The smooth approximation is

        smM(a) = (1/β) * log Σ_i exp(β·c_i) ,   c = a*a

   Its gradient is
        ∇ smM = 2 * ( w * a )[0:n] ,          w_i = softmax_i(β·c_i)

   where `*` denotes linear convolution.
   We use an Adam‑style update with a diminishing step size and
   project back onto the simplex after each step.
   The smoothing parameter β is increased geometrically
   (continuation) so that the optimiser first moves in a well‑behaved
   direction and later concentrates on the true maximum.

3. **Time budgeting** – ~30 % of the total wall‑time is spent on the
   random search, the remainder is split equally among the elite
   candidates for refinement.

The routine returns the best sequence found (as a plain Python list)
before the deadline.  All numbers are non‑negative and the sequence is
automatically projected onto the simplex, so the security checks in
`evaluate_sequence` are satisfied.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    using an FFT based convolution.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Shape generators (families used in the random phase)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
    a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Convolution helpers used by the smooth‑max optimiser
# ----------------------------------------------------------------------


def linear_conv(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a∗b using FFT.
    Returned length is len(a)+len(b)-1.
    """
    L = next_pow2(len(a) + len(b) - 1)
    fa = np.fft.rfft(a, n=L)
    fb = np.fft.rfft(b, n=L)
    conv = np.fft.irfft(fa * fb, n=L)
    return conv[: len(a) + len(b) - 1]


def compute_self_conv(a: np.ndarray) -> np.ndarray:
    """Return a∗a (length 2·n‑1)."""
    return linear_conv(a, a)


def compute_grad(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max w.r.t. a:
        g = 2·(w∗a)[0:n]
    where w has length 2·n‑1 (the soft‑max weights over the convolution)
    """
    g_full = linear_conv(w, a)            # length 3·n‑2
    return 2.0 * g_full[: a.shape[0]]


# ----------------------------------------------------------------------
# Smooth‑max descent with Adam‑style momentum
# ----------------------------------------------------------------------


def smooth_descent(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    beta_init: float = 0.5,
    beta_factor: float = 1.2,
    eta0: float = 0.2,
    b1: float = 0.9,
    b2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the simplex, minimising a smooth approximation
    of max(conv(seq,seq)).
    Returns the best sequence seen and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]
    a = project_simplex(a)

    m = np.zeros_like(a)          # 1st moment
    v = np.zeros_like(a)          # 2nd moment
    beta = beta_init
    best_seq = a.copy()
    best_val = fast_score(a)
    it = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        it += 1

        # 1) self‑convolution
        conv = compute_self_conv(a)                 # length 2n‑1
        max_c = conv.max()

        # 2) soft‑max weights (numerically stable)
        shifted = conv - max_c
        exp_w = np.exp(beta * shifted)
        w = exp_w / exp_w.sum()

        # 3) gradient
        g = compute_grad(a, w)                      # shape (n,)

        # 4) Adam update
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * (g * g)
        m_hat = m / (1 - b1 ** it)
        v_hat = v / (1 - b2 ** it)
        eta = eta0 / (np.sqrt(it) + 1.0)
        a_new = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection back onto simplex
        a = project_simplex(a_new)

        # 6) evaluate & keep the best
        cur_val = fast_score(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # 7) increase smoothing gradually
        if it % 200 == 0:
            beta = min(beta * beta_factor, 200.0)

        # 8) occasional random jitter to escape plateaus
        if no_improve >= 4000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact score for the best sequence obtained
    best_val = fast_score(best_seq)
    return best_seq, best_val
