"""
Advanced search for a low‑value step‑function.

The algorithm works in three phases:

0. **Initialisation** – try to reuse a previously discovered sequence,
   otherwise fall back to a modest two‑sided power‑law.

1. **Broad stochastic search** – random shapes from several families
   (power‑law, exponential, two‑sided power‑law, Beta‑pdf, Gaussian,
   triangular, block‑constant) are generated with varying lengths.
   Each shape is normalised to Σaᵢ=1 and scored with a fast FFT‑based
   surrogate `fast_score = 2·n·max(conv)`.  The best few candidates are
   kept in an elite pool.

2. **Smooth‑gradient refinement** – every elite candidate is refined by a
   projected Adam optimiser that minimises
        f(a)=2·n·max(a∗a)                (a lives on the probability simplex).
   The non‑smooth max is approximated by a temperature‑controlled
   soft‑max:
        M_β(a)= (1/β)·log Σₖ exp(β·(a∗a)ₖ).
   Its gradient can be computed in O(n log n) via a second FFT.
   The optimiser uses an adaptive step size, a slowly increasing β
   (continuation), and occasional random perturbations to escape plateaus.
   The best sequence found during refinement replaces the global best.

Finally the best candidate is validated with the provided `evaluate_sequence`
function and returned as a plain Python list.
"""

import time
import random
import numpy as np
import math

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fast_score(seq: np.ndarray) -> float:
    """
    Fast surrogate for `evaluate_sequence` when Σaᵢ = 1:
        2·n·max(conv(seq,seq))
    Uses FFT convolution.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int, p_left: float, p_right: float, w_left: float = 1.0, w_right: float = 1.0
) -> np.ndarray:
    """a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric Gaussian centred in the middle of the sequence."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    return np.exp(-0.5 * ((idx - centre) / sigma) ** 2)


def generate_triangular(n: int) -> np.ndarray:
    """Triangular shape, peak in the centre."""
    i = np.arange(n, dtype=np.float64) + 1
    return np.minimum(i, n - i + 1).astype(np.float64)


def generate_block(n: int, num_blocks: int = 5) -> np.ndarray:
    """Piecewise‑constant blocks with random heights."""
    if num_blocks < 1:
        num_blocks = 1
    # random cut points
    cuts = sorted(random.sample(range(1, n), min(num_blocks - 1, n - 1)))
    cuts = [0] + cuts + [n]
    seq = np.empty(n, dtype=np.float64)
    for a, b in zip(cuts[:-1], cuts[1:]):
        height = random.random()
        seq[a:b] = height
    return seq


# ----------------------------------------------------------------------
# Smooth‑gradient (Adam) refinement on the simplex
# ----------------------------------------------------------------------


def refine_sequence_smooth(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    eta0: float = 0.02,
    beta_start: float = 5.0,
    beta_end: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """
    Refine a candidate sequence using a projected Adam optimiser with a
    soft‑max approximation of the max‑convolution.
    Returns the best sequence seen and its fast_score.
    """
    a = np.asarray(seq, dtype=np.float64).copy()
    n = a.shape[0]
    a = project_simplex(a)                     # enforce Σa=1, a≥0
    best_seq = a.copy()
    best_score = fast_score(best_seq)

    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow = 1.0
    beta2_pow = 1.0

    # FFT lengths used repeatedly
    fft_len_conv = next_pow2(2 * n)            # for a*a
    fft_len_grad = next_pow2(3 * n)            # for p * a_rev

    it = 0
    no_improve = 0

    while time.time() < start_time + time_limit:
        it += 1

        # ----- (1) current convolution a*a -----
        fa = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2 * n - 1]

        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ----- (2) soft‑max of convolution entries -----
        M = conv.max()
        # temperature schedule – linear continuation towards beta_end
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / 20000.0)
        # stable exponentiation
        exp_arg = beta * (conv - M)               # ≤0
        exp_arg = np.clip(exp_arg, -700, 0.0)     # avoid underflow
        p = np.exp(exp_arg)
        p_sum = p.sum()
        if p_sum == 0.0:                          # degenerate case
            p = np.full_like(p, 1.0 / p.size)
        else:
            p /= p_sum

        # ----- (3) gradient of the smoothed max -----
        # grad_soft = 2 * Σ_i a_i p_{i+j}
        a_rev = a[::-1]
        fp = np.fft.rfft(p, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_full = np.fft.irfft(fp * fa_rev, n=fft_len_grad)
        grad_soft = 2.0 * conv_full[n - 1 : n - 1 + n]   # length n
        grad = 2.0 * n * grad_soft                        # ∇f

        # ----- (4) Adam update -----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        eta = eta0 / (np.sqrt(it) + 1.0)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = project_simplex(a)

        # ----- (5) occasional random jitter -----
        if no_improve >= 3000:
            sigma = 0.02 / np.sqrt(n)
            a = project_simplex(a + sigma * np.random.randn(n))
            no_improve = 0

    # final exact score for the best candidate
    best_score = fast_score(best_seq)
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function with a low bound.
    Returns a plain list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin for the final check

    # ---------------------------------------------------------------
    # 0) Starting point – reuse a previously saved sequence if possible
    # ---------------------------------------------------------------
    try:
        init_seq = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init_seq.ndim != 1 or init_seq.size == 0:
            raise ValueError
    except Exception:
        # fallback: modest two‑sided power law
        init_len = 8000
        init_seq = generate_two_sided_power_law(init_len, 0.55, 0.55, 1.0, 1.0)

    init_seq = np.maximum(init_seq, 0.0)
    if init_seq.sum() == 0.0:
        init_seq = np.full_like(init_seq, 1.0 / init_seq.size)
    else:
        init_seq = init_seq / init_seq.sum()          # normalise Σa=1

    best_seq = init_seq.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, score={best_score:.6f}")

    # ---------------------------------------------------------------
    # 1) Broad stochastic search
    # ---------------------------------------------------------------
    length_choices = [4000, 6000, 8000, 10000, 12000, 16000, 20000]
    families = [
        "power",
        "exp",
        "twosided",
        "beta",
        "gaussian",
        "triangular",
        "block",
    ]
    top_k: list[tuple[float, np.ndarray]] = [(best_score, best_seq.copy())]
    top_k_size = 7

    rand_deadline = start + 0.55 * budget_s
    while time.time() < rand_deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.10, 1.20)
            seq = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.10, 1.20)
            pr = random.uniform(0.10, 1.20)
            wl = random.uniform(0.2, 5.0)
            wr = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "beta":
            alpha = random.uniform(0.05, 1.50)
            beta = random.uniform(0.05, 1.50)
            seq = generate_beta_shape(n, alpha, beta)

        elif family == "gaussian":
            sigma = random.uniform(0.1 * n, 0.5 * n)
            seq = generate_gaussian(n, sigma)

        elif family == "triangular":
            seq = generate_triangular(n)

        else:  # "block"
            blk = random.randint(2, 8)
            seq = generate_block(n, blk)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s

        sc = fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[{time.time() - start:.1f}s] new global best ({family}) n={n} score={sc:.6f}")

        # maintain elite pool
        if len(top_k) < top_k_size:
            top_k.append((sc, seq.copy()))
        else:
            worst_idx = max(range(len(top_k)), key=lambda i: top_k[i][0])
            if sc < top_k[worst_idx][0]:
                top_k[worst_idx] = (sc, seq.copy())

    top_k.sort(key=lambda x: x[0])

    # ---------------------------------------------------------------
    # 2) Momentum / Adam refinement on elite candidates
    # ---------------------------------------------------------------
    time_remaining = deadline - time.time()
    if time_remaining <= 0:
        final_seq = best_seq.tolist()
        final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
        print(f"Time exhausted. Final reported score = {final_score:.6f}")
        return final_seq

    per_candidate = max(1.0, time_remaining / len(top_k))

    for idx, (sc, seq) in enumerate(top_k):
        if time.time() > deadline:
            break
        print(
            f"[refine] elite #{idx + 1}/{len(top_k)} (init score={sc:.6f}, len={seq.size})"
        )
        refined_seq, refined_score = refine_sequence_smooth(
            seq, per_candidate, time.time()
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[refine] improved global best → {best_score:.6f}")

    # ---------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # ---------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
