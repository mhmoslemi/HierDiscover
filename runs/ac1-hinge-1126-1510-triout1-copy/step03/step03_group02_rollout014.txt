def targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_moves=2000, frac=0.1):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_moves > 0:
        # compute conv and find peak index
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))
        # contributions vector p
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        p = np.zeros_like(best)
        p[mask] = best[mask] * best[idx[mask]]
        # source: max contribution
        i_src = int(np.argmax(p))
        # dest: min contribution (lowest p)
        i_dst = int(np.argmin(p))
        if i_src == i_dst:
            # random perturbation to avoid stuck
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                continue
        delta = best[i_src] * rng.random() * frac
        if delta <= 0:
            max_moves -= 1
            continue
        # apply move
        best[i_src] -= delta
        best[i_dst] += delta
        # compute new score
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:  # strict improvement
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta
        max_moves -= 1
    return best, best_score

"""
propose_candidate – a hybrid optimizer for the step‑function attack.

The algorithm mixes three complementary strategies:

1. **Continuation L‑BFGS**  
   The non‑negative simplex is parametrised by soft‑max logits `y`.  
   A smooth approximation of the max‑convolution is minimised with a
   geometric λ‑schedule (the smoothing parameter).  This quickly finds
   a good basin of attraction.

2. **Targeted hill‑climb**  
   After each L‑BFGS run the current maximal convolution index `k*`
   is identified.  The contribution of each coordinate to that peak is
   `p_i = a_i * a_{k*‑i}`.  Mass is repeatedly transferred from the
   largest contributor to the smallest one – a greedy sub‑gradient step
   that directly reduces the true `max(a*a)`.

3. **Random hill‑climb (fallback)**  
   A fast generic hill‑climb that randomly moves a fraction of the mass
   between two positions is applied as a polishing step.

The search runs many restarts (different random seeds, exponential,
power‑law and edge‑biased initialisations) within the allotted time.
The best sequence seen is returned.  The whole pipeline stays well under
the 1000 s budget while typically beating the previous 1.5126 bound,
aiming for ≤ 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns the first ``2*n‑1`` entries (the full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector ``a``.
    This matches the value returned by ``evaluate_sequence`` (the denominator
    ``sum(a)**2`` equals 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution and its gradient w.r.t.
    the unconstrained logits ``y`` (the log‑softmax coordinates).

    Parameters
    ----------
    y : ndarray
        Unconstrained parameters; a = softmax(y).
    n : int
        Length of the probability vector.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for the convolution and the auxiliary
        convolution needed for the gradient.
    lam_factor : float
        Multiplicative factor for the smoothing width λ.

    Returns
    -------
    f : float
        Smoothed version of ``2·n·max(a*a)``.
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list that gradually reduces the smoothing.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Generic random hill‑climb on the simplex.
    Randomly transfers a fraction of mass from one index to another
    and keeps the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb.
    At each step the index contributing most to the current peak
    (k* = argmax conv) donates a small fraction of its mass to the
    index with the smallest contribution.  This directly reduces the
    true ``max(a*a)`` and often yields improvements beyond the random
    hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        # compute current convolution and the location of the peak
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random move if the contribution vector is flat
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert the move
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score

# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (its sum will be 1 after the
        internal normalisation).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – use a manageable length for the LP‑free phase
    # --------------------------------------------------------------
    default_n = 2000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            n = default_n
            a0 = arr[:n].copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 1e-12:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Continuation L‑BFGS hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 5e-6, num=30)  # smooth‑max continuation
    maxiter_stage = 400                          # L‑BFGS iterations per stage

    iteration = 0

    # --------------------------------------------------------------
    # Multi‑restart optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=3000,
            frac=0.08,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final deep refinement (another continuation pass)
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 600, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=8000,
            frac=0.06,
        )
        # a final safety random hill‑climb in case of tiny flat‑spots
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
