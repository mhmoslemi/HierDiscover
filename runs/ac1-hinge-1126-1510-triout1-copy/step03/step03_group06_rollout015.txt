def build_convolution_constraint_matrix(f):
    n = len(f)
    row = []
    col = []
    data = []
    for k in range(2*n-1):

            row.append(k)

            data.append(f[i])
    # create csr
    A = scipy.sparse.csr_matrix((data, (row, col)), shape=(2*n-1, n))

"""
Hybrid LP‑driven refinement + sub‑gradient fine‑tuning.

The score is   2·n·max(conv(a,a)) / (sum(a)²).
Because the expression is homogeneous we keep the vector on the
simplex Σa = √(2·n); then the score is simply the peak of the
self‑convolution (M = max(conv)).  Starting from a reasonable
initial shape we repeatedly:

1. Compute the current peak M = max(conv(f,f)).
2. Solve the linear program   max Σ b_j  s.t.  f*b ≤ M,  b ≥ 0 .
   The LP exploits the Toeplitz structure of the convolution matrix
   and is built with a sparse CSR representation.
3. Rescale the LP solution b to the required ℓ₁–norm (√(2·n))
   obtaining a candidate g.
4. Mix the current iterate f with g (convex combination) and
   renormalise – this step provably never increases M.
5. Occasionally run a short projected sub‑gradient descent
   (active‑peak gradient) to polish the sequence.

The loop stops when the time budget expires or when the target
score 1.503 is reached.  The best sequence found is returned as a
plain Python list of floats.
"""

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σ x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # all entries are zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based)
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the full linear convolution a*a and its maximal entry.
    Uses a power‑of‑two FFT length for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper that returns only the maximal convolution entry."""
    _, mx = compute_conv_and_max(a)
    return mx


# ----------------------------------------------------------------------
# Objective / score
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    """Return the evaluation score used by `evaluate_sequence`."""
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    max_b = compute_max_conv(a)
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the simplex (fine‑tuning)
# ----------------------------------------------------------------------
def subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run projected sub‑gradient descent on the simplex (Σa = 1).

    Parameters
    ----------
    a0 : np.ndarray
        Starting point.
    max_iters : int
        Upper bound on iterations.
    step0 : float
        Base step size (scaled by 1/√iteration).
    deadline : float
        Unix timestamp after which the routine must stop.

    Returns
    -------
    best_seq : np.ndarray
        Best sequence (still on the simplex).
    best_score : float
        Corresponding 2·n·max(conv) score (since Σa = 1).
    """
    a = a0.copy()
    n = a.shape[0]
    a = a / a.sum()                     # enforce Σa = 1

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b          # Σa = 1 ⇒ denominator = 1

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            # occasional report
            print(f"[subgrad it {it}] new best={score:.6f}")

        # active peak index
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        # sub‑gradient: grad_i = 2 * a_{k-i} for i in the active window
        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        # diminishing step size
        step = step0 / np.sqrt(it)

        # gradient step + projection
        a = project_simplex(a - step * grad)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (increase ℓ₁‑norm under the same M)
# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   (f * b)_k ≤ M  ∀k,   b ≥ 0.
    The convolution matrix is built in sparse CSR format exploiting
    the Toeplitz structure (each column is a shifted copy of ``f``).
    Returns the optimal b (as a dense ndarray) or ``None`` on failure.
    """
    n = f.shape[0]

    # ----- build the (2n‑1) × n constraint matrix A -----
    # For column j (0‑based) the non‑zero rows are j … j+n‑1,
    # with entry f[i] at row i+j.
    i = np.arange(n, dtype=np.int64)[:, None]      # shape (n,1)
    j = np.arange(n, dtype=np.int64)[None, :]      # shape (1,n)

    rows = (i + j).ravel()                         # length n²
    cols = np.tile(j.ravel(), n)                   # repeat each column index n times
    data = np.repeat(f, n)                         # each f_i repeated across all columns

    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # ----- bounds and objective -----
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)   # (f * b) ≤ M
    c = -np.ones(n, dtype=np.float64)                # minimise –Σb ≡ maximise Σb

    # ----- solve -----
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=(0, None),
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        print("[LP] optimisation failed:", res.message)
        return None

    return res.x

                # early exit if target reached
                if global_best_score < 1.503:
                    print("[global] target reached – terminating early")
                    deadline = time.time()  # force outer loops to stop
                    break

            # ----- decide whether to accept the new iterate -----
            # keep the improvement; otherwise keep previous a (helps stability)
            if score_new <= compute_score(a) * 0.999:
                a = a_new
            else:
                # occasional random jitter to escape shallow traps
                if np.random.rand() < 0.07:
                    jitter = np.random.normal(scale=0.01, size=n)
                    a = np.maximum(a + jitter, 0.0)
                    a = a * (target_sum / a.sum())

            # ----- short sub‑gradient polishing every few LP steps -----
            if lp_iter % 3 == 0:
                time_left = deadline - time.time()
                if time_left > 2.0:
                    # allocate a modest number of sub‑gradient iterations
                    sub_iters = min(4000, int(time_left / 0.0012))
                    seq_sg, _ = subgradient_descent(
                        a / a.sum(),  # work on the simplex
                        max_iters=sub_iters,
                        step0=0.6,
                        deadline=deadline,
                    )
                    # renormalise back to √(2n)
                    seq_sg = seq_sg * (target_sum / seq_sg.sum())
                    sg_score = compute_score(seq_sg)
                    if sg_score < best_score_local:
                        best_score_local = sg_score
                        best_seq_local = seq_sg.copy()
                        a = seq_sg  # continue from the polished point
                        print(f"[subgrad polish] improved to {sg_score:.6f}")

            # safety net – avoid endless loops
            if lp_iter >= 30:
                # after a reasonable number of LP updates we move on
                break

    # ------------------------------------------------------------------
    # Fallback (very unlikely) – generate a simple power‑law sequence.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-0.55)
        a = a / a.sum()
        a = a * np.sqrt(2.0 * n)
        global_best_seq = a

    # Return plain Python list as required by the harness
    return [float(x) for x in global_best_seq.tolist()]
