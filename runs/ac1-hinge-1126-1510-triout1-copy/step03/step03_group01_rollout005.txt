"""
Propose a step‑function with a low evaluation score.

The algorithm combines several ideas:

*   Continuous optimisation on the probability simplex using a soft‑max
    parametrisation and L‑BFGS, with a continuation schedule that
    gradually reduces a smooth‑max approximation of the convolution
    supremum.
*   Many random restarts with diverse initialisations (random, exponential,
    power‑law, edge‑heavy, Sidon‑set based) and occasional Gaussian
    perturbations of the current best solution.
*   After each L‑BFGS run a fast greedy “targeted hill‑climb’’ moves mass
    from indices that contribute most to the current maximal convolution
    value towards low‑mass indices.
*   A final long refinement with a finer smoothing schedule and a final
    polishing hill‑climb.
*   The whole search is bounded by the supplied time budget (≈1000 s).

The produced list of non‑negative floats is ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2 n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. the
    unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor, i.e. a relative smoothing width.
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # convolution of a (reversed) with w  -> gradient w.r.t a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point y (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set (pairwise sums distinct) – modest size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict:
                if cand * 2 in sums:
                    conflict = True
                else:
                    chosen.append(cand)
                    for c in chosen:
                        sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        if k > 0:
            a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode!r}")
    return y0


def hillclimb_targeted(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially moves probability mass
    away from indices that contribute most to the currently maximal
    convolution value.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # percentile of low‑mass indices that we consider as destinations
    low_percentile = 20.0

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current convolution and its worst index
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each index to that worst convolution entry
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros(n)
        contrib[mask] = best[mask] * best[j[mask]]

        # pick a donor index proportionally to its contribution
        total = contrib.sum()
        if total == 0.0:
            i = rng.integers(0, n)
        else:
            probs = contrib / total
            i = rng.choice(n, p=probs)

        # pick a receiver index among the low‑mass tail
        threshold = np.percentile(best, low_percentile)
        low_candidates = np.where(best <= threshold)[0]
        if low_candidates.size == 0:
            j_to = rng.integers(0, n)
        else:
            j_to = rng.choice(low_candidates)

        if i == j_to:
            continue

        # move up to 10 % of the donor mass toward the receiver
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j_to] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j_to] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    The method runs under a fixed wall‑time budget, repeatedly optimising
    random restarts with L‑BFGS (continuation on a smooth‑max proxy) and
    a targeted greedy hill‑climb.  The best sequence found before the
    deadline is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # problem size – use a reasonably large n, but honour any existing
    # longer construction stored in the global namespace.
    # --------------------------------------------------------------
    n = 8000
    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 2000:
                n = min(max(5000, int(tmp.size)), 12000)
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise with the uniform distribution (baseline)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)            # probability simplex (sum = 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)         # soft‑max representation
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    # A relatively fine continuation schedule
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
    ]

    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]

    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ---------- continuation L‑BFGS ----------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_schedule=lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ---------- targeted greedy hill‑climb ----------
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hillclimb_targeted(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final refinement with a finer schedule (more stages, larger caps)
    # --------------------------------------------------------------
    fine_lam_schedule = [
        0.3, 0.15, 0.07, 0.04, 0.02, 0.01,
        0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001,
    ]
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final polishing hill‑climb (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hillclimb_targeted(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
