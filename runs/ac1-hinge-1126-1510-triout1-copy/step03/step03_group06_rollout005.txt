"""
Hybrid LP‑guided refinement + projected sub‑gradient descent.

The objective is
    score(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

Because the expression is homogeneous we keep a on the probability simplex
(a_i ≥ 0, Σ a_i = 1).  The algorithm consists of three phases:

1. **LP‑guided expansion** – given a current shape f,
   we solve the linear programme

        maximise   Σ g_i
        subject to conv(f, g) ≤ M   (pointwise)
                    g_i ≥ 0

   where M = max(conv(f, f)).  The solution g is normalised to Σ g_i = 1
   and mixed with f:  f ← (1‑t) f + t g.  Repeating this a few times
   quickly reduces the peak of the self‑convolution.

2. **Projected sub‑gradient descent** on the simplex, following the
   gradient of the *active* peak of conv(f, f).  A diminishing step‑size
   schedule and simplex projection (Duchi et al.) are used.

3. **Light stochastic refinement** (simulated‑annealing style) moving a
   tiny amount of mass between two coordinates, accepting occasional
   uphill moves.

Multiple restarts with different lengths, initial power‑law shapes and
perturbations of the current global best are performed until the time
budget expires.  The best sequence found is returned as a plain Python
list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Simplex projection (Duchi et al. 2008)
# ----------------------------------------------------------------------
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def compute_conv_and_max(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Return the full linear convolution a*a and its maximum entry.
    Uses an FFT based method for speed.
    """
    n = a.shape[0]
    fft_len = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return conv, float(conv.max())


def compute_max_conv(a: np.ndarray) -> float:
    """Convenient wrapper that returns only the maximal convolution entry."""
    _, mx = compute_conv_and_max(a)
    return mx


# ----------------------------------------------------------------------
# Linear‑program to find a large‑sum partner g with bounded convolution
# ----------------------------------------------------------------------
def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve
        maximise   Σ g_i
        subject to   f * g ≤ M   (pointwise)
                     g_i ≥ 0
    returning the normalised solution (Σ g = 1).  f is assumed to lie on
    the simplex and to be non‑negative.
    """
    n = f.shape[0]
    # Build the (2n‑1) × n lower‑triangular Toeplitz matrix that
    # implements the linear convolution f * g.
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = toeplitz(c, r)                     # dense (2n‑1) × n matrix
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Linprog solves a *x ≤ b ; we minimise -Σ g_i to obtain a maximise.
    res = linprog(
        c=-np.ones(n, dtype=np.float64),
        A_ub=A,
        b_ub=b,
        bounds=[(0.0, None)] * n,
        method="highs",
    )

    if not res.success:
        # If the LP fails, fall back to the current shape.
        return np.copy(f)

    g = res.x
    s = g.sum()
    if s <= 0.0:
        return np.copy(f)
    return g / s


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the simplex
# ----------------------------------------------------------------------
def subgradient_descent(
    a0: np.ndarray,
    max_iters: int,
    step0: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run projected sub‑gradient descent on the simplex, returning the best
    sequence found and its (scaled) score 2·n·max(conv).
    """
    a = a0.copy()
    a = a / a.sum()
    n = a.shape[0]

    best_seq = a.copy()
    best_score = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv, max_b = compute_conv_and_max(a)
        score = 2.0 * n * max_b

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[subgrad it {it}] new best={score:.6f}")

        # active peak index
        k = int(np.argmax(conv))
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)

        grad = np.zeros_like(a)
        grad[idx] = 2.0 * a[k - idx]

        step = step0 / np.sqrt(it)
        a = project_simplex(a - step * grad)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Light stochastic local refinement (simulated‑annealing style)
# ----------------------------------------------------------------------
def local_refine(a0: np.ndarray, steps: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Random walk moving a tiny amount of mass between two coordinates,
    accepting improvements (or occasional uphill moves).  Returns the best
    sequence encountered and its scaled score.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_score = 2.0 * n * compute_max_conv(a)

    temperature = 0.02

    for _ in range(steps):
        if time.time() >= deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # direction of mass transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        delta = np.random.uniform(0.0, 0.05) * a[src]
        if delta == 0.0:
            continue

        new = a.copy()
        new[src] -= delta
        new[dst] += delta
        new = np.maximum(new, 0.0)
        s = new.sum()
        if s == 0.0:
            continue
        new /= s

        new_score = 2.0 * n * compute_max_conv(new)

        if new_score < best_score or np.random.rand() < np.exp((best_score - new_score) / temperature):
            a = new
            if new_score < best_score:
                best_score = new_score
                best_seq = new.copy()

        temperature = max(temperature * 0.9999, 1e-6)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative height sequence.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Determine a reasonable set of candidate lengths.
    # ------------------------------------------------------------------
    try:
        n0 = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821
        if n0 < 500:
            n0 = 2000
    except Exception:
        n0 = 2000

    length_candidates = sorted(
        set(
            [
                max(1500, n0 // 2),
                n0,
                min(3000, n0 * 2),
            ]
        )
    )

    global_best_seq = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Search over lengths; each length gets several independent restarts.
    # ------------------------------------------------------------------
    for n in length_candidates:
        # limit total restarts per length to keep inside time budget
        restart = 0
        while time.time() < deadline and restart < 30:
            restart += 1

            # ---- initialise a fresh candidate -------------------------
            # power‑law start
            exponent = np.random.uniform(0.45, 0.80)
            idx = np.arange(1, n + 1, dtype=np.float64)
            a0 = idx ** (-exponent)
            a0 /= a0.sum()

            # occasionally perturb the current global best if lengths match
            if (
                global_best_seq is not None
                and len(global_best_seq) == n
                and np.random.rand() < 0.25
            ):
                a0 = np.array(global_best_seq, copy=True)
                a0 += 0.02 * np.random.randn(n)
                a0 = np.maximum(a0, 0.0)
                s = a0.sum()
                a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)

            # ------------------------------------------------------------------
            # Phase 1: LP‑guided expansion (a few alternating steps)
            # ------------------------------------------------------------------
            a = a0.copy()
            lp_steps = 6
            t_initial = 0.5
            for lp_it in range(lp_steps):
                if time.time() >= deadline:
                    break
                # current peak
                conv, max_b = compute_conv_and_max(a)
                M = max_b
                # solve LP for partner g
                g = solve_lp_max_sum(a, M)
                # mix (use a decaying step size)
                t = t_initial * (0.7 ** lp_it)
                a = (1.0 - t) * a + t * g
                # ensure exact simplex (numerical drift)
                a = project_simplex(a)

            # ------------------------------------------------------------------
            # Phase 2: projected sub‑gradient descent (moderate budget)
            # ------------------------------------------------------------------
            remaining = deadline - time.time()
            # roughly 0.001 s per iteration for n≈2000 → allocate accordingly
            max_iters = int(min(8000, max(0, remaining / 0.0012)))
            if max_iters > 0:
                a, _ = subgradient_descent(a, max_iters, step0=0.8, deadline=deadline)

            # ------------------------------------------------------------------
            # Phase 3: local stochastic refinement (if time permits)
            # ------------------------------------------------------------------
            remaining = deadline - time.time()
            if remaining > 0.3:
                refine_steps = int(min(6000, remaining / 0.0008))
                a, _ = local_refine(a, refine_steps, deadline)

            # ------------------------------------------------------------------
            # Evaluate final candidate
            # ------------------------------------------------------------------
            final_score = 2.0 * n * compute_max_conv(a)   # Σa == 1
            if final_score < global_best_score:
                global_best_score = final_score
                global_best_seq = a.copy()
                print(
                    f"[len={n}] new global best score={final_score:.6f} "
                    f"(restart {restart})"
                )
                if final_score < 1.503:
                    # We have reached the target – finish early.
                    deadline = time.time()
                    break

        # early exit if target already met
        if global_best_score < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – construct a modest power‑law if nothing was found.
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n = 2000
        idx = np.arange(1, n + 1, dtype=np.float64)
        a = idx ** (-0.55)
        a /= a.sum()
        global_best_seq = a

    # Return as plain Python list of floats
    return [float(x) for x in global_best_seq.tolist()]
