"""
Propose a candidate step‑function (non‑negative heights) that minimises

    2 n·max(conv(a,a)) / (sum(a)**2)

The search consists of three stages:

1️⃣ **Broad random exploration** – several thousand candidates are drawn from
   many parametric families (power‑law, exponential, two‑sided power‑law,
   arcsine‑like, cosine/Hann, Gaussian, Beta‑shaped).  The best candidate
   (lowest fast evaluation) is kept.

2️⃣ **Smoothed exponential‑gradient descent** – starting from the best
   candidate we minimise a smooth approximation of the ℓ∞‑norm of the
   self‑convolution:

        Lβ(a) = (1/β)·log( Σk exp( β·(a*a)_k ) ).

   The gradient of Lβ can be written as a correlation between the
   normalised weights w_k∝exp(β·(a*a)_k) and the current sequence a.
   An exponentiated‑gradient update (which automatically respects
   non‑negativity) is performed, followed by normalisation onto the
   simplex.  The smoothing parameter β is increased slowly so that the
   optimisation focuses more and more on the actual maximum.  A small
   amount of jitter is added periodically to escape plateaus.

3️⃣ **Light local perturbation** – a few random “mass‑transfer” moves are
   tried; improvements are kept.  This often yields a final tiny gain.

The whole routine respects a 1000 s budget; if the time runs out the
best sequence found so far is returned.  The final sequence is verified
with the official `evaluate_sequence` function before being returned.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2) using FFT.
    The sequence is assumed non‑negative and already normalised (sum=1),
    therefore the denominator is 1 and the score reduces to 2·n·max(conv).
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(np.max(conv.real))
    return float(2.0 * n * max_conv / (s * s))


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_arcsine_like(n: int) -> np.ndarray:
    """Discrete arcsine‑like law: a_i ∝ 1/√(i+1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    """Hann window: 0.5·(1‑cos(2πi/(n‑1)))."""
    i = np.arange(n, dtype=np.float64)
    return 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))


def generate_gaussian_window(n: int, sigma_frac: float) -> np.ndarray:
    """Gaussian centred on the middle, σ = sigma_frac·n."""
    sigma = sigma_frac * n
    mid = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    return np.exp(-0.5 * ((i - mid) / sigma) ** 2)


def generate_beta_distribution(
    n: int, alpha_left: float, alpha_right: float
) -> np.ndarray:
    """
    Discrete Beta(α_left,α_right) shape on {0,…,n‑1}:
    a_i ∝ ( (i+0.5)/n )^{α_left‑1} · (1‑(i+0.5)/n)^{α_right‑1}.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # avoid zero raised to a negative power
    eps = np.finfo(float).tiny
    x = np.clip(x, eps, 1 - eps)
    left = np.power(x, alpha_left - 1.0)
    right = np.power(1.0 - x, alpha_right - 1.0)
    return left * right


def generate_mixed_distribution(
    n: int,
    base_func,
    base_kwargs: dict,
    mix_weight: float,
) -> np.ndarray:
    """
    Linear mixture of a base distribution and the uniform distribution.
    mix_weight ∈ [0,1] is the weight of the base component.
    """
    base = base_func(n, **base_kwargs)
    uniform = np.ones(n, dtype=np.float64)
    mix = mix_weight * base + (1.0 - mix_weight) * uniform
    return mix


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a low‑score height sequence under the given time budget.
    Returns the sequence as a plain Python list of floats.
    """
    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # 0️⃣  Load a previous construction if available
    try:
        init = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if init.ndim != 1 or init.size == 0:
            raise ValueError
    except Exception:
        # fallback: a medium‑size arcsine‑like sequence
        init = generate_arcsine_like(8000)

    init = np.maximum(init, 0.0)
    if init.sum() < 1e-12:
        init = np.full_like(init, 1.0 / init.size)
    else:
        init = init / init.sum()  # normalise to sum‑1 (the scale is irrelevant)

    best_seq = init.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={best_seq.size}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Randomised family search (broad exploration)
    # ------------------------------------------------------------------
    length_choices = [2000, 4000, 6000, 8000, 10000, 12000]
    families = [
        "power",
        "exp",
        "twosided",
        "arcsine",
        "cosine",
        "gaussian",
        "beta",
        "mix",
    ]

    # reserve roughly 250 s for this phase (or 20 % of the budget, whichever is smaller)
    random_deadline = min(deadline - 300.0, start + max(100.0, 0.20 * budget_s))

    while time.time() < random_deadline:
        n = random.choice(length_choices)

        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.20, 1.00)          # broaden the exponent range
            cand = generate_power_law(n, p)

        elif family == "exp":
            lam = random.uniform(1e-4, 3e-2)
            cand = generate_exponential(n, lam)

        elif family == "twosided":
            pl = random.uniform(0.20, 1.00)
            pr = random.uniform(0.20, 1.00)
            wl = random.uniform(0.5, 2.0)
            wr = random.uniform(0.5, 2.0)
            cand = generate_two_sided_power_law(n, pl, pr, wl, wr)

        elif family == "arcsine":
            cand = generate_arcsine_like(n)

        elif family == "cosine":
            cand = generate_cosine_window(n)

        elif family == "gaussian":

            cand = generate_gaussian_window(n, sigma_frac)

            al = random.uniform(0.1, 2.5)
            ar = random.uniform(0.1, 2.5)

        elif family == "mix":

                p = random.uniform(0.2, 1.0)
                base_kwargs = {"p": p}
                base_func = generate_power_law
            elif base_family == "exp":
                lam = random.uniform(1e-4, 3e-2)
                base_kwargs = {"lam": lam}
                base_func = generate_exponential
            else:  # beta
                al = random.uniform(0.1, 2.5)
                ar = random.uniform(0.1, 2.5)
                base_kwargs = {"alpha_left": al, "alpha_right": ar}
                base_func = generate_beta_distribution
            mix_weight = random.uniform(0.2, 0.9)
            cand = generate_mixed_distribution(n, base_func, base_kwargs, mix_weight)

        else:
            continue

        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s  # project onto the simplex

        sc = fast_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(
                f"[{time.time() - start:.1f}s] new random best "
                f"({family}) n={n} fast_score={sc:.6f}"
            )

    # ------------------------------------------------------------------
    # 2️⃣  Smoothed exponential‑gradient descent
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # FFT lengths (chosen once because n stays fixed)
    fft_len_conv = next_pow2(2 * n)
    fft_len_corr = next_pow2(3 * n)  # for w * a_rev (length ≈ 3n)

    # hyper‑parameters
    eta0 = 5.0               # initial step‑size multiplier
    beta0 = 0.05             # initial smoothing parameter
    beta_max = 5.0           # final smoothing (focus on the true max)
    beta_inc = 5000          # increase β every this many iterations
    eval_interval = 1000     # how often to compute the cheap score
    jitter_interval = 20000  # periodic jitter to escape plateaus
    jitter_scale = 0.005     # relative amplitude of jitter

    iteration = 0

    while time.time() < deadline - 20.0:
        iteration += 1

        # ---- self‑convolution ----
        Fa = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(Fa * Fa, n=fft_len_conv)[: 2 * n - 1]

        # ---- smoothed max (soft‑max) weights ----
        current_beta = min(beta_max, beta0 * (1.0 + iteration / beta_inc))
        # stabilise the exponent by subtracting the max value
        max_conv = np.max(conv)
        w = np.exp(current_beta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient (correlation of w with a) ----
        # correlation g_i = 2 * Σ_k w_{i+k} a_k  (0≤i<n)
        Wf = np.fft.rfft(w, n=fft_len_corr)

        corr = np.fft.irfft(Wf * Ar, n=fft_len_corr)
        g = 2.0 * corr[n - 1 : n - 1 + n]  # slice gives length n

        # ---- multiplicative (exponentiated) update ----
        eta = eta0 / np.sqrt(iteration)   # decaying step size
        a = a * np.exp(-eta * g)

        # ---- normalise onto the probability simplex ----
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ---- occasional jitter ----
        if iteration % jitter_interval == 0:
            noise = jitter_scale * np.random.randn(n) * a
            a = np.clip(a + noise, 0.0, None)
            a = a / a.sum()

        # ---- periodic cheap evaluation ----
        if iteration % eval_interval == 0:
            cur_score = fast_score(a)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[{time.time() - start:.1f}s] iter {iteration:,} "
                    f"new best fast_score={cur_score:.6f}"
                )

    # ------------------------------------------------------------------
    # 3️⃣  Light local perturbation (random mass swaps)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    perturb_deadline = deadline - 2.0
    attempts = 0
    while time.time() < perturb_deadline:
        i, j = random.sample(range(n), 2)
        eps = 1e-4 * a[i]  # move a tiny relative amount
        if eps <= 0.0:
            continue
        a[i] -= eps
        a[j] += eps
        # re‑normalise (the total is unchanged up to floating‑point noise)
        a = a / a.sum()
        cur_score = fast_score(a)
        attempts += 1
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[{time.time() - start:.1f}s] perturb {attempts:,} "
                f"new best fast_score={cur_score:.6f}"
            )
        else:
            # revert the move
            a[j] -= eps
            a[i] += eps
            a = a / a.sum()

    # ------------------------------------------------------------------
    # 4️⃣  Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    final_score = evaluate_sequence(final_seq)  # type: ignore[name-defined]
    print(f"Finished. Best score reported by evaluate_sequence = {final_score:.6f}")
    return final_seq
