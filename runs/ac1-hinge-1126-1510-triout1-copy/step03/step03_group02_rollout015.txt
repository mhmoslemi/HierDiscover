def proj_simplex(v):
    n = len(v)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    n = len(a)

"""
Propose a low‑score step‑function sequence.

The algorithm consists of three phases:

1. **Coarse shape search** – a low‑dimensional parametrisation (K anchor
   heights) is optimised with SciPy's differential evolution.  The
   anchors are interpolated to a full length‑`n` vector and normalised.
   This quickly discovers a promising non‑uniform shape.

2. **Fine‑grained projected gradient descent** – the exact sub‑gradient
   of the objective `max_k (a*a)_k` (the max of the linear
   auto‑convolution) is computed and a projected gradient step on the
   probability simplex is taken.  The step size decays as 1/√t.
   Because the gradient points to coordinates that increase the current
   peak, the descent pushes mass away from them and reduces the peak.

3. **Aggressive greedy hill‑climb** – random mass transfers between
   coordinates are accepted only when they strictly improve the fast
   score.  This polishing stage extracts the last bits of improvement.

All operations are performed with FFT‑based convolution (`conv_fft`) for
speed.  The whole search respects the 1000 s time budget and returns the
best sequence found (a list of non‑negative floats).  The final score
is obtained by calling the provided `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the full convolution of length 2*n‑1 (real part only).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector
    (sum(a) == 1):
        2 * n * max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def make_sequence_from_anchors(p: np.ndarray, n: int) -> np.ndarray:
    """
    Interpolate a low‑dimensional anchor vector `p` onto a full
    length‑`n` sequence and normalise it to sum‑to‑one.
    """
    k = len(p)
    xp = np.linspace(0, n - 1, k)
    a = np.interp(np.arange(n, dtype=float), xp, p)
    a = np.maximum(a, 0.0)
    # avoid the pathological all‑zero case
    a += 1e-12
    a /= a.sum()
    return a


def max_conv_subgradient(a: np.ndarray, max_inds: np.ndarray, n: int) -> np.ndarray:
    """
    Sub‑gradient of the function  max_k (a*a)_k  w.r.t. `a`.
    `max_inds` contains the indices k where the maximum is attained.
    """
    g = np.zeros_like(a)
    for k in max_inds:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if i < j:
                g[i] += a[j]
                g[j] += a[i]
            elif i == j:
                g[i] += 2.0 * a[i]
    return g


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { w | w_i >= 0, sum w_i = 1 }.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative -> uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly moves a fraction of the mass from one index to another
    and keeps the move only if the fast score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function sequence with a low
    evaluation score, respecting a 1000 s time budget.

    list[float]

    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Problem size – try to use a fairly large n (more DOF → lower bound)
    # --------------------------------------------------------------
    n = 8000
    try:
        prev = globals().get("height_sequence_1")
        if prev is not None:
            arr = np.asarray(prev, dtype=float).ravel()
            if arr.ndim == 1 and arr.size > n:
                n = int(arr.size)
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)  # unused in this version

    # --------------------------------------------------------------
    # Baseline (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Coarse shape optimisation with Differential Evolution
    # --------------------------------------------------------------
    K = 30  # number of anchor points
    # initialise a guess from the previous best sequence if available
    if "height_sequence_1" in globals():
        prev_seq = np.asarray(globals()["height_sequence_1"], dtype=float)
        idx = np.linspace(0, prev_seq.size - 1, K).astype(int)
        guess = prev_seq[idx]
        guess = np.maximum(guess, 0.0)

    else:

    bounds = [(0.0, 5.0) for _ in range(K)]

    def de_obj(p):

    popsize = 15

    init_pop[0] = guess * 5.0

    def de_callback(xk, convergence):

    de_res = differential_evolution(
        de_obj,

        strategy="best1bin",
        maxiter=30,
