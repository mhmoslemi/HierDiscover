"""
Propose a low‑score non‑negative step sequence.

The search combines three ideas:
1. Random restarts with a fast projected sub‑gradient descent that directly
   minimises the non‑smooth objective 2·n·max(conv(a,a)).
2. A long hill‑climb that mixes pure random mass transfers with
   gradient‑directed moves (using the sub‑gradient of the active
   convolution constraints) to shave off the remaining tiny amount.
3. A final polishing pass (still hill‑climb) exploiting every remaining
   millisecond.

All steps respect the simplex (sum = 1, a_i ≥ 0).  The implementation
uses FFT‑based convolutions for O(n log n) speed and carefully splits the
1000 s time budget.  The returned list is ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projected sub‑gradient descent
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):

    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_candidate = simplex_projection(a - step * g)
            cand_score = fast_score(a_candidate, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb (random + gradient‑directed moves)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb that alternates between:
      * a random small mass transfer between two coordinates,
      * a gradient‑directed transfer from the index with largest sub‑gradient
        contribution (i.e. most responsible for the current max) to the one
        with the smallest contribution.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        if rng.random() < 0.3:                     # gradient‑directed move
            # compute sub‑gradient of the current max‑convolution
            fa = np.fft.fft(best, fft_len)
            conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - eps).astype(np.float64)

            rev = best[::-1]
            fm = np.fft.fft(mask, fft_len)
            fr = np.fft.fft(rev, fft_len)
            cg = np.real(np.fft.ifft(fm * fr))
            g = 2.0 * cg[(n - 1):(n - 1 + n)]

            i_high = int(np.argmax(g))
            i_low = int(np.argmin(g))
            if i_high != i_low and best[i_high] > 1e-12:
                delta = best[i_high] * rng.random() * 0.04
                cand = best.copy()
                cand[i_high] -= delta
                cand[i_low] += delta
                new_score = fast_score(cand, fft_len, n)
                if new_score < best_score:
                    best = cand
                    best_score = new_score
        else:                                      # plain random transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                it += 1
                continue
            delta = best[i] * rng.random() * 0.05
            if delta <= 0.0:
                it += 1
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            new_score = fast_score(cand, fft_len, n)
            if new_score < best_score:
                best = cand
                best_score = new_score
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random point on the simplex."""
    a = rng.random(n)
    a /= a.sum()
    return a


def edge_heavy_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass on the two borders, tiny mass in the centre."""
    edge_frac = rng.uniform(0.010, 0.035)                # proportion of indices that become “edges”
    m_edge = max(1, int(n * edge_frac))
    a = np.full(n, 1e-12)                               # tiny background to stay away from zeros
    big = np.sqrt(n / 2.0) / n
    a[:m_edge] = big
    a[-m_edge:] = big
    a /= a.sum()
    return a


def decreasing_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Monotonically decreasing shape (more mass on the left)."""
    rate = rng.uniform(0.0008, 0.004)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (a little larger than the previous 2000 gives the
    # algorithm more freedom without killing speed)
    # ------------------------------------------------------------------
    max_n = 3000
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0.sum()
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)          # for all convolutions
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many short sub‑gradient runs from different starts
    # ------------------------------------------------------------------
    sub_deadline_total = start_time + 0.55 * budget_s
    while time.time() < sub_deadline_total:
        # pick a random generator
        mode = rng.choice(['rand', 'edge', 'dec'])
        if mode == 'rand':
            a_start = random_initial(n, rng)
        elif mode == 'edge':
            a_start = edge_heavy_initial(n, rng)
        else:
            a_start = decreasing_initial(n, rng)

        # allocate a modest slice of time for this run
        run_deadline = min(sub_deadline_total,
                           time.time() + 0.25 * (deadline - time.time()))
        a_opt, score_opt = subgradient_optimize(a_start,
                                                run_deadline,
                                                fft_len,
                                                fft_len,
                                                n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[sub‑grad‑restart] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive hill‑climb (random + gradient directed)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[hill‑climb] starting intensive climb")
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=200000)
        print(f"[hill‑climb] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (as much as time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final‑polish] extra hill‑climb")
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=50000)
        print(f"[final‑polish] best score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)          # external function
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
