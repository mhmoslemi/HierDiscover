"""
Propose a low‑evaluation step‑function sequence.

Algorithm overview
------------------
1. Initialise a probability vector `a` (sum = 1).  If a previously
   discovered sequence is available via `height_sequence_1` we reuse it,
   otherwise we start from the uniform distribution.

2. A first “smooth” optimisation phase uses the same L‑BFGS + soft‑max
   continuation that was used before.  This quickly brings the
   convolution’s maximum down to the vicinity of the current state‑of‑the‑art.

3. Refine the vector by a dedicated sub‑gradient descent on the *exact*
   max‑convolution (not a smooth approximation).  For the index `i*`
   where the convolution attains its maximum we have
       ∂max/∂a_j = 2·a_{i*−j}   (if the index exists, otherwise 0).
   We move the probabilities in the opposite direction of this sub‑gradient,
   project back onto the probability simplex, and accept the move only if
   the maximum convolution is reduced.  A diminishing step‑size and occasional
   random perturbations avoid stagnation.

4. A short greedy hill‑climb (the one from the reference code) finishes
   the polishing.

All heavy work (FFT‑based convolutions) is performed with powers‑of‑two
lengths for speed.  The whole search respects the 1000 s budget; the
best sequence found is finally evaluated with the supplied
`evaluate_sequence` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    Returns the full convolution (real part only).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS helpers (kept from the reference implementation)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """Smooth‑max objective 2·n·smooth_max(conv) and its gradient w.r.t. y."""
    a = softmax(y)

    # ---- convolution a*a (full length) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                    # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # gradient w.r.t. a: g_i = Σ_j a_j * w_{i+j}
    # (convolution of a reversed with w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """Successive L‑BFGS runs with decreasing smoothing."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (identical to the reference version)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep the move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def max_conv_and_gradient(a: np.ndarray, fft_len: int) -> tuple[float, np.ndarray, int]:
    """Return (max_conv, sub‑gradient vector, index_of_max)."""
    n = a.shape[0]
    conv = conv_fft(a, fft_len, n)
    i_max = int(np.argmax(conv))
    max_val = float(conv[i_max])

    # sub‑gradient: g_j = 2 * a_{i_max - j} if the index exists
    idx = np.arange(n)
    q = i_max - idx
    mask = (q >= 0) & (q < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[q[mask]]
    return max_val, g, i_max


def subgradient_descent(
    a: np.ndarray,
    fft_len: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on max(conv).  Returns the best vector
    found and its fast_score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        max_val, g, _ = max_conv_and_gradient(a, fft_len)

        # diminishing step size
        eta = 0.5 / np.sqrt(it + 1)

        # gradient step and projection onto the simplex
        a_candidate = a - eta * g
        a_candidate = project_simplex(a_candidate)

        # evaluate candidate
        cand_score = fast_score(a_candidate, fft_len, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[sgd] iter={it} improved fast score = {best_score:.6f}")

        # accept only if we improved the current point
        if cand_score < 2.0 * n * max_val:
            a = a_candidate
        else:
            # occasional random shake‑up to escape flat regions
            if rng.random() < 0.01:
                noise = rng.normal(scale=0.02, size=n)
                a = a * np.exp(noise)
                a = a / a.sum()

        it += 1

    return best_a, best_score
