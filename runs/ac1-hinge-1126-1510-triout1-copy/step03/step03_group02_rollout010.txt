def directed_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=3000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        max_idx = int(np.argmax(conv))
        # compute contributions only on valid range
        low = max(0, max_idx - (n-1))
        high = min(n-1, max_idx)
        # contributions vector
        # contributions[i] = best[i] * best[max_idx - i] for i in [low, high]
        contrib = np.zeros(n)
        indices = np.arange(low, high+1)
        contrib[indices] = best[indices] * best[max_idx - indices]
        # choose source proportionally to contrib
        total_contrib = contrib.sum()
        if total_contrib == 0:
            # fallback to random source/target
            src = rng.integers(n)
            tgt = rng.integers(n)
        else:
            # random choice weighted by contrib
            src = rng.choice(n, p=contrib/total_contrib)
            # choose target where contrib is small (or a is small)
            # compute weight for target: (1 - contrib_norm) * (1 - a_norm)
            # simpler: choose random but reject if a[tgt] > a[src] maybe
            # we'll pick target among bottom K smallest a
            # find indices sorted by a
            # precomputed sorted? just sample random then check condition
            # sample until we find
            attempts = 0
            while attempts < 5:
                tgt = rng.integers(n)
                if best[tgt] < best[src] * 0.5:
                    break
                attempts += 1
        # compute delta as fraction of source
        delta = best[src] * rng.random() * 0.2  # up to 20% of source
        if delta <= 0:
            continue
        # apply move
        best[src] -= delta
        best[tgt] += delta

"""
Propose a candidate step‑function sequence.

The algorithm builds on the previous multi‑restart L‑BFGS approach but
adds two key refinements:

1. **Geometric smoothing continuation** – a longer λ‑schedule that drives
   the smooth‑max (log‑sum‑exp) approximation down to 1e‑5, allowing the
   optimiser to approach the true max more closely.

2. **Gradient‑guided hill‑climbing** – after each L‑BFGS run we compute the
   exact autocorrelation (via FFT), locate the index where it attains its
   maximum and move mass away from the pair of coordinates that contribute
   most to this peak.  The source index is chosen with probability proportional
   to its contribution a_i·a_{k−i}; the target is chosen among the smallest
   entries.  This directed move is far more effective than a blind random
   swap.

All intermediate vectors are kept on the probability simplex (sum = 1);
symmetrisation is applied after each refinement because the optimal
solution is known to be symmetric.  The final sequence is evaluated with
`evaluate_sequence` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) using FFT.
    a is assumed to have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector (sum = 1).
    This is exactly the value returned by ``evaluate_sequence`` (the denominator
    ``sum(a)**2`` equals 1 after normalisation).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max of the auto‑convolution and its gradient w.r.t. the unrestricted
    parameters y (the log‑soft‑max coordinates).

    The smooth‑max is ``max_c + λ·log Σ exp((c_i‑max_c)/λ)`` with
    λ = max_c·lam_factor (plus a tiny epsilon).  Gradients are obtained by the
    chain rule through the soft‑max.
    """
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient of smooth max ----------
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 200,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list (geometric schedule) that
    gradually reduces the smoothing of the max‑operator.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force exact left–right symmetry and renormalise to sum‑1."""
    a_sym = (a + a[::-1]) * 0.5
    s = a_sym.sum()
    if s > 0:
        a_sym /= s
    return a_sym


def directed_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb on the simplex.

    At each step we locate the maximal autocorrelation index ``k`` and move
    a fraction of the mass from a source coordinate that contributes heavily
    to ``conv[k]`` to a target coordinate that is currently small.  Moves are
    kept only if they strictly improve the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        # indices that actually appear in the sum for conv[k]
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        idx = np.arange(low, high + 1)
        contrib = best[idx] * best[k - idx]            # a_i * a_{k-i}
        tot = contrib.sum()
        if tot == 0:
            src = rng.integers(n)
        else:
            src = rng.choice(n, p=contrib / tot)       # source weighted by contribution

        # pick a target among the currently smallest entries
        # (we try a few random draws until a sufficiently smaller entry is found)
        for _ in range(5):
            tgt = rng.integers(n)
            if best[tgt] < 0.7 * best[src]:
                break
        else:
            tgt = rng.integers(n)                       # fallback

        if src == tgt:
            continue

        # move up to 15 % of the source mass
        delta = best[src] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        best[src] -= delta
        best[tgt] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[src] += delta
            best[tgt] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    ``evaluate_sequence`` score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 6000
    try:
        prev = globals().get("height_sequence_1")
        if isinstance(prev, (list, np.ndarray)):
            arr = np.asarray(prev, dtype=float)
            if arr.ndim == 1 and arr.size >= 1000:
                n = max(n, int(arr.size))
    except Exception:
        pass

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # initialise with the uniform distribution
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-5, num=30)   # smoother → tighter
    maxiter_stage = 200

    iteration = 0

    # --------------------------------------------------------------
    # multi‑restart optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline - 6.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.5)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.05)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, centre_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        a_opt = symmetrize(a_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- directed hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = directed_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_a = symmetrize(best_a)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # final deep refinement with the same continuation schedule
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_a = symmetrize(best_a)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final polishing hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = directed_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_a = symmetrize(best_a)

    # --------------------------------------------------------------
    # official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
