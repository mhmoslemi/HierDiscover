"""
Propose a low‑value step sequence.

The evaluation metric
    E(a) = 2·n·max(a*a) / (sum(a))²
is scale‑invariant, so we restrict the search to the probability simplex
(sum = 1, a_i ≥ 0).  The program consists of four phases

1. **Parametric shape exploration** – generate many candidates from a few
   simple families (Gaussian, Laplace, Beta‑like, random piece‑wise constant)
   on a fixed length `n`.  Each candidate is normalised and evaluated with a
   fast FFT based score `2·n·max(conv)`.  The best candidate is retained.

2. **Projected sub‑gradient refinement** – starting from the best shape,
   perform a diminishing‑step sub‑gradient descent on the true non‑smooth
   objective (the max‑convolution) while projecting onto the simplex.
   This phase exploits the exact gradient of a smoothed max and yields a
   monotone improvement.

3. **Simulated annealing** – a short non‑local search that moves a random
   fraction of mass between two coordinates; uphill moves are accepted with
   a slowly decreasing temperature.  It helps escape shallow local minima.

4. **Final greedy hill‑climb** – a fast local optimiser that transfers a
   small fraction of mass between two positions only when the score improves.
   The resulting sequence is returned.

All convolutions are performed by FFT (O(n log n)).  The routine respects the
supplied time budget (default 1000 s) and prints progress information.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already summing to 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x ≥ 0, sum x = 1 }.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient optimisation of the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using a sub‑gradient method.
    Each step projects back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = full_conv(a, fft_len_conv)
    best_score = float(2.0 * n * conv.max())
    step0 = 1.0
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient: g = 2 * (mask * a_rev) restricted to the original n entries
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        # keep the best iterate seen so far
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simple simulated annealing that moves a fraction of mass between two
    positions.  Accepts improving moves always, uphill moves with a
    temperature‑dependent probability.  Temperature decays geometrically.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-12:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    move a random fraction of one coordinate to another while preserving the sum.
    Accept only moves that improve the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Parametric shape generators (all return a normalised probability vector)
# ----------------------------------------------------------------------


def shape_gaussian(n: int, sigma: float) -> np.ndarray:
    """Symmetric discrete Gaussian (truncated to length n)."""
    idx = np.arange(n)
    mid = (n - 1) / 2.0
    a = np.exp(- ((idx - mid) ** 2) / (2.0 * sigma ** 2))
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def shape_laplace(n: int, scale: float) -> np.ndarray:
    """Symmetric discrete Laplace (double‑exponential) shape."""
    idx = np.arange(n)
    mid = (n - 1) / 2.0
    a = np.exp(- np.abs(idx - mid) / scale)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def shape_beta(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑like shape a_i ∝ (i+1)^p·(n-i)^q."""
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** p) * ((n + 1 - i) ** q)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def shape_piecewise(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piece‑wise constant distribution."""
    segs = rng.integers(2, 7)          # between 2 and 6 segments
    breakpoints = np.sort(rng.integers(0, n + 1, size=segs - 1))
    heights = rng.random(segs) + 0.1   # avoid exact zeros
    a = np.empty(n, dtype=float)
    start = 0
    for bp, h in zip(np.append(breakpoints, n), heights):
        a[start:bp] = h
        start = bp
    a /= a.sum()
    return a

    # ------------------------------------------------------------------
    subgrad_deadline = start_time + 0.70 * budget_s
    if time.time() < subgrad_deadline:
        best_a, best_score = subgradient_optimize(best_a,
                                                  subgrad_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for fine‑grained exploration
    # ------------------------------------------------------------------
    sa_deadline = start_time + 0.90 * budget_s
    if time.time() < sa_deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 sa_deadline,
                                                 rng)
        print(f"[sim‑anneal] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a,
                                      best_score,
                                      fft_len_conv,
                                      n,
                                      deadline,
                                      max_iter=3000)

    # ------------------------------------------------------------------
    # Report final evaluation using the provided evaluator
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:          # in the sandbox evaluate_sequence may not be present
        eval_val = float('nan')
    print(f"[final] evaluate_sequence = {eval_val:.6f}")

    # Return the raw list (the evaluator will re‑normalise it anyway)
    return final_a.tolist()
