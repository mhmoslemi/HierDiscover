"""
propose_candidate
-----------------
Search for a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

where `conv` is the linear convolution of the sequence with itself.
The objective is invariant to overall scaling, so we keep the vector on the
probability simplex (sum = 1).

The algorithm is a multi‑restart smooth‑max optimisation:

*   The variables are represented in the unconstrained log‑softmax space
    (`a_i = softmax(y_i)`).  This automatically satisfies non‑negativity
    and the sum‑to‑one constraint.
*   A smooth approximation to the `max` (soft‑max) is minimised with a
    geometric continuation schedule on the smoothing factor.
*   Each L‑BFGS run is seeded from a diverse family of initial
    distributions (`random`, `exp`, `power`, `edge`, `sidon`,
    `beta`, `arcsine`, `sparse`).  The current best solution is also
    perturbed occasionally.
*   After every L‑BFGS phase a greedy hill‑climb moves a small
    fraction of mass between random positions and accepts only strict
    improvements of the exact `fast_score = 2·n·max(conv)`.
*   When the time budget is almost exhausted a finer continuation
    schedule and a final intensive hill‑climb polish the best
    candidate.

The whole search respects the 1000 s time limit; if the limit is hit
the currently best sequence is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum == 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and gradient w.r.t. y.

    Parameters
    ----------
    y : ndarray
        Unconstrained log‑softmax variables.
    n, fft_len_conv, fft_len_g : int
        Problem size and FFT lengths.
    lam_factor : float
        Smoothing factor λ = max(conv)·lam_factor.

    Returns
    -------
    f : float
        Smoothed objective value.
    grad_y : ndarray
        Gradient w.r.t. y.
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c            # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth‑max -----
    w = exp_shift / Z                 # ∂smooth_max / ∂conv_i

    # convolution of reversed a with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the n‑length central part (indices n‑1 … 2n‑2)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Accept the move only if the exact fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12,
                     "disp": False},
        )
        y = res.x
    return y


def init_arcsine(n: int) -> np.ndarray:
    """Arc‑sine (Beta(½,½)) shaped distribution – heavy at the ends."""
    i = np.arange(n, dtype=float)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = np.maximum(a, eps)
    a /= a.sum()
    return np.log(a + eps)


def init_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random Beta‑like shape a_i ∝ (i+1)^p · (n‑i)^q with p,q∈[‑2,2]."""
    p = rng.uniform(-2.0, 2.0)
    q = rng.uniform(-2.0, 2.0)
    i = np.arange(n, dtype=float)
    a = ((i + 1.0) ** p) * ((n - i) ** q)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    eps = 1e-12
    return np.log(a + eps)


def init_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    """Sparse random support with Dirichlet weights."""
    # expected support size ≈ 2·√n (adjustable)
    K = max(2, int(np.sqrt(n) * rng.uniform(1.5, 3.0)))
    idx = rng.choice(n, size=K, replace=False)
    w = rng.random(K) + 1e-3
    w /= w.sum()
    a = np.zeros(n, dtype=float)
    a[idx] = w
    eps = 1e-12
    return np.log(a + eps)


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a starting point `y` (pre‑softmax) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set (pairwise sums distinct) – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "beta":
        y0 = init_beta(n, rng)
    elif mode == "arcsine":
        y0 = init_arcsine(n)
    elif mode == "sparse":
        y0 = init_sparse(n, rng)
    else:
        raise ValueError(f"unknown mode `{mode}`")
    return y0

    try:
        init_seq = globals().get("height_sequence_1", None)
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            tmp = np.asarray(init_seq, dtype=float)
            if tmp.ndim == 1 and tmp.size >= 1000:
                n = max(n, int(tmp.size))
    except Exception:
        pass

    # FFT lengths – powers of two for speed
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initial uniform baseline
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # probability vector
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} uniform fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # geometric decay of the smoothing factor
    lam_schedule = list(np.geomspace(0.5, 0.001, num=14))
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon",
             "beta", "arcsine", "sparse", "best"]

    # ------------------------------------------------------------------
    # multi‑restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            # small Gaussian perturbation of the current best
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # ---------- continuation L‑BFGS ----------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_schedule=lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ---------- aggressive hill‑climb ----------
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final refinement: finer smoothing schedule
    # ------------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.3, 5e-4, num=18))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 500, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.5g} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final greedy polishing (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # the harness provides this function
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence for possible reuse by later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
