def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    start=time.time()
    deadline = start+budget_s-5

    n = 5000 # maybe bigger if time allows; we calibrate based on time.

    # Initial Catalan distribution
    a = np.empty(n, dtype=np.float64)
    a[0]=1.0
    for i in range(1,n):
        a[i]=a[i-1] * (2*i-1)/(2*i)

    # Normalize sum to 1 (optional) but leaving as is fine. We'll keep as is.
    cur_seq = a.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"Initial score: {best_score:.6f}")

    # parameters
    max_iters = 10_000_000
    it=0
    # set step size factor minimal
    # we maintain small epsilon for moves: define epsilon = 0.01 * a[i] maybe.

    while time.time() < deadline and it < max_iters:
        it+=1

        # compute convolution and max index
        n_cur = cur_seq.size
        # compute conv and max
        # Re-use FFT each loop for speed; just call _fast_score returns max conv? Not directly.

        # We'll compute conv explicitly:
        L = 1 << ((2*n_cur-1).bit_length())
        f = np.fft.rfft(cur_seq, n=L)
        conv = np.fft.irfft(f * f, n=L)[:2*n_cur-1]
        M = conv.max()
        k_star = conv.argmax()
        # compute derivative for each i
        # derivative = a_{k_star - i} if 0<=k_star-i<n else 0, except for self term factor 2
        # Let's vectorize:
        # i indices 0..n-1
        idx = np.arange(n_cur)
        partner = k_star - idx
        mask = (partner >=0) & (partner < n_cur)
        deriv = np.zeros_like(cur_seq)
        deriv[mask] = cur_seq[partner[mask]]
        # double count self term when partner == idx
        self_mask = mask & (partner == idx)
        deriv[self_mask] *= 2.0
        # choose source index from those with large derivative and with some mass
        # We may select top candidates within early region to focus on early mass.
        # We'll filter idx with deriv > 0 and cur_seq > 0.
        candidate_mask = (deriv > 0) & (cur_seq > 1e-12)
        if not np.any(candidate_mask):
            continue
        # choose source with highest derivative * value perhaps:
        # weight = deriv * cur_seq (mass moved weighted by improvement)
        weights = deriv * cur_seq
        # limit to first half (or early region) maybe we restrict to idx< n//4 to ensure early.
        early_region = n_cur // 4
        cand_region_mask = (idx < early_region) & candidate_mask
        if np.any(cand_region_mask):
            src_pool = np.where(cand_region_mask)[0]
            # take max weight among them
            src = src_pool[np.argmax(weights[src_pool])]
        else:
            src = np.where(candidate_mask)[0][np.argmax(weights[candidate_mask])]
        # choose target far in tail:

import time, numpy as np

def _fast_score(seq):
    n = seq.size
    s = seq.sum()
    if s < 0.01: return float('inf')
    L = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2*n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)

def _conv_max(seq):
    n = seq.size
    L = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2*n - 1]
    max_val = float(conv.max())
    max_idx = int(conv.argmax())
    return max_val, max_idx

def generate_catalan(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1) / (2.0*i)
    return a

def generate_powerlaw(n, p):
    idx = np.arange(1, n+1, dtype=np.float64)  # 1..n
    a = 1.0/ (idx ** p)
    a = a / a.sum()  # normalize to sum=1
    return a

def generate_exponential(n, lam):
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a = a / a.sum()
    return a

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # safety margin

    # ---------- Phase 1: quick exploration of simple parametric families ----------
    best_seq = None
    best_score = float('inf')
    # try a few lengths
    lengths = [1500, 2500, 3500]  # moderate sizes
    for n in lengths:
        # Catalan
        seq = generate_catalan(n)
        seq = seq / seq.sum()  # optional scaling
        score = _fast_score(seq)
        if score < best_score:
            best_score, best_seq = score, seq.copy()
            print(f"Init Catalan n={n} score={score:.6f}")
        # power law exponents
        for p in np.linspace(0.5, 2.5, 9):
            seq = generate_powerlaw(n, p)
            score = _fast_score(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"Power p={p:.2f} n={n} score={score:.6f}")
        # exponential
        for lam in np.linspace(0.0005, 0.02, 9):
            seq = generate_exponential(n, lam)
            score = _fast_score(seq)
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"Exp lam={lam:.4f} n={n} score={score:.6f}")

    # If all attempts failed (unlikely) fall back to Catalan with larger n
    if best_seq is None:
        n = 2500
        best_seq = generate_catalan(n)
        best_score = _fast_score(best_seq)

    # ---------- Phase 2: local refinement ----------
    cur_seq = best_seq.copy()
    cur_score = best_score

    n = cur_seq.size
    early_limit = max(1, n // 6)   # early region for sources
    tail_start = max(early_limit + 1, 4 * n // 5)  # target region
    max_iters = 10_000_000
    it = 0

    # Some annealing schedule
    init_temp = best_score * 0.5
    while time.time() < deadline and it < max_iters:
        it += 1

        # compute current max convolution and its index
        max_b, k_star = _conv_max(cur_seq)

        # compute derivative array (sensitivity of max to each a_i)
        idx = np.arange(n)
        partner = k_star - idx
        mask = (partner >= 0) & (partner < n)
        deriv = np.zeros_like(cur_seq)
        deriv[mask] = cur_seq[partner[mask]]
        # double contribution when i == partner
        self_mask = mask & (partner == idx)
        deriv[self_mask] *= 2.0

        # define a score for moving mass: we want high derivative * current value
        move_impact = deriv * cur_seq

        # source candidate: early region with biggest impact

        # target candidate: far tail where derivative is small

                print(f"Iter {it:07d} new best {best_score:.6f}")

        # occasional random perturbation (shake)

            if shake_score < best_score:

                best_seq = shake_seq.copy()
                cur_seq = shake_seq
                cur_score = shake_score
                print(f"Iter {it:07d} shake improved to {best_score:.6f}")

        # early exit if target reached

            print(f"Target reached at iteration {it}, score {best_score:.6f}")
            break

    print(f"Finished after {it} iterations. Best score {best_score:.6f}")
    # Clip to allowed bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()

"""
Propose a step‑function (list of non‑negative floats) that minimises the
evaluation metric defined in ``evaluate_sequence``.

The algorithm works in two phases:

1. **Exploratory phase** – generate a few simple families of sequences
   (Catalan / half‑binomial, power‑law, exponential) for several lengths,
   keep the best one as a warm start.  If a global variable
   ``height_sequence_1`` is present it is also considered.

2. **Refinement phase** – a fast gradient‑guided stochastic search.
   At each iteration we compute the self‑convolution, locate the index
   where the maximum occurs and evaluate the sensitivity of that maximum
   to every coefficient (the derivative is simply the partner coefficient
   in the convolution).  We then move a fraction of mass from a coefficient
   with high sensitivity (large impact on the peak) to a coefficient that
   does not influence the current peak (derivative zero) and accept the
   move if it improves the score (or with a small Simulated‑Annealing
   probability).  The step size shrinks over time and occasional tiny
   random “shakes’’ help to escape plateaus.

The whole routine respects the 1000 s time limit, prints progress and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the provided one, but without safety checks)
def _fast_score(seq: np.ndarray) -> float:
    """Return the metric 2·n·max(conv) / (sum(seq)**2)."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return float('inf')
    # next power of two for a fast FFT
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Helper that returns both the maximum value of the convolution and its index
def _conv_max(seq: np.ndarray):
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


# ----------------------------------------------------------------------
# Simple families of sequences ------------------------------------------------
def _catalan(n: int) -> np.ndarray:
    """Half‑binomial (Catalan‑type) coefficients a_i = C(2i,i)/4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    # Normalise only to avoid the <0.01 guard – scaling does not affect the score
    a = a / a.sum()
    return a


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (normalised to unit sum)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a = a / a.sum()
    return a


def _exponential(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)  (normalised to unit sum)."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000) -> list[float]:
    """Search for a low‑score step function within the given time budget."""
    np.random.seed(seed)
    start = time.time()
    # keep a few seconds margin for the wrapper
    deadline = start + budget_s - 5.0

    # ------------------------------------------------------------------
    # Phase 1 – quick sweep over a few parametric families
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # if a previous construction is available, try it first
    if 'height_sequence_1' in globals():
        try:
            seq0 = np.asarray(height_sequence_1, dtype=np.float64)
            if seq0.size > 0:
                seq0 = np.clip(seq0, 0.0, 1000.0)
                if seq0.sum() >= 0.01:
                    seq0 = seq0 / seq0.sum()          # normalise – scaling is irrelevant
                    sc = _fast_score(seq0)
                    best_seq, best_score = seq0.copy(), sc
                    print(f"Seeded from height_sequence_1, score={sc:.6f}")
        except Exception:
            pass

    lengths = [1500, 2500, 3500]  # modest sizes – quick to evaluate
    for n in lengths:
        # Catalan
        cand = _catalan(n)
        sc = _fast_score(cand)
        if sc < best_score:
            best_seq, best_score = cand.copy(), sc
            print(f"Catalan n={n} → score={sc:.6f}")

        # Power‑law exponents
        for p in np.linspace(0.5, 2.5, 9):
            cand = _power_law(n, p)
            sc = _fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"Power‑law p={p:.2f}, n={n} → score={sc:.6f}")

        # Exponential decay
        for lam in np.linspace(0.0005, 0.0200, 9):
            cand = _exponential(n, lam)
            sc = _fast_score(cand)
            if sc < best_score:
                best_seq, best_score = cand.copy(), sc
                print(f"Exponential λ={lam:.4f}, n={n} → score={sc:.6f}")

    # Ensure we have a starting point
    if best_seq is None:
        n = 2500
        best_seq = _catalan(n)
        best_score = _fast_score(best_seq)

    cur_seq = best_seq.copy()
    cur_score = best_score
    n = cur_seq.size

    # ------------------------------------------------------------------
    # Phase 2 – gradient‑guided stochastic refinement
    # ------------------------------------------------------------------
    # dynamic source/target selection based on the current convolution peak
    max_iters = 10_000_000
    it = 0
    # initial temperature for simulated‑annealing component
    init_temp = best_score * 0.5

    while time.time() < deadline and it < max_iters:
        it += 1

        # 1) current convolution peak and its index
        cur_max, k_star = _conv_max(cur_seq)

        # 2) sensitivity of the peak to each coefficient:
        #    d(conv[k_star])/d a_i = a_{k_star-i} (2*a_i when i == k_star-i)
        idx = np.arange(n)
        partner = k_star - idx
        valid = (partner >= 0) & (partner < n)
        deriv = np.zeros_like(cur_seq)
        deriv[valid] = cur_seq[partner[valid]]
        self_mask = valid & (partner == idx)
        deriv[self_mask] *= 2.0

        # 3) Choose a source (mass that actually influences the peak)
        src_mask = (cur_seq > 1e-12) & (deriv > 0)
        if not np.any(src_mask):
            break
        # impact = derivative * current amount – larger impact ⇒ larger reduction
        impact = deriv * cur_seq
        src_candidates = np.where(src_mask)[0]
        src = src_candidates[np.argmax(impact[src_candidates])]

        # 4) Choose a target that does NOT influence the current peak
        tgt_mask = (deriv == 0)
        if np.any(tgt_mask):
            # among those, pick the smallest coefficient (to stay flat)
            tgt_candidates = np.where(tgt_mask)[0]
            tgt = tgt_candidates[np.argmin(cur_seq[tgt_candidates])]
        else:
            # fallback – pick something far from the source
            tgt = (src + n // 2) % n

        if tgt == src:
            continue

        # 5) Propose moving a fraction of the source amount
        #    step size shrinks linearly from ~30 % to ~1 %
        elapsed = time.time() - start
        frac = 0.30 * (1.0 - elapsed / (budget_s - 5.0)) + 0.01
        delta = cur_seq[src] * frac
        # respect the 0 – 1000 bounds
        if cur_seq[tgt] + delta > 1000.0:
            delta = 1000.0 - cur_seq[tgt]
        if delta <= 0.0:
            continue

        trial_seq = cur_seq.copy()
        trial_seq[src] -= delta
        trial_seq[tgt] += delta

        trial_score = _fast_score(trial_seq)

        # 6) Acceptance rule
        temp = init_temp * max(0.0, 1.0 - elapsed / (budget_s - 5.0))
        accept = False
        if trial_score < cur_score:
            accept = True
        elif temp > 1e-12:
            prob = np.exp((cur_score - trial_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_seq = trial_seq
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_seq = trial_seq.copy()
                print(f"Iter {it:07d} – new best score: {best_score:.6f}")

        # 7) Periodic tiny random “shake’’ to jump out of tiny plateaus
        if it % 20000 == 0:
            shake_seq = cur_seq.copy()
            shake_idxs = np.random.choice(n, size=10, replace=False)
            perturb = np.random.randn(10) * 0.001 * shake_seq[shake_idxs]
            shake_seq[shake_idxs] = np.clip(shake_seq[shake_idxs] + perturb, 0.0, 1000.0)
            shake_score = _fast_score(shake_seq)
            if shake_score < best_score:
                best_score = shake_score
                best_seq = shake_seq.copy()
                cur_seq = shake_seq
                cur_score = shake_score
                print(f"Iter {it:07d} – shake improved score to {best_score:.6f}")

        # early exit if we already beat the target
        if best_score < 1.503:
            print(f"Target 1.503 reached at iteration {it}, score {best_score:.6f}")
            break

    print(f"Finished after {it} iterations – best score {best_score:.6f}")
    # final safety clip
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return best_seq.tolist()
