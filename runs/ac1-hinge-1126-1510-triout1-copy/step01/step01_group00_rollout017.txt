"""
Propose a step‑function sequence with a low evaluation score.

The algorithm combines several ideas that differ from the previous
sub‑gradient implementation:

*   A richer initialisation pool (power‑laws, symmetric Beta‑shapes,
    exponential decay and random vectors).  Symmetric Beta families
    include the classic arcsine‐law (α=β=0.5) which is known to flatten
    self‑convolution.

*   An adaptive AdaGrad projected gradient descent on the probability
    simplex.  The per‑coordinate step size  η / (√G + ε) automatically
    shrinks directions with large accumulated gradients, allowing larger
    moves in flat directions.

*   Occasional random perturbations when progress stalls, helping to
    escape shallow local minima.

*   Time‑aware loops – the whole search respects the given budget
    (default 1000 s).

The objective (scale‑invariant) is
    score = 2·n·max(conv(a,a)) / (sum(a)²)  = 2·n·max(conv(p,p))
for a probability vector p (sum(p)=1).  Hence we work directly with a
normalized vector and project onto the simplex after each update.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog  # optional, kept for possible extensions
from scipy.sparse import diags


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (identical to evaluate_sequence)."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    fft_seq = np.fft.fft(seq, n=L)
    conv = np.real(np.fft.ifft(fft_seq * fft_seq))[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the time budget and prints progress.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Choose the length.  Use the previously found construction if present.
    try:
        n = len(height_sequence_1)   # type: ignore
        print(f"Using length from height_sequence_1: n = {n}")
    except Exception:
        n = 6000           # fallback length
        print(f"No global sequence found – using default length n = {n}")

    target_sum = math.sqrt(2.0 * n)          # normalisation used by the reference

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Helper factories for families of initial vectors (sum = 1)
    ind = np.arange(1, n + 1, dtype=np.float64)

    def make_powerlaw(p):
        w = np.power(ind, p)
        return w / w.sum()

    def make_beta(alpha, beta):
        # p_i ∝ i^{α-1} (n-i+1)^{β-1}
        w = np.power(ind, alpha - 1.0) * np.power(n - ind + 1.0, beta - 1.0)
        # protect against infinities (α or β may be <1)
        w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
        if w.sum() == 0.0:
            w = np.ones_like(w)
        return w / w.sum()

    def make_exponential(lam):
        w = np.exp(-lam * (ind - 1.0)) + np.exp(-lam * (n - ind))
        return w / w.sum()

    def make_arcsine():
        w = 1.0 / np.sqrt(ind * (n - ind + 1.0))
        return w / w.sum()

    # ------------------------------------------------------------------
    # Phase 1 – a diverse set of starting points
    candidates = []

    # Power‑law (both increasing and decreasing)
    for p in np.linspace(-2.0, 2.0, 41):
        candidates.append(make_powerlaw(p))

    # Symmetric Beta shapes (including arcsine and uniform)
    alphas = np.linspace(0.2, 2.0, 10)   # 0.2 … 2.0
    betas = alphas
    for a_val in alphas:
        for b_val in betas:
            candidates.append(make_beta(a_val, b_val))

    # Exponential decays (symmetric)
    for lam in [0.001, 0.005, 0.01, 0.02, 0.05]:
        candidates.append(make_exponential(lam))

    # Explicit arcsine (α = β = 0.5)
    candidates.append(make_arcsine())

    # Random vectors (few, to keep runtime modest)
    for _ in range(15):
        w = rng.random(n)
        candidates.append(w / w.sum())

    print(f"--- Phase 1: evaluating {len(candidates)} candidate shapes ---")
    for idx, a in enumerate(candidates):
        if time.time() > deadline:
            break
        sc = _score(a * target_sum)
        if sc < best_score:
            best_score = sc
            best_seq = a * target_sum
            print(f"[init {idx:03d}] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – AdaGrad projected sub‑gradient descent
    if best_seq is None:
        # fallback to uniform distribution
        a = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = a * target_sum
        best_score = _score(best_seq)
        print("Fallback to uniform distribution.")
    else:
        a = best_seq / best_seq.sum()   # work on the normalized probability vector

    # AdaGrad state
    G = np.zeros(n, dtype=np.float64)    # accumulated squared gradients
    eps = 1e-12
    eta0 = 0.8                           # initial global step factor
    max_iters = 20000
    it = 0
    no_improve = 0
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    print("--- Phase 2: AdaGrad sub‑gradient descent ---")
    while time.time() < deadline and it < max_iters:
        # ----- self‑convolution (FFT) -----
        fft_a = np.fft.fft(a, n=L_conv)
        conv_full = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        max_val = conv_full.max()

        # ----- active set (indices touching the maximum) -----
        tol = max_val * 1e-12 + 1e-14
        act = (conv_full >= max_val - tol).astype(np.float64)

        # ----- gradient of the maximal entry (correlation) -----
        fft_act = np.fft.fft(act, n=L_corr)
        fft_a_rev = np.fft.fft(a[::-1], n=L_corr)
        corr_full = np.real(np.fft.ifft(fft_act * fft_a_rev))
        grad = 2.0 * corr_full[n - 1:n - 1 + n]   # length‑n sub‑gradient

        # ----- AdaGrad step (per‑coordinate learning rates) -----
        G += grad * grad
        step = eta0 / (np.sqrt(G) + eps)          # element‑wise step sizes
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # ----- evaluate -----
        cur_score = _score(a_new * target_sum)
        if cur_score + 1e-9 < best_score:  # strict improvement
            best_score = cur_score
            best_seq = a_new * target_sum
            no_improve = 0
            print(f"Iter {it:5d}: improved score = {best_score:.6f}")
        else:
            no_improve += 1

        # ----- occasional random perturbation to escape plateaus -----
        if no_improve >= 3000:
            # add a tiny jitter and re‑project
            jitter = 0.001 * rng.random(n)
            a_new = _project_simplex(a_new + jitter)
            no_improve = 0
            print(f"Iter {it:5d}: jitter applied to escape stagnation")

        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (plain Python list)
    final_seq = best_seq.tolist()
    print(f"--- Finished: best score = {best_score:.6f} after {it} iterations ---")
    return final_seq
