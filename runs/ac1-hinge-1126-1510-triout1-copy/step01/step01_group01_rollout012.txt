def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simulated-annealing style optimizer that searches over
    probability distributions of a given length to minimise
    2*n*max(conv(a,a)), i.e. the evaluation function.
    Starts from a family of Beta(α,α) (arcsine‑like) sequences,
    runs a parameter sweep for different shapes, then refines the
    best candidate by many small mass‑transfer moves using an
    incremental convolution update.
    """
    import time, numpy as np
    start = time.time()
    np.random.seed(seed)

    # choose length n (tune for speed vs precision)
    n = 2000
    m = 2*n - 1
    nfft = 1 << ((m-1).bit_length())

    # Helper definitions
    def conv_max(arr):
        fx = np.fft.rfft(arr, n=nfft)
        conv = np.fft.irfft(fx * fx, n=nfft)[:m]
        return conv.max()

    # 1. PARAMETER SWEEP
    best_seq = None
    best_score = np.inf

    # Beta family (symmetric)
    alphas = np.concatenate([np.linspace(0.05, 0.9, 150),
                             np.linspace(0.9, 2.0, 150)])
    x = (np.arange(n) + 0.5) / n  # central points in (0,1)
    for a in alphas:
        # shape: x^{a-1} (1-x)^{a-1}
        # avoid overflow: compute using log
        log_vals = (a - 1) * (np.log(x) + np.log(1 - x))
        vals = np.exp(log_vals)
        seq = vals / vals.sum()
        # Ensure non‑negative and finite
        if not np.isfinite(seq).all():
            continue
        score = 2.0 * n * conv_max(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            # print progress
            print(f"[sweep] α={a:.3f}, score={score:.6f}")

    # Additional families can be added similarly if time left
    # We'll add exponential family
    if time.time() - start < budget_s * 0.2:  # reserve a bit of time
        gammas = np.linspace(0.001, 0.02, 50)
        for g in gammas:
            vals = np.exp(-g * np.arange(n))
            vals = vals + vals[::-1]  # make it symmetric
            seq = vals / vals.sum()
            score = 2.0 * n * conv_max(seq)
            if score < best_score:

"""
Propose a candidate step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a))²

where `conv` is the linear convolution.  The algorithm works in two phases:

1. **Parametric sweep** – generate many candidate families (Beta,
   mixtures of the arcsine law with uniform, exponential and power‑law
   shapes) on a moderate length (≈2000).  The best sequence from this
   sweep is used as a starting point.

2. **Simulated‑annealing local search** – repeatedly move a small amount of
   mass from a source index to a destination index.  The convolution
   is updated incrementally (O(n) work) using the exact formula

        c_new = c + 2·a·Δ + Δ·Δ

   where Δ has two non‑zero entries (−δ at the source, +δ at the
   destination).  Moves are accepted if they improve the score or,
   with a temperature that cools linearly, if they are slightly worse.
   The step size is reduced when the search stagnates.

The whole routine respects the 1000 s time budget, prints occasional
progress messages and returns the best sequence found (as a plain
Python list of floats).  The returned vector is normalised to L¹‑norm 1,
which makes the score identical to `evaluate_sequence`.
"""

import time
import numpy as np


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients:  binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _conv_max_fft(a: np.ndarray, nfft: int, m: int) -> float:
    """Maximum entry of a*a using FFT (fast for the sweep phase)."""
    fx = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return float(conv.max())


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Simulated‑annealing optimiser for the step‑function evaluation.
    Returns a list of non‑negative heights (the best sequence found
    before the time budget expires).
    """
    start_time = time.time()
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # Basic parameters – they balance speed and resolution.
    # -----------------------------------------------------------------
    n = 2000                     # length of the candidate sequence
    m = 2 * n - 1                # length of the self‑convolution
    nfft = 1 << ((m - 1).bit_length())   # next power‑of‑two for FFT

    # -----------------------------------------------------------------
    # Helper to evaluate a sequence (assumes sum≈1, which we keep).
    # -----------------------------------------------------------------
    def score_of(a: np.ndarray) -> float:
        """Return 2·n·max(conv(a,a))."""
        return 2.0 * n * _conv_max_fft(a, nfft, m)

    # -----------------------------------------------------------------
    # 1️⃣  PARAMETRIC SWEEP – get a good starting point.
    # -----------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ---- (a) symmetric Beta(α,α) family --------------------------------
    alphas = np.concatenate([np.linspace(0.05, 0.9, 120),
                             np.linspace(0.9, 2.0, 120)])
    x = (np.arange(n) + 0.5) / n                     # points in (0,1)
    for α in alphas:
        # log‑space to avoid overflow/underflow
        log_vals = (α - 1.0) * (np.log(x) + np.log(1.0 - x))
        seq = np.exp(log_vals)
        if not np.isfinite(seq).all():
            continue
        seq /= seq.sum()
        sc = score_of(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[sweep] Beta α={α:.3f} → score={sc:.6f}")

    # ---- (b) mixture of arcsine law with uniform -----------------------
    arcsine = 1.0 / np.sqrt(x * (1.0 - x))
    arcsine /= arcsine.sum()
    uniform = np.ones(n) / n
    for lam in np.linspace(0.0, 1.0, 21):
        seq = lam * arcsine + (1.0 - lam) * uniform
        seq /= seq.sum()
        sc = score_of(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[sweep] mix λ={lam:.2f} → score={sc:.6f}")

    # ---- (c) symmetric exponential decay --------------------------------
    for γ in np.linspace(0.001, 0.02, 30):
        vals = np.exp(-γ * np.arange(n))
        vals = vals + vals[::-1]          # make it symmetric
        seq = vals / vals.sum()
        sc = score_of(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[sweep] expo γ={γ:.5f} → score={sc:.6f}")

    # ---- (d) symmetric power‑law ---------------------------------------
    for p in np.linspace(0.5, 3.0, 30):
        vals = np.arange(1, n + 1, dtype=np.float64) ** (-p)
        vals = vals + vals[::-1]
        seq = vals / vals.sum()
        sc = score_of(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[sweep] power p={p:.2f} → score={sc:.6f}")

    # ---- (e) random Dirichlet draws (quick stochastic exploration) -----
    for _ in range(200):
        seq = np.random.dirichlet(np.ones(n))
        sc = score_of(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"[sweep] rand Dirichlet → score={sc:.6f}")

    # ---- fallback (very unlikely) ---------------------------------------
    if best_seq is None:
        raw = _catalan_sequence(n)
        best_seq = raw / raw.sum()
        best_score = score_of(best_seq)
        print(f"[fallback] Catalan init score={best_score:.6f}")

    print(f"[init] length={n}, best_score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 2️⃣  LOCAL SEARCH – simulated annealing with incremental updates.
    # -----------------------------------------------------------------
    a_cur = best_seq.copy()                 # current candidate
    c_cur = np.convolve(a_cur, a_cur)       # exact convolution (length 2n‑1)
    cur_score = 2.0 * n * c_cur.max()

    # Sanity check (they should match up to numerical noise)
    if abs(cur_score - best_score) > 1e-12:
        best_score = cur_score
        best_seq = a_cur.copy()

    # Annealing schedule
    deadline = start_time + budget_s - 2.0  # a small safety margin
    T0 = best_score * 0.01                 # start temperature (≈1 % of score)
    step_frac = 0.05                       # up‑to‑5 % of source mass per move
    max_no_improve = 5000                  # after this we shrink the step

    it = 0
    no_improve = 0

    while time.time() < deadline:
        it += 1

        # ----- pick two distinct indices --------------------------------
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a_cur[i] <= 0.0:
            continue

        # ----- random amount to move ------------------------------------
        delta = step_frac * a_cur[i] * np.random.rand()
        if delta <= 0.0:
            continue

        # ----- propose the move (copy‑on‑write) -------------------------
        a_new = a_cur.copy()
        c_new = c_cur.copy()

        # linear part: 2·a·Δ  (Δ has -δ at i, +δ at j)
        c_new[i:i + n] += -2.0 * delta * a_new
        c_new[j:j + n] +=  2.0 * delta * a_new

        # quadratic part: Δ·Δ
        c_new[2 * i] += delta * delta
        c_new[2 * j] += delta * delta
        c_new[i + j] -= 2.0 * delta * delta

        # update the probability vector
        a_new[i] -= delta
        a_new[j] += delta

        # ----- evaluate the new candidate --------------------------------
        new_score = 2.0 * n * c_new.max()

        # ----- temperature (linear cooling) -----------------------------
        remaining = deadline - time.time()
        temperature = T0 * max(0.0, remaining / (budget_s - 2.0))

        # ----- Metropolis acceptance rule --------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        elif temperature > 0.0:
            prob = np.exp(-(new_score - cur_score) / temperature)
            if np.random.rand() < prob:
                accept = True

        # ----- commit or reject -------------------------------------------
        if accept:
            a_cur = a_new
            c_cur = c_new
            cur_score = new_score
            no_improve = 0
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[improve] it={it:,} t={int(time.time() - start_time)}s "
                      f"score={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= max_no_improve:
                step_frac *= 0.5
                no_improve = 0
                if step_frac < 1e-6:        # step too small → stop early
                    break

    print(f"[final] length={n}, best_score={best_score:.6f}, iterations={it}")

    # Final normalisation (protect against tiny drift)
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq = np.ones(n) / n
    else:
        best_seq = best_seq / total

    # Return a plain Python list as required by the external driver.
    return best_seq.tolist()
