Algorithm
---------
1. **Initialisation** – start from the best known construction
   (`height_sequence_1` if present) otherwise from the
   “sqrt‑difference” profile.  Normalise the L1 norm to 1
   (the objective is scale‑invariant).

2. **LP improvement phase** – repeated:
      a. Compute the current convolution peak `M = max(a*a)`.
      b. Build the linear program  
         maximise   Σ g_i
         subject to  (a * g)_k ≤ M   for all k = 0 … 2n‑2,
                     g_i ≥ 0.
         The matrix of the constraints is a Toeplitz matrix whose
         columns are shifted copies of `a`.
      c. Solve the LP (SciPy HiGHS).  Let `g0` be the optimum and
         `S = Σ g0`.  Scale it to the same L1 norm as `a`:
         `g = g0 / S`.
      d. Find the largest mixing factor `t ∈ (0,1]` such that
         `max( ((1‑t)·a + t·g) * ((1‑t)·a + t·g) ) ≤ M`.
         A short binary search (10 steps) is used.
      e. Replace `a ← (1‑t)·a + t·g` (renormalise to sum = 1) and
         keep the best sequence seen.

   The LP step can increase the total mass under the same
   convolution bound; mixing then strictly reduces the peak.
   The loop stops when time is exhausted or when a step cannot
   improve the peak.

4. **Return** – the best sequence found is rescaled to the full
   `[0, 1000]` range (scaling does not affect the score) and

def _conv_max(seq: np.ndarray) -> float:
    """Return the maximum value of the full self‑convolution of `seq`."""
    return np.convolve(seq, seq).max()


def _build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g)[k] = (a * g)[k].
    Each column j of A is a shifted copy of `a`.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contributes a shifted by j rows (from j to j+n‑1)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _mix_factor(a: np.ndarray, g: np.ndarray, M_target: float,
                max_iter: int = 10, eps: float = 1e-12) -> float:
    """
    Find the largest t ∈ (0, 1] such that the self‑convolution peak of
    (1‑t)·a + t·g does not exceed M_target (up to a tiny tolerance).
    Simple binary search; returns 0 when no feasible t > 0 exists.
    """
    lo, hi = 0.0, 1.0
    best = 0.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        mixed = (1.0 - mid) * a + mid * g
        mixed /= mixed.sum()          # keep L1 = 1
        if _conv_max(mixed) <= M_target * (1.0 + 1e-9):
            best = mid
            lo = mid
        else:
            hi = mid
        if hi - lo < eps:
            break
    return best


def _lp_step(a: np.ndarray, M: float, linprog_solver) -> np.ndarray | None:
    """
    Perform one LP step: maximise Σ g_i under (a * g) ≤ M.
    Returns the optimal g (not yet normalised) or None on failure.
    """
    n = a.shape[0]
    # Build constraint matrix A (shape (2n‑1, n))
    A = _build_constraint_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # Objective: maximise Σ g_i  →  minimise -Σ g_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # Solve with HiGHS (fast interior‑point) – SciPy’s default method
    res = linprog_solver(c, A_ub=A, b_ub=b, bounds=bounds,
                         method="highs", options={"presolve": True})
    if not res.success:
        return None
    return res.x


def _fallback_gradient_descent(initial, time_limit):
    """
    Simple projected gradient descent on the peak index (the method used
    in the previous run).  Used only when SciPy is unavailable.
    """
    a = initial.copy()
    n = a.size
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    start = time.time()
    step_scale = 0.02
    temperature = 1.0
    min_step = 1e-5

    while time.time() - start < time_limit:
        # locate peak index
        conv = np.convolve(a, a)
        k_max = int(conv.argmax())

        # gradient of the peak w.r.t. a
        grad = np.zeros_like(a)
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        if i_low <= i_high:
            i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
            j_vals = k_max - i_vals
            grad[i_vals] = a[i_vals] + a[j_vals]

        # zero‑sum direction (preserve total mass)
        direction = -grad + grad.mean()

        step = step_scale * temperature * (np.linalg.norm(direction) / n + 1e-12)
        step = max(step, min_step)

        a_cand = a + step * direction
        a_cand = np.clip(a_cand, 0.0, None)
        if a_cand.sum() == 0.0:
            continue
        a_cand /= a_cand.sum()

        score = evaluate_sequence(a_cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_cand.copy()
            a = a_cand
            step_scale = max(step_scale * 0.99, min_step)
            temperature = max(temperature * 0.99, 0.1)
            continue

        # occasional uphill move
        if np.random.rand() < 0.02 * temperature:
            a = a_cand

        # tiny random jitter
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            delta = a[idx] * (np.random.rand() - 0.5) * 0.1
            a[idx] = max(0.0, a[idx] + delta)
            a /= a.sum()

    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    `evaluate_sequence`.  The routine first tries an LP‑based
    monotone‑decreasing peak refinement, then falls back to a short
    stochastic gradient descent / local‑move polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for finalisation

    # -------------------- 1. Initialise --------------------
    try:
        # use the globally supplied construction if it exists
        init = np.array(height_sequence_1, dtype=np.float64)
        n = init.size
    except Exception:
        # default to sqrt‑difference shape (known good start)
        n = 1542
        init = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - \
               np.sqrt(np.arange(0, n, dtype=np.float64))

    # enforce non‑negativity, normalise L1 to 1 (scale‑invariant)
    init = np.clip(init, 0.0, None)
    if init.sum() == 0.0:
        init[0] = 1.0
    a = init / init.sum()

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init]  L1={a.sum():.6f}  n={n}  score={best_score:.6f}")

    # -------------------- 2. LP improvement phase --------------------
    # try to import SciPy lazily; if not available we fall back.
    try:
        from scipy.optimize import linprog as _linprog
    except Exception:
        _linprog = None

    if _linprog is not None:
        # parameters for the iterative scheme
        t_initial = 0.5          # start mixing weight
        t_min = 1e-3
        mixing_decay = 0.95      # reduce t a bit after each successful step
        lp_max_iter = 30         # cap on LP steps (time‑controlled anyway)

        lp_iteration = 0
        while time.time() < deadline and lp_iteration < lp_max_iter:
            M = _conv_max(a)                 # current peak
            # solve LP: maximise Σ g_i s.t. a * g ≤ M
            g0 = _lp_step(a, M, _linprog)
            if g0 is None:
                print("[LP]  Solver failed – breaking LP phase.")
                break
            S = g0.sum()
            if S <= 1e-12:
                print("[LP]  Trivial solution – breaking.")
                break

            # Scale to same L1 norm as `a` (which is 1)
            g = g0 / S

            # Find feasible mixing factor
            t = t_initial
            t_feas = _mix_factor(a, g, M)
            if t_feas < t_min:
                # No feasible mixing – reduce ambition and retry
                t_initial *= 0.5
                if t_initial < t_min:
                    print("[LP]  Mixing weight too small – terminating LP phase.")
                    break
                continue

            # Apply mixing
            a_new = (1.0 - t_feas) * a + t_feas * g
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() == 0.0:
                break
            a_new /= a_new.sum()

            # Evaluate improvement
            new_score = evaluate_sequence(a_new.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f"[LP]  Iter {lp_iteration:02d}  t={t_feas:.4f}  "
                      f"score={best_score:.6f}  time={time.time()-start_time:.1f}s")

            # Accept the step (the mixing factor guarantees non‑increase of peak)
            a = a_new
            # slowly decrease the aggressive mixing weight
            t_initial = max(t_initial * mixing_decay, t_min)
            lp_iteration += 1

    else:
        print("[info] SciPy not available – skipping LP phase.")

    # -------------------- 3. Gradient descent / local refinement --------------------
    # Use a short gradient‑descent run if any time remains.
    remaining = deadline - time.time()
    if remaining > 5.0:
        gd_time = min(remaining - 4.0, 30.0)  # leave a few seconds for polishing
        print(f"[gd]  Running fallback gradient descent for {gd_time:.1f}s")
        best_from_gd = _fallback_gradient_descent(best_seq, gd_time)
        gd_score = evaluate_sequence(best_from_gd.tolist())
        if gd_score < best_score:
            best_score = gd_score
            best_seq = best_from_gd.copy()
            print(f"[gd]  Improved score to {best_score:.6f}")

    # -------------------- 4. Final random‑move polishing --------------------
    # Short stochastic hill‑climbing (≈5 s) to squeeze out remaining gains.
    polish_deadline = min(deadline, time.time() + 5.0)
    a = best_seq.copy()
    while time.time() < polish_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = min(a[i], a[i] * 0.02)
        if eps <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new /= a_new.sum()
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            a = a_new
            print(f"[polish] improved to {best_score:.6f}")

    # -------------------- 5. Rescale to [0,1000] and return --------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"[final] returning sequence with score {best_score:.6f}")
    return final_seq.tolist()
