"""
Propose a low‑evaluation step sequence.

The algorithm builds on the smooth‑max L‑BFGS approach but adds
a multi‑stage annealing schedule for the smoothing factor,
several diversified random initialisations, a fast greedy hill‑climb
and a simple sub‑gradient descent on the true max‑convolution.
A larger problem size (n≈4000) is used to obtain a finer
approximation of the continuous optimum.  The search runs until the
time budget expires and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using FFTs.
    a must have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Quick evaluation of the objective for a sequence that already sums to 1:
    score = 2 * n * max(conv(a,a)).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective (2*n*smooth_max(conv)) and its gradient
    with respect to the unconstrained vector y (a = softmax(y)).
    """
    a = softmax(y)                       # simplex point, sum(a)=1

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # g_vec_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # needed slice corresponds to indices (n‑1)…(n‑1+n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple greedy hill‑climb on the simplex.
    Randomly moves a fraction of mass from one coordinate to another.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 5 % of the mass at i is transferred
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    # reference: Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_descent(a: np.ndarray,
                        fft_len: int, n: int,
                        deadline: float,
                        max_iter: int = 300,
                        step0: float = 0.1) -> (np.ndarray, float):
    """
    Sub‑gradient descent on the true (non‑smooth) max‑convolution.
    Uses a simple diminishing step size and projects back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        # exact convolution and index of the maximum
        conv = np.convolve(best, best)
        k_star = int(np.argmax(conv))

        # gradient of max(conv) w.r.t. a
        g = np.zeros_like(best)
        idx = np.arange(n)
        k_minus_idx = k_star - idx
        mask = (k_minus_idx >= 0) & (k_minus_idx < n)
        g[mask] = 2.0 * best[k_minus_idx[mask]]

        # diminishing step size
        step = step0 / np.sqrt(iter_cnt + 1)

        a_new = best - step * g
        a_new = project_onto_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best = a_new
        iter_cnt += 1
    return best, best_score


def generate_initial_a(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random feasible sequence (sum = 1, non‑negative) for a given mode."""
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)

    elif mode == 'random':
        a = rng.random(n)
        a /= a.sum()

    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()

    elif mode == 'power':
        p = rng.uniform(0.3, 2.0)
        a = (np.arange(1, n + 1) ** (-p))
        a /= a.sum()

    elif mode == 'edge':
        m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, a_center)
        a[:m_edge] = a_edge
        a[-m_edge:] = a_edge
        a /= a.sum()

    elif mode == 'sin':
        x = np.arange(n) + 0.5
        a = np.sin(np.pi * x / n)
        a = np.maximum(a, 0.0)
        a /= a.sum()

    elif mode == 'gaussian':
        sigma = rng.uniform(0.08 * n, 0.20 * n)
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)
        a /= a.sum()

    else:
        # fallback: pure random
        a = rng.random(n)
        a /= a.sum()

    return a


def optimize_with_schedule(a_start: np.ndarray,
                           schedule: list[float],
                           n: int,
                           fft_len_conv: int,
                           fft_len_g: int) -> np.ndarray:
    """
    Run a short L‑BFGS optimisation for each λ‑factor in `schedule`,
    chaining the solution as the start for the next stage.
    """
    # initialise y = log(a) (softmax inverse)
    y = np.log(a_start + 1e-12)

    for lam_factor in schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam_factor),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 150,
                                'gtol': 1e-10,
                                'disp': False})
        y = res.x  # warm‑start next stage

    return softmax(y)

            print(f"[{iteration}] mode={mode:<9}  improved fast score = {best_score:.6f}")

        # ----- greedy hill‑climb (short, time‑bounded) -----
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=2000)

        # ----- sub‑gradient descent on the exact max -----
        sg_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = subgradient_descent(best_a,
                                                 fft_len_conv, n,
                                                 sg_deadline,
                                                 max_iter=300,
                                                 step0=0.08)

        iteration += 1

    # --------------------------------------------------------------
    # Final verification with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
