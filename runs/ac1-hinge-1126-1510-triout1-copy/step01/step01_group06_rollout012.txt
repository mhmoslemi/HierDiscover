def simplex_projection(v):
    """Return the Euclidean projection of v onto the probability simplex."""
    # v is 1-D np.ndarray
    # algorithm: sort v in descending order.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_gradient(a, active_idxs):
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for k in active_idxs:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # i range
        i_vals = np.arange(i_low, i_high + 1)
        j_vals = k - i_vals
        # j_vals in [0, n-1]
        grad[i_vals] += 2.0 * a[j_vals]
    return grad

    a = a0.copy()
    cur_score = evaluate_sequence(a.tolist())
    best_seq_local = a.copy()
    best_score_local = cur_score

    n = a.shape[0]
    start = time.time()
    deadline = start + time_budget

    # hyperparameters
    step_scale = 0.2  # initial step scale
    min_step = 1e-7
    max_step = 5.0
    temperature = 1.0
    # we can track no improvement steps to reduce step
    iter_cnt = 0
    while time.time() < deadline and time.time() < global_deadline:
        # conv and gradient
        conv = np.convolve(a, a)
        max_b = conv.max()
        # find active indices (within absolute tolerance)
        # tolerance:
        tol = 1e-12
        active_idxs = np.where(conv >= max_b - tol)[0]
        # compute gradient
        grad = _gradient_from_active(a, active_idxs)
        # compute direction preserving sum
        direction = -grad + grad.mean()
        # step size
        step = step_scale * np.linalg.norm(direction)
        step = np.clip(step, min_step, max_step)

        candidate = a + step * direction
        # enforce non-negative and sum 1 via clipping+renorm
        candidate = np.maximum(candidate, 0.0)
        total = candidate.sum()
        if total <= 0:
            # fallback to uniform
            candidate = np.full_like(candidate, 1.0 / n)
            total = 1.0
        candidate /= total

        # acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # simulated annealing
            if temperature > 1e-8:
                prob = math.exp(-(cand_score - cur_score) / temperature)

"""
Propose a coefficient sequence that minimizes the evaluation function.

The algorithm works in three phases:

1. **Diverse initialisations** – a small grid of analytic families
   (square‑root‑difference, power‑law, exponential) together with a few
   random Dirichlet vectors are normalised to sum to one and evaluated.
   The best few are kept for further optimisation.

2. **Projected sub‑gradient descent** – starting from each selected
   initial vector we repeatedly minimise the infinity‑norm of the
   autocorrelation (the quantity `max(conv(a,a))`).  For the active
   convolution indices (the ones attaining the current maximum) the exact
   gradient is known analytically:
   `∂max/∂a_i = 2 * a_{k-i}`.  The direction is orthogonalised by
   subtracting its mean so that the sum of the coefficients stays constant.
   After a step we project back onto the probability simplex (non‑negative
   and unit‑sum) and accept the move if it improves the score or with a
   simulated‑annealing probability.

3. **Polishing** – a short random‑move local search that only keeps
   improving steps.

All numbers are kept non‑negative and the final vector is scaled to the
full `[0,1000]` range (the metric is homogeneous, this does not change the
score).  The routine respects the 1000‑second time budget.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    `{x | x >= 0, sum(x) = 1}`.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _gradient_from_active(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Exact gradient of the (sub‑)function `max_k conv(a,a)[k]`
    with respect to `a`, summed over the active indices `active_idxs`.
    The formula is:
        d conv_k / d a_i = 2 * a_{k-i}   (if 0 ≤ k-i < n, else 0)
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        i = np.arange(i_low, i_high + 1, dtype=np.int64)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _run_local_search(a_start: np.ndarray, allocated_seconds: float,
                      global_deadline: float, polish: bool = False):
    """
    Sub‑gradient descent with optional final polishing.
    Returns the best sequence found and its score.
    """
    a = a_start.copy()
    cur_score = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    best_score = cur_score

    n = a.shape[0]

    # hyper‑parameters (they adapt during the run)
    step_scale = 0.2          # base step factor
    min_step = 1e-7
    max_step = 5.0
    temperature = 1.0
    it = 0

    start = time.time()
    deadline = start + allocated_seconds

    while time.time() < deadline and time.time() < global_deadline:
        # 1) compute convolution and the active set
        conv = np.convolve(a, a)
        max_b = conv.max()
        # numerical tolerance – treat near‑max values as active
        active = np.where(conv >= max_b - 1e-12)[0]

        # 2) gradient of the active peak(s)
        grad = _gradient_from_active(a, active)

        # 3) direction that keeps the sum of a unchanged
        direction = -grad + grad.mean()          # sum(direction) == 0

        # 4) step length (scaled by the norm of the direction)
        step = step_scale * np.linalg.norm(direction)
        step = np.clip(step, min_step, max_step)

        # 5) candidate and projection onto the simplex
        cand = a + step * direction
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0.0:
            # fall back to a tiny uniform perturbation
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand /= cand.sum()

        cand_score = evaluate_sequence(cand.tolist())

        # acceptance rule (deterministic improvement + SA)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if temperature > 1e-8:
                prob = math.exp(-(cand_score - cur_score) / temperature)
                if random.random() < prob:
                    accept = True

        if accept:
            a = cand
            cur_score = cand_score
            # remember global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                # become a bit more aggressive
                step_scale = min(step_scale * 1.05, max_step)
                temperature = max(temperature * 0.99, 0.001)
        else:
            # on rejection shrink the step size a little
            step_scale = max(step_scale * 0.95, min_step)

        # occasional random “mass‑transfer” move – helps to escape flat regions
        if random.random() < 0.02:
            i, j = random.sample(range(n), 2)
            eps = min(a[i], a[i] * random.random() * 0.1)
            if eps > 0:
                cand2 = a.copy()
                cand2[i] -= eps
                cand2[j] += eps
                cand2 = np.maximum(cand2, 0.0)
                cand2 /= cand2.sum()
                cand2_score = evaluate_sequence(cand2.tolist())
                if cand2_score < cur_score or (
                        temperature > 0 and random.random() <
                        math.exp(-(cand2_score - cur_score) / temperature)):
                    a = cand2
                    cur_score = cand2_score
                    if cand2_score < best_score:
                        best_score = cand2_score
                        best_seq = cand2.copy()
                        step_scale = min(step_scale * 1.02, max_step)

        # cooling schedule
        it += 1
        if it % 100 == 0:
            temperature = max(temperature * 0.95, 1e-4)

    # ------------------------------------------------------------------
    # Optional polishing phase (pure improvement moves only)
    # ------------------------------------------------------------------
    if polish:
        polish_deadline = time.time() + max(1.0, (global_deadline - time.time()) / 2)
        a = best_seq.copy()
        cur_score = best_score
        while time.time() < polish_deadline:
            i, j = random.sample(range(n), 2)
            eps = min(a[i], a[i] * random.random() * 0.05)
            if eps <= 0:
                continue
            cand = a.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            cand_score = evaluate_sequence(cand.tolist())
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(f'Polish improved: {best_score:.6f}')

    return best_seq, best_score

def _init_sqrt_diff(n: int) -> np.ndarray:
    """Square‑root‑difference profile (the classic starting point)."""
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(
        np.arange(0, n, dtype=np.float64))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _init_power(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """Power‑law decay a_i ∝ (i + offset + 1)^(-p)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1.0) ** (-p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _init_exponential(n: int, decay: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-decay * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-decay * i)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _init_dirichlet(n: int, alpha: float = 1.0) -> np.ndarray:
    """Random vector drawn from a Dirichlet(alpha) distribution."""
    v = np.random.gamma(alpha, 1.0, size=n)
    v /= v.sum()
    return v

            best_score = sc
            best_seq = seq.copy()
            print(f'  New global best from {init_name}: {best_score:.6f}')

    # ------------------------------------------------------------------
    # final polishing phase (pure improvement moves)
    # ------------------------------------------------------------------
    time_left = deadline - time.time()
    if time_left > 0:
        print(f'\nPolishing phase with ~{time_left:.1f}s remaining')
        best_seq, best_score = _run_local_search(best_seq, time_left,
                                                 deadline, polish=True)

    # ------------------------------------------------------------------
    # scale to the [0,1000] range (does not affect the score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f'\nReturning final sequence with score {best_score:.6f}')
    return final_seq.tolist()
