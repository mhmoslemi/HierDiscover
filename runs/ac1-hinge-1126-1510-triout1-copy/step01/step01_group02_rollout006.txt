def simplex_projection(v):
    # returns projection onto the probability simplex
    n = v.shape[0]
    # sort
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
```

"""
Gradient‐descent on the max‑convolution peak.

The evaluation objective is
    E(a) = 2·n·max(a*a) / (∑a)² .
Because E is homogeneous of degree 0, we may restrict ourselves to the
probability simplex (non‑negative entries that sum to 1).  For a given
sequence a the convolution peak is attained at some index k*.  The
partial derivative of max(a*a) w.r.t. a_i (when the peak occurs at k*)
is 2·a_{k*‑i} (or 0 if the partner index lies outside the support).
Using the chain rule we obtain the gradient of the full objective.
After centring the gradient (to keep the total mass constant) we take a
small step opposite to the gradient, clip negatives and project back onto
the simplex.  A simple simulated‑annealing acceptance rule and occasional
random “shakes’’ help to escape plateaus.

The routine runs for the allotted wall‑clock time (≈1000 s) and returns
the best sequence found.  It starts from the Catalan (half‑binomial)
coefficients – the classical construction – but works with any
initialisation that is present in the global variable ``height_sequence_1``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence, but without
# the safety checks).  This is used repeatedly inside the optimiser.
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a 1‑D numpy array."""
    n = seq.size
    s = seq.sum()
    if s < 0.01:          # protect against division by zero
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):            # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    # maximum total L1 change per iteration (adapted linearly)
    max_move_initial = 0.02        # move ≈2 % of the total mass at the start

    while time.time() < deadline:
        it += 1

        # ---- compute the current convolution peak ---------------------------------
        f = np.fft.rfft(cur_seq, n=L)
        conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
        M = conv.max()
        k_star = int(conv.argmax())
        S = cur_seq.sum()          # should be 1.0 (up to round‑off)

        # ---- gradient of the objective (centered to keep sum constant) ----------
        indices = np.arange(n)
        partners = k_star - indices
        mask = (partners >= 0) & (partners < n)
        g = np.zeros_like(cur_seq)
        g[mask] = 2.0 * cur_seq[partners[mask]]

        c1 = 2.0 * n / (S * S)
        c2 = 4.0 * n * M / (S * S * S)
        raw_grad = c1 * g - c2
        raw_grad -= raw_grad.mean()          # enforce zero‑sum direction

        # ---- propose a new point -------------------------------------------------
        # direction opposite to the gradient
        direction = -raw_grad

        # limit total L1 movement (linearly decreasing)
        elapsed = time.time() - start_time
        fraction = max(0.0, 1.0 - elapsed / (budget_s - 5.0))
        max_move = max_move_initial * fraction
        total_move = np.abs(direction).sum()
        if total_move > max_move:
            direction = direction * (max_move / total_move)

        trial = cur_seq + direction
        trial = np.maximum(trial, 0.0)        # clip negatives
        trial = _project_simplex(trial)       # project back onto the simplex

        trial_score = _fast_score(trial)

        # ---- acceptance (simulated annealing) ------------------------------------
        if trial_score < cur_score:
            accept = True
        else:
            # temperature decays linearly to zero
            temp = init_temp * fraction
            if temp > 1e-12:
                prob = np.exp((cur_score - trial_score) / temp)
                accept = np.random.rand() < prob
            else:
                accept = False

        if accept:
            cur_seq = trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_seq = trial.copy()
                print(f"Iter {it:07d} – new best score: {best_score:.6f}")

        # ---- occasional random shake to avoid stagnation -------------------------
        if it % 3000 == 0:
            # tiny Gaussian perturbation on a random subset
            shake = cur_seq.copy()
            idxs = np.random.choice(n, size=min(100, n), replace=False)
            shake[idxs] += np.random.randn(len(idxs)) * 0.001
            shake = np.maximum(shake, 0.0)
            shake = _project_simplex(shake)
            shake_score = _fast_score(shake)
            if shake_score < best_score:
                best_score = shake_score
                best_seq = shake.copy()
                cur_seq = shake
                cur_score = shake_score
                print(f"Iter {it:07d} – shake improved score to {best_score:.6f}")

        # ---- early exit if target reached -----------------------------------------
        if best_score < 1.503:
            print(f"Target reached at iteration {it}, score {best_score:.6f}")
            break

    print(f"Finished after {it} iterations. Best score: {best_score:.6f}")
    # convert to plain Python list for the evaluator
    return best_seq.tolist()
```

This implementation follows the plan described in the docstring, runs a constrained gradient‑descent / simulated‑annealing loop for up to the allotted time, and returns the best step‑function found.
