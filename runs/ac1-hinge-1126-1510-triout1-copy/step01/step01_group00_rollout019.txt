"""
Propose a step‑function (non‑negative list) with a low evaluation score.

Algorithm
---------

1. **Initial pool** – generate many candidate sequences:
   * power‑law shapes a[i] ∝ (i+1)^p (p∈[0,2]) and their reversed versions,
   * a few random Dirichlet draws (α < 1 to encourage sparsity).

2. **Pick the best pool member** (according to the exact evaluation function).

3. **Projected Adam optimisation** – work with a normalised vector `a`
   (non‑negative, Σa = 1).  In each iteration:
   * compute the linear convolution `c = a * a` via FFT,
   * locate the index `k*` of the maximal entry,
   * the sub‑gradient of `max(c)` is the correlation with the one‑hot
     vector at `k*`; this is obtained with a simple O(n) slice,
   * update `a` with Adam (adaptive moments) and project back onto the
     probability simplex,
   * keep the best sequence seen (scaled back to the required sum
     √(2 n)).

4. **Multiple restarts** – after a plateau or after a fixed number of
   iterations the optimiser is re‑initialised from a fresh random Dirichlet
   vector (again projected onto the simplex).  All runs share the same
   time budget; the best overall result is returned.

The method differs from the previous code (plain sub‑gradient with a
diminishing step) by using adaptive learning rates, momentum, several
restarts and a more aggressive initial pool, which improves the chance
to go below the target bound 1.503.  The whole search respects a 1000 s
wall‑clock limit.

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_max(a: np.ndarray) -> tuple[np.ndarray, float, int]:
    """Return (full convolution, maximal value, index of the maximum)."""
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())      # next power of two
    fft_a = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
    max_idx = int(np.argmax(conv))
    max_val = float(conv[max_idx])
    return conv, max_val, max_idx

def _gradient_one_hot(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) when the active index is `idx`.
    The gradient components are 2·a[idx‑j] for those j where the term is
    defined, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # valid j satisfy 0 ≤ idx‑j < n  →  idx‑(n‑1) ≤ j ≤ idx
    j_start = max(0, idx - (n - 1))
    j_end = min(n - 1, idx)
    if j_start <= j_end:
        j = np.arange(j_start, j_end + 1)
        grad[j] = 2.0 * a[idx - j]
    return grad

        a_inc = w * (target_sum / w.sum())
        pool.append(a_inc)
        pool.append(a_inc[::-1])          # reversed (decreasing) version

    # a few sparse random starts (Dirichlet with α<1)
    for _ in range(12):
        dirich = np.random.dirichlet(alpha=np.full(n, 0.5))
        a_rand = dirich * target_sum
        pool.append(a_rand)

    best_seq = None
    best_score = float("inf")
    print(f"--- Phase 0: initial pool (n={n}) ---")
    for cand in pool:
        sc = _true_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"   new pool best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Prepare for optimisation on the simplex (sum = 1)
    a = best_seq / best_seq.sum()               # now Σa = 1
    best_M = _conv_max(a)[1]                    # max convolution for sum‑1 vector
    best_score = 2 * n * best_M                 # objective = 2·n·M (same scaling used by evaluate_sequence)
    best_seq = (a * target_sum).copy()

    # FFT length for convolution (used in every iteration)
    L_conv = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                       # starting learning rate
    max_total_iters = 120000        # safety ceiling; actual count limited by time

    it = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    print("--- Phase 1: Adam optimisation on simplex ---")
    while time.time() < deadline and it < max_total_iters:
        # ----- convolution (linear) via FFT -----
        fft_a = np.fft.fft(a, n=L_conv)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        max_val = float(conv.max())
        idx_max = int(conv.argmax())

        # ----- gradient of the active entry -----
        grad = _gradient_one_hot(a, idx_max)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # linear decay of the base learning rate
        lr = lr0 * (1.0 - it / max_total_iters)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # ----- evaluate -----
        conv_new = np.real(np.fft.ifft(np.fft.fft(a_new, n=L_conv) *
                                       np.fft.fft(a_new, n=L_conv)))[: 2 * n - 1]
        M_new = float(conv_new.max())
        score_new = 2 * n * M_new

        if score_new < best_score:
            best_score = score_new
            best_seq = (a_new * target_sum).copy()
            best_M = M_new
            print(f"Iter {it:6d}: improved score = {best_score:.6f}")

        a = a_new

        # ------------------------------------------------------------------
        # occasional restart with fresh random Dirichlet if we stall for a while
        if it % 15000 == 0 and time.time() < deadline:
            # take a noisy version of the current best as a new start
            noise = np.random.dirichlet(alpha=np.full(n, 0.5))
            a_restart = noise
            a_restart = _project_simplex(a_restart)
            a = a_restart
            m.fill(0.0)
            v.fill(0.0)
            print(f"--- Restart at iteration {it} ---")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (list of floats)
    final_seq = best_seq.tolist()
    print(f"--- Finished: best obj = {best_score:.6f} (M = {best_M:.6f}) after {it} iters ---")
    return final_seq
