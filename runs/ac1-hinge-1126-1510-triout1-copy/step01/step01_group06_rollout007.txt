    def gen_sqrt_diff(n):
        a = np.sqrt(np.arange(1, n+1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
        a = a / a.sum()
        return a

    def gen_power_raw(p, offset):
        i = np.arange(n, dtype=np.float64)
        a_raw = (i + offset) ** (-p)
        a_raw[0] = (offset) ** (-p) if offset>0 else 1.0
        a = a_raw / a_raw.sum()
        return a

    def gen_power_diff(p, offset):
        i = np.arange(n, dtype=np.float64)
        a = (i + offset) ** (-p) - (i+1 + offset) ** (-p)
        a = np.clip(a, 0, None)
        if a.sum() == 0:
            return None
        return a / a.sum()

    def gen_beta(alpha, beta_param):
        i = np.arange(1, n+1, dtype=np.float64)
        a_raw = (i**(alpha-1)) * ((n - i + 1) ** (beta_param-1))
        a_raw = np.clip(a_raw, 0, None)
        a = a_raw / a_raw.sum()

"""
Propose a coefficient sequence that minimizes the evaluation function.

The algorithm works in three stages:

1️⃣  **Parameter sweep** – generate candidates from a few simple
    parametric families (sqrt‑difference, power‑law, power‑law
    differences, Beta‑type) and keep the best one.

2️⃣  **Smoothed‑max projected gradient descent** – treat the maximum
    entry of the autocorrelation (a∗a) as a soft‑max (temperature
    controlled by a β parameter).  The gradient of the soft‑max is
    obtained efficiently via a single convolution, then a step is
    taken and the new point is projected onto the probability simplex
    (non‑negative, sum = 1).  β is increased during the run so the
    optimisation gradually focuses on the true maximum.

3️⃣  **Local polishing** – tiny random mass‑transfer moves are tried
    while the time budget allows; any improvement is kept.

All numbers stay non‑negative and the final sequence is scaled to the
full [0, 1000] range (the evaluation metric is scale‑invariant).
"""

import time
import numpy as np

def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself."""
    return np.convolve(a, a)

def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights for the entries of b.
    The maximum entry receives weight ≈1 when beta is large.
    """
    m = b.max()
    exp_beta = np.exp(beta * (b - m))          # shift for numerical stability
    return exp_beta / exp_beta.sum()

def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k (a∗a)[k] w.r.t. a.
    Equivalent to 2 * (w ∗ reversed(a))[n‑1 : n‑1+n].
    """
    n = a.shape[0]
    conv = np.convolve(w, a[::-1])
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Implementation follows the algorithm of
    Wang & Carreira‑Perpiñán (2013), O(n log n).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gen_sqrt_diff(n: int) -> np.ndarray:
    """Baseline a_i = sqrt(i+1) - sqrt(i) (sum‑normalised)."""
    a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(np.arange(0, n, dtype=np.float64))
    a /= a.sum()
    return a

def _gen_power_raw(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i + offset)^{-p} , normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p)
    a /= a.sum()
    return a

def _gen_power_diff(n: int, p: float, offset: float) -> np.ndarray:
    """a_i = (i+offset)^{-p} - (i+1+offset)^{-p} , clipped & normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset) ** (-p) - (i + 1 + offset) ** (-p)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

def _gen_beta(n: int, alpha: float, beta_param: float) -> np.ndarray:
    """Symmetric Beta‑type shape (decreasing for α<1)."""
    i = np.arange(1, n + 1, dtype=np.float64)      # 1‑based for simplicity
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta_param - 1.0))
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        return None
    a /= a.sum()
    return a

    Return a list of non‑negative heights that (hopefully) gives a low
    evaluation score.  The function respects the provided time budget.
    """

    start = time.time()
    # leave a small safety margin
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1️⃣  Determine length of the step function
    # ------------------------------------------------------------------
    try:

    for p in np.linspace(0.2, 2.0, 9):

            trials.append(('pow_raw', p, off))

    # power‑law differences
    for p in np.linspace(0.2, 2.0, 9):
        for off in (0.0, 0.5, 1.0, 2.0, 5.0):
            trials.append(('pow_diff', p, off))

    # Beta‑type shapes
    for alpha in np.linspace(0.2, 1.5, 7):

    # ------------------------------------------------------------------
    polish_deadline = time.time() + min(5.0, deadline - time.time())
    a = best_seq.copy()

        a_new = _project_onto_simplex(a_new)  # ensures non‑negativity & sum=1

    # 5️⃣  Rescale to the full [0, 1000] range (scale does not affect score)
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Returning sequence of length {len(final_seq)} with final score {best_score:.6f}")
    return final_seq.tolist()
