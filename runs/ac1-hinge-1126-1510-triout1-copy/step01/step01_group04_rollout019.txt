"""
Propose a low‑score step‑function.

The algorithm proceeds in two stages:

1. **Broad hyper‑parameter sweep** – we try many lengths, exponents and
   offsets for the classic “arcsine‑type” profile a[i] = (i+1+offset)^(-exp).
   The evaluation function is cheap for a few hundred candidates, so we
   can afford to explore lengths up to 20 000 and a small grid around the
   known optimal exponent 0.5.  The best candidate from this sweep is kept
   as a seed.

2. **Projected‑gradient refinement** – starting from the best seed we
   repeatedly move mass away from the indices that contribute to the
   currently maximal convolution entry.  The gradient of the active
   convolution entry is computed analytically, then projected onto the
   hyper‑plane Σa = constant (the ℓ₁‑norm is preserved).  A diminishing
   step size together with occasional tiny random jitter keeps the
   optimisation stable.  The loop stops when the time budget (≈ 1000 s)
   is exhausted.

All numbers are kept non‑negative and inside the range required by
``evaluate_sequence``; the implementation never writes to disk and does
not depend on any external files.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# helper functions – defined at module level (no closures)
# ----------------------------------------------------------------------


def _initial_sequence(length: int,
                      exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Return a non‑negative sequence of the given length with the shape

        a[i] = (i + 1 + offset)^{-exponent} .

    The values are clipped to [0, 1000] because ``evaluate_sequence`` does
    the same internally.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _gradient_of_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the auto‑convolution entry at index ``k`` with respect
    to the vector ``a``.
    For the entry (a*a)[k] = Σ a[i]·a[k‑i] the derivative w.r.t. a[j] is
    a[k‑j] + a[j] (the two summands coincide when 2j = k).  We return the
    result as a vector where the coefficient for ``j`` is 2·a[k‑j] whenever
    the pair (j, k‑j) lies inside the valid range.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k - j
        grad[j] = 2.0 * a[i]
    return grad


def _project_onto_sum_hyperplane(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Rescale a non‑negative vector so that its ℓ₁‑norm equals ``target_sum``.
    The operation is simply a multiplicative rescaling; negativity is
    already removed before calling this routine.
    """
    s = np.sum(x)
    if s == 0.0:
        return x
    return x * (target_sum / s)
