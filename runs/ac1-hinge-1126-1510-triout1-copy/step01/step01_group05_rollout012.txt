"""
Propose a low‑evaluation step sequence.

The optimizer works on the probability simplex (sum‑to‑one) by
representing the coefficients a_i = softmax(y_i).  A smooth
approximation of the maximum convolution entry is minimised with a
continuation schedule that gradually reduces the smoothing parameter.
Each stage uses L‑BFGS‑B.  After each L‑BFGS run a greedy mass‑transfer
hill‑climb refines the solution.  The schedule is repeated with many
random restarts and occasional perturbations of the current best.
The final result is returned as a plain list of non‑negative floats
(every entry sums to 1, so the official evaluator will return
2·n·max(a*a)).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum=1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t.
    the unconstrained vector y.
    The smoothing width is lam = max(conv) * lam_factor.
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                           # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb: repeatedly move a random fraction
    of mass from i to j (keeping the sum = 1).  Keep a move only
    if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 120,
) -> np.ndarray:
    """
    Run a sequence of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ---------- continuation L‑BFGS ----------
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=120,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ---------- aggressive local hill‑climb ----------
        # allocate a short time slice for a deeper greedy search
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # final refinement: run the continuation schedule one more time
    # ------------------------------------------------------------------
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 250, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final refine] lam={lam:.4f} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # final greedy polishing (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
