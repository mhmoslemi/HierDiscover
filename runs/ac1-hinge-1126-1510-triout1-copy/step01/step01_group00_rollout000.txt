def gradient_softmax(a, probs, L_corr):
    # a_rev = a[::-1]
    fft_probs = np.fft.fft(probs, n=L_corr)
    fft_a_rev = np.fft.fft(a[::-1], n=L_corr)
    corr_full = np.real(np.fft.ifft(fft_probs * fft_a_rev))
    grad = 2.0 * corr_full[n-1 : n-1 + n]  # size n
    return grad

"""
propose_candidate
-----------------

A fast “soft‑max + Adam” optimiser for the step‑function problem.

The evaluation function (after the usual normalisation
 Σ a_i = √(2 n)) equals the maximum entry of the linear
convolution a∗a.  Minimising the max‑convolution is a highly
non‑convex problem.  The algorithm below approximates the
non‑smooth max by a smooth log‑sum‑exp (soft‑max) with a
temperature β that is increased during optimisation (continuation
method).  For the smooth objective the gradient is an easy
correlation and can be computed with FFTs.
An Adam style update is applied and the iterate is projected
onto the probability simplex {x≥0, Σx=1}.  The final vector is
scaled to Σa = √(2 n) – this is exactly the scaling used by the
reference code, so the returned value equals the original
objective.

Key ideas compared with the previous code:
* smooth approximation of the max (β‑annealing) → richer gradient;
* Adam momentum → faster convergence and escape from shallow
  local minima;
* occasional random restarts (tiny perturbations) to explore
  additional basins;
* use of real‑FFT (`rfft`/`irfft`) for speed.

The routine respects a user supplied time budget (default 1000 s)
and prints progress information.  The best sequence found is
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: maximum entry of the linear convolution a*a (full length)
def _max_convolution(a: np.ndarray, L_conv: int) -> float:
    """Return max_{k} (a * a)[k] using FFT of length L_conv."""
    # real FFT is a bit faster for real data
    fft_a = np.fft.rfft(a, n=L_conv)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L_conv)[: 2 * a.shape[0] - 1]
    return float(conv_full.max())


# ----------------------------------------------------------------------
# Helper: gradient of the smoothed max (log‑sum‑exp) w.r.t. a
def _gradient_softmax(a: np.ndarray, probs: np.ndarray, L_corr: int) -> np.ndarray:
    """
    Gradient of (1/β)·log∑exp(β·c_k) where c = a*a.
    probs is the softmax weight vector (same length as c).
    Returns a vector of length n.
    """
    n = a.shape[0]
    # correlation of probs with a (valid part) = 2 * (B a) where B_ij = probs_{i+j}
    # Implement with FFT: corr = ifft( fft(probs) * fft(a_rev) )
    a_rev = a[::-1]
    fft_probs = np.fft.rfft(probs, n=L_corr)
    fft_a_rev = np.fft.rfft(a_rev, n=L_corr)
    corr_full = np.fft.irfft(fft_probs * fft_a_rev, n=L_corr)
    # slice the central n entries (indices n-1 … n-1+n-1)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad
