"""
propose_candidate
-----------------

Optimization of a non‑negative step‑function (vector a) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Using the observation that after fixing the normalisation
    Σ a_i = √(2n)
the objective reduces to the infinity‑norm of the self‑convolution.
The problem is convex: minimise   t   subject to   a ≥ 0,
Σ a = √(2n) and   a * a ≤ t (element‑wise).

We solve it with a smooth approximation of the max‐norm (log‑sum‑exp)
and a projected Adam optimiser on the probability simplex (Σx = 1,
x ≥ 0).  The vector x = a / √(2n) lives on the simplex; after optimisation
we simply rescale it back.

Key ideas
~~~~~~~~~
* **Smooth max** – replace max(c) by (1/α)·log∑exp(αc).  The gradient is a
  weighted correlation of the probability weights with the current vector.
* **Adam with Euclidean projection** – adaptive step sizes, followed by
  projection onto the simplex (the projection routine works in O(n log n)).
* **Multiple starts** – we start from the known construction (if present),
  from a uniform vector and from a random Dirichlet vector, each getting a
  share of the time budget.
* **Time‑aware loop** – the function stops a few seconds before the given
  budget, guaranteeing a result is returned.

The implementation uses only NumPy (no external solvers) and stays well
within the allowed resources.  The returned list contains non‑negative
floats whose sum is √(2 n), ready to be passed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]                       # sorted decreasing
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(x: np.ndarray, L: int) -> np.ndarray:
    """Linear convolution of x with itself using FFT of length L."""
    fx = np.fft.fft(x, n=L)
    conv_full = np.real(np.fft.ifft(fx * fx))[: 2 * x.size - 1]
    return conv_full


# ----------------------------------------------------------------------
def _correlation_fft(w: np.ndarray, x_rev: np.ndarray, L: int) -> np.ndarray:
    """
    Correlation of weight vector w (length 2n‑1) with x.
    Returns a length‑n vector:   (w ⋆ x)[k] = Σ_i w_{i+k} x_i .
    """
    fw = np.fft.fft(w, n=L)
    f_rev = np.fft.fft(x_rev, n=L)
    corr_full = np.real(np.fft.ifft(fw * f_rev))
    n = x_rev.size
    # slice the centre part that corresponds to indices 0…n‑1
    return corr_full[n - 1 : n - 1 + n]

    Search for a step‑function (non‑negative list) that minimises the
    evaluation function.  The algorithm runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine length n (use existing construction if available)

    target_sum = math.sqrt(2.0 * n)     # the scaling used by the reference

    # Pre‑compute FFT lengths (next powers of two for speed)
    L_conv = 1 << ((2 * n - 1).bit_length())          # >= 2n‑1
    L_corr = 1 << ((3 * n - 2).bit_length())          # >= 3n‑2

    # ------------------------------------------------------------------

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------

        # max(conv(x,x)) * target_sum**2  == evaluate_sequence(x*target_sum)
        conv = _conv_fft(x, L_conv)
        return target_sum ** 2 * conv.max()

    # ------------------------------------------------------------------
    # Optimisation routine (Adam on a smooth approximation of max)
    def _run_opt(x0: np.ndarray, time_alloc: float):
        nonlocal best_seq, best_score

        # Normalise to simplex
        x = x0.copy()
        if x.sum() == 0:

        else:

        # Adam state

        # smoothing parameter α for log‑sum‑exp (start small, increase)
        alpha = 5.0
        alpha_max = 50.0
        alpha_increase = 1.02   # multiplicative increase every 500 iters

        it = 0
        while time.time() < deadline and (time.time() - start_time) < time_alloc:
            it += 1
