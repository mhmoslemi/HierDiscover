"""
search for a low‑score non‑negative step function.

The algorithm works in two phases.
1.  A coarse sweep over a few lengths and power‑law seeds
    a[i] ≈ (i+1+δ)^{-p} (p≈0.5) is performed; the best few seeds are kept.
2.  Each retained seed is refined by projected gradient descent on a
    *smooth* approximation of the max‑convolution term.
    The smooth max is implemented with a soft‑max (temperature λ) which is
    slowly increased, turning the objective from a smooth average into the
    true supremum.  After each gradient step the vector is projected onto the
    feasible set (non‑negative entries with a fixed ℓ₁‑norm).  Random jitter
    is added occasionally to escape plateaus.

All operations use FFT‑based convolution for speed, allowing sequences of
tens of thousands of entries to be optimized well within the 1000 s budget.
The best sequence found (as a plain list of floats) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level as required
# ----------------------------------------------------------------------


def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Build a power‑law profile a[i] = (i + 1 + offset)^(-exponent).
    Values are clipped to the [0,1000] range required by evaluate_sequence.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    return np.clip(seq, 0.0, 1000.0)


def _target_sum(length: int) -> float:
    """Target ℓ₁‑norm (the scaling that makes the objective equal max‑conv)."""
    return np.sqrt(2.0 * length)


def _normalize_sequence(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Clip to non‑negative and rescale so that the sum equals target_sum.
    """
    x = np.maximum(x, 0.0)
    cur = x.sum()
    if cur == 0.0:
        # should never happen, but keep the vector unchanged
        return x
    return x * (target_sum / cur)


def _autocorr_fft(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using a real FFT.
    Returns a vector of length 2*len(a)-1.
    """
    n = a.shape[0]
    m = 1 << ((2 * n - 1).bit_length())      # next power of two
    A = np.fft.rfft(a, n=m)
    C = np.fft.irfft(A * A, n=m)
    return C[: 2 * n - 1]


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Soft‑max weights w_k ∝ exp(lam * conv_k) that approximate the max.
    The subtraction of max(conv) guarantees numerical stability.
    """
    shifted = conv - conv.max()               # stabilise the exponentials
    exp_vals = np.exp(lam * shifted)
    return exp_vals / exp_vals.sum()


def _gradient_softmax(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max objective Σ_k w_k (a*a)_k w.r.t. a.
    The formula reduces to
        g_i = 2 * Σ_k w_k a_{k-i},
    which is a correlation of w with the reversed a.
    """
    n = a.shape[0]
    rev_a = a[::-1]                           # reversed a
    # length of the linear convolution w * rev_a is len(w)+n-1 = 3n-2
    m = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, n=m)
    A_rev = np.fft.rfft(rev_a, n=m)
    c = np.fft.irfft(W * A_rev, n=m)          # linear convolution
    c = c[: len(w) + n - 1]                  # keep the needed part
    start = n - 1
    g = 2.0 * c[start : start + n]            # slice of length n
    return g


def _score_from_seq(a: np.ndarray) -> float:
    """
    Direct computation of the evaluation function (avoids the slower
    evaluate_sequence call).  For a feasible vector this equals
        2 * n * max(a*a) / (sum(a)**2)
    """
    n = a.shape[0]
    conv = _autocorr_fft(a)
    maxb = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * maxb / (sum_a * sum_a)

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises
        2 * n * max(a*a) / (sum(a)**2)

    The search combines a quick grid over power‑law seeds and a
    smooth‑max projected gradient descent.  It respects the given time
    budget and prints intermediate progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # tiny safety margin

    # --------------------------------------------------------------
    # 1) Coarse grid of seeds (power‑law shapes)
    # --------------------------------------------------------------
    lengths = [1500, 3000, 5000, 8000, 12000, 20000]
    exponents = [0.48, 0.50, 0.52]
    offsets = [0.0, 0.2, 0.5, 1.0]

    candidate_seqs = []      # list of (score, array)
    best_score = float("inf")
    best_seq = None

    print("=== coarse grid search over power‑law seeds ===")
    for n in lengths:
        target = _target_sum(n)
        for p in exponents:
            for off in offsets:
                a = _initial_sequence(n, exponent=p, offset=off)
                a = _normalize_sequence(a, target)
                scr = _score_from_seq(a)

                # keep up to eight promising seeds
                if len(candidate_seqs) < 8:
                    candidate_seqs.append((scr, a.copy()))
                    candidate_seqs.sort(key=lambda x: x[0])
                else:
                    if scr < candidate_seqs[-1][0]:
                        candidate_seqs[-1] = (scr, a.copy())
                        candidate_seqs.sort(key=lambda x: x[0])

                if scr < best_score:
                    best_score = scr
                    best_seq = a.copy()
                    print(f"new coarse best: len={n:5d}, p={p:.2f}, off={off:.1f} → {scr:.6f}")

                if time.time() > deadline:
                    break
            if time.time() > deadline:
                break
        if time.time() > deadline:
            break

    if best_seq is None:
        # fallback: uniform vector of length 1000 (guaranteed feasible)
        best_seq = np.ones(1000, dtype=np.float64)
        best_seq = _normalize_sequence(best_seq, _target_sum(1000))
        best_score = _score_from_seq(best_seq)

    print(f"Coarse phase best score: {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Refined optimisation with smooth‑max gradient descent
    # --------------------------------------------------------------
    print("=== refined optimisation (smooth‑max gradient descent) ===")
    for idx, (seed_score, a0) in enumerate(candidate_seqs):
        n = a0.shape[0]
        target = _target_sum(n)
        a = a0.copy()

        # hyper‑parameters (they can be tweaked per candidate if desired)
        step0 = 0.20                   # initial step size
        lam = 1.0                      # start temperature for soft‑max
        lam_max = 15.0
        lam_factor = 1.001             # multiplicative increase per iteration
        max_iters = 60000              # generous upper bound
        iteration = 0

        while time.time() < deadline and iteration < max_iters:
            iteration += 1

            # current convolution & objective value
            conv = _autocorr_fft(a)
            maxb = conv.max()
            cur_sum = a.sum()
            cur_score = 2.0 * n * maxb / (cur_sum * cur_sum)

            # global best update
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"candidate {idx} iter {iteration:5d}: new best {best_score:.6f} (len={n})")
                if best_score < 1.503:
                    # reached the target – stop early
                    break

            # smooth‑max weights and gradient
            w = _softmax_weights(conv, lam)
            grad = _gradient_softmax(a, w)

            # project gradient onto the hyperplane Σ a = constant
            grad -= np.mean(grad)

            # diminishing step size
            step = step0 / (1.0 + iteration * 0.001)
            a_new = a - step * grad

            # enforce feasibility
            a_new = _normalize_sequence(a_new, target)

            # occasional jitter to help escape shallow plateaus
            if iteration % 500 == 0:
                jitter = np.random.randn(n) * 1e-4
                a_new = _normalize_sequence(a_new + jitter, target)

            a = a_new

            # tighten soft‑max (approach true max)
            lam = min(lam * lam_factor, lam_max)

        if best_score < 1.503:
            break   # goal achieved – no need to process remaining seeds

    print(f"=== finished === final best score: {best_score:.6f} (length {len(best_seq)})")
    # Return a plain Python list, as required by the surrounding harness
    return best_seq.tolist()
