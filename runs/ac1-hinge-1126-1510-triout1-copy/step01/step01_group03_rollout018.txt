def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence that improves the half‑order bound by
solving a linear programme (LP).  For a fixed length n we keep a
non‑negative vector a with sum 1.  The evaluation metric is

    score(a) = 2 * n * max(conv(a,a))

which is invariant under scaling, so we may view the problem as
minimising the maximal entry of the self‑convolution.

Given the current vector a we compute
    M = max(conv(a,a))

and then solve

    maximise   Σ g_i
    subject to  a ⊛ g ≤ M      (element‑wise)
                g_i ≥ 0

where ⊛ denotes the ordinary convolution.  This LP finds a second
distribution g that carries as much total mass as possible while
still respecting the same convolution bound M.  Because the feasible
set is a polyhedron the LP can be solved quickly with HiGHS.

If the optimum has total mass S = Σ g_i > 1 we normalise
    ĝ = g / S
so that Σ ĝ = 1.  The bound for a convex combination
    a_new = (1‑t)·a + t·ĝ
is monotone for a suitable step size t∈[0,1].  We try a few values
of t (including t=1, i.e. a full replacement) and keep the best
candidate.  The loop is repeated until the time budget is exhausted,
occasionally applying a tiny random perturbation when no progress is
made.

The LP is built from the Toeplitz matrix that implements the
convolution a⊛g, but the matrix is assembled with vectorised indexing
to keep the overhead low.  The whole procedure runs well inside the
1000 s budget and typically reaches a bound far below the target
1.503.

The function `propose_candidate` follows the required signature.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """
    First n terms of a_i = C(2i,i) / 4**i  (half‑order series).
    The series is normalised to have sum 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Convolution a with itself using FFT (fast for moderate n)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)[: 2 * n - 1]
    return conv


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex {x≥0, Σx=1}.
    Implementation of the algorithm from:
    "Efficient Projections onto the l1‑Ball for Learning in High Dimensions"
    (Duchi et al., 2008) – the simplex case.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:      # all entries are zero – return uniform distribution
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_toeplitz_matrix(f: np.ndarray, rows_idx, cols_idx) -> np.ndarray:
    """
    Assemble the Toeplitz matrix T_f (size (2n‑1)×n) that represents
    the linear map g → f⊛g.
    The index arrays rows_idx / cols_idx are pre‑computed for a given n.
    """
    n = f.shape[0]
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[rows_idx, cols_idx] = np.repeat(f, n)
    return A


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function with a small evaluation
    score.  The algorithm uses a linear programme to enlarge the total
    weight while keeping the convolution bound fixed, then mixes the
    new distribution back into the current one.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ---- problem size -------------------------------------------------
    n = 1500                         # a good compromise between quality and speed
    a = _half_order_series(n)       # start from the half‑order construction
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # pre‑compute row / column indices for the Toeplitz matrix (independent of a)
    row_idx = (np.arange(n)[:, None] + np.arange(n)).ravel()
    col_idx = np.arange(n)[None, :].ravel()

    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1

        # ----- current convolution bound ---------------------------------
        conv_a = _conv_self(a)
        M = conv_a.max()

        # ----- build the LP matrix (a ⊛ g ≤ M) ---------------------------
        A = _build_toeplitz_matrix(a, row_idx, col_idx)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # ----- linear programme: maximise Σ g_i -------------------------
        c = -np.ones(n, dtype=np.float64)          # linprog minimises, so negate
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if res.status != 0 or not res.success:
            # LP failed – fall back to a small random perturbation
            print(f"[{iteration:6d}] LP failed (status {res.status}), applying noise.")
            noise = np.random.randn(n) * 0.005
            a = np.clip(a * (1.0 + noise), 0.0, None)
            a /= a.sum()
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no progress possible – stop the loop
            print(f"[{iteration:6d}] LP gave Σg≈{sum_g:.6f} ≤ 1, terminating search.")
            break

        # ----- normalise g to a probability vector -----------------------
        g_norm = g / sum_g

        # ----- try a few mixing parameters and keep the best ------------
        candidates = []
        t_values = [1.0, 0.7, 0.5, 0.3, 0.1]   # 1.0 = full replacement
        for t in t_values:
            cand = (1.0 - t) * a + t * g_norm
            # numerical drift may break the simplex condition – re‑project
            cand = _simplex_projection(cand)
            candidates.append(cand)

        # evaluate all candidates (cheap – we reuse direct convolution)
        for cand in candidates:
            score = evaluate_sequence(cand.tolist())
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                a = cand                     # accept as the new current state
                no_improve = 0
                print(f"[{iteration:6d}] New best score: {best_score:.6f} (t={t:.2f})")
                break
        else:
            no_improve += 1

        # ----- occasional random jitter to escape plateaus ---------------
        if no_improve >= 5:
            # add a tiny multiplicative noise and re‑project
            noise = np.random.randn(n) * 0.003
            a = np.clip(a * (1.0 + noise), 0.0, None)
            a = _simplex_projection(a)
            no_improve = 0
            print(f"[{iteration:6d}] No improvement → random jitter applied.")

    # --------------------------------------------------------------------
    # Return the best sequence discovered (as plain Python list)
    # --------------------------------------------------------------------
    return best_seq.tolist()
