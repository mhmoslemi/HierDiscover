import numpy as np
import time
import random

def generate_power_law(n: int, p: float) -> np.ndarray:
    """Return a normalized power‑law sequence a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq_sum = seq.sum()
    if seq_sum == 0:
        return np.full(n, 0.0, dtype=np.float64)
    seq /= seq_sum
    return seq

def fast_score(seq: np.ndarray) -> float:
    """Fast approximation of evaluate_sequence (no clipping)."""
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()  # next power of two
    fft_a = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:m]
    maxb = conv.max()
    return 2.0 * n * maxb / (s * s)

"""Multi‑stage stochastic search for a low‑score non‑negative step function.

Stage 1 – a cheap grid search over sequence length and a power‑law exponent.
Stage 2 – a simple simulated‑annealing style local search that repeatedly moves a
small amount of mass from one index to another while keeping the total sum
constant.  All work respects the supplied time budget (default 1000 s).

The returned sequence is a plain list of Python floats, ready for
``evaluate_sequence`` to be called by the harness.
"""

import numpy as np
import time
import random

def generate_power_law(n: int, p: float) -> np.ndarray:
    """Return a normalized power‑law sequence a_i = (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    total = seq.sum()
    if total == 0.0:
        return np.full(n, 0.0, dtype=np.float64)
    seq /= total
    return seq


def fast_score(seq: np.ndarray) -> float:
    """
    Fast analogue of ``evaluate_sequence`` (without clipping checks).

    The formula is 2 * n * max(conv(seq, seq)) / (sum(seq) ** 2)
    where the convolution is the ordinary linear convolution.
    """
    seq = np.asarray(seq, dtype=np.float64)
    n = seq.shape[0]
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    m = 2 * n - 1
    # zero‑pad to the next power of two for an efficient FFT
    size = 1 << (m - 1).bit_length()
    fft_a = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:m]
    maxb = conv.max()
    return 2.0 * n * maxb / (s * s)
