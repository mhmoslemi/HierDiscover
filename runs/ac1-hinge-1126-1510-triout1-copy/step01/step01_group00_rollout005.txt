            beta = beta_start + (beta_end - beta_start) * it / max_iters_per_restart
            # stability: shift by the max before exponentiation
            w_exp = np.exp(beta * (conv - conv_max))
            w = w_exp / w_exp.sum()             # soft‑max over convolution entries

            # ------- gradient w.r.t. a (2·corr(w,a)) -------
            corr_full = _correlation_full(w, a[::-1], L_corr)
            # valid correlation segment of length n
            grad_a = 2.0 * corr_full[n - 1 : n - 1 + n]

            # ------- gradient w.r.t. soft‑max parameters v -------
            # ∂obj/∂v = a ◦ (grad_a – ⟨grad_a , a⟩)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # ------- Adam update -------
            t = it + 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

            m_hat = m / (1.0 - beta1 ** t)
            v2_hat = v2 / (1.0 - beta2 ** t)

            v -= lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    # -------------------------------------------------------------
    # Fallback in the unlikely case nothing was found
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)

    # Clip to the range enforced by evaluate_sequence (just to be safe)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"--- Finished: best score = {best_score:.6f} (n={n}) ---")
    return best_seq.tolist()
