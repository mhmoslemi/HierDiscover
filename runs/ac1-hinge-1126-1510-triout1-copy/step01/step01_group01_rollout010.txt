def simplex_projection(v):
    """Project vector v onto the probability simplex (nonnegative and sum=1)."""
    # Implementation: sort v descending, compute cumulative sum, find rho ...
    # Standard algorithm from Duchi et al. 2008.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def _catalan_sequence(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k-1] * (2.0*k)*(2.0*k-1.0) / (4.0*k*k)
    return a

def _project_simplex(v):
    # Project v onto the probability simplex.
    # Implementation per Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full(n, 1.0/n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Gradient‑descent / exponentiated‑gradient optimizer for the
    self‑convolution bound.

    The method starts from the Catalan (central‑binomial) coefficients,
    then repeatedly applies an exponentiated‑gradient step that reduces
    the maximum entry of the linear self‑convolution while keeping the
    vector on the probability simplex.  A temperature‑like parameter
    beta gradually sharpens the focus on the current peak.  The loop
    stops when the time budget expires and the best sequence seen is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # Length of the candidate – use the same length that gave the
    # previous record, unless a global sequence is already provided.
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        n = a.shape[0]
        a = np.maximum(a, 0.0)          # ensure non‑negative
        if a.sum() == 0.0:
            a = np.ones(n, dtype=np.float64) / n
        else:
            a = a / a.sum()
    else:
        n = 6402
        a_raw = _catalan_sequence(n)
        a = a_raw / a_raw.sum()   # ℓ1‑normalised to 1

    best_seq = a.copy()
    # initial evaluation (exact)
    c_initial = np.convolve(a, a)
    best_score = float(2.0 * n * c_initial.max())
    print(f"[init] length={n}, score={best_score:.6f}")

    # FFT configuration – the convolution a_rev * w needs length 3n‑2.
    Nfft = 1 << ((3 * n - 2 - 1).bit_length())   # next power of two

    # Hyper‑parameters
    beta_start = 1.0               # exponent on the convolution values
    beta_end   = 30.0              # final exponent (more peaked)
    lr_start   = 0.3               # initial multiplicative‑step size
    min_lr     = 1e-4              # lower bound for safety
    no_improve = 0
    IMPROVE_PATIENCE = 5000        # when to inject jitter

    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # Fraction of the total elapsed time (0..1)
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)

        # Gradually increase beta and decay the step size.
        beta = beta_start + (beta_end - beta_start) * frac
        lr = max(lr_start * (0.5 ** frac), min_lr)

        # ---- 1. self‑convolution (linear) ----
        a_f = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(a_f * a_f, n=Nfft)
        conv = conv_full[: 2 * n - 1]               # length 2n‑1
        max_c = conv.max()
        # ---- 2. build the weighting vector w ----
        # Use a powered‑softmax w_i ∝ (conv_i)^beta  (beta ≥ 1)
        # Adding a tiny epsilon avoids division by zero.
        eps = 1e-30
        w_raw = (conv + eps) ** beta
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # Convolution of reversed a with w:
        a_rev_f = np.fft.rfft(a[::-1], n=Nfft)
        w_f = np.fft.rfft(w, n=Nfft)

        conv_rev = conv_rev_full[: 3 * n - 2]       # length 3n‑2
        grad = a + conv_rev[n - 1 : 2 * n - 1]      # shape (n,)

        # ---- 4. exponentiated‑gradient update ----
        # a_i ← a_i * exp(‑η * grad_i)  then renormalise.
        a_new = a * np.exp(-lr * grad)

        a = a_new / a_new.sum()

        if score < best_score - 1e-12:

"""
Gradient‑exponentiated‑gradient optimizer for the self‑convolution bound.

The algorithm starts from the Catalan (central‑binomial) coefficients,
normalised to a probability vector (ℓ¹‑norm = 1).  At each iteration it
computes the linear self‑convolution a*a, builds a weighting vector
w_i ∝ (a*a)_i^β (β grows from 1 to ≈30) and forms the gradient of the
softened max‑convolution:
    g = a + (a_rev * w)_{slice}
where a_rev is a reversed copy of a and the slice extracts the n entries
needed for the gradient.  An exponentiated‑gradient step
    a ← a·exp(‑η·g)
followed by ℓ¹‑renormalisation keeps a on the probability simplex.
The learning rate η decays while β increases, first smoothing the
objective then focusing on the current peak.  If no improvement is seen
for many iterations a tiny random jitter is added (projected back onto
the simplex) to escape plateaus.  The loop runs until the time budget is
exhausted and the best sequence found is returned.
"""

import numpy as np
import time


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients a_k = binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all entries are non‑positive
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Returns the best list of heights discovered
    before the time limit expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety buffer so we always exit cleanly
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Initialise the sequence.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.ones_like(a) / a.size
        else:
            a = a / a.sum()
        n = a.size
    else:
        n = 6402
        a_raw = _catalan_sequence(n)
        a = a_raw / a_raw.sum()               # ℓ¹‑normalised to 1

    best_seq = a.copy()
    # Exact initial score (avoids the overhead of evaluate_sequence)
    conv0 = np.convolve(a, a)
    best_score = float(2.0 * n * conv0.max())
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT length (must accommodate convolutions of lengths
    #    2n‑1 and 3n‑2).
    # ------------------------------------------------------------------
    Nfft = 1 << ((3 * n - 2 - 1).bit_length())   # next power of two ≥ 3n‑2

    # ------------------------------------------------------------------
    # 3. Hyper‑parameters
    # ------------------------------------------------------------------
    beta_start = 1.0          # exponent on convolution values (warm‑up)
    beta_end   = 30.0         # final (sharp) exponent
    lr_start   = 0.30         # initial multiplicative step size
    min_lr     = 1e-4         # lower bound to keep progress possible
    no_improve = 0
    IMPROVE_PATIENCE = 5000   # inject jitter after this many non‑improving steps
    iteration = 0

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)   # 0 → 1

        # annealing of β and learning‑rate η
        beta = beta_start + (beta_end - beta_start) * frac
        lr = max(lr_start * (0.5 ** frac), min_lr)

        # ---- a) self‑convolution a*a (linear, zero‑padded) ----
        a_f = np.fft.rfft(a, n=Nfft)
        conv_full = np.fft.irfft(a_f * a_f, n=Nfft)
        conv = conv_full[: 2 * n - 1]               # length 2n‑1
        max_c = conv.max()

        # ---- b) weighting vector w ∝ (conv)^β  (softened max) ----
        eps = 1e-30
        w_raw = (conv + eps) ** beta
        w_sum = w_raw.sum()

            w = w_raw / w_sum

        grad = a + conv_rev[n - 1 : 2 * n - 1]

        a_new = a * np.exp(-lr * grad)
        if a_new.sum() == 0.0:                      # numerical safeguard

        a = a_new / a_new.sum()                     # stays on the simplex

        # ---- e) bookkeeping ----
        score = 2.0 * n * max_c                      # exact objective (∑a = 1)
        if score < best_score - 1e-12:

            best_seq = a.copy()
            no_improve = 0

            no_improve += 1

        # ---- f) occasional jitter to escape plateaus ----
        if no_improve >= IMPROVE_PATIENCE:
            jitter = 0.001 * np.random.rand(n)
            a = _project_simplex(a + jitter)
            no_improve = 0
            print(f"[jitter] it={iteration} t={int(elapsed)}s")

    # ------------------------------------------------------------------
    # 5. Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"[final] length={n}, best_score={best_score:.6f}")
    return best_seq.tolist()
