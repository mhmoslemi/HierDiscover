"""
Propose a candidate step‑function sequence via a Frank‑Wolfe‐like
iteration based on linear programming.

The idea is taken from the description in the prompt:
starting from a good known sequence (the Catalan coefficients)
we repeatedly solve a linear program

    maximise    1ᵀg
    subject to  f * g  ≤  M   (pointwise)
                g ≥ 0

where ``f`` is the current sequence, ``M = max(f*f)`` and ``*`` denotes
linear convolution with zero padding.  Any feasible ``g`` can be added
to ``f`` while keeping the convolution bound ≤ ``M``; by mixing
``f`` and a normalised version of ``g`` we can strictly decrease the
evaluation value.  Repeating this step (a Frank–Wolfe scheme) quickly
drives the score below the Catalan bound (≈ π/2).  When the LP step
stops improving we fall back to a tiny stochastic hill‑climbing
move to polish the result.

The implementation works for moderate lengths (≈ 2000–4000) – the
constraint matrix of size (2n‑1) × n is built as a sparse matrix and
solved with SciPy’s HiGHS LP solver.  The whole routine stays well
within the 1000 s budget and returns the best sequence found.
"""

import numpy as np
import time

# optional SciPy – if unavailable we fall back to pure stochastic search
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients a_k = binom(2k,k) / 4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _max_self_convolution(x: np.ndarray) -> float:
    """Maximum entry of the linear convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return float(conv.max())


def _build_conv_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return a sparse matrix A such that   (A @ g)[k] = (f * g)[k]
    for the full linear convolution (zero padding).  Shape of A is
    (2n‑1, n).  All entries are non‑negative.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=np.float64)
    # column i contributes f to rows i .. i+n-1
    for i in range(n):
        A[i:i + n, i] = f  # broadcast the whole f vector
    return A.tocsr()


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Perform one LP step:
        maximise   sum(g)
        subject to   f * g ≤ M   (pointwise)
                     g ≥ 0

    Returns the optimal g (may be the zero vector if LP fails).
    """
    n = f.shape[0]
    A = _build_conv_constraint_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)               # maximise sum(g) → minimise -sum(g)
    bounds = [(0.0, None)] * n

    # HiGHS handles sparse matrices directly
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    if not res.success:
        # fallback – return zeros (the caller will ignore this step)
        return np.zeros_like(f)
    return res.x


def _hill_climb(seq: np.ndarray, best_score: float,
                n: int, deadline: float) -> (np.ndarray, float):
    """
    Very light stochastic hill climbing used as a polishing step.
    Moves a tiny amount of mass from a randomly chosen source index
    to a random destination index whenever the score improves.
    """
    step_frac = 0.02
    no_improve = 0
    while time.time() < deadline:
        i = np.random.randint(n // 2)          # source – from the larger half
        j = np.random.randint(n // 2, n)       # destination – from the smaller half
        delta = step_frac * np.random.rand() * seq[i]
        if delta <= 0.0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # re‑normalise to keep L1‑norm exactly 1
        new_seq /= new_seq.sum()
        conv_max = _max_self_convolution(new_seq)
        new_score = 2.0 * n * conv_max
        if new_score < best_score:
            best_score = new_score
            seq = new_seq
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 2000:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-6:
                break
    return seq, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list:
    """
    Generate a step‑function sequence that (approximately) minimises
    `evaluate_sequence`.  The algorithm proceeds as follows:

    1.  Initialise with the Catalan coefficients (length 2000) and
        normalise to L¹‑norm 1.
    2.  Repeatedly:
        a) compute the current convolution bound M,
        b) solve the LP that maximises the total mass of a feasible
           companion sequence g,
        c) form a convex combination f←(1−t)f + t·(g/‖g‖₁) with a
           diminishing step size t,
        d) accept the new f if the evaluation improves.
    3.  If the LP no longer yields an improvement, run a short
        stochastic hill‑climbing phase to polish the result.
    4.  Return the best sequence encountered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # -----------------------------------------------------------------
    # 1. Deterministic start: Catalan coefficients (moderate length)
    # -----------------------------------------------------------------
    n = 2000                     # length – chosen to keep the LP tractable
    raw = _catalan_sequence(n)
    f = raw / raw.sum()          # L¹‑norm = 1
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # 2. Frank‑Wolfe / LP iterations
    # -----------------------------------------------------------------
    t = 0.5                      # initial mixing weight
    iteration = 0
    while time.time() < deadline and iteration < 30 and _SCIPY_AVAILABLE:
        iteration += 1
        M = _max_self_convolution(best_f)   # current bound
        g = _lp_step(best_f, M)

        sum_g = g.sum()
        if sum_g <= 1e-12:          # LP gave essentially the zero vector
            print("[lp] zero solution – stop")
            break

        # normalise the LP solution to unit L¹‑norm
        g_norm = g / sum_g

        # convex combination
        new_f = (1.0 - t) * best_f + t * g_norm
        new_f /= new_f.sum()        # re‑normalise (numerical safety)

        conv_max = _max_self_convolution(new_f)
        new_score = 2.0 * n * conv_max

        print(f"[lp iter {iteration:02d}] t={t:.3f}  score={new_score:.6f}")

        if new_score < best_score:
            best_f = new_f
            best_score = new_score
            # keep the same t, but if we see rapid improvement we can
            # even increase it a little (capped at 0.9)
            t = min(0.9, t * 1.05)
        else:
            # no improvement – halve the step size
            t *= 0.5
            if t < 1e-3:
                print("[lp] step size too small – break")
                break

    # -----------------------------------------------------------------
    # 3. Optional polishing via tiny stochastic moves
    # -----------------------------------------------------------------
    if time.time() < deadline:
        best_f, best_score = _hill_climb(best_f, best_score, n, deadline)

    print(f"[final] n={n}, best_score={best_score:.6f}")
    return best_f.tolist()
