    """
    Gradient-descent on a smooth approximation of the max self‑convolution.
    Starts from the previous best sequence (if present) or the sqrt‑difference
    profile, keeps the vector on the probability simplex and iteratively
    reduces the peak of a*a.  A simple step‑size schedule, increasing λ
    (log‑sum‑exp temperature) and occasional random perturbations are used.
    The best sequence found is scaled to the [0,1000] range before returning.
    """

    # Setup RNG
    np.random.seed(seed)

    # Determine length and initialise
    try:
        init = np.array(height_sequence_1, dtype=np.float64)
        n = len(init)
        a = init.copy()
        # protect against cases where sum is zero (should not happen)

    a = a / a.sum()

    # Helper for simplex projection

        # Duchi et al. 2008
        v = np.asarray(v, dtype=np.float64)

        u = np.sort(v)[::-1]

            return np.full_like(v, 1.0 / n_)
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
        w = np.maximum(v - theta, 0.0)
        return w

    # Evaluation routine (max conv)
    def max_convolution(x):
        c = np.convolve(x, x)
        return c.max()

    # Start with best known
    best_a = a.copy()
    best_max = max_convolution(best_a)
    best_score = evaluate_sequence((best_a * 1000.0).tolist())
    print(f'Initial max {best_max:.8f}, score {best_score:.6f}, n={n}')

    start = time.time()
    deadline = start + budget_s - 5.0   # leave a few seconds margin

    # Parameters
    lam = 1.0            # smoothness (log‑sum‑exp temperature)
    base_alpha = 0.8    # initial step size (in ℓ2 sense)
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # current convolution and its max
        conv = np.convolve(a, a)
        max_conv = conv.max()

        start_idx = len(w) - 1

    """Project vector v onto the probability simplex {x>=0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v

    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)

def smooth_gradient(a, lam):
    conv = np.convolve(a, a)
    maxc = conv.max()
    if maxc == 0:
        return np.zeros_like(a)
    rel = conv / maxc
    w = np.exp(lam * (rel - 1.0))
    # correlation
    corr = np.correlate(a, w, mode='full')
    return 2.0 * corr[len(w)-1 : len(w)-1 + len(a)]
```

The algorithm works in three stages:

1.  A smoothed gradient descent on a log‑sum‑exp approximation of the
    peak of the convolution a*a.  The gradient is obtained analytically

3.  A short random‑move polishing phase around the best sequence found.

The returned sequence is scaled to the full [0, 1000] range (the metric
is scale‑invariant).  The whole search stops before the supplied time
budget (≈ 1000 s) and the best sequence discovered is returned.

import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def max_convolution(x: np.ndarray) -> float:
    """Maximum value of the linear convolution x*x."""
    return np.convolve(x, x).max()


def smooth_gradient(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp smoothed peak of a*a.
    lam – temperature (larger => sharper approximation).
    Returns a vector of the same length as a.
    """
    conv = np.convolve(a, a)                # length 2n‑1
    max_c = conv.max()
    if max_c == 0.0:
        return np.zeros_like(a)

    # relative values (in [0,1]), then soft‑max weights
    rel = conv / max_c
    w = np.exp(lam * (rel - 1.0))           # w_max = 1, others ≤ 1

    # gradient = 2 * (a ⋆ w)  (cross‑correlation)
    corr = np.correlate(a, w, mode='full')
    start = len(w) - 1                      # lag 0 corresponds to this index
    grad = 2.0 * corr[start:start + a.shape[0]]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative coefficient sequence that minimizes
    `evaluate_sequence`.  The routine respects the time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin – keep a few seconds for returning the result
    deadline = start_time + max(budget_s - 5.0, 1.0)

    # Normalise to the probability simplex (scale‑invariant objective)
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # --------------------------------------------------------------
    # 2. Track the best solution seen so far
    # --------------------------------------------------------------
    best_a = a.copy()
    best_max = max_convolution(best_a)
    best_score = evaluate_sequence((best_a * 1000.0).tolist())
    print(f"[init] score={best_score:.6f}, max_conv={best_max:.8f}, n={n}")

    # --------------------------------------------------------------
    # 3. Phase A – smoothed gradient descent
    # --------------------------------------------------------------
    lam = 1.0                # start temperature
    base_alpha = 0.5         # initial step size (ℓ₂‑scaled)
    it = 0
    phase_a_deadline = start_time + (deadline - start_time) * 0.60
    while time.time() < phase_a_deadline:
        # gradient of the smoothed objective
        grad = smooth_gradient(a, lam)

        # step size slowly decays with the iteration counter
        alpha = base_alpha / (1.0 + it * 1e-3)
        a_candidate = a - alpha * grad
        a_candidate = project_onto_simplex(a_candidate)

        new_max = max_convolution(a_candidate)
        if new_max < best_max:
            best_max = new_max
            best_a = a_candidate.copy()
            best_score = evaluate_sequence((best_a * 1000.0).tolist())
            print(f"[grad] it={it:05d}  max={best_max:.8f}  score={best_score:.6f}")

        # accept if the true max decreased, otherwise keep current a
        if new_max <= max_convolution(a):
            a = a_candidate
        else:
            # occasional uphill move to escape shallow plateaus
            if np.random.rand() < 0.001:
                a = a_candidate

        # heat schedule – make the approximation sharper over time
        if it % 200 == 0 and lam < 30.0:
            lam = min(lam * 1.4, 30.0)

        # tiny random jitter every few hundred steps to avoid deadlocks
        if it % 500 == 0:
            idx = np.random.randint(n)
            a[idx] = max(0.0, a[idx] + np.random.randn() * 1e-4)
            a = project_onto_simplex(a)

        it += 1

    # --------------------------------------------------------------
    # 4. Phase B – LP refinement (if SciPy is available)
    # --------------------------------------------------------------
    try:
        import scipy.sparse as sp
        from scipy.optimize import linprog
        scipy_ok = True
    except Exception:
        scipy_ok = False

    if scipy_ok:
        # Pre‑compute the row/column pattern of the Toeplitz constraint matrix
        i_idx = np.arange(n)[:, None]          # shape (n,1)
        j_idx = np.arange(n)[None, :]          # shape (1,n)
        rows = (i_idx + j_idx).ravel()         # k = i + j, 0 … 2n‑2
        cols = np.repeat(np.arange(n), n)      # column index j repeats for each i
        # The data entries are the current f_i; they will be rebuilt each LP call.

        t = 0.12            # convex‑combination weight
        lp_iter = 0
        phase_b_deadline = deadline - 5.0  # leave a few seconds for polishing
        while time.time() < phase_b_deadline and lp_iter < 30:
            M = max_convolution(a)            # current convolution peak
            # Build the constraint matrix with the *current* f entries
            data = np.repeat(a, n)            # f_i repeated n times (row‑major order)
            A_ub = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

            # Linear program: maximise Σ g_j  ⇔  minimise −Σ g_j
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            bounds = [(0, None)] * n

            res = linprog(
                c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                method="highs",
                options={"presolve": True, "time_limit": max(phase_b_deadline - time.time(), 0.0)},
            )
            if not res.success:
                print(f"[LP] iteration {lp_iter}: solve failed ({res.message})")
                break

            g = res.x
            if g.sum() == 0.0:
                break
            g = g / g.sum()                     # re‑normalise to sum = 1

            # convex update
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = project_onto_simplex(a_candidate)

            new_max = max_convolution(a_candidate)
            if new_max < best_max:
                best_max = new_max
                best_a = a_candidate.copy()
                best_score = evaluate_sequence((best_a * 1000.0).tolist())
                print(f"[LP] it={lp_iter:02d}  max={best_max:.8f}  score={best_score:.6f}")

            # Accept the update if it reduced the true peak
            if new_max < M:
                a = a_candidate
            else:
                # shrink the mixing factor when the move is not beneficial
                t = max(t * 0.9, 0.02)

            lp_iter += 1
            # slowly tighten the mixing weight
            if lp_iter % 5 == 0:
                t = max(t * 0.93, 0.02)

    # --------------------------------------------------------------
    # 5. Final polishing – tiny random mass transfers
    # --------------------------------------------------------------
    polish_deadline = time.time() + min(5.0, deadline - time.time())
    a = best_a.copy()
    while time.time() < polish_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps = a[i] * 0.02
        if eps <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new = project_onto_simplex(a_new)

        new_max = max_convolution(a_new)
        if new_max < best_max:
            best_max = new_max
            best_a = a_new.copy()
            best_score = evaluate_sequence((best_a * 1000.0).tolist())
            print(f"[polish] improved to score={best_score:.6f}")

        a = a_new

    # --------------------------------------------------------------
    # 6. Return the best sequence (scaled to the 0‑1000 range)
    # --------------------------------------------------------------
    final_seq = best_a * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Returning sequence length={len(final_seq)} with final score={best_score:.6f}")
    return final_seq.tolist()
