"""
Propose a candidate sequence for the convolution‑maximisation problem.

The algorithm works in two phases:

1. **Coarse global search** –  we generate families of simple analytic
   sequences
        a_i ∝ (i+1+δ)^(-α)      (shifted power‑law, α≈0.5)
        a_i ∝ binom(2i,i)/4^i   (Catalan, the known asymptotic optimum)
        a_i ∝ 1/√(x_i(1‑x_i))   (discrete arcsine shape)
   for a few different lengths (≈2 k … 12 k).  Every candidate is
   normalised to ℓ¹‑norm = 1 and its score `2·n·max(conv(a,a))` is
   evaluated with a fast real‑FFT.  The best sequence from this stage
   becomes the seed for the second phase.

2. **Refinement by simulated‑annealing hill‑climbing** –  starting from the
   best seed we repeatedly transfer a small amount of mass from a
   randomly chosen source index to a random destination index, re‑normalise
   the vector and accept the move if it improves the score or with a
   temperature‑dependent probability (standard annealing).  The step size
   shrinks together with the temperature, allowing fine‑grained
   optimisation when the search gets close to a local minimum.

Both phases respect the 1000 s time budget; the routine prints progress
and finally returns the best sequence found as a plain list of floats.
"""

import math
import time

import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _max_self_convolution(x: np.ndarray) -> float:
    """Maximum entry of the linear convolution x*x (real FFT, fast)."""
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()  # next power of two
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return float(conv.max())


def _score_normalised(seq: np.ndarray) -> float:
    """
    For a normalised (ℓ¹‑norm = 1) non‑negative sequence returns the
    evaluation score `2·n·max(conv(seq,seq))`.  This is exactly the value
    produced by `evaluate_sequence` when all entries are ≤1000.
    """
    n = seq.shape[0]
    return 2.0 * n * _max_self_convolution(seq)


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients a_k = binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a / a.sum()


def _arcsine_sequence(n: int) -> np.ndarray:
    """Discrete arcsine law on {0,…,n‑1}: 1/√(x(1‑x)),  x = (i+0.5)/n."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid division by zero – x never equals 0 or 1
    a = 1.0 / np.sqrt(x * (1.0 - x))
    return a / a.sum()


def _shifted_power_law_sequence(n: int, alpha: float, shift: float) -> np.ndarray:
    """
    a_i ∝ (i + 1 + shift)^(-alpha).  The +1 prevents a division by zero
    when shift is 0.  The result is ℓ¹‑normalised.
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(i + 1.0 + shift, alpha)
    return a / a.sum()


def _candidate_score(seq: np.ndarray) -> float:
    """Convenient wrapper – normalises then returns the score."""
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total == 0.0:
        return np.inf
    seq = seq / total
    return _score_normalised(seq)


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine performs a fast global sweep over a
    few analytic families of sequences and subsequently refines the best
    candidate with simulated‑annealing hill‑climbing.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 4.0   # safety margin for finalisation

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – global sweep over simple parametric families
    # ------------------------------------------------------------------
    lengths = [2000, 4000, 6402, 8000, 12000]   # varied lengths, all feasible
    alphas = np.linspace(0.45, 0.55, 9)        # around the theoretical α=0.5
    shifts = np.linspace(0.0, 5.0, 11)        # 0 … 5 in 0.5 steps

    for n in lengths:
        if time.time() >= deadline:
            break

        # ----- Catalan (already known good baseline) -----
        seq = _catalan_sequence(n)
        sc = _score_normalised(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[baseline] n={n:5d} Catalan   score={sc:.6f}")

        # ----- Arcsine -----
        seq = _arcsine_sequence(n)
        sc = _score_normalised(seq)
        if sc < best_score:
            best_score, best_seq = sc, seq.copy()
            print(f"[baseline] n={n:5d} Arcsine   score={sc:.6f}")

        # ----- Shifted power‑law grid -----
        for alpha in alphas:
            for shift in shifts:
                if time.time() >= deadline:
                    break
                seq = _shifted_power_law_sequence(n, alpha, shift)
                sc = _score_normalised(seq)
                if sc < best_score:
                    best_score, best_seq = sc, seq.copy()
                    print(
                        f"[grid   ] n={n:5d} α={alpha: .3f} shift={shift: .2f} "
                        f"score={sc:.6f}"
                    )

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing refinement
    # ------------------------------------------------------------------
    if best_seq is None:                     # very unlikely, but keep a fallback
        best_seq = np.ones(1, dtype=np.float64)
        best_score = _score_normalised(best_seq)

    n = best_seq.shape[0]
    anneal_start = time.time()
    # linear temperature from 1 → 0 (scaled later)
    while time.time() < deadline:
        # pick two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or best_seq[i] <= 0.0:
            continue

        # annealing schedule (simple linear cooling)
        elapsed = time.time() - anneal_start
        total = deadline - anneal_start
        progress = min(1.0, elapsed / total)
        temperature = 1.0 - progress           # decreases from 1 to 0

        # step size adapts with temperature – smaller steps later
        max_frac = 0.15 * (temperature + 0.01)  # between 0.0015 and 0.15
        delta = max_frac * best_seq[i] * np.random.rand()
        if delta <= 0.0:
            continue

        # propose new sequence
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()               # renormalise to ℓ¹ = 1

        # quick score using the FFT helper
        new_score = _score_normalised(new_seq)

        # accept if improvement or with Metropolis probability
        if new_score < best_score:
            best_score, best_seq = new_score, new_seq
            print(
                f"[anneal] t={int(time.time() - start_time)}s "
                f"score={best_score:.6f}"
            )
        else:
            # probability = exp(-(Δ)/T); protect against division by zero
            if temperature > 1e-12:
                accept_prob = math.exp(-(new_score - best_score) / temperature)
                if np.random.rand() < accept_prob:
                    best_seq = new_seq

    # ------------------------------------------------------------------
    # Final tidy‑up and return
    # ------------------------------------------------------------------
    best_seq = np.maximum(best_seq, 0.0)
    best_seq /= best_seq.sum()                      # ensure exact ℓ¹ = 1
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] length={len(best_seq)} best_score={best_score:.6f}")
    print(f"[official] evaluate_sequence → {final_score:.6f}")

    return best_seq.tolist()
