def project_simplex(v):
    # returns projection onto simplex of sum=1
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v, s=1.0):
    """Projection of v onto the simplex {x >= 0, sum(x) = s}."""
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Proposed optimizer for the non‑negative step‑function problem.

The algorithm combines multiple diverse initialisations (power‑law,
exponential, random Dirichlet, the classic square‑root difference
profile) and a projected gradient descent with momentum on the
convolution peak.  At each iteration the gradient of the active
convolution entry is computed, a momentum term is updated and the new
candidate is projected onto the probability simplex (non‑negative,
unit‑sum).  Simulated‑annealing‐style uphill moves and periodic random
restarts help to escape local minima.  Finally a short random‑pair
local‑search polish is performed.  The routine respects the 1000 s

def _project_onto_simplex(v, s=1.0):
    """Euclidean projection of v onto the simplex {x >= 0, sum(x) = s}."""
    # Code adapted from Duchi et al. (2008)
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    """Return (max value, index) of the full autocorrelation a*a."""
    conv = np.convolve(a, a)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx

    # ----------------------------------------------------------------------

    final_seq = np.clip(final_seq, 0.0, 1000.0)
    print(f"Returning sequence with final score {best_score:.6f}")
    return final_seq.tolist()
