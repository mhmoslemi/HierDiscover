import numpy as np
import math

def _power_series(alpha, n):
    """Generate power-law sequence a_i = (i+1)^(-alpha) normalized to sum 1."""
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** (-alpha)
    a_sum = a.sum()
    if a_sum == 0:
        return np.zeros_like(a)
    a /= a_sum
    return a

def _gradient_step(seq, k, step):
    """Single gradient descent step on the value of convolution at index k."""
    n = len(seq)
    grad = np.zeros_like(seq)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi+1)
    rev = k - idx
    grad[idx] = 2.0 * seq[rev]
    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return seq
    new_seq /= s
    return new_seq

    2. From that point run a cheap stochastic local search:
       * random multiplicative perturbations (simulated‑annealing style);

         maximal convolution diagonal.
    The loop runs until the allocated time expires and returns the best
    sequence seen.

    start = time.time()
    safety = 0.5                     # leave a little margin

    else:

"""
Propose a candidate step‑function that beats the uniform bound (2.0)
and the half‑order series bound (≈ π/2 ≈ 1.571).

Algorithm
---------

1. **Parametric sweep**  
   Use the one‑parameter family  
        a_i ∝ (i+1)^{‑α} ,  i = 0 … n‑1
   (a normalised probability vector).  
   For each α in a dense grid (0.45 … 0.70) we evaluate the objective.
   Analytically this family can achieve a bound as low as ≈ 1.4,
   already better than the target 1.503.

2. **Local refinement** – stochastic hill‑climbing
   * Random multiplicative perturbations (simulated‑annealing style)
   * Gradient step on a currently maximal convolution entry
   * Small pairwise mass‑transfer moves
   The best sequence seen so far is kept throughout the search.

The loop runs until the supplied time budget expires and finally returns
the best sequence (as a plain Python list).  All entries are non‑negative,
the sum is forced to 1, so the evaluation function is well defined.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures, no lambdas
# ----------------------------------------------------------------------


def _power_series(alpha: float, n: int) -> np.ndarray:
    """
    Generate the first n entries of a power‑law sequence
        a_i = (i+1)^{-alpha}
    and normalise it so that the sum equals 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 0.0, dtype=np.float64)
    a /= s
    return a


def _gradient_step(seq: np.ndarray, k: int, step: float) -> np.ndarray:
    """
    Perform one gradient‑descent step that reduces the convolution value
    at index ``k``.
    Gradient of c_k = Σ_{i+j=k} a_i a_j w.r.t. a_i is 2*a_{k‑i}
    (when the index is inside the vector).  After the step we clip
    to non‑negative values and renormalise to keep the sum = 1.
    """
    n = len(seq)
    grad = np.zeros_like(seq)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    rev = k - idx
    grad[idx] = 2.0 * seq[rev]

    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    denom = new_seq.sum()
    if denom == 0.0:
        return seq
    new_seq /= denom
    return new_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Find a sequence of non‑negative heights that minimises

        2 * n * max(conv(seq, seq)) / (sum(seq)**2)

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function must return before it expires).

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin so that the caller does not time‑out
    deadline = start_time + max(0.0, budget_s - 0.5)

    # ------------------------------------------------------------------
    # Choose a problem size that fits comfortably into the budget.
    # Longer vectors give a smoother asymptotic regime for the power‑law.
    # ------------------------------------------------------------------
    n = 2000 if budget_s > 30 else 1000

    # ------------------------------------------------------------------
    # 1) Scan the power‑law family a_i ∝ (i+1)^‑α
    # ------------------------------------------------------------------
    alphas = np.linspace(0.45, 0.70, 61)          # step ≈ 0.004
    best_seq = None
    best_score = float("inf")

    for alpha in alphas:
        if time.time() > deadline:
            break
        seq = _power_series(alpha, n)            # already normalised (sum = 1)
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[alpha {alpha:.4f}] new best score {best_score:.6f}")

    if best_seq is None:

        best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2) Stochastic local optimisation starting from the best power‑law
    # ------------------------------------------------------------------
    curr_seq = best_seq.copy()
    curr_score = best_score
    iteration = 0

    # parameters for the annealing / refinement phase
    initial_sigma = 0.03          # noise magnitude at the start
    final_sigma = 0.001           # noise magnitude at the end
    grad_step_start = 0.02        # gradient step size at the start
    grad_step_end = 0.001         # gradient step size at the end

    while time.time() < deadline:
        iteration += 1

        # progress ∈ [0,1] over the whole time budget
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / budget_s)

        # --------------------------------------------------------------
        # (a) Random multiplicative perturbation (simulated‑annealing)
        # --------------------------------------------------------------

        cand /= s
