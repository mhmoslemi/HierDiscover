    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    step_frac = 0.2
    iteration = 0
    while time.time() < deadline:
        n = len(a)
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1)
        # compute products
        products = a[idx_range] * a[max_idx - idx_range]
        # exclude indices where product is zero maybe
        idx_local = np.argmax(products)
        i_max = idx_range[idx_local]
        j_max = max_idx - i_max
        # Determine removal amount
        max_removal = min(a[i_max], a[j_max])
        if max_removal <= 0:
            # can't remove; break or random move
            # fallback random move
            src = np.random.choice(n, p=a/a.sum())
            dest = np.random.randint(0, n-1)
            if dest >= src:
                dest += 1
            delta = a[src] * 0.01
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dest] += delta
            # renorm (sum unchanged)
        else:
            delta = max_removal * step_frac
            # Ensure delta not too small
            if delta < 1e-12:
                # reduce step_frac
                step_frac *= 0.9
                continue
            # Remove from i_max, j_max, add to ends
            dest1 = 0 if i_max != 0 else 1
            dest2 = n-1 if j_max != n-1 else n-2
            a_new = a.copy()
            a_new[i_max] -= delta
            a_new[j_max] -= delta
            a_new[dest1] += delta
            a_new[dest2] += delta
            # Clip
        # Ensure nonneg
        # Normalize (sum unchanged)
        a_new = np.maximum(a_new, 0.0)
        sum_a = a_new.sum()
        if sum_a == 0:
            break
        a_new /= sum_a
        # Compute new score
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

"""
Propose a candidate step‑function that beats the previous best (~1.5766).

Algorithm
---------
1. **Parametric search** – explore sequences of the form
       a_i ∝ (i + 1 + c)^{-p}
   for a handful of lengths `n`, exponents `p` and offsets `c`.
   The parameters are cheap to evaluate (one convolution per candidate) and
   give a wide variety of shapes, including the classic `p≈0.5` power law
   and its smoothed variants (c>0).  The best triple (n, p, c) is kept.

2. **Local refinement** – starting from the best parametric candidate,
   perform a simple sub‑gradient‑type hill‑climb.
   * Find the convolution index `k*` where the peak occurs.
   * Locate the pair (i, j) = (i, k*–i) that contributes the most to this
     peak (`a_i * a_j` is maximal).
   * Move a small fraction of mass from those two indices to the
     far‑ends of the vector (indices 0 and n‑1).  This reduces the large
     central product while preserving the total sum.
   * Accept the move only if the evaluation score improves (or with a tiny
     probability for uphill moves – simulated‑annealing style).
   * The step size decays over time, and the loop stops when the time budget
     is exhausted.

The whole search respects the 1000 s budget, prints progress, and returns
the best sequence (as a plain Python list) found so far.
"""

import numpy as np
import time
import random

def generate_shifted_power_law(n: int, p: float, c: float) -> np.ndarray:
    """
    Generate a non‑negative sequence of length ``n`` of the form
        a_i = ((i + 1) + c)^{-p}
    and normalise it so that sum(a) = 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) + c
    seq = idx ** (-p)
    total = seq.sum()
    if total == 0.0:
        # fallback to uniform – never happens for the ranges we use
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq /= total
    return seq


def hill_climb(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climb starting from ``a`` (which must already be
    normalised to sum = 1).  Returns the best sequence and its score.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = float("inf")
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        # evaluate_sequence should be available; if not, just return
        return best_seq, best_score

    # parameters for the climb
    step_frac = 0.20          # initial fraction of the removable mass
    iteration = 0

    while time.time() < deadline:
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))

        # range of i such that i + (k_max-i) = k_max and both in [0,n)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        prod = a[idx] * a[k_max - idx]

        # pick the pair contributing the most to the peak
        loc = int(np.argmax(prod))
        i_star = idx[loc]
        j_star = k_max - i_star

        # amount we can safely move
        removable = min(a[i_star], a[j_star])
        if removable <= 0.0:
            # fallback: random small move
            src = np.random.choice(n, p=a / a.sum())
            dest = random.randrange(n - 1)
            if dest >= src:
                dest += 1
            delta = a[src] * 0.01
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dest] += delta
        else:
            delta = removable * step_frac
            # destination indices – we push mass to both ends
            dest1 = 0 if i_star != 0 else 1
            dest2 = n - 1 if j_star != n - 1 else n - 2
            a_new = a.copy()
            a_new[i_star] -= delta
            a_new[j_star] -= delta
            a_new[dest1] += delta
            a_new[dest2] += delta

        # projection onto the feasible set (non‑negative, sum = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            break
        a_new /= s

        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # defensive – treat as infinite
            score = float("inf")

        # simple deterministic acceptance with a tiny simulated‑annealing tail
        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            a = a_new
            print(f"[hill] it={iteration:06d}, score={score:.6f}")
        else:
            # accept with a small probability that decays
            T = max(1e-4 * (1.0 - iteration / 200000.0), 1e-7)
            if np.random.rand() < np.exp(-(score - best_score) / T):
                a = a_new

        iteration += 1
        # slowly decay the step size – never let it become too tiny
        step_frac = max(step_frac * 0.9995, 1e-5)

        # occasional sanity break
        if iteration % 10000 == 0 and time.time() >= deadline:
            break

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score height sequence within the given time budget.

    The search proceeds in two phases:
      1. A coarse/fine grid over the parameters (n, p, c) of a shifted
         power‑law shape.
      2. A hill‑climb from the best grid candidate.
    The function returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # 1) GRID SEARCH OVER (n, p, c)
    # ------------------------------------------------------------------
    # If a previous construction exists, reuse its length as a baseline.
    try:
        n_start = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821
        if n_start < 500:
            n_start = 3000
    except Exception:
        n_start = 3000

    # A handful of lengths to test – larger n generally helps but costs more.
    n_grid = [2000, 3000, 4000, 5000, 6000, 8000]
    # Ensure we test the current length as well (if not already listed)
    if n_start not in n_grid:
        n_grid.append(n_start)

    # Exponent range – centred around the known sweet spot p≈0.5
    p_grid_coarse = np.linspace(0.45, 0.55, 21)   # 0.45 … 0.55
    # Offset c – smooths the head of the power law
    c_grid_coarse = np.linspace(0.0, 30.0, 7)    # 0, 5, 10, 15, 20, 25, 30

    best_seq = None
    best_score = float("inf")
    best_params = None  # tuple (n, p, c)

    print("[*] Starting coarse grid search...")
    for n in n_grid:
        if time.time() >= deadline:
            break
        for p in p_grid_coarse:
            if time.time() >= deadline:
                break
            for c in c_grid_coarse:
                if time.time() >= deadline:
                    break
                seq = generate_shifted_power_law(n, p, c)
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    score = float("inf")
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (n, p, c)
                    print(f"[grid] new best n={n}, p={p:.4f}, c={c:.2f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) FINE‑GRID REFINEMENT AROUND THE BEST PARAMETERS
    # ------------------------------------------------------------------
    if best_params is not None and time.time() < deadline:
        n_best, p_best, c_best = best_params
        # Narrow windows
        p_fine = np.linspace(max(0.3, p_best - 0.04), min(0.7, p_best + 0.04), 21)
        c_fine = np.linspace(max(0.0, c_best - 10.0), c_best + 10.0, 21)

        print("[*] Starting fine grid refinement...")
        for p in p_fine:
            if time.time() >= deadline:
                break
            for c in c_fine:
                if time.time() >= deadline:
                    break
                seq = generate_shifted_power_law(n_best, p, c)
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    score = float("inf")
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_params = (n_best, p, c)
                    print(f"[fine] new best n={n_best}, p={p:.5f}, c={c:.2f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) LOCAL HILL‑CLIMB FROM THE BEST GRID POINT
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("[*] Starting hill‑climb local optimisation...")
        # Ensure exact normalisation (evaluate_sequence is scale‑invariant,
        # but we keep sum=1 for numerical stability)
        best_seq = best_seq / best_seq.sum()
        refined_seq, refined_score = hill_climb(best_seq, deadline)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"[*] Hill‑climb improved score to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) SAFETY NET – fallback to uniform if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = n_start
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[*] Fallback to uniform sequence (score = {:.6f})".format(best_score))

    # Return as plain Python list as required by the harness.
    print("[*] Final best score = {:.6f}".format(best_score))
    return best_seq.tolist()
