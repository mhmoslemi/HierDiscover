    import numpy as np, time, scipy.optimize as opt

    budget = budget_s - 0.5
    deadline = time.time() + budget

    try: n = len(height_sequence_1)
    except: n = 4000

    # initial - use power law p=0.5 ~ best
    a = generate_power_law(n, 0.5)
    # ensure sum normalized to 1
    a = a/ a.sum()

    best_score = evaluate_sequence(a.tolist())
    best_seq = a.copy()
    print("initial score:", best_score)

    # gradient descent loop parameters
    step = 0.5 # maybe start moderate
    # Use backtracking line search to guarantee descent.
    # Will adapt: step reduces if no improvement
    while time.time() < deadline:
        # compute conv and max
        conv = np.convolve(a, a)
        idx = np.argmax(conv)
        max_val = conv[idx]

        # compute gradient g:
        g = np.zeros_like(a)
        # Determine i range:
        i_min = max(0, idx - (n - 1))
        i_max = min(n - 1, idx)
        # For i in [i_min, i_max]:
        # g[i] = 2 * a[idx - i]
        # vectorized:
        i_range = np.arange(i_min, i_max + 1)
        g[i_range] = 2.0 * a[idx - i_range]

        # gradient step: a_new = a - step * g
        # This may lead to sum < 1; we project onto simplex after step
        a_new = a - step * g
        # project onto simplex sum=1
        a_new = project_simplex(a_new, s=1.0)

        # compute new max conv
        conv_new = np.convolve(a_new, a_new)
        max_new = conv_new.max()
        # compute score (2*n*max_new)
        score_new = 2 * n * max_new

        if score_new < best_score - 1e-9:
            best_score = score_new
            best_seq = a_new.copy()
            print(f"[+] Improved to score {best_score:.6f}, step {step:.4f}")
            a = a_new
            # maybe increase step a bit
            step = min(step * 1.2, 1.0) # enlarge step but not exceed 1.0
        else:
            # no improvement: reduce step
            step *= 0.5
            if step < 1e-6:
                # step too small, break or random restart
                # random perturb a a bit
                step = 0.1
                # random perturb: add small noise and re-project
                noise = np.random.randn(n) * 0.01

                # keep a unchanged
                pass

        if time.time() >= deadline:
            break

    # final evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print("final score:", final_score)
    return best_seq.tolist()

"""
Propose a candidate step‑function sequence by directly minimizing the
self‑convolution peak.

The score is 2 * n * max(conv(a, a)) / (sum(a)**2).  By fixing sum(a)=1 the
objective reduces to 2 * n * max(conv(a, a)).  The algorithm performs a
projected (multiplicative) sub‑gradient descent on the simplex:

* Start from a power‑law (or a random Dirichlet) vector.
* In each iteration compute the self‑convolution and pick the few largest
  entries (by default the 5 biggest peaks).
* The sub‑gradient of the maximum peak is the sum of the gradients of those
  peaks (for a peak at lag k the gradient is 2*a_{k‑i} for each index i that
  participates in the lag).
* Update the vector multiplicatively:  a_i ← a_i * exp(-η * g_i) and
  renormalise to keep ∑a_i = 1.
* Use a simple back‑tracking line‑search on the learning rate η.
* If progress stalls, inject a small random perturbation (restart) to
  escape plateaus.

The loop runs until the allotted time budget (≈1000 s) is exhausted and the
best sequence found is returned.  The final score is obtained by calling the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np


def generate_power_law(n: int, p: float) -> np.ndarray:
    """
    Return a non‑negative sequence of length ``n`` following a power‑law
    a_i ∝ (i+1)^{-p} and normalised so that the sum equals 1.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = s}.  Implements the algorithm of
    Duchi et al. (2008).  Not used in the main loop (we employ a
    multiplicative update), but kept for fallback safety.
    """
    if s <= 0:
        raise ValueError("Simplex radius must be positive")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho_arr = u - cssv / (np.arange(1, n + 1))
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
