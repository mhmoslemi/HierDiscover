        best_score = evaluate_sequence(best_seq.tolist())

    a = best_seq.copy()

    L = 1 << ((2 * n - 1).bit_length())

    beta1 = 0.9

    iteration = 0

    # compute initial score

    cur_score = best_score
    while time.time() < deadline:
        iteration += 1
        # compute convolution via FFT
        A = np.fft.rfft(a, L)
        conv_full = np.fft.irfft(A * A, L)[:2 * n - 1]
        # compute maximum
        max_val = conv_full.max()
        # identify active indices (where conv close to max)
        # tolerance: maybe absolute 1e-10 * max_val or relative 1e-6
        tol = max(1e-12, max_val * 1e-9)

        # But simplest: loop over active (should be few, maybe 1-3)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad

        # renormalize to maintain total_mass
        s = a_new.sum()
        if s == 0.0:

            a_new *= (total_mass / s)

        # occasional jitter

            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= (total_mass / a_new.sum())

        # evaluate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception as e:
            # ignore errors

        # track best
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"iter {iteration:6d}: new best {best_score:.6f}")

        # acceptance rule: accept if improvement or if not much worse (like 0.0005% increase)

                cur_score = score

        lr *= 0.9999  # small decay

    # final return best_a.tolist()
    print(f"finished after {iteration} iterations, best score {best_score:.6f}")
    return best_a.tolist()

"""
Propose a low‑score step function.

The algorithm works in three stages:

1. **Coarse parametric sweep** –  generate sequences of the form
   a[i] = (i+1+offset)^(-exponent) for several lengths, exponents and
   offsets.  Every candidate is evaluated by ``evaluate_sequence`` and the
   best one is kept as a seed.

2. **Fast Adam‑style projected optimisation** –  starting from the best
   seed we minimise the maximal autocorrelation M = max(a*a).  The
   objective is 2·n·M/‖a‖₁², which is invariant under a global scaling, so
   the ℓ₁‑norm is kept constant.  The gradient of the active convolution
   entry (or the average of a few active entries) is used, the update is
   performed with Adam momentum, the result is projected onto the
   non‑negative simplex and renormalised to the original total mass.
   Convolutions are computed with an FFT (O(n log n)) to allow many
   iterations within the 1000 s budget.  Approximate scores are used for
   the inner loop; an exact call to ``evaluate_sequence`` is performed only
   when a genuine improvement is detected.

3. **Length trimming** –  after optimisation the tail of the sequence is
   optionally truncated (with renormalisation) to see whether a shorter
   sequence yields a lower bound.

The final list of coefficients is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return the full autocorrelation a*a using an FFT of length L."""
    A = np.fft.rfft(a, L)
    conv_full = np.fft.irfft(A * A, L)[: 2 * a.size - 1]
    # Numerical noise may create tiny negative values
    return np.maximum(conv_full, 0.0)


def _active_gradient(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximum (or near‑maximum) entry of a*a.
    The gradient for a single index k is
        g_j = 2 * a_{k-j}  for 0 ≤ k-j < n  else 0.
    When several active indices are present we average their gradients.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active_idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # indices j = lo … hi
        js = np.arange(lo, hi + 1, dtype=np.int64)
        grad[js] += 2.0 * a[k - js]
    # average over the number of active constraints
    grad /= max(1, active_idxs.size)
    return grad


def _project_onto_simplex(x: np.ndarray, total_mass: float) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the simplex
    {y ≥ 0, Σ y = total_mass}.  This implementation follows the classic
    O(n log n) algorithm.
    """
    if total_mass <= 0:
        return np.zeros_like(x)

    # ensure non‑negativity first
    x = np.maximum(x, 0.0)
    # sort descending
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - total_mass
    ind = np.arange(len(x)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen)
        tau = 0.0
    else:
        rho = ind[cond][-1] - 1
        tau = cssv[cond][-1] / (rho + 1)
    w = np.maximum(x - tau, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a sequence of non‑negative heights that minimises the given score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # a small safety margin

    # ------------------------------------------------------------------
    # 1) Coarse parametric sweep (arcsine‑type profiles)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    # candidate lengths – keep them moderate for fast evaluation
    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    exponents = np.linspace(0.45, 0.55, 11)          # around the known 0.5
    offsets = np.linspace(0.0, 2.0, 11)             # small horizontal shift

    print("=== parametric sweep ===")
    for n in lengths:
        idx = np.arange(n, dtype=np.float64) + 1.0  # i+1 part
        for p in exponents:
            base = np.power(idx, -p)
            for off in offsets:
                seq = np.power(idx + off, -p)          # (i+1+off)^{-p}
                # Clip to the allowed range (never really needed)
                seq = np.clip(seq, 0.0, 1000.0)
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(
                        f"len={n:4d}  p={p:.3f}  off={off:.2f} → score={score:.6f}"
                    )

    # also try the previously found sequence if it exists
    if "height_sequence_1" in globals():
        try:
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"height_sequence_1 improves to {score:.6f}")
        except Exception:
            pass

    # fallback – uniform vector (guaranteed to be valid)
    if best_seq is None:
        n = 3000
        best_seq = np.ones(n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"fallback uniform: score={best_score:.6f}")

    print(f"Initial best: length={len(best_seq)} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Fast Adam‑style projected optimisation
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.size
    total_mass = float(np.sum(a))
    # factor for the exact objective (used only for reporting)
    const_factor = 2.0 * n / (total_mass * total_mass)

    # FFT length – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.15                     # base learning rate
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    cur_score = best_score
    best_a = a.copy()
    iteration = 0

    # early‑stop helper
    def _approx_score(max_val):
        """Fast approximation of the objective using the FFT max."""
        return const_factor * max_val

    print("=== start gradient optimisation ===")
    while time.time() < deadline:
        iteration += 1

        # --------------- convolution via FFT -----------------
        conv_full = _conv_fft(a, L)
        max_val = float(conv_full.max())

        # ------- identify active (near‑max) indices ---------
        # tolerance is tiny relative to the maximum
        tol = max(1e-12, max_val * 1e-9)
        active = np.where(conv_full >= max_val - tol)[0]

        # ------------------- gradient -----------------------
        grad = _active_gradient(a, active)

        # ------------------- Adam update --------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # tentative new point
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)          # non‑negativity
        a_new = _project_onto_simplex(a_new, total_mass)

        # occasional tiny jitter to escape plateaus
        if iteration % 500 == 0:
            jitter = np.random.randn(n) * (total_mass / n) * 0.004
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = _project_onto_simplex(a_new, total_mass)

        # ----------------------------------------------------
        # evaluate (approx.) and decide on acceptance
        # ----------------------------------------------------
        approx_sc = _approx_score(max_val)

        # if the approximate score beats the best by any margin,
        # verify with the exact (slow) evaluator.
        if approx_sc < best_score * 0.9999:
            try:
                exact_sc = evaluate_sequence(a_new.tolist())
            except Exception:
                exact_sc = float("inf")
            if exact_sc < best_score:
                best_score = exact_sc
                best_a = a_new.copy()
                print(
                    f"iter {iteration:6d}: NEW BEST approx={approx_sc:.6f} exact={exact_sc:.6f}"
                )
        # acceptance rule – allow slight uphill moves for stability
        if approx_sc <= cur_score * 1.001:
            a = a_new
            cur_score = approx_sc
        else:
            # with a small probability accept a worse move (simulated‑annealing flavour)
            if np.random.rand() < 0.005:
                a = a_new
                cur_score = approx_sc

        # decay the learning rate very gently
        lr *= 0.99997

        # safety break if we are extremely close to the time limit
        if time.time() > deadline - 0.5:
            break

    # ------------------------------------------------------------
    # 3) Optional trimming of a trailing low‑mass tail
    # ------------------------------------------------------------
    final_seq = best_a.copy()
    final_score = best_score
    for ratio in (0.9, 0.8, 0.7, 0.6):
        cut = int(np.round(ratio * n))
        if cut < 2:
            continue
        truncated = final_seq[:cut]
        if truncated.sum() == 0:
            continue
        # renormalise to the original total_mass
        truncated = _project_onto_simplex(truncated, total_mass)
        try:
            sc = evaluate_sequence(truncated.tolist())
        except Exception:
            continue
        if sc < final_score:
            final_score = sc
            final_seq = truncated.copy()
            print(f"trim to {cut} elements → score={sc:.6f}")

    print(
        f"=== finished ===\n"
        f"iterations      : {iteration}\n"
        f"final length    : {len(final_seq)}\n"
        f"final best score: {final_score:.6f}"
    )
    return final_seq.tolist()
