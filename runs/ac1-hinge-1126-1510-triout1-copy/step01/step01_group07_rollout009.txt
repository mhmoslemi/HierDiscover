def fast_score(seq):
    n = len(seq)
    # Normalize sum to 1
    total = np.sum(seq)
    if total == 0:
        return np.inf
    seq_norm = seq / total
    # Compute convolution via FFT of length m = 1 << (2*n-1).bit_length()
    m = 1 << ((2*n-1).bit_length())
    fft_a = np.fft.rfft(seq_norm, n=m)
    conv = np.fft.irfft(fft_a*fft_a, n=m)[:2*n-1]
    max_conv = np.max(conv)
    # Score = 2*n*max_conv / (total**2) ??? Since seq_norm total=1, denominator 1.
    # Using original seq (not normalized) would be 2 * n * max(conv) / (total**2)

"""
Propose a candidate step‑function that beats the current best score (~1.5766).

Algorithm
---------

1.  Use a cheap yet accurate *fast_score* which evaluates the objective
    `2 * n * max(conv(a, a)) / (sum(a)**2)` via FFT.  The score is
    invariant under scaling, so the sequence is only required to be
    non‑negative.

2.  The search space is parametrised by a *shifted power‑law*
        a_i = (i + shift) ** (‑p)          (i = 0 … n‑1)
    where `p > 0` and `shift ≥ 0.5`.  This family includes the previous
    pure power‑law (shift = 1) and allows us to smooth the first few
    coefficients.

3.  A set of lengths (taken from the previous solution if available
    plus a few standard values) is explored.  For each length we perform
    a coarse grid search over `p` (0.45 … 0.75) and keep the best
    candidate.

4.  Around the globally best `(n, p, shift)` a finer grid is searched
    (Δp ≈ 0.02, Δshift ≈ 0.5).  Afterwards a short random‑sampling phase
    explores the neighbourhood further.

5.  The best raw sequence is finally evaluated with the supplied
    `evaluate_sequence` to obtain the authoritative score and then
    returned as a plain Python list.

The whole procedure respects the 1000 s time budget and prints progress
information.
"""

import time
import numpy as np

def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective using FFT.

    The function works with any non‑negative sequence `seq`.  The score is
    `2 * n * max(conv(seq, seq)) / (sum(seq)**2)`.  Because scaling cancels,
    we normalise `seq` to sum to 1, compute the convolution by FFT and
    return `2 * n * max(conv_norm)`.
    """
    n = seq.shape[0]
    total = np.sum(seq)
    if total <= 0.0:
        return float("inf")
    # normalise – this does not affect the final score
    a = seq / total
    # length for FFT (next power of two >= 2*n-1)
    m = 1 << ((2 * n - 1).bit_length())
    # real‑valued FFT is sufficient (seq is real)
    fft = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fft * fft, n=m)[: 2 * n - 1]
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv)


def generate_power_law_shift(n: int, p: float, shift: float = 1.0) -> np.ndarray:
    """Return the (i+shift)^(-p) sequence for i=0…n‑1."""
    idx = np.arange(n, dtype=np.float64) + shift
    seq = idx ** (-p)
    return seq
