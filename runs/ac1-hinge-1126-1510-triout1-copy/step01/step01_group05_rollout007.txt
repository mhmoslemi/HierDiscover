def hill_climb(a, score, fft_len, n, deadline, max_iter=2000):
    best = a.copy()
    best_score = score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        if rng.random() < 0.7:   # random move
            i,j = rng.integers(0,n,2)
            if i == j: continue
            delta = best[i] * rng.random() * 0.1
            if delta <= 0: continue
            best[i] -= delta
            best[j] += delta
        else:
            # gradient directed
            maxc, g = max_conv_and_subgrad(best)
            # find i where g_i largest positive (should increase conv) and j where small.
            i = np.argmax(g)
            j = np.argmin(g)
            if i == j: continue
            delta = best[i] * 0.05
            if delta <= 0: continue
            best[i] -= delta
            best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep
        else:
            # revert
            # for random move revert
            if rng.random() < 0.7:
                best[i] += delta
                best[j] -= delta
            else:
                best[i] += delta
                best[j] -= delta
    return best, best_score

"""
Propose a low‑evaluation step‑function sequence.

The search combines three ideas:

1. **Smooth‑max L‑BFGS** – the objective 2·n·max(a∗a) is approximated by a
   soft‑max (controlled by a λ‑parameter).  Starting from many random
   points we run a few L‑BFGS steps, decreasing λ over time to sharpen
   the approximation.

2. **Gradient‑guided hill climbing** – after each L‑BFGS run a fast greedy
   mass‑transfer hill‑climb (O(n log n) per move via FFT) refines the
   candidate.  Occasionally the move direction is taken from the sub‑gradient
   of the exact max‑convolution.

3. **Mirror‑descent polishing** – when the time budget is almost exhausted,
   we run a short projected‑gradient (mirror‑descent) loop directly on the
   soft‑max parameters y, using the exact sub‑gradient of the max‑convolution
   for a very fine local optimisation.

All steps respect the simplex (non‑negative, sum‑to‑one) constraints, so the
result can be handed to ``evaluate_sequence`` without further processing.
The algorithm works for any length ``n``; here we use ``n=3500`` which
offers a good trade‑off between solution quality and speed within the
1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a∗a (length 2·n‑1) using FFT.
    The caller ensures ``fft_len`` is a power of two ≥ 2·n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector ``a``:
    2·n·max(a∗a).  ``a`` must already sum to 1 (or any positive constant –
    the ratio is scale‑invariant).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def max_conv_and_subgrad(a: np.ndarray) -> (float, np.ndarray):
    """
    Compute (max_convolution, sub‑gradient) of the exact max‑convolution.
    The sub‑gradient g satisfies ∂max(a∗a)/∂a = g.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    k0 = int(np.argmax(conv))
    max_c = conv[k0]

    # sub‑gradient: for each i, g[i] = 2·a[j] where j = k0‑i (if in range)
    g = np.zeros_like(a)
    lo = max(0, k0 - (n - 1))
    hi = min(k0, n - 1)
    j = np.arange(lo, hi + 1, dtype=int)
    i = k0 - j
    g[i] = 2.0 * a[j]

    return max_c, g


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max surrogate of 2·n·max(a∗a) and its gradient w.r.t. the
    unconstrained variables ``y`` (a = softmax(y)).
    The smoothing width λ = lam_factor·max(conv) + 1e‑12.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice the part that corresponds to the gradient w.r.t. a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    # The factor 2·n is omitted (it only rescales the step size);
    # L‑BFGS adapts the step length automatically.
    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Greedy hill‑climb on the simplex.
    Moves a random fraction (≤ 10 %) of mass from one coordinate to another.
    Occasionally (≈ 20 % of attempts) a gradient‑guided move is performed.
    The sum stays exactly 1, and the fast O(n log n) score is used for acceptance.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        if rng.random() < 0.8:                 # random mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = best[i] * rng.random() * 0.10   # up to 10 % of i‑mass
            if delta <= 0.0:
                continue
            best[i] -= delta
            best[j] += delta
        else:                                   # gradient‑guided move
            _, g = max_conv_and_subgrad(best)
            i = int(np.argmax(g))               # heavy‑gradient index
            j = int(np.argmin(g))               # light‑gradient index
            if i == j or best[i] <= 0.0:
                continue
            delta = best[i] * 0.05
            best[i] -= delta
            best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert the move
            if rng.random() < 0.8:
                best[i] += delta
                best[j] -= delta
            else:
                best[i] += delta
                best[j] -= delta

    return best, best_score


def mirror_descent_refine(a: np.ndarray,
                          fft_len: int,
                          n: int,
                          deadline: float,
                          steps: int = 3000) -> (np.ndarray, float):
    """
    Fine‑grained mirror‑descent (projected gradient) on the soft‑max parameters.
    Uses the exact sub‑gradient of the true max‑convolution.
    """
    y = np.log(a + 1e-12)                     # initialise log‑weights
    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    for t in range(steps):
        if time.time() >= deadline:
            break
        # exact sub‑gradient of max(a∗a)
        maxc, g = max_conv_and_subgrad(best)
        # gradient w.r.t. y (soft‑max Jacobian)
        grad_y = best * (g - np.dot(best, g))
        # step‑size schedule – modest decay
        step = 0.5 / np.sqrt(t + 1.0)
        y -= step * grad_y
        best = softmax(y)
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for final eval

    # problem size – 3500 works well within the time limit
    n = 3500

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # start from the uniform distribution (baseline)
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast score = {best_score:.6f}")

    rng = np.random.default_rng(seed)
    iteration = 0
    lam_factor = 0.60          # initial smoothing, will be adapted

    # ------------------------------------------------------------------
    # Global optimisation – many random restarts with L‑BFGS + hill‑climb
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- pick an initial point on the simplex (via y) ----
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'arcsine'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.015)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 1.5)
            a0 = (np.arange(1, n + 1) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'edge':
            # a few heavy entries at the borders, the rest very flat
            m_edge = max(1, int(n * rng.uniform(0.008, 0.02)))
            heavy = np.sqrt(n / 2.0) / n
            flat = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, flat)
            a0[:m_edge] = heavy
            a0[-m_edge:] = heavy
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # arcsine‑like distribution
            i = np.arange(n, dtype=float) + 0.5
            a0 = 1.0 / np.sqrt(i * (n + 0.5 - i))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)

        # ---- run a short L‑BFGS optimisation on the smooth surrogate ----
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam_factor),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 300,
                                'gtol': 1e-9,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        # ---- keep the best candidate ------------------------------------------------
        improved = False
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            improved = True
            print(f"[{iteration}] mode={mode:<8}  L‑BFGS fast score = {best_score:.6f}")

        # ---- adaptive smoothing: tighten after a success -------------------------------
        if improved:
            lam_factor = max(lam_factor * 0.85, 0.02)   # make smoothing sharper
        else:
            lam_factor = min(lam_factor * 1.05, 0.80)   # keep a broader view

        # ---- local greedy refinement -------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=2500)

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing – mirror‑descent on the best candidate
    # ------------------------------------------------------------------
    final_deadline = min(deadline, time.time() + 30.0)
    best_a, best_score = mirror_descent_refine(best_a,
                                               fft_len_conv,
                                               n,
                                               final_deadline,
                                               steps=4000)

    # ------------------------------------------------------------------
    # Return the sequence (the official evaluator will clip / normalise)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)   # official, slower, final check
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
