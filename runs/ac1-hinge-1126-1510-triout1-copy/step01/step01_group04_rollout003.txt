def compute_score(a):
    a = np.clip(a, 0, 1000)  # maybe after negativity is already managed
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    # compute convolution using FFT:
    n = a.shape[0]
    size = 1 << (2*n-1).bit_length()
    # compute FFT
    fft_a = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:2*n-1]  # real conv
    max_b = conv.max()
    score = 2.0 * n * max_b / (sum_a*sum_a)
    return score

def compute_gradient_max(a, active_idxs):

"""
Propose a step‑function (non‑negative vector) with a very low
evaluate_sequence() score.

Algorithm
---------
1. **Fast score** – Uses an FFT based convolution (O(n log n)) that
   reproduces the exact behaviour of ``evaluate_sequence`` but is
   orders of magnitude faster than the built‑in naïve convolution.

2. **Coarse‑grid + random search** – Scan a modest set of lengths,
   exponents *p* (a[i] ∝ (i+δ)^‑p) and offsets δ.  The 30‑odd best
   candidates are kept for refinement.  A short random search over the
   same parameter space is also performed to avoid missing a good
   region.

3. **Refinement** – For each promising seed a *projected accelerated
   gradient descent* (momentum) is applied directly on the vector.
   The objective is the maximal entry of the auto‑convolution; the
   sub‑gradient is obtained analytically (see ``_gradient_max``).  The
   update is projected onto the hyper‑plane Σa = const and onto the
   non‑negative orthant.  A diminishing step size together with occasional
   Gaussian jitter helps to escape plateaus.

4. **Time management** – All work respects the supplied ``budget_s``
   (default 1000 s).  The routine stops a few seconds before the
   deadline and returns the best sequence found.

The code uses only the standard library, ``numpy`` and ``scipy`` (for the
optional linear‑programming fallback – not required for the main loop).
It never touches the filesystem and prints progress information only.
"""

import time
import math
import numpy as np
from scipy import optimize  # imported for possible fall‑backs only


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Return a non‑negative vector of the given length with the shape

        a[i] = (i + 1 + offset) ** (-exponent)

    Values are clipped to the [0, 1000] range required by
    ``evaluate_sequence``.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _fast_score(a: np.ndarray) -> float:
    """
    Compute the same objective as ``evaluate_sequence`` but with an FFT
    based convolution for speed.
    """
    a = np.clip(a, 0.0, 1000.0)          # the security‑clamps
    s = a.sum()
    if s < 0.01:
        return float("inf")
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def _conv_and_max(a: np.ndarray):
    """
    Return (conv, max_value, indices_of_max) for the auto‑convolution
    of ``a``.  ``conv`` is a 1‑D array of length 2*n‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_val = conv.max()
    # treat all indices whose value is within 1e‑12 of the maximum as active
    active = np.where(conv >= max_val - 1e-12)[0]
    return conv, max_val, active


def _gradient_max(a: np.ndarray, active_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry (or the sum of all
    active entries) w.r.t. the vector ``a``.
    For a fixed index k the gradient component i is
        ∂/∂a_i (a*a)[k] = 2 * a_{k-i}
    (out‑of‑range indices contribute 0).  The gradient for a set of
    active indices is the sum of the individual gradients.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in active_idxs:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i = np.arange(lo, hi + 1, dtype=np.int64)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


def _project_onto_sum_hyperplane(x: np.ndarray, total: float) -> np.ndarray:
    """
    Scale a non‑negative vector so that its ℓ1‑norm equals ``total``.
    The scaling factor is total / sum(x) (if sum > 0).
    """
    cur = x.sum()
    if cur == 0.0:
        return x.copy()
    return x * (total / cur)

# Main candidate proposer
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises
        2 * n * max(conv(a, a)) / (sum(a)**2)

    The search comprises:
        * a fast grid + random sweep over (length, exponent, offset);
        * accelerated projected gradient descent started from the
          best few seeds;
        * occasional stochastic jitter to escape flat regions.
    The function respects the supplied time budget (seconds) and
    returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1) Coarse sweep + random sampling over the parametric family
    # ------------------------------------------------------------------
    print("=== Stage 1: coarse grid + random search ===")
    param_candidates = []                     # (score, length, p, off)
    best_score = float("inf")
    best_seq = None

    # modest length set – long enough to capture the asymptotic shape,
    # short enough for a quick evaluation.
    lengths = [1200, 2000, 3000, 5000]

    # deterministic grid
    exponents_grid = np.linspace(0.35, 0.80, 10)   # 0.35 … 0.80
    offsets_grid = np.linspace(0.0, 5.0, 11)       # 0 … 5

    for n in lengths:
        for p in exponents_grid:
            for off in offsets_grid:
                seq = _initial_sequence(n, exponent=p, offset=off)
                sc = _fast_score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                # keep a short ordered list of the best few seeds
                param_candidates.append((sc, n, p, off))
    # random sampling to diversify the pool
    for _ in range(400):
        n = np.random.choice(lengths)
        p = np.random.uniform(0.30, 0.90)
        off = np.random.uniform(0.0, 7.0)
        seq = _initial_sequence(n, exponent=p, offset=off)
        sc = _fast_score(seq)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
        param_candidates.append((sc, n, p, off))

    # keep the 8 best parameter triples
    param_candidates.sort(key=lambda x: x[0])
    top_params = param_candidates[:8]
    print(f"Coarse search best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine each promising seed with accelerated projected gradient
    # ------------------------------------------------------------------
    print("\n=== Stage 2: accelerated projected gradient refinement ===")
    # A fixed final length – we re‑instantiate every seed at this length
    final_len = 5000
    total_mass_target = None                     # will be set per seed

    overall_best_seq = best_seq
    overall_best_score = best_score

    # allocate roughly 80 % of remaining time to refinement
    refinement_deadline = deadline - 2.0

    # iterate over the selected seeds
    for idx, (seed_score, seed_len, seed_p, seed_off) in enumerate(top_params, 1):

        a = _project_onto_sum_hyperplane(a, a.sum())   # just to be explicit
        total_mass_target = a.sum()

        best_local_seq = a.copy()
        best_local_score = cur_score

        # momentum variables
        momentum = np.zeros_like(a)
        mu = 0.85                              # momentum coefficient
        base_step = 0.5                        # initial step magnitude

        it = 0
        stagnation_counter = 0

        while time.time() < refinement_deadline and it < 25000:
            it += 1

            conv, max_val, active = _conv_and_max(a)

            # gradient of the active maximal entry (sum over all ties)
            grad = _gradient_max(a, active)

            # project gradient onto the hyper‑plane Σa = total_mass_target
            grad -= grad.mean()

            a_new = a - momentum

                cur_score = _fast_score(a)
