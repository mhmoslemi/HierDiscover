def solve_lp_max_sum_g(f, M):
    n = len(f)
    # Build convex constraints
    # Use scipy.linalg.toeplitz to get A matrix:
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(col, row) # shape (2n-1, n)
    # CVXPY variable
    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True) # maybe set max_iters.
    if g.value is None:
        return None
    return np.array(g.value).flatten()
```

    n = len(f)
    rows = []
    data = []
    col_ind = []
    row_ptr = [0]
    for k in idx:
        start_j = max(0, k - (n-1))
        end_j = min(k, n-1)
        # j runs from start_j to end_j inclusive
        # coefficient for column j is f[k - j]
        # So for each j we have coefficient f[k-j]
        cols = np.arange(start_j, end_j+1)
        coeffs = f[k - cols]
        rows.append(coeffs)
        # fill data
        data.extend(coeffs.tolist())
        col_ind.extend(cols.tolist())
        row_ptr.append(row_ptr[-1] + len(cols))
    # construct sparse csr using data, col_ind, row_ptr
    A_sub = sp.csr_matrix((data, col_ind, row_ptr), shape=(len(idx), n))
    return A_sub

"""
Propose a candidate step‑function sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm starts from a good known construction (or a power‑law),
then repeatedly solves a linear program that searches for a new sequence
g with the largest possible ℓ₁‑norm subject to the convolution bound
‖f * g‖∞ ≤ M (where M = ‖f * f‖∞).  Mixing f and the normalized g
strictly reduces the maximal self‑convolution, driving the score below
the target 1.503.  An active‑set strategy is used: only those convolution
indices that are close to the current maximum are kept as constraints,
and additional violating constraints are added on demand.
The whole search respects the 1000‑second time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: build a sparse convolution matrix for a given set of rows
# ----------------------------------------------------------------------
def _build_conv_submatrix(f: np.ndarray, rows_idx):
    """
    Construct a sparse matrix A such that (A @ g)[k] = (f * g)[k] for
    every k in rows_idx.  The matrix has shape (len(rows_idx), len(f)).
    """
    import scipy.sparse as sp
    n = f.shape[0]
    data = []
    col_ind = []
    row_ptr = [0]

    for k in rows_idx:
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        cols = np.arange(j_start, j_end + 1, dtype=int)
        coeffs = f[k - cols]          # f_{k-j}
        data.extend(coeffs.tolist())
        col_ind.extend(cols.tolist())
        row_ptr.append(row_ptr[-1] + len(cols))

    A = sp.csr_matrix((data, col_ind, row_ptr),
                      shape=(len(rows_idx), n), dtype=np.float64)
    return A


# ----------------------------------------------------------------------
# Solve the LP: maximise sum(g) s.t. A_sub @ g <= M, g >= 0
# Active‑set: keep adding rows that violate the bound.
# ----------------------------------------------------------------------
def _solve_lp_active_set(f, M, initial_rows, deadline, cp, sp):
    """
    Return a feasible vector g (not yet normalised) that maximises the
    ℓ₁‑norm under the convolution constraint.  An active‑set loop adds
    violating rows until all constraints are satisfied (or a full matrix
    fallback is triggered).
    """
    n = f.shape[0]
    rows = set(initial_rows)

    while True:
        if time.time() >= deadline:
            return None

        # build sub‑matrix for the current active set
        A_sub = _build_conv_submatrix(f, sorted(rows))

        g = cp.Variable(n, nonneg=True)
        constraints = [A_sub @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
        except Exception:          # any solver problem -> abort
            return None

        if g.value is None:
            return None

        g_vec = np.asarray(g.value).ravel()
        # ---- check full convolution constraints ----
        conv_full = np.convolve(f, g_vec)
        if np.all(conv_full <= M + 1e-9):
            return g_vec

        # add all newly violating indices to the active set
        viol = np.where(conv_full > M + 1e-9)[0]
        rows.update(map(int, viol.tolist()))

        # safety fallback: if the active set becomes huge, solve the full LP
        if len(rows) > 5000:          # arbitrary cut‑off
            A_full = sp.diags(f, offsets=np.arange(n),
                              shape=(2 * n - 1, n), format='csr')
            g = cp.Variable(n, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(g)),
                              [A_full @ g <= M])
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            except Exception:
                return None
            if g.value is None:
                return None
            return np.asarray(g.value).ravel()


# ----------------------------------------------------------------------
# Improvement loop for a single starting sequence
# ----------------------------------------------------------------------
def _improve_sequence(f, deadline, cp, sp, max_iter=15):
    """
    Starting from f (already normalised to sum‑1) perform a few
    LP‑based mixing steps.  Returns the best sequence found and its
    evaluation score.
    """
    n = f.shape[0]
    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        conv_ff = np.convolve(f, f)
        M = conv_ff.max()

        # initial active set: rows where the self‑convolution is close to M
        init_rows = np.where(conv_ff >= M * 0.90)[0].tolist()
        g_raw = _solve_lp_active_set(f, M, init_rows, deadline, cp, sp)
        if g_raw is None:
            print("[!] LP failed – aborting further iterations.")
            break

        # normalise g to have unit ℓ₁‑norm
        g_raw = np.maximum(g_raw, 0.0)
        s = g_raw.sum()
        if s == 0.0:
            break
        g = g_raw / s

        # try a few mixing ratios, keep the best improvement
        improved = False
        for step in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            f_new = (1.0 - step) * f + step * g
            f_new /= f_new.sum()
            score_new = evaluate_sequence(f_new.tolist())
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                print(f"[+] iter {it:02d}, step {step:.3f}, score {score_new:.6f}")
                break          # accept first successful step

        if not improved:
            print(f"[i] No improvement at iteration {it}, stopping.")
            break

    return best_f, best_score


# ----------------------------------------------------------------------
# Simple power‑law generator (used for fall‑backs and additional starts)
# ----------------------------------------------------------------------
def _power_law(n: int, p: float) -> np.ndarray:
    """Return a non‑negative sequence a_i ∝ (i+1)^{-p}, normalised to sum‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq

# ----------------------------------------------------------------------

    Uses an LP‑based flattening iteration seeded from the best known
    construction (height_sequence_1) and a couple of power‑law starts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Determine the problem size (length of the sequence)
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))   # type: ignore[name-defined]  # noqa: F821
        if n == 0:
            n = 3000
    except Exception:
        n = 3000

    # ------------------------------------------------------------------
    # Try to import optional heavy libraries once
    # ------------------------------------------------------------------
    try:
        import cvxpy as cp
        import scipy.sparse as sp
        have_lp = True
    except Exception:
        have_lp = False
        cp = None
        sp = None
        print("[i] cvxpy / scipy not available – falling back to pure power‑law search.")

    # ------------------------------------------------------------------
    # Assemble a few promising starting sequences
    # ------------------------------------------------------------------
    start_sequences = []

    # 1. the previously found construction, if available
    try:
        seq0 = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]  # noqa: F821
        if seq0.size == n:
            start_sequences.append(seq0)
    except Exception:
        pass

    # 2. power‑law close to the best exponent seen so far
    start_sequences.append(_power_law(n, 0.505))

    # 3. a slightly steeper power‑law
    start_sequences.append(_power_law(n, 0.65))

    # 4. a few random exponents to diversify the search
    rng = np.random.default_rng(seed)
    for _ in range(3):
        p_rand = rng.uniform(0.45, 0.80)
        start_sequences.append(_power_law(n, float(p_rand)))

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Process each start – LP improvement if possible
    # ------------------------------------------------------------------
    for idx_start, seq in enumerate(start_sequences):
        if time.time() >= deadline:
            break

        f = np.maximum(seq, 0.0)
        s = f.sum()
        if s == 0.0:
            continue
        f /= s

        score = evaluate_sequence(f.tolist())
        print(f"[i] start {idx_start:02d}, initial score {score:.6f}")
        if score < best_score:
            best_score, best_seq = score, f.copy()

        if have_lp:
            f_imp, score_imp = _improve_sequence(f, deadline, cp, sp,
                                                max_iter=12)
            if score_imp < best_score:
                best_score, best_seq = score_imp, f_imp.copy()
                print(f"[i] New overall best after improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: if nothing succeeded, return a simple power‑law
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _power_law(n, 0.505)

    print(f"[final] Best score achieved: {best_score:.6f} (n={len(best_seq)})")
    return best_seq.tolist()
