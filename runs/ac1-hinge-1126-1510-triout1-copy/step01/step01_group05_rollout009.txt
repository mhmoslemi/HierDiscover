def objective_lp_and_grad(y, n, fft_len_conv, fft_len_g, p):
    a = softmax(y)
    # compute conv
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    # compute S = sum(conv**p)
    conv_pow_p = conv ** p
    S = conv_pow_p.sum()
    if S <= 0:
        # return large value
        f = np.inf
        grad_y = np.zeros_like(y)
        return f, grad_y
    f = 2.0 * n * (S ** (1.0 / p))
    # compute w = conv**(p-1)
    w = conv ** (p - 1)
    # cross-correlation a_rev * w
    a_rev_fft = np.fft.fft(a[::-1], fft_len_g)
    w_fft = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(a_rev_fft * w_fft))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]
    factor = 4.0 * n * (S ** (1.0 / p - 1.0))
    grad_a = factor * g_vec
    # chain rule for softmax
    mean_g = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_g)
    return f, grad_y
```

def hill_climb_targeted(a, current_score, fft_len_conv, n, deadline, max_iter=5000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute conv and argmax
        conv = conv_fft(best, fft_len_conv, n)
        k_max = int(np.argmax(conv))
        # compute weights for source selection proportional to a[i] * a[k_max-i]
        # produce weight array w_source length n
        idx = np.arange(n)
        counterpart = k_max - idx
        mask = (counterpart >=0) & (counterpart < n)
        w_source = np.zeros(n)
        w_source[mask] = best[idx[mask]] * best[counterpart[mask]]
        total = w_source.sum()
        if total == 0:
            # fallback random
            i = rng.integers(0, n)
        else:
            prob = w_source / total
            i = rng.choice(n, p=prob)
        # destination choose index with low contribution to max
        # We can choose j uniformly at random, or weighted by inverse of w_source
        # Let's try uniform j != i
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # move delta from i to j. Use fraction up to maybe 20% of a[i]
        if best[i] <= 0:
            continue
        max_frac = 0.2
        delta = best[i] * rng.random() * max_frac
        # apply move
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len_conv, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep move
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

def run_adam(y0, n, fft_len_conv, fft_len_g, p, max_iters, deadline, lr=0.1):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_y = y.copy()
    best_score = np.inf
    while t < max_iters and time.time() < deadline:
        t += 1
        f, grad = objective_lp_and_grad(y, n, fft_len_conv, fft_len_g, p)
        # Convert grad to steps
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # Evaluate fast score occasionally, say every 10 iterations
        if t % 10 == 0:
            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_y = y.copy()
    # Return best a and its fast score
    a_best = softmax(best_y)
    final_score = fast_score(a_best, fft_len_conv, n)
    return a_best, final_score
```

"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
* Work on the probability simplex (sum = 1) by representing the
  coefficients as a = softmax(y).  This guarantees non‑negativity.
* Use three complementary optimisation stages per restart:
  1. Multi‑phase L‑BFGS on a smooth‑max (log‑sum‑exp) objective with a
     decreasing smoothing factor.  This quickly produces a good
     “coarse” shape.
  2. Adam optimiser on an L^p‑norm approximation of the true max
     (p grows large, typically 10‑20).  The L^p objective
     2·n·(∑(a*a)^p)^{1/p} directly targets the maximum convolution
     entry while remaining differentiable.
  3. A greedy, targeted hill‑climb that moves mass away from the
     positions that contribute most to the current maximal convolution
     entry (i.e. from the heavy side of the “autocorrelation” peak) to
     low‑impact positions.
* After each candidate is generated we enforce left‑right symmetry
  (a ← (a+a[::-1])/2) – the problem is symmetric and symmetry often
  reduces the peak.
* Multiple random restarts are performed until the time budget (≈1000 s)
  is exhausted.  The best sequence found is returned and evaluated by
  the supplied ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum=1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·n‑1) using FFT of length ``fft_len``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a sequence that already sums to 1:
    score = 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) objective – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective (log‑sum‑exp) and its gradient w.r.t. ``y``.
    Returns f = 2·n·smooth_max(a*a) and ∇_y f.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of conv
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12                # smoothing width
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                                 # ∂smooth_max/∂conv_i

    # g_i = Σ_j a_j·w_{i+j}  (i = 0..n‑1)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L^p‑norm objective (p>1) – used by Adam optimiser
# ----------------------------------------------------------------------


def objective_lp_and_grad(y: np.ndarray, n: int,
                         fft_len_conv: int, fft_len_g: int,
                         p: int = 12):
    """
    L^p‑norm approximation of max(a*a) and its gradient w.r.t. ``y``.
    f = 2·n·( Σ (a*a)^p )^{1/p}
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # L^p norm of the convolution entries
    conv_pow_p = conv ** p
    S = conv_pow_p.sum()
    if S <= 0.0:
        # degenerate case – return large value and zero gradient
        return float('inf'), np.zeros_like(y)

    # smooth objective value
    f = 2.0 * n * (S ** (1.0 / p))

    # gradient w.r.t. a (see analysis for derivation)
    # w = conv^{p‑1}
    w = conv ** (p - 1)

    # cross‑correlation a_rev * w (same trick as in smooth‑max)
    a_rev_fft = np.fft.fft(a[::-1], fft_len_g)
    w_fft = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(a_rev_fft * w_fft))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # factor = 4·n·S^{1/p‑1} = 4·n / S^{(p‑1)/p}
    s_root = S ** (1.0 / p)                     # ≈ smooth max
    factor = 4.0 * n / (s_root ** (p - 1))

    grad_a = factor * g_vec

    # chain rule through the soft‑max
    mean_g = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser (generic – works with any objective that returns f,g)
# ----------------------------------------------------------------------


def run_adam(y0: np.ndarray, n: int,
             fft_len_conv: int, fft_len_g: int,
             p: int, max_iters: int,
             deadline: float, lr: float = 0.15) -> (np.ndarray, float):
    """
    Run Adam on the L^p‑norm objective starting from ``y0``.
    Returns the best sequence (as a probability vector) and its fast score.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_score = float('inf')
    best_y = y.copy()
    t = 0

    while t < max_iters and time.time() < deadline:
        t += 1
        f, grad = objective_lp_and_grad(y, n, fft_len_conv, fft_len_g, p)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Check the true max‑convolution score every few steps
        if t % 10 == 0:
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_y = y.copy()

    a_best = softmax(best_y)
    final_score = fast_score(a_best, fft_len_conv, n)
    return a_best, final_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (mass transfer focussed on the max‑conv index)
# ----------------------------------------------------------------------


def hill_climb_targeted(a: np.ndarray, current_score: float,
                        fft_len_conv: int, n: int,
                        deadline: float,
                        max_iter: int = 4000) -> (np.ndarray, float):
    """
    Greedy hill‑climb that preferentially moves mass away from the
    positions contributing most to the current maximal convolution entry.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # recompute the convolution and the index of its maximum
        conv = conv_fft(best, fft_len_conv, n)
        k_max = int(np.argmax(conv))

        # contribution weight for each index i: a[i] * a[k_max‑i]
        idx = np.arange(n)
        counterpart = k_max - idx
        mask = (counterpart >= 0) & (counterpart < n)
        w_source = np.zeros(n)
        w_source[mask] = best[idx[mask]] * best[counterpart[mask]]
        total = w_source.sum()

        if total > 0.0:
            # choose source i with probability proportional to its contribution
            probs = w_source / total
            i = rng.choice(n, p=probs)
        else:
            # fallback: random source
            i = rng.integers(0, n)

        # choose a destination j (avoid i)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        if best[i] <= 0.0:
            continue

        # move a random fraction (up to 20 %) of mass from i to j
        delta = best[i] * rng.random() * 0.20
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len_conv, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # keep the improvement
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    The function respects a 1000‑second budget; after that it returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for final eval

    # ------------------------------------------------------------------
    # Problem size – try to stay comparable to the known baseline (2000)
    # ------------------------------------------------------------------
    try:
        # existing construction may be available as a global variable
        base_seq = np.array(height_sequence_1, dtype=float)
        n = len(base_seq)
        base_seq = base_seq / base_seq.sum()
        y_base = np.log(base_seq + 1e-12)
    except Exception:
        n = 2000
        base_seq = np.full(n, 1.0 / n)
        y_base = np.zeros(n)                     # uniform → softmax yields uniform

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial best candidate
    # ------------------------------------------------------------------
    best_a = base_seq.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] base fast score = {best_score:.6f}")

    # also keep uniform as a fallback (just in case)
    uniform_a = np.full(n, 1.0 / n)
    uniform_score = fast_score(uniform_a, fft_len_conv, n)
    if uniform_score < best_score:
        best_a = uniform_a
        best_score = uniform_score
        print(f"[init] uniform fast score = {uniform_score:.6f}")

    rng = np.random.default_rng(seed)
    iteration = 0

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        time_left = deadline - time.time()
        if time_left < 4.0:      # not enough time for a full restart
            break

        # ----------- pick a starting point for the optimisation -------------
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'base'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p_ex = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1) ** (-p_ex))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'edge':
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # 'base'
            y0 = y_base.copy()

        # ------------------------------------------------------------------
        # 1) Multi‑phase L‑BFGS on the smooth‑max objective
        # ------------------------------------------------------------------
        lam_factors = [0.5, 0.2, 0.05]          # decreasing smoothing
        maxiters_phase = [80, 80, 80]          # modest iteration budget
        y = y0.copy()
        for lam, it in zip(lam_factors, maxiters_phase):
            # early‑exit if time is running out
            if time.time() >= deadline:
                break
            res = minimize(fun=lambda yv: objective_and_grad(
                               yv, n, fft_len_conv, fft_len_g, lam),
                           x0=y,
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': it,
                                    'gtol': 1e-9,
                                    'disp': False})
            y = res.x

        a_lbfgs = softmax(y)
        score_lbfgs = fast_score(a_lbfgs, fft_len_conv, n)

        # ------------------------------------------------------------------
        # 2) Adam on the L^p‑norm objective (p chosen randomly)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            p_choice = int(rng.choice([8, 12, 16, 20]))
            adam_deadline = min(deadline, time.time() + 1.0)   # ~1 s for Adam
            a_adam, score_adam = run_adam(y, n,
                                          fft_len_conv, fft_len_g,

        else:

        # pick the better of the two

            a_candidate = a_adam
            cand_score = score_adam
        else:
            a_candidate = a_lbfgs
            cand_score = score_lbfgs

        # ------------------------------------------------------------------
        # 3) Targeted hill‑climb refinement
        # ------------------------------------------------------------------
        if time.time() < deadline:

            a_candidate, cand_score = hill_climb_targeted(
                a_candidate, cand_score,

                hc_deadline,

        # Enforce left‑right symmetry (does not increase the max‑conv)

        iteration += 1

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
