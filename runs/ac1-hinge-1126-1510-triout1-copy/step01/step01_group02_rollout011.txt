"""
Linear‑programming based optimisation of a non‑negative step function.

The algorithm iteratively enlarges the ℓ¹‑norm of the sequence while keeping the
peak of the self‑convolution (the quantity M = ||a*a||_∞) almost unchanged.

1.  Initialise with the Catalan (half‑binomial) coefficients, which have
    M = 1 and a relatively small sum.
2.  For the current sequence f compute M = max(f ∗ f).
3.  Solve a linear programme
        maximise   Σ b_j
        subject to A_f @ b ≤ M   and   b ≥ 0
   where A_f is the (2n‑1)×n convolution matrix of f.  The solution b is a
   direction that increases the total sum without increasing any entry of
   the convolution beyond the current bound M.
4.  Mix a small fraction t of b into f:
        f ← (1‑t)·f + t·b     (t≈0.05)
   The new peak satisfies
        M_new ≤ (1‑t²)·M + t²·max(b ∗ b),
   so with a modest t the peak grows only slightly while the sum grows
   considerably, thereby reducing the evaluation metric
        2·n·M / (Σ a_i)² .
5.  Keep the best sequence seen, repeat until the time budget expires or the
   target score (1.503) is reached.

The LP is solved with SciPy’s HiGHS linear‑programming routine, which works
well with the sparse convolution matrix.  The routine returns a plain Python
list ready for the external `evaluate_sequence` call.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def _max_convolution(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using a fast FFT‑based convolution."""
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())   # next power of two
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the (2n‑1)×n sparse matrix A such that A @ b = conv(f, b).
    Row k of A contains the coefficients f_i for all i with 0≤i≤n‑1 and
    0≤k‑i≤n‑1.
    """
    n = f.size
    # column index (the variable index) repeats 0..n‑1 for each row offset i
    col = np.tile(np.arange(n, dtype=np.int32), n)
    # row index is i + j
    row = np.repeat(np.arange(n, dtype=np.int32), n) + col
    data = np.repeat(f, n)
    return sp.csr_matrix((data, (row, col)), shape=(2 * n - 1, n))


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """Return a step‑function (list of non‑negative floats) with a low score."""
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1.  Starting sequence – Catalan (half‑binomial) coefficients.
    # ------------------------------------------------------------------
    n = 1500                                   # a size that balances speed & quality
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    f = a.copy()
    cur_score = evaluate_sequence(f.tolist())
    best_seq = f.copy()
    best_score = cur_score
    print(f"Initial Catalan sequence (n={n}) – score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main optimisation loop.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current peak of the self‑convolution
        M_f = _max_convolution(f)

        # build the convolution matrix of the current sequence
        A = _build_conv_matrix(f)

        # linear programme – maximise Σ b_j  subject to A·b ≤ M_f , b ≥ 0
        c = -np.ones(n, dtype=np.float64)                     # minimise –Σ b_j
        ub = np.full(2 * n - 1, M_f, dtype=np.float64)

        lp_res = opt.linprog(
            c,
            A_ub=A,
            b_ub=ub,
            bounds=(0, None),
            method="highs",
            options={"presolve": True},
        )

        if not lp_res.success:
            print(f"LP failed at iteration {iteration}: {lp_res.message}")
            break

        g = lp_res.x
        sum_f = f.sum()
        sum_g = g.sum()

        # if the LP cannot increase the ℓ¹‑norm we are done
        if sum_g <= sum_f * (1.0 + 1e-8):
            print(f"No ℓ¹‑norm increase at iteration {iteration}; terminating.")
            break

        # peak of g ∗ g – needed for a safe mixing factor
        M_g = _max_convolution(g)

        # ------------------------------------------------------------------
        # 3.  Mix a small fraction of g into f.
        # ------------------------------------------------------------------
        # start with a modest step size; shrink if the move worsens the score
        t = 0.05
        mixed = False
        for attempt in range(5):
            f_candidate = (1.0 - t) * f + t * g
            # enforce the hard limits required by evaluate_sequence
            f_candidate = np.clip(f_candidate, 0.0, 1000.0)

            cand_score = evaluate_sequence(f_candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = f_candidate.copy()
                print(f"Iter {iteration:04d} – new best score: {best_score:.6f}")

            if cand_score < cur_score:
                f = f_candidate
                cur_score = cand_score
                mixed = True
                break
            else:
                t *= 0.5    # try a smaller step

        if not mixed:
            # nothing helpful – stop the loop early
            print(f"Iteration {iteration} – no improving mix found.")
            break

        # early termination if target reached
        if best_score < 1.503:
            print(f"Target achieved at iteration {iteration}.")
            break

        # free the large matrix before the next iteration
        del A

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    return best_seq.tolist()
