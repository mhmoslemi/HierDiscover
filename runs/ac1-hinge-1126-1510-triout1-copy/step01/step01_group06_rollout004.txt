def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Hybrid search for a low‑autocorrelation step sequence.

The algorithm consists of three phases:

1. **Random parametric sweep** – generate many candidate shapes of the
   form  a_i = (i+δ+1)^γ – (i+δ)^γ  (γ∈(0,1], δ≥0), normalize and keep the
   best one (fast coarse optimisation).

2. **Deterministic‑annealing gradient descent** – minimise a smooth
   approximation of the infinity‑norm of the autocorrelation
   (soft‑max of the convolution).  An Adam optimiser together with a
   projection onto the probability simplex (non‑negative, sum = 1) is used.
   The “temperature’’ β of the soft‑max is increased slowly so that the
   method first moves in a smooth landscape and later concentrates on the
   true peak.

3. **Local polishing** – tiny mass‑transfer moves (i→j) are tried
   repeatedly to squeeze out any remaining improvement.

All numbers are kept non‑negative and the final sequence is scaled to
the 0‑1000 range required by ``evaluate_sequence`` (the scaling does not
affect the score because the metric is homogeneous).
"""

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    # Implementation of the algorithm from
    # Duchi, Shalev‑Shwartz, Singer, Chandra (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the rho index
    cond = u > cssv / ind
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – re‑normalise to exact sum 1
    w_sum = w.sum()
    if w_sum > 0:
        w = w / w_sum
    return w


def _score_of_sequence(seq: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a non‑negative sequence with sum≈1."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_conv = np.max(conv)
    # sum(seq) is (almost) 1 – we keep it in the formula for safety
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return normalized soft‑max weights for a given convolution vector."""
    # Shift by the maximum to avoid overflow/underflow.
    c_max = np.max(conv)
    shifted = conv - c_max
    w = np.exp(beta * shifted)
    w_sum = w.sum()
    if w_sum == 0:
        # Extremely unlikely but keep numerics stable
        w = np.ones_like(w) / w.size
    else:
        w /= w_sum
    return w

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Length of the step function (use the global if it exists)
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # noqa: F821  (global may be provided)
    except Exception:
        n = 1542                     # default used previously

    # ------------------------------------------------------------------
    # Phase 1 – random parametric sweep (cheap coarse search)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    sweep_iters = 2000
    for _ in range(sweep_iters):
        # Sample parameters:
        gamma = np.random.uniform(0.30, 0.80)          # exponent γ ∈ (0,1)
        delta = np.random.uniform(0.0, 2000.0)         # shift δ ≥ 0
        i = np.arange(n, dtype=np.float64)
        # a_i = (i+δ+1)^γ – (i+δ)^γ  (discrete derivative of (x+δ)^γ)
        a = np.power(i + delta + 1.0, gamma) - np.power(i + delta, gamma)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            continue
        a /= a.sum()                                 # project onto simplex
        sc = _score_of_sequence(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
    print(f"Phase 1 – sweep best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic‑annealing gradient descent (Adam)
    # ------------------------------------------------------------------
    # If for any reason the sweep failed, fall back to the classic sqrt‑diff start.
    if best_seq is None:
        a = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(
            np.arange(0, n, dtype=np.float64)
        )
        a = np.clip(a, 0.0, None)
        a /= a.sum()
        best_seq = a.copy()
        best_score = _score_of_sequence(best_seq)

    a = best_seq.copy()
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.15                 # base learning rate
    # Soft‑max temperature schedule
    beta = 5.0                # start (very smooth)
    beta_max = 150.0
    beta_inc = 5.0            # additive increase every few hundred iters
    iter_cnt = 0
    last_improve = 0

    while time.time() < deadline:
        # ---------- convolution and soft‑max ----------
        conv = np.convolve(a, a)
        w = _softmax_weights(conv, beta)

        # ---------- gradient of the smoothed max ----------
        # grad_j = 2 * sum_i w_{i+j} * a_i   (valid mode ↔ correlation)
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # ---------- Adam update ----------
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        # Project back onto the simplex (keeps non‑negativity & sum=1)
        a_new = _simplex_projection(a_new)

        # Occasionally inject a tiny random jitter to avoid stagnation
        if np.random.rand() < 0.008:
            i, j = np.random.choice(n, 2, replace=False)
            eps_move = min(a_new[i], a_new[i] * 0.02)
            a_new[i] -= eps_move
            a_new[j] += eps_move
            a_new = _simplex_projection(a_new)

        a = a_new

        # ---------- temperature (β) schedule ----------
        if iter_cnt % 500 == 0 and beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

        # ---------- learning‑rate decay ----------
        if iter_cnt % 1000 == 0:
            lr = max(lr * 0.94, 0.005)

        # ---------- evaluate ----------
        score = _score_of_sequence(a)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            last_improve = iter_cnt
            print(
                f"Iter {iter_cnt:6d} – new best {best_score:.6f} (β={beta:.1f}, lr={lr:.4f})"
            )
        # If no improvement for a long while, do a small random reset
        if iter_cnt - last_improve > 5000:
            noise = np.random.rand(n) * 0.01
            a = a + noise
            a = _simplex_projection(a)
            last_improve = iter_cnt

    # ------------------------------------------------------------------
    # Phase 3 – short local polishing (tiny mass transfers)
    # ------------------------------------------------------------------
    polish_deadline = time.time() + min(5.0, deadline - time.time())
    a = best_seq.copy()
    while time.time() < polish_deadline:
        i, j = np.random.choice(n, 2, replace=False)
        eps_move = min(a[i], a[i] * 0.015)
        if eps_move <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= eps_move
        a_new[j] += eps_move
        a_new = _simplex_projection(a_new)
        sc = _score_of_sequence(a_new)
        if sc < best_score:
            best_score = sc
            best_seq = a_new.copy()
            a = a_new
            print(f"Polish improved: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the 0‑1000 range required by the runner
    # ------------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    print(f"Returning sequence – final score {best_score:.6f}")
    return final_seq.tolist()
