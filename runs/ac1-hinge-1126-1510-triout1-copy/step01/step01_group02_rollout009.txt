def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    n = 3000   # try 3000 maybe better than 2500
    # compute Catalan half binomial
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)

    # compute conv via FFT
    L = 1 << ((2*n - 1).bit_length())
    f = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(f * f, n=L)[:2*n-1]

    sum_a = a.sum()
    sum_a2 = sum_a * sum_a
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / sum_a2
    best_score = cur_score
    best_a = a.copy()
    # quick print
    print(f"Init n={n}, sum={sum_a:.6f}, score={cur_score:.6f}")

    # parameters
    temp0 = cur_score * 0.5
    it = 0

    tail_start = n // 2
    tail_idxs = np.arange(tail_start, n)
    epsilon = 1e-12

    while time.time() < deadline:
        it += 1
        # compute max index
        K = int(conv.argmax())
        # compute source weights contributions
        lo = max(0, K - (n - 1))
        hi = min(n - 1, K)
        # indices where partner in range
        i_vals = np.arange(lo, hi + 1)
        j_vals = K - i_vals
        contrib = a[i_vals] * a[j_vals]    # vectorized
        total_contrib = contrib.sum()
        if total_contrib <= 0:
            # fallback: pick random src among first quarter
            src = np.random.randint(0, n // 4)
        else:
            # probabilities proportional to contrib
            probs = contrib / total_contrib
            src = np.random.choice(i_vals, p=probs)

        if a[src] <= 1e-12:
            continue

        # select target: choose among tail sorted by low a
        tail_a = a[tail_start:]
        # probability inverse to a
        inv = 1.0 / (tail_a + epsilon)
        inv /= inv.sum()
        tgt = np.random.choice(tail_idxs, p=inv)

        if tgt == src:
            continue

        # delta: random fraction
        frac = np.random.rand() * 0.4 + 0.1   # 0.1 to 0.5
        delta = min(frac * a[src], 1000.0 - a[tgt])
        if delta <= 0:
            continue

        # Save old max for potential acceptance later (but we compute new after update)
        # Apply incremental update
        # Note: we need to use old a values for conv updates
        # We'll store a_src_old = a[src] (old)
        # update conv slices using old a (including a[tgt]) before modifying a
        # Using vectorized slice updates
        # Use conv_view to avoid copy later
        # NB: For overlapping slices, using separate updates works

        conv[src:src+n] -= 2.0 * delta * a   # uses old a

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The algorithm works as a fast stochastic local search that repeatedly moves a
fraction of mass from indices that contribute heavily to the current peak of
the self‑convolution to indices that have a weak influence on that peak.
Only the convolution array is updated incrementally, so each iteration costs
O(n) instead of recomputing the whole convolution.

Features
--------
* Starts from the global variable ``height_sequence_1`` if it exists,
  otherwise from the Catalan (half‑binomial) coefficients.
* In each iteration:
    – Finds the index `K` where the convolution attains its current maximum.
    – Computes the contribution `w_i = a_i * a_{K-i}` of every index to that
      maximum.
    – Picks a source index `src` with probability proportional to `w_i`.
    – Picks a target index `tgt` among the right‑hand half, biased toward the
      smallest values (to spread the mass).
    – Moves a random fraction (10‑50 %) of the source value to the target.
    – Accepts the move if it improves the score, otherwise with a Metropolis
      probability that follows a linearly decreasing temperature schedule.
* Keeps the best sequence ever seen and stops early if the target bound
  `1.503` is reached.

The whole search runs for ``budget_s`` seconds (default 1000 s).  The final
sequence is returned as a plain Python list of floats, ready for
``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def catalan_sequence(n: int) -> np.ndarray:
    """Return the Catalan (half‑binomial) coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def self_convolution(seq: np.ndarray) -> np.ndarray:
    """Return the (linear) self‑convolution of ``seq`` using FFT."""
    n = seq.size
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    ft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(ft * ft, n=L)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence with a low score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the sequence ----------------------------------------
    try:
        # Use the globally provided result if it exists and looks reasonable.
        init = np.array(height_sequence_1, dtype=np.float64)
        if init.ndim != 1 or init.size < 10:
            raise ValueError
        # Clip to the allowed range.
        a = np.clip(init, 0.0, 1000.0)
        n = a.size
        print(f"Initialised from global height_sequence_1 (n={n})")
    except Exception:
        # Default: Catalan coefficients, length 3000 (a good trade‑off).
        n = 3000
        a = catalan_sequence(n)
        print(f"Initialised with Catalan coefficients (n={n})")

    # compute the initial self‑convolution and the constant sum
    conv = self_convolution(a)
    sum_a = a.sum()
    sum_a2 = sum_a * sum_a
    cur_max = conv.max()
    cur_score = 2.0 * n * cur_max / sum_a2

    best_seq = a.copy()
    best_score = cur_score
    best_max = cur_max

    print(f"Initial score: {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Parameters for the stochastic optimisation
    # ------------------------------------------------------------------
    init_temp = cur_score * 0.5          # initial temperature
    tail_start = n // 2                  # we will mainly target the second half
    tail_len = n - tail_start
    eps = 1e-12

    iteration = 0
    report_interval = 2000               # print progress every N iterations

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ---- identify the current peak of the convolution ----------------
        K = int(conv.argmax())                     # index of the maximal value

        # ---- compute contributions w_i = a_i * a_{K-i} -----------------
        lo = max(0, K - (n - 1))
        hi = min(n - 1, K)
        i_vals = np.arange(lo, hi + 1, dtype=np.int64)
        j_vals = K - i_vals
        contrib = a[i_vals] * a[j_vals]           # contribution of each i
        total_contrib = contrib.sum()
        if total_contrib <= 0.0:
            # Degenerate case – fall back to a random source in the first half
            src = np.random.randint(0, n // 2)
        else:
            # sample src proportionally to its contribution
            cum = np.cumsum(contrib)
            r = np.random.rand() * cum[-1]
            src = i_vals[np.searchsorted(cum, r)]

        if a[src] <= eps:
            continue    # nothing to move from this index

        # ---- choose a target index in the right half --------------------
        # bias towards the smallest current values to spread the mass
        tail_vals = a[tail_start:]
        inv_weights = 1.0 / (tail_vals + eps)
        cum_tgt = np.cumsum(inv_weights)
        r2 = np.random.rand() * cum_tgt[-1]
        tgt_offset = np.searchsorted(cum_tgt, r2)
        tgt = tail_start + int(tgt_offset)

        if tgt == src:
            continue

        # ---- propose a mass transfer ------------------------------------
        frac = np.random.rand() * 0.4 + 0.1                 # 0.1 … 0.5
        delta = min(frac * a[src], 1000.0 - a[tgt])
        if delta <= eps:
            continue

        # ---- apply the move on a *copy* of the convolution -------------
        conv_trial = conv.copy()
        # update the two convolution slices (uses the *old* a values)
        conv_trial[src:src + n] -= 2.0 * delta * a
        conv_trial[tgt:tgt + n] += 2.0 * delta * a
        # delta‑squared point contributions
        conv_trial[src + src] += delta * delta
        conv_trial[tgt + tgt] += delta * delta
        conv_trial[src + tgt] -= 2.0 * delta * delta

        new_max = conv_trial.max()
        new_score = 2.0 * n * new_max / sum_a2

        # ---- temperature schedule ----------------------------------------
        elapsed = time.time() - start_time
        temp = init_temp * max(0.0, 1.0 - elapsed / (budget_s - 5.0))

        # ---- acceptance test ---------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        elif temp > 1e-12:
            prob = math.exp(-(new_score - cur_score) / temp)
            if np.random.rand() < prob:
                accept = True

        if accept:
            # Commit the move
            a[src] -= delta
            a[tgt] += delta
            conv = conv_trial
            cur_score = new_score
            cur_max = new_max

            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                best_max = new_max
                print(f"Iter {iteration:07d} – new best score: {best_score:.6f}")

        # ---- occasional random shake (to escape plateaus) ---------------
        if iteration % 5000 == 0:
            shake = a.copy()
            # pick a tiny set of indices and perturb them a bit
            idxs = np.random.randint(0, n, size=6)
            shake[idxs] = np.maximum(0.0, shake[idxs] + np.random.randn(6) * 0.02)
            shake_conv = self_convolution(shake)
            shake_score = 2.0 * n * shake_conv.max() / sum_a2
            if shake_score < best_score:
                best_score = shake_score
                best_seq = shake.copy()
                conv = shake_conv
                a = shake
                cur_score = shake_score
                cur_max = shake_conv.max()
                print(f"Iter {iteration:07d} – shake improved score to {best_score:.6f}")

        # ---- report progress ------------------------------------------------
        if iteration % report_interval == 0:
            print(f"Iter {iteration:07d} – cur score {cur_score:.6f}, best {best_score:.6f}")

        # ---- early termination if target achieved --------------------------
        if best_score < 1.503:
            print(f"Target reached at iteration {iteration} – score {best_score:.6f}")
            break

    print(f"Finished after {iteration} iterations. Best score: {best_score:.6f}")
    return best_seq.tolist()
