def fast_score(seq):
    n = len(seq)
    sz = 1
    while sz < 2*n-1:
        sz <<= 1
    # FFT
    seq_f = np.fft.rfft(seq, sz)
    conv = np.fft.irfft(seq_f * seq_f, sz)[:2*n-1]  # real part
    max_b = conv.max()

def _full_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Return the full discrete convolution of ``seq`` with itself using FFT.
    Length of the result is ``2*len(seq)-1``.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    # real FFT is sufficient because the input is real
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f * f, size)[: 2 * n - 1]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast version of ``evaluate_sequence`` – it assumes the sequence already
    satisfies all security checks (non‑negative, not too large, sum>0.01).
    Returns the same quantity: 2 * n * max(conv) / (sum**2).
    """
    n = seq.shape[0]
    conv = _full_convolution(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


def _generate_power_law_shift(n: int, p: float, delta: float) -> np.ndarray:
    """
    Power‑law with an optional offset.
    a_i ∝ (i + 1 + delta)^{-p},  i = 0 … n‑1, then normalised to sum 1.
    """
    idx = np.arange(n, dtype=np.float64) + delta + 1.0
    seq = idx ** (-p)
    total = seq.sum()
    if total <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return seq / total


def _generate_exponential(n: int, lam: float) -> np.ndarray:
    """
    Pure exponential decay a_i ∝ exp(-lam * i), normalised to sum 1.
    """
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    total = seq.sum()
    if total <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return seq / total

    size = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two
    f = np.fft.rfft(seq, size)
    conv = np.fft.irfft(f * f, size)[: 2 * n - 1]
    return conv


def _fast_score(seq: np.ndarray) -> float:
    """
    Fast analogue of ``evaluate_sequence`` (the input already satisfies the
    security checks).  Returns 2·n·max(conv)/sum**2.
    """
    n = seq.shape[0]
    conv = _full_convolution(seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a ** 2))


def _generate_power_law_shift(n: int, p: float, delta: float) -> np.ndarray:
    """
    a_i ∝ (i + 1 + delta)^{-p}, i = 0 … n‑1, normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64) + delta + 1.0
    seq = idx ** (-p)
    total = seq.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return seq / total


def _generate_exponential(n: int, lam: float) -> np.ndarray:
    """
    a_i ∝ exp(-lam·i), normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    total = seq.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return seq / total


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises the
    evaluation function, respecting the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ----------------------------------------------------------------------
    # 1.  Determine the length of the candidate vector.
    # ----------------------------------------------------------------------
    try:
        n = len(height_sequence_1)   # type: ignore[name-defined]  # noqa: F821
        if n <= 0:
            n = 3000
    except Exception:
        n = 3000

    best_seq = None
    best_score = float("inf")

    # ----------------------------------------------------------------------
    # 2.  Coarse global search over simple parametric families.
    # ----------------------------------------------------------------------
    # Shifted power‑law
    p_vals = np.linspace(0.40, 0.74, 18)          # 0.40 … 0.74
    delta_vals = np.linspace(0.0, 20.0, 11)       # offset up to 20
    for p in p_vals:
        for delta in delta_vals:
            if time.time() >= deadline:
                break
            seq = _generate_power_law_shift(n, p, delta)
            score = _fast_score(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[+] PL  p={p: .4f} δ={delta: .1f}  score={score: .6f}")

    # Pure exponential decay
    lam_vals = np.linspace(0.0005, 0.0100, 15)
    for lam in lam_vals:
        if time.time() >= deadline:
            break
        seq = _generate_exponential(n, lam)
        score = _fast_score(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[+] EXP λ={lam: .6f}  score={score: .6f}")

    # ----------------------------------------------------------------------
    # 3.  Refinement with exponentiated‑gradient descent on the simplex.
    # ----------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        step = 0
        while time.time() < deadline and step < 10000:
            conv = _full_convolution(a)
            max_idx = int(np.argmax(conv))

            # gradient of the active quadratic form (2·a_{k‑max‑i})
            i = np.arange(n)
            j = max_idx - i
            mask = (j >= 0) & (j < n)
            g = np.zeros_like(a)
            g[mask] = 2.0 * a[j[mask]]

            # diminishing step size
            eta = 0.6 / (1.0 + step * 0.001)

            # exponentiated‑gradient update (keeps positivity)
            a = a * np.exp(-eta * g)
            total = a.sum()
            if total == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
                total = 1.0
            a /= total

            # periodic evaluation
            if step % 20 == 0:
                score = _fast_score(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[+] GD step {step:5d} idx={max_idx:5d}  score={score: .6f}")

            step += 1

    # ----------------------------------------------------------------------
    # 4.  Light simulated‑annealing style random perturbations.
    # ----------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        cur_score = _fast_score(a)

        total_sa_iters = 3000
        for it in range(total_sa_iters):
            if time.time() >= deadline:
                break

            # simple linear cooling
            temp = 0.1 * (1.0 - it / total_sa_iters) + 0.001

            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a tiny amount of mass from i to j
            delta = np.random.uniform(0.0, min(a[i], 0.005))
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_score = _fast_score(a_new)

            # Metropolis acceptance
            if new_score < cur_score or np.exp((cur_score - new_score) / max(temp, 1e-12)) > np.random.rand():
                a = a_new
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = a.copy()
                    print(f"[+] SA iter {it:5d}  score={new_score: .6f}")

    # ----------------------------------------------------------------------
    # 5.  Safety net – fall back to uniform if nothing succeeded.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Optional final check with the official evaluator (harmless if missing)
    try:
        official = evaluate_sequence(best_seq.tolist())
        print(f"[+] Official evaluation: {official:.6f}")
    except Exception:
        pass

    # Return a plain Python list as required by the harness.
    return best_seq.tolist()
