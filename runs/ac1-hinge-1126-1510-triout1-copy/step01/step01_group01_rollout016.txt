from scipy.optimize import linprog

def lp_step(f, M):
    n = len(f)
    m = 2*n - 1
    # Build A_ub (m x n)
    # For each k row: A[k, j] = f[k-j] if 0 <= k-j < n else 0

    # Build as sparse matrix for efficiency
    rows = []
    cols = []
    data = []
    for k in range(m):
        j_start = max(0, k-(n-1))
        j_end = min(n-1, k)  # inclusive
        # iterate j in range
        for j in range(j_start, j_end+1):
            i = k - j
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
    # upper bounds ub vector
    ub = np.full(m, M)
    # objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    # bounds for variables: (0, None)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=ub, bounds=bounds, method='highs')
    b = res.x
    return b
```

def project_simplex(v):
    # Ensure sum(v) = 1, v>=0.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    vector is projected onto the probability simplex (non‑negative, sum = 1).
    A small simulated‑annealing acceptance rule permits occasional uphill
    moves, helping to escape plateaux.  Hyper‑parameters (learning rate,

            # occasional mild perturbation to shake the state
            if np.random.rand() < 0.20:
                noise = np.random.normal(scale=step_frac, size=n_target)
                cur_seq = np.maximum(best_seq + noise, 0.0)
                cur_seq /= cur_seq.sum()
                cur_score = 2.0 * n_target * _max_self_convolution(cur_seq)

    print(f"[final] length={n_target}, best_score={best_score:.6f}")
    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
