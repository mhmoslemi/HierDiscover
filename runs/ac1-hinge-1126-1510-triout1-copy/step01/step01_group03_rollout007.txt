Algorithm to project onto simplex:
```
def simplex_projection(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1]/rho
    w = np.maximum(v - theta, 0)
    return w
```

That's O(n log n). Acceptable for n up to 3000.

def convolve_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv
```

def simplex_projection(v, s):
    # ensure sum(v) = s, v_i >= 0
    # reference: https://github.com/ConstrainedOptimization/ProjectionOntoSimplex
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        # all zeros?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def half_order_series(n):

The evaluation function is scale‑invariant, so we fix the ℓ₁‑norm

* we build a smooth “max” using a soft‑max over the convolution values,

* we update the vector with Adam and project it back onto the simplex
  {x ≥ 0, Σx = √(2·n)}.

A linear schedule raises the soft‑max temperature (β) from a low value
to a high value, turning the smooth max into the true max.  Random
multiplicative noise is injected every few hundred iterations to escape

# ----------------------------------------------------------------------


def half_order_series(n: int) -> np.ndarray:
    """
    Generate the first n coefficients a_i = C(2i,i) / 4**i using the
    stable recurrence a_i = a_{i‑1} * ((2i‑1)*(2i)) / (4*i*i).
    The series is *not* normalised.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    return a


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the full discrete convolution of real vectors x and y using
    an FFT of length that is a power of two (fast for n up to a few
    thousand).  Output length is len(x)+len(y)-1.
    """
    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()         # next power of two
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    conv = np.fft.irfft(Z, size)[:n]
    return conv


def simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project v onto the simplex {x ≥ 0, Σ x = s}.  Implements the
    O(n log n) algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("target sum s must be non‑negative")
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero after projection
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """
    Numerically stable soft‑max with temperature beta.
    """
    x_shift = x - np.max(x)               # shift for stability
    exp_beta = np.exp(beta * x_shift)
    return exp_beta / np.sum(exp_beta)

            a_new = a_new * (1.0 + noise)
            a_new = np.maximum(a_new, 0.0)
            a_new = simplex_projection(a_new, target_sum)

        a = a_new

        # ------------------------------------------------------------------
        # 4) periodic evaluation of the true objective
        # ------------------------------------------------------------------
        if iteration % 200 == 0 or (time.time() > deadline - 5):
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[{iteration:6d}] new best: {best_score:.6f}, "
                      f"max_conv={max_conv:.6f}")
                # early exit if we have already reached the target bound
                if best_score < 1.503:
                    print("[info] target bound achieved – terminating early.")
                    break

    # final safety evaluation
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[final] best score={final_score:.6f} after {iteration} iterations")
    return best_seq.tolist()
