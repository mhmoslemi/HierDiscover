def project_simplex(v):
    # returns projection of v onto simplex {x>=0, sum(x)=1}
    v = np.array(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a low‑evaluation step sequence.

The algorithm works on a probability simplex (sum = 1) using a
soft‑max parametrisation a = softmax(y).  It repeatedly
* draws diverse starting points,
* minimises a smooth approximation of the maximal auto‑convolution via
  L‑BFGS (the smoothing parameter λ is decreased gradually),
* refines the result with a projected sub‑gradient step on the *exact*
  max‑convolution,
* finishes with a fast greedy hill‑climb (mass transfers between two
  coordinates).

All operations are performed with FFT‑based convolutions (O(n log n)).
The routine respects the 1000 s time budget and returns the best
sequence found as a plain list of non‑negative floats.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) via FFT.
    a must be a 1‑D array of length n, fft_len ≥ 2*n‑1 and a power of two.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a sequence that already sums to 1."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective for L‑BFGS.

    Returns:
        f  – smooth objective value (2·n·smooth_max)
        g  – gradient w.r.t. y (shape n)
    """
    # simplex mapping
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # convolution (a reversed) * w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Duchi et al. (2008) algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.zeros_like(v)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_step(a: np.ndarray, step: float,
                     fft_len: int, n: int) -> np.ndarray:
    """
    One projected sub‑gradient step on the exact objective
    max(conv(a,a)). The gradient is derived from the active
    convolution index/indices.
    """
    conv = conv_fft(a, fft_len, n)
    max_c = conv.max()
    # identify active positions (allow a small tolerance)
    active = np.isclose(conv, max_c, atol=1e-12)
    # if far too many active points, keep only the top 10
    if active.sum() > 10:
        idx = np.argpartition(conv, -10)[-10:]
        active[:] = False
        active[idx] = True
    # weight vector w (averaged over active indices)
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.sum()

    # gradient = 2 * (a_rev * w) restricted to n entries
    fa_rev = np.fft.fft(a[::-1], fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]
    grad = 2.0 * g_vec

    # gradient step + projection onto simplex
    a_new = a - step * grad
    a_proj = project_simplex(a_new)
    return a_proj


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple greedy hill‑climb that transfers a random fraction of mass
    from one coordinate to another while preserving the sum.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

            # add Gaussian noise to current best and renormalise
            a_noisy = best_a + rng.normal(scale=0.02, size=n)
            a_noisy = np.clip(a_noisy, 0, None)
            if a_noisy.sum() < 1e-12:
                a_noisy = np.full(n, 1.0 / n)
            else:
                a_noisy = a_noisy / a_noisy.sum()
            y0 = np.log(a_noisy + 1e-12)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # edge‑heavy heuristic
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)

        # ----- L‑BFGS optimisation with decreasing smoothing -----
        lam_seq = [0.6, 0.4, 0.2, 0.1]   # gradually sharpen the objective
        y_cur = y0.copy()
        for lam in lam_seq:
            if time.time() >= deadline:
                break
            # note: we reuse the same FFT length for both convolutions
            res = minimize(fun=objective_and_grad,
                           x0=y_cur,
                           args=(n,
                                 fft_len,      # for a*a
                                 fft_len,      # for gradient convolution
                                 lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 350,
                                    'gtol': 1e-10,
                                    'disp': False})
            y_cur = res.x

        a_opt = softmax(y_cur)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode}  fast score = {best_score:.6f}")

        # ----- sub‑gradient refinement on the exact max‑convolution -----
        # Perform a short schedule of decreasing step sizes.
        step = 0.6
        a_sg = best_a.copy()
        for _ in range(150):
            if time.time() >= deadline:
                break
            a_sg = subgradient_step(a_sg, step, fft_len, n)
            step *= 0.985
        sg_score = fast_score(a_sg, fft_len, n)
        if sg_score < best_score:
            best_score = sg_score
            best_a = a_sg.copy()
            print(f"[{iteration}] sub‑grad improved fast score = {best_score:.6f}")

        # ----- greedy hill‑climb for final polishing -----
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, n,
                                        hc_deadline,
                                        max_iter=3000)

        iteration += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best candidate (list of floats)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # final evaluation using the official (security‑checked) function
    try:
        final_val = evaluate_sequence(final_seq)
        print(f"[final] official evaluation = {final_val:.6f}")
    except Exception as e:
        print("[final] evaluation error:", e)

    return final_seq
