def smooth_gradient(a, alpha):
    n = a.shape[0]
    conv = convolve_self(a)
    max_c = conv.max()
    # compute exp_shift = exp(alpha * (conv - max_c))
    exp_shift = np.exp(alpha * (conv - max_c))
    w = exp_shift / exp_shift.sum()
    # compute convolution a * w_rev via FFT
    w_rev = w[::-1]
    # compute next pow2 size
    size = 1 << ((n + (2*n-1) - 1).bit_length())
    A = np.fft.rfft(a, size)
    W = np.fft.rfft(w_rev, size)
    grad_full = np.fft.irfft(A * W, size)[:n + (2*n-1) - 1]  # length 3n-2
    # extract grad indices:
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * grad_full[idx]
    return grad
```

import time
import numpy as np
import scipy.optimize as opt

def _next_pow2(x):
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _convolve_self(a):
    """Return convolution a*a using FFT (full length 2n-1)."""
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    C = A * A
    conv_full = np.fft.irfft(C, size)
    return conv_full[: 2 * n - 1]

def _objective(a):
    """Compute the objective 2*n*max(conv) / (sum(a)^2)."""
    n = a.shape[0]
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    conv = _convolve_self(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)

def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """Generate a non‑negative power‑law sequence a_i = (i+1+offset)^(-exponent)."""
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    # Clip to the range required by evaluate_sequence.
    seq = np.clip(seq, 0.0, 1000.0)
    return seq

def _objective_params(params, length):
    """Helper for SciPy: objective for a given (exponent, offset)."""
    exp, off = params
    # enforce simple bounds – outside the domain we give a huge penalty
    if not (0.1 < exp < 0.999) or off < 0.0:
        return 1e6
    a = _initial_sequence(length, exponent=exp, offset=off)
    return _objective(a)

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a step‑function that minimises
        2 * n * max(a*a) / (sum(a)**2)

    Strategy:
      1. Coarse grid over length, exponent and offset (power‑law shapes).
      2. For the best grid point run a cheap local optimisation
         (Nelder‑Mead on exponent and offset) keeping the length fixed.
      3. Return the best sequence found.
    The whole search respects the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1) coarse grid search
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    best_len = None
    best_exp = None
    best_off = None

    # reasonable lengths – larger gives a slightly lower bound but costs more time
    lengths = [2000, 4000, 8000, 12000, 16000]
    exponents = np.linspace(0.55, 0.90, 8)   # 0.55 … 0.90
    offsets = [0.0, 0.2, 0.5, 1.0, 2.0]

    print("=== coarse grid search ===")
    for n in lengths:
        for exp in exponents:
            for off in offsets:
                if time.time() >= deadline:
                    break
                a = _initial_sequence(n, exponent=float(exp), offset=float(off))
                score = _objective(a)
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    best_len = n
                    best_exp = float(exp)
                    best_off = float(off)
                    print(f"new grid best: n={n} exp={exp:.3f} off={off:.1f} → {score:.6f}")

    # --------------------------------------------------------------
    # 2) local refinement (Nelder‑Mead) for the best length
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform vector
        best_seq = np.ones(1000, dtype=np.float64)
        best_score = _objective(best_seq)
        best_len = 1000
        best_exp = 0.5
        best_off = 0.0
        print("fallback to uniform sequence")

    print(f"grid finished – best length={best_len}, exp={best_exp:.4f}, off={best_off:.4f}, score={best_score:.6f}")

    # refine only if we still have time
    if time.time() < deadline:
        # use the length we already found as the search space
        x0 = np.array([best_exp, best_off])
        # Nelder‑Mead does not need gradients; we cap the number of iterations
        options = {'maxiter': 250, 'disp': False, 'xatol': 1e-6, 'fatol': 1e-6}
        res = opt.minimize(_objective_params, x0, args=(best_len,),
                           method='Nelder-Mead', options=options)
        exp_opt, off_opt = res.x
        a_opt = _initial_sequence(best_len, exponent=exp_opt, offset=off_opt)
        score_opt = _objective(a_opt)
        print(f"local optimisation: exp={exp_opt:.6f}, off={off_opt:.6f}, score={score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()
            best_exp = exp_opt
            best_off = off_opt

    # --------------------------------------------------------------
    # 3) very short random‑jitter polishing (optional)
    # --------------------------------------------------------------
    # We perform a few random perturbations of the exponent/offset pair
    # to escape a possible remaining local minimum.
    for _ in range(10):
        if time.time() >= deadline:

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two not less than *x*."""
    return 1 << (x - 1).bit_length()


def _convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full self‑convolution a*a using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, size)
    C = A * A
    conv_full = np.fft.irfft(C, size)
    return conv_full[: 2 * n - 1]


def _objective(a: np.ndarray) -> float:
    """
    Compute the evaluation function used in the competition:
        2 * n * max(a*a) / (sum(a)**2)
    Returns ``inf`` when the sum is too small.
    """
    n = a.shape[0]
    sum_a = float(np.sum(a))
    if sum_a < 0.01:
        return float('inf')
    conv = _convolve_self(a)
    max_b = float(np.max(conv))
    return 2.0 * n * max_b / (sum_a * sum_a)


def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Power‑law step function a[i] = (i + 1 + offset)^(-exponent),
    clipped to the range required by ``evaluate_sequence``.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _objective_params(params, length):
    """
    Wrapper for SciPy optimisation.
    ``params`` = (exponent, offset).  Returns a large penalty if the
    parameters are outside a sensible domain.
    """
    exp, off = params
    # simple bounds – anything outside is penalised heavily
    if not (0.1 < exp < 0.999) or off < 0.0:
        return 1e6
    a = _initial_sequence(length, exponent=exp, offset=off)
    return _objective(a)

    Find a non‑negative step function that minimises

        2 * n * max(a*a) / (sum(a)**2)

    Approach
    --------
    1. Coarse grid search over length, exponent and offset for the
       family a[i]=(i+1+offset)^(-exponent).  The objective is evaluated
       with a fast FFT‑based convolution.
    2. For the best grid point run a short Nelder‑Mead optimisation on
       (exponent, offset) while keeping the length fixed.
    3. Perform a few random “jitter” trials around the refined parameters.
    4. Verify the final candidate with the official ``evaluate_sequence``.
    The whole routine respects the supplied time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # --------------------------------------------------------------
    # 1) Coarse grid search
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    best_len = None
    best_exp = None
    best_off = None

    # Lengths are modest enough to keep the search fast but large enough
    # to benefit from the asymptotic improvement of power‑law shapes.
    lengths = [2000, 4000, 8000, 12000, 16000]
    exponents = np.linspace(0.55, 0.90, 8)   # 0.55 … 0.90
    offsets = [0.0, 0.2, 0.5, 1.0, 2.0]

    for n in lengths:
