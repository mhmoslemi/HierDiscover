    max_val = conv.max()
    max_idx = conv.argmax()

def proj_simplex(v):
    # algorithm per Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

"""
Propose a low‑evaluation step sequence.

The search combines three complementary ideas:

1.  **Smooth‑max L‑BFGS** – a softmax mapping (softmax(y))
    guarantees a non‑negative vector that sums to one.
    The objective 2·n·smooth_max(a∗a) is differentiable;
    a short L‑BFGS run (with a decreasing smoothing factor)
    provides a fast descent from many random starts.

2.  **Exponentiated‑gradient refinement** – after the L‑BFGS phase
    the true max‑convolution value is reduced by a simple
    projected‑gradient step on the simplex.
    The sub‑gradient of max(a∗a) at the active index k is
        g_i = 2·a_{k‑i}   (zero outside the support).
    Using a multiplicative update  a←a·exp(−η·g)  followed by
    renormalisation stays on the simplex and drives the peak
    towards the average 1/(2n‑1).

3.  **Greedy hill‑climbing** – a short random mass‑transfer
    hill‑climb finishes the optimisation, cleaning up any
    remaining tiny irregularities.

The whole procedure respects the 1000 s budget, repeatedly
restarts from diverse initialisations, and keeps the best
sequence found.  The final sequence is returned as a plain
list of floats and is evaluated with the supplied
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2·n‑1) via FFT.
    Input ``a`` must already have length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective 2·n·max(a∗a) for a sum‑one vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def projection_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective and its gradient w.r.t. the un‑constrained vector y.
    Returns (f, grad_y) where ``f`` = 2·n·smooth_max(a∗a) and ``a = softmax(y)``.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth‑max of the convolution entries
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth‑max w.r.t. convolution entries
    w = exp_shift / Z                     # shape (2n‑1,)

    # convolution (reverse(a)) * w   →  length n vector
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]  # needed slice

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple random hill‑climb that transfers a small fraction of mass
    between two coordinates while keeping the total mass = 1.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * 0.05          # up to 5 % of mass at i
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def exp_gradient_refine(a: np.ndarray,
                       fft_len: int, n: int,
                       max_iters: int,
                       eta0: float,
                       decay: float,
                       deadline: float) -> np.ndarray:
    """
    Exponentiated‑gradient refinement of a on the simplex.
    Uses the sub‑gradient of max(a∗a) (active convolution index).
    """
    a_cur = a.copy()
    eta = eta0

    for _ in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a_cur, fft_len, n)
        max_val = conv.max()
        k = int(conv.argmax())

        # sub‑gradient g_i = 2·a_{k‑i} (zero outside range)
        g = np.zeros_like(a_cur)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx_i = np.arange(low, high + 1)
        g[idx_i] = 2.0 * a_cur[k - idx_i]

        # multiplicative update and renormalisation
        a_cur = a_cur * np.exp(-eta * g)
        s = a_cur.sum()
        if s == 0.0:
            # numerical safeguard – fall back to uniform
            a_cur = np.full_like(a_cur, 1.0 / n)
        else:
            a_cur /= s

        # adapt step‑size
        eta *= decay
        if eta < 1e-12:
            break

    return a_cur

# Main entry point used by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function tries to finish before this).

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Reserve a small safety margin for the final evaluation call.
    deadline = time.time() + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – a little larger than the previous 2000 for extra
    # flexibility, still cheap enough for many FFTs.
    # --------------------------------------------------------------
    n = 2500

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise the best candidate.
    # If a previous construction is available (global variable), use it.
    # --------------------------------------------------------------
    prev = globals().get('height_sequence_1')
    if prev is not None and isinstance(prev, (list, tuple)):
        best_a = np.array(prev, dtype=float)
        if best_a.size != n:
            # resize by interpolation if size mismatches
            # simple linear interpolation onto the new length
            x_old = np.linspace(0, 1, best_a.size)
            x_new = np.linspace(0, 1, n)
            best_a = np.interp(x_new, x_old, best_a)
        best_a = np.maximum(best_a, 0.0)
        sum_a = best_a.sum()
        if sum_a == 0.0:
            best_a = np.full(n, 1.0 / n)
        else:
            best_a /= sum_a
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start score = {best_score:.6f}")

    iteration = 0
    # Upper‑level random‑restart loop
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) Choose an initial point for the smooth‑max L‑BFGS phase
        # ------------------------------------------------------------------
        mode = rng.choice(['random', 'exp', 'power', 'edge', 'spike'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'power':
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == 'edge':
            # heavy entries near both ends (1‑3 % of the positions)
            m_edge = max(1, int(n * rng.uniform(0.01, 0.03)))
            a_center = 1.0 / np.sqrt(2.0) / n
            a_edge = np.sqrt(n / 2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # spike – a few isolated large entries
            a0 = np.full(n, 1e-12)
            num_spikes = max(1, int(n * 0.005))
            positions = rng.choice(n, size=num_spikes, replace=False)
            weights = rng.random(num_spikes) + 0.5
            a0[positions] = weights
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)

        # ------------------------------------------------------------------
        # 2) Smooth‑max optimisation (multiple decreasing λ phases)
        # ------------------------------------------------------------------
        y_cur = y0.copy()
        # a modest schedule of smoothing parameters
        for lam_factor in (0.5, 0.2, 0.1, 0.05):
            res = minimize(fun=objective_and_grad,
                           x0=y_cur,
                           args=(n, fft_len_conv, fft_len_g, lam_factor),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'gtol': 1e-10,
                                    'ftol': 1e-12,
                                    'disp': False})
            y_cur = res.x
            # abort early if we run out of time
            if time.time() >= deadline:
                break

        a_opt = softmax(y_cur)                     # already normalized
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] mode={mode}  L‑BFGS improved score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3) Exponentiated‑gradient refinement on the simplex
        # ------------------------------------------------------------------
        refine_deadline = min(deadline, time.time() + 0.8)   # keep a margin
        a_refined = exp_gradient_refine(best_a,
                                        fft_len_conv, n,
                                        max_iters=5000,
                                        eta0=0.12,
                                        decay=0.995,
                                        deadline=refine_deadline)
        refined_score = fast_score(a_refined, fft_len_conv, n)
        if refined_score < best_score:
            best_score = refined_score
            best_a = a_refined.copy()
            print(f"[{iteration}]  exp‑grad refined score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4) Short greedy hill‑climb to polish the result
        # ------------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=3000)

        iteration += 1

    # ----------------------------------------------------------------------
    # Final verification with the official evaluator (clamps negatives, etc.)
    # ----------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
