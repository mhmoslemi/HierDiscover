"""
Propose a low‑evaluation step sequence.

This implementation improves on the previous baseline by:
* Using a richer family of initial shapes (power‑law, Beta‑distribution,
  symmetric “arcsine” shape) and a dense grid search over their parameters.
* Running a multi‑stage L‑BFGS optimisation where the smooth‑max
  smoothing parameter λ is gradually reduced (0.30 → 0.10 → 0.04).
* Refining the result with a sub‑gradient descent that works on the
  true ∞‑norm (max convolution) and a fast greedy hill‑climb.
* Adapting the sequence length: if a previous sequence is available
  (global variable ``height_sequence_1``) it is reused; otherwise a
  length of 3000 gives a good trade‑off between quality and speed.
All operations stay in the 0‑1 simplex (soft‑max mapping), because the
objective 2·n·max(b)/(sum(a)²) is invariant under uniform scaling.
The function returns the best sequence found before the time budget
expires.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector summing to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns an array of length 2·n‑1 (the “full” convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv) for a sequence that
    already sums to 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Objective (smooth‑max) used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max objective (2·n·smooth_max(a*a)) and its gradient w.r.t. y.
    The smoothing width is λ = lam_factor·max_conv + ε.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # g_i = Σ_j a_j * w_{i+j}   (i = 0..n‑1)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (non‑smooth) objective
# ----------------------------------------------------------------------
def max_subgrad_descent(a0: np.ndarray, fft_len: int, n: int,
                       steps: int, lr0: float,
                       deadline: float) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on 2·n·max(conv) using the exact max.
    The variables are kept in the simplex via a soft‑max (y‑variables).
    """
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)

    # start from the log‑space representation of a0
    y = np.log(best_a + 1e-12)
    lr = lr0
    for _ in range(steps):
        if time.time() > deadline:
            break

        a = softmax(y)
        conv = conv_fft(a, fft_len, n)
        k = int(conv.argmax())                     # index of the current max

        # gradient of max conv w.r.t. a (sub‑gradient)
        g_a = np.zeros_like(a)
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        g_a[mask] = 2.0 * a[idx[mask]]

        # chain rule through soft‑max
        mean_g = np.dot(a, g_a)
        grad_y = a * (g_a - mean_g)

        # simple back‑tracking line‑search
        y_new = y - lr * grad_y
        a_new = softmax(y_new)
        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            y = y_new
            lr = min(lr * 1.25, 1.0)          # slightly increase step‑size on success
        else:
            lr *= 0.5                         # shrink step‑size on failure
            if lr < 1e-6:
                break

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (random mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple greedy mass‑transfer hill‑climb.
    At each step move a random fraction (≤5 %) of the mass from one
    coordinate to another, keeping the total sum exactly 1.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05          # move up to 5 % of the mass at i
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Shape generators for the initial points
# ----------------------------------------------------------------------
def shape_power(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """a_i ∝ (i+1+offset)^{-p}."""
    idx = np.arange(n, dtype=float) + 1.0 + offset
    a = idx ** (-p)
    return a / a.sum()


def shape_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type shape: a_i ∝ x^{alpha} (1-x)^{beta},  x ∈ (0,1)."""
    x = (np.arange(n, dtype=float) + 0.5) / n   # mid‑point rule
    # avoid zero to the negative power
    eps = 1e-12
    xa = np.power(np.maximum(x, eps), alpha)
    xb = np.power(np.maximum(1.0 - x, eps), beta)
    a = xa * xb
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    return a / a.sum()


def shape_symmetric_arcsine(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine distribution:
       a_i ∝ 1/√i (i starts at 1)."""
    idx = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(idx)
    return a / a.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (the evaluator will
        compute the final score).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # 1. Choose problem size
    # ------------------------------------------------------------------
    # If a previous sequence exists, reuse its length (often already good)
    global height_sequence_1
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, list):
        n = len(height_sequence_1)
        init_a = np.maximum(0.0, np.array(height_sequence_1, dtype=float))
        if init_a.sum() < 1e-12:
            init_a = np.full(n, 1.0 / n)
        else:
            init_a = init_a / init_a.sum()
    else:
        n = 3000                 # default problem size
        init_a = np.full(n, 1.0 / n)

    # pre‑computed FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2. Evaluate the starting point
    # ------------------------------------------------------------------
    best_a = init_a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] length={n}  fast score = {best_score:.6f}")

    rng = np.random.default_rng(seed)
    iteration = 0

    # ------------------------------------------------------------------
    # 3. Create a pool of promising deterministic seeds
    # ------------------------------------------------------------------
    deterministic_candidates = []

    # a) Pure arcsine shape
    deterministic_candidates.append(shape_symmetric_arcsine(n))

    # b) Power‑law family
    for p in np.linspace(0.35, 0.65, 7):
        for off in [0.0, 0.5, 1.0, 2.0]:
            deterministic_candidates.append(shape_power(n, p, offset=off))

    # c) Beta family (symmetric and slightly asymmetric)
    alphas = [-0.7, -0.5, -0.3, -0.1, 0.0, 0.2]
    betas = alphas
    for a_ in alphas:
        for b_ in betas:
            deterministic_candidates.append(shape_beta(n, a_, b_))

    # evaluate each deterministic start once (cheap)
    for cand in deterministic_candidates:
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[init‑grid] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main search loop (random restarts + multi‑stage L‑BFGS)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- pick a random initial shape -------------------------------------------------
        mode = rng.choice(['random', 'power', 'beta', 'arc'])
        if mode == 'random':
            y0 = rng.standard_normal(n)
        elif mode == 'power':
            p = rng.uniform(0.3, 0.7)
            off = rng.uniform(0.0, 2.5)
            a0 = shape_power(n, p, offset=off)
            y0 = np.log(a0 + 1e-12)
        elif mode == 'beta':
            a_ = rng.uniform(-0.8, 0.4)
            b_ = rng.uniform(-0.8, 0.4)
            a0 = shape_beta(n, a_, b_)
            y0 = np.log(a0 + 1e-12)
        else:   # 'arc' – deterministic arcsine start
            a0 = shape_symmetric_arcsine(n)
            y0 = np.log(a0 + 1e-12)

        # ----- multi‑stage L‑BFGS (decreasing smoothing) ----------------------------------
        current_y = y0
        for lam in (0.30, 0.12, 0.04):            # progressively tighter approximation
            if time.time() > deadline:
                break
            res = minimize(fun=objective_and_grad,
                           x0=current_y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': 300,
                                    'gtol': 1e-12,
                                    'disp': False})
            current_y = res.x

        a_opt = softmax(current_y)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt

            print(f"[{iteration}] mode={mode}  L‑BFGS improved fast score = {best_score:.6f}")

        # ----- sub‑gradient descent on the true max ---------------------------------------

        hc_deadline = min(deadline, time.time() + 0.4)

    # ------------------------------------------------------------------
    final_seq = best_a.tolist()

    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq
