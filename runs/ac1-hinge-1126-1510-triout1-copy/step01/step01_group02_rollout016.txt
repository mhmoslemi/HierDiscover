def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, scipy.sparse as sp, scipy.optimize as opt

    # parameters
    n = 1024
    target_sum = np.sqrt(2.0 * n)

    # Catalan coefficients (half-binomial)
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)

    # scale to target sum
    a = a * (target_sum / a.sum())
    cur = a
    # compute score with fast convolution
    def fast_score(seq):
        s = seq.sum()
        if s < 0.01:
            return np.inf
        n = seq.size
        L = 1 << ((2*n - 1).bit_length())
        f = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(f * f, n=L)[:2*n-1]
        return 2.0 * n * conv.max() / (s * s)

    cur_score = fast_score(cur)
    best_seq = cur.copy()
    best_score = cur_score
    print(f"Initial score {cur_score:.6f}")

    start = time.time()
    deadline = start + budget_s - 5.0

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        M = np.max(np.convolve(cur, cur))

        # Build constraints matrix A for conv(cur, g) <= M
        # shape: (2*n-1, n)
        i = np.arange(n, dtype=np.int64)
        j = np.arange(n, dtype=np.int64)
        I, J = np.meshgrid(i, j, indexing='ij')
        rows = (I + J).ravel()
        cols = I.ravel()
        data = cur[J.ravel()]
        A = sp.csr_matrix((data, (rows, cols)), shape=(2*n - 1, n))

        # RHS
        b = np.full(2*n - 1, M, dtype=np.float64)

        # linear program: maximize sum(g) -> minimize -1⋅g
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iteration {iteration}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print(f"Invalid sum_g at iteration {iteration}")
            break

        # scale g to have same L1 norm as cur (target_sum)
        scale = target_sum / sum_g
        g_scaled = g * scale

        # try a few mixing factors
        improved = False
        best_local_seq = cur
        best_local_score = cur_score

        for t in (1.0, 0.5, 0.2, 0.1):
            cand = (1-t) * cur + t * g_scaled
            cand_score = fast_score(cand)
            if cand_score < best_local_score - 1e-12:
                best_local_seq = cand
                best_local_score = cand_score
                improved = True

        if improved:
            cur = best_local_seq
            cur_score = best_local_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"Iter {iteration:04d} – new best score {best_score:.6f}")
            # early stop if target reached
            if best_score < 1.503:
                print(f"Target reached after {iteration} iterations, score {best_score:.6f}")
                break
        else:
            # not improved, maybe try random perturbation
            # generate small jitter
            jitter = np.random.randn(n) * 1e-3
            cand = np.maximum(0.0, cur + jitter)
            cand = cand * (target_sum / cand.sum())
            cand_score = fast_score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.copy()
                    print(f"Iter {iteration:04d} – jitter improved score {best_score:.6f}")

        # stop if iterations exceed some limit
        if iteration >= 100:
            # maybe you want to break; but hold for time.
            pass

    print(f"Finished after {iteration} iterations. Best score {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a high‑quality step‑function using a deterministic LP “direction” step.
The algorithm starts from the Catalan (half‑binomial) coefficients, which give
the classic bound π/2≈1.5708.  In each iteration we solve a linear program that
maximises the total ℓ₁‑mass of a candidate vector   g  under the constraint
‖f ∗ g‖∞≤M, where f is the current sequence and M=‖f ∗ f‖∞.  The solution g
has a much larger ℓ₁‑norm; scaling it back to the original ℓ₁‑norm produces a
new sequence whose self‑convolution peak is reduced.  A short line‑search over
mixing factors (t∈{1,0.5,0.2,0.1}) is performed to guarantee improvement.
The process is repeated until the time budget expires or the target score
1.503 is reached.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative heights (floats) that minimises the
    ``evaluate_sequence`` score.  The routine uses a linear‑programming
    refinement of the classic Catalan construction.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # small safety margin

    # ------------------------------------------------------------------
    # 1.  Build the initial Catalan (half‑binomial) sequence.
    # ------------------------------------------------------------------
    n = 1024                     # length – can be increased if time permits
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    # Normalise the ℓ₁‑norm to √(2n); this matches the normalisation used
    # in the reference description and keeps the denominator of the score
    # constant throughout the run.
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    cur = a.copy()

    # ------------------------------------------------------------------
    # Helper: fast evaluation of the score (identical to evaluate_sequence)
    # ------------------------------------------------------------------
    def fast_score(seq: np.ndarray) -> float:
        s = seq.sum()
        if s < 0.01:
            return np.inf
        nn = seq.size
        L = 1 << ((2 * nn - 1).bit_length())
        f_fft = np.fft.rfft(seq, n=L)
        conv = np.fft.irfft(f_fft * f_fft, n=L)[: 2 * nn - 1]
        return 2.0 * nn * conv.max() / (s * s)

    cur_score = fast_score(cur)
    best_seq = cur.copy()
    best_score = cur_score
    print(f"Initial Catalan score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Pre‑compute the index pattern for the convolution matrix.
    #     The matrix A satisfies  A @ g  =  f ∗ g .
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    I, J = np.meshgrid(i, j, indexing='ij')
    row_idx = (I + J).ravel()          # shape (n*n,)
    col_idx = I.ravel()
    J_flat = J.ravel()                 # to fetch the appropriate f‑coefficients

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop.  Each round solves a linear program that
    #     pushes mass as far to the right as possible while respecting the
    #     current convolution ceiling M.
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # Current convolution ceiling
        M = np.max(np.convolve(cur, cur))

        # Build the sparse constraint matrix A for the inequality  f∗g ≤ M.
        data = cur[J_flat]                         # coefficient cur[j] for each (i,j)
        A = sp.csr_matrix((data, (row_idx, col_idx)),
                          shape=(2 * n - 1, n))

        # Right‑hand side (same bound for every row)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise Σ g_i   ⇔   minimise –Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iteration {it}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print(f"Degenerate solution at iteration {it}")
            break

        # Scale g back to the original ℓ₁‑norm.
        scale = target_sum / sum_g
        g_scaled = g * scale

        # Try a short line‑search over mixing factors.
        improved = False
        best_local = cur
        best_local_score = cur_score

        for t in (1.0, 0.5, 0.2, 0.1):
            cand = (1.0 - t) * cur + t * g_scaled
            # Numerical safeguard: keep non‑negative and re‑normalise.
            cand = np.maximum(0.0, cand)
            cand = cand * (target_sum / cand.sum())
            cand_score = fast_score(cand)
            if cand_score < best_local_score - 1e-12:
                best_local = cand
                best_local_score = cand_score
                improved = True

        if improved:
            cur = best_local
            cur_score = best_local_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"Iter {it:05d} – new best score: {best_score:.6f}")

            if best_score < 1.503:
                print(f"Target reached at iteration {it} with score {best_score:.6f}")
                break
        else:
            # If LP direction did not improve, inject a tiny random perturbation.
            jitter = np.random.randn(n) * 1e-4
            cand = np.maximum(0.0, cur + jitter)
            cand = cand * (target_sum / cand.sum())
            cand_score = fast_score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.copy()
                    print(f"Iter {it:05d} – random jitter improved score to {best_score:.6f}")

        # Occasionally log progress even if nothing changed.
        if it % 20 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {it:05d} – elapsed {elapsed:.1f}s – best score {best_score:.6f}")

    print(f"Finished after {it} iterations. Best score {best_score:.6f}")
    # Return as plain Python list for the external evaluator.
    return best_seq.tolist()

"""
Propose a step‑function with a lower evaluation score than the Catalan baseline.

The idea is to improve the classic Catalan (half‑binomial) coefficients by a
deterministic “LP direction” step.  Starting from the Catalan sequence `f`,
let `M = ||f*f||_∞`.  We solve the linear program

    max   Σ g_i
    s.t.  (f * g)_k ≤ M   for all k = 0 … 2n‑2
          g_i ≥ 0

which finds a vector `g` that can carry a much larger ℓ₁‑mass while keeping the
convolution with the current `f` bounded by the same peak `M`.  After scaling
`g` back to the original ℓ₁‑norm we mix it with `f` (testing several mixing
factors) and keep the best improvement.  The process is repeated until the
time budget expires or the target score 1.503 is reached.

All helper functions are defined at module level and no closures are used.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def fast_score(seq: np.ndarray) -> float:
    """Fast version of the given evaluation metric (no safety checks)."""
    s = seq.sum()
    if s < 0.01:
        return np.inf
    n = seq.size
    # next power of two for efficient FFT
    L = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)


def propose_candidate(seed: int = 0, budget_s: int = 1000):
    """
    Return a list of non‑negative floats (the step heights) that attempts to
    minimise ``evaluate_sequence``.  The method starts from the Catalan
    coefficients and repeatedly performs an LP‑based mass‑reallocation step
    followed by a short line‑search on the mixing factor.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise with Catalan (half‑binomial) coefficients.
    # ------------------------------------------------------------------
    n = 1024                                 # length of the sequence
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    # Normalise the ℓ₁‑norm to √(2n); this keeps the denominator of the score
    # constant throughout the optimisation.
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    cur = a.copy()

    cur_score = fast_score(cur)
    best_seq = cur.copy()
    best_score = cur_score
    print(f"Initial Catalan score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Pre‑compute index pattern for the convolution matrix.
    #     For any vector f, the matrix A with rows k and columns i defined by
    #         A[k,i] = f[k‑i]  (interpreted as 0 when k‑i is out of range)
    # satisfies  A @ g = f * g .
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.int64)
    j = np.arange(n, dtype=np.int64)
    I, J = np.meshgrid(i, j, indexing='ij')
    row_idx = (I + J).ravel()          # length n*n
    col_idx = I.ravel()
    J_flat = J.ravel()                 # used to fetch the appropriate f‑coeff.

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop.
    # ------------------------------------------------------------------
    it = 0
    while time.time() < deadline:
        it += 1

        # Current peak of the self‑convolution (the current bound M).
        M = np.max(np.convolve(cur, cur))

        # Build the sparse matrix A for the inequality  (cur * g) ≤ M .
        data = cur[J_flat]                         # cur[j] appears in row i+j, col i
        A = sp.csr_matrix((data, (row_idx, col_idx)),
                          shape=(2 * n - 1, n))

        # Right‑hand side vector (identical bound for every row).
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # Linear programme: maximise Σ g_i  ⇔  minimise –Σ g_i .
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iteration {it}: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            print(f"Degenerate LP solution at iteration {it}")
            break

        # Scale g back to the target ℓ₁‑norm.
        scale = target_sum / sum_g
        g_scaled = g * scale

        # ------------------------------------------------------------------
        # 4.  Line‑search over mixing factors t ∈ (0,1] for the candidate
        #     sequence  (1‑t)·cur + t·g_scaled .
        # ------------------------------------------------------------------
        improved = False
        best_local_seq = cur
        best_local_score = cur_score

        # test a fine grid of mixing coefficients
        for t in np.linspace(0.0, 1.0, 21)[1:]:   # skip t=0 (no change)
            cand = (1.0 - t) * cur + t * g_scaled
            cand = np.maximum(0.0, cand)                     # enforce non‑negativity
            cand = cand * (target_sum / cand.sum())          # re‑normalise
            cand_score = fast_score(cand)

            if cand_score < best_local_score - 1e-12:
                best_local_seq = cand
                best_local_score = cand_score
                improved = True

        if improved:
            cur = best_local_seq
            cur_score = best_local_score
            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.copy()
                print(f"Iter {it:05d} – new best score: {best_score:.6f}")

            if best_score < 1.503:
                print(f"Target reached at iteration {it} with score {best_score:.6f}")
                break
        else:
            # If the LP direction did not help, add a tiny random perturbation.
            jitter = np.random.randn(n) * 1e-4
            cand = np.maximum(0.0, cur + jitter)
            cand = cand * (target_sum / cand.sum())
            cand_score = fast_score(cand)
            if cand_score < cur_score:
                cur = cand
                cur_score = cand_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.copy()
                    print(f"Iter {it:05d} – jitter improved score to {best_score:.6f}")

        # Periodic progress report (does not affect the optimisation).
        if it % 20 == 0:
            elapsed = time.time() - start_time
            print(f"Iter {it:05d} – elapsed {elapsed:.1f}s – best score {best_score:.6f}")

    print(f"Finished after {it} iterations. Best score {best_score:.6f}")
    return best_seq.tolist()
