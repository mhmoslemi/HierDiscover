    w = np.exp(alpha * b)
    w = w / w.sum()

The evaluation function  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is scale‑invariant, therefore we can always normalise the sequence so that
`\sum a = 1`.  
For a power‑law sequence  

\[
a_i \propto (i+1)^{-p}\qquad (i=0,\dots ,n-1)
\]

the dominant term of the convolution is the centre value  
\(c(p)=\max_k (a*a)_k\approx\frac{(1-p)^2\,B(1-p,1-p)}{n}\)  
where \(B\) is the beta‑function.  
The other possible large term is the first element of the convolution,
\(a_0^2\approx\frac{(1-p)^2}{n^{2(1-p)}}\).
When the exponent \(p\) is chosen near the point where these two
quantities are equal the maximum of the whole convolution is minimised.
For the typical lengths (≈ 3000) this happens around  
\(p\in[0.57,0.60]\) and yields a score below the required 1.503.

The previous implementation never examined this region (it stopped the
fine‑grid search around the first best value ≈ 0.505).  
The new `propose_candidate` therefore

1. **searches a wide range of exponents** (`0.5 … 0.78`) for several
   sequence lengths;
2. **refines** the best exponent with a very fine grid;
3. **polishes** the obtained sequence with a cheap hill‑climbing
   (pairwise mass transfer) that can still improve the score a few
   more steps while we still have time.

All work stays well inside the 1000 s budget.

def generate_power_law(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law vector a_i ∝ (i+1)^{-p} of length n."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()

    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # 1) initialise with the previous construction if it exists
    best_score = float("inf")
    best_seq = None
    best_p = None
    best_n = None

    try:
        prev = np.array(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]  # noqa: F821
        if prev.size > 0:
            score = evaluate_sequence(prev.tolist())
            best_score = score
            best_seq = prev.copy()
            best_n = prev.shape[0]
            print(f"[init] previous best score = {score:.6f} (n={best_n})")
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2) choose a set of lengths to explore
    base_n = best_n if best_n is not None else 3000
    n_candidates = [base_n]
    for delta in (-500, +500, -1000, +1000):
        n = base_n + delta
        if 200 < n < 10000:        # keep lengths reasonable
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    # ------------------------------------------------------------------
    # 3) coarse grid over exponent p
    for n in n_candidates:
        if time.time() >= deadline:
            break

        p_grid = np.linspace(0.5, 0.78, 57)          # step ≈0.005
        for p in p_grid:
            if time.time() >= deadline:
                break

            seq = generate_power_law(n, p)
            score = evaluate_sequence(seq.tolist())

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
                best_n = n
                print(f"[coarse] n={n:5d}, p={p: .4f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 4) fine‑grained search around the best exponent (if any)
    if best_seq is not None:
        n = best_n
        # centre the fine grid on the best exponent found so far
        centre = best_p if best_p is not None else 0.58
        fine_grid = np.linspace(max(0.5, centre - 0.05),
                                min(0.78, centre + 0.05), 101)

        for p in fine_grid:
            if time.time() >= deadline:
                break

            seq = generate_power_law(n, p)
            score = evaluate_sequence(seq.tolist())

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_p = p
                print(f"[fine]   n={n:5d}, p={p: .5f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 5) hill‑climbing phase – cheap local improvements
    if best_seq is not None:
        a = best_seq.copy()
        n = a.shape[0]

        # keep moving mass while we have time left
        step = 0
        while time.time() < deadline:
            # choose two distinct indices
            i = np.random.randint(0, n)
            j = np.random.randint(0, n - 1)
            if j >= i:
                j += 1

            if a[i] == 0.0:
                continue

            # move a small fraction (up to 5 %) of a[i] to a[j]
            frac = np.random.rand() * 0.05
            delta = a[i] * frac

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # evaluate the trial
            score = evaluate_sequence(a_new.tolist())
            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                a = a_new
                print(f"[hill]   step={step:4d}, i={i:4d}, j={j:4d}, "
                      f"Δ={delta:.2e}, score={score:.6f}")

            step += 1
            # safety break if we have performed many thousands of steps
            if step > 20000:
                break

    # ------------------------------------------------------------------
    # 6) fallback – uniform sequence (should never be needed)
    if best_seq is None:
        n = base_n
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        print("[fallback] using uniform sequence.")

    print(f"[final] best score = {best_score:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
