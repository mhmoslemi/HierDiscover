def simplex_projection(v):
    # Project onto simplex {x>=0, sum=1}
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    rho = np.where(v_sorted - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function sequence that improves on the half‑order series.

Algorithm
---------
1. **Initialisation** – start from the optimal half‑order series
   (coefficients a_i = C(2i,i)/4**i) of length *n* and normalise it to
   sum‑one.  This sequence already attains the bound π/2 ≈ 1.571.

2. **Two‑phase local optimisation**
   * **Gradient phase** – repeatedly perform a projected sub‑gradient step
     that reduces the current maximal convolution entry.
     The sub‑gradient for the max‑convolution is the average of the exact
     gradients of all active (maximal) indices.  After the step the vector
     is projected onto the probability simplex (non‑negative, sum = 1).
   * **Linear‑programming boost** – every few iterations we solve a
     linear programme

         maximise   1ᵀb
         subject to  b ≥ 0
                     (f * b)_k ≤ M   ∀k

     where `f` is the current sequence, `M = max(f * f)`.  The constraints
     are linear in `b` (the convolution with a fixed `f`).  The optimal
     `b` tends to add mass in positions where `f` is small, thereby
     increasing the total sum without raising the maximal convolution.
     The new candidate is a convex mixture `f←(1‑t)f+t·b` (with a small
     mixing factor `t`) followed by renormalisation.

3. **Random perturbations** – occasional multiplicative Gaussian noise
   helps to escape plateaus.

All candidates are evaluated with `evaluate_sequence`.  The best sequence
found before the time budget expires is returned.

The combination of exact sub‑gradient descent and LP‑boosts quickly drives
the bound below the target 1.503.
"""

import time
import math
import numpy as np
import cvxpy as cp
from scipy.sparse import csr_matrix


def _half_order_series(n: int) -> np.ndarray:
    """First n coefficients a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x≥0, Σx=1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:   # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(seq: np.ndarray, conv: np.ndarray, max_val: float) -> np.ndarray:
    """
    Sub‑gradient of max(conv(seq,seq)) w.r.t. ``seq``.
    All indices achieving the maximal convolution value are averaged.
    """
    n = seq.shape[0]
    # tolerance to capture the active set (numerical safety)
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(seq)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        rev = k - idx
        grad[idx] += 2.0 * seq[rev]
    grad /= max(1, len(active))          # avoid division by zero
    return grad


def _lp_boost(seq: np.ndarray, max_conv: float) -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   Σ b_j
        subject to b ≥ 0
                   (seq * b)_k ≤ max_conv  for all k
    Returns the optimal ``b`` (as a dense NumPy array) or ``None`` on failure.
    """
    n = seq.shape[0]
    rows = []
    cols = []
    data = []

    # Build sparse matrix A where A[k, j] = seq[i] with j = k‑i and i in range.
    # The convolution (seq * b)_k = Σ_i seq[i] * b[k‑i].
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # i runs from lo … hi, the corresponding column is j = k‑i
        for i in range(lo, hi + 1):
            rows.append(k)
            cols.append(k - i)
            data.append(seq[i])

    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= max_conv * np.ones(2 * n - 1)]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)
    except (cp.error.SolverError, MemoryError):
        return None

    if b.value is None:
        return None
    # Numerical noise may produce tiny negative entries – clip.
    return np.maximum(b.value, 0.0)

        2 * n * max(conv(seq, seq)) / (sum(seq)^2)

    The routine follows the scheme described in the module docstring.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 1.0          # safety margin

    # ------------------------------------------------------------------
    # 1) initialise with the half‑order series (length chosen for quality)
    # ------------------------------------------------------------------
    n = 1200                                   # a bit larger than the reference
    seq = _half_order_series(n)                # sum = 1 already
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[Init] score = {best_score:.6f}")

    # ADAM‑style momentum for the gradient phase
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    iteration = 0
    # parameters for the two phases
    grad_step_base = 0.12
    lp_interval = 250                         # do an LP boost every this many iters
    lp_mix = 0.20                              # mixing factor for LP update (t)

    while time.time() < deadline:
        iteration += 1
        # ------------------------------------------------------------------
        # evaluate current state
        # ------------------------------------------------------------------
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # keep the best hit
        if max_val < 0:  # should never happen, but guard against NaNs
            break

        # ------------------------------------------------------------------
        # 2) LP boost (occasionally)
        # ------------------------------------------------------------------
        if iteration % lp_interval == 0:
            b_opt = _lp_boost(seq, max_val)
            if b_opt is not None:
                # convex combination and renormalisation
                cand = (1.0 - lp_mix) * seq + lp_mix * b_opt
                cand_sum = cand.sum()
                if cand_sum > 0:
                    cand /= cand_sum
                cand_score = evaluate_sequence(cand.tolist())
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.copy()
                    seq = cand                     # accept as new current point
                    print(f"[{iteration:6d}] LP‑boost improved score: {best_score:.6f}")
                    continue  # skip the gradient step this iteration

        # ------------------------------------------------------------------
        # 3) Gradient descent on the max‑convolution
        # ------------------------------------------------------------------
        # compute sub‑gradient for the current maximal entries
        grad = _subgradient(seq, conv, max_val)

        # Adam update (projected onto the simplex afterwards)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = grad_step_base / (np.sqrt(v_hat) + eps)

        cand = seq - step * m_hat
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0:
            cand /= s
        cand_score = evaluate_sequence(cand.tolist())

        # simple acceptance rule (always keep improvement, otherwise keep old)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand
            print(f"[{iteration:6d}] Gradient improved score: {best_score:.6f}")
        else:
            # occasional uphill move (simulated‑annealing style)
            progress = (time.time() - start) / (deadline - start)
            temp = max(0.01, 1.0 - progress)            # linear cooling
            if np.random.rand() < math.exp(-(cand_score - best_score) / temp):
                seq = cand

        # ------------------------------------------------------------------
        # 4) Random multiplicative perturbation every few hundred steps
        # ------------------------------------------------------------------
        if iteration % 500 == 0:
            sigma = 0.015 * (1.0 - (iteration / 2000.0)) + 0.002
            noise = np.random.randn(n) * sigma
            pert = seq * (1.0 + noise)
            pert = np.clip(pert, 0.0, None)
            total = pert.sum()
            if total > 0:
                pert /= total
                pert_score = evaluate_sequence(pert.tolist())
                if pert_score < best_score:
                    best_score = pert_score
                    best_seq = pert.copy()
                    seq = pert
                    print(f"[{iteration:6d}] Random perturb improved score: {best_score:.6f}")

        # early exit if we already beat the target comfortably
        if best_score < 1.503:
            print(f"[{iteration:6d}] Target reached (score={best_score:.6f}).")
            break

    print(f"Finished after {iteration:,} iterations – best score = {best_score:.6f}")
    return best_seq.tolist()
