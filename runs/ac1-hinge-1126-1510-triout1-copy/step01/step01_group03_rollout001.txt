def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

def solve_lp(a, M):
    import scipy.optimize
    n = len(a)
    m = 2*n - 1
    # construct convolution matrix A
    col = np.concatenate([a, np.zeros(n-1, dtype=np.float64)])

    else:

"""
Propose a candidate sequence for the self‑convolution max‑norm problem.

Algorithm (Frank‑Wolfe style)
--------------------------------
1. Start from the classic “half‑order’’ series a_i = C(2i,i)/4**i,
   normalised to sum = 1.
2. Repeatedly improve the sequence:
   * Compute M = max(conv(a,a)).
   * Solve a linear program:
       maximise   Σ b_j
       subject to  a * b  ≤  M    (full convolution, element‑wise)
                   b ≥ 0
     The LP gives a feasible direction b that pushes mass away from
     the currently maximal convolution entry.
   * Normalise b to a probability vector (sum = 1).
   * Perform a line‑search on the convex combination
         a_new = (1‑γ)·a + γ·b      (γ ∈ {0.1,0.25,0.5,0.75,1.0})
     and keep the best γ that decreases the evaluation score.
   * Occasionally inject a small random multiplicative perturbation
     (projected back onto the simplex) to escape shallow plateaus.
3. Keep the best sequence seen throughout the run and return it
   before the time budget expires.

The optimisation is performed on the simplex, so the objective
`2·n·max(conv)/sum²` is invariant to scaling – we keep the sum fixed at 1.
The LP is dense but small (≈2 n × n) and solved with SciPy’s HiGHS
solver, which is fast enough for n≈800–1000 within the 1000 s budget.
"""

def _half_order_series(n: int) -> np.ndarray:
    """First n terms of a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from
    # Wang & Carreira‑Perpiñán (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolution_matrix(a: np.ndarray) -> np.ndarray:
    """Return the dense matrix A such that A @ b = np.convolve(a, b)."""
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
    row = np.concatenate([a[:1], np.zeros(n - 1, dtype=a.dtype)])
    return scipy.linalg.toeplitz(col, row)


def _solve_lp(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   a * b ≤ M   (full convolution, elementwise),
    b ≥ 0.
    Returns the optimal b (not normalised) or None if LP fails.
    """
    n = a.shape[0]
    # Build constraint matrix A_ub (shape (2n‑1, n))
    A = _convolution_matrix(a)
    c = -np.ones(n, dtype=np.float64)          # maximise Σ b  → minimise –Σ b
    b_ub = np.full(A.shape[0], M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    evaluate_sequence(...).  Returns the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1) initialise – half‑order series
    # ------------------------------------------------------------------
    n = 800  # a comfortable size for the LP (≈1.3 MiB matrix)
    a = _half_order_series(n)

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # Parameters for occasional random “shake’’
    shake_interval = 30
    shake_sigma0 = 0.02
    shake_sigma_final = 0.001

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # --------------------------------------------------------------
        # 2) compute current max‑convolution M
        # --------------------------------------------------------------
        conv = np.convolve(a, a)               # full convolution, length 2n‑1
        M = conv.max()
        # --------------------------------------------------------------
        # 3) solve the LP to obtain a feasible direction b
        # --------------------------------------------------------------
        if _SCIPY_AVAILABLE:
            b = _solve_lp(a, M)
        else:
            b = None

        # If LP failed for any reason, fall back to a simple gradient step
        if b is None or b.sum() == 0.0:
            # Simple sub‑gradient descent on the maximal index
            max_idx = np.argmax(conv)
            # gradient of that entry w.r.t. a is 2*a_rev at the selected index
            grad = np.zeros_like(a)
            lo = max(0, max_idx - (n - 1))
            hi = min(n - 1, max_idx)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            rev = max_idx - idx
            grad[idx] = 2.0 * a[rev]
            # elementary step size schedule
            step = 0.01 / math.sqrt(iteration)
            cand = a - step * grad
            cand = _project_onto_simplex(cand)
        else:
            # Normalise b to a probability vector (sum = 1)
            b_norm = b / b.sum()
            # ----------------------------------------------------------
            # 4) line‑search on the convex combination (a ↦ (1‑γ)a+γb)
            # ----------------------------------------------------------
            gammas = [0.1, 0.25, 0.5, 0.75, 1.0]
            cand = a  # placeholder
            cand_score = best_score
            improved = False
            for gamma in gammas:
                a_new = (1.0 - gamma) * a + gamma * b_norm
                # a_new already sums to 1; ensure non‑negativity (numerical)
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()
                score = evaluate_sequence(a_new.tolist())
                if score < cand_score:
                    cand = a_new
                    cand_score = score
                    improved = True
            if not improved:
                # no improvement with the tested gammas – keep current a
                cand = a

        # --------------------------------------------------------------
        # 5) occasional random perturbation (“shake”)
        # --------------------------------------------------------------
        if iteration % shake_interval == 0:
            progress = (time.time() - start) / (budget_s - 2.0)
            sigma = shake_sigma0 * (1.0 - progress) + shake_sigma_final
            noise = np.random.randn(n) * sigma
            noisy = a * (1.0 + noise)
            noisy = np.maximum(noisy, 0.0)
            noisy = _project_onto_simplex(noisy)
            noisy_score = evaluate_sequence(noisy.tolist())
            if noisy_score < best_score:
                cand = noisy
                cand_score = noisy_score
                print(f"[{iteration:5d}] shake improved score: {cand_score:.6f}")

        # --------------------------------------------------------------
        # 6) accept the candidate if it improves the current sequence
        # --------------------------------------------------------------
        curr_score = evaluate_sequence(cand.tolist())
        if curr_score < best_score:
            best_score = curr_score
            best_seq = cand.copy()
            print(f"[{iteration:5d}] new best score: {best_score:.6f}")

        a = cand  # continue optimisation from the latest candidate

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score = {best_score:.6f}")
    return best_seq.tolist()
