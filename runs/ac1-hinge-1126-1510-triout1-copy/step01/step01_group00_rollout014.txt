        max_conv = conv.max()

            best_score = cur_score
            best_seq = (a * target_sum).copy()
            print(f"Iter {iteration} new best score {best_score:.6f}")
        # smoothing
        if smoothing is None:
            smoothing = max_conv * 0.5 + 1e-12
        else:
            smoothing = max(smoothing * 0.99, max_conv * 1e-4)

            w[conv.argmax()] = 1.0

        dot = a.dot(grad_a)

        # Adam update
        m_t = beta1 * m_t + (1 - beta1) * grad_p
        v_t = beta2 * v_t + (1 - beta2) * (grad_p ** 2)
        m_hat = m_t / (1 - beta1 ** (iteration + 1))
        v_hat = v_t / (1 - beta2 ** (iteration + 1))

        max_norm = 5.0
        step_norm = np.linalg.norm(step)
        if step_norm > max_norm:
            step = step * (max_norm / step_norm)
        p = p - step
        iteration += 1
    return best_seq, best_score

"""
propose_candidate
-----------------
Optimises the step‑function heights by directly minimising the
sup‑norm of the self‑convolution on the probability simplex.
The algorithm works with the normalised vector (∑a = 1) and uses a
smooth approximation of the max‑operator (soft‑max) together with an
Adam‑style mirror‑descent update in the log‑probability (softmax) space.
Multiple initialisations (previous best, uniform, random) are tried and
the best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def softmax(p: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum = 1)."""
    shift = p - np.max(p)
    e = np.exp(shift)
    return e / e.sum()


# ----------------------------------------------------------------------
def compute_full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·n‑1) using FFT.
    Returns a real array of length 2·n‑1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())            # next power of two
    A = np.fft.fft(a, n=L)
    conv = np.real(np.fft.ifft(A * A))[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def compute_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the (smoothed) max‑convolution w.r.t. the probability
    vector a.  w is the weight vector (same length as the full convolution
    a*a).  The gradient has length n.
    """
    n = a.shape[0]
    L = 1 << ((len(w) + n - 1).bit_length())      # enough for linear correlation
    W = np.fft.fft(w, n=L)
    A_rev = np.fft.fft(a[::-1], n=L)
    corr_full = np.real(np.fft.ifft(W * A_rev))
    # slice the valid part (indices n‑1 … n‑1+n‑1)
    grad = 2.0 * corr_full[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def run_optimization(p0: np.ndarray,
                     n: int,
                     target_sum: float,
                     deadline: float,
                     best_seq: np.ndarray,
                     best_score: float,
                     max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam‑mirror‑descent on the log‑probability vector p0.
    Returns the best (scaled) sequence and its score encountered.
    """
    # Adam state
    m = np.zeros_like(p0)
    v = np.zeros_like(p0)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.5                     # base learning rate

    p = p0.copy()
    smoothing = None

    for it in range(1, max_iters + 1):
        # stop if time ran out
        if time.time() >= deadline:
            break

        a = softmax(p)                     # a ≥ 0, ∑a = 1
        conv = compute_full_convolution(a)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv     # exactly what evaluate_sequence would return

        # keep the best encountered sequence (scaled to the required sum)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = (a * target_sum).copy()
            print(f"Iter {it:6d}: new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # smoothing for a soft‑max approximation of the max‑operator
        if smoothing is None:
            smoothing = max_conv * 0.5 + 1e-12
        else:
            # decay but never below a tiny fraction of the current max
            smoothing = max(smoothing * 0.99, max_conv * 1e-4)

        # compute weight vector w ≈ softmax(conv / smoothing)
        if smoothing < 1e-15:
            # essentially a pure sub‑gradient: put all weight on the max index
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            shift = conv - max_conv        # ≤ 0
            exp_vals = np.exp(shift / smoothing)
            w = exp_vals / exp_vals.sum()

        # --------------------------------------------------------------
        # gradient of the smoothed objective w.r.t. a
        grad_a = compute_gradient(a, w)

        # mirror‑descent gradient on the log‑probability (softmax) space
        dot = a.dot(grad_a)
        grad_p = a * (grad_a - dot)        # Jacobian of softmax multiplied by ∂L/∂a

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (slowly decreasing)
        lr = lr0 / (1.0 + it / 5000.0)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # optional clipping of huge updates (keeps p numerically stable)
        max_norm = 5.0
        step_norm = np.linalg.norm(step)
        if step_norm > max_norm:
            step = step * (max_norm / step_norm)

        p -= step

    return best_seq, best_score
