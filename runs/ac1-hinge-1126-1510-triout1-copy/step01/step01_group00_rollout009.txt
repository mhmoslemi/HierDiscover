# Helper: Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]               # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):               # all non‑positive → uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast convolution (linear) via FFT – returns full length 2·n‑1 array
def _conv_fft(a: np.ndarray, L_fft: int) -> np.ndarray:
    """Linear convolution of a with itself using a pre‑allocated FFT length."""
    fft_a = np.fft.fft(a, n=L_fft)
    conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Fast correlation (valid mode) via FFT.
#   corr = a * w  (where w is a weight vector of length 2·n‑1)
def _corr_fft(w: np.ndarray, a_rev: np.ndarray, L_fft: int) -> np.ndarray:
    """Correlation of w (length 2·n‑1) with a (via a_rev) using FFT."""
    fft_w = np.fft.fft(w, n=L_fft)
    fft_a = np.fft.fft(a_rev, n=L_fft)
    full = np.real(np.fft.ifft(fft_w * fft_a))
    n = a_rev.shape[0]
    # valid part: indices n‑1 … n‑1+n‑1
    return full[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Objective used by the competition (scale‑invariant, sum can be arbitrary)
def _score(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    n = seq.shape[0]
    L = 1 << ((2 * n - 1).bit_length())
    conv = _conv_fft(seq, L)
    return float(2.0 * n * conv.max() / (seq.sum() ** 2))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation function.  The function respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Determine the length n (use the global if available)
    try:
        n = len(height_sequence_1)                 # type: ignore
    except Exception:
        n = 6000                                    # a comfortable default

    target_sum = math.sqrt(2.0 * n)                 # the normalisation used later
    unit_target = 1.0                               # we work on the simplex Σa = 1

    # pre‑computed FFT lengths
    L_conv = 1 << ((2 * n - 1).bit_length())       # for self‑convolution
    L_corr = 1 << ((3 * n - 2).bit_length())       # for correlation (w * a_rev)

    # ------------------------------------------------------------------
    # 1) GENERATE DIVERSE STARTING POINTS
    candidates = []

    # a) uniform distribution
    cand = np.full(n, 1.0 / n)
    candidates.append(cand)

    # b) previous best if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=float)      # type: ignore
        prev = prev.astype(float)
        if prev.size == n:
            candidates.append(prev / prev.sum())
    except Exception:
        pass

    # c) power‑law sweep (both forward and reversed)
    exponents = np.linspace(0.0, 2.5, 51)           # 0 … 2.5 in steps of 0.05
    for p in exponents:
        w = np.power(np.arange(1, n + 1, dtype=float), p)
        a = w / w.sum()
        candidates.append(a)
        candidates.append(a[::-1])

    # d) random Dirichlet‑like draws (Gamma with shape 0.5 → sparse)
    for _ in range(30):
        g = np.random.gamma(shape=0.5, scale=1.0, size=n)
        a = g / g.sum()
        candidates.append(a)

    # ------------------------------------------------------------------
    # 2) QUICK pre‑evaluation, keep the best few
    best_score = float("inf")
    best_seq = None
    evaluated = []
    for idx, a in enumerate(candidates):
        if time.time() > deadline:
            break
        sc = _score(a * target_sum)               # scale to the required sum
        evaluated.append((sc, a.copy()))
        if sc < best_score:
            best_score, best_seq = sc, a.copy()
    # keep top K for refinement
    K = 6
    evaluated.sort(key=lambda x: x[0])
    top_candidates = [seq for _, seq in evaluated[:K]]

    print(f"--- Generated {len(candidates)} candidates, best pre‑score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) REFINEMENT – Adam optimiser with smoothed max‑norm
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    alpha = 0.12                     # base learning rate
    lambda_start = 5.0               # initial smoothing parameter
    lambda_max = 1500.0
    max_iters_per_restart = 5000

    for restart, a0 in enumerate(top_candidates):
        if time.time() > deadline:
            break
        a = a0.copy()                # a lives on the simplex Σa = 1
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        t = 0

        # adam loop
        while time.time() < deadline and t < max_iters_per_restart:
            t += 1

            # ---- convolution of current iterate ----------
            conv = _conv_fft(a, L_conv)
            max_c = conv.max()

            # ---- gradually increase smoothing λ ----------
            lam = min(lambda_max, lambda_start * (1.0 + 0.001 * t))

            # ---- softmax‑style weights over the conv entries ----------
            # w_k ∝ exp(λ·(c_k - max_c))  → max weight = 1
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:               # numerical safety (should not happen)
                w = np.ones_like(w) / w.size
                w_sum = 1.0
            w_norm = w / w_sum

            # ---- gradient = 2·correlation(w_norm, a) ----------
            grad = _corr_fft(w_norm, a[::-1], L_corr)
            grad *= 2.0

            # ---- Adam update ----------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = alpha * m_hat / (np.sqrt(v_hat) + eps)

            a_new = a - step

            # ---- projection back onto the probability simplex ----------
            a_new = _project_simplex(a_new)

            # ---- occasional evaluation ----------
            if t % 10 == 0:
                cur_score = _score(a_new * target_sum)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a_new * target_sum
                    print(
                        f"[restart {restart}] iter {t:5d} – new best = {best_score:.6f}"
                    )
                # optional: break early if improvement stalls
                # (not enforced here)

            a = a_new

        print(
            f"[restart {restart}] finished after {t} iters, local best = {_score(a * target_sum):.6f}"
        )

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (converted to plain Python list)
    if best_seq is None:
        # as a final safety, fall back to a uniform construction
        best_seq = np.full(n, target_sum / n)

    final_seq = best_seq.tolist()
    print(f"--- Finished overall – best score = {best_score:.6f} (n={n}) ---")
    return final_seq
