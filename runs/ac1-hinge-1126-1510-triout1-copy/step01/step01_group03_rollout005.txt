"""
Propose candidate sequence using a soft‑max‑smoothed gradient descent.
The algorithm works on the simplex (non‑negative entries with sum = 1).
It repeatedly:
    1. Computes the self‑convolution c = a * a.
    2. Forms a soft‑max distribution over the convolution entries:
           p_k ∝ exp(β·(c_k – max(c))).
       For β→∞ this concentrates on the maximal entry;
       we start with a small β and increase it during the run.
    3. The gradient of the smoothed objective
           f(a) = Σ_k p_k·c_k
       w.r.t. a_i is   g_i = 2·Σ_k p_k·a_{k‑i} .
       This can be obtained efficiently with one extra convolution.
    4. Perform a multiplicative (mirror‑descent) update
           a_i ← a_i·exp(–η·g_i)    and renormalise.
    5. Occasionally add a tiny random multiplicative perturbation
       (simulated‑annealing style) to avoid local minima.
The step‑size η and the soft‑max temperature β are scheduled
linearly with the elapsed time.  The best sequence seen so far
(and its evaluation score) is kept and returned when the time
budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _half_order_series(n: int) -> np.ndarray:
    """First n coefficients of a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # recurrence a_i = a_{i‑1} * ((2i‑1)*(2i)) / (4*i*i)
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _exp_series(n: int, lam: float = 0.99) -> np.ndarray:
    """Geometric series a_i = lam**i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    a /= a.sum()
    return a


def _score_from_seq(seq: np.ndarray) -> float:
    """
    Compute the evaluation function (2 * n * max(conv(seq,seq))) for a
    sequence that is already normalised to sum = 1.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    return float(2 * n * conv.max())


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Return a probability vector proportional to exp(beta * (conv - max(conv))).
    This is numerically stable because the maximum is subtracted.
    """
    max_c = conv.max()
    # all entries are ≤ max_c, so exponent is in [0,1]
    exp_vals = np.exp(beta * (conv - max_c))
    return exp_vals / exp_vals.sum()


def _gradient_via_convolution(a: np.ndarray, p: np.ndarray) -> np.ndarray:
    """
    Compute the gradient g_i = 2 * Σ_k p_k * a_{k-i} efficiently.
    a : length n, p : length 2n‑1 (probability weights).
    Returns an array of length n.
    """
    # reverse p to turn the linear combination into a convolution
    p_rev = p[::-1]
    conv_full = np.convolve(a, p_rev)          # length 3n‑2
    n = len(a)
    # indices 2n‑2‑i for i=0..n‑1 give the required sums
    idx = (2 * n - 2) - np.arange(n)
    grad = 2.0 * conv_full[idx]
    return grad

# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    the supplied evaluation function.  The algorithm runs for up to
    `budget_s` seconds and returns the best sequence found.

    The core optimisation method is a soft‑max‑smoothed gradient descent
    on the simplex, with a decreasing multiplicative step‑size and a modest
    simulated‑annealing perturbation to escape plateaus.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a tiny safety margin for the final return
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1) Choose a problem size and initialise a good starting point.
    #      Larger n gives more degrees of freedom; 1500 is a reasonable
    #      trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 1500
    a = _half_order_series(n)                      # good deterministic start
    # a small admixture of an exponential tail often helps breaking symmetry
    a = 0.90 * a + 0.10 * _exp_series(n, lam=0.995)
    a = a / a.sum()                               # ensure exact sum = 1

    best_seq = a.copy()
    best_score = _score_from_seq(best_seq)

    # ------------------------------------------------------------------
    # 2) Hyper‑parameters for the schedule.
    # ------------------------------------------------------------------
    iteration = 0
    # start with a modest soft‑max temperature and increase it linearly
    beta_start, beta_end = 5.0, 30.0
    # start with a relatively large step‑size and shrink it to a tiny value
    eta_start, eta_end = 0.6, 0.005
    # how often to inject random noise (in iterations)
    noise_interval = 7
    # after how many iterations without improvement we restart from a new random point
    stagnation_limit = 5000
    last_improvement_iter = 0

    while time.time() < deadline:
        iteration += 1
        elapsed = time.time() - start_time
        prog = elapsed / (budget_s - 0.5)   # 0 … 1

        # --------------------------------------------------------------
        # 2a) Current soft‑max temperature β and step size η
        # --------------------------------------------------------------
        beta = beta_start + prog * (beta_end - beta_start)
        eta = eta_start + prog * (eta_end - eta_start)

        # --------------------------------------------------------------
        # 2b) Compute convolution and soft‑max weights
        # --------------------------------------------------------------
        conv = np.convolve(a, a)               # length 2n‑1
        p = _softmax_weights(conv, beta)       # probability vector, sum(p)=1

        # --------------------------------------------------------------
        # 2c) Gradient (mirror‑descent style) and multiplicative update
        # --------------------------------------------------------------
        grad = _gradient_via_convolution(a, p)  # length n
        # multiplicative update – always stays non‑negative
        a = a * np.exp(-eta * grad)

        # re‑normalise to the simplex
        s = a.sum()
        if s == 0.0:                # numerical safety – revert to best known
            a = best_seq.copy()
            continue
        a = a / s

        # --------------------------------------------------------------
        # 2d) Occasionally add a tiny random multiplicative perturbation
        # --------------------------------------------------------------
        if iteration % noise_interval == 0:
            sigma = 0.004 * (1.0 - prog) + 0.0005   # slowly shrinking noise
            noise = np.random.randn(n) * sigma
            cand = a * (1.0 + noise)
            cand = np.clip(cand, 0.0, None)
            cand_sum = cand.sum()
            if cand_sum > 0.0:
                cand /= cand_sum
                cand_score = _score_from_seq(cand)
                if cand_score < best_score:
                    best_score, best_seq = cand_score, cand.copy()
                    a = cand
                    last_improvement_iter = iteration
                    print(f"[{iteration:6d}] Random perturbation improved score: {best_score:.6f}")
                else:
                    # accept with a very small probability to keep exploration
                    if np.random.rand() < 0.001:
                        a = cand

        # --------------------------------------------------------------
        # 2e) Evaluate current sequence and keep the best one
        # --------------------------------------------------------------
        cur_score = _score_from_seq(a)
        if cur_score < best_score:
            best_score, best_seq = cur_score, a.copy()
            last_improvement_iter = iteration
            print(f"[{iteration:6d}] New best score: {best_score:.6f}")

        # --------------------------------------------------------------
        # 2f) Restart if we have been stagnant for a long time
        # --------------------------------------------------------------
        if iteration - last_improvement_iter > stagnation_limit:
            # fresh random Dirichlet start, blended with the best so far
            random_start = np.random.dirichlet(np.ones(n))
            a = 0.8 * best_seq + 0.2 * random_start
            a = a / a.sum()
            last_improvement_iter = iteration
            print(f"[{iteration:6d}] Restarted after stagnation (score {best_score:.6f})")

    # ----------------------------------------------------------------------
    # 3) Return the best sequence discovered (as plain Python list)
    # ----------------------------------------------------------------------
    return best_seq.tolist()
