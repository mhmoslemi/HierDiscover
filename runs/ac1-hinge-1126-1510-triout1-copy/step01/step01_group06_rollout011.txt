"""
Propose a coefficient sequence that (hopefully) beats the previous best
(≈1.576) for the given evaluation metric.

The algorithm mixes three ideas:

1. **Diverse initial seeds** – square‑root‑difference, exponential,
   power‑law, random Dirichlet draws and, if available, the previously
   discovered best sequence (`height_sequence_1`).

2. **Simulated‑annealing style local search** – at each step we locate the
   convolution peak, compute the exact gradient of that peak with respect
   to the coefficients and move a small amount of mass from the most
   “responsible’’ index to the least responsible one.  The move is always
   accepted if it improves the objective; otherwise it may be accepted
   with a temperature‑dependent probability to escape shallow minima.

3. **Linear‑programming refinement** – after the stochastic phase we solve
   a linear program (CVXPY) that maximises the ℓ¹‑norm of a new vector
   `b` under the *current* convolution bound.  Mixing the current vector
   with the scaled optimiser (`b`) provably does not increase the peak
   and typically reduces it further.

The code respects the 1000 s budget, prints progress, and finally returns
the best sequence scaled to the full `[0,1000]` range (the scaling does not
affect the objective because the metric is homogeneous).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_max_and_index(seq: np.ndarray):
    """Return (max_value, index_of_max) of the full self‑convolution."""
    conv = np.convolve(seq, seq)
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _score_from_seq(seq: np.ndarray) -> float:
    """Fast evaluation of the metric for a sequence with sum == 1."""
    # score = 2 * n * max(conv(seq,seq))  because sum(seq) == 1
    n = seq.shape[0]
    max_b, _ = _conv_max_and_index(seq)
    return 2.0 * n * max_b


def _gradient_of_peak(seq: np.ndarray, k_max: int):
    """
    Exact gradient of the convolution value at index `k_max` with respect to
    the entries of `seq`.  For i ≠ j the contribution is 2*seq[j];
    for i == j it is 2*seq[i].
    """
    n = seq.shape[0]
    grad = np.zeros_like(seq)

    i_low = max(0, k_max - (n - 1))
    i_high = min(k_max, n - 1)

    i_vals = np.arange(i_low, i_high + 1, dtype=np.int64)
    j_vals = k_max - i_vals

    mask_lt = i_vals < j_vals
    mask_eq = i_vals == j_vals

    # i < j pairs – add 2*seq[j] to grad[i] and 2*seq[i] to grad[j]
    i_lt = i_vals[mask_lt]
    j_lt = j_vals[mask_lt]
    grad[i_lt] += 2.0 * seq[j_lt]
    grad[j_lt] += 2.0 * seq[i_lt]

    # i == j (the centre of an odd‑length convolution)
    if np.any(mask_eq):
        i_eq = i_vals[mask_eq]
        grad[i_eq] += 2.0 * seq[i_eq]

    return grad


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1) × n matrix `A` such that (A @ b) equals the linear
    convolution f * b (f is treated as constant, b is the variable).
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f to rows j … j+n‑1
    for j in range(n):
        A[j : j + n, j] = f[: rows - j]
    return A


def _lp_max_l1_under_bound(f: np.ndarray, bound: float):
    """
    Solve the LP:
        maximise   sum(b)
        s.t.       f * b <= bound    (element‑wise, linear convolution)
                b >= 0
    Returns the optimal vector `b` (as a 1‑D numpy array) or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover – cvxpy should be available per rules
        return None

    n = f.shape[0]
    A = _build_convolution_matrix(f)                # (2n‑1) × n
    b = cp.Variable(n, nonneg=True)

    rhs = np.full(2 * n - 1, bound, dtype=np.float64)
    constraints = [A @ b <= rhs]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    # ECOS is fast and available in the default CVXPY installation
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)

    if b.value is None:
        return None
    return np.asarray(b.value).reshape(-1)


def _lp_refine(seq: np.ndarray, mix: float = 0.5, rounds: int = 3):
    """
    Apply a few LP‑based mixing steps to the given sequence.
    `mix` is the interpolation factor between the current vector and the LP
    optimiser (0 < mix ≤ 1).  Returns a (potentially) improved sequence.
    """
    a = seq.copy()
    n = a.shape[0]

    for _ in range(rounds):
        max_b, _ = _conv_max_and_index(a)
        b_opt = _lp_max_l1_under_bound(a, max_b)
        if b_opt is None:
            break
        s = b_opt.sum()
        if s <= 0.0:
            break
        # scale to unit ℓ¹‑norm
        g = b_opt / s
        # mixing step – convex combination preserves the ℓ¹‑norm (=1)
        a = (1.0 - mix) * a + mix * g

        # optional small decay of mix to avoid over‑mixing later
        mix *= 0.7

    # make sure the result is still a proper probability vector
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total > 0:
        a /= total
    return a

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑autocorrelation step‑function using
    - several diversified initialisations,
    - a simulated‑annealing mass‑transfer local optimiser,
    - a final LP‐based refinement.

    Returns a list of heights (floats) in the range [0,1000].
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0        # small safety margin

    # --------------------------------------------------------------
    # 1. Determine the length of the step function (default 1542)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)               # global variable from the runner
    except Exception:
        n = 1542

    # --------------------------------------------------------------
    # 2. Build a collection of promising start vectors (all ℓ¹‑normalised)
    # --------------------------------------------------------------
    starts = []

    # a) previously discovered best (if dimensions match)
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.shape[0] == n:
            prev = np.clip(prev, 0.0, None)
            prev_sum = prev.sum()
            if prev_sum > 0:
                starts.append(prev / prev_sum)
    except Exception:
        pass

    # b) square‑root‑difference (the classic low‑peak profile)
    a_sqrt = np.sqrt(np.arange(1, n + 1, dtype=np.float64)) - np.sqrt(
        np.arange(0, n, dtype=np.float64)
    )
    starts.append(a_sqrt / a_sqrt.sum())

    # c) exponential decays with several rates
    i = np.arange(n, dtype=np.float64)
    for lam in (0.001, 0.0025, 0.005, 0.01, 0.02):
        a_exp = np.exp(-lam * i)
        starts.append(a_exp / a_exp.sum())

    # d) power‑law differences (i+1)**p - i**p for various p
    for p in np.linspace(0.2, 0.9, 8):
        a_pow = (i + 1) ** p - i ** p
        starts.append(a_pow / a_pow.sum())

    # e) a few random Dirichlet draws (fully stochastic)
    rng = np.random.default_rng(seed)
    for _ in range(4):
        a_rand = rng.dirichlet(np.ones(n))
        starts.append(a_rand)

    # Choose the best of the seeds (by cheap internal score)
    best_seq = None
    best_score = np.inf
    for s in starts:
        sc = _score_from_seq(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()
    print(f"[Init] Best seed score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Simulated‑annealing / gradient‑guided local search
    # --------------------------------------------------------------
    a = best_seq.copy()
    curr_score = best_score

    # SA schedule parameters
    temp_start = 0.8
    temp_end = 1e-4
    step_start = 0.09            # move up to ~9 % of the source mass
    step_end = 0.001

    no_improve_cnt = 0
    IMPROVE_LIMIT = 4000          # after this many non‑improvements we jitter

    while time.time() < deadline:
        # compute convolution peak and gradient
        max_b, k_max = _conv_max_and_index(a)
        grad = _gradient_of_peak(a, k_max)

        # pick source (high gradient) and sink (low gradient) indices
        src_idx = int(np.argmax(grad))
        dst_idx = int(np.argmin(grad))
        if src_idx == dst_idx:
            # fallback to a random pair
            src_idx, dst_idx = rng.choice(n, 2, replace=False)

        # step size decays with time
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = temp_start * (1 - frac) + temp_end * frac
        step = step_start * (1 - frac) + step_end * frac

        # amount of mass to transfer
        delta = step * a[src_idx]
        if delta <= 0.0:
            delta = step * 1e-3   # very small fallback

        # propose new vector (mass transfer)
        a_prop = a.copy()
        a_prop[src_idx] -= delta
        a_prop[dst_idx] += delta

        # keep numerical sanity
        a_prop = np.clip(a_prop, 0.0, None)
        # ℓ¹‑sum stays 1 (up to floating error)
        total = a_prop.sum()
        if total > 0:
            a_prop /= total

        # evaluate the proposal (fast internal metric)
        prop_score = _score_from_seq(a_prop)

        # acceptance rule
        accept = False
        if prop_score < curr_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(prop_score - curr_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            curr_score = prop_score
            # record global best
            if prop_score < best_score:
                best_score = prop_score
                best_seq = a_prop.copy()
                print(f"[{time.time()-start_time:5.1f}s] New best = {best_score:.6f}")
                no_improve_cnt = 0
            else:
                no_improve_cnt += 1
        else:
            no_improve_cnt += 1

        # spontaneous jitter to keep dynamics alive
        if rng.random() < 0.001:
            i1, i2 = rng.choice(n, 2, replace=False)
            eps = 0.02 * a[i1] * (rng.random() - 0.5)
            a[i1] = max(0.0, a[i1] + eps)
            a[i2] = max(0.0, a[i2] - eps)
            a /= a.sum()

        # if we have been stuck for a while, re‑seed from a random start
        if no_improve_cnt >= IMPROVE_LIMIT:
            # take the current best, add a tiny random perturbation
            a = best_seq.copy()
            perturb = rng.normal(scale=0.01, size=n)
            a = np.clip(a + perturb, 0.0, None)
            a /= a.sum()
            curr_score = _score_from_seq(a)
            no_improve_cnt = 0

    # --------------------------------------------------------------
    # 4. LP‑based refinement (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0:
        # one or two refinement rounds are cheap
        refined = _lp_refine(best_seq, mix=0.5, rounds=2)
        refined_score = _score_from_seq(refined)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"[LP] Refined best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Scale to the [0,1000] range required by the evaluator
    # --------------------------------------------------------------
    final_seq = best_seq * 1000.0
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    # Final verification (optional but cheap)
    try:
        # the external evaluator will be called by the system; we only
        # compute a sanity check to avoid accidental overflow.
        from __main__ import evaluate_sequence  # noqa: F401
        test_score = evaluate_sequence(final_seq.tolist())
        print(f"[Done] Final reported score = {test_score:.6f}")
    except Exception:
        # in the competition harness evaluate_sequence is already available
        pass

    return final_seq.tolist()
