def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Simulated-annealing / gradient-guided local search for a sequence minimizing
    `evaluate_sequence`.  Starts from several seed shapes (Catalan, uniform,
    power‑law) and iteratively moves small amounts of mass from indices that
    contribute most to the current maximal self‑convolution towards indices with
    the smallest values.  The step size is gradually reduced and a simple
    temperature schedule permits occasional uphill moves.
    """
    import numpy as np, time, math

    np.random.seed(seed)
    total_time = budget_s
    safety = 2.0
    start_time = time.time()
    deadline = start_time + total_time - safety

    # Choose length (might be provided via kwargs else default)
    n = kwargs.get('length', 6402)
    # If length exceeds reasonable limit make sure not too large
    if n > 20000:
        n = 20000

    # Pre‑compute FFT size
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()
    
    def conv_full(x):
        """Return full linear convolution a*a as a 1‑D np.ndarray."""
        fx = np.fft.rfft(x, n=nfft)
        conv = np.fft.irfft(fx * fx, n=nfft)[:m]
        return conv

    def score_of(x):
        conv = conv_full(x)
        return 2.0 * n * conv.max()

    #--- generate a list of starting sequences ---------------------------------
    start_seqs = []

    # 1. Catalan (central‑binomial) coefficients
    def catalan_seq(k):
        a = np.empty(k, dtype=np.float64)
        a[0] = 1.0
        for i in range(1, k):
            a[i] = a[i-1] * (2.0 * i) * (2.0 * i - 1.0) / (4.0 * i * i)
        return a
    raw = catalan_seq(n)
    seq0 = raw / raw.sum()
    start_seqs.append(seq0)

    # 2. Uniform
    seq_uniform = np.full(n, 1.0/n, dtype=np.float64)
    start_seqs.append(seq_uniform)

    # 3‑5. Random power‑law candidates
    for _ in range(3):
        p = np.random.uniform(0.3, 0.8)     # exponent
        offset = np.random.uniform(0.0, 5.0)  # shift to avoid singularity
        idx = np.arange(n, dtype=np.float64) + 1.0 + offset
        raw = idx ** (-p)
        seq = raw / raw.sum()
        start_seqs.append(seq)

    # 6. Random beta‑distribution shape (α in (0.3, 2.0))
    for _ in range(2):
        alpha = np.random.uniform(0.3, 2.0)
        x = (np.arange(n) + 0.5) / n
        raw = (x**(alpha-1)) * ((1.0 - x)**(alpha-1))
        seq = raw / raw.sum()
        start_seqs.append(seq)

    best_seq = None
    best_sc = np.inf

    # Time allocation per start
    remaining_time = deadline - time.time()
    if remaining_time <= 0:
        # not enough time; just return first candidate
        return start_seqs[0].tolist()

    time_per_start = remaining_time / len(start_seqs)

    # Main loop over each initial sequence
    for init_idx, init_seq in enumerate(start_seqs):
        # restart timer for this run
        run_deadline = time.time() + time_per_start
        seq = init_seq.copy()
        conv = conv_full(seq)
        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = seq.copy()
            print(f"[init {init_idx}] score={cur_sc:.6f}")

        # local search parameters
        step_frac = 0.05
        no_improve = 0
        # simple exponential cooling temperature
        initial_temp = best_sc * 0.02
        while time.time() < run_deadline:
            # locate current peak of convolution
            k_max = int(conv.argmax())
            # contributions to the peak
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx_range = np.arange(lo, hi+1)
            contrib = seq[idx_range] * seq[k_max - idx_range]
            # source index: the one with largest contribution
            i_src = idx_range[int(contrib.argmax())]

            # destination index: global minimum of sequence (avoid being the source)
            i_dst = int(np.argmin(seq))
            if i_dst == i_src:
                # second smallest if needed
                sorted_idx = np.argsort(seq)
                i_dst = int(sorted_idx[1]) if n > 1 else i_src

            # randomise the amount transferred
            delta = step_frac * np.random.rand() * seq[i_src]
            if delta <= 0.0:
                continue

            new_seq = seq.copy()
            new_seq[i_src] -= delta
            new_seq[i_dst] += delta
            # (numerical drift) re‑normalise rarely
            # sum unchanged, but we protect against round‑off
            if (new_seq[i_src] < -1e-12) or (new_seq[i_dst] < 0):
                continue

            # evaluate
            new_conv = conv_full(new_seq)
            new_sc = 2.0 * n * new_conv.max()

            # acceptance test (simulated annealing)
            accept = False
            if new_sc < cur_sc - 1e-12:
                accept = True
            else:
                # temperature decreasing linearly
                elapsed = time.time() - start_time
                total_el = deadline - start_time
                temp = max(0.0, initial_temp * (1.0 - elapsed / total_el))
                if temp > 0.0 and np.random.rand() < math.exp(-(new_sc - cur_sc) / temp):
                    accept = True

            if accept:
                seq = new_seq
                conv = new_conv
                cur_sc = new_sc
                if new_sc < best_sc - 1e-12:
                    best_sc = new_sc
                    best_seq = new_seq.copy()
                    no_improve = 0
                    print(f"[improve step {init_idx}] t={int(time.time()-start_time)}s score={best_sc:.6f}")
                else:
                    # still accepted uphill move – do not reset counter
                    pass
            else:
                no_improve += 1

            # shrink step size if stuck
            if no_improve >= 3000:
                step_frac *= 0.5
                no_improve = 0
                if step_frac < 1e-6:
                    break

        # optionally tidy floating‑point noise
        if best_seq is not None:
            # force exact normalisation to avoid tiny drift
            best_seq = best_seq / best_seq.sum()

        # early exit if time almost exhausted
        if time.time() > deadline - 1.0:
            break

    print(f"[final] best_score={best_sc:.6f} length={n}")
    return best_seq.tolist()

import time
import math
import numpy as np


def _catalan_sequence(n: int) -> np.ndarray:
    """First n terms of binom(2k,k)/4**k (central‑binomial coefficients)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def _conv_full(seq: np.ndarray, nfft: int) -> np.ndarray:
    """Full linear convolution of seq with itself (real FFT version)."""
    fx = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)
    # keep only the first (2*len(seq)-1) entries
    m = 2 * len(seq) - 1
    return conv[:m]


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Simulated‑annealing / gradient‑guided local search for a non‑negative
    sequence that minimises `evaluate_sequence`.  Several seed shapes are
    generated (Catalan, uniform, random power‑law and beta‑like families) and
    each is locally refined.  In every iteration the index that contributes the
    most to the current maximal self‑convolution is identified and a small
    amount of mass is moved from that index to the globally smallest entry.
    A simple temperature schedule permits occasional uphill moves, the step
    size shrinks when the search stalls, and the best sequence seen so far is
    returned.
    """
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # General time handling
    # ----------------------------------------------------------------------
    start_time = time.time()
    safety_margin = 2.0                # keep a couple of seconds spare
    deadline = start_time + budget_s - safety_margin
    if deadline <= start_time:
        # not enough time – fall back to a trivial sequence
        return [1.0]

    # ----------------------------------------------------------------------
    # Length of the sequence (allow override via kwargs)
    # ----------------------------------------------------------------------
    n = int(kwargs.get("length", 6402))
    # Cap the length to keep the per‑iteration cost reasonable
    n = max(100, min(n, 20000))

    # FFT size (next power of two for the convolution length)
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()

    def conv_full(x):
        """Convenient wrapper that returns the full convolution."""
        return _conv_full(x, nfft)

    def score_of(x):
        """Score = 2·n·max(x*x) for a sequence with ℓ¹‑norm 1."""
        conv = conv_full(x)
        return 2.0 * n * conv.max()

    # ----------------------------------------------------------------------
    # Build a diverse set of initial candidates
    # ----------------------------------------------------------------------
    init_seqs = []

    # 1) Catalan (central‑binomial) coefficients
    raw = _catalan_sequence(n)
    init_seqs.append(raw / raw.sum())

    # 2) Uniform distribution
    init_seqs.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 3‑5) Random power‑law shapes a_i ∝ (i+1+offset)^(-p)
    for _ in range(3):
        p = np.random.uniform(0.3, 0.8)          # exponent
        offset = np.random.uniform(0.0, 5.0)     # tiny shift to avoid singularity
        idx = np.arange(n, dtype=np.float64) + 1.0 + offset
        a = idx ** (-p)
        init_seqs.append(a / a.sum())

    # 6‑7) Beta‑like shapes a_i ∝ x^{α-1} (1‑x)^{α-1}
    for _ in range(2):
        alpha = np.random.uniform(0.3, 2.0)
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
        # avoid all‑zero case when α is very small
        if a.sum() == 0:
            a = np.ones_like(a)
        init_seqs.append(a / a.sum())

    # ----------------------------------------------------------------------
    # Search parameters (shared across all starts)
    # ----------------------------------------------------------------------
    best_seq = None
    best_sc = np.inf
    target_score = 1.503                       # stop early if we reach it

    # Rough time split among the different initialisations
    remaining = deadline - time.time()
    if remaining <= 0:
        return init_seqs[0].tolist()
    time_per_start = remaining / len(init_seqs)

    for init_id, seq in enumerate(init_seqs):
        # ------------------------------------------------------------------
        # Local search for this initialisation
        # ------------------------------------------------------------------
        run_deadline = time.time() + time_per_start
        cur_seq = seq.copy()
        conv = conv_full(cur_seq)
        cur_sc = 2.0 * n * conv.max()

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = cur_seq.copy()
            print(f"[init {init_id}] score={cur_sc:.6f}")

        # Adaptive parameters
        step_frac = 0.10                # start with fairly large moves
        no_improve = 0
        # temperature for simulated annealing – 2 % of current best at start
        init_temp = max(best_sc, cur_sc) * 0.02

        while time.time() < run_deadline:
            # --------------------------------------------------------------
            # Identify the peak of the current convolution
            # --------------------------------------------------------------
            k_max = int(conv.argmax())
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx_range = np.arange(lo, hi + 1, dtype=np.int64)

            # contribution of each i to the peak (a_i * a_{k_max-i})
            contrib = cur_seq[idx_range] * cur_seq[k_max - idx_range]
            i_src = int(idx_range[int(contrib.argmax())])

            # --------------------------------------------------------------
            # Choose a destination (the smallest entry, never equal to src)
            # --------------------------------------------------------------
            i_dst = int(np.argmin(cur_seq))
            if i_dst == i_src:
                # second smallest if needed
                sorted_idx = np.argsort(cur_seq)
                i_dst = int(sorted_idx[1]) if n > 1 else i_src

            # --------------------------------------------------------------
            # Propose a move (mass transfer)
            # --------------------------------------------------------------
            delta = step_frac * np.random.rand() * cur_seq[i_src]
            if delta <= 0.0:
                continue

            new_seq = cur_seq.copy()
            new_seq[i_src] -= delta
            new_seq[i_dst] += delta

            # guard against tiny negative values caused by rounding
            if new_seq[i_src] < -1e-12 or new_seq[i_dst] < 0:
                continue

            # --------------------------------------------------------------
            # Evaluate the new candidate
            # --------------------------------------------------------------
            new_conv = conv_full(new_seq)
            new_sc = 2.0 * n * new_conv.max()

            # --------------------------------------------------------------
            # Acceptance test – deterministic improvement + SA uphill moves
            # --------------------------------------------------------------
            accept = False
            if new_sc < cur_sc - 1e-12:
                accept = True
            else:
                # linear cooling
                elapsed = time.time() - start_time
                total = deadline - start_time
                temp = max(0.0, init_temp * (1.0 - elapsed / total))
                if temp > 0.0 and np.random.rand() < math.exp(-(new_sc - cur_sc) / temp):
                    accept = True

            if accept:
                cur_seq = new_seq
                conv = new_conv
                cur_sc = new_sc
                if new_sc < best_sc - 1e-12:
                    best_sc = new_sc
                    best_seq = new_seq.copy()
                    no_improve = 0
                    print(f"[improve {init_id}] t={int(time.time() - start_time)}s  score={best_sc:.6f}")
                    if best_sc <= target_score:
                        print("[target reached]")
                        # normalise and exit immediately
                        best_seq = best_seq / best_seq.sum()
                        return best_seq.tolist()
                else:
                    # uphill but accepted – do not reset stall counter
                    pass
            else:
                no_improve += 1

            # --------------------------------------------------------------
            # Shrink step size when we get stuck
            # --------------------------------------------------------------
            if no_improve >= 3000:
                step_frac *= 0.5
                no_improve = 0
                if step_frac < 1e-6:
                    break

            # occasional tiny re‑normalisation to suppress floating‑point drift
            if (time.time() - start_time) % 30 < 0.01:
                cur_seq /= cur_seq.sum()

        # ------------------------------------------------------------------
        # End of this start – move on to the next one (if time permits)
        # ------------------------------------------------------------------
        # normalise the globally best sequence before the next round
        if best_seq is not None:
            best_seq = best_seq / best_seq.sum()

        if time.time() > deadline - 1.0:
            break

    # ----------------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = best_seq / best_seq.sum()   # ensure exact ℓ¹‑norm 1
    print(f"[final] best_score={best_sc:.6f}  length={n}")
    return best_seq.tolist()
