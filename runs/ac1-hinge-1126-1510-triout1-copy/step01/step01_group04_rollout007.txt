"""
Propose a step‑function whose normalized ℓ₁‑mass is 1.
The algorithm proceeds in three stages:

1. **Seed selection** – try a handful of power‑law seeds
   a[i] = (i+1+δ)⁻ᵖ (different lengths, exponents p and offsets δ)
   and keep the best one according to ``evaluate_sequence``.

2. **Active‑set projected gradient descent with momentum** –  
   at each iteration the convolution ``c = a*a`` is computed.  
   All indices whose value is within 0.1 % of the current maximum are
   treated as *active*.  The sub‑gradient of the active convolution
   entries is accumulated, projected onto the hyperplane Σa = 1,
   and a Nesterov‑style momentum step is taken.  Non‑negativity and the
   ℓ₁‑constraint are enforced after each step.  Occasionally a tiny
   random jitter is added to escape plateaus.

3. **Return** the best sequence seen within the time budget.
"""

import time
import numpy as np
import itertools
import random

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _powerlaw_sequence(length: int,
                       exponent: float = 0.5,
                       offset: float = 0.0) -> np.ndarray:
    """
    Return a non‑negative power‑law sequence a[i] = (i+1+offset)^(-exponent).
    The vector is *not* normalised; the caller may normalise it as needed.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    return np.clip(seq, 0.0, 1000.0)


def _project_to_simplex(x: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """
    Project a non‑negative vector onto the affine hyperplane Σx = target_sum.
    Since we never create negative entries, the operation reduces to a simple
    scaling.
    """
    s = x.sum()
    if s == 0.0:
        return x
    return x * (target_sum / s)


def _active_set_gradient(a: np.ndarray,
                         conv: np.ndarray,
                         rel_tol: float = 1e-3) -> np.ndarray:
    """
    Compute the summed sub‑gradient of all convolution entries that lie
    within ``rel_tol`` relative distance of the maximum.
    For a given active index k the sub‑gradient is
        g_j = 2 * a_{k‑j}  (if 0 ≤ k‑j < n)
    The result is projected later onto the sum‑zero hyperplane.
    """
    n = a.shape[0]
    max_val = conv.max()
    # consider all k with conv[k] >= max_val * (1 - rel_tol)
    thresh = max_val * (1.0 - rel_tol)
    active = np.where(conv >= thresh)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # j runs from lo..hi, i = k-j
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k - j
        grad[j] += 2.0 * a[i]
    return grad


def _evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:
        return float("inf")

# Main entry point
# ----------------------------------------------------------------------

                      budget_s: int = 1000,

    """

    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds

    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # 0) initialise from a globally stored solution if available
    # ------------------------------------------------------------------
    global height_sequence_1
    if "height_sequence_1" in globals():

        if init.size > 0 and np.all(np.isfinite(init)):

            best_seq = a0.copy()
            best_score = _evaluate(best_seq)
            print(f"Initial seed from height_sequence_1 : score={best_score:.6f}")
        else:
            best_seq = None
    else:
        best_seq = None

    # ------------------------------------------------------------------
    # 1) coarse grid search over length / exponent / offset

        random.shuffle(combos)            # random order – better distribution of time
        best_seq = None
        best_score = float("inf")

        print("=== coarse hyper‑parameter scan ===")
        for length, exp, off in combos:

                break

            seq = _powerlaw_sequence(length, exponent=exp, offset=off)
            seq = _project_to_simplex(seq, target_sum=1.0)   # fix ℓ₁‑norm = 1
            score = _evaluate(seq)

            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print("  ** new best **")

    # ------------------------------------------------------------------
    # 2) refined optimisation – active‑set projected gradient with momentum
    # ------------------------------------------------------------------
    a_cur = best_seq.copy()
    n = a_cur.shape[0]

    print("\n=== finished ===")
