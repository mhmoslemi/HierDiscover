"""
Propose a candidate sequence using an alternating linear‑programming
procedure followed by a short simulated‑annealing refinement.
The LP step solves, for a fixed sequence `f`, the problem
    maximize 1ᵀ g  subject to  g ≥ 0,  f * g ≤ M
where M = max(f*f) and `*` denotes discrete convolution.
The solution `g` concentrates mass where `f` is small, increasing the
total sum without raising the bound M.  By mixing `g` into `f` we
strictly reduce the value of the evaluation function.
After a few LP iterations a quick stochastic local search (random
multiplicative perturbations plus occasional gradient moves) cleans up
the remaining peaks.
The function runs for ≤ budget_s seconds and returns the best sequence
found (list of non‑negative floats).
"""

import time
import math
import numpy as np
import scipy.sparse as sp

# optional: import cvxpy, fallback to None if unavailable
try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def half_order_series(n: int) -> np.ndarray:
    """First n coefficients of a_i = C(2i,i)/4**i, normalized to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix C such that C @ g = f * g.
    Each diagonal d (0 ≤ d < n) equals the constant f[d].
    """
    n = f.shape[0]
    offsets = np.arange(n, dtype=np.int32)          # 0 … n‑1
    # each diagonal consists of n copies of the corresponding coefficient
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    C = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return C


def gradient_step(seq: np.ndarray, max_index: int, step: float) -> np.ndarray:
    """Single gradient step w.r.t. the maximal convolution entry."""
    n = len(seq)
    grad = np.zeros_like(seq)
    lo = max(0, max_index - (n - 1))
    hi = min(n - 1, max_index)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    rev = max_index - idx
    grad[idx] = 2.0 * seq[rev]
    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return seq
    return new_seq / s


def _lp_step(f: np.ndarray, M: float, solver: str = "ECOS") -> np.ndarray | None:
    """
    Solve the LP
        max 1ᵀ g
        s.t. g ≥ 0,
             C_f @ g ≤ M·1,
    where C_f is the convolution matrix of f.
    Returns the optimal g (unnormalised) or None on failure.
    """
    if cp is None:   # pragma: no cover
        return None
    n = f.shape[0]
    C = convolution_matrix(f)
    g = cp.Variable(n)
    constraints = [
        g >= 0,

    except Exception:   # pragma: no cover
        return None
    if g.value is None:
        return None
    g_opt = np.asarray(g.value).astype(np.float64).ravel()
    # numerical cleanup
    g_opt[g_opt < 0.0] = 0.0

    # ------------------------------------------------------------------

        f_new = (1.0 - t) * f + t * g_norm

        conv_new = np.convolve(f_new, f_new)
        M_new = conv_new.max()

            best_score = score_new
            best_seq = f_new.copy()
            print(f"[lp {lp_iterations:02d}] New best score: {best_score:.6f}")

        # we are a bit tolerant to allow occasional worsening that may help later
        if M_new <= M * 0.999 or score_new < curr_score:
            f = f_new
            M = M_new
            curr_score = score_new
            # print status
            print(f"[lp {lp_iterations:02d}] Accepted new f (M={M:.8e}, score={curr_score:.6f})")
        else:
            # reject – keep the old f
            print(f"[lp {lp_iterations:02d}] Rejected (M unchanged)")

    # ------------------------------------------------------------------
    # 3) final stochastic refinement (simulated annealing + gradients)
    # ------------------------------------------------------------------
    # start from the best sequence found so far
    seq = best_seq.copy()
    curr_score = best_score
    n = len(seq)

    # annealing schedule parameters
    total_runtime = deadline - time.time()
    if total_runtime <= 0:
        return best_seq.tolist()

    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # linear progress for temperature
        elapsed = time.time() - start_time
        progress = min(1.0, elapsed / budget_s)
        temp = 0.5 * (1e-5 / 0.5) ** progress   # geometric decay from 0.5 to 1e‑5

        else:

            curr_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[anneal {iteration:06d}] New best (random) score: {best_score:.6f}")

            max_val = conv.max()
            max_idxs = np.where(conv == max_val)[0]
            k = np.random.choice(max_idxs)

            cand_score = evaluate_sequence(cand.tolist())

                best_seq = cand.copy()
                seq = cand
                curr_score = cand_score
                print(f"[anneal {iteration:06d}] Gradient improvement score: {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return best sequence found
    # ----------------------------------------------------------------------
    print(f"Finished. Best score = {best_score:.6f}")
    return best_seq.tolist()

"""
Propose a candidate sequence using an alternating linear‑programming
procedure followed by a short simulated‑annealing refinement.

The LP step solves, for a fixed sequence `f`,
    maximise   1ᵀ g
    subject to g ≥ 0,
               f * g ≤ M,
where * denotes discrete convolution and M = max(f*f).
The optimiser `g` puts mass where `f` is small, increasing the total
mass without raising the bound M.  By mixing `g` into `f` the value of

    2·n·max(conv(seq,seq)) / (sum(seq)²)

strictly decreases.  After a handful of LP iterations a quick stochastic
search (random multiplicative perturbations + occasional gradient moves)
flattens the remaining peaks.

The function respects the global time budget (default 1000 s) and
returns the best sequence found (list of non‑negative floats).
"""

import time
import math
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Optional import of CVXPY – used for the linear programmes.
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
except Exception:   # pragma: no cover
    cp = None


def half_order_series(n: int) -> np.ndarray:
    """First n coefficients of a_i = C(2i,i)/4**i, normalised to sum = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def convolution_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n matrix C such that C @ g = f * g.
    Each diagonal d (0 ≤ d < n) consists of the constant f[d].
    """
    n = f.shape[0]
    offsets = np.arange(n, dtype=np.int32)          # 0 … n‑1
    # every diagonal is just a constant vector equal to the corresponding entry of f
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    C = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return C


def gradient_step(seq: np.ndarray, max_index: int, step: float) -> np.ndarray:
    """
    Perform a gradient‑descent step that reduces the convolution value at
    `max_index`.  The gradient of the entry c_k = Σ a_i a_{k-i} w.r.t. a_i
    equals 2·a_{k-i} (when the index is in range).
    """
    n = len(seq)
    grad = np.zeros_like(seq)

    lo = max(0, max_index - (n - 1))
    hi = min(n - 1, max_index)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    rev = max_index - idx
    grad[idx] = 2.0 * seq[rev]

    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s == 0.0:
        return seq
    return new_seq / s


def _lp_step(f: np.ndarray, M: float, solver: str = "ECOS") -> np.ndarray | None:
    """
    Solve the linear programme
        maximise   1ᵀ g
        subject to g ≥ 0,
                   C_f @ g ≤ M·1,
    where C_f is the convolution matrix of `f`.
    Returns the (unnormalised) optimiser `g` or None on failure.
    """
    if cp is None:   # pragma: no cover
        return None

    n = f.shape[0]
    C = convolution_matrix(f)

    g = cp.Variable(n)
    constraints = [
        g >= 0,
        C @ g <= M * np.ones(2 * n - 1, dtype=np.float64)
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver, warm_start=True, verbose=False, max_iters=5000)

        return None

        return None
    g_opt = np.asarray(g.value).astype(np.float64).ravel()
    g_opt[g_opt < 0.0] = 0.0
    return g_opt
