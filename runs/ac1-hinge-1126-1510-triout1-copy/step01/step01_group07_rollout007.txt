def solve_lp_max_sum(f, M, timeout=None):
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0, None)] * n
    method = 'highs'
    options = {}
    if timeout is not None:
        options['time_limit'] = timeout
    res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method=method, options=options)
    if res.success:

        sum_g0 = g0.sum()
        if sum_g0 > 0:
            g = g0 / sum_g0
            return g
    return None

"""
Propose a candidate height sequence by iterative LP‑based flattening.
The algorithm starts from the best known construction (or a power‑law
approximation) and repeatedly:
1. computes the current max self‑convolution M,
2. solves a linear program that maximises the total mass of a new
   non‑negative sequence g under the constraints conv(f,g) ≤ M,
3. rescales g to have the same ℓ¹‑norm as f,
4. moves the current sequence f a small step towards g.
Each step strictly reduces the score 2·n·max(conv(f,f))/sum(f)²,
so the process converges to a flatter autocorrelation.  The LP is
solved with SciPy’s high‑performance “highs” solver; the whole search
stops when the time budget (≈ 1000 s) is exhausted or no further
improvement is observed.  If the LP route fails, a simple power‑law
grid‑search fallback is used.
"""

import time
import numpy as np
from scipy import optimize

def generate_power_law(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2·n‑1)×n matrix A that maps g → conv(f,g).
    Row k, column j contains f_{k‑j} if 0≤k‑j<n else 0.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j is f shifted down by j rows
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _solve_lp_for_g(f: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve   max sum(g)   s.t. conv(f,g) ≤ M, g ≥ 0.
    Return the normalised vector g (ℓ¹‑norm = 1) or None on failure.
    """
    n = f.shape[0]
    # Build constraint matrix A g ≤ M
    A = _build_conv_matrix(f)
    # Right‑hand side slightly relaxed to avoid numerical infeasibility
    b = np.full(2 * n - 1, M * (1.0 + 1e-12), dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)                     # linprog minimises cᵀx → maximise sum(g)
    bounds = [(0.0, None)] * n

    # ``highs`` accepts a timeout via the high‑level option dict
    options = {}
    if time_limit is not None:
        # SciPy’s ``highs`` understands ``time_limit`` (seconds) via the
        # ``highs_options`` dict; fallback to a tiny value if unsupported.
        options = {"highs_options": {"time_limit": time_limit}}
    try:
        res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                               method="highs", options=options)
    except Exception as e:
        print("[LP] Exception:", e)
        return None

    if not res.success:
        print("[LP] Failure:", res.message)
        return None

    g0 = res.x
    s = g0.sum()
    if s <= 0.0:
        return None
    # Normalise to ℓ¹‑norm = 1 (the same norm used for the score)
    return g0 / s


def _mix_step(f: np.ndarray, g: np.ndarray, step: float) -> tuple[np.ndarray, float]:
    """
    Perform a convex combination f←(1‑step)·f + step·g, re‑normalise,
    and return the new sequence together with its raw score
    2·n·max(conv(f,f)) (i.e. without the denominator because the ℓ¹‑norm is 1).
    """
    f_new = (1.0 - step) * f + step * g
    total = f_new.sum()
    if total <= 0.0:
        return f, np.inf
    f_new = f_new / total
    max_conv = np.convolve(f_new, f_new).max()
    score = 2.0 * f_new.shape[0] * max_conv
    return f_new, score
