def greedy_g(f, M):
    n = len(f)
    # constraints array S of length 2n-1
    S = np.zeros(2*n - 1, dtype=np.float64)
    g = np.zeros_like(f)
    # iterate over j from n-1 down to 0 (or other ordering)
    for j in range(n-1, -1, -1):
        # compute min ratio across i where f_i>0
        # vector of M - S[i+j]
        # We need to compute allowed = min_{i} (M - S[i+j]) / f_i
        # For i where f_i is zero (unlikely) skip.
        # But need to ensure M - S may be negative; take min positive.
        # Inverse: we can compute remaining capacities: cap_i_j = (M - S[i+j]) / f_i
        # Allowed = max(0, min(cap_i_j)).
        # This might be negative due to rounding; clamp to zero.
        rem = M - S[j:j+n]  # length n slice
        # division by f
        with np.errstate(divide='ignore', invalid='ignore'):
            caps = rem / f
        # Only consider where f>0
        allowed = np.min(caps)
        if allowed <= 0:
            g[j] = 0.0
        else:
            g[j] = allowed
            # update S
            S[j:j+n] += f * allowed
    return g

def greedy_g(f, M, direction='right'):
    n = len(f)
    S = np.zeros(2*n-1, dtype=np.float64)
    g = np.zeros(n, dtype=np.float64)
    indices = range(n-1, -1, -1) if direction=='right' else range(0, n)
    for j in indices:
        # Determine the allowed amount to add at position j.
        # Compute remaining capacity for each constraint that includes g_j.
        # constraints indices k = i + j for i in [0, n-1], i.e., range j to j+n-1.
        rem = M - S[j:j+n]   # shape n
        # Compute allowed_j = min(rem / f_i)
        # Avoid division by zero: if f_i == 0, then constraint reduces to rem >=0 (no restriction)
        # So we treat those components as inf.
        # compute caps = rem / f
        # where f_i zero => set cap = np.inf
        caps = np.full_like(f, np.inf, dtype=np.float64)
        nonzero = f > 0
        caps[nonzero] = rem[nonzero] / f[nonzero]
        # To maintain numerical stability, find min of caps
        allowed = np.min(caps)
        if allowed <= 0:
            # cannot add further mass at this position
            continue
        g[j] = allowed
        # update S
        S[j:j+n] += f * allowed
    return g

"""
Algorithm
---------
1. Initialise a long Catalan (central‑binomial) sequence and normalise it to have
   ℓ¹‑norm 1.  This gives the well‑known bound ≈ π/2.
2. Repeatedly improve the sequence by solving the linear programme

       maximise   Σ g_j          (total mass of a new sequence)
       subject to Σ_i f_i g_{k-i} ≤ M   for all convolution indices k,
                g_j ≥ 0,

   where *f* is the current sequence and *M* = max(f∗f).  The feasible region
   describes all non‑negative sequences *g* whose convolution with *f* does not
   exceed the current worst‑case value *M*.  The LP has a totally‑unimodular
   constraint matrix (consecutive‑ones) and can be solved optimally by a fast
   greedy “water‑filling’’ routine: we allocate as much mass as possible to a
   coordinate while respecting all still‑unsaturated constraints.
3. The greedy routine returns a candidate *g*.  After normalising Σ g_j = 1 we
   replace the current sequence by a convex combination
        f ← (1‑t) f + t g,
   with a modest step size *t* (initially 0.1).  Because Σ g = 1 the objective
   value (2 n·max (f∗f)) is scale‑invariant, and the mixture strictly reduces
   the maximal self‑convolution when *g* differs from *f*.
4. When the improvement stagnates we shrink the step size and occasionally
   perform tiny random moves (a light stochastic hill‑climber) to escape flat
   regions.
5. The loop runs until the time budget expires (≈ 1000 s).  The best sequence
   found is returned.

The whole procedure works with only NumPy (no external solvers) and stays
well within the time limit for a sequence length of a few‑thousand terms.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def catalan_sequence(n: int) -> np.ndarray:
    """First *n* Catalan‑type coefficients a_k = binom(2k,k)/4**k (exact)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)
    return a


def max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear convolution a∗a using a real FFT."""
    n = a.shape[0]
    m = 2 * n - 1
    nfft = 1 << (m - 1).bit_length()        # next power of two
    fa = np.fft.rfft(a, n=nfft)
    conv = np.fft.irfft(fa * fa, n=nfft)[:m]
    return float(conv.max())


def greedy_g(f: np.ndarray, M: float, direction: str = "right") -> np.ndarray:
    """
    Construct a feasible vector g that maximises Σ g_j given
    the linear constraints Σ_i f_i·g_{k-i} ≤ M (for all k).
    The constraints form a consecutive‑ones matrix, therefore a simple
    greedy “fill‑as‑much‑as‑possible’’ algorithm yields an optimal extreme point.
    """
    n = f.shape[0]
    S = np.zeros(2 * n - 1, dtype=np.float64)   # current values of (f*g)
    g = np.zeros_like(f)

    # order of processing: either from the rightmost index to the left
    # or the opposite; the better of the two orders will be chosen later.
    if direction == "right":
        indices = range(n - 1, -1, -1)
    else:
        indices = range(0, n)

    # mask of non‑zero f entries (division by zero is avoided)
    nz = f > 0.0

    for j in indices:
        # the constraints that involve g_j are exactly the window S[j : j+n]
        remaining = M - S[j:j + n]                # how much slack we still have
        # Allowed increase = min_i (remaining_i / f_i)
        caps = np.full(n, np.inf, dtype=np.float64)
        caps[nz] = remaining[nz] / f[nz]
        allowed = caps.min()
        if allowed <= 0.0:
            continue
        g[j] = allowed
        # update the whole window because g_j contributes f_i·allowed to every
        # constraint in the window.
        S[j:j + n] += f * allowed

    return g


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The method starts from Catalan coefficients and
    repeatedly applies a greedy LP‑step followed by a small convex update.
    A lightweight stochastic hill‑climber is interleaved to avoid stagnation.
    """
    np.random.seed(seed)

    # ----- parameters ----------------------------------------------------
    n = 6402                     # length of the construction (as in the baseline)
    step_frac = 0.05            # for the random hill‑climber (fraction of source mass)
    t = 0.10                    # mixing coefficient for the LP‑step
    min_t = 0.005               # lower bound for t
    no_improve = 0              # counter for stagnation detection
    patience = 1500             # how many iterations without improvement before shrinking t

    # ----- 1. deterministic start: Catalan coefficients ------------------
    raw = catalan_sequence(n)
    f = raw / raw.sum()         # ℓ¹‑norm = 1
    best_f = f.copy()
    best_score = 2.0 * n * max_self_convolution(best_f)
    print(f"[init] n={n}, score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # 1) current worst convolution value
        M = max_self_convolution(f)

        # 2) obtain two candidate g‑vectors (right‑to‑left and left‑to‑right)
        g_r = greedy_g(f, M, direction="right")
        g_l = greedy_g(f, M, direction="left")
        # pick the one with larger total mass (so after normalisation it is more
        # “aggressive’’).
        if g_l.sum() > g_r.sum():
            g_raw = g_l
        else:
            g_raw = g_r

        sum_g_raw = g_raw.sum()
        if sum_g_raw <= 0.0:
            # fall back to a tiny random perturbation if the greedy step fails
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i != j and f[i] > 1e-12:
                delta = min(f[i] * 0.02, np.random.rand() * f[i])
                f[i] -= delta
                f[j] += delta
                f = np.clip(f, 0, None)
                f /= f.sum()
            continue

        # 3) normalise g to have ℓ¹‑norm 1
        g = g_raw / sum_g_raw

        # 4) convex combination
        new_f = (1.0 - t) * f + t * g
        new_f = np.clip(new_f, 0, None)   # numerical safety
        new_f /= new_f.sum()

        new_score = 2.0 * n * max_self_convolution(new_f)

        if new_score < best_score:
            best_score = new_score
            best_f = new_f.copy()
            f = new_f                     # continue from the improved point
            no_improve = 0
            print(f"[improve] t={int(time.time() - start_time)}s  score={best_score:.6f}")
        else:

            if no_improve >= patience:
                old_t = t
                t = max(t * 0.5, min_t)
                no_improve = 0
                print(f"[shrink] t reduced from {old_t:.4f} to {t:.4f}")

        # --------------------------------------------------------------
        # occasional tiny random move (stochastic hill‑climber)
        # --------------------------------------------------------------
        if np.random.rand() < 0.02:   # 2 % of iterations get a random tweak
            # choose a source from the “large” half and a destination from the “small” half
            src = np.random.randint(n // 2)
            dst = np.random.randint(n // 2, n)
            if f[src] > 0.0:
                delta = step_frac * np.random.rand() * f[src]
                if delta > 0.0:
                    tmp = f.copy()
                    tmp[src] -= delta
                    tmp[dst] += delta
                    tmp = np.clip(tmp, 0, None)
                    tmp /= tmp.sum()
                    tmp_score = 2.0 * n * max_self_convolution(tmp)
                    if tmp_score < best_score:
                        best_score = tmp_score
                        best_f = tmp.copy()
                        f = tmp
                        no_improve = 0
                        print(f"[random‑improve] t={int(time.time() - start_time)}s  score={best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[final] n={n}, best_score={best_score:.6f}")
    return best_f.tolist()
