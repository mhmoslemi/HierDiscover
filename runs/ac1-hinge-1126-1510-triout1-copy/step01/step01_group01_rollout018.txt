def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence for `evaluate_sequence`.

The method starts from the classic Catalan (central‑binomial) coefficients,
normalised to ℓ¹‑norm 1, which already gives a value ≈ π/2 ≈ 1.5708.
It then performs an aggressive hill‑climbing / simulated‑annealing search
that repeatedly transfers a *large* fraction of mass from the currently
largest entry to the smallest entry (typically the far‑right tail).  This
reduces the dominant contribution of the first coefficient to the
self‑convolution and therefore lowers the maximal convolution value.
Only moves that improve the score are kept; occasional uphill moves are
accepted according to a temperature schedule to escape shallow plateaus.
The search stops when the time budget (≈ 1000 s) is exhausted or when the
step size becomes negligible.  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _catalan_sequence(n: int) -> np.ndarray:
    """First n Catalan‑type coefficients a_k = binom(2k,k)/4**k."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # recurrence a_k = a_{k‑1} * (2k)(2k‑1) / (4 k²) = a_{k‑1} * (1 - 1/(2k))
        a[k] = a[k - 1] * (1.0 - 1.0 / (2.0 * k))
    return a


def _max_self_convolution(x: np.ndarray) -> float:
    """Maximum entry of the linear convolution x*x using a real FFT."""
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return float(conv.max())


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Compute the value returned by `evaluate_sequence` for a sequence that
    already satisfies the ℓ¹‑norm = 1 constraint.
    """
    n = seq.shape[0]
    conv_max = _max_self_convolution(seq)
    return 2.0 * n * conv_max


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The routine starts from the Catalan construction
    and then repeatedly moves mass from the current largest coefficient to the
    current smallest one, accepting improving moves and occasionally accepting
    worse moves with a temperature schedule (simulated annealing).
    """
    rng = np.random.default_rng(seed)

    # ---- 1.  Deterministic start: Catalan coefficients -----------------
    n = 6402                     # length of the construction (the given one)
    raw = _catalan_sequence(n)
    seq = raw / raw.sum()        # normalise to ℓ¹‑norm 1

    cur_seq = seq.copy()
    cur_score = _score_from_sequence(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score

    print(f"[init] length={n}, score={cur_score:.6f}")

    # ---- 2.  Aggressive mass‑transfer hill‑climbing -------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # safety margin
    step_frac = 0.5                              # initial fraction of source mass to move
    temperature = cur_score * 0.05               # initial temperature for annealing
    cooling = 0.9995
    no_improve = 0
    max_no_improve = 5000

    while time.time() < deadline:
        # indices of current extreme entries
        i_max = int(np.argmax(cur_seq))          # source (largest)
        i_min = int(np.argmin(cur_seq))          # destination (smallest)

        # random amount to move – up to step_frac of the source entry
        delta = step_frac * rng.random() * cur_seq[i_max]
        if delta <= 0.0:
            continue

        # propose new sequence by moving delta from i_max to i_min
        new_seq = cur_seq.copy()
        new_seq[i_max] -= delta
        new_seq[i_min] += delta

        # renormalise to exactly sum = 1 (prevents drift caused by FP noise)
        s = new_seq.sum()
        if s != 0.0:
            new_seq /= s

        # evaluate the new candidate (fast: only max convolution needed)
        new_score = _score_from_sequence(new_seq)

        # acceptance test (simulated annealing)
        improve = new_score < cur_score
        if improve or rng.random() < np.exp((cur_score - new_score) / max(temperature, 1e-12)):
            cur_seq = new_seq
            cur_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        # keep track of the best ever found
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            elapsed = int(time.time() - start_time)
            print(f"[improve] t={elapsed}s  score={best_score:.6f}")

        # adapt the step size – shrink when we have been stuck for a while
        if no_improve >= max_no_improve:
            step_frac *= 0.5
            temperature *= 0.5
            no_improve = 0
            if step_frac < 1e-6:
                break

        # slowly cool down the temperature
        temperature *= cooling

    print(f"[final] length={n}, best_score={best_score:.6f}")
    # Return a plain Python list as required by the evaluation harness
    return best_seq.tolist()
