def proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]

    w = np.maximum(v - theta, 0)
    return w

def project_onto_simplex(v):
    """Project onto probability simplex (sum to 1, >=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv/np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all negative; project to uniform? but should not happen.
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, math

    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0

    # candidate n values
    n_list = [500, 800, 1000, 1500, 2000]
    best_score = float('inf')
    best_a = None
    best_n = None

    for n in n_list:
        fft_len = next_pow2(2*n-1)
        # compute log(1-p) efficiently, p = c/n
        c_vals = np.linspace(1.5, 2.5, 300)  # fine grid
        for c in c_vals:
            p = c / n
            # ensure p < 1
            if p <= 0 or p >= 1:
                continue
            # compute a_i
            i = np.arange(n)
            # use exp(i*log1p(-p))
            log1mp = math.log1p(-p)
            a_raw = np.exp(i * log1mp)   # (1-p)**i
            # normalize
            sum_raw = a_raw.sum()
            a = a_raw / sum_raw
            # compute conv
            conv = conv_fft(a, fft_len, n)
            max_conv = conv.max()
            score = 2.0 * n * max_conv
            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
                # print progress
                print(f"[grid] n={n}, c={c:.5f}, score={score:.6f}")

                # check time
                if time.time() > deadline:
                    break
        if time.time() > deadline:
            break

    # optional further local refinement via hill climb
    if best_a is not None and time.time() < deadline:
        # do hill climb with small number of steps
        fft_len = next_pow2(2*best_n-1)
        best_score = fast_score(best_a, fft_len, best_n)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len, best_n,
                                        deadline,
                                        max_iter=5000)
        print(f"[refine] after hill climb score={best_score:.6f}")

    # final evaluation
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")
    return final_seq

range of n (500‑2000) and c (≈1.5‑2.5) to locate the best member of this
family.  The best candidate is then polished with a short greedy
hill‑climb that transfers a small fraction of mass between two positions
while keeping the vector on the probability simplex.
All heavy lifting (convolution, max‑convolution, projection) is done with
NumPy/FFT, so thousands of candidates can be examined well within the
1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself (length 2*n‑1) using FFT.
    a is assumed real and length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # keep only the first 2*n‑1 real entries
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a sequence that already sums to 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # largest i satisfying the condition
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple greedy mass‑transfer hill‑climb.
    Moves a random fraction (up to 5 %) of mass from i to j, keeps the sum
    exactly 1 via the projection step, and accepts the move only if the
    objective improves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a fraction of the mass at i toward j
        delta = best[i] * rng.random() * 0.05
        if delta == 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project to keep a valid probability vector
        cand = project_onto_simplex(cand)

        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
            # keep improvement
        # otherwise reject

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence found (non‑negative, sum > 0).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for final eval

    # ------------------------------------------------------------------
    # 1) Grid search over the exponential family a_i ∝ (1‑p)^i
    # ------------------------------------------------------------------
    # Candidate lengths – a trade‑off between resolution and speed
    n_candidates = [500, 800, 1000, 1500, 2000]
    best_score = float('inf')
    best_a = None
    best_n = None

    # Grid for the constant c (p = c / n).  The theoretical optimum
    # is around c ≈ target·e / 2 ≈ 2.04, we search a broader interval.
    c_vals = np.linspace(1.5, 2.5, 300)   # 300 points → good accuracy

    for n in n_candidates:
        fft_len = next_pow2(2 * n - 1)
        # Pre‑compute the index vector
        idx = np.arange(n, dtype=np.float64)

        for c in c_vals:
            if time.time() >= deadline:
                break
            p = c / n
            if not (0.0 < p < 1.0):
                continue

            # a_i = (1‑p)^i  (unnormalised)
            log1mp = math.log1p(-p)            # more stable than np.log(1-p)
            a_raw = np.exp(idx * log1mp)

            # Normalise to the simplex
            a = a_raw / a_raw.sum()

            # Fast objective (no scaling needed because sum(a)=1)
            score = fast_score(a, fft_len, n)

            if score < best_score:
                best_score = score
                best_a = a.copy()
                best_n = n
                print(f"[grid] n={n}, c={c:.5f}, p={p:.6f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional local refinement with greedy hill‑climb
    # ------------------------------------------------------------------
    if best_a is not None and time.time() < deadline:
        fft_len = next_pow2(2 * best_n - 1)
        # recompute exact score (might differ slightly from fast_score)
        best_score = fast_score(best_a, fft_len, best_n)
        # give the hill‑climb the remaining time (or up to 0.8 s)
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        best_n,
                                        hc_deadline,
                                        max_iter=8000)
        print(f"[refine] after hill climb score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")

    return final_seq
