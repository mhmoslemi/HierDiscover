"""
Propose a candidate step‑function using an alternating linear‑programming
refinement.  The idea is:

1. Start from a reasonable shape (the previously best sequence or a power‑law
   with exponent ≈0.5) and normalise it to L1‑norm 1.
2. For the current shape f compute M = max(conv(f,f)).  By construction
   conv(f,f) ≤ M for every shift.
3. Solve a linear program that maximises Σ g_j subject to the *linear*
   constraints   (f * g)_k ≤ M   for all k and   g ≥ 0.
   Because the convolution is linear in g this LP is cheap to formulate.
   The optimum g tries to put as much mass as possible into “slack’’ spots
   of the convolution.
4. Mix the new shape g into the old one (with a line‑search on the mixing
   coefficient) and keep the mix if it improves the evaluation score.
5. Repeat steps 2‑4 until the time budget runs out or no further improvement
   is observed.

If the LP approach fails to beat the target, a cheap power‑law grid search
is used as a safety net.  The returned sequence is a plain Python list of
non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix, csr_matrix

# ----------------------------------------------------------------------
# Helper: generate a power‑law sequence and normalise it to sum 1
def generate_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq

# ----------------------------------------------------------------------
# Helper: build a sparse matrix that represents convolution with a fixed
# vector f.  For a given f (length n) the matrix A (shape (2n‑1, n))
# satisfies   A @ g = np.convolve(f, g)
def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    n = f.shape[0]
    # row indices: each column i contributes to rows i … i+n‑1
    row = (np.arange(n, dtype=np.int32).reshape(-1, 1) +
           np.arange(n, dtype=np.int32)).ravel()
    col = np.repeat(np.arange(n, dtype=np.int32), n)
    data = np.tile(f, n)
    return coo_matrix((data, (row, col)),
                      shape=(2 * n - 1, n), dtype=np.float64).tocsr()

# ----------------------------------------------------------------------
# Helper: one LP iteration – maximise Σg under f * g ≤ M
def solve_for_g(f: np.ndarray, M: float, time_limit: float = 15.0) -> np.ndarray | None:
    """
    Returns an optimal g (as a NumPy array) or None if the LP fails.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)                # (2n‑1) × n, sparse
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog solves a *minimisation* problem → use -c to maximise Σg
    c = -np.ones(n, dtype=np.float64)

    # HiGHS (default) respects a "time_limit" entry in options (seconds)
    # If the SciPy version does not support it the solver will just ignore.
    opts = {"presolve": True, "time_limit": time_limit}

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0, None)] * n,
                  method="highs",
                  options=opts)

    if not res.success:
        return None
    g = res.x
    return g

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Uses an alternating LP refinement for up to a few seconds, then falls back
    to a fine power‑law grid if necessary.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # tiny safety margin

    # --------------------------------------------------------------
    # Determine the length.  Prefer the previously discovered length,
    # otherwise use a medium size (≈2000) that keeps the LP manageable.
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]  # noqa: F821
        if n <= 0:
            n = 2000
    except Exception:
        n = 2000

    # --------------------------------------------------------------
    # Initialise the shape.
    try:
        # height_sequence_1 may already be a good start.
        f = np.array(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]  # noqa: F821
        if f.size != n:
            # If the stored sequence length differs, truncate or pad.
            if f.size > n:
                f = f[:n]
            else:
                pad = np.full(n - f.size, 0.0)
                f = np.concatenate([f, pad])
    except Exception:
        # Fallback: a power‑law close to the previous best exponent.
        f = generate_power_law(n, p=0.505)

    # Normalise to L1‑norm 1 (scaling does not affect the score).
    if f.sum() == 0:
        f = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        f = f / f.sum()

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main alternating LP loop.
    iteration = 0
    while time.time() < deadline and iteration < 12:
        iteration += 1
        # Compute current convolution maximum.
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Solve the LP to get a candidate g.
        lp_time = min(15.0, max(0.5, (deadline - time.time()) / 4.0))
        g = solve_for_g(f, M, time_limit=lp_time)
        if g is None:
            print("[lp] LP failed – stopping LP phase.")
            break

        # Normalise g to have L1‑norm 1 (shape only; magnitude is absorbed by mixing).
        if g.sum() == 0:
            print("[lp] Solver returned all‑zero g – aborting.")
            break
        g = g / g.sum()

        # Try mixing f and g with a simple back‑tracking line search on t.
        improved = False
        t = 0.5
        while t > 1e-4 and time.time() < deadline:
            f_new = (1.0 - t) * f + t * g
            # Re‑normalise (again scale‑invariant, but keeps numbers well‑behaved)
            f_new = f_new / f_new.sum()
            score_new = evaluate_sequence(f_new.tolist())
            if score_new < best_score:
                best_score = score_new
                best_seq = f_new.copy()
                f = f_new
                print(f"[iter {iteration}] t={t:.4f}, improved score={best_score:.6f}")
                improved = True
                break
            t *= 0.5

        if not improved:
            # No improvement with this g – stop early.
            print(f"[iter {iteration}] No improvement found, exiting LP loop.")
            break

    # --------------------------------------------------------------
    # If we did not reach the target, perform a cheap power‑law sweep
    # as a safety net (this is very fast).
    if best_score > 1.503 and time.time() < deadline:
        print("[fallback] Starting fine power‑law grid search.")
        # Try a finer grid around the best exponent found earlier (≈0.505).
        p_grid = np.linspace(0.45, 0.60, 31)
        for p in p_grid:
            if time.time() >= deadline:
                break
            seq = generate_power_law(n, p)
            sc = evaluate_sequence(seq.tolist())
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print(f"[fallback] p={p:.5f}, new best score={best_score:.6f}")

    # Final sanity check: guarantee non‑negative finite result.
    final_seq = np.maximum(best_seq, 0.0).tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] length={len(final_seq)}, final score={final_score:.6f}")

    return final_seq
