import numpy as np
import time

def _max_self_convolution(x):
    n = x.shape[0]
    m = 2*n - 1
    nfft = 1 << ((m - 1).bit_length())
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return conv.max()

def _powerlaw_sequence(n, alpha):
    i = np.arange(1, n+1, dtype=np.float64)
    seq = i ** (-alpha)
    seq /= seq.sum()
    return seq

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a decreasing power‑law step function a_i ∝ (i+1)^{-α}

    golden‑section line search for the exponent α, evaluates each
    candidate with an FFT‑based convolution, and finally returns the
    best sequence found within the time budget.
    """
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a safety margin

    # length of the step function; keep the same as the reference solution
    n = kwargs.get('n', 6402)

    # --------------------------------------------------------------
    left, right = 0.30, 0.80   # reasonable interval for a decreasing shape
    # evaluate the boundaries to make sure the optimum lies inside

    def evaluate_alpha(alpha):
        seq = _powerlaw_sequence(n, alpha)
        conv_max = _max_self_convolution(seq)
        score = 2.0 * n * conv_max
        return score, seq

    # initial evaluations
    fa, seq_a = evaluate_alpha(left)
    fb, seq_b = evaluate_alpha(right)
    # ensure we have a proper bracketing; if not, keep expanding

    # First interior points

    b = right - resphi * (right - left)
    fa_mid, seq_a_mid = evaluate_alpha(a)
    fb_mid, seq_b_mid = evaluate_alpha(b)

    best_score = min(fa, fb, fa_mid, fb_mid)
    best_seq = None
    if best_score == fa:
        best_seq = seq_a
        best_alpha = left
    elif best_score == fb:
        best_seq = seq_b
        best_alpha = right
    elif best_score == fa_mid:
        best_seq = seq_a_mid
        best_alpha = a
    else:
        best_seq = seq_b_mid
        best_alpha = b

            # the minimum lies in [left, b]
            right = b

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def _max_self_convolution(x: np.ndarray) -> float:
    """
    Return the maximum entry of the linear convolution x*x using FFT.
    x must be a one‑dimensional real array.
    """
    n = x.shape[0]
    m = 2 * n - 1
    nfft = 1 << ((m - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=nfft)
    conv = np.fft.irfft(fx * fx, n=nfft)[:m]
    return float(conv.max())


def _powerlaw_sequence(n: int, alpha: float) -> np.ndarray:
    """
    Return a decreasing sequence a_i ∝ (i+1)^{-alpha}, normalised to ℓ¹‑norm 1.
    """
    i = np.arange(1, n + 1, dtype=np.float64)   # 1‑based index to avoid division by zero
    seq = i ** (-alpha)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        # fallback – should never happen for alpha>0
        seq = np.full_like(seq, 1.0 / n)
    else:
        seq = seq / seq_sum
    return seq


def _score_for_alpha(alpha: float, n: int) -> tuple[float, np.ndarray]:
    """
    Evaluate a power‑law candidate.
    Returns (score, normalised sequence) where
        score = 2 * n * max(a * a)   (equivalent to evaluate_sequence for ℓ¹‑norm 1)
    """
    seq = _powerlaw_sequence(n, alpha)
    conv_max = _max_self_convolution(seq)
    score = 2.0 * n * conv_max
    return score, seq

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative decreasing step function that minimises
    `evaluate_sequence`.  The search is performed on the one‑parameter
    family a_i ∝ (i+1)^{-α}.  A golden‑section line search finds the best
    exponent α, after which a lightweight hill‑climbing fine‑tunes the
    coefficients while keeping the monotone shape.  The best sequence
    encountered before the time budget expires is returned.
    """
    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin so the caller never sees a timeout
    deadline = start_time + budget_s - 0.5

    # Length of the step function – the reference solution used 6402.
    n = int(kwargs.get("n", 6402))

    # ------------------------------------------------------------------
    # 1) Bracket the optimum exponent α and run a golden‑section search.
    # ------------------------------------------------------------------
    left, right = 0.30, 0.80          # a reasonable interval for a decreasing shape
    # evaluate the interval ends (mostly for debugging)
    left_score, left_seq = _score_for_alpha(left, n)
    right_score, right_seq = _score_for_alpha(right, n)

    # golden‑section constants
    phi = (1.0 + 5.0 ** 0.5) / 2.0          # ≈1.618
    resphi = 2.0 - phi                     # ≈0.382

    # initial interior points
    a = left + resphi * (right - left)
    b = right - resphi * (right - left)
    fa, seq_a = _score_for_alpha(a, n)
    fb, seq_b = _score_for_alpha(b, n)

    # keep track of the best candidate seen so far
    best_score = min(left_score, right_score, fa, fb)
    if best_score == left_score:
        best_seq, best_alpha = left_seq, left
    elif best_score == right_score:
        best_seq, best_alpha = right_seq, right
    elif best_score == fa:
        best_seq, best_alpha = seq_a, a
    else:
        best_seq, best_alpha = seq_b, b

    # main optimisation loop
    while time.time() < deadline and (right - left) > 1e-6:
        if fa < fb:
            # the minimum lies in [left, b]
            right, fb, seq_b = b, fa, seq_a
            b = a
            fb = fa
            a = left + resphi * (right - left)
            fa, seq_a = _score_for_alpha(a, n)
        else:
            # the minimum lies in [a, right]
            left, fa, seq_a = a, fb, seq_b
            a = b
            fa = fb
            b = right - resphi * (right - left)
            fb, seq_b = _score_for_alpha(b, n)

        # update global best
        for s, seq, al in ((fa, seq_a, a), (fb, seq_b, b),
                           (left_score, left_seq, left),
                           (right_score, right_seq, right)):
            if s < best_score:
                best_score, best_seq, best_alpha = s, seq, al

        # recompute the scores at the interval ends only when they change
        left_score, left_seq = (fa, seq_a) if left == a else left_score, left_seq
        right_score, right_seq = (fb, seq_b) if right == b else right_score, right_seq

    print(f"[init] length={n}, α≈{best_alpha:.6f}, score≈{best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional tiny local refinement – move a small amount of mass.
    # ------------------------------------------------------------------
    # This stage respects the non‑increasing property of the sequence.
    step_frac = 0.02
    no_improve = 0
    while time.time() < deadline:
        # pick a source from the first half (larger values) and a destination
        # from the second half (smaller values)
        i = np.random.randint(n // 2)
        j = np.random.randint(n // 2, n)
        delta = step_frac * np.random.rand() * best_seq[i]
        if delta <= 0.0:
            continue

        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # preserve monotonicity: a_i ≥ a_{i+1}
        if new_seq[i] < new_seq[i + 1]:
            continue
        if j > 0 and new_seq[j] > new_seq[j - 1]:
            continue

        # renormalise (round‑off may have changed the sum)
        new_seq /= new_seq.sum()
        conv_max = _max_self_convolution(new_seq)
        score = 2.0 * n * conv_max

        if score < best_score:
            best_score, best_seq = score, new_seq
            no_improve = 0
            print(f"[improve] t={int(time.time() - start_time)}s α≈{best_alpha:.6f} "
                  f"score={best_score:.6f}")
        else:
            no_improve += 1

        # shrink step size if stuck for a while
        if no_improve >= 2000:
            step_frac *= 0.5
            no_improve = 0
            if step_frac < 1e-6:
                break

    # ------------------------------------------------------------------
    # 3) Return the best sequence found.
    # ------------------------------------------------------------------
    final_seq = best_seq.tolist()
    print(f"[final] length={n}, α≈{best_alpha:.6f}, best_score={best_score:.6f}")
    return final_seq
