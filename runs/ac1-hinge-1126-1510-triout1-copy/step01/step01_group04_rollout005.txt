"""
Propose a low‑score step‑function.

The algorithm has three stages:

1. **Broad parametric sweep** – generate many candidates from two
   simple families
       * power‑law   a[i] = (i+1+δ)^(-α)
       * product‑law a[i] = ((i+1+δ)*(n‑i+δ))^(-α)
   for a handful of lengths, exponents α and offsets δ.  Keep the best
   (lowest value of the evaluation function).

2. **Accelerated projected‑gradient descent** – start from the best
   candidate and minimise  max(conv(a,a)) while keeping ∑a fixed.
   The update uses an Adam‑style momentum scheme, the gradient being
   the sub‑gradient of the active convolution entry(s).  After each
   step the vector is projected onto the simplex (non‑negative, same
   ℓ₁‑norm).

3. **Random‑pair annealing** – when the gradient loop stalls, repeatedly
   move a tiny amount of mass from a random index to another random
   index; accept the move if it improves the score (or with a tiny
   probability to escape plateaus).

All work is bounded by the supplied time budget (default 1000 s).  The
returned sequence is a plain Python list of non‑negative floats ready
for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _initial_power(n: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    Power‑law family: a[i] = (i + 1 + offset)^{-exponent}.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    return np.clip(seq, 0.0, 1000.0)


def _initial_product(n: int, exponent: float = 0.5, offset: float = 0.0) -> np.ndarray:
    """
    Product‑law family: a[i] = ((i+1+offset)*(n-i+offset))^{-exponent}.
    The +1 moves the range away from zero to avoid singularities.
    """
    i = np.arange(n, dtype=np.float64)
    left = i + 1.0 + offset
    right = (n - i) + offset
    seq = np.power(left * right, -exponent)
    return np.clip(seq, 0.0, 1000.0)


def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective (identical to evaluate_sequence
    for valid non‑negative vectors that stay inside the [0,1000] clamp).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    total = a.sum()
    if total < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (total * total)


def _project_onto_simplex(x: np.ndarray, total: float) -> np.ndarray:
    """
    Scale the non‑negative vector so that its ℓ₁‑norm equals *total*.
    The vector is first clipped to non‑negative, then uniformly scaled.
    """
    x = np.maximum(x, 0.0)
    s = x.sum()
    if s == 0.0:
        return x
    return x * (total / s)


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of the maximal entry of the autocorrelation
    (a*a).  If several entries share the maximal value, their sub‑gradients
    are summed.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_val = conv.max()
    # choose a tiny tolerance to capture a few active indices
    tol = max_val * 1e-12 + 1e-12
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k - j
        # derivative of conv[k] w.r.t. a[j] is 2*a[i]
        grad[j] += 2.0 * a[i]
    return grad

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative height sequence) with a
    minimal value of ``evaluate_sequence``.  The routine first performs
    a coarse parametric search, then refines the best candidate with an
    Adam‑style projected gradient descent and finishes with a short
    random‑pair annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse parametric sweep
    # ------------------------------------------------------------------
    best_seq = None

    beta2 = 0.999

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    no_improve_cnt = 0

    while time.time() < deadline:

        # project the gradient onto the hyper‑plane Σa = total_mass
        grad -= grad.mean()

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # diminishing learning rate
        lr = base_lr / (1.0 + iteration * 1e-4)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new, total_mass)

        # evaluate
        cur_score = _score(a_new)

            best_seq = a_new.copy()
            no_improve_cnt = 0
            print(f"iter {iteration:6d}: improved score = {best_score:.6f}")
        else:
            no_improve_cnt += 1

        # accept a new point if it does not worsen the objective dramatically
        if cur_score <= best_score * 1.001:
            a = a_new
        else:

            if np.random.rand() < 0.02:
                a = best_seq.copy()

        # occasional jitter to help escape flat regions
        if iteration % 500 == 0:

            a = np.maximum(a + jitter, 0.0)
            a = _project_onto_simplex(a, total_mass)

        # stop early if we have not improved for a long stretch
        if no_improve_cnt > 5000:
            print("no improvement for many iterations – terminating Phase 2")
            break

    print(f"Phase 2 finished – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Random‑pair annealing (optional, only if we still have time)
    # ------------------------------------------------------------------
    print("=== Phase 3: random‑pair annealing ===")
    a = best_seq.copy()                              # start from the best so far
    total_mass = a.sum()
    n = a.shape[0]

    # annealing schedule parameters
    epsilon = total_mass * 1e-4                       # typical mass to move
    temp_start = 0.02
    temp_end = 0.0005
    anneal_iters = 0

    while time.time() < deadline:
        anneal_iters += 1

        # linear temperature schedule
        progress = (time.time() - start_time) / (deadline - start_time)
        temperature = temp_start * (1.0 - progress) + temp_end * progress

        if a[i] <= 0.0:
