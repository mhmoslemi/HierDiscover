def project_simplex(v, s=1.0):
    """Project onto simplex {x >=0, sum x = s}."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _optimize(a, deadline, alpha=10.0, lr=0.2):
    n = len(a)
    total = a.sum()
    if total == 0:
        a = np.ones_like(a) / n
    else:
        a = a / total
    L = 1
    while L < 2*n - 1:
        L <<= 1
    best_a = a.copy()
    best_score = 2.0 * n * np.max(np.fft.irfft(np.fft.rfft(a, n=L) * np.fft.rfft(a, n=L), n=L)[:2*n-1])
    # Adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    iteration = 0
    step_lr = lr
    while time.time() < deadline:
        iteration += 1
        # compute convolution via FFT
        A = np.fft.rfft(a, n=L)
        b_full = np.fft.irfft(A * A, n=L)[:2*n-1]
        max_b = b_full.max()
        # compute softmax weights
        exp_arg = alpha * (b_full - max_b)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0:
            # fallback to one-hot at max index
            w = np.zeros_like(b_full)
            w[np.argmax(b_full)] = 1.0
        else:
            w /= w_sum
        # gradient via convolution
        a_rev = a[::-1]
        B = np.fft.rfft(a_rev, n=L)
        W = np.fft.rfft(w, n=L)
        conv_rev = np.fft.irfft(B * W, n=L)
        grad = 2.0 * conv_rev[n-1:n-1+n]
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = step_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        sum_new = a_new.sum()
        if sum_new == 0:
            a_new = a.copy()
        else:
            a_new /= sum_new
        # evaluate new best
        A2 = np.fft.rfft(a_new, n=L)
        b2_full = np.fft.irfft(A2 * A2, n=L)[:2*n-1]
        score_new = 2.0 * n * b2_full.max()
        if score_new + 1e-12 < best_score:
            best_score = score_new
            best_a = a_new.copy()
            # optionally print
            if iteration % 500 == 0:
                print(f"  n={n} iter {iteration} score={best_score:.6f}")
        # accept some moves:
        if score_new <= 1.001 * (2.0 * n * max_b):
            a = a_new
        else:
            # occasional random restart to best
            if np.random.rand() < 0.01:
                a = best_a.copy()
        # occasional jitter
        if iteration % 2000 == 0:
            jitter = np.random.randn(n) * 1e-4
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()
    return best_a, best_score

import time
import numpy as np
# assume evaluate_sequence is already imported

def _next_pow_two(x):
    """Return the smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()

def _project_onto_simplex(v, s=1.0):
    """Projection of vector v onto the simplex {x >= 0, sum x = s}."""
    # Duchi et al. algorithm
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        theta = cssv[-1] / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _softmax_weights(b, alpha):
    """Smooth approximation of the max: returns normalized softmax weights."""
    # b is 1‑d array
    max_b = b.max()
    # exponent must be non‑positive; shift by max_b for stability
    exp_arg = alpha * (b - max_b)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0:
        # fallback: one‑hot on the maximal entry
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    return w

def _gradient_max_conv(a, w, L):
    """
    Compute ∇  Σ_k w_k (a * a)[k] using FFT.
    a : array of length n (sum = 1, non‑negative)
    w : softmax weights, length 2n‑1
    L : FFT length (power of two ≥ 2n‑1)
    """
    n = a.shape[0]
    # reverse a
    a_rev = a[::-1]
    # FFTs
    B = np.fft.rfft(a_rev, n=L)
    W = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(B * W, n=L)
    # slice the part that corresponds to gradient entries
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad

def _score_from_autocorr(max_b, n):
    """Score when the ℓ₁‑norm is 1 (the denominator is 1)."""
    return 2.0 * n * max_b

def _optimize_sequence(a0, deadline, alpha=10.0, lr=0.2, jitter_prob=0.01):
    """
    Perform a fast projected (Adam) descent on the smooth max objective.
    Returns the best sequence found and its score (as defined by evaluate_sequence).
    """
    a = a0.copy()
    n = a.shape[0]
    # enforce sum = 1
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= s

    # FFT size
    L = _next_pow_two(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # keep best
    # initial score
    A = np.fft.rfft(a, n=L)
    b_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    max_b = b_full.max()
    best_score = _score_from_autocorr(max_b, n)
    best_a = a.copy()
    it = 0

    while time.time() < deadline:
        it += 1

        # compute convolution (full)
        A = np.fft.rfft(a, n=L)
        b_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]

        max_b = b_full.max()
        # smooth weights
        w = _softmax_weights(b_full, alpha)

        # gradient of smooth objective
        grad = _gradient_max_conv(a, w, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        sum_new = a_new.sum()
        if sum_new == 0.0:
            a_new = a.copy()
        else:
            a_new /= sum_new

        # occasional jitter to escape flat plateaus
        if np.random.rand() < jitter_prob:
            jitter = np.random.randn(n) * 1e-4
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new /= a_new.sum()

        # evaluate candidate
        A2 = np.fft.rfft(a_new, n=L)
        b2_full = np.fft.irfft(A2 * A2, n=L)[: 2 * n - 1]
        max_b2 = b2_full.max()
        score2 = _score_from_autocorr(max_b2, n)

        if score2 < best_score:
            best_score = score2
            best_a = a_new.copy()
            if it % 500 == 0:
                print(f"    iter {it:6d}  best score {best_score:.6f}")

        # accept if not significantly worse; this stabilises the walk
        if score2 <= best_score * 1.001:
            a = a_new
        else:
            # with small probability jump back to the best seen so far
            if np.random.rand() < 0.02:
                a = best_a.copy()

    return best_a, best_score

def _initial_arcsine_like(n, exponent=0.5, offset=0.0):
    """
    Produce a monotone decreasing sequence a_i = (i+1+offset)^{-exponent}.
    The result is already normalised to unit ℓ₁‑norm.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    a = np.power(idx, -exponent)
    # Clip to the bounds used by evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() == 0.0:
        a = np.ones_like(a) / n
    else:
        a = a / a.sum()
    return a

def _upscale_sequence(a_small, N):
    """
    Linear interpolation of a_small onto a new length N.
    The returned sequence is renormalised to unit ℓ₁‑norm.
    """
    n = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n)
    x_big = np.linspace(0.0, 1.0, N)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    if a_big.sum() == 0.0:
        a_big = np.ones_like(a_big) / N
    else:
        a_big = a_big / a_big.sum()
    return a_big

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
        2 * n * max(a*a) / (sum(a)**2)

    The search consists of:
      * a quick grid over short lengths and a few (exponent,offset) seeds;
      * a smooth‑max projected Adam optimisation (FFT‑based) for each seed;
      * an optional up‑scaling phase that refines the best seed on a larger
        length (up to 10 000 terms).
    The routine respects the time budget and prints progress.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Coarse grid on modest lengths
    # ------------------------------------------------------------------
    short_lengths = [1200, 2000, 3000]
    exponents = [0.45, 0.5, 0.55, 0.6]
    offsets = [0.0, 0.2, 0.5, 1.0]

    print("=== coarse grid stage ===")
    for n in short_lengths:
        for exp in exponents:
            for off in offsets:
                if time.time() >= deadline:
                    break
                a0 = _initial_arcsine_like(n, exponent=exp, offset=off)
                # give each run a slice of the remaining time
                sub_deadline = min(deadline, time.time() + 8.0)  # at most 8 s per seed
                a_opt, score_opt = _optimize_sequence(a0, sub_deadline,
                                                      alpha=12.0, lr=0.25,
                                                      jitter_prob=0.02)
                # evaluate with the official routine for safety
                try:
                    eval_score = evaluate_sequence(a_opt.tolist())
                except Exception:
                    continue
                if eval_score < best_score:
                    best_score = eval_score
                    best_seq = a_opt.copy()
                    print(f"new overall best: len={n} exp={exp} off={off} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Upscaling & long‑run refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        # aim for a longer vector, keep the shape obtained so far
        upscale_targets = [5000, 8000, 10000]
        for N in upscale_targets:
            if time.time() >= deadline:
                break
            a_up = _upscale_sequence(best_seq, N)
            sub_deadline = min(deadline, time.time() + 25.0)   # more time for longer runs
            a_opt, score_opt = _optimize_sequence(a_up, sub_deadline,
                                                  alpha=15.0, lr=0.2,
                                                  jitter_prob=0.015)
            try:
                eval_score = evaluate_sequence(a_opt.tolist())
            except Exception:
                continue
            if eval_score < best_score:
                best_score = eval_score
                best_seq = a_opt.copy()
                print(f"upscale improved: N={N} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback (should never fire)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(1000, dtype=np.float64) / 1000.0
        best_score = evaluate_sequence(best_seq.tolist())

    print("=== finished ===")
    print(f"final best score (official) = {best_score:.6f}  length = {len(best_seq)}")

    # convert to plain python list of floats as required
    return best_seq.tolist()

"""
Smooth‑max projected Adam optimizer for the autoconvolution objective.

We minimise
    score = 2 * n * max(a*a) / (sum(a)**2)

which is scale‑invariant.  The algorithm works with the unit‑ℓ₁‑norm
(sum(a)=1).  At each iteration we:
  * compute the full self‑convolution by FFT,
  * form a soft‑max over the convolution entries (parameter α) – a smooth
    approximation of the max,
  * obtain the gradient of the smooth objective via a second FFT
    (the convolution of the reversed a with the soft‑max weights),
  * perform an Adam update,
  * clip to non‑negativity and renormalise (projection onto the simplex).

A coarse grid of short lengths provides several seeds; the best seed is
up‑scaled to longer lengths (up to 10 000) and refined again.
The routine respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow_two(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _softmax_weights(b: np.ndarray, alpha: float) -> np.ndarray:
    """
    Smooth approximation of the max:   w_k = exp(α(b_k‑max_b))/Σ.
    Returns a probability vector that sums to one.
    """
    max_b = b.max()
    # shift = max_b makes the largest exponent zero → exp(0)=1
    exp_arg = alpha * (b - max_b)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0:          # numerical safeguard
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    return w


def _gradient_max_conv(a: np.ndarray, w: np.ndarray, L: int) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a*a)[k] with respect to a.
    Uses the identity  ∂/∂a_j Σ_k w_k (a*a)[k] = 2·(a_rev * w)[j]
    where * denotes linear convolution and a_rev is a reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    B = np.fft.rfft(a_rev, n=L)
    W = np.fft.rfft(w, n=L)
    conv_rev = np.fft.irfft(B * W, n=L)          # length L (real)
    # indices n‑1 … n‑1+(n‑1) contain the desired n‑component gradient
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad


def _score_from_autocorr(max_b: float, n: int) -> float:
    """Score for a sequence with sum(a)=1."""
    return 2.0 * n * max_b


def _initial_arcsine_like(n: int, exponent: float = 0.5,
                         offset: float = 0.0) -> np.ndarray:
    """
    Produce a decreasing seed a_i = (i+1+offset)^(-exponent),
    then normalise to unit ℓ₁‑norm.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    a = np.power(idx, -exponent)
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / n
    else:
        a /= total
    return a


def _upscale_sequence(a_small: np.ndarray, N: int) -> np.ndarray:
    """
    Linear interpolation of a_small onto length N and re‑normalisation.
    """
    n = a_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n)
    x_big = np.linspace(0.0, 1.0, N)
    a_big = np.interp(x_big, x_small, a_small)
    a_big = np.maximum(a_big, 0.0)
    total = a_big.sum()
    if total == 0.0:
        a_big = np.ones_like(a_big) / N
    else:
        a_big /= total
    return a_big


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _optimize_sequence(a0: np.ndarray,
                       deadline: float,
                       alpha: float = 10.0,
                       lr: float = 0.2,
                       jitter_prob: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Fast projected Adam descent on the smooth‑max objective.
    Returns the best sequence found and its (official) score.
    """
    a = np.maximum(a0, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.ones_like(a) / a.shape[0]
    else:
        a /= total

    n = a.shape[0]
    L = _next_pow_two(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # initialise best
    A = np.fft.rfft(a, n=L)
    b_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    best_score = _score_from_autocorr(b_full.max(), n)
    best_a = a.copy()
    it = 0

    while time.time() < deadline:
        it += 1

        # current convolution
        A = np.fft.rfft(a, n=L)
        b_full = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
        max_b = b_full.max()

        # soft‑max weights over the convolution entries
        w = _softmax_weights(b_full, alpha)

        # gradient of the smooth objective
        grad = _gradient_max_conv(a, w, L)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = a.copy()
        else:
            a_new /= s

        # occasional jitter to escape plateaus
        if np.random.rand() < jitter_prob:
            jitter = np.random.randn(n) * 1e-4
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new /= a_new.sum()

        # evaluate candidate (exact max via FFT, no extra Python call)
        A2 = np.fft.rfft(a_new, n=L)
        b2_full = np.fft.irfft(A2 * A2, n=L)[: 2 * n - 1]
        score2 = _score_from_autocorr(b2_full.max(), n)

        if score2 < best_score:
            best_score = score2
            best_a = a_new.copy()
            if it % 500 == 0:
                print(f"    iter {it:6d}  best score {best_score:.6f}")

        # accept if not dramatically worse (helps stability)
        if score2 <= best_score * 1.001:
            a = a_new
        else:
            # with a small chance reset to the best seen so far
            if np.random.rand() < 0.02:
                a = best_a.copy()

    return best_a, best_score
