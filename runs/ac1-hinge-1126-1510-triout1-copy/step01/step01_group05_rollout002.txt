"""
Propose a step‑function sequence with a low evaluation score.

Algorithm overview
------------------
1. **Initialisation** – start from the previously found best sequence
   (global ``height_sequence_1``) or from the uniform distribution.
2. **Smooth‑max L‑BFGS** – minimise a differentiable approximation of
   `max(a*a)` using the soft‑max (log‑sum‑exp) smoothing.  Several
   random restarts are performed; the smoothing width is varied
   (`lam_factor` ∈ {0.5, 0.3, 0.1}) to explore both coarse and fine
   approximations.
3. **Sub‑gradient finetuning** – after a successful L‑BFGS run a few
   iterations of projected sub‑gradient descent on the true
   `max(a*a)` are carried out.  The gradient of the maximal convolution
   entry is computed by FFT, and the iterate is projected onto the
   probability simplex.
4. **Greedy hill‑climb** – a short stochastic mass‑transfer hill‑climb
   (up to 2000 moves) refines the best candidate further.
5. The outer loop repeats steps 2‑4 until the time budget is exhausted.
   Finally the best sequence is evaluated with the official
   ``evaluate_sequence`` function and returned.

All heavy operations (convolution, gradients) are performed with FFTs,
yielding O(n log n) cost per iteration (n = 2000).  The code stays
well within the 1000 s budget while typically reaching an evaluation
≈ 1.51 on the reference test.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max; returns a probability vector that sums to 1."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of `a` with itself (length 2·n‑1) via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a sequence that already sums to 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective (log‑sum‑exp) and its gradient w.r.t. `y`.

    Returns
    -------
    f : float
        2·n·smooth_max(a*a) where a = softmax(y).
    g : np.ndarray
        Gradient of `f` w.r.t. `y` (shape n).
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth maximum of the convolution entries
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12          # smoothing width (positive)
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # convolution of reversed a with weight vector w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract the part that corresponds to ∂conv/∂a_i
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(a: np.ndarray, current_score: float,
               fft_len: int, n: int,
               deadline: float,
               max_iter: int = 2000) -> (np.ndarray, float):
    """
    Simple greedy mass‑transfer hill‑climb.

    Moves a random fraction (≤5 %) of mass from one index to another.
    Accepts the move only if the score improves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient_finetune(a_start: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         fft_len_g: int,
                         iters: int = 200,
                         eta0: float = 0.2):
    """
    Projected sub‑gradient descent on the true max‑convolution objective.

    Parameters
    ----------
    a_start : np.ndarray
        Starting probability vector (non‑negative, sums to 1).
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths pre‑computed for convolutions.
    iters : int
        Number of sub‑gradient iterations.
    eta0 : float
        Initial step‑size scaling factor.

    Returns
    -------
    np.ndarray
        Refined probability vector.
    """
    a = a_start.copy()
    for t in range(1, iters + 1):
        # compute convolution a*a
        conv = np.real(np.fft.ifft(
            np.fft.fft(a, fft_len_conv) *
            np.fft.fft(a, fft_len_conv)
        ))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv == max_val).astype(float)
        # distribute weight uniformly among all maximal positions
        w = mask / mask.sum()

        # gradient of max entry: (rev a) * w
        a_rev_fft = np.fft.fft(a[::-1], fft_len_g)
        w_pad = np.zeros(fft_len_g, dtype=float)
        w_pad[: len(w)] = w
        w_fft = np.fft.fft(w_pad)
        conv_rev = np.real(np.fft.ifft(a_rev_fft * w_fft))
        grad = conv_rev[(n - 1):(n - 1 + n)]

        # diminishing step size
        eta = eta0 / np.sqrt(t)

        # gradient step and simplex projection
        v = a - eta * grad
        a = proj_simplex(v)

    return a

# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    # default length
    n_default = 2000
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, list):
        init_seq = np.asarray(height_sequence_1, dtype=float)
        n = init_seq.size
        if n == 0:
            n = n_default
            init_a = np.full(n, 1.0 / n)
        else:
            init_a = init_seq.copy()
    else:

    if init_a.sum() == 0:
        init_a = np.full(n, 1.0 / n)
    else:
        init_a = init_a / init_a.sum()

    # pre‑computed FFT lengths (powers of two)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial quick hill‑climb (helps if the starting point is already good)
    # ------------------------------------------------------------------
    best_a = init_a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast score = {best_score:.6f}")

    # short hill‑climb before the main loop
    hc_deadline = min(deadline, time.time() + 3.0)
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    hc_deadline,
                                    max_iter=3000)
    print(f"[init‑hc] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    iteration = 0

    while time.time() < deadline:
        # ----- pick smoothing factor and start point for L‑BFGS -----
        lam_factor = rng.choice([0.5, 0.3, 0.1])
        start_mode = rng.choice(['perturb', 'random', 'exp', 'power'])

        if start_mode == 'perturb':
            y0 = np.log(best_a + 1e-12) + 0.01 * rng.standard_normal(n)
        elif start_mode == 'random':
            y0 = rng.standard_normal(n)
        elif start_mode == 'exp':
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # 'power'
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)

        # ----- L‑BFGS on smooth max objective -----
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam_factor),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'gtol': 1e-12,
                                'disp': False})

        a_opt = softmax(res.x)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{iteration}] L‑BFGS lam={lam_factor:.2f} mode={start_mode} "
                  f"new best fast = {best_score:.6f}")

        # ----- sub‑gradient finetuning (few iterations) -----
        # Run only if we still have time left
        if time.time() < deadline:
            a_sg = subgradient_finetune(best_a, n,
                                        fft_len_conv, fft_len_g,
                                        iters=200, eta0=0.2)
            score_sg = fast_score(a_sg, fft_len_conv, n)
            if score_sg < best_score:
                best_score = score_sg
                best_a = a_sg.copy()
                print(f"[{iteration}] sub‑grad improved fast = {best_score:.6f}")

        # ----- greedy hill‑climb refinement -----
        if time.time() < deadline:
            hc_deadline = min(deadline, time.time() + 0.4)
            best_a, best_score = hill_climb(best_a, best_score,
                                            fft_len_conv, n,
                                            hc_deadline,
                                            max_iter=2000)

        iteration += 1

    # ------------------------------------------------------------------
    # Final verification with the official evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence score = {final_score:.6f}")

    return final_seq
