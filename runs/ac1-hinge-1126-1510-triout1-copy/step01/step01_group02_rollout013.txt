"""
Propose a step function with a lower evaluation score than the Catalan baseline.

Algorithm
---------
1. Initialise the sequence with the truncated Catalan (half‑binomial) coefficients.
   This gives a self‑convolution that is (almost) constant = 1, i.e. a score ≈π/2.

2. Repeatedly improve the sequence by a *gradient‑guided mass transfer*:
   * Compute the self‑convolution with an FFT, locate the index `kmax` where the
     convolution attains its maximum `M`.
   * For each position `i` the contribution of `a[i]` to that peak is
       `c[i] = a[i] * a[kmax‑i]`  (zero if the partner index is out of range).
   * Choose a source index `src` with the largest contribution `c[src]`
     (this is the coefficient that hurts the peak most).
   * Choose a destination index `tgt` for which the partner value
     `a[kmax‑tgt]` is minimal (ideally zero) and that still has capacity
     (we keep all entries ≤ `CAP = 0.90` so that squares stay below 1).
   * Move as much mass as possible from `src` to a *set* of low‑partner
     destinations, respecting the cap.  The move is performed in one step
     and the new score is evaluated.
   * Accept the move if it improves the current score; otherwise accept it
     with a small simulated‑annealing probability (temperature linearly
     decays to zero).  The best sequence ever seen is stored.

3. Every few thousand iterations a tiny random shake (Gaussian noise on a
   handful of entries) is tried to escape possible plateaus.

The procedure stops when the time budget (≈1000 s) expires or when the
target score `1.503` is reached.  The returned list contains only non‑negative
floats (the security checks inside `evaluate_sequence` will accept it).

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation identical to the hidden `evaluate_sequence` but without
# safety checks – used many times inside the optimiser.
def _fast_score(seq: np.ndarray) -> float:
    n = seq.size
    s = seq.sum()
    if s < 0.01:
        return np.inf
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
    return 2.0 * n * conv.max() / (s * s)

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a score
    below the target 1.503.  The routine uses a greedy, gradient‑guided
    local search with a small simulated‑annealing component.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0               # safety margin

    # ------------------------------------------------------------------
    # 1.  Catalan (half‑binomial) start – length 2500 is a good compromise.
    n = 2500
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)

    cur_seq = a.copy()
    cur_score = _fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    print(f"Initial Catalan score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Parameters
    CAP = 0.90                     # maximum value any entry may attain
    init_temp = best_score * 0.5   # start temperature for SA
    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # 3.  Main optimisation loop
    while time.time() < deadline:
        it += 1

        # ----- 3.1  current convolution and peak location -------------
        L = 1 << ((2 * n - 1).bit_length())
        f = np.fft.rfft(cur_seq, n=L)
        conv = np.fft.irfft(f * f, n=L)[: 2 * n - 1]
        k_max = int(conv.argmax())                # index of the maximum
        max_b = conv.max()

        # ----- 3.2  compute contribution of each coefficient to the peak
        idx = k_max - np.arange(n)                # partner indices
        partner = np.zeros_like(cur_seq)
        mask = (idx >= 0) & (idx < n)
        partner[mask] = cur_seq[idx[mask]]
        # contribution to the peak = a[i] * a[kmax-i]
        contrib = cur_seq * partner

        # ----- 3.3  source selection (largest contribution) ----------
        # ignore zero‑contribution entries
        src_candidates = np.where(cur_seq > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmax(contrib[src_candidates])]

        # ----- 3.4  destination selection (small partner, room to grow) ----------
        # we want indices where partner value is tiny (ideally zero) and
        # the entry is below the cap.
        tgt_mask = cur_seq < CAP
        if not np.any(tgt_mask):

            CAP = min(CAP + 0.05, 1000.0)
            tgt_mask = cur_seq < CAP

        min_partner = partner[tgt_mask].min()
        tgt_candidates = np.where(tgt_mask & (partner == min_partner))[0]

        if tgt_candidates.size == 0:
            # fallback – any slot with room
            tgt_candidates = np.where(tgt_mask)[0]

        # ----- 3.5  perform a mass transfer from src to several targets ----
        trial = cur_seq.copy()
        remaining = trial[src]

        # shuffle the target list to obtain a spread
        np.random.shuffle(tgt_candidates)

        for tgt in tgt_candidates:
            if remaining <= 1e-12:
                break
            room = CAP - trial[tgt]
            if room <= 1e-12:
                continue
            delta = min(remaining, room)
            trial[src] -= delta
            trial[tgt] += delta
            remaining -= delta

        # If something is still left (very unlikely) dump it to a random free slot
        if remaining > 1e-12:

                room = CAP - trial[tgt]
                delta = min(remaining, room)
                trial[src] -= delta
                trial[tgt] += delta
                remaining -= delta

        # ----- 3.6  evaluate the trial sequence -----------------------
        trial_score = _fast_score(trial)

        # ----- 3.7  record best solution --------------------------------
        if trial_score < best_score:
            best_score = trial_score
            best_seq = trial.copy()
            last_improve = it
            print(f"Iter {it:07d} – new best score: {best_score:.6f}")

        # ----- 3.8  acceptance (greedy + simulated annealing) ----------
        accept = False
        if trial_score < cur_score:
            accept = True
        else:
            elapsed = time.time() - start_time
            temp = init_temp * (1.0 - elapsed / (budget_s - 5.0))
            if temp > 1e-12:
                prob = np.exp((cur_score - trial_score) / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_seq = trial
            cur_score = trial_score

        # ----- 3.9  occasional random shaking -------------------------
        if it % 5000 == 0:
            shake = cur_seq.copy()
            idxs = np.random.randint(0, n, size=8)
            shake[idxs] = np.maximum(0.0, shake[idxs] + np.random.randn(8) * 0.02)
            shake_score = _fast_score(shake)
            if shake_score < best_score:
                best_score = shake_score
                best_seq = shake.copy()
                cur_seq = shake
                cur_score = shake_score
                last_improve = it
                print(f"Iter {it:07d} – shake improved score to {best_score:.6f}")

        # ----- 3.10 early exit if target reached ----------------------
        if best_score < 1.503:
            print(f"Target reached at iteration {it}, score {best_score:.6f}")
            break

        # ----- 3.11 safeguard – stop if no improvement for a long time --
        if it - last_improve > 200_000:
            # after a long plateau we stop to avoid wasting time
            print("No improvement for many iterations – terminating early.")
            break

    print(f"Finished after {it} iterations. Best score: {best_score:.6f}")
    # The runner expects a plain Python list
    return best_seq.tolist()
