Propose a candidate step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three parts:

1. **Broad parametric sweep** – generate simple families of shapes
   (power‑law, exponential, “U‑shaped” sum of two exponentials, etc.),
   normalise each to unit sum and keep the best one.

2. **Adam descent on a smooth approximation of the max** –  
   The objective `max(conv)` is replaced by a soft‑max
   `log‑sum‑exp(α·conv) / α`.  For a given α the gradient is
   `2 * correlation(softmax_weights, a)`.  Adam updates are applied
   on the simplex (projection onto the probability simplex after each
   step).  α is increased from a small to a large value so that the
   optimisation first sees a smooth landscape and then focusses on the
   true supremum norm.

3. **Local random moves** – after the Adam phase we perform many cheap
   perturbations that move a small amount of mass from a large entry to
   a random entry; any improvement is kept.  This helps to escape the
   final plateau of the gradient method.

All steps respect the supplied time budget (default 1000 s).  The
returned sequence is scaled to the normalisation used by the reference

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}. """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    print(f"--- Phase 1: parametric sweep (n={n}) ---")
    exponents = np.linspace(0.0, 2.0, 41)                 # power‑law exponents
    lambdas = np.linspace(0.0, 5.0, 41)                  # exponential decay rates
    def _evaluate_and_maybe_update(a_raw):
        nonlocal best_seq, best_obj
        a_norm = a_raw / a_raw.sum()
        fft_a = np.fft.fft(a_norm, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        obj = conv.max()
        if obj < best_obj:
            best_obj, best_seq = obj, a_norm.copy()
            print(f"  new best | obj={obj:.8f} | score={2 * n * obj:.6f}")

    # ---- Power‑law (increasing & decreasing) ----
    for p in exponents:
        if time.time() > deadline:
            break
        base = np.arange(1, n + 1, dtype=np.float64) ** p
        _evaluate_and_maybe_update(base)               # increasing
        _evaluate_and_maybe_update(base[::-1])          # decreasing

    # ---- Pure exponential decay (and its mirror) ----
    for lam in lambdas:
        if time.time() > deadline:
            break
        exp_shape = np.exp(-lam * np.arange(n, dtype=np.float64))
        _evaluate_and_maybe_update(exp_shape)          # decreasing
        _evaluate_and_maybe_update(exp_shape[::-1])    # increasing

    # ---- U‑shaped: sum of two exponentials (high at both ends) ----
    for lam in lambdas[::4]:    # fewer points – already expensive
        if time.time() > deadline:
            break
        left = np.exp(-lam * np.arange(n, dtype=np.float64))
        right = left[::-1]
        u_shape = left + right
        _evaluate_and_maybe_update(u_shape)

    # ------------------------------------------------------------------
    # 2.   Adam descent on a smoothed max (soft‑max) of the convolution.
    # Work with a unit‑sum vector; the evaluation score is   2·n·max(conv).
    a = best_seq.copy()
    best_obj = np.real(np.fft.ifft(np.fft.fft(a, n=L) *
                                   np.fft.fft(a, n=L)))[: 2 * n - 1].max()
    best_seq = a.copy()

    # Adam hyper‑parameters
    max_iters = 30000
    lr0 = 0.08
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    alpha_start, alpha_end = 4.0, 120.0    # smoothing parameter schedule

    print(f"--- Phase 2: Adam descent with smooth‑max (max {max_iters} iters) ---")
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---- current convolution ----
        fft_a = np.fft.fft(a, n=L)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        cur_max = conv.max()

        # ---- smooth‑max weights (softmax) ----
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        # to improve numerical stability we shift by the maximum value
        scaled = alpha * conv
        shift = scaled.max()
        exp_scaled = np.exp(scaled - shift)
        w = exp_scaled / exp_scaled.sum()                # length 2n‑1, sum = 1

        # ---- gradient of smooth‑max (2 * correlation) ----
        # FFT of reversed a via phase factor (no extra FFT)
        fft_a_rev = np.conj(fft_a) * phase
        fft_w = np.fft.fft(w, n=L)
        corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr[n - 1 : n - 1 + n]            # length n

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _project_simplex(a_new)

        # ---- evaluate new point ----
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[: 2 * n - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj, best_seq = obj_new, a_new.copy()
            print(f"Iter {it:5d}: improved | obj={obj_new:.8f} | score={2*n*obj_new:.6f}")

        a = a_new

    # ------------------------------------------------------------------
    # 3.   Local random moves (cheap hill‑climbing).
    print("--- Phase 3: random local moves ---")
    a_current = best_seq.copy()
    while time.time() < deadline:
        # Choose source index proportionally to its mass (large entries more likely)
        i = np.random.choice(n, p=a_current)
        j = np.random.randint(0, n)
        if i == j:
            continue
        # Transfer a small random fraction of the source mass
        delta = np.random.rand() * 0.02 * a_current[i]   # up to 2 % of a[i]
        if delta <= 0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue   # safety

        # Simple evaluation (single FFT)
        fft_an = np.fft.fft(a_new, n=L)
        conv_new = np.real(np.fft.ifft(fft_an * fft_an))[: 2 * n - 1]
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj, best_seq = obj_new, a_new.copy()
            a_current = a_new
            print(f"  move improved | obj={obj_new:.8f} | score={2*n*obj_new:.6f}")

    # ------------------------------------------------------------------
    # Final scaling to the reference normalisation (sum = sqrt(2·n))
    final_seq = (best_seq * target_sum).tolist()
    final_score = evaluate_sequence(final_seq)  # just for the final log line
    print(f"--- Finished: best score = {final_score:.6f} (n={n}) after {it} iterations ---")
    return final_seq
