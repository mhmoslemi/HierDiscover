```
def project_to_simplex(v, z):
    if np.sum(v) == z and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""Propose a candidate step sequence via deterministic subgradient descent
augmented by occasional random perturbations and a linear‑programming
mixing step.  The algorithm starts from the half‑order central binomial
coefficients (which already give a bound ≈ π/2) and iteratively flattens
the self‑convolution by moving mass away from the currently maximal
convolution entry.  After each update the vector is projected onto the
probability simplex (sum = 1) to keep the evaluation invariant under
scaling.  Random multiplicative noise is added periodically to escape
plateaus and a small LP‑based mixing step (solving max ∑b subject to
a∗b ≤ current bound) is performed a few times for larger jumps.  The
best sequence seen within the time budget (default 1000 s) is returned.
"""

import time
import math
import numpy as np

# optional import of cvxpy for LP steps
try:
    import cvxpy as cp
    _HAS_CVX = True
except Exception:
    _HAS_CVX = False


def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i) / 4**i,
    normalised so that sum(a) == 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_to_simplex(v: np.ndarray, z: float = 1.0) -> np.ndarray:
    """Euclidean projection of vector v onto the simplex {x>=0, sum(x)=z}."""
    # algorithm from https://arxiv.org/pdf/1309.1541.pdf
    if v.sum() == z and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho_arr = u - cssv / np.arange(1, len(v) + 1)
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _solve_lp_mix(a: np.ndarray, M: float) -> np.ndarray | None:
    """Given current sequence a (sum(a)=1) and current max convolution M,
    solve the linear programme
        maximise   1ᵀ b
        subject to b >= 0,
                   (a * b)_k ≤ M   for all k,
    where * denotes discrete convolution.
    Returns the optimal b as a NumPy array (already satisfying sum(b)=1)
    or None if the solver fails.
    """
    if not _HAS_CVX:
        return None
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    constraints = []
    # Build convolution constraints a * b <= M
    for k in range(2 * n - 1):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        if lo > hi:
            continue
        a_slice = a[lo:hi + 1]
        b_slice = b[k - np.arange(lo, hi + 1)]
        expr = cp.sum(cp.multiply(a_slice, b_slice))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=10000, abstol=1e-9, reltol=1e-9)
    except Exception:
        return None
    if b.value is None:
        return None
    b_val = np.array(b.value).flatten()
    # Project onto the simplex to enforce exact sum = 1 (tiny numerical errors may exist)
    b_val = _project_to_simplex(b_val, 1.0)
    return b_val


def propose_candidate(seed: int = 42, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step sequence with a low evaluation score.

    The routine runs for at most `budget_s` seconds (leaving a small margin)
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # 1) Initialise – half‑order series, normalised to sum = 1.
    # --------------------------------------------------------------
    n = 600                     # a size that balances quality & LP cost
    a = _half_order_series(n)  # sum == 1
    best_a = a.copy()
    # evaluation for a is simply the max of its self‑convolution:
    best_max = np.convolve(a, a).max()
    best_score = 2.0 * n * best_max  # same as evaluate_sequence for sum=1
    print(f"[init] n={n}, initial max={best_max:.8f}, score≈{best_score:.8f}")

    # --------------------------------------------------------------
    # 2) Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    # step‑size for subgradient updates (initial magnitude)
    step0 = 0.01
    # LP mixing will be attempted every few thousand iterations
    lp_interval = 3000
    # occasional random perturbations every 200 iterations
    noise_interval = 200

    while time.time() < deadline:
        iteration += 1
        # ---- subgradient step on the current maximal convolution entry ----
        conv = np.convolve(a, a)
        max_idx = int(np.argmax(conv))
        max_val = float(conv[max_idx])

        # gradient of max entry (sub‑gradient of the max‑function)
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        rev = max_idx - idx
        grad = np.zeros_like(a)
        # 2 * a_{k-i}
        grad[idx] = 2.0 * a[rev]

        # diminishing step size
        step = step0 / math.sqrt(iteration)

        # try a forward move; if it does not decrease the max, backtrack
        for _ in range(5):
            cand = a - step * grad
            cand = _project_to_simplex(cand, 1.0)
            cand_max = np.convolve(cand, cand).max()
            if cand_max <= max_val + 1e-14:  # accept non‑increasing
                a = cand
                max_val = cand_max
                break
            step *= 0.5   # backtrack a bit

        # track the best solution seen so far
        if max_val < best_max:
            best_max = max_val
            best_a = a.copy()
            best_score = 2.0 * n * best_max
            print(f"[{iteration:6d}] new best max={best_max:.8f} score≈{best_score:.8f}")

        # ---- occasional random noise to escape plateaus ----
        if iteration % noise_interval == 0:
            # noise strength slowly decays
            progress = (time.time() - start_time) / (budget_s - 2.0)
            sigma = 0.02 * (1.0 - progress) + 0.001
            noise = np.random.randn(n) * sigma
            cand = a * (1.0 + noise)
            cand = np.clip(cand, 0.0, None)
            cand = _project_to_simplex(cand, 1.0)
            cand_max = np.convolve(cand, cand).max()
            if cand_max < best_max:
                best_max = cand_max
                best_a = cand.copy()
                best_score = 2.0 * n * best_max
                a = cand
                print(f"[{iteration:6d}] noise improvement max={best_max:.8f}")

        # ---- occasional LP‑based mixing step (if cvxpy is available) ----
        if _HAS_CVX and iteration % lp_interval == 0:
            # Solve the LP to obtain a complementary vector b
            b = _solve_lp_mix(a, max_val)
            if b is not None:
                # Mix the current vector and the LP solution; mixing factor t
                t = 0.3
                cand = (1.0 - t) * a + t * b
                cand = _project_to_simplex(cand, 1.0)
                cand_max = np.convolve(cand, cand).max()
                if cand_max < best_max:
                    best_max = cand_max
                    best_a = cand.copy()
                    best_score = 2.0 * n * best_max
                    a = cand
                    print(f"[{iteration:6d}] LP mix improvement max={best_max:.8f}")

        # quick time‑check to avoid overrunning the deadline
        if time.time() > deadline:
            break

    # --------------------------------------------------------------
    # 3) Return the best sequence found
    # --------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations, best score = {final_score:.8f}")
    return best_a.tolist()

            cand_max = np.convolve(cand, cand).max()
            if cand_max <= max_val + 1e-14:
                a = cand
                max_val = cand_max
                break
            step *= 0.5

        # store best solution found so far
        if max_val < best_max:
            best_max = max_val
            best_a = a.copy()
            best_score = 2.0 * n * best_max
            print(f"[{iteration:6d}] new best max={best_max:.8f} score≈{best_score:.8f}")

        # ---- Occasional random perturbation to escape plateaux ----
        if iteration % noise_interval == 0:
            # noise strength decays as we approach the time limit
            progress = (time.time() - start_time) / (budget_s - 2.0)
            sigma = 0.02 * (1.0 - progress) + 0.001
            noise = np.random.randn(n) * sigma
            cand = a * (1.0 + noise)
            cand = np.clip(cand, 0.0, None)
            cand = _project_to_simplex(cand, 1.0)
            cand_max = np.convolve(cand, cand).max()
            if cand_max < best_max:
                best_max = cand_max
                best_a = cand.copy()
                best_score = 2.0 * n * best_max
                a = cand
                print(f"[{iteration:6d}] noise improvement max={best_max:.8f}")

        # ---- Periodic LP‑based mixing step (if cvxpy is available) ----
        if _HAS_CVX and iteration % lp_interval == 0:
            b = _solve_lp_mix(a, max_val)
            if b is not None:
                t = 0.3                     # mixing coefficient
                cand = (1.0 - t) * a + t * b
                cand = _project_to_simplex(cand, 1.0)
                cand_max = np.convolve(cand, cand).max()
                if cand_max < best_max:
                    best_max = cand_max
                    best_a = cand.copy()
                    best_score = 2.0 * n * best_max
                    a = cand
                    print(f"[{iteration:6d}] LP mix improvement max={best_max:.8f}")

        # quick time check
        if time.time() > deadline:
            break

    # ------------------------------------------------------------------
    # 3) Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_a.tolist())
    print(f"Finished after {iteration} iterations, best score = {final_score:.8f}")
    return best_a.tolist()
