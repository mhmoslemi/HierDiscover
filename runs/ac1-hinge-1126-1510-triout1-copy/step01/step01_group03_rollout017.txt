def propose_candidate(seed=0,budget_s=1000,n=2000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2
    # initial a = half-order series length n normalized sum=1
    a = half_order_series(n) # returns array sum=1
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    # parameters
    step = 0.1
    step_decay = 0.9995   # per iteration
    lambda_val = 10.0
    lambda_inc = 0.001
    it = 0
    while time.time() < deadline:
        it+=1
        # compute convolution via FFT
        conv = np.fft.ifft(np.fft.fft(a, 2*n) * np.fft.fft(a, 2*n)).real[:2*n-1]
        # compute max conv
        maxc = conv.max()
        # evaluate score maybe every few iterations
        if it % 100 == 0:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best = a.copy()
                print(f"[{it}] Score improved: {best_score}")
        # compute weights softmax
        # Might adjust lambda over time
        lambda_val = min(200, lambda_val + lambda_inc)
        exp_vals = np.exp(lambda_val * (conv - maxc))
        w = exp_vals / exp_vals.sum()
        # compute gradient
        # compute convolution of w and reversed a
        grad_full = np.convolve(w, a[::-1])
        grad = 2.0 * grad_full[n-1:n-1+n]  # slice length n
        # gradient descent step
        a = a - step * grad
        # project onto simplex
        # maybe clip negativity before projection to avoid numerical issues
        a = project_simplex(a)
        # optional random perturb step occasionally
        if it % 500 == 0:
            # small random multiplicative noise to escape stagnation
            noise = np.random.randn(n) * 0.01
            a = a * (1 + noise)
            a = project_simplex(a)
        # decay step
        step *= step_decay
    # final evaluate best
    return best.tolist()

def convolve_fft(x, y, size=None):
    if size is None:
        size = len(x) + len(y) - 1
    N = 1 << (size-1).bit_length()  # next power of two for speed
    fx = np.fft.rfft(x, N)
    fy = np.fft.rfft(y, N)
    conv = np.fft.irfft(fx * fy, N)[:size]
    return conv

def project_simplex(v):
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho: last i where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def convolve_fft(a, b):
    size = len(a) + len(b) - 1
    nfft = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=nfft)
    fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(fa * fb, n=nfft)[:size]
    return conv

def gradient_softmax(a, lam):
    n = len(a)
    conv = convolve_fft(a, a)
    maxc = conv.max()
    # compute softmax weights
    # use subtract max for numerical stability
    w = np.exp(lam * (conv - maxc))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # compute gradient: 2 * convolution of w with reversed a
    a_rev = a[::-1]
    grad_full = convolve_fft(w, a_rev)
    # slice [n-1 : n-1 + n] gives length n
    grad = 2.0 * grad_full[n-1:n-1+n]
    return grad, conv

def project_simplex(v):
    """Project onto the probability simplex (non-negative, sum to 1)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _half_order_series(n: int) -> np.ndarray:
    """Return the normalized half‑order series a_i = C(2i,i)/4**i (sum = 1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a

"""
Projected‑gradient descent on a smooth approximation of the
L∞‑norm of the self‑convolution.

The algorithm starts from the half‑order series (the classic
construction that achieves the π/2 ≈ 1.571 bound).  At each iteration we
compute a soft‑max weighting of the convolution entries, obtain the
gradient of the weighted sum (which concentrates on the current
maximal entry for large temperatures), take a step opposite to the
gradient and project the result onto the probability simplex
(non‑negative, sums to 1).  The temperature (λ) is slowly increased so
that the optimisation focuses more and more on the true maximum.
Occasional random perturbations and small mass‑transfer moves are
performed to escape plateaus.  The best sequence seen within the
time budget is returned.

The whole optimisation runs in O(n log n) per iteration thanks to FFT
convolutions, allowing thousands of iterations in the 1000 s budget.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ------------------------------------------------------------

def _half_order_series(n: int) -> np.ndarray:
    """
    First n coefficients a_i = C(2i,i)/4**i (central binomial /
    4^i), normalised so that sum_i a_i = 1.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    # recurrence a_i = a_{i-1} * ((2i-1)*(2i)) / (4*i*i)
    for i in range(1, n):
        a[i] = a[i - 1] * ((2 * i - 1) * (2 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution using FFT (real inputs).  Returns an array of
    length len(x)+len(y)-1.
    """
    size = len(x) + len(y) - 1
    nfft = 1 << (size - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, n=nfft)
    fy = np.fft.rfft(y, n=nfft)
    conv = np.fft.irfft(fx * fy, n=nfft)[:size]
    return conv


def _softmax_gradient(a: np.ndarray, lam: float):
    """
    Gradient of the soft‑max approximation of
        max_k (a * a)_k
    with temperature parameter lam.
    Returns (gradient, max_convolution_value).
    """
    n = a.shape[0]
    conv = _convolve_fft(a, a)               # length 2n‑1
    max_val = conv.max()

    # soft‑max weights (numerically stable)
    exp_w = np.exp(lam * (conv - max_val))
    w = exp_w / exp_w.sum()

    # gradient = 2 * (w * a_rev)  (where * denotes convolution)
    grad_full = _convolve_fft(w, a[::-1])    # length 3n‑2
    grad = 2.0 * grad_full[n - 1:n - 1 + n]  # slice of length n
    return grad, max_val


def _random_mass_transfer(a: np.ndarray, frac: float = 0.01) -> np.ndarray:
    """
    Move a small fraction (frac) of mass from a random early index
    to a random late index.  Used to break symmetry.
    """
    n = a.shape[0]
    # define early / late zones
    early_zone = np.arange(0, n // 4)
    late_zone = np.arange(3 * n // 4, n)

    i = np.random.choice(early_zone)
    j = np.random.choice(late_zone)

    delta = frac * a[i]               # amount to transfer
    if delta <= 0.0:
        return a

    b = a.copy()
    b[i] -= delta
    b[j] += delta
    # project back to simplex (ensures non‑negativity & unit sum)
    b = _project_simplex(b)
    return b

# Main optimisation routine
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Returns a list of non‑negative heights that (empirically) beats the
    uniform bound 2.0 and improves upon the half‑order construction.
    Uses projected gradient descent on a soft‑max approximation of the
    max‑convolution, with occasional random perturbations.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # 1) Initial sequence – half‑order series (good starting point)
    # ------------------------------------------------------------------
    n = 2000                                   # a balance between quality & speed
    seq = _half_order_series(n)                # already sums to 1

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] Half‑order score (n={n}): {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Hyper‑parameters for the optimisation loop
    # ------------------------------------------------------------------
    lam = 5.0                 # initial soft‑max temperature (smooth)
    lam_target = 200.0        # final temperature (sharp)
    # linear increase of λ over the whole run
    lam_inc = (lam_target - lam) / max(1.0, (deadline - start_time))

    eta = 0.4                 # step size for projected gradient descent
    eta_decay = 0.9995        # exponential decay of η

    iteration = 0

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # ----- gradient step on the smoothed max‑convolution ----------
        grad, max_conv = _softmax_gradient(seq, lam)
        cand = seq - eta * grad
        cand = _project_simplex(cand)

        seq = cand
        lam = min(lam + lam_inc, lam_target)
        eta *= eta_decay

        # ----- occasional cheap evaluation (FFT based) -----------------
        if iteration % 20 == 0:
            conv = _convolve_fft(seq, seq)
            approx_score = 2.0 * n * conv.max() / (np.sum(seq) ** 2)  # sum(seq) = 1
            # only call the expensive security‑checked evaluator if we
            # see a promising improvement
            if approx_score < best_score * 1.05:
                true_score = evaluate_sequence(seq.tolist())
                if true_score < best_score:
                    best_score = true_score
                    best_seq = seq.copy()
                    print(f"[{iteration:6d}] New best score: {best_score:.6f}")

        # ----- random perturbations to escape plateaus -----------------
        if iteration % 150 == 0:
            # multiplicative Gaussian noise
            noise = np.random.randn(n) * 0.005
            pert = seq * (1.0 + noise)
            pert = np.maximum(pert, 0.0)
            seq = _project_simplex(pert)

        # ----- directed mass‑transfer move (early → late) -------------
        if iteration % 300 == 0:
            cand = _random_mass_transfer(seq, frac=0.015)
            # evaluate only if it looks potentially better
            conv = _convolve_fft(cand, cand)
            approx = 2.0 * n * conv.max() / (np.sum(cand) ** 2)
            if approx < best_score * 1.02:
                true_score = evaluate_sequence(cand.tolist())
                if true_score < best_score:
                    best_score = true_score
                    best_seq = cand.copy()
                    seq = cand
                    print(f"[{iteration:6d}] Mass‑transfer improved score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final report and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"Finished after {iteration} iterations. Final best score: {final_score:.6f}")
    return best_seq.tolist()
