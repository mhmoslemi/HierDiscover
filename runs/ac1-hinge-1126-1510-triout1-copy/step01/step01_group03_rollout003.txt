"""
Propose a candidate sequence that improves the classic Π/2 ≈ 1.571 bound.
The algorithm implements the linear‑programming based “exchange step’’ described
in the problem statement:

1. Start from the half‑order series (Catalan / 4**i) normalized to sum = 1.
2. Repeatedly:
   a. Compute M = max(conv(f,f)).
   b. Solve the LP  max Σ b_j  s.t.  conv(f,b) ≤ M  and  b ≥ 0.
      (conv(f,b) is linear in b, so the constraints are linear.)
   c. Scale the LP solution to the same ℓ₁‑norm as f
      (so the mixing keeps the sum fixed).
   d. Mix the current vector f with the scaled LP solution using a small
      mixing factor t that slowly decays.
   e. Keep the best sequence seen so far.
3. Occasionally add a tiny random perturbation and a simple sub‑gradient
   step on the currently maximal convolution entry to escape plateaus.

All operations respect the non‑negativity and ℓ₁‑normalisation required by
`evaluate_sequence`.  The LP is solved with SciPy’s HiGHS implementation,
which is fast enough for n≈1000 and the 1000 s time budget.

The result is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


def _half_order_series(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4**i,
    normalised so that Σ a_i = 1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        # a_i = a_{i-1} * ((2i-1)*(2i)) / (4*i*i)
        a[i] = a[i - 1] * ((2.0 * i - 1.0) * (2.0 * i)) / (4.0 * i * i)
    a /= a.sum()
    return a


def _conv_max(seq: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of `seq`."""
    return np.convolve(seq, seq).max()


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_j
    s.t.    conv(f,b) <= M   (component‑wise)
            b_j >= 0
    Returns the optimal b (may be all‑zero if infeasible).
    """
    n = len(f)
    # Build the Toeplitz matrix A such that (A @ b)[k] = (conv(f,b))[k]
    # First column = f padded with zeros, first row = [f[0], 0, …, 0]
    first_col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    first_row = np.zeros(n, dtype=f.dtype)
    first_row[0] = f[0]
    A = toeplitz(first_col, first_row)   # shape (2n-1, n)

    # RHS: every entry must be ≤ M
    rhs = np.full(2 * n - 1, M, dtype=f.dtype)

    # Objective: maximise Σ b  → minimise -Σ b
    c = -np.ones(n, dtype=f.dtype)

    # Bounds: b_j ≥ 0
    bounds = [(0, None)] * n

    # Solve LP with HiGHS (fast for medium size)
    res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method='highs')
    if not res.success:
        # Infeasible or numerical trouble – return zeros
        return np.zeros_like(f)
    return res.x

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    n = kwargs.get("n", 1000)   # length of the sequence

    seq = init.copy()                     # current iterate, sum = 1

    # --------------------------------------------------------------
    # 3) return the best sequence found
    # --------------------------------------------------------------
    print(f"[Done] Best score achieved: {best_score:.6f}")
