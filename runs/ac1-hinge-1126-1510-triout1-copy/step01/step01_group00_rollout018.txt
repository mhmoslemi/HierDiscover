def propose_candidate(seed=...), budget_s=1000):

    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    # Determine n: maybe choose n as len(height_sequence_1) else fallback to 3000.
    try: n = len(height_sequence_1)
    except NameError: n = 3000

    # Define function to generate initial vectors:
    def init_random_dirichlet(alpha):
        a = np.random.gamma(alpha, 1.0, size=n)
        a = a / a.sum()
        return a

    def init_powerlaw(p):
        w = np.arange(1, n+1) ** p
        a = w / w.sum()
        return a

    def init_triangular():
        a = np.concatenate([np.linspace(0, 1, n//2, endpoint=False),
                            np.linspace(1, 0, n - n//2, endpoint=True)])
        a = a / a.sum()
        return a

    def init_arcsine():
        i = np.arange(n)
        # shift to avoid division by zero
        eps = 1e-9
        x = (i + 0.5) / n  # scaled to (0,1)
        a = 1.0 / np.sqrt(x * (1 - x) + eps)
        a = a / a.sum()
        return a

    def init_from_prev():
        seq = globals().get('height_sequence_1', None)
        if seq is not None:
            a = np.array(seq, dtype=float)
            a = a / a.sum()
            # if length mismatched, pad/truncate
            if a.shape[0] != n:
                if a.shape[0] > n:
                    a = a[:n]
                else:
                    a = np.pad(a, (0, n-a.shape[0]), 'constant')
                a = a / a.sum()
            return a
        else:
            return init_random_dirichlet(1.0)

    initializers = [init_from_prev, lambda: init_random_dirichlet(0.5),
                   lambda: init_random_dirichlet(2.0),
                   lambda: init_powerlaw(0.5),
                   lambda: init_powerlaw(1.0),
                   lambda: init_powerlaw(1.5),
                   init_triangular,
                   init_arcsine]

    best_seq = None
    best_score = float('inf')

    # For each initializer, run optimization
    for idx, init_func in enumerate(initializers):
        if time.time() > deadline:
            break
        a = init_func()
        # Run Adam optimizer
        # hyperparameters
        alpha = 0.5    # base learning rate
        beta1 = 0.9
        beta2 = 0.999
        eps_opt = 1e-8
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # Set starting p
        p = 4.0
        # Set schedule to raise p after some iterations
        num_iters = 4000
        # Determine convolution FFT lengths (re-use to avoid recompute each iter)
        L_conv = 1 << ((2 * n - 1).bit_length())
        L_corr = 1 << ((3 * n - 2).bit_length())

        for it in range(num_iters):
            if time.time() > deadline:
                break

            # Compute convolution using FFT
            fft_a = np.fft.fft(a, n=L_conv)
            conv = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]

            # Compute objective Lp with current p
            # define weight w = (conv + 1e-12) ** (p - 1)
            eps_conv = 1e-12
            w = np.power(conv + eps_conv, p - 1)

            # correlation gradient
            fft_w = np.fft.fft(w, n=L_corr)
            a_rev = a[::-1]
            fft_a_rev = np.fft.fft(a_rev, n=L_corr)
            corr = np.real(np.fft.ifft(fft_w * fft_a_rev))
            grad = 2.0 * corr[n-1:n-1+n]

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)

            m_hat = m / (1 - beta1 ** (it + 1))
            v_hat = v / (1 - beta2 ** (it + 1))

            step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)

            a_new = a - step
            # project onto simplex to keep sum=1 & nonneg
            a_new = _project_simplex(a_new)

            # evaluate with original objective (max conv)
            # we can compute maxconv directly
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv  # since sum=1

            # Keep best
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a_new.copy()
                print(f"Init {idx}: iter {it}, new best score {best_score:.6f}")

            a = a_new

            # increase p gradually
            if it % 800 == 0 and p < 40:
                p = min(40, p * 1.5)

        # after finishing this initializer, maybe run a few more subgradient steps to fine tune 
        # using original subgradient approach (similar to previous algorithm) for refinement
        a_ref = a.copy()
        # run simple subgradient with diminishing step
        for it2 in range(2000):
            if time.time() > deadline:
                break
            # compute conv and gradient as before using active set
            fft_a = np.fft.fft(a_ref, n=L_conv)
            conv_full = np.real(np.fft.ifft(fft_a * fft_a))[:2*n-1]
            maxv = conv_full.max()
            eps = maxv * 1e-12 + 1e-14
            act = (conv_full >= maxv - eps).astype(np.float64)

            fft_act = np.fft.fft(act, n=L_corr)
            corr_full = np.real(np.fft.ifft(fft_act * fft_a_rev))
            grad2 = 2.0 * corr_full[n-1:n-1+n]

            step2 = 0.5 / np.sqrt(it2 + 1.0)
            a_ref = _project_simplex(a_ref - step2 * grad2)

            cur_score2 = 2.0 * n * maxv
            if cur_score2 < best_score:
                best_score = cur_score2
                best_seq = a_ref.copy()
                print(f"Init {idx}: subgrad iter {it2} new best score {best_score:.6f}")

        # ensure best_seq stored in proper scaling to avoid any zeros after projection
        # continue to next initializer

    # After processing all inits, produce final sequence:
    # Rescale to any sum, keep as list of floats.
    if best_seq is None:
        # fallback uniform
        best_seq = np.full(n, 1.0 / n)

    final_seq = best_seq.tolist()

    # Optionally clamp values to 0..1000

    final_seq = [float(min(max(x, 0.0), 1000.0)) for x in final_seq]

    print(f"--- Finished all initializations: best score = {best_score:.6f} ---")
    return final_seq

"""
Propose a step‑function (non‑negative sequence) that minimizes
2·n·max(conv(a,a))/sum(a)^2.

Algorithm:
1. Choose a length n (uses the global `height_sequence_1` if present,
   otherwise a fallback 3000).
2. Generate a variety of initial candidates (random Dirichlet,
   power‑law, triangular, Gaussian, sine, previous best, …) each
   normalised to sum = 1.
3. For each candidate run a short Adam optimiser on a smooth
   approximation of the L∞‑norm of the self‑convolution:
      F_p(a) = ( Σ (conv_i)^p )^{1/p}   (p starts at 4 and grows).
   The gradient is obtained analytically via FFT‑based correlation.
   After every update the vector is projected onto the probability
   simplex (non‑negative, Σ = 1).
4. After the Adam phase a few iterations of the classic sub‑gradient
   descent (the method used in the reference) are performed for fine
   tuning.
5. The best sequence seen so far (according to the exact score) is
   remembered.  The whole process stops when the time budget (≈1000 s)
   is exhausted and the best sequence is returned as a plain list.
"""

import math

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the probability simplex (non‑negative, sum = 1). """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x: int) -> int:
    """ Return the smallest power of two >= x. """
    return 1 << (x - 1).bit_length()

def _score(seq: np.ndarray) -> float:
    """Exact evaluation used for bookkeeping (scale‑invariant)."""
    n = seq.shape[0]
    L = _next_pow2(2 * n - 1)
    fft = np.fft.fft(seq, n=L)
    conv = np.real(np.fft.ifft(fft * fft))[: 2 * n - 1]
    max_conv = conv.max()
    sum_a = seq.sum()
    return float(2.0 * n * max_conv / (sum_a * sum_a))

    if isinstance(seq, list) and len(seq) > 0:

    a = w / w.sum()
    return a

def _init_triangular(n: int) -> np.ndarray:
    half = n // 2
    up = np.linspace(0.0, 1.0, half, endpoint=False)
    down = np.linspace(1.0, 0.0, n - half, endpoint=True)
    a = np.concatenate([up, down])
    return a / a.sum()

    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0

    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.sin(np.pi * i / (n + 1))

def _adam_optimize(a: np.ndarray, n: int, L_conv: int, L_corr: int,
                   deadline: float, best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the smooth Lp approximation of the max‑convolution.
    The vector a is assumed to sum to 1.
    """
    # hyper‑parameters
    alpha = 0.05           # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps_opt = 1e-8
    eps_conv = 1e-12

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # start with a moderate exponent and increase slowly
    p = 4.0
    max_iters = 8000
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- convolution a * a (linear) -----
        fft_a = np.fft.fft(a, n=L_conv)
        conv = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv   # exact score because sum(a)=1

        # keep the best vector seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam] iter {it:5d}: improved score = {best_score:.6f}")

        # ----- weights for a smooth approximation of the L∞ norm -----
        w = np.power(conv + eps_conv, p - 1.0)   # shape (2n‑1,)

        # ----- gradient via correlation (FFT) -----
        fft_w = np.fft.fft(w, n=L_corr)
        fft_a_rev = np.fft.fft(a[::-1], n=L_corr)
        corr_full = np.real(np.fft.ifft(fft_w * fft_a_rev))
        grad = 2.0 * corr_full[n - 1: n - 1 + n]   # length n

        # ----- Adam update and simplex projection -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = alpha * m_hat / (np.sqrt(v_hat) + eps_opt)

        a = _project_simplex(a - step)

        # ----- slowly increase the exponent p (more focus on the max) -----
        if (it + 1) % 2000 == 0 and p < 30.0:
            p = min(30.0, p * 1.5)

        it += 1

    return best_seq, best_score


def _subgradient_refine(a: np.ndarray, n: int, L_conv: int, L_corr: int,
                        deadline: float, best_seq: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """
    Classic projected sub‑gradient descent (the method used in the reference implementation).
    """
    max_iters = 12000
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution
        fft_a = np.fft.fft(a, n=L_conv)
        conv_full = np.real(np.fft.ifft(fft_a * fft_a))[: 2 * n - 1]
        max_val = conv_full.max()
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[SubG] iter {it:5d}: improved score = {best_score:.6f}")

        # active set (indices attaining the maximum up to tolerance)
        eps = max_val * 1e-12 + 1e-14
        act = (conv_full >= max_val - eps).astype(np.float64)

        # gradient = 2 * correlation(act, a)
        fft_act = np.fft.fft(act, n=L_corr)
        fft_a_rev = np.fft.fft(a[::-1], n=L_corr)
        corr_full = np.real(np.fft.ifft(fft_act * fft_a_rev))
        grad = 2.0 * corr_full[n - 1: n - 1 + n]

        # diminishing step size
        step = 0.5 / math.sqrt(it + 1.0)

        a = _project_simplex(a - step * grad)
        it += 1

    return best_seq, best_score

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of `v` onto the probability simplex
        (non‑negative components, sum = 1). """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _next_pow2(x: int) -> int:
    """ Smallest power of two ≥ `x`. """
    return 1 << (x - 1).bit_length()

    conv = np.real(np.fft.ifft(fft * fft))[: 2 * n - 1]
    max_conv = conv.max()
