def fast_conv_max(a):
    # compute convolution using FFT with length next power of 2 for speed
    n = len(a)
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]
    return conv.max()

def propose_candidate(...):
    import time, numpy as np

    deadline = time.time() + budget_s - 0.5

    # Choose initial n: if previous sequence exists, set n = len prev + 500 maybe; else start 5000.

    # We consider n values like n0 if >0 else maybe 5000, up to maybe 10000 if time.
    # We'll try increasing n gradually.

    best_score = float('inf')
    best_seq = None

    # We'll define a function to evaluate fast and track
    def eval_seq(seq):
        return 2.0 * len(seq) * fast_conv_max(seq) / (seq.sum()**2)

    # Explore multiple lengths
    lengths = [n0] if n0 > 0 else []
    # add some lengths: 2000, 3000, 4000, 6000, 8000, 10000 (bounded by time)
    lengths += [2000, 3000, 4000, 6000, 8000, 10000]

    # Ensure lengths unique and sorted
    lengths = sorted(set([l for l in lengths if l > 0]))

    for n in lengths:
        if time.time() >= deadline:
            break
        # Grid search p
        p_grid = np.linspace(0.5, 0.9, 41)  # 0.5 to 0.9 step 0.01
        for p in p_grid:
            if time.time() >= deadline: break
            seq = np.arange(1, n+1, dtype=np.float64) ** (-p)
            seq /= seq.sum()
            score = eval_seq(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[grid] n={n}, p={p:.4f}, score={score:.6f}")

    # Local refinement
    if best_seq is not None:
        seq = best_seq.copy()
        n = len(seq)
        # Simulated annealing style: gradually reduce step size.
        step = 0.05 * (1.0 / n)  # typical magnitude shift per move
        temperature = 1.0
        while time.time() < deadline:
            # pick two indices randomly
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # propose moving epsilon from i to j
            epsilon = np.random.exponential(scale=step)
            # ensure we don't make a_i negative
            if epsilon > seq[i]:
                epsilon = seq[i] * np.random.rand()
            new_seq = seq.copy()
            new_seq[i] -= epsilon
            new_seq[j] += epsilon
            # Normalize not needed if sum unchanged (epsilon moved)
            # Evaluate
            new_score = eval_seq(new_seq)
            # Accept if better or with probability exp(-(new_score - score)/temp)
            if new_score < best_score or np.random.rand() < np.exp(-(new_score - best_score)/temperature):
                seq = new_seq
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    print(f"[anneal] updated best score={best_score:.6f}")
            # decay temperature and step
            temperature = max(temperature * 0.9995, 1e-4)
            step = max(step * 0.9995, 1e-9)
    # Return best_seq as list
    if best_seq is None:
        # fallback uniform
        n = 3000
        best_seq = np.full(n, 1.0/n)
    return best_seq.tolist()

def fast_convolution(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n-1]

def evaluate_fast(a):
    n = a.shape[0]
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    conv = fast_convolution(a)
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a**2))

def project_simplex(v):
    # Projects onto the probability simplex {x >=0, sum x =1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works in three stages:

1. **Coarse grid search** – for several candidate lengths we scan a
   power‑law family  a_i ∝ (i+1)^{-p}  (p in a sensible interval) and keep
   the best sequence found.

2. **Fine‑grid refinement** – around the best exponent we evaluate a finer
   grid to improve the score.

3. **Projected gradient descent** – starting from the best sequence we
   repeatedly identify the shift `k` where the self‑convolution is maximal,
   compute the gradient of that entry, take a step opposite to the gradient
   and project back onto the probability simplex (non‑negative, sum = 1).
   The step size is adapted dynamically.  This directly reduces the peak
   of the convolution and therefore the evaluation score.

All heavy computations (convolution, gradient) use FFT for speed.
The routine respects the 1000 s time budget and returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def generate_power_law_seq(n: int, p: float) -> np.ndarray:
    """Return a non‑negative sequence a_i ∝ (i+1)^{-p} with sum(a)=1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def fast_convolution(seq: np.ndarray) -> np.ndarray:
    """
    Compute the discrete self‑convolution of ``seq`` using FFT.
    Returns the full convolution array of length 2*len(seq)-1.
    """
    n = seq.shape[0]
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    fft_seq = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft_seq * fft_seq, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clip to zero for safety
    np.maximum(conv, 0, out=conv)
    return conv


def evaluate_fast(seq: np.ndarray) -> float:
    """
    Fast analogue of ``evaluate_sequence`` – assumes all entries are already
    non‑negative, finite and far from the clipping limits.
    """
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    conv = fast_convolution(seq)
    max_b = conv.max()
    return float(2.0 * len(seq) * max_b / (sum_a ** 2))


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector ``v`` onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimizes the evaluation
    function within the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Determine a set of candidate lengths
    # ------------------------------------------------------------------
    lengths = []
    try:
        n0 = len(height_sequence_1)                 # type: ignore[name-defined]
        if n0 > 0:
            lengths.append(n0)
    except Exception:
        n0 = 0

    # Additional lengths – larger n gives a lower theoretical bound
    for L in (2000, 3000, 4000, 6000, 8000, 10000):
        if L not in lengths:
            lengths.append(L)
    # Stay safely below a size that would make repeated FFTs too slow
    lengths = [L for L in lengths if L <= 12000]
    lengths = sorted(set(lengths))

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 2. Coarse and fine grid search over power‑law exponent p
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        # Coarse grid (step ~0.01)
        p_grid = np.linspace(0.48, 0.72, 25)           # 25 candidates
        for p in p_grid:
            if time.time() >= deadline:
                break
            seq = generate_power_law_seq(n, p)
            score = evaluate_fast(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[grid] n={n:5d}, p={p:.4f}, score={score:.6f}")

        # Fine grid centred around the best p for this length (if we have one)
        if best_seq is not None and len(best_seq) == n:
            # Scan a finer interval – cheap because the grid is small
            fine_grid = np.linspace(0.49, 0.71, 21)
            for p in fine_grid:
                if time.time() >= deadline:
                    break
                seq = generate_power_law_seq(n, p)
                score = evaluate_fast(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"[fine] n={n:5d}, p={p:.5f}, score={score:.6f}")

    # ------------------------------------------------------------------
    # 3. Local refinement via projected gradient descent on the best seq
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        cur_score = best_score
        n = a.shape[0]

        # Initial step size – a fraction of a typical entry (≈1/n)
        step = 1.0
        iteration = 0
        # Limit iterations to keep within the time budget
        while time.time() < deadline and step > 1e-7 and iteration < 50000:
            conv = fast_convolution(a)
            k = int(conv.argmax())
            # Gradient of the active convolution entry (k where conv is maximal)
            if k == 0:
                grad = 2.0 * a
            else:
                grad = np.zeros_like(a)
                grad[: n - k] += a[k:]
                grad[k:] += a[: n - k]

            # Take a step opposite to the gradient and re‑project onto simplex
            cand = a - step * grad
            cand = project_onto_simplex(cand)

            cand_score = evaluate_fast(cand)

            if cand_score < cur_score - 1e-12:
                # Successful improvement – accept and slightly increase step
                a = cand
                cur_score = cand_score
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_seq = cand.copy()
                    print(
                        f"[grad] iter={iteration:5d}, n={n}, k={k}, "
                        f"score={cand_score:.6f}"
                    )
                step *= 1.05
            else:
                # No improvement – shrink step
                step *= 0.7

            iteration += 1

    # ------------------------------------------------------------------
    # 4. Safety net – uniform sequence if nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_fast(best_seq)
        print(f"[fallback] uniform seq, n={n}, score={best_score:.6f}")

    # Return a plain Python list as required by the external harness
    return best_seq.tolist()
