def simplex_projection(v):
    # ensure nonnegative and sum to 1
    # input: v (np.ndarray)
    # returns projected vector.
    # Clip negative? Actually algorithm handles.
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def subgrad_max_reduce(a, n, fft_len_conv, fft_len_g, steps=200, step_size=0.2):
    best = a.copy()
    best_score = fast_score(best,...)
    for it in range(steps):
        conv = conv_fft(best, fft_len_conv, n)
        max_val = conv.max()
        idx = np.flatnonzero(conv >= max_val - 1e-12) # within tolerance
        # create weight vector w
        w = np.zeros(2*n-1, dtype=float)
        w[idx] = 1.0 / len(idx)
        # compute gradient subg = 2 * cross_corr(best, w)
        rev_best = best[::-1]
        fa_rev = np.fft.fft(rev_best, fft_len_g)
        fw = np.fft.fft(w, fft_len_g)
        geom = np.real(np.fft.ifft(fa_rev * fw))
        g = 2.0 * geom[n-1:n-1+n]
        # direction
        dir = -g + np.mean(g)  # broadcast mean to vector
        # step
        new = best + step_size * dir
        # project onto simplex
        new = simplex_projection(new)
        new_score = fast_score(new, fft_len_conv, n)
        if new_score < best_score:
            best = new
            best_score = new_score
        else:
            # reduce step size?
            step_size *= 0.9
        # optionally break if step small
    return best, best_score
```

def simulated_annealing(a, n, fft_len_conv, deadline):
    best = a.copy()
    best_score = fast_score(best,...)
    cur = a.copy()
    cur_score = best_score
    T0 = 0.01 maybe.
    iteration = 0
    while time.time() < deadline:
        # propose move
        i,j = rng.integers(0,n,2)
        if i==j: continue
        # transfer delta = cur[i] * rng.random() * 0.1
        delta = cur[i] * rng.random() * 0.1
        if delta==0: continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur,...)
        dE = new_score - cur_score
        # acceptance probability
        T = T0 * (0.999 ** iteration)
        if dE < 0 or rng.random() < np.exp(-dE / (T+1e-12)):
            cur_score = new_score
            # keep new
        else:
            cur[i] += delta
            cur[j] -= delta
        if cur_score < best_score:
            best_score = cur_score
            best = cur.copy()
            print(f"[anneal] new best = {best_score}")
        iteration += 1
    return best,best_score

"""
Improved search for a low‐evaluation step sequence.

The algorithm works on the probability simplex (a ≥ 0, sum a = 1) and
optimises a smooth approximation of the ∞‑norm of the self‑convolution
via L‑BFGS.  A schedule of decreasing smoothing factors is used, after
which a cheap sub‑gradient descent on the true max‑convolution and a
large greedy hill‑climb finish the refinement.  Multiple random restarts
and a few deterministic “shapes” (exponential, power‑law, edge‑heavy,
half‑sine) are explored.  The best sequence found before the deadline is
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector (sums to 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*n‑1) using FFTs.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) for a vector that already
    sums to 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Duchi et al. (2008) algorithm.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
    reg: float = 0.0,
):
    """
    Smooth‑max objective (2·n·smooth_max(a*a)) and its gradient w.r.t. y.

    Parameters
    ----------
    y : ndarray (n,)
        Unconstrained parameters.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths pre‑computed for the two convolutions.
    lam_factor : float
        Smoothing width λ = lam_factor·max(conv) (+1e‑12).
    reg : float
        Optional L2 regularisation weight on a (helps to avoid spikes).

    Returns
    -------
    f : float
        Smoothed objective value.
    g : ndarray (n,)
        Gradient w.r.t. y.
    """
    # ----- map to the simplex -----
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # g_vec = Σ_i w_i * a_{i‑k}   (k = 0..n‑1)  →  cross‑correlation
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]  # length n

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    # optional small L2 regularisation on a
    if reg > 0.0:
        # ∂/∂a (reg·||a||²) = 2·reg·a
        grad_a_reg = 2.0 * reg * a
        # propagate through soft‑max Jacobian
        mean_reg = np.dot(a, grad_a_reg)
        grad_y += a * (grad_a_reg - mean_reg)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> (np.ndarray, float):
    """
    Greedy hill‑climb: repeatedly move a random fraction (≤10 %) of the mass
    from one coordinate to another as long as the objective improves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        # transfer up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgrad_max_reduce(
    a: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    steps: int = 150,
    step_size: float = 0.4,
) -> (np.ndarray, float):
    """
    Sub‑gradient descent on the true max‑convolution.
    At each iteration the gradient of max_k (a*a)_k is approximated by
    averaging the gradients of all indices attaining the current maximum.
    After a gradient step we project back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)
    for it in range(steps):
        conv = conv_fft(best, fft_len_conv, n)
        max_val = conv.max()
        # indices where the maximum is attained (tolerance for rounding)
        idx = np.flatnonzero(conv >= max_val - 1e-14)
        if idx.size == 0:
            break

        # weight vector in convolution space (average over all active maxima)
        w = np.zeros(2 * n - 1, dtype=float)
        w[idx] = 1.0 / idx.size

        # gradient of max conv: g = 2 * (rev_a * w)  (cross‑correlation)
        rev = best[::-1]
        fa_rev = np.fft.fft(rev, fft_len_g)
        fw = np.fft.fft(w, fft_len_g)
        g_full = np.real(np.fft.ifft(fa_rev * fw))
        g = 2.0 * g_full[(n - 1) : (n - 1 + n)]

        # direction that preserves the sum (projected gradient)
        direction = -g + np.mean(g)

        # tentative update
        cand = best + step_size * direction
        cand = simplex_projection(cand)

        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            # keep step size
        else:
            # shrink step size on failure
            step_size *= 0.92
            if step_size < 1e-6:
                break

    return best, best_score

        elif mode == "power":

            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":

            a_edge = np.sqrt(n / 2.0) / n
