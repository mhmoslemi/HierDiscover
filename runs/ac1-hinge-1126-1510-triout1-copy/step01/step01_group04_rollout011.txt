def smoothed_gradient(a, alpha):
    conv = np.convolve(a, a)
    max_val = conv.max()
    # compute weights
    # to prevent underflow, maybe compute exponent with shift: w_i = exp(alpha*(conv[i] - max_val)), which yields <=1.
    weights = np.exp(alpha * (conv - max_val))
    weights_sum = weights.sum()
    if weights_sum == 0:
        weights = np.zeros_like(weights)
    else:
        weights /= weights_sum

    w_rev = weights[::-1]
    grad_full = np.convolve(a, w_rev)
    # slice
    n = a.shape[0]
    grad = 2.0 * grad_full[n-1:2*n-1]
    return grad

def _generate_sequence(n: int, exponent: float, offset: float) -> np.ndarray:
    """Return a non‑negative power‑law sequence a[i] = (i+1+offset)^(-exponent)."""

    # Clip to the limits required by evaluate_sequence (never hit for our ranges)
    return np.clip(seq, 0.0, 1000.0)


def _project_onto_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Simple Euclidean projection onto the non‑negative simplex
    {y ≥ 0, Σy = target_sum}.  For speed we clip negatives and then
    rescale to the required sum – this respects both constraints.
    """
    x = np.maximum(x, 0.0)
    s = np.sum(x)
    if s == 0.0:
        return x
    return x * (target_sum / s)


def _gradient_of_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution entry (index k_max) with respect
    to the vector a.  For a*a we have ∂(a*a)[k]/∂a_j = 2*a[k‑j] when the
    index is valid, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    if lo <= hi:
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k_max - j
        grad[j] = 2.0 * a[i]
    return grad

def propose_candidate(budget_s: int = 1000, seed: int = 42) -> list[float]:
    """
    Search for a sequence that minimises
        2 * n * max(a*a) / (sum(a)**2)
    using a cheap grid over power‑law parameters followed by a brief
    projected‑gradient polishing loop.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # 1) Coarse grid search over length / exponent / offset
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    lengths = [2000, 3000, 4000, 5000, 6000, 8000]
    exponents = np.arange(0.52, 0.71, 0.02)   # 0.52, 0.54, ..., 0.70
    offsets = [0.0, 0.2, 0.5, 1.0, 1.5, 2.0]

    print("=== grid search ===")
    for n in lengths:
        if time.time() > deadline:
            break
        for p in exponents:
            for off in offsets:
                seq = _generate_sequence(n, exponent=p, offset=off)
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"new grid best: n={n:4d} p={p:.2f} off={off:3.1f} → {score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional: try the previously‑found global sequence if present
    # ------------------------------------------------------------------
    try:
        prev = np.array(height_sequence_1, dtype=np.float64)
        prev_score = evaluate_sequence(prev.tolist())
        if prev_score < best_score:
            best_score = prev_score
            best_seq = prev.copy()
            print(f"replaced by previous best: score={prev_score:.6f}")
    except Exception:
        # height_sequence_1 may be missing – ignore silently
        pass

    if best_seq is None:
        # ultimate fallback – uniform vector (guaranteed to be valid)
        best_seq = np.ones(1000, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print("fallback to uniform vector")

    print(f"=== grid finished – best score {best_score:.6f} (len={len(best_seq)}) ===")

    # ------------------------------------------------------------------
    # 3) Short projected‑gradient refinement (few hundred iterations)
    # ------------------------------------------------------------------
    a = best_seq.copy()
    total_sum = float(np.sum(a))
    n = a.shape[0]
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # current convolution and its maximal entry
        conv = np.convolve(a, a)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # sub‑gradient of the active entry
        grad = _gradient_of_max_conv(a, k_max)

        # keep the sum constant by removing the mean component (optional)
        grad -= grad.mean()

        # simple diminishing step size
        step = 0.02 / (1.0 + iteration * 0.001)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)                     # enforce non‑negativity
        a_new = _project_onto_simplex(a_new, total_sum)   # keep ℓ₁‑norm fixed

        # occasional tiny jitter to escape flat regions
        if iteration % 200 == 0:
            jitter = np.random.randn(n) * 1e-3
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new = _project_onto_simplex(a_new, total_sum)

        # evaluate the candidate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # numeric trouble – discard the move
            continue

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"iter {iteration:5d}: improved score = {best_score:.6f}")

        # accept the move if it does not worsen the objective significantly
        if score <= best_score * 1.001:
            a = a_new
        else:
            # occasional rollback to the current best (helps escape plateaus)
            if np.random.rand() < 0.05:
                a = best_seq.copy()

    print("=== finished ===")
    print(f"final best score: {best_score:.6f}  length: {len(best_seq)}")
    return best_seq.tolist()
