def annealing_local_search(a_start, start_score, fft_len_conv, fft_len_g, n, deadline):
    best_a = a_start.copy()
    best_score = start_score
    cur_a = a_start.copy()
    cur_score = start_score
    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time
    init_temp = 0.001 * best_score  # maybe 0.001 of score

    while time.time() < deadline:
        # compute gradient
        fa = np.fft.fft(cur_a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = cur_a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[n - 1: n - 1 + n]

        # select source and target
        # we can select randomly among top 10% high gradient indices
        # and bottom 10% low gradient
        # to incorporate diversity.
        # get sorted indices
        # we can compute percentile thresholds.

        # top indices: those with grad > median maybe? but we can just pick argmax/min.
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break
        max_transfer = cur_a[i_hi]
        if max_transfer <= 1e-12:
            # pick random source with positive mass
            pos = np.where(cur_a > 1e-12)[0]
            if pos.size == 0:
                break
            i_hi = int(rng.choice(pos))
            max_transfer = cur_a[i_hi]
        # choose delta
        frac = rng.random() * 0.3
        delta = max_transfer * frac
        if delta <= 0.0:
            continue
        # create new candidate
        cand = cur_a.copy()
        cand[i_hi] -= delta
        cand[i_lo] += delta
        # Ensure non-negativity - possible delta > a[i_hi] ?
        # But delta <= a_i, ok.

        cand_score = fast_score(cand, fft_len_conv, n)

        # temperature schedule:
        elapsed = time.time() - start_time
        remaining = max(deadline - time.time(), 0.0)
        # linear cooling: T = init_temp * remaining / total_time
        T = init_temp * remaining / total_time if total_time > 0 else 0.0

        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(cand_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
        # Occasionally add noise
        if rng.random() < 0.01:
            cur_a = add_noise(cur_a, sigma=rng.uniform(0.0005, 0.005), rng=rng)
            cur_a = cur_a / cur_a.sum()
            cur_score = fast_score(cur_a, fft_len_conv, n)

    return best_a, best_score

"""
Search for a non‑negative step sequence that minimizes the evaluation
function 2·n·max(conv(a,a))/(sum(a)²).  The algorithm consists of:

1.  A quick parametric sweep over many analytic windows (power‑law,
    exponential, Gaussian, beta, sinusoidal, Hann/Hamming/Blackman).
2.  Random‑restart L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution, with a decreasing smoothing parameter schedule.
3.  Projected sub‑gradient descent with momentum on the true max‑convolution.
4.  Exponentiated‑gradient descent on the simplex.
5.  An annealing‑guided local search that repeatedly moves a small
    fraction of mass from the coordinate with the largest sub‑gradient
    (most harmful) to the one with the smallest (most helpful), accepting
    uphill moves according to a cooling schedule.
6.  A final deterministic hill‑climb (the same routine as previously
    used) for polishing.

All phases respect the global 1000 s budget.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a distribution a that already
    satisfies ∑a = 1.  The function works in O(n log n) time.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best `a` found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a (random) fraction of mass from the
    coordinate with the largest sub‑gradient to the one with the smallest.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # move a random fraction (up to 15 %) of the mass from i_hi to i_lo
        frac = rng.random() * 0.15
        delta = best[i_hi] * frac
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Candidate families for the cheap sweep
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def candidate_hann(n: int) -> np.ndarray:
    i = np.arange(n)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


def candidate_hamming(n: int) -> np.ndarray:
    i = np.arange(n)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def candidate_blackman(n: int) -> np.ndarray:
    i = np.arange(n)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Annealing‑guided local search
# ----------------------------------------------------------------------


def annealing_local_search(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Stochastic local search that moves mass from the most ‘harmful’
    coordinate (largest sub‑gradient) to the most ‘helpful’ one (smallest).
    Uphill moves are occasionally accepted according to a linear cooling schedule.
    """
    best_a = a_start.copy()
    best_score = start_score
    cur_a = a_start.copy()
    cur_score = start_score

    rng = np.random.default_rng()
    start_time = time.time()
    total_time = deadline - start_time
    init_temp = 0.001 * best_score  # a few ‑3 of the objective

    while time.time() < deadline:
        # convolution and mask for max locations
        fa = np.fft.fft(cur_a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = cur_a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source (largest grad) and target (smallest grad)
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # ensure source has non‑zero mass
        if cur_a[i_hi] <= 0.0:
            pos = np.where(cur_a > 1e-12)[0]
            if pos.size == 0:
                break
            i_hi = int(rng.choice(pos))

        # random fraction of the source mass (up to 30 %)
        frac = rng.random() * 0.30
        delta = cur_a[i_hi] * frac
        if delta <= 0.0:
            continue

        cand = cur_a.copy()
        cand[i_hi] -= delta
        cand[i_lo] += delta

        cand_score = fast_score(cand, fft_len_conv, n)

        # linear cooling schedule
        remaining = max(deadline - time.time(), 0.0)
        T = init_temp * remaining / total_time if total_time > 0 else 0.0

        accept = False
        if cand_score < cur_score:
            accept = True
        elif T > 0.0:
            prob = np.exp(-(cand_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()

        # occasional small random perturbation to escape plateaus
        if rng.random() < 0.01:
            cur_a = add_noise(cur_a, sigma=rng.uniform(5e-4, 5e-3), rng=rng)
            cur_a = cur_a / cur_a.sum()
            cur_score = fast_score(cur_a, fft_len_conv, n)

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise with a known construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    sweep_deadline = start_time + 0.02 * budget_s  # ~20 s

    def try_update(a: np.ndarray):
        nonlocal best_a, best_score
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sweep] improved fast_score={best_score:.6f}")

    # Power‑law families
    for p in np.linspace(-2.0, 5.0, 15):
        try_update(candidate_power_law(n, p))
        try_update(candidate_rev_power_law(n, p))

    # Exponential families
    for lam in np.linspace(0.0005, 0.02, 12):
        try_update(candidate_exponential(n, lam, reverse=False))
        try_update(candidate_exponential(n, lam, reverse=True))

    # Gaussian windows
    for sigma_frac in np.linspace(0.15, 0.45, 7):
        sigma = sigma_frac * n
        try_update(candidate_gaussian(n, sigma))

    # Beta‑like shapes
    for alpha in np.linspace(0.2, 5.0, 12):
        try_update(candidate_beta_symmetric(n, alpha))

    # Sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        try_update(candidate_sine_power(n, p))

    # Classic window functions
    try_update(candidate_hann(n))
    try_update(candidate_hamming(n))
    try_update(candidate_blackman(n))

    # Quick random restarts while time permits
    while time.time() < sweep_deadline:
        try_update(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.38 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006, 0.003,
        0.001, 0.0005, 0.0002, 0.0001,
    ]
    maxiter_per_lam = 1200  # a bit more than before

    while time.time() < phase1_end:
        if rng.random() < 0.2:
            a_start = add_noise(best_a,
                                sigma=rng.uniform(0.001, 0.015),
                                rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.58 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.78 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – annealing‑guided local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = annealing_local_search(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            deadline,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – deterministic hill‑climb for final polishing
    # ------------------------------------------------------------------
    best_a, best_score = gradient_guided_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_iter=40000,
    )
    print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
