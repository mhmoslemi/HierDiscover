"""
Propose a low‑evaluation step‑function sequence.

The optimizer works in four stages (within a 1000 s budget):
1.  Several short L‑BFGS runs on a log‑sum‑exp smoothed max‑convolution.
    This gives a good basin and helps escape poor local minima.
2.  A long projected sub‑gradient descent on the true L∞‑objective.
    The sub‑gradient of the max‑convolution is obtained by convolving the
    mask of maximal entries with the reversed current vector.
    An adaptive step size (∝ 1/√t) together with occasional tiny Gaussian
    perturbations yields steady improvement.
3.  Exponentiated‑gradient descent (mirror descent on the simplex) for
    fine‑grained polishing.
4.  A greedy hill‑climb that moves a random fraction of mass from a
    larger entry to a smaller entry, updating the full convolution
    incrementally (O(n) per move).  The move size decays over time.

All operations stay on the probability simplex (sum = 1, non‑negative);
the final sequence is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a (sum = 1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge',
         'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Phase‑1: L‑BFGS on log‑sum‑exp smoothed max‑convolution
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) wrt logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑prop through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS for each λ; return best a and its fast_score."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Phase‑2: Projected sub‑gradient descent on the true L∞‑objective
# ----------------------------------------------------------------------


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent using the exact max‑convolution.
    The step size is η = η0 / √t with occasional Gaussian perturbations.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    eta0 = 0.5          # initial step‑size factor
    perturb_interval = 1500
    sigma_perturb = 1e-4

    while time.time() < deadline:
        # FFT of a and its reversal
        a_fft = np.fft.fft(a, fft_len)
        a_rev = a[::-1]
        a_rev_fft = np.fft.fft(a_rev, fft_len)

        # full self‑convolution
        conv_full = np.real(np.fft.ifft(a_fft * a_fft))[: 2 * n - 1]

        max_val = conv_full.max()
        # mask of positions attaining the max (tolerance for numerical noise)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv_full >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution w.r.t. a
        mask_fft = np.fft.fft(mask, fft_len)
        grad_full = np.real(np.fft.ifft(mask_fft * a_rev_fft))
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # projected sub‑gradient step
        a_candidate = a - eta * grad
        a_candidate = simplex_projection(a_candidate)

        # evaluate
        score = fast_score(a_candidate, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()

        a = a_candidate
        it += 1

        # occasional tiny random perturbation to escape flat regions
        if it % perturb_interval == 0:
            a = add_noise(a, sigma_perturb, rng)
            a = simplex_projection(a)

        # optional decay of eta0 (helps fine tuning)
        if it % 5000 == 0 and eta0 > 0.02:
            eta0 *= 0.9

    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the L∞‑objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score

# Phase‑4: Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction_start: float = 0.20,
                           max_fraction_end: float = 0.02,
                           recompute_every: int = 1500) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that transfers a random fraction of mass from a
    larger entry to a smaller entry.  The move size decays linearly from
    max_fraction_start to max_fraction_end as time runs out.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)  # exact full convolution
    diff = np.zeros_like(conv)

    it = 0
    accepted = 0
    start_time = time.time()

    while time.time() < deadline:
        # linearly decay allowed fraction
        elapsed = time.time() - start_time
        total = deadline - start_time
        frac = max_fraction_end + (max_fraction_start - max_fraction_end) * (1 - elapsed / total)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger to smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1

    return best_a, best_score

    # ------------------------------------------------------------------

            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:

            best_a = a_opt.copy()

    # ------------------------------------------------------------------
    # Phase 2 – long projected sub‑gradient descent on the true max‑conv.
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_descent(
            best_a, phase2_end, n, fft_len, rng
        )
        print(f"[sub‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=10.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – intensive greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction_start=0.20,
            max_fraction_end=0.02,
            recompute_every=1500,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
