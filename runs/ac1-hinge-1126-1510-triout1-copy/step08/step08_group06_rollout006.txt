"""
Solution for the step‑function attack.

The search proceeds in five phases:

  1. Cheap parametric sweep over a large family of analytic shapes
     (power‑law, exponential, Gaussian, beta‑like, sinusoidal, triangular
     and logistic windows).  The best result is kept as the starting point.
  2. Aggressive multi‑restart L‑BFGS on a log‑sum‑exp smooth approximation
     of the max‑convolution.  The smoothing factor λ is decreased in a fine
     schedule and a large number of iterations per λ are allowed.
  3. Projected sub‑gradient descent with momentum (many more iterations than
     the reference implementation).
  4. Exponentiated‑gradient descent on the simplex (again with a generous
     iteration budget).
  5. Deterministic hill‑climb that repeatedly transfers a small fraction of
     mass from the coordinate with the largest sub‑gradient to the one with
     the smallest sub‑gradient.  The transferred fraction is chosen by a
     short line‑search, guaranteeing a strict improvement when possible.

All phases respect a global 1000 s time budget.  The final sequence is
returned as a plain list of floats and is evaluated with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective when sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the self‑convolution where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient w.r.t. a lives on indices n‑1 … 2n‑2 (the central n values)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y

# Optimisation primitives

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        scr = fast_score(a, fft_len_conv, n)
        if scr < best_score:
            best_score = scr
            best_a = a.copy()
            print(f"[L‑BFGS] improved fast_score={best_score:.6f}")
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decays like 1/√t, a slightly larger prefactor than before
        step = 0.9 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.4 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb.  Each iteration transfers a fraction of the
    mass from the coordinate with the largest sub‑gradient (most harmful)
    to the one with the smallest sub‑gradient (most helpful).  The transferred
    fraction is chosen by a tiny line‑search over a few candidates.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    # pre‑allocate buffers for FFTs to avoid repeated allocations
    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo or best[i_hi] <= 0.0:
            break

        # try a handful of fractions and keep the best improvement
        best_local = best_score
        best_delta = 0.0
        best_candidate = None
        for frac in (0.02, 0.05, 0.10, 0.15):
            delta = best[i_hi] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i_hi] -= delta
            cand[i_lo] += delta
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_local:
                best_local = cand_score
                best_delta = delta
                best_candidate = cand

        if best_candidate is not None:
            best = best_candidate
            best_score = best_local
        else:
            # if none of the tried fractions improve, fall back to a tiny random move
            delta = best[i_hi] * 0.005
            cand = best.copy()
            cand[i_hi] -= delta
            cand[i_lo] += delta
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "dirichlet", "exp", "power", "logistic", "triangular"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "dirichlet":
        # concentration parameter 1 (flat) gives a reasonable spread
        a = rng.dirichlet(np.ones(n))
    elif mode == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "logistic":
        centre = rng.integers(0, n)
        scale = rng.uniform(0.05 * n, 0.15 * n)
        x = np.arange(n)
        a = 1.0 / (1.0 + np.exp((x - centre) / scale))
    else:  # triangular
        peak = rng.uniform(0.5, 2.0) / n
        a = np.maximum(0.0, 1.0 - np.abs(np.arange(n) - n // 2) / (n // 2))
        a = a * peak
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise with the previously known best if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # keep a small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (many analytic families)
    # ------------------------------------------------------------------
    def try_update(a: np.ndarray):
        nonlocal best_a, best_score
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sweep] improved fast_score={best_score:.6f}")

    # Power‑law families (including decreasing)
    for p in np.linspace(-3.0, 5.0, 25):
        try_update(candidate_power_law(n, p))
        try_update(candidate_rev_power_law(n, p))

    # Exponential families (forward and reversed)
    for lam in np.linspace(0.0002, 0.03, 20):
        try_update(candidate_exponential(n, lam, reverse=False))
        try_update(candidate_exponential(n, lam, reverse=True))

    # Gaussian windows
    for sigma_frac in np.linspace(0.12, 0.45, 10):
        sigma = sigma_frac * n
        try_update(candidate_gaussian(n, sigma))

    # Symmetric beta‑like shapes
    for alpha in np.linspace(0.2, 6.0, 15):
        try_update(candidate_beta_symmetric(n, alpha))

    # Sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0, 8.0]:
        try_update(candidate_sine_power(n, p))

    # Triangular & logistic families
    for pf in [0.01, 0.02, 0.03]:
        try_update(candidate_triangular(n, peak_frac=pf))
    for centre in np.linspace(0.2, 0.8, 5):
        for scale in np.linspace(0.05, 0.2, 5):
            try_update(candidate_logistic(n, centre_frac=centre, scale_frac=scale))

    # A few Dirichlet random restarts (quick)
    while time.time() < start_time + 0.08 * budget_s:
        try_update(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Allocate time for the remaining phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS gets the biggest chunk
    phase2_end = start_time + 0.78 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.93 * budget_s   # exponentiated gradient
    # remaining time → deterministic hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS continuation with finer schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.13, 0.09, 0.06,
        0.04, 0.025, 0.018, 0.012, 0.008, 0.005,
        0.0035, 0.0025, 0.0015, 0.001, 0.0007,
        0.0005, 0.00035, 0.0002, 0.00012, 0.00008,
        0.00005, 0.00003, 0.00001,
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # jitter around the current best occasionally, otherwise a fresh Dirichlet sample
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.02), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new global best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum (more iters)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (more iters)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic gradient‑guided hill‑climb with line‑search
    # ------------------------------------------------------------------
    best_a, best_score = gradient_guided_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_iter=150000,
    )
    print(f"[hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the sequence for a possible next run (useful in a multi‑run setting)
    globals()["height_sequence_2"] = final_seq

    return final_seq
