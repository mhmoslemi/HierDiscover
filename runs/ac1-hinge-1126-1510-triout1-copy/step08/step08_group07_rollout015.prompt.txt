def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is normalised (∑a = 1):
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. convolution entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ----- back‑propagation through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to the derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    # start in the log‑softmax space
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(

    fft_len_g: int,

def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 150000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb that, for each chosen (source, target) pair,
    tries a few different mass‑transfer fractions and keeps the best.
    Only strictly improving moves are retained.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    # fractions of the source mass to try – they are dense enough to see
    # tiny improvements.
    fractions = [0.005, 0.01, 0.02, 0.04, 0.07, 0.10, 0.15]

    while time.time() < deadline and it < max_iter:
        # ----- current convolution and active mask -----
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # a slightly relaxed tolerance helps the gradient to see a larger
        # active set while still focusing on the true peak.
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the smoothed max (size n) -----
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- choose source / target based on gradient percentiles -----
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        # ----- try several fractions of the source mass -----
        improved = False
        best_local_score = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta

            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local_score - 1e-12:
                best_local_score = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local_score

        it += 1

    return best, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s
