from scipy.optimize import linprog
def solve_lp(f, M):
    n=len(f)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # i varies max(0, k-(n-1)) <= i <= min(n-1, k)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # constraints: A @ b <= M
    # objective: maximize sum(b) -> minimize -sum(b)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=(0, None), method='highs')
    b_opt = res.x
    return b_opt, res.success, res.fun
```

        # update temperature

        j = rng.integers(0, n)
        if i == j: continue
        max_frac = 0.1 # up to 10% of a[i]
        delta = a[i] * rng.random() * max_frac
        if delta <= 0:
            continue
        a[i] -= delta
        a[j] += delta
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score

            # accept move

            # accept with probability exp(-(score-best_score)/temp)
            if rng.random() < np.exp(-(score - best_score) / temp):
                # accept move
                pass
            else:
                # reject
                a[i] += delta
                a[j] -= delta
        # temperature decay
        it += 1
        if it % 1000 == 0:
            temp *= 0.995  # geometric decay
            temp = max(temp, 1e-8)
    return best_a, best_score

"""
Improved search for a low‑score non‑negative step sequence.

Algorithm overview
------------------
1. **Parametric sweep** – evaluate a wide collection of analytic families
   (power‑law, exponential, Gaussian, β‑like, sine‑power, central‑binomial,
   arcsine, tent).  The best candidate is kept as a warm‑start.
2. **Random‑restart L‑BFGS** – smooth the max‑convolution with a
   log‑sum‑exp surrogate and minimise it using L‑BFGS.  The schedule of
   smoothing parameters is long and each run performs many iterations;
   many restarts are performed until the allocated time elapses.
3. **Projected sub‑gradient descent** (with momentum) and **exponentiated‑
   gradient descent** further refine the solution.
4. **Simulated annealing** – a cheap stochastic local search that moves a
   random fraction of mass between two coordinates, accepting worsening
   moves with a decaying temperature.  This often pushes the solution
   past the deterministic hill‑climb barrier.
5. **Deterministic gradient‑guided hill‑climb** – final polishing step.
6. **Evaluation** – the best sequence found is fed to the provided
   `evaluate_sequence` and returned.

All heavy work respects the global time budget (default 1000 s).  The
implementation avoids closures and uses only NumPy, SciPy and the
standard library.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.special import gammaln

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1 (the simplex constraint).  Runs in O(n log n).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best `a` found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a small fraction of mass from the
    coordinate with the largest sub‑gradient (most harmful) to the one with
    the smallest sub‑gradient (most helpful).  Only strictly improving moves
    are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # move a random fraction (up to 15 %) of the mass from i_hi to i_lo
        frac = rng.random() * 0.15  # 0 → 0.15
        delta = best[i_hi] * frac
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep move
        else:
            # revert
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_score


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style stochastic local search.
    Randomly transfers a fraction of mass between two positions;
    worsening moves are accepted with a temperature that decays over time.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    rng = np.random.default_rng()
    it = 0
    temp = 1e-2  # initial temperature

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move up to 10 % of the mass from i to j
        max_frac = 0.10
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        a[i] -= delta
        a[j] += delta

        cur_score = fast_score(a, fft_len_conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # always accept improving move
        else:
            # accept with probability exp(-(Δ)/temp)
            if rng.random() < np.exp(-(cur_score - best_score) / max(temp, 1e-12)):
                # keep the worsening move
                pass
            else:
                # revert
                a[i] += delta
                a[j] -= delta

        # temperature schedule
        it += 1
        if it % 1000 == 0:
            temp *= 0.995
            temp = max(temp, 1e-8)

    return best_a, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Candidate families used in the cheap sweep
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def candidate_central_binomial(n: int) -> np.ndarray:
    """a_i ∝ binom(2i, i) / 4^i  (i starts at 0)."""
    i = np.arange(n, dtype=float)
    # log( binom(2i,i) ) = gammaln(2i+1) - 2*gammaln(i+1)
    log_coeff = gammaln(2 * i + 1) - 2 * gammaln(i + 1) - i * np.log(4.0)
    a = np.exp(log_coeff)
    a /= a.sum()
    return a


def candidate_arcsine(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt(i * (n - i + 1)), i = 1..n."""
    i = np.arange(1, n + 1, dtype=float)
    a = 1.0 / np.sqrt(i * (n - i + 1))
    a /= a.sum()
    return a


def candidate_tent(n: int) -> np.ndarray:
    """Symmetric tent (triangular) shape."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=float)
    if n % 2 == 0:
        a = np.concatenate([inc, inc[::-1]])
    else:
        dec = np.arange(half, 0, -1, dtype=float)
        a = np.concatenate([inc, np.array([half + 1], dtype=float), dec])
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a previously saved sequence if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (various families)
    # ------------------------------------------------------------------
    # power‑law families
    for p in np.linspace(-2.0, 5.0, 15):
        a = candidate_power_law(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] power p={p:.2f} → {best_score:.6f}")

        a = candidate_rev_power_law(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] rev‑power p={p:.2f} → {best_score:.6f}")

    # exponential families
    for lam in np.linspace(0.0005, 0.02, 12):
        for rev in (False, True):
            a = candidate_exponential(n, lam, reverse=rev)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[sweep] exp lam={lam:.4f} rev={rev} → {best_score:.6f}")

    # Gaussian windows
    for sigma_frac in np.linspace(0.15, 0.45, 7):
        sigma = sigma_frac * n
        a = candidate_gaussian(n, sigma)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] gauss σ={sigma_frac:.2f} → {best_score:.6f}")

    # symmetric beta‑like shapes
    for alpha in np.linspace(0.2, 5.0, 12):
        a = candidate_beta_symmetric(n, alpha)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] beta α={alpha:.2f} → {best_score:.6f}")

    # sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        a = candidate_sine_power(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] sine p={p:.1f} → {best_score:.6f}")

    # central‑binomial family
    a = candidate_central_binomial(n)
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[sweep] central‑binomial → {best_score:.6f}")

    # arcsine family
    a = candidate_arcsine(n)
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[sweep] arcsine → {best_score:.6f}")

    # tent shape
    a = candidate_tent(n)
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[sweep] tent → {best_score:.6f}")

    # quick random restarts
    while time.time() < start_time + 0.10 * budget_s:
        a = random_initial(n, rng)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] random → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time allocation for the remaining phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.85 * budget_s   # exponentiated‑gradient
    # remaining time → simulated annealing + final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS continuation
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
        0.0002,
        0.0001,
    ]
    maxiter_per_lam = 1600  # a bit more than the previous 800

    while time.time() < phase1_end:
        # jitter around the current best with probability 0.2
        if rng.random() < 0.2:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, deadline, fft_len_conv, n
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – deterministic gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    best_a, best_score = gradient_guided_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_iter=40000,
    )
    print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
