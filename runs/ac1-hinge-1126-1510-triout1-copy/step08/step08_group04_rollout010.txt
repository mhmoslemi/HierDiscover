```python
def gradient_descent(a_start, deadline, n, fft_len_conv, fft_len_g):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    # schedule for step size: initial eta = 0.1, decays 1/sqrt(it+1)
    while time.time() < deadline:
        # compute convolution and mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
        max_val = conv.max()
        # tolerance for mask: entries within 1e-12 relative
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        # compute gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_rev = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_rev[(n - 1):(n - 1 + n)]
        # step-size
        eta = 0.5 / np.sqrt(it + 1)  # tuned
        # additive update with projection onto simplex
        a_new = a - eta * g
        # Ensure non-negative; projection will handle negativity
        a_new = simplex_projection(a_new)
        # evaluate
        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score
```

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
        return float(2.0 * n * conv.max())

Propose a step‑function sequence with a low evaluation score.

Algorithm
---------

1. **Coarse search on a small problem** (n≈1024).  
   * Generate a diverse pool of starting vectors (uniform, interpolated
     previous solution, exponentials, power‑laws and random draws).  
   * Run a fast exponentiated‑gradient descent that works on the exact
     max‑convolution (the sub‑gradient is obtained by a simple slice
     operation, no extra FFT).  
   * Keep the best small‑scale vector.

2. **Upscale** the best small vector to the target size (n=4000) by linear
   interpolation and renormalise to sum = 1.

3. **Fine‑tune on the full size** using many rounds of exponentiated‑gradient
   descent.  The gradient is again computed from the current max‑convolution
   indices by a linear slice, which is O(n) and very cheap.  An adaptive
   step size  η≈0.5/√t  is used.  Every few thousand iterations a tiny
   Gaussian perturbation is injected to escape plateaus.

4. **Final polishing** – a greedy hill‑climbing phase transfers a small
   fraction of mass between random coordinates, accepting only improving
   moves.

All steps respect the simplex constraints (non‑negative entries, sum = 1).
The routine stops just before the 1000 s budget expires and returns the
best sequence found.  The returned list is evaluated with the official
`evaluate_sequence` function.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random distribution on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "logit_normal",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # logit_normal
        logits = rng.normal(0.0, 1.0, size=n)
        e = np.exp(logits - np.max(logits))
        a = e / e.sum()
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a on the simplex."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """Greedy hill‑climb moving a modest random fraction of mass."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects the 1000 s budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.02  # safety margin

    # ------------------------------------------------------------------
    # Phase A – coarse optimisation on a small instance
    # ------------------------------------------------------------------
    n_small = 1024
    fft_len_small = next_pow2(2 * n_small - 1)

    candidates = []

    # uniform
    candidates.append(np.full(n_small, 1.0 / n_small))

    # interpolated previous solution if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n_small)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            candidates.append(a)

    # exponential family
    for r in np.linspace(0.90, 0.9999, 12):
        a = (1.0 - r) * (r ** np.arange(n_small))
        s = a.sum()
        if s > 0:
            a /= s
        candidates.append(a)

    # power‑law family
    for p_exp in np.linspace(0.2, 2.5, 12):
        a = np.arange(1, n_small + 1, dtype=float) ** (-p_exp)
        a /= a.sum()
        candidates.append(a)

    # random draws
    for _ in range(20):
        candidates.append(random_initial(n_small, rng))

    def exp_grad_opt(a_start: np.ndarray, iters: int,
                     n: int, fft_len: int):
        """Exact exponentiated‑gradient descent on the hard max‑convolution."""
        a = a_start.copy()
        best = a.copy()
        best_score = fast_score(a, fft_len, n)
        for t in range(iters):
            # convolution
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-12
            mask_idx = np.where(conv >= max_val - eps)[0]

            # gradient (linear slice, no extra FFT)
            rev = a[::-1]
            g = np.zeros_like(a)
            for idx in mask_idx:
                offset = idx - (n - 1)
                if offset >= 0:
                    g[offset:] += 2.0 * rev[: n - offset]
                else:
                    off = -offset
                    g[: n - off] += 2.0 * rev[off:]

            eta = 1.0 / np.sqrt(t + 1.0)
            a = a * np.exp(-eta * g)
            s = a.sum()
            if s <= 0:
                a = best.copy()
            else:
                a /= s

            score = fast_score(a, fft_len, n)
            if score < best_score:
                best_score = score
                best = a.copy()
        return best, best_score

    best_small = None
    best_small_score = np.inf
    small_iters = 2000   # iterations per run on the small instance

    for a0 in candidates:
        a_opt, sc = exp_grad_opt(a0, small_iters, n_small, fft_len_small)
        if sc < best_small_score:
            best_small_score = sc
            best_small = a_opt

    # ------------------------------------------------------------------
    # Phase B – upscale to the target size
    # ------------------------------------------------------------------
    n_large = 4000
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    a_large = np.interp(x_large, x_small, best_small)
    a_large = np.maximum(a_large, 0.0)
    a_large /= a_large.sum()

    # ------------------------------------------------------------------
    # Phase C – intensive exponentiated‑gradient on the full size
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n_large - 1)

    best_a = a_large.copy()
    best_score = fast_score(best_a, fft_len_conv, n_large)
    print(f"[init] n={n_large}, fast_score={best_score:.6f}")

    it = 0
    while time.time() < deadline:
        # occasional perturbation to escape local plateaus
        if it % 5000 == 0 and it > 0:
            sigma = rng.uniform(0.0005, 0.003)
            best_a = add_noise(best_a, sigma, rng)

        # convolution and mask for the max entry
        fa = np.fft.rfft(best_a, fft_len_conv)
        conv = np.fft.irfft(fa * fa)[: 2 * n_large - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask_idx = np.where(conv >= max_val - eps)[0]

        # gradient via slice operation
        rev = best_a[::-1]
        g = np.zeros_like(best_a)
        for idx in mask_idx:
            offset = idx - (n_large - 1)
            if offset >= 0:
                g[offset:] += 2.0 * rev[: n_large - offset]
            else:
                off = -offset
                g[: n_large - off] += 2.0 * rev[off:]

        # adaptive step size
        eta = 0.5 / np.sqrt(it + 1.0)
        a_new = best_a * np.exp(-eta * g)
        s = a_new.sum()
        if s <= 0:
            a_new = best_a.copy()
        else:
            a_new /= s

        score = fast_score(a_new, fft_len_conv, n_large)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            print(f"[exp‑grad] iter {it:,} new best fast_score={best_score:.6f}")
        else:
            best_a = a_new.copy()
        it += 1

    # ------------------------------------------------------------------
    # Phase D – final polishing with greedy hill‑climbing
    # ------------------------------------------------------------------
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n_large,
                                    deadline,
                                    max_iter=50000)
    print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # provided by the environment
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
