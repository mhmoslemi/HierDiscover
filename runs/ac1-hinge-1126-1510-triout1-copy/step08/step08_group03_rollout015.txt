"""
Hybrid optimiser with an additional linear‑programming refinement
----------------------------------------------------------------
1. Initialise a population of candidate step functions (uniform, random,
   edge‑biased, interpolated previous best).
2. Heavy L‑BFGS on a log‑sum‑exp smooth approximation of the max‑convolution
   (multiple random restarts, continuation on the smoothing parameter).
3. Linear‑programming refinement: for the current a, maximise Σg under the
   linear constraints a∗g ≤ max(a∗a).  The optimiser returns a more spread
   distribution g which is convex‑combined with a.
4. Long exponentiated‑gradient descent on the exact max‑convolution (mirror
   descent on the simplex, adaptive step size).
5. Deterministic hill‑climb moving mass from the most harmful coordinate to
   the least harmful one, with a shrinking step size and occasional random
   swaps for escaping plateaus.
6. Return the best sequence before the global deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a⋆a) when a has been normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # exact convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float,
              rng: np.random.Generator) -> np.ndarray:
    """
    Add Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """
    Uniform initialise (kept for compatibility).
    """
    return np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Guided hill‑climb (deterministic improvement) + random swaps
# ----------------------------------------------------------------------
def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 0.001,
                      swap_trials: int = 500) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb moving mass from the coordinate that
    contributes most to the current maximal convolution entry to a
    coordinate with the smallest partner value.  After the deterministic
    phase a small number of random swaps are attempted to escape
    plateaus.
    """
    step = init_step
    while time.time() < deadline:
        # --- deterministic move ---
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        max_idx = np.argmax(conv)

        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # pick destination with smallest partner (ignore i_max)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a, score = a_new, new_score
            # keep step size for next move
            continue
        else:
            step = max(step * 0.8, min_step)

    # --- random swaps ----------------------------------------------------
    rng = np.random.default_rng()
    while time.time() < deadline and swap_trials > 0:
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] < 1e-12:
            swap_trials -= 1
            continue
        # move a small fraction from i to j
        delta = min(a[i] * 0.05, 0.001)
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a, score = a_new, new_score
        swap_trials -= 1

    return a, score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def lp_refine(a: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 12,
              t_init: float = 0.5,
              t_decay: float = 0.8) -> tuple[np.ndarray, float]:
    """
    For the current a, solve:
        maximise   Σ g_j
        subject to a∗g ≤ M   (M = max(a∗a)),   g ≥ 0
    and convex‑combine the solution with a.
    """
    fft_len = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = t_init

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current maximal convolution value
        conv = full_conv(a, fft_len)
        M = conv.max()

        # Build the Toeplitz matrix representing convolution with a
        # (lower‑triangular Toeplitz, shape (2n‑1) × n)
        offsets = -np.arange(n)                     # -0, -1, …, -(n‑1)
        data = [np.full(n, a[d]) for d in range(n)]  # each diagonal has n entries
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise Σ g → minimise -Σ g
        c = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None) for _ in range(n)]

        # Solve with HiGHS (fast for sparse problems)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # fall back to stopping the refinement
            print("[LP] failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g_norm = g / sum_g

        # convex combination
        a_candidate = (1.0 - t) * a + t * g_norm
        a_candidate = np.clip(a_candidate, 0.0, None)
        a_candidate /= a_candidate.sum()

        cand_score = fast_score(a_candidate, fft_len, n)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate                     # continue from improved point
            # be a little bolder after a success
            t = min(t * 1.1, 0.9)
            print(f"[LP] iter {it:02d} – new best score = {best_score:.6f}")
        else:
            t = max(t * t_decay, 0.05)          # become more conservative

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep the original length if a previous solution exists
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        n = max(prev.size, 4000)            # at least 4000 entries
    else:
        n = 4000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise candidates
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and prev.size == n:
        a_start = prev / prev.sum()
    elif 'height_sequence_1' in globals():
        # interpolate previous sequence to the new length
        old_x = np.linspace(0, 1, prev.size)
        new_x = np.linspace(0, 1, n)
        a_start = np.interp(new_x, old_x, prev)
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
    else:
        a_start = np.full(n, 1.0 / n)

    a0 = a_start.copy()
    a_spec = spectral_init(n)          # uniform – kept for compatibility
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    phase1_deadline = start_time + 0.25 * budget_s   # L‑BFGS
    phase2_deadline = start_time + 0.55 * budget_s   # LP refinement
    phase3_deadline = start_time + 0.85 * budget_s   # exp‑gradient
    phase4_deadline = start_time + 0.98 * budget_s   # hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation (multiple random restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.4:
            a_start = random_initial(n, rng)
        elif r < 0.8:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.015), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = lp_refine(
            best_a,
            n,
            phase2_deadline,
            max_iters=15,
            t_init=0.6,
            t_decay=0.85,
        )
        print(f"[LP‑refine] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (long run)
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – guided hill‑climb + random swaps
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_deadline,
            init_step=0.20,
            min_step=0.001,
            swap_trials=800,
        )
        print(f"[guided hill‑climb] final score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
