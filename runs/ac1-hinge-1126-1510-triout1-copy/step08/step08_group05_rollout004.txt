"""
Improved optimizer for the step‑function attack.

Main ideas
----------

*   Keep the original multi‑phase pipeline (heavy L‑BFGS, hill‑climb,
    fine L‑BFGS) but add a powerful **exponentiated‑gradient (mirror)
    descent** phase.  The sub‑gradient of the max‑convolution is obtained
    exactly (as in the original Adam implementation) and an update of
    the form

        a_i ← a_i · exp(‑η·g_i)   ,   a ← a / ∑a

    is performed.  This respects the simplex constraint without an
    explicit projection and generally yields larger objective reductions
    per iteration than the plain Adam steps.

*   Allocate the 1000 s budget conservatively:
        – 0 – 0.25 budget : several random L‑BFGS restarts (the
          “heavy” phase).
        – 0.25 – 0.70 budget : exponentiated‑gradient descent.
        – 0.70 – 0.90 budget : aggressive hill‑climb (the original
          implementation).
        – remaining time    : short fine‑tuned L‑BFGS.

*   The code works for any sequence length; the default is 4000
    (the length used in the reference run).  All heavy lifting is
    performed with FFT‑based convolutions, so the optimizer stays well
    inside the 1000 s limit.

The function ``propose_candidate`` returns the best list of non‑negative
floats found before the deadline.  The external ``evaluate_sequence``
function is called only once at the end for the final score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised
    (∑a = 1).  Returns 2·n·max(a*a) .
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (logits → a)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits, safe
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (adaptive moment estimation) using the true
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam updates
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 0:
                a = a / s

    return best_a, best_score


# ----------------------------------------------------------------------
# Aggressive gradient‑guided hill‑climb (kept from reference)
# ----------------------------------------------------------------------


def targeted_hill_climb_aggressive(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that uses the sub‑gradient to move mass from
    high‑gradient indices to low‑gradient ones.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        best_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(best_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        high_thr = np.percentile(g, 90)
        low_thr = np.percentile(g, 10)
        high_idx = np.where(g >= high_thr)[0]
        low_idx = np.where(g <= low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        improved = False
        for frac in (0.03, 0.07, 0.12):
            delta = best[i] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            new_score = fast_score(cand, fft_len_conv, n)
            if new_score < best_score:
                best = cand
                best_score = new_score
                improved = True
                break
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror) descent – the new phase
# ----------------------------------------------------------------------


def exponentiated_gradient_descent(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Mirror descent on the simplex using an exponentiated‑gradient update.
    The sub‑gradient of the max‑convolution is used.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # convolution and mask of active entries
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # step‑size schedule (inverse‑sqrt)
        it += 1
        eta = 0.5 / np.sqrt(it)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / n)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        # occasional jitter to escape tiny plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.004)
            a = best + rng.normal(0.0, sigma, size=n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum()

    return best, best_score


# ----------------------------------------------------------------------
# Main driver – the function that will be called by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser for the step‑function attack.
    Returns a list of non‑negative floats that (hopefully) improves
    upon the reference bound 1.506544.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (we keep the original 4000 – larger sizes would
    # increase the constant factor 2·n in the objective).
    # ------------------------------------------------------------------
    n = 4000

    # initialise from a previous solution if present, otherwise random
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2  # safety margin

    phase1_end = start + 0.25 * (budget_s - 0.2)   # heavy L‑BFGS
    phase2_end = start + 0.70 * (budget_s - 0.2)   # exponentiated‑gradient
    phase3_end = start + 0.90 * (budget_s - 0.2)   # aggressive hill‑climb
    # remainder → fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-05, 1e-05,
    ]
    maxiter_per_lam = 800  # a bit more than in the reference

    while time.time() < phase1_end:
        # random start (biased diversely)
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.015)
            a_start = best_a + rng.normal(0.0, sigma, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exponentiated_gradient_descent(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[EGD] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb (kept from reference)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = targeted_hill_climb_aggressive(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iter=70000,
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine L‑BFGS polishing with very small λ
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [1e-05, 5e-06, 2e-06, 1e-06, 5e-07]
        fine_maxiter = 500
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
