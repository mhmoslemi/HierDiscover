"""
Optimise a non‑negative step function for the evaluation metric

    score = 2·n·max(conv(a,a))  (a is normalised so sum(a)=1)

The optimiser combines several ideas that proved useful in earlier runs:

* a diverse set of initialisations (uniform, edge‑heavy, spectral,
  random Dirichlet and a continuation of the best solution so far);
* a continuation of L‑BFGS on a smooth log‑sum‑exp approximation of the
  max‑convolution (multiple λ stages, many iterations);
* Adam‑style mirror descent on the *logits* of the probability vector;
* an exact sub‑gradient / exponentiated‑gradient descent on the true
  max‑convolution;
* a long greedy hill‑climb that moves a small random amount of mass
  between two positions only when the objective improves.

All heavy work uses FFT‑based convolution (real FFT) for speed.
The routine respects the 1000 s time budget, always returning the best
sequence found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast score (exact objective for sum(a)=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed maximum of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt conv entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Square‑root of a uniform spectrum, projected onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u = np.zeros(L)
    u[:target_len] = uniform
    S = np.fft.fft(u)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Various random draws on the simplex."""
    mode = rng.choice(
        ["uniform", "dirichlet", "exp", "power", "edge", "bimodal", "beta"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "dirichlet":
        a = rng.dirichlet(np.ones(n))
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        # concentrate mass at the two extremes
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # beta distribution – very flexible shape
        a = rng.beta(0.5, 0.5, size=n)
    a /= a.sum()
    return a


def edge_heavy_init(n: int) -> np.ndarray:
    """Large values at both ends, low values in the centre."""
    i = np.arange(n, dtype=float)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    a /= a.sum()
    return a


def parametric_alpha_init(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^‑α + (n‑i)^‑α  (edge‑heavy with a single parameter)."""
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-alpha) + (n - i) ** (-alpha)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS continuation (smooth max)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # very rare – continue with the current best
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    lam_factor: float = 0.05,
):
    """
    Mirror‑descent (Adam) on the logits y with the log‑sum‑exp
    approximation of the max‑convolution. Returns best a found.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_y = y.copy()
    best_score = fast_score(a_start, fft_len, n)

    while time.time() < deadline:
        t += 1
        # gradient of smooth max w.r.t. logits
        f, grad_y = objective_and_grad(
            y, n, fft_len, fft_len, lam_factor=lam_factor
        )
        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.5 / np.sqrt(t)  # slowly decreasing step size
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_y = y.copy()
    best_a = softmax(best_y)
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """
    Exact exponentiated‑gradient descent (mirror descent with KL‑divergence).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        # step‑size schedule (larger initial step)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (polish)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 200_000,
):
    """
    Greedy hill‑climb: move a random fraction of the mass from one
    coordinate to another; accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move at most 10 % of the source mass, could be zero
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                       # good compromise between quality & speed
    fft_len = next_pow2(2 * n - 1)  # power‑of‑two for FFT speed

    # --------------------------------------------------------------
    # Build a pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # 0) previous best, if available
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n))

    # 2) spectral guess
    candidates.append(spectral_init(n))

    # 3) random Dirichlet‑like draw
    candidates.append(random_initial(n, rng))

    # 4) edge‑heavy deterministic init
    candidates.append(edge_heavy_init(n))

    # 5) a few parametric α‑shapes
    for a in [0.5, 1.0, 1.5, 2.0]:
        candidates.append(parametric_alpha_init(n, alpha=a))

    # pick the best of the raw pool
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on a smoothed max (continuation)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.001,
        0.0005,
        0.0002,
        0.0001,
    ]
    maxiter_per_lam = 1200  # a bit more than the previous 800

    while time.time() < phase1_deadline:
        # a fresh random start for each L‑BFGS run
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed objective (deeper refinement)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = adam_optimize(
            best_a, phase2_deadline, n, fft_len, rng, lam_factor=0.02
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, rng
        )
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (heavy polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=250_000,
        )
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result (converted to a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
