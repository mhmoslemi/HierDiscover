def random_shape(n, rng):
    mode = rng.choice([...]) (list of shape types)
    if mode == 'exp':
        # a_i ∝ exp(-α * i)
        alpha = rng.uniform(0.0005, 0.01) # some range
        a = np.exp(-alpha * np.arange(n))
    elif mode == 'exp_center':
        # a_i ∝ exp(-α * (i - n//2)^2) (Gaussian)
        sigma = rng.uniform(0.5, 0.25*n) # maybe use sqrt?
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - n/2)/sigma)**2)
    elif mode == 'sin_pow':
        p = rng.uniform(0.1, 4.0)
        a = np.sin(np.pi * (np.arange(1, n+1) / (n+1)))**p
    elif mode == 'triangular':
        # Linear ramp up and down
        height = rng.uniform(0.5, 5.0)
        # base width fraction
        frac = rng.uniform(0.6, 1.0)
        # create triangle shape
        x = np.linspace(-1,1,n)
        a = np.maximum(1 - np.abs(x)/frac, 0) * height
    elif mode == 'beta':
        # Beta distribution shape on [0,1]
        a = rng.beta(rng.uniform(0.2, 5.0), rng.uniform(0.2,5.0), size=n)
    elif mode == 'bimodal':
        # mixture of two gaussians
        sigma1 = rng.uniform(0.01*n, 0.1*n)
        sigma2 = rng.uniform(0.01*n, 0.1*n)
        mu1 = rng.uniform(0, n*0.4)
        mu2 = rng.uniform(n*0.6, n)
        i = np.arange(n)
        gauss1 = np.exp(-0.5*((i-mu1)/sigma1)**2)
        gauss2 = np.exp(-0.5*((i-mu2)/sigma2)**2)
        a = gauss1 + gauss2
        # optionally weight
        a *= rng.uniform(0.5,1.5)
    else:
        a = np.full(n, 1.0/n)
    # ensure non-negative and normalize
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a_sum
    return a

"""
Step‑function optimizer for the score 2·n·max(a*a)/(sum(a)^2).

Algorithm (≈1000 s budget)
-------------------------
1. **Initialisation** – reuse a previous solution if provided, otherwise start
   with a uniform distribution.

2. **Coarse random‑shape search** – generate many synthetic shapes
   (exponential, Gaussian, power‑law, sinusoidal, triangular, beta,
   bimodal, sparse).  Keep the top few candidates according to a fast
   FFT‑based proxy for the true score.

3. **Smooth L‑BFGS refinement** – for the current best candidate run a
   continuation of L‑BFGS on a log‑sum‑exp smoothed version of the max‑convolution.
   This quickly reaches a good basin.

4. **Exponentiated‑gradient descent** – perform a long EG‑descent on the exact
   L∞‑objective (sub‑gradient taken from the set of maximising lags).

5. **Greedy hill‑climb with incremental convolution updates** – move a
   fraction of mass from a larger entry to a smaller one; the convolution
   is updated in O(n) per move, with periodic re‑evaluation.

6. **Final evaluation** – compute the official score with
   `evaluate_sequence()` and return the best list found.

The search respects the 1000 s time limit and logs intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic FFT utilities (shared by several phases)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the true score: 2·n·max(a*a) (sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Random shape generators (coarse search)
# ----------------------------------------------------------------------


def random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a normalized non‑negative shape from a random family."""
    mode = rng.choice(
        [
            "uniform",
            "exp",
            "exp_center",
            "power",
            "sin_pow",
            "triangular",
            "beta",
            "bimodal",
            "sparse",
        ]
    )
    i = np.arange(n, dtype=float)

    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "exp":
        # a_i ∝ exp(-α i)
        alpha = rng.uniform(0.0005, 0.02)
        a = np.exp(-alpha * i)

    elif mode == "exp_center":
        # Gaussian‑like centred at n/2
        sigma = rng.uniform(0.02 * n, 0.2 * n)
        a = np.exp(-0.5 * ((i - n / 2) / sigma) ** 2)

    elif mode == "power":
        # a_i ∝ (i+1)^-p  (p>0)
        p = rng.uniform(0.3, 3.0)
        a = (i + 1.0) ** (-p)

    elif mode == "sin_pow":
        # a_i ∝ sin^p (π*i/(n+1))
        p = rng.uniform(0.2, 4.0)
        a = np.sin(np.pi * (i + 1) / (n + 1)) ** p

    elif mode == "triangular":
        # Symmetric triangle, height controlled by a factor
        width_frac = rng.uniform(0.5, 1.0)  # fraction of full width where >0
        height = rng.uniform(0.5, 3.0)
        x = (i - (n - 1) / 2) / ((n - 1) / 2)  # in [-1,1]
        a = np.maximum(1.0 - np.abs(x) / width_frac, 0.0) * height

    elif mode == "beta":
        # Discrete Beta distribution on {0,…,n‑1}
        a = rng.beta(rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0), size=n)

    elif mode == "bimodal":
        # Mixture of two Gaussians
        sigma1 = rng.uniform(0.01 * n, 0.1 * n)
        sigma2 = rng.uniform(0.01 * n, 0.1 * n)
        mu1 = rng.uniform(0.0, 0.4 * n)
        mu2 = rng.uniform(0.6 * n, n)
        g1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        blend = rng.uniform(0.3, 0.7)
        a = blend * g1 + (1 - blend) * g2

    else:  # sparse
        m = rng.integers(int(0.01 * n), int(0.07 * n) + 1)
        a = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=m, replace=False)
        a[idx] = rng.random(m) + 1e-6

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed L‑BFGS (log‑sum‑exp surrogate of max‑convolution)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective: 2·n·smooth_max(conv(a,a)), where a = softmax(y).
    Returns (value, gradient w.r.t. y).
    """
    # softmax -> probability vector on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z   # ∂smooth_max/∂conv_i

    # backprop through convolution: ∂conv/∂a = rev(a) * w   (linear conv)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]   # gradient w.r.t. a[i]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f_val, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run L‑BFGS on the smoothed objective for each λ in lam_schedule.
    Starts each stage from the previous result.
    Returns best vector (on simplex) and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logit representation

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 10.0,
):
    """
    Sub‑gradient EG descent on 2·n·max(conv(a,a))/(sum a)^2.
    Step size η = η0 / sqrt(t). Returns best vector found.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # full convolution via FFT (fast enough)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # size n

        eta = eta0 / np.sqrt(it + 1.0)

        # EG update (clipping exponent)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(
    a: np.ndarray,
    curr_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_fraction: float = 0.12,
    recompute_every: int = 1500,
):
    """
    Greedy hill‑climb moving a fraction of mass from a larger entry to a
    smaller one.  The convolution is updated incrementally (O(n) per move)
    and recomputed exactly every `recompute_every` accepted moves.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)  # exact full convolution
    diff = np.zeros_like(conv)
    accept_cnt = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger to smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] <= 0.0:
            continue

        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of convolution
        diff.fill(0.0)
        diff[src : src + n] -= best_a
        diff[dst : dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accept_cnt += 1

            if accept_cnt % recompute_every == 0:
                conv = np.convolve(best_a, best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – trade‑off between granularity and speed
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous solution if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0.0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # Phase lengths (percent of total budget)
    phase1_end = start_time + 0.15 * budget_s  # random shape search
    phase2_end = phase1_end + 0.25 * budget_s  # L‑BFGS refinement
    phase3_end = phase2_end + 0.35 * budget_s  # EG descent
    # remaining → hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – random shape exploration (keep top‑k candidates)
    # --------------------------------------------------------------
    TOP_K = 8
    top_candidates = [(best_a.copy(), best_score)]  # list of (a, score)

    while time.time() < phase1_end:
        a_rand = random_shape(n, rng)
        score = fast_score(a_rand, fft_len)

        # Update global best
        if score < best_score:
            best_score = score
            best_a = a_rand.copy()
            print(f"[rand] new global best fast_score={best_score:.6f}")

        # Insert into top‑k list
        if len(top_candidates) < TOP_K or score < top_candidates[-1][1]:
            # Insert while keeping list sorted by score
            inserted = False
            for idx, (_, sc) in enumerate(top_candidates):
                if score < sc:
                    top_candidates.insert(idx, (a_rand.copy(), score))
                    inserted = True
                    break
            if not inserted:
                top_candidates.append((a_rand.copy(), score))
            if len(top_candidates) > TOP_K:
                top_candidates.pop()  # drop worst

    # --------------------------------------------------------------
    # Phase 2 – L‑BFGS continuation on the current global best
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [
            0.5,
            0.35,
            0.25,
            0.18,
            0.12,
            0.08,
            0.05,
            0.03,
            0.018,
            0.01,
            0.006,
            0.003,
            0.0015,
            0.0008,
            0.0004,
            0.0002,
            0.0001,
            0.00005,
        ]
        maxiter_per_lam = 600
        best_a, best_score = lbfgs_continuation(
            best_a,
            lam_schedule,
            maxiter_per_lam,
            phase2_end,
            fft_len,
            n,
        )
        print(f"[L‑BFGS] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (hard max)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            eta0=10.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – intensive greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.15,
            recompute_every=1200,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Occasionally re‑try refinement on a few top random candidates
    # (This block runs only if time permits)
    # --------------------------------------------------------------
    extra_time = deadline - time.time()
    if extra_time > 20.0 and len(top_candidates) > 1:
        # Allocate a few seconds per extra candidate
        per_candidate = max(5.0, extra_time / (len(top_candidates) - 1))
        for idx, (cand_a, cand_score) in enumerate(top_candidates[1:], start=1):
            if time.time() >= deadline:
                break
            # Short L‑BFGS on the candidate
            cand_end = time.time() + min(per_candidate, deadline - time.time())
            cand_a_ref, cand_score_ref = lbfgs_continuation(
                cand_a,
                lam_schedule[:5],  # fewer λ's for speed
                400,
                cand_end,
                fft_len,
                n,
            )
            # Short EG descent
            if time.time() < cand_end:
                cand_a_ref, cand_score_ref = exp_gradient_optimize(
                    cand_a_ref,
                    cand_end,
                    n,
                    fft_len,
                    rng,
                    eta0=8.0,
                )
            # Final hill‑climb short
            if time.time() < cand_end:
                cand_a_ref, cand_score_ref = hill_climb_incremental(
                    cand_a_ref,
                    cand_score_ref,
                    fft_len,
                    n,
                    cand_end,
                    rng,
                    max_fraction=0.10,
                    recompute_every=1500,
                )
            if cand_score_ref < best_score:
                best_score = cand_score_ref
                best_a = cand_a_ref.copy()
                print(f"[extra #{idx}] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
