"""

The search consists of several stages within the 1000 s budget:

def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # keep only the linear part
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is normalised (∑a = 1):
        2·n·max(a*a)
    """
    conv = full_conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def smoothed_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12  # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. convolution entries -----
    w = exp_shift / Z  # d smooth_max / d conv_i

    # ----- back‑propagation through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to the derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

            print("[lbfgs] exception:", e)
            continue

        y = res.x
        # back to the simplex via softmax
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient
    of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- convolution a*a -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the max‑convolution (size n) -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- exponentiated‑gradient update -----
        eta = 0.20 / np.sqrt(it + 1.0)  # step‑size schedule
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def random_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    frac_schedule: list[float] = None,
    max_iter: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Pure random local search: repeatedly move a random fraction of mass
    from a source index to a destination index, accepting only strictly
    improving moves.  The fraction schedule is annealed (large moves early,
    tiny moves later).
    """
    if frac_schedule is None:
        frac_schedule = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]

    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # choose a fraction according to an annealing schedule
        # early iterations prefer larger fractions
        prob = (it + 1) / max_iter
        # linear interpolation between first and last fraction
        frac = np.interp(prob, [0.0, 1.0], [frac_schedule[0], frac_schedule[-1]])

        # pick source with non‑zero mass
        src_candidates = np.flatnonzero(best > 1e-14)
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        # pick destination different from source
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = best[i] * frac
        if delta <= 0.0:
            it += 1
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        it += 1

    return best, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex, including several
    parametric families (uniform, exponential, power‑law, Gaussian bump,
    arcsine, symmetric Beta).
    """
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "gaussian",
            "arcsine",
            "beta",
            "edge",
            "bimodal",
            "cosine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "gaussian":
        centre = n // 2
        sigma = rng.uniform(0.05, 0.25) * n
        i = np.arange(n)
        a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    elif mode == "arcsine":
        i = np.arange(n, dtype=float) + 0.5
        a = 1.0 / np.sqrt(i * (n - i + 1.0))
    elif mode == "beta":
        # symmetric Beta(alpha, alpha) discretisation
        alpha = rng.uniform(0.2, 4.0)
        i = np.arange(1, n + 1, dtype=float)
        a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001

    a = np.maximum(a, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a previous solution if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # small safety margin

    # allocate roughly 45 % to L‑BFGS (exploratory),
    # 20 % to exponentiated‑gradient,
    # 15 % to random hill‑climb,
    # 15 % to final polish L‑BFGS.
    phase1_dead = start_time + 0.45 * budget_s
    phase2_dead = start_time + 0.65 * budget_s
    phase3_dead = start_time + 0.80 * budget_s
    phase4_dead = deadline

    # ------------------------------------------------------------------
    # Phase 1 – generate seeds and run L‑BFGS continuation
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.2,
        0.1,
        0.05,
        0.02,
        0.01,
        0.005,
        0.002,
        0.001,
        0.0005,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
        1e-06,
        5e-07,
        2e-07,
        1e-07,
    ]
    maxiter_per_lam = 1500

    # keep a small pool of promising candidates
    pool = []
    pool_scores = []

    # generate an initial batch of diverse candidates
    for _ in range(20):
        a_start = random_initial(n, rng)
        score = fast_score(a_start, fft_len_conv, n)
        pool.append(a_start)
        pool_scores.append(score)

    # also add the previous best
    pool.append(best_a.copy())
    pool_scores.append(best_score)

    # sort pool by score and keep top 5
    top_idx = np.argsort(pool_scores)[:5]
    pool = [pool[i] for i in top_idx]
    pool_scores = [pool_scores[i] for i in top_idx]

    # run L‑BFGS on each pooled candidate
    for a_start in pool:
        if time.time() >= phase1_dead:
            break
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_dead,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score - 1e-12:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_dead:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_dead, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – random hill‑climb with annealed fractions
    # ------------------------------------------------------------------
    if time.time() < phase3_dead:
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_dead,
            frac_schedule=[0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005],
        )
        print(f"[random hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final fine L‑BFGS polish with ultra‑small lambda
    # ------------------------------------------------------------------
    if time.time() < phase4_dead:
        a_final, fine_score = lbfgs_continuation(
            best_a,
            [5e-08, 2e-08, 1e-08, 5e-09, 1e-09],
            maxiter_per_lam=600,
            deadline=phase4_dead,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score - 1e-12:
            best_score = fine_score
            best_a = a_final.copy()
            print(f"[final L‑BFGS] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # Store the result in a global variable for possible reuse
    globals()["height_sequence_1"] = final_seq

    return final_seq
