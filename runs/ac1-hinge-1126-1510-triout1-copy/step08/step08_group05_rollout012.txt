def targeted_hill_climb_line_search(a, current_score, fft_len_conv, n, deadline, max_iter=50000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    fractions = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute active mask and gradient as before
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        frev = np.fft.fft(rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(frev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source and target
        high_idx = np.where(grad > np.percentile(grad, 90))[0]
        low_idx = np.where(grad < np.percentile(grad, 10))[0]
        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = int(rng.choice(high_idx))
            j = int(rng.choice(low_idx))
        if i == j:
            it += 1
            continue
        # search best fraction
        best_local = None
        best_local_score = best_score
        for frac in fractions:
            delta = best[i] * frac
            if delta <= 0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            # renormalize to keep sum 1 (critical)
            s = cand.sum()
            if s == 0:
                continue
            cand /= s
            score = fast_score(cand, fft_len_conv, n)
            if score < best_local_score:
                best_local_score = score
                best_local = cand
        if best_local is not None:
            best = best_local
            best_score = best_local_score
        it += 1
    return best, best_score

"""
Propose a step‑function (non‑negative vector) that minimises the
provided ``evaluate_sequence`` objective.

The method is a four‑phase hybrid optimiser:

1. **Heavy L‑BFGS** – optimisation of a smooth log‑sum‑exp
   surrogate of the max‑convolution.  A decreasing schedule of the
   smoothing parameter λ guides the solution towards the true
   non‑smooth objective.

2. **Linear‑programming improvement** – given the current vector *f*,
   solve a linear programme that finds a partner *g* with maximal
   ℓ₁‑norm under the constraint ``f * g ≤ max(f*f)`` (convolution
   interpreted as linear constraints).  ``g`` is renormalised and
   mixed with *f*; this step is very effective at lowering the
   max‑convolution.

3. **Aggressive hill‑climb with line‑search** – using the exact
   sub‑gradient of the max‑convolution we transfer a fraction of mass
   from high‑gradient indices to low‑gradient ones, trying a small
   grid of transfer fractions and keeping the best improving move.

4. **Fine L‑BFGS polish** – a short continuation with very small
   λ values to polish the final candidate.

All phases respect a 1000 s budget and work with the vector already
normalised to unit ℓ₁‑norm (the objective is scale‑invariant).  The
returned list is ready for ``evaluate_sequence``.
"""

import time

from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised
    (∑a = 1).  Returns ``2·n·max(a*a)``.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse distribution on the simplex."""
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal"])
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)            # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming based improvement
# ----------------------------------------------------------------------


def lp_improvement_step(f: np.ndarray, n: int, fft_len: int, max_iter: int = 10):
    """
    Perform a few LP steps that try to enlarge the ℓ₁‑norm of a partner g
    while keeping ``f * g`` bounded by the current max‑convolution.
    Returns the updated vector and its fast_score.
    """
    best = f.copy()
    best_score = fast_score(best, fft_len, n)

    for it in range(max_iter):
        # current bound M
        conv_f = full_conv(best, fft_len)
        M = conv_f.max()

        # Build sparse constraint matrix A (size (2n‑1) × n)
        # each diagonal offset i contains the constant best[i]
        offsets = np.arange(n)
        A = diags(best, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear programme: maximise sum(b)  ⇒ minimise -sum(b)
        c = -np.ones(n)
        b_ub = M * np.ones(2 * n - 1)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            # LP failed – abort further LP steps
            print("[LP] optimisation failed:", res.message)
            break

        b = res.x
        s = b.sum()
        if s < 1e-12:
            break

        # Normalise the partner to unit ℓ₁‑norm
        g = b / s

        # Mix with the current vector (mixing factor gradually reduces)
        t = 0.5 / (it + 1)          # decreasing step size
        cand = (1 - t) * best + t * g
        cand /= cand.sum()          # enforce unit sum

        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[LP iter {it}] improved score = {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Aggressive hill‑climb with line‑search
# ----------------------------------------------------------------------


def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 60000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that transfers a fraction of mass from high‑gradient
    to low‑gradient indices.  For each chosen pair a small line‑search
    over several fractions is performed; only moves that strictly improve
    the fast_score are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    fractions = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
    it = 0

    while time.time() < deadline and it < max_iter:
        # convolution and active mask
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        frev = np.fft.fft(rev, fft_len)
        conv_g = np.real(np.fft.ifft(frev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source (high gradient) and target (low gradient)
        high_idx = np.where(grad > np.percentile(grad, 90))[0]
        low_idx = np.where(grad < np.percentile(grad, 10))[0]
        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = int(rng.choice(high_idx))
            j = int(rng.choice(low_idx))

        if i == j:
            it += 1
            continue

        # try several transfer fractions, keep the best improvement
        improved = False
        best_local = None
        best_local_score = best_score

        for frac in fractions:
            delta = best[i] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            # renormalise to keep sum = 1
            s = cand.sum()
            if s == 0:
                continue
            cand /= s
            score = fast_score(cand, fft_len, n)
            if score < best_local_score:
                best_local_score = score
                best_local = cand
                improved = True

        if improved:
            best = best_local
            best_score = best_local_score

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimisation for the step sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000  # primary size – a good compromise between quality and speed

    # initialise from a previously discovered construction if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()           # normalise to ℓ₁‑norm = 1

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2          # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS (smooth surrogate)
    # ------------------------------------------------------------------
    phase1_deadline = start + 0.25 * (budget_s - 0.2)
    lam_schedule = [
        0.6, 0.4, 0.25, 0.15, 0.09, 0.05,
        0.03, 0.015, 0.008, 0.004, 0.002, 0.001,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_deadline:
        # occasional random perturbation to keep diversity
        if rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.012)
            a_start = a0 + rng.normal(0.0, sigma, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len,
            fft_len,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best score={best_score:.6f}")

        if best_score <= 1.503:      # early exit if target already reached
            break

    # ------------------------------------------------------------------
    # Phase 2 – linear‑programming improvement
    # ------------------------------------------------------------------
    phase2_deadline = start + 0.60 * (budget_s - 0.2)
    if time.time() < phase2_deadline:
        best_a, best_score = lp_improvement_step(
            best_a, n, fft_len, max_iter=12
        )
        print(f"[LP phase] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb with line‑search
    # ------------------------------------------------------------------
    phase3_deadline = start + 0.85 * (budget_s - 0.2)
    if time.time() < phase3_deadline:
        best_a, best_score = targeted_hill_climb_line_search(
            best_a,
            best_score,
            fft_len,
            n,
            phase3_deadline,
            max_iter=70000,
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine‑tuned L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [1e-05, 5e-06, 2e-06, 1e-06, 5e-07]
        fine_maxiter = 500
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len,
            fft_len,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
