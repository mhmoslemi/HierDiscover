"""
Propose a low‑score non‑negative step sequence.

Algorithm
---------
1.  Choose a moderately large size (default n = 8000).  A larger
    n gives a smaller theoretical lower bound 2·n/(2n‑1) → 1,
    so we have room to improve over the 1.5‑range achieved before.
2.  Initialise a few diverse candidates:
        * Uniform / interpolated previous best.
        * Spectral “sqrt‑uniform’’ guess.
        * Random Dirichlet‑like draw.
        * Random‑phase spectral guess (constant magnitude, random phase).
3.  Refine each start with a sequence of L‑BFGS runs.
    The objective is a *smoothed* version of the true score:
        f(a) = 2·n·( smooth_max(conv(a,a)) + μ·Var(conv(a,a)) )
    where `smooth_max` is a log‑sum‑exp approximation of max,
    and a small variance penalty (μ≈0.02) pushes the convolution
    towards a flat shape, lowering its peak.
4.  Long exponentiated‑gradient descent on the exact max‑convolution.
5.  Projected sub‑gradient descent (the “new ingredient’’) for further polishing.
6.  A final greedy hill‑climb that moves a tiny random fraction of mass
    between coordinates and accepts only improving moves.
7.  Return the best sequence found before the deadline.

All heavy work uses real FFTs; the optimisation works on the
soft‑max parametrisation (so the L‑1 norm is always 1) and projects
onto the probability simplex when required.  The routine respects the
1000 s budget and prints progress information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp max) with variance penalty
# ----------------------------------------------------------------------


def smooth_max_var_obj_grad(y: np.ndarray, n: int,
                            fft_len_conv: int, fft_len_g: int,
                            lam: float, mu: float):
    """
    Returns (objective, gradient) for the smooth surrogate:
        f = 2·n·(smooth_max(conv) + μ·Var(conv))

    Parameters
    ----------
    y : np.ndarray
        Log‑softmax parameters (a = softmax(y)).
    lam : float
        Smoothing factor for the max (larger ⇒ smoother).
    mu : float
        Weight of the variance penalty (0 ≤ μ).
    """
    a = softmax(y)

    # ----- convolution a*a (full) ------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    # ----- smooth maximum -------------------------------------------
    max_c = conv.max()
    # lam_scaling keeps the smoothing proportional to the current scale
    lam_scaled = max_c * lam + 1e-12
    shifted = conv - max_c                      # ≤ 0
    exp_shift = np.exp(shifted / lam_scaled)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_scaled * np.log(Z)

    # ----- variance term ---------------------------------------------
    avg = conv.mean()
    var = ((conv - avg) ** 2).mean()

    # ----- total objective -------------------------------------------
    f = 2.0 * n * (smooth_max + mu * var)

    # ----- gradient w.r.t. conv --------------------------------------
    # derivative of smooth_max
    w_max = exp_shift / Z                      # d smooth_max / d conv_i
    # derivative of variance: 2*(conv_i-avg)/M, where M = len(conv)
    m = conv.size
    w_var = 2.0 * (conv - avg) / m
    # combined weight
    w = w_max + mu * w_var

    # ----- back‑propagate through the convolution (a*a) -------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n) – the factor 2 accounts for the two
    # symmetric contributions of a in the bilinear form
    g_vec = 2.0 * conv_rev[(n - 1):(n - 1 + n)]

    # ----- chain rule through the soft‑max ---------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len_conv: int,
                 fft_len_g: int,
                 n: int,
                 mu: float):
    """
    Repeated L‑BFGS over a schedule of smoothing parameters.
    Returns best (a, score) seen.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=smooth_max_var_obj_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam, mu),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] lam={lam:.4g} new best score={best_score:.6f}")

    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # diminishing step‑size
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

    it = 0
    step0 = 0.3

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # direction that reduces the max while preserving the sum
        d = -(g - g.mean())  # sum(d)=0

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        score_new = fast_score(a_new, fft_len, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: relocate a small random fraction of mass.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of sqrt of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    sqrt_amp = np.sqrt(np.abs(S))
    sqrt_phase = np.angle(S) / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random palette on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                       # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def random_phase_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Constant‑magnitude FFT with random phase, then project onto simplex.
    """
    L = next_pow2(2 * n - 1)
    # magnitude = 1 for all frequencies (except DC, which we set to sqrt(n))
    mag = np.ones(L // 2 + 1)
    # ensure the DC component yields a non‑zero sum after IFFT
    mag[0] = np.sqrt(L)
    phases = rng.uniform(0.0, 2 * np.pi, size=mag.shape)
    A = mag * np.exp(1j * phases)
    # enforce Hermitian symmetry for a real signal
    if L % 2 == 0:
        # even length – the Nyquist bin must be real
        A[-1] = mag[-1] * np.exp(1j * phases[-1]).real
    a_complex = np.fft.irfft(A, n=L)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # ------------------------------------------------------------------
    # Global RNG
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimension – a larger n gives a tighter theoretical bound.
    # ------------------------------------------------------------------
    n = 8000                              # feel free to lower if time is tight
    fft_len = next_pow2(2 * n - 1)        # power‑of‑two for FFT speed

    # ------------------------------------------------------------------
    # Initialise a variety of candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1) Interpolated previous best (if any)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2) Uniform start (simple baseline)
    candidates.append(np.full(n, 1.0 / n))

    # 3) Spectral sqrt‑uniform guess
    candidates.append(spectral_init(n))

    # 4) Random Dirichlet‑style draw
    candidates.append(random_initial(n, rng))

    # 5) Random‑phase spectral guess
    candidates.append(random_phase_init(n, rng))

    # Keep the best among the raw starts
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS refinement with variance penalty
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005, 0.0002]
    maxiter_per_lam = 600
    mu = 0.02                     # variance‑penalty weight

    while time.time() < phase1_deadline:
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_refine(a_start, lam_schedule,
                                     maxiter_per_lam,
                                     phase1_deadline,
                                     fft_len, fft_len, n, mu)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the exact max
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (tightening)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a, best_local_score = subgradient_optimize(best_a, deadline, n, fft_len)
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = a
        print(f"[subgrad] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=30000)
        print(f"[hill‑climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the final sequence (list of floats)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
