def exp_gradient_optimize(a_start, deadline, n, fft_len):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # identify entries achieving max (mask)

        # gradient of hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n-1):(n-1+n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        a = a * exp_term
        s = a.sum()

            a = best_a.copy()
        else:
            a = a / s
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def hill_climb(mask, counts, m, max_iters, deadline, n, rng):
    # mask: bool array of length n, counts: int array length 2n-1
    best_mask = mask.copy()
    best_counts = counts.copy()
    best_max = counts.max()
    best_score = 2 * n * best_max / (m*m)

    cur_mask = mask.copy()
    cur_counts = counts.copy()
    cur_max = best_max

    cur_idx = np.where(cur_mask)[0]

    it = 0
    while time.time() < deadline and it < max_iters:
        # randomly choose removal
        i_remove = int(rng.choice(cur_idx))
        # choose addition
        # we generate random index until not in mask
        i_add = int(rng.integers(0, n))
        if cur_mask[i_add]:
            # attempt a few times to find a non-mask index
            tries = 0
            while tries < 5 and cur_mask[i_add]:
                i_add = int(rng.integers(0, n))
                tries += 1
            if cur_mask[i_add]:
                it += 1
                continue
        # compute new counts copy
        new_counts = cur_counts.copy()
        # removal contributions
        idxs = cur_idx
        sums = i_remove + idxs
        same = (idxs == i_remove)
        dec = np.where(same, 1, 2)

        # addition contributions
        idxs_without = idxs[idxs != i_remove]
        sums_add = i_add + idxs_without
        np.add.at(new_counts, sums_add, 2)
        new_counts[2*i_add] += 1

        new_max = new_counts.max()
        new_score = 2 * n * new_max / (m*m)

        # acceptance rule: accept if new_score <= cur_score or accept with small probability
        cur_score = 2 * n * cur_max / (m*m)
        if new_score < cur_score:
            # accept
            cur_mask[i_remove] = False
            cur_mask[i_add] = True
            cur_counts = new_counts
            cur_max = new_max
            cur_idx = np.where(cur_mask)[0]
        # also record best
        if new_score < best_score:
            best_score = new_score
            best_mask = cur_mask.copy()
            best_counts = cur_counts.copy()
            best_max = cur_max
        it += 1
    return best_mask, best_counts, best_score

Combinatorial construction for a low‑∞‑norm self‑convolution.

We build a binary support set `S ⊂ {0,…,n‑1}` of size `m` (≈150) such that
the linear convolution of its indicator vector with itself has a tiny
maximum entry.  The step‑function heights are taken equal on `S`
(`a_i = 1/m` for i∈S, 0 otherwise); this respects the required
normalisation (∑a_i = 1) and yields a simple, non‑negative sequence.

Construction pipeline
----------------------
1. **Greedy support construction** – iteratively add the index that
   causes the smallest increase of the current maximum convolution
   count.  Counts are kept as integers (ordered pair counts), so the
   objective `max_counts / m²` equals the true evaluation metric.

   support with an unused index, accepting only moves that lower the
   current maximum count.  The best mask seen during the walk is kept.
3. **Fine‑tuning of the weights** – a short exponentiated‑gradient
   descent (projected on the simplex) starting from the equal‑weight
   vector can further reduce the exact max‑convolution.
4. The best vector found before the time budget expires is returned.

The algorithm respects the 1000 s overall budget, works entirely with
NumPy, and never writes to the file system.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility routines
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised
    (∑a_i = 1).  Convolution is performed via FFT.
    """
    conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) *
                               np.fft.fft(a, fft_len)))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (uses the exact max‑convolution)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy construction of a binary mask with small max convolution count
# ----------------------------------------------------------------------


def greedy_mask(m: int, n: int, rng: np.random.Generator):
    """
    Build a support set of size m by repeatedly adding the index that
    yields the smallest possible increase of the current maximum ordered
    pair count (i.e. the max entry of the integer convolution of the
    indicator vector with itself).
    Returns the boolean mask and the final count array.
    """
    mask = np.zeros(n, dtype=bool)
    counts = np.zeros(2 * n - 1, dtype=np.int32)

    # first element – random
    first = int(rng.integers(0, n))
    mask[first] = True
    cur_idx = np.array([first], dtype=np.int32)
    counts[2 * first] = 1          # self‑pair (ordered) contributes 1
    cur_max = 1

    for _ in range(1, m):
        candidates = np.where(~mask)[0]               # indices not yet used
        P = candidates.astype(np.int32)               # shape (C,)

        # sums of each candidate with all current indices (broadcast)
        sums = P[:, None] + cur_idx[None, :]           # (C, k)
        cnts = counts[sums]                            # (C, k) – current counts at those sums
        max_existing = cnts.max(axis=1) + 2            # each new ordered pair adds 2
        self_vals = counts[2 * P] + 1                  # contribution of the new self‑pair

        cand_max = np.maximum(max_existing, self_vals)
        cand_max = np.maximum(cand_max, cur_max)       # also compare with the unchanged maximum

        best_i = int(np.argmin(cand_max))
        best_p = int(P[best_i])

        # update data structures with the chosen index
        mask[best_p] = True
        sums_add = best_p + cur_idx
        counts[sums_add] += 2
        counts[2 * best_p] += 1

        cur_idx = np.append(cur_idx, best_p)
        cur_max = counts.max()

    return mask, counts


# ----------------------------------------------------------------------
# Local swap‑based hill‑climb on a fixed‑size mask
# ----------------------------------------------------------------------


def hill_climb(mask: np.ndarray,
               counts: np.ndarray,
               m: int,
               max_iters: int,
               deadline: float,
               n: int,
               rng: np.random.Generator):
    """
    Randomly swap one element of the support with a non‑support element.
    Accept only moves that improve the current maximum count.
    The best mask (and its score) encountered is returned.
    """
    best_mask = mask.copy()
    best_counts = counts.copy()
    best_max = counts.max()
    best_score = 2.0 * n * best_max / (m * m)

    cur_mask = mask.copy()
    cur_counts = counts.copy()
    cur_max = best_max
    cur_idx = np.where(cur_mask)[0]

    it = 0
    while time.time() < deadline and it < max_iters:
        # pick an element to remove
        i_remove = int(rng.choice(cur_idx))
        # pick an element to add (simple rejection sampling)
        i_add = int(rng.integers(0, n))
        tries = 0
        while cur_mask[i_add] and tries < 5:
            i_add = int(rng.integers(0, n))
            tries += 1
        if cur_mask[i_add]:
            it += 1
            continue

        # propose the swap
        new_counts = cur_counts.copy()

        # removal part (subtract contributions of i_remove)
        idxs = cur_idx                     # includes i_remove
        sums_rem = i_remove + idxs
        same = (idxs == i_remove)
        dec = np.where(same, 1, 2)          # self‑pair contributes 1, others 2
        np.subtract.at(new_counts, sums_rem, dec)

        # addition part (add contributions of i_add)
        idxs_without = idxs[idxs != i_remove]
        sums_add = i_add + idxs_without
        np.add.at(new_counts, sums_add, 2)
        new_counts[2 * i_add] += 1         # self‑pair of the new index

        new_max = new_counts.max()
        new_score = 2.0 * n * new_max / (m * m)
        cur_score = 2.0 * n * cur_max / (m * m)

        # accept if strictly better for the current state
        if new_score < cur_score:
            cur_mask[i_remove] = False
            cur_mask[i_add] = True
            cur_counts = new_counts
            cur_max = new_max
            cur_idx = np.where(cur_mask)[0]

        # keep the globally best mask seen
        if new_score < best_score:
            best_score = new_score
            best_mask = cur_mask.copy()
            best_counts = cur_counts.copy()
            best_max = cur_max

        it += 1

    return best_mask, best_counts, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    n = 4000                         # length of the step function
    fft_len = next_pow2(2 * n - 1)   # FFT length for convolutions

    best_score = np.inf
    best_a = np.full(n, 0.0)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1: combinatorial mask search
    # ------------------------------------------------------------------
    # try several support sizes; larger support yields smaller weights
    # but may increase the maximal count – we explore a range.
    m_candidates = list(range(140, 201, 5))  # 140,145,…,200

    for m in m_candidates:
        # allocate roughly 10 % of the remaining budget to each m
        phase_deadline = min(deadline, time.time() + (deadline - start_time) * 0.10)
        attempts = 0
        while time.time() < phase_deadline and attempts < 12:
            attempts += 1

            # ----- greedy construction -----
            mask, counts = greedy_mask(m, n, rng)
            cur_max = counts.max()
            cur_score = 2.0 * n * cur_max / (m * m)

            # ----- local swap improvement -----
            mask, counts, cur_score = hill_climb(mask, counts, m,
                                                max_iters=5000,
                                                deadline=deadline,
                                                n=n, rng=rng)

            if cur_score < best_score:
                best_score = cur_score
                best_a = mask.astype(float) / m
                print(f"[candidate] m={m} attempt={attempts} score={best_score:.6f}")

                # stop early if we already beat the target
                if best_score <= 1.503:
                    break
        if best_score <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 2: fine‑tune the weights (optional, time‑permitting)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_score > 1.5:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, fft_len, rng)
        print(f"[refine] after exponentiated‑gradient score={best_score:.6f}")
