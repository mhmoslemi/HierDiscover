"""
Propose a step‑function sequence with a low evaluation score.

The algorithm is a multi‑phase search:
1. **Broad parametric sweep** over several families (exponential,
   power‑law, beta, sinusoid, random Dirichlet).  The best few candidates
   are retained.
2. **Smooth‑max optimisation** – each retained candidate is refined by
   a continuation of L‑BFGS on a log‑sum‑exp approximation of the
   max‑convolution.  A fine λ‑schedule (down to 1e‑6) and many inner
   iterations are used.
3. **Projected sub‑gradient descent** on the true max‑convolution,
   starting from the best result of step 2.
4. **Pairwise mixing** – convex combinations of the top candidates are
   examined on a fine grid; mixing can often flatten the convolution
   further.
5. **Simulated‑annealing hill‑climb** with larger moves (up to 30 % of
   the source mass) to escape any remaining local minima.
6. Final evaluation with the provided `evaluate_sequence`.
All phases respect a global 1000 s budget; intermediate best scores are
printed for diagnostics.
"""

from __future__ import annotations
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures, all top‑level)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)−1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a sequence that already satisfies ∑a = 1."""
    conv = full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def smooth_obj_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(a*a) with a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth max w.r.t. each convolution entry
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    grad_conv = np.real(np.fft.ifft(fa_rev * fw))
    # only the central n entries influence the gradient
    g_vec = grad_conv[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=smooth_obj_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam, "gtol": 1e-14, "disp": False},
        )
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
    return best_a, best_score


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with a simple adaptive step scheme.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # true convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # mask of active constraints (allow a tiny tolerance)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        # step size – start larger and decay as 1/√it
        step = 0.8 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * g)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a random *fraction* of the mass
    from a source index to a destination index.  The temperature
    decays linearly with the remaining time budget.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = 0.02 * start_score  # initial temperature
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # linearly decreasing temperature
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        if a[i] == 0.0:
            continue
        # pick a different destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector drawn from a simple family."""
    t = np.arange(n, dtype=float)

    if family == "exp":
        # rate in a wide range, include very slow decay
        rate = rng.uniform(0.0001, 0.01)
        a = np.exp(-rate * t)

    elif family == "power":
        # power‑law with optional reversal
        p = rng.uniform(-4.0, 4.0)
        inc = rng.choice([True, False])
        if inc:
            a = (t + 1) ** p
        else:
            a = (n - t) ** p

    elif family == "beta":
        a = rng.uniform(0.1, 5.0, size=2)
        alpha, beta = a
        # discrete Beta(α,β) on {0,…,n‑1}
        a = (t + 1) ** (alpha - 1) * (n - t) ** (beta - 1)

    elif family == "sin":
        # sin‑shaped (concave) with optional exponent
        exp = rng.uniform(0.5, 3.0)
        a = np.sin(np.pi * (t + 0.5) / n) ** exp
        a = np.maximum(a, 0.0)

    elif family == "mix":
        # mixture of two exponentials (left‑right asymmetry)
        rate1 = rng.uniform(0.0001, 0.02)
        rate2 = rng.uniform(0.0001, 0.02)
        a = np.exp(-rate1 * t) + np.exp(-rate2 * (n - 1 - t))

    else:  # random Dirichlet‑like
        a = rng.random(n) ** rng.uniform(0.3, 2.0)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
    keep_top: int = 10,
) -> list[tuple[np.ndarray, float]]:
    """
    Generate many random candidates from the families above, keep the
    `keep_top` best (by fast_score) and return them.
    """
    best = []  # list of (a, score), sorted by score asc
    start = time.time()
    deadline = start + budget_s * 0.15  # 15 % of total budget

    families = ["exp", "power", "beta", "sin", "mix", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if len(best) < keep_top:
            best.append((a, sc))
            best.sort(key=lambda x: x[1])
        elif sc < best[-1][1]:
            best[-1] = (a, sc)
            best.sort(key=lambda x: x[1])

    return best

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (slightly larger than the previous 4000 to give a
    # finer discretisation – still cheap to FFT)
    # ------------------------------------------------------------------
    n = 5000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial candidate – try to reuse a previously stored construction
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # Resample to current length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            s1 = fast_score(a1, fft_len_conv, n)
            if s1 < best_score:
                best_a, best_score = a1, s1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep (keep several elite candidates)
    # ------------------------------------------------------------------
    elite = parametric_sweep(n, fft_len_conv, budget_s, rng, keep_top=12)
    for a, sc in elite:
        if sc < best_score:
            best_a, best_score = a.copy(), sc
            print(f"[sweep] new elite fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # Allocate fractions of the total budget
    phase1_end = start_time + 0.40 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.90 * budget_s   # pairwise mixing
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation from each elite candidate
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.4, 0.25, 0.15, 0.10, 0.06, 0.04, 0.025, 0.015,
        0.008, 0.004, 0.002, 0.001, 5e-04, 2e-04, 1e-04, 5e-05,
        2e-05, 1e-05, 5e-06,
    ]
    maxiter_per_lam = 1200  # more inner iterations than the previous run

    while time.time() < phase1_end:
        # choose a start point – either a random elite or a jittered best
        if rng.random() < 0.35:
            a_start = elite[rng.integers(len(elite))][0].copy()
        else:
            a_start = best_a.copy()
            a_start = a_start * rng.normal(1.0, 0.03, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (more aggressive)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_descent(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – pairwise convex mixing of elite candidates (including best)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        # ensure we have a list of unique elite vectors (including the current best)
        candidates = [best_a] + [c[0] for c in elite]
        # prune near‑duplicates (L2 distance < 1e‑6)
        unique = []
        for a in candidates:
            if not any(np.linalg.norm(a - u) < 1e-6 for u in unique):
                unique.append(a)
        candidates = unique

        # grid of mixing coefficients
        taus = np.linspace(0.0, 1.0, 81)  # step 0.0125
        for i in range(len(candidates)):
            for j in range(i + 1, len(candidates)):
                a1, a2 = candidates[i], candidates[j]
                for t in taus:
                    a_mix = (1 - t) * a1 + t * a2
                    a_mix /= a_mix.sum()
                    sc = fast_score(a_mix, fft_len_conv, n)
                    if sc < best_score:
                        best_score, best_a = sc, a_mix.copy()
        print(f"[mixing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing hill‑climb (global search)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
