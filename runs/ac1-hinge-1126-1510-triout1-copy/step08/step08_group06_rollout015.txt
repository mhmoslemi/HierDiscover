"""
search for a step sequence with a low evaluation score.

The algorithm works in three stages:

1. Fast parametric sweep over several families (power‑law, exponential,
   Gaussian, beta‑like, sine‑power).  For the best family we perform a
   simple 1‑D local search on its continuous parameter(s).

2. A large‑scale greedy hill‑climb on the true max‑convolution.
   The move “transfer a small amount of mass from the most harmful
   coordinate to the most helpful one” is performed many thousands of
   times; the convolution is updated incrementally (O(n) per move) so
   that millions of moves fit easily into the time budget.

3. The best sequence found is finally evaluated with the provided
   `evaluate_sequence` and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a sequence already
    normalised to sum(a) = 1.
    Returns 2·n·max(conv) where conv = a*a (linear convolution).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)          # length 2·n‑1
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate distributions (all return a probability vector)
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)          # avoid exact zeros
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector with a variety of shapes.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                               # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# 1‑D parametric refinement (simple local search on a single scalar)
# ----------------------------------------------------------------------


def refine_parameter(
    maker,
    n,
    start,
    low,
    high,
    deadline,
    rng,
    steps=200,
):
    """
    Very cheap hill‑climb on the scalar parameter of a family.
    Returns the best parameter and the associated vector.
    """
    param = start
    best_vec = maker(n, param)
    best_sc = fast_score(best_vec)

    while time.time() < deadline and steps > 0:
        # propose a multiplicative perturbation (up to ±10 %)
        delta = (rng.random() - 0.5) * 0.2
        cand = param * (1.0 + delta)
        cand = max(min(cand, high), low)
        vec = maker(n, cand)
        sc = fast_score(vec)
        if sc < best_sc:
            best_sc = sc
            best_vec = vec
            param = cand
        steps -= 1
    return best_vec, param, best_sc


# ----------------------------------------------------------------------
# Incremental hill‑climb on the true max‑convolution
# ----------------------------------------------------------------------


def gradient_from_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution when a unique maximiser k_max is assumed.
    Returns g_i = 2·a[k_max−i] whenever the index lies in the support, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    if k_max < n:
        # indices 0 … k_max are active
        g[: k_max + 1] = 2.0 * a[k_max::-1]
    else:
        low = k_max - (n - 1)               # first i with a non‑zero term
        g[low:] = 2.0 * a[low:][::-1]
    return g


def hill_climb_incremental(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy (with tiny amount of stochasticity) hill‑climb that repeatedly
    transfers a fraction of mass from the coordinate with the largest
    sub‑gradient to the one with the smallest.  The convolution is
    updated incrementally in O(n) time per move.
    """
    a = a0.copy()
    n = a.shape[0]
    conv = np.convolve(a, a)                # length 2·n‑1
    max_val = conv.max()
    best_a = a.copy()
    best_max = max_val
    rng = np.random.default_rng()
    start = time.time()
    it = 0
    recompute_every = 5000

    while time.time() < deadline and it < max_iter:
        # Occasionally inject a small amount of noise to escape plateaus
        if it % 1000 == 0 and rng.random() < 0.02:
            a = a + rng.normal(0.0, 1e-4, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = np.convolve(a, a)
            max_val = conv.max()
            continue

        # ----- decide source (src) and destination (dst) -----
        if rng.random() < 0.8:                     # mostly gradient‑guided
            k_max = int(np.argmax(conv))
            g = gradient_from_max(a, k_max)
            src = int(np.argmax(g))                # most harmful
            dst = int(np.argmin(g))                # most helpful
        else:                                       # random pair
            src = rng.integers(n)
            dst = rng.integers(n)
            while dst == src:
                dst = rng.integers(n)

        # ensure we can take mass from src
        if a[src] <= 1e-15:
            it += 1
            continue

        # --- fractional move (larger early, smaller later) ---
        remaining = deadline - time.time()
        total = deadline - start
        frac_max = 0.20 * (remaining / total) + 0.001   # shrink as time runs out
        delta = a[src] * rng.random() * frac_max
        if delta <= 0.0:
            it += 1
            continue

        # ----- incremental convolution update -----
        conv_new = conv.copy()
        # shift of destination (adds mass)
        conv_new[dst: dst + n] += 2.0 * delta * a
        # shift of source (removes mass)
        conv_new[src: src + n] -= 2.0 * delta * a
        # delta‑squared terms
        conv_new[2 * dst] += delta * delta
        conv_new[2 * src] += delta * delta
        conv_new[dst + src] -= 2.0 * delta * delta

        new_max = conv_new.max()

        # ----- accept if improvement (or with a tiny annealing probability) -----
        if new_max < max_val:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            max_val = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            # occasional uphill moves (very low temperature)
            T = max_val * 0.001 * (remaining / total + 1e-12)
            if rng.random() < np.exp(-(new_max - max_val) / T):
                a[src] -= delta
                a[dst] += delta
                conv = conv_new
                max_val = new_max

        # periodic full recomputation to avoid drift
        if it % recompute_every == 0:
            conv = np.convolve(a, a)
            max_val = conv.max()
            # renormalise to keep the sum exactly one (tiny floating‑point drift)
            s = a.sum()
            if s != 0.0:
                a /= s

        it += 1

    return best_a, best_max


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative step sequence within the given time budget.
    Returns the sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (the original challenge used n = 4000)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Global time handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin for final evaluation

    # ------------------------------------------------------------------
    # 1. cheap parametric sweep + 1‑D refinement
    # ------------------------------------------------------------------
    best_a = None
    best_sc = np.inf

    def try_update(vec):
        nonlocal best_a, best_sc
        vec = np.maximum(vec, 0.0)
        if vec.sum() == 0.0:
            return
        vec = vec / vec.sum()
        sc = fast_score(vec)
        if sc < best_sc:
            best_sc = sc
            best_a = vec.copy()
            print(f"[sweep] improved fast_score={best_sc:.6f}")

    # ---- families ----------------------------------------------------
    # power law (both directions)
    for p in np.linspace(-2.0, 5.0, 15):
        try_update(candidate_power_law(n, p))
        try_update(candidate_rev_power_law(n, p))

    # exponential (forward and reverse)
    for lam in np.linspace(0.0005, 0.02, 12):
        try_update(candidate_exponential(n, lam, reverse=False))
        try_update(candidate_exponential(n, lam, reverse=True))

    # Gaussian windows
    for sigma_frac in np.linspace(0.15, 0.45, 7):
        sigma = sigma_frac * n
        try_update(candidate_gaussian(n, sigma))

    # symmetric beta‑like shapes
    for alpha in np.linspace(0.2, 5.0, 12):
        try_update(candidate_beta_symmetric(n, alpha))

    # sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        try_update(candidate_sine_power(n, p))

    # a few random restarts (quick)
    while time.time() < start_time + 0.07 * budget_s:
        try_update(random_initial(n, rng))

    # If we already have a saved construction, use it as a possible start
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            try_update(a0)

    # ------------------------------------------------------------------
    # 1‑D refinement for the best families (optional, cheap)
    # ------------------------------------------------------------------
    # This extracts the parameter that produced the current best_a (if any)
    # and performs a tiny local search around it.  We only do this if we
    # still have plenty of time left.
    refined_deadline = start_time + 0.18 * budget_s
    if time.time() < refined_deadline and best_a is not None:
        # Try to infer which family produced best_a by checking a few signatures.
        # For simplicity we just run a few generic refinements on a few families.
        # The gains are modest but cheap.
        families = [
            (candidate_power_law, -2.0, 5.0),
            (candidate_rev_power_law, -2.0, 5.0),
            (candidate_exponential, 0.0005, 0.02),
            (candidate_gaussian, 0.15 * n, 0.45 * n),
            (candidate_beta_symmetric, 0.2, 5.0),
            (candidate_sine_power, 0.5, 6.0),
        ]
        for maker, lo, hi in families:
            # start from the current best parameter (mid‑point) – the refinement
            # may find a slightly better point even if this family was not the origin.
            vec, _, sc = refine_parameter(
                maker, n, (lo + hi) / 2.0, lo, hi, refined_deadline, rng, steps=120
            )
            if sc < best_sc:
                best_sc = sc
                best_a = vec.copy()
                print(f"[refine] new best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # 2. large‑scale greedy hill‑climb (the core optimiser)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    if best_a is None:                # fallback – uniform distribution
        best_a = np.full(n, 1.0 / n)

    print(f"[init] start fast_score={fast_score(best_a):.6f}")

    best_a, best_max = hill_climb_incremental(best_a, hill_deadline)

    final_fast = 2.0 * n * best_max
    print(f"[hill‑climb] best fast_score={final_fast:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
