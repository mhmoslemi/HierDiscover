```
def generate_shape(n, rng):
    family = rng.choice(['exp', 'power', 'sinepow', 'tri', 'uniform'])
    if family == 'exp':
        alpha = rng.uniform(0.001, 0.02)
        a = np.exp(-alpha * np.arange(n))
    elif family == 'power':
        p = rng.uniform(0.2, 2.5)
        a = 1.0 / ((np.arange(1, n+1, dtype=float))**p)
    elif family == 'sinepow':
        p = rng.uniform(0.5, 5.0)
        a = np.sin(np.pi * (np.arange(1, n+1) / (n+1)))**p
    elif family == 'tri':
        # triangular shape: max at centre, zero at edges.
        i = np.arange(n)
        a = 1.0 - np.abs(i - (n-1)/2) / ((n-1)/2)
        # optionally raise to power
        power = rng.uniform(0.5, 3.0)
        a = a**power
    else:
        a = np.full(n, 1.0)
    # ensure nonneg
    a = np.maximum(a, 0.0)
    # Normalize sum to sqrt(2n)
    total = a.sum()
    if total == 0:
        return np.full(n, 1.0/ n)
    a = a * (np.sqrt(2*n) / total)
    return a
```

"""
Propose candidate sequence to minimize evaluate_sequence.

Algorithm Overview:
- Random parametric shape search (exp, power, sine^p, triangular) to find a good
  starting distribution.
- Light L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution,
  with many λ‑stages.
- Long exponentiated‑gradient descent on the true ‑∞ objective.
- Adaptive greedy hill‑climb with decreasing move size and occasional
  neighbor‑exchange moves.
- Final evaluation with the official scorer.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and gradient for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax to stay on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the L∞‑objective.
    Uses step size η = η₀ / √t (η₀ can be tuned via eta0 argument).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# ----------------------------------------------------------------------
# Adaptive greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------
def hill_climb_adaptive(a: np.ndarray,
                        curr_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_frac_initial: float = 0.12,
                        recompute_every: int = 1500,
                        neighbor_prob: float = 0.1):
    """
    Greedy hill‑climb that moves a fraction of mass from a source index
    to a destination index.  The allowed fraction decays linearly with the
    remaining time, allowing large exploratory steps early and fine
    adjustments later.  A small fraction of moves swaps mass between
    neighboring indices to smooth the distribution.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)                       # exact full conv
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0
    start_time = time.time()
    total_time = deadline - start_time

    while time.time() < deadline:
        # adapt move size
        remaining = max(deadline - time.time(), 0.0)
        frac = max_frac_initial * (remaining / total_time)  # linear decay
        # pick source and destination
        i = rng.integers(0, n)
        if rng.random() < neighbor_prob:
            # neighbor move: destination is i±1 (wrap around)
            j = (i + rng.choice([-1, 1])) % n
        else:
            j = rng.integers(0, n)
        if i == j:
            continue
        # ensure we move from larger to smaller
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear approximation)
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # periodic exact recompute to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score

# Parametric shape generators

def shape_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.001, 0.02)
    a = np.exp(-alpha * np.arange(n, dtype=float))
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def shape_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 2.5)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def shape_sinepow(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 5.0)
    x = np.arange(1, n + 1, dtype=float) / (n + 1)
    a = np.sin(np.pi * x) ** p
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def shape_triangular(n: int, rng: np.random.Generator) -> np.ndarray:
    # symmetric triangular shape that vanishes at the boundaries
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    power = rng.uniform(0.5, 3.0)
    a = a ** power
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def shape_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    # two Gaussian-like bumps placed symmetrically
    centre = n // 2
    width = rng.uniform(0.05, 0.15) * n
    left_center = centre - width
    right_center = centre + width
    sigma = width / 2.0
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - left_center) / sigma) ** 2) + np.exp(-0.5 * ((x - right_center) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # initialise with previous global if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0_sum = a0.sum()
            a0 = a0 / a0_sum if a0_sum > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    # ------------------------------------------------------------------
    # Phase 0: fast parametric shape search (≈15% of budget)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.15 * budget_s
    shape_funcs = [shape_exp, shape_power, shape_sinepow, shape_triangular, shape_bimodal]
    while time.time() < phase0_end:
        func = rng.choice(shape_funcs)
        cand = func(n, rng)
        score = fast_score(cand, fft_len)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            print(f"[shape] new best fast_score={best_score:.6f} (func={func.__name__})")

    # ------------------------------------------------------------------
    # Phase 1: L‑BFGS on smoothed max‑convolution (≈25% of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 1200
    while time.time() < phase1_end:
        # occasionally perturb the current best, otherwise fresh random start
        if rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2: exponentiated‑gradient descent (≈45% of budget)
    # ------------------------------------------------------------------

"""
Search for a low‑score non‑negative step sequence.

Algorithm
---------
1. **Fast parametric shape search** – generate a large number of
   candidate distributions (exponential, power‑law, sine‑powered,
   triangular, bimodal) on the simplex and keep the best one.

2. **L‑BFGS on a smoothed max‑convolution** – minimise a
   log‑sum‑exp approximation of the true objective.  A schedule of
   decreasing smoothing parameters (λ) is used, each stage starting from
   the previous result.

3. **Exponentiated‑gradient descent** – perform long‑running EG updates
   directly on the hard ‑∞ objective (gradient obtained from the active
   constraints of the convolution).

4. **Adaptive greedy hill‑climb** – repeatedly move a fraction of mass
   from a large entry to a smaller one.  The allowed fraction decays
   linearly with the remaining time, and a small proportion of moves
   swaps mass between neighbouring indices to smooth the distribution.
   Convolution is updated incrementally and recomputed exactly from time
   to time to avoid drift.

5. **Final evaluation** – the best sequence is returned and scored with
   the official `evaluate_sequence` function.

The whole procedure respects the 1000 s budget and prints progress
information.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and project back onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    return noisy / s if s > 0 else a.copy()


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Log‑sum‑exp smoothed max‑convolution.
    Variables are logits y (softmax(y) gives the probability vector).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax → probability vector a (∑a = 1)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # full convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # derivative of smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """Run L‑BFGS over a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    EG descent directly on 2·n·max(a*a)/(∑a)².
    Step size ηₜ = η₀ / √t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        a = a / s if s > 0 else best_a.copy()

        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adaptive greedy hill‑climb with incremental updates
# ----------------------------------------------------------------------
def hill_climb_adaptive(a: np.ndarray,
                        curr_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        max_frac_initial: float = 0.12,
                        recompute_every: int = 1500,
                        neighbor_prob: float = 0.12):
    """
    Greedy hill‑climb.  The allowed move fraction decays linearly with the
    remaining time, allowing coarse moves early and fine tweaks later.
    A small fraction of moves swaps mass between neighbouring indices.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)  # exact full convolution
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)

    while time.time() < deadline:
        # linear decay of the maximal allowed fraction
        remaining = max(deadline - time.time(), 0.0)
        cur_frac = max_frac_initial * (remaining / total_time)

        i = rng.integers(0, n)
        if rng.random() < neighbor_prob:
            j = (i + rng.choice([-1, 1])) % n
        else:
            j = rng.integers(0, n)
        if i == j:
            continue

        # always move mass from the larger to the smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue

        delta = best_a[src] * rng.random() * cur_frac
        if delta <= 0.0:
            continue

        # linearised incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric shape generators (used in the fast search phase)
# ----------------------------------------------------------------------
def shape_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    alpha = rng.uniform(0.001, 0.02)
    a = np.exp(-alpha * np.arange(n, dtype=float))
    return a / a.sum()


def shape_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 2.5)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    return a / a.sum()


def shape_sinepow(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.5, 5.0)
    x = np.arange(1, n + 1, dtype=float) / (n + 1)
    a = np.sin(np.pi * x) ** p
    return a / a.sum()


def shape_triangular(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    power = rng.uniform(0.5, 3.0)
    a = a ** power
    return a / a.sum()


def shape_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = n // 2
    width = rng.uniform(0.05, 0.15) * n
    sigma = width / 2.0
    left_c = centre - width
    right_c = centre + width
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - left_c) / sigma) ** 2) + np.exp(-0.5 * ((x - right_c) / sigma) ** 2)
    return a / a.sum()

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 points give a good accuracy vs speed trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():

            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)

    start_time = time.time()

    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
