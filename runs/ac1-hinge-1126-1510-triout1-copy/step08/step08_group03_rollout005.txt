def local_random_search(a, cur_score, fft_len, n, deadline, rng,
                        max_iters=1e9):
    it = 0
    step = 0.05  # fraction of mass to move
    while time.time() < deadline and it < max_iters:
        # compute convolution for current a to get max index and partner for sampling
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
        max_val = conv.max()
        max_idx = np.argmax(conv)

        # compute partner vector as before
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        # source index with probability ~ contrib (higher) 
        prob_source = contrib / contrib.sum()
        i_src = rng.choice(n, p=prob_source)
        # destination index choose from low partner values
        low_idxs = np.argsort(partner)[:max(1, int(0.02*n))]
        i_dst = rng.choice(low_idxs)
        if i_src == i_dst:
            continue
        # define delta as step * a[i_src]
        delta = step * a[i_src]
        if delta <= 0:
            break
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        # projection onto simplex
        s = a_new.sum()
        if s <= 0:
            continue
        a_new /= s
        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            # maybe increase step for faster moves?
            step = min(step * 1.1, 0.5)
        else:
            step = max(step * 0.9, 0.0005)
        it += 1
    return a, cur_score

"""
Hybrid global‐local optimization of a non‑negative step sequence.

The pipeline follows four stages inside the 1000 s budget:

1.  Diverse initialisations (uniform, spectral, random, sinusoidal,
    and a warm‑start from the previous run) on a chosen length `n`
    (default 8000).  All vectors are projected onto the probability
    simplex (sum = 1).

2.  A multi‑restart L‑BFGS optimisation of a log‑sum‑exp smoothed
    version of the max–convolution.  The smoothing factor (`lam_factor`)
    is decreased gradually (lam‑schedule) to approach the true max.

3.  Long exponentiated‑gradient descent on the exact max‑convolution.
    The update `a ← a·exp(−η·g)` keeps the vector on the simplex.

4.  A deterministic guided hill‑climb that moves mass from the most
    harmful coordinate to the least harmful one, followed by a
    stochastic targeted local search that repeatedly transfers a small
    amount of mass from a high‑contribution index to a low‑contribution
    index (accepting only strict improvements).

All heavy work uses FFT‑based convolutions (`O(n log n)`).  The best
vector found is finally evaluated with the supplied `evaluate_sequence`
function and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when sum(a)=1:
    returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (mixture of families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'sinusoidal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sinusoidal
        p = rng.uniform(0.5, 2.0)
        a = np.sin(np.pi * np.arange(n) / (n - 1)) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """
    Initialise a by taking the inverse FFT of the square‑root of the FFT
    of a uniform target (length 2·n‑1).  The result is projected onto
    the simplex.
    """
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximal self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the hard max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
            s = a.sum()
        a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.20,
                      min_step: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Deterministic improvement: move a fraction of mass from the index that
    contributes most to the current maximal convolution entry to an index
    with the smallest partner value.  Accept only strict improvements.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero out‑of‑range)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: index with smallest partner (avoid i_max)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size for next move
        else:
            step = max(step * 0.8, min_step)

    return a, score


def local_random_search(a: np.ndarray,
                        cur_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        init_frac: float = 0.05,
                        min_frac: float = 5e-4) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic refinement.
    At each iteration we pick a source index proportional to its
    contribution to the current worst convolution entry and a destination
    from the set of indices with smallest partner values.
    A small fraction of mass is transferred; the move is accepted only
    if it strictly lowers the objective.
    """
    step = init_frac
    while time.time() < deadline:
        # compute convolution and the critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner vector for this index
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        total_contrib = contrib.sum()
        if total_contrib <= 0.0:
            break

        # source sampled with probability proportional to its contribution
        prob_src = contrib / total_contrib
        i_src = rng.choice(n, p=prob_src)

        # a small set of destinations with minimal partner values
        low_set = np.argsort(partner)[: max(1, int(0.02 * n))]
        i_dst = rng.choice(low_set)
        if i_src == i_dst:
            continue

        delta = step * a[i_src]
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s <= 0.0:
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            step = min(step * 1.1, 0.5)      # become a bit bolder after success
        else:
            step = max(step * 0.9, min_frac)  # shrink after failure
    return a, cur_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (larger n gives more freedom, FFT length stays small)
    # ------------------------------------------------------------------
    n = 8000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # warm‑start from the previous run if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_warm = np.interp(x_new, x_old, prev)
            a_warm = np.maximum(a_warm, 0.0)
            a_warm /= a_warm.sum()
            candidates.append(a_warm)

    # uniform, spectral, random, sinusoidal
    candidates.append(np.full(n, 1.0 / n))
    candidates.append(spectral_init(n))
    candidates.append(random_initial(n, rng))

    # additional sinusoidal family
    p = rng.uniform(0.5, 2.0)
    sin_shape = np.sin(np.pi * np.arange(n) / (n - 1)) ** p
    sin_shape = sin_shape / sin_shape.sum()
    candidates.append(sin_shape)

    # pick the best among the pool as the starting point
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best initial fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety buffer

    phase1_deadline = start_time + 0.20 * budget_s   # L‑BFGS
    phase2_deadline = start_time + 0.80 * budget_s   # exp‑gradient
    phase3_deadline = start_time + 0.97 * budget_s   # hill‑climb + local search

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS on smoothed objective
    # ------------------------------------------------------------------
    lam_schedule = [
        0.50, 0.30, 0.20, 0.12, 0.07, 0.04,
        0.025, 0.015, 0.008, 0.004, 0.002,
        0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.33:
            a_start = random_initial(n, rng)
        elif r < 0.66:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.02), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (long run)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – guided hill‑climb + stochastic refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        # deterministic hill‑climb
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_deadline,
            init_step=0.30,
            min_step=0.0005,
        )
        # stochastic targeted local search using the remaining budget
        best_a, best_score = local_random_search(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
            init_frac=0.05,
            min_frac=5e-4,
        )
        print(f"[local refinement] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluator
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
