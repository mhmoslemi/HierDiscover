        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        if score < best_score:
            best_score = score
            best_y = y.copy()
    a_best = softmax(best_y)
    return a_best, best_score

def guided_hill_climb_refined(a, score, fft_len, n, deadline):
    step_frac = 0.5
    min_step = 1e-4
    while time.time() < deadline:
        # compute conv and find max_idx
        conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len)*np.fft.fft(a, fft_len)))[:2*n-1]
        max_idx = np.argmax(conv)
        # compute partner and contribution same as before
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n-1))
        i_end = min(n-1, max_idx)
        idx = np.arange(i_start, i_end+1)
        partner[idx] = a[max_idx - idx]
        contrib = a * partner
        i_max = np.argmax(contrib)
        # choose i_min as index with smallest partner value not equal i_max
        # we can choose among few candidates with lowest partner
        sorted_by_partner = np.argsort(partner)
        i_min = sorted_by_partner[0]
        if i_min == i_max:
            i_min = sorted_by_partner[1]
        # Try delta fractions decreasing
        improved = False
        # fraction list
        for frac in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            delta = a[i_max] * frac
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i_max] -= delta
            a_new[i_min] += delta
            # renorm
            # ensure non-negative and sum ~1
            s = a_new.sum()
            a_new /= s
            new_score = fast_score(a_new, fft_len, n)
            if new_score < score:
                a = a_new
                score = new_score
                improved = True
                break
        if not improved:
            # reduce step frac or break if too small
            step_frac = max(step_frac * 0.5, min_step)
            if step_frac <= min_step:
                break
    return a, score

4.  Guided hill‑climb with an adaptive line‑search that moves mass from the
    most harmful coordinate to the least harmful one.
5.  Tiny random “swap” moves as a final polishing step.

All phases respect the simplex constraint (non‑negative entries, sum = 1)
and use FFT‑based convolutions for speed.  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when a has sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisation families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (several mixed families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """FFT‑based spectral initialisation – loosely inspired by a flat target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12          # smoothing width
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, always starting from the
    previous result.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # exact convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def adam_optimize(
    y_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smoothed objective.
    Operates in logit space (y) to keep the simplex automatically.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    best_y = y.copy()
    best_score = float("inf")

    # start with a moderate smoothing factor and gradually reduce it
    lam_factors = [0.10, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    lam_idx = 0
    next_lam_change = time.time() + (deadline - time.time()) / len(lam_factors)

    while time.time() < deadline:
        if time.time() >= next_lam_change and lam_idx + 1 < len(lam_factors):
            lam_idx += 1
            next_lam_change = time.time() + (deadline - time.time()) / (len(lam_factors) - lam_idx)

        lam_factor = lam_factors[lam_idx]

        f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule – slower as we near the end
        base_lr = 0.05
        lr = base_lr / np.sqrt(t)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_y = y.copy()
            print(f"[Adam] new best score={best_score:.6f}")

    best_a = softmax(best_y)
    return best_a, best_score


# ----------------------------------------------------------------------
# Guided hill‑climb with adaptive line search
# ----------------------------------------------------------------------


def guided_hill_climb_refined(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_step: float = 0.4,
    min_step: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Deterministic improvement: move a fraction of mass from the coordinate
    that contributes most to the current maximal convolution entry to a
    coordinate with the smallest partner value.  Several candidate fractions
    are tried (line‑search) and the best improvement is kept.
    """
    step = init_step
    fractions = [0.5, 0.35, 0.25, 0.15, 0.10, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005]

    while time.time() < deadline:
        # current convolution
        conv = full_conv(a, fft_len)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # partner[i] = a[max_idx - i] for indices that line up
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: index with smallest partner (avoiding i_max)
        sorted_by_partner = np.argsort(partner)
        i_min = int(sorted_by_partner[0] if sorted_by_partner[0] != i_max else sorted_by_partner[1])

        improved = False
        for frac in fractions:
            delta = a[i_max] * frac
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i_max] -= delta
            a_new[i_min] += delta

            # re‑normalise (numerical drift)
            s = a_new.sum()
            if s <= 0.0:
                continue
            a_new /= s

            new_score = fast_score(a_new, fft_len, n)
            if new_score < score - 1e-12:
                a = a_new
                score = new_score
                improved = True
                # keep the successful step size (no need to shrink)
                break

        if not improved:
            step = max(step * 0.7, min_step)
            if step <= min_step:
                break
    return a, score


def random_swap_refine(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
    max_trials: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Tiny random pair‑wise swaps, kept only if they strictly improve the score.
    Acts as a final polishing step.
    """
    best_a = a.copy()
    best_score = score
    trial = 0
    while time.time() < deadline and trial < max_trials:
        i, j = rng.integers(0, n, size=2)
        if i == j or best_a[i] < 1e-12:
            trial += 1
            continue
        # move a small fraction from i to j
        frac = rng.uniform(0.01, 0.2)
        delta = best_a[i] * frac
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_a = a_new
            best_score = new_score
            # reset trial counter to keep searching after improvement
            trial = 0
        else:
            trial += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for gradient‑back‑prop

    # --------------------------------------------------------------
    # Initial candidates (uniform / spectral / random / previous)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    a0 = a_start.copy()
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split (seconds)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    phase1_deadline = start_time + 0.25 * budget_s   # L‑BFGS heavy
    phase2_deadline = start_time + 0.70 * budget_s   # exp‑grad long
    phase3_deadline = start_time + 0.90 * budget_s   # Adam + refined hill‑climb
    phase4_deadline = deadline                       # final polishing

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002, 0.00001,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.45:
            a_start = random_initial(n, rng)
        elif r < 0.85:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.012), rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (long run)
    # --------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam optimisation + guided refined hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_deadline:
        # --- Adam on smoothed objective -----------------------------------
        y_start = np.log(best_a + 1e-12)
        best_a_adam, score_adam = adam_optimize(
            y_start,
            phase3_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        if score_adam < best_score:
            best_a, best_score = best_a_adam, score_adam
            print(f"[Adam] improved score={best_score:.6f}")

        # --- Refined guided hill‑climb ------------------------------------
        best_a, best_score = guided_hill_climb_refined(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_deadline,
            init_step=0.4,
            min_step=1e-4,
        )
        print(f"[guided hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final random‑swap polishing
    # --------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = random_swap_refine(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_deadline,
            rng,
            max_trials=5000,
        )
        print(f"[random swap] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official evaluator
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
