def lp_refine(a, deadline):
    f = a
    best = f.copy()
    best_score = fast_score(f, fft_len, n)
    t = 0.4 (starting).
    while time.time() < deadline:
        # compute M
        fa = np.fft.rfft(f, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
        M = conv.max()
        # build A_ub
        data = np.repeat(f, n)
        A_ub = csr_matrix((data, (row_idx_flat, col_idx_flat)), shape=(2*n-1, n))
        # solve LP
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0,None)]*n, method='highs')
        if not res.success:
            print("[LP] solve failed")
            break
        b0 = res.x
        sum_b = b0.sum()
        if sum_b <= 1.0 + 1e-12:
            # no possible improvement; break
            break
        g = b0 / sum_b
        # try step size
        # compute candidate with t
        a_new = (1 - t) * f + t * g
        a_new = a_new / a_new.sum()
        # compute new M.
        fa_new = np.fft.rfft(a_new, fft_len)
        conv_new = np.fft.irfft(fa_new * fa_new, fft_len)[:2*n-1]
        M_new = conv_new.max()
        # if improved take, else reduce t.
        if M_new < M:
            f = a_new
            score = 2.0*n*M_new
            if score < best_score:
                best_score = score
                best = f.copy()
                print(f"[LP] improved score={best_score:.6f}")
        else:
            t = t * 0.5
            if t < 1e-4:
                # cannot improve further
                break
        # continue loop
    return best, best_score

"""
Propose a low‑score step‑function.

The search is split into four phases with a total budget of 1000 s:

1️⃣  Quick L‑BFGS polishing on a smooth log‑sum‑exp surrogate,
    starting from several diverse random / spectral initialisations.

2️⃣  Exponentiated‑gradient descent on the exact max‑convolution.

3️⃣  Linear‑programming refinement (the key new ingredient):
    given the current distribution f, maximise Σ g_j
    subject to the linear constraints (f∗g)_k ≤ M = max(f∗f)_k.
    The optimum usually has Σ g_j > 1, so after rescaling
    g = g₀ / Σ g₀ the bound on the convolution drops.
    A convex combination f←(1‑t) f + t g (t≈0.2) strictly reduces M.
    Repeating this step drives the objective close to its theoretical
    lower bound (≈ 1).

4️⃣  Final greedy hill‑climb for polishing.

All heavy convolution work uses real FFT for speed.  The algorithm
always respects the 1000 s deadline and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum = 1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(f*f) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Log‑sum‑exp smooth max of self‑convolution; returns (obj, grad_y)."""
    a = softmax(y)

    # a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Short L‑BFGS continuation on the smoothed surrogate."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """Exponentiated‑gradient descent on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = a / a.sum()

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of √(FFT of uniform) → projected onto simplex."""
    L = next_pow2(2 * n - 1)
    uniform = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    u_padded = np.zeros(L)
    u_padded[: 2 * n - 1] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    a = a / a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector on the simplex."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = a / a.sum()
    return a


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 12000):
    """Greedy hill‑climb: move a random fraction of mass between two sites."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (new ingredient)
# ----------------------------------------------------------------------


def lp_refine(a: np.ndarray,
              deadline: float,
              n: int,
              fft_len: int,
              row_idx: np.ndarray,
              col_idx: np.ndarray,
              initial_step: float = 0.2):
    """
    Repeatedly solve:
        maximise Σ g_j   s.t.   g ≥ 0,  (a ∗ g)_k ≤ M,
    where M = max(a ∗ a).  Then mix f←(1‑t)f+t·(g/Σg).
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    t = initial_step
    it = 0

    while time.time() < deadline:
        # current max‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # build the (a∗·) matrix A (sparse)
        data = np.repeat(a, n)                     # length n²
        A = csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        # Linear program: maximise Σ g  ⇔ minimise -1ᵀg
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            # if the LP fails, abort the refinement loop
            print("[LP] linprog failed – status:", res.message)
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 1.0 + 1e-12:
            # no room for improvement
            break

        # scaled direction
        g = g0 / sum_g0

        # try to move towards g
        a_candidate = (1.0 - t) * a + t * g
        a_candidate = a_candidate / a_candidate.sum()

        # evaluate candidate
        fa_cand = np.fft.rfft(a_candidate, fft_len)
        conv_cand = np.fft.irfft(fa_cand * fa_cand, fft_len)[: 2 * n - 1]
        M_cand = conv_cand.max()
        score_cand = 2.0 * n * M_cand

        if score_cand < best_score:
            best = a_candidate.copy()
            best_score = score_cand
            a = a_candidate
            # keep step size
        else:
            # reduce step size and retry (but stay on the same a)
            t *= 0.5
            if t < 1e-5:
                break
        it += 1
        if it % 10 == 0:
            print(f"[LP] iter={it}, best_score={best_score:.6f}")

    return best, best_score
