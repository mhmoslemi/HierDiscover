"""
search for a step‑function sequence that minimises
    2·n·max(convolution)/sum^2

The algorithm has four phases:
  1. Heavy multi‑restart L‑BFGS on a smooth log‑sum‑exp approximation.
  2. Simple exponentiated‑gradient (mirror) descent on the simplex.
  3. Fine‑grained gradient‑guided hill‑climb.  A move transfers a
     variable amount of mass from a “high‑gradient’’ index to a
     “low‑gradient’’ one.  The exact improvement for a given amount is
     evaluated analytically (no extra FFT) and a tiny golden‑section
     line‑search finds the best delta.
  4. An ultra‑small‑lambda polish with L‑BFGS.

All phases respect non‑negativity and the ∑a=1 normalisation, so the
external ``evaluate_sequence`` is scale‑invariant.  The total wall‑time
is limited to 1000 s; the budget is split dynamically between the
phases.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast objective when a is on the simplex (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine", "triangular"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "cosine":
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    else:  # triangular bump
        a = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                            np.linspace(1, 0, n - n // 2)])
        a += 0.001
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray, lam_schedule, maxiter_per_lam: int,
                       deadline: float, fft_len_conv: int,
                       fft_len_g: int, n: int):
    """
    Run L‑BFGS for each lambda in ``lam_schedule``.  Each stage starts from
    the previous result.  Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)                      # log‑softmax space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam,
                         "gtol": 1e-12,
                         "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def exp_gradient_optimize(a0: np.ndarray, deadline: float,
                         fft_len_conv: int, fft_len_g: int, n: int):
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient
    of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.20 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def compute_subgradient(a: np.ndarray, conv: np.ndarray,
                        fft_len_g: int, eps_factor: float = 1e-8):
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * eps_factor + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return g


def hill_climb_line_search(a_start: np.ndarray, conv_start: np.ndarray,
                           n: int, fft_len_g: int, deadline: float):
    """
    Fine‑grained hill‑climb.  In each iteration we move a variable amount
    of mass from a “high‑gradient’’ index to a “low‑gradient’’ one.
    The optimal amount is found by a cheap golden‑section search that
    evaluates the max‑convolution analytically (no FFT).
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    conv = conv_start.copy()
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline:
        # compute sub‑gradient on the current active set
        g = compute_subgradient(a, conv, fft_len_g)

        # choose source (large g) and target (small g)
        topk = max(1, int(0.02 * n))
        lowk = max(1, int(0.02 * n))
        high_idx = np.argpartition(-g, topk - 1)[:topk]
        low_idx = np.argpartition(g, lowk - 1)[:lowk]
        i = int(rng.choice(high_idx))
        j = int(rng.choice(low_idx))
        if i == j or a[i] <= 1e-14:
            it += 1
            continue

        # pre‑compute the linear effect of moving a delta from i to j
        delta_vec = np.zeros(2 * n - 1, dtype=np.float64)
        # source contribution (removal)
        delta_vec[i:i + n] -= 2.0 * a
        # target contribution (addition)
        delta_vec[j:j + n] += 2.0 * a

        max_delta = a[i]              # we cannot move more than what we have
        if max_delta <= 0.0:
            it += 1
            continue

        # ----- golden‑section line search -----
        phi = (1.0 + 5.0 ** 0.5) / 2.0
        invphi = 1.0 / phi
        lo, hi = 0.0, max_delta
        # initial interior points
        c = hi - (hi - lo) * invphi
        d = lo + (hi - lo) * invphi

        def eval_max(delta):
            tmp = conv + delta * delta_vec
            # the quadratic self‑terms
            tmp[2 * i] += delta * delta
            tmp[2 * j] += delta * delta
            return tmp.max()

        f_c = eval_max(c)
        f_d = eval_max(d)

        # limit to ~25 iterations (more than enough for machine precision)
        for _ in range(25):
            if f_c < f_d:
                hi = d
                d = c
                f_d = f_c
                c = hi - (hi - lo) * invphi
                f_c = eval_max(c)
            else:
                lo = c
                c = d
                f_c = f_d
                d = lo + (hi - lo) * invphi
                f_d = eval_max(d)
            if hi - lo < 1e-12 * max_delta:
                break
        delta_opt = (lo + hi) / 2.0
        new_max = eval_max(delta_opt)

        # accept only genuine improvements
        if new_max < conv.max() - 1e-14:
            # apply the move permanently
            a[i] -= delta_opt
            a[j] += delta_opt
            # keep numerical safety
            a[i] = max(a[i], 0.0)
            a[j] = max(a[j], 0.0)

            conv += delta_opt * delta_vec
            conv[2 * i] += delta_opt * delta_opt
            conv[2 * j] += delta_opt * delta_opt

            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Starting point (use previous solution if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1               # safety margin

    phase1_end = start_time + 0.30 * budget_s           # L‑BFGS heavy
    phase2_end = start_time + 0.55 * budget_s           # exponentiated‑gradient
    phase3_end = start_time + 0.95 * budget_s           # fine hill‑climb
    # the last few seconds are for the ultra‑small λ polish

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS with decreasing smoothing
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 1e-06,
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # occasional perturbation of the current best
        if rng.random() < 0.30:
            a_start = add_noise(best_a, rng.uniform(0.001, 0.015), rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – precise gradient‑guided hill‑climb with line search
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        conv_start = full_conv(best_a, fft_len_conv)
        best_a, best_score = hill_climb_line_search(
            best_a, conv_start, n, fft_len_g, phase3_end
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final ultra‑small‑lambda L‑BFGS polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, fine_score = lbfgs_continuation(
            best_a,
            [1e-07],
            maxiter_per_lam=800,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score:
            best_score = fine_score
            best_a = a_final.copy()
            print(f"[final L‑BFGS] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
