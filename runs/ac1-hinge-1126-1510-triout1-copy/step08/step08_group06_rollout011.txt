def annealing_optimize(a0, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    T0 = 0.01
    while time.time() < deadline:
        # schedule temperature decreasing over time
        t = (deadline - time.time()) / (deadline - start_time)  # fraction left
        T = T0 * t
        # propose move
        i = rng.integers(0,n)
        j = rng.integers(0,n)
        if i==j:
            continue
        delta = min(a[i], rng.random() * 0.1 * a[i])  # up to 10% of a[i]
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # compute score
        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
        if sc_new < fast_score(a, fft_len, n):
            a = a_new
        else:
            # accept with probability exp(-(sc_new - sc_old)/T)
            if rng.random() < np.exp(-(sc_new - fast_score(a, fft_len, n))/T):
                a = a_new

"""
Hybrid optimizer for the max‑convolution objective.

The algorithm proceeds in four stages

1.  A cheap parametric sweep over several analytic families (power‑law,
    exponential, Gaussian, beta‑like, sinusoidal).  Any improvement is
    kept.

2.  A large number of short L‑BFGS continuations with a smoothed
    ``log‑sum‑exp`` approximation of the max‑convolution.  The
    continuation uses a schedule of decreasing smoothing parameters; each
    run starts either from a jittered version of the current best or from
    a random *sparse* distribution (few non‑zero entries).

3.  Projected sub‑gradient descent with momentum performed on the true
    max‑convolution for the bulk of the remaining time.

4.  A deterministic hill‑climb that repeatedly shifts a small fraction
    of mass from the coordinate with the largest sub‑gradient to the
    smallest one.  The step size is slowly decayed, guaranteeing many
    fine‑grained updates.

All phases respect the global time budget (default 1000 s).  The final
sequence (a list of non‑negative floats) is returned and also evaluated
with the provided ``evaluate_sequence`` function so that the exact score
is printed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a sequence that already satisfies
    sum(a) = 1.  Returns 2·n·max(conv) where conv = a*a.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. y)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best `a` found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_sc = fast_score(a_candidate, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc = cand_sc
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_sc


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb: repeatedly move a small fraction of mass
    from the coordinate with the largest sub‑gradient to the one with the
    smallest sub‑gradient.  Only strictly improving moves are kept.
    """
    best = a.copy()
    best_sc = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo:
            break

        # fraction decays with iterations
        frac = 0.15 * max(0.1, 1.0 - it / max_iter)
        delta = best[i_hi] * frac
        if delta <= 0.0:
            it += 1
            continue

        best[i_hi] -= delta
        best[i_lo] += delta

        new_sc = fast_score(best, fft_len, n)
        if new_sc < best_sc:
            best_sc = new_sc
        else:
            # revert move
            best[i_hi] += delta
            best[i_lo] -= delta

        it += 1

    return best, best_sc


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def random_sparse_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a sparse distribution: few non‑zero entries placed uniformly."""
    # choose support size between 70 and 200 (adjustable)
    k = int(rng.integers(70, 200))
    idx = rng.choice(n, size=k, replace=False)
    weights = rng.random(k) + 1e-8  # ensure positivity
    a = np.zeros(n, dtype=float)
    a[idx] = weights
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Candidate families for the cheap sweep
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the provided
    evaluation function.  The routine respects a 1000 s budget (or the
    supplied *budget_s*) and returns the best list of floats it found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (fixed to 4000 as in the reference run)
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)  # length for all FFT‑based convolutions

    # ------------------------------------------------------------------
    # Initialise from a previously saved global if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                # interpolate onto the required size
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_sc = fast_score(best_a, fft_len, n)
    print(f"[init] fast_score={best_sc:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # reserve a small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    def try_update(a):
        nonlocal best_a, best_sc
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print(f"[sweep] improved fast_score={best_sc:.6f}")

    # power‑law and reversed
    for p in np.linspace(-2.0, 5.0, 15):
        try_update(candidate_power_law(n, p))
        try_update(candidate_rev_power_law(n, p))

    # exponential families
    for lam in np.linspace(0.0005, 0.02, 12):
        try_update(candidate_exponential(n, lam, reverse=False))
        try_update(candidate_exponential(n, lam, reverse=True))

    # Gaussian windows
    for sigma_frac in np.linspace(0.15, 0.45, 7):
        sigma = sigma_frac * n
        try_update(candidate_gaussian(n, sigma))

    # beta‑like symmetric
    for alpha in np.linspace(0.2, 5.0, 12):
        try_update(candidate_beta_symmetric(n, alpha))

    # sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        try_update(candidate_sine_power(n, p))

    # a few quick random starts
    while time.time() < start_time + 0.07 * budget_s:
        try_update(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Allocate windows for the remaining phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.60 * budget_s    # L‑BFGS restarts
    phase2_end = start_time + 0.85 * budget_s    # long sub‑gradient phase
    # remaining time → deterministic hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS continuations from diverse starts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.25,
        0.125,
        0.0625,
        0.03,
        0.015,
        0.008,
        0.004,
        0.002,
        0.001,
        0.0005,
        0.00025,
        0.00012,
        0.00006,
        0.00003,
    ]
    maxiter_per_lam = 500

    while time.time() < phase1_end:
        # choose start: jittered best or a fresh sparse candidate
        if rng.random() < 0.3:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.012), rng=rng)
        else:
            a_start = random_sparse_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len,
            n,
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.6f} (t={time.time()-start_time:.1f}s)")

    # ------------------------------------------------------------------
    # Phase 2 – long projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len, fft_len, n
        )
        print(f"[sub‑gradient] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic gradient‑guided hill‑climb (fine tuning)
    # ------------------------------------------------------------------
    best_a, best_sc = gradient_guided_hill_climb(
        best_a,
        best_sc,
        fft_len,
        n,
        deadline,
        max_iter=50000,
    )
    print(f"[final hill‑climb] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
