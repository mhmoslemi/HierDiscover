def propose_candidate(seed: int = 42, budget_s: int = 1000):
    import time
    import numpy as np
    rng = np.random.default_rng(seed)
    n = 4000
    max_c_allowed = 2
    best_set = []
    best_m = 0
    deadline = time.time() + budget_s - 0.1
    attempts = 0
    while time.time() < deadline:
        # random order of indices
        order = rng.permutation(n)
        counts = np.zeros(2*n-1, dtype=np.int16)
        S = []
        for i in order:
            # compute self count
            if counts[2*i] + 1 > max_c_allowed:
                continue
            if len(S) > 0:
                S_arr = np.array(S, dtype=np.int32)
                sums = i + S_arr
                # For each sum, adding i adds 2 contributions
                if np.any(counts[sums] + 2 > max_c_allowed):
                    continue
                # All constraints satisfied
                counts[sums] += 2
            # Add self count
            counts[2*i] += 1
            S.append(i)
        m = len(S)
        if m > best_m:
            best_m = m
            best_set = S
            print(f"[attempt {attempts}] new best size {best_m}")
        attempts += 1
        # if we reach theoretical bound, break early
        if best_m >= int(np.sqrt((max_c_allowed)*(2*n-1))) + 5:
            break
    # Build final sequence
    seq = np.full(n, 10.0, dtype=np.float64)  # baseline small value
    seq[best_set] = 1000.0
    final_score = evaluate_sequence(seq.tolist())
    print(f"[final] size={best_m}, max_c_allowed={max_c_allowed}, score={final_score}")
    return seq.tolist()

Search for a step function with very low evaluation score by exploiting the
per‑element 1000‑cap. The idea is to place a moderate number of entries at
the maximal value 1000 while keeping the ordered convolution count
`# { (i,j) ∈ S×S : i+j = k }` bounded by a small constant C (C=2). 
If the set `S` has size m then the convolution maximum is at most
C·10⁶ while the sum of the sequence is 1000·m, giving a score 
2 n · C / m².  For n=4000 the theoretical bound is m ≤ √(C·(2n‑1)),
so with C=2 we can reach a score around 1.0, well below the target 1.5.
The routine repeatedly builds such a set by a greedy random‑order
insertion, keeps the best (largest) set found and finally returns the
corresponding sequence (1000 on the chosen positions, 0 elsewhere). 
The whole search runs well within the 1000 s budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """Return a list of non‑negative heights that achieves a low score.

    The algorithm constructs a sparse set of indices S on which the value
    1000 is placed.  The set is built greedily while enforcing that for every
    possible convolution position k the number of ordered pairs
    (i,j)∈S×S with i+j=k never exceeds a small bound C (C=2).  This limits
    the convolution maximum to C·10⁶, while the sum is 1000·|S|, yielding a
    very small value of the evaluation function.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem parameters
    # --------------------------------------------------------------
    n = 4000                     # length of the step function
    C = 2                        # allowed maximum count per convolution index
    max_allowed = int(np.sqrt(C * (2 * n - 1)) + 1)   # theoretical ceiling

    best_set = []
    best_m = 0
    best_counts = None

    deadline = time.time() + budget_s - 0.05   # a tiny safety margin

    attempt = 0
    while time.time() < deadline:
        # random ordering of indices – different order gives a different greedy
        order = rng.permutation(n)
        counts = np.zeros(2 * n - 1, dtype=np.int16)   # ordered‑pair counters
