def refine_with_lbfgs(a_start, n, fft_len, deadline):
    lam = 0.001 # small smoothing
    maxiter = 200
    y = np.log(a_start + 1e-12)
    try:
        res = minimize(fun=objective_and_grad, x0=y,
                       args=(n, fft_len, lam),
                       method='L-BFGS-B', jac=True,
                       options={'maxiter': maxiter, 'gtol':1e-12, 'disp': False})
    except Exception:
        return a_start
    a = np.exp(res.x - np.max(res.x))
    a = a / a.sum()
    return a

        if a0.sum() == 0: a0 = np.full(n,1/n)
        else: a0 = a0 / a0.sum()
        best_a = a0.copy()
        best_score = fast_score(best_a, fft_len)
else:
    best_a = np.full(n, 1.0/n)
    best_score = fast_score(best_a, fft_len)

"""
Propose a low‑score step‑function.

The algorithm works in three stages within the 1000 s budget:

1.  **Exploratory families** – simple parametric families (power‑law,
    exponential) are scanned on the simplex to obtain a good starting
    point.  This costs only a few seconds.

2.  **Mirror‑descent (exponentiated‑gradient)** – many short
    restarts of an exponentiated‑gradient descent on the true L∞‑objective
    are performed.  Each restart runs until a short sub‑deadline (≈10–15 s)
    and updates the global best sequence.

    A refined L‑BFGS pass on the smooth log‑sum‑exp surrogate is applied
    to the current best point to accelerate convergence.

3.  **Fine‑grained greedy hill‑climb** – the best sequence is refined by
    moving tiny fractions of mass from a source index to a destination
    index.  Convolution updates are performed incrementally (O(n) per
    move) and the exact convolution is recomputed periodically to avoid
    drift.

All steps keep the coefficients non‑negative and normalised to sum = 1,
so the official ``evaluate_sequence`` can be called at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def generate_power(n: int, p: float) -> np.ndarray:
    """Power‑law distribution a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    a = idx ** (-p)
    return a / a.sum()


def generate_exponential(n: int, r: float) -> np.ndarray:
    """Exponential distribution a_i ∝ r^{i}."""
    if abs(r - 1.0) < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = r ** np.arange(n, dtype=np.float64)
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and L‑BFGS refinement
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed max‑convolution: log‑sum‑exp proxy.
    Returns (value, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_refine(a_start: np.ndarray,
                 n: int, fft_len: int,
                 sub_deadline: float,
                 lam: float = 0.001) -> np.ndarray:
    """
    A short L‑BFGS run on the smoothed objective starting from a_start.
    """
    y = np.log(a_start + 1e-12)
    maxiter = 300
    try:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter,
                                'gtol': 1e-12,
                                'disp': False})
    except Exception:
        return a_start
    if not res.success:
        return a_start
    a = np.exp(res.x - np.max(res.x))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          sub_deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true L∞‑objective.
    Returns best a, best fast_score seen before sub_deadline.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < sub_deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Fast greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.08,
                           recompute_every: int = 1500) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index.  Convolution is updated incrementally
    (O(n) per move) and periodically recomputed exactly.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    diff = np.zeros_like(conv)                  # reusable buffer
    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger to smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            # accept move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # periodically recompute to keep numerical error bounded
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1

    return best_a, best_score

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 1234, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 points gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – try to use a previously discovered solution if present.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 0.0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin
    hill_time = 200.0                         # reserve for final hill‑climb
    phase1_deadline = deadline - hill_time

    # ------------------------------------------------------------------
    # Phase 1 – quick parametric scan (power‑law & exponential families)
    # ------------------------------------------------------------------
    for p in np.linspace(0.0, 1.5, 61):
        a = generate_power(n, p)
        s = fast_score(a, fft_len)
        if s < best_score:
            best_score, best_a = s, a.copy()
            print(f"[param‑power] p={p:.3f}  fast_score={best_score:.6f}")

    for r in np.linspace(0.2, 0.99, 40):
        a = generate_exponential(n, r)
        s = fast_score(a, fft_len)
        if s < best_score:
            best_score, best_a = s, a.copy()
            print(f"[param‑exp] r={r:.3f}  fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many short exponentiated‑gradient runs (random restarts)
    # ------------------------------------------------------------------
    sub_time = 12.0          # seconds per restart
    while time.time() < phase1_deadline - sub_time:
        a_start = random_initial(n, rng)
        # run a short EG optimisation
        a_opt, s_opt = exp_gradient_optimize(
            a_start, time.time() + sub_time, n, fft_len, rng, eta0=20.0
        )
        if s_opt < best_score:
            best_score, best_a = s_opt, a_opt.copy()
            print(f"[exp‑grad‑restart] new best fast_score={best_score:.6f}")

        # occasional L‑BFGS refinement on the current best
        if rng.random() < 0.15:
            a_ref = lbfgs_refine(best_a, n, fft_len, time.time() + sub_time / 2, lam=0.001)
            s_ref = fast_score(a_ref, fft_len)
            if s_ref < best_score:
                best_score, best_a = s_ref, a_ref.copy()
                print(f"[lbfgs‑refine] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – intensive greedy hill‑climb on the remaining budget
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_score = fast_score(best_a, fft_len)
        best_a, best_score = hill_climb_incremental(
            best_a,
            cur_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.10,
            recompute_every=1200,
        )
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
