"""
Propose a step function that minimizes the provided evaluation function.

The search proceeds in several stages:

1. **Initial families** – generate a few deterministic families (uniform,
   symmetric Beta, exponential, power‑law) and keep the best few on a fast
   FFT based score.
2. **Intensive L‑BFGS refinement** – start from the current best sequence,
   add a small Gaussian perturbation, and run a continuation of L‑BFGS on a
   log‑sum‑exp smoothed version of the max‑convolution objective.
   The λ‑schedule is taken down to 1e‑6 and each sub‑problem gets a large
   iteration budget.
3. **Exact exponentiated‑gradient descent** – fine‑tune the solution with the
   true max‑convolution sub‑gradient (the method used in the reference
   implementation).
4. **Greedy hill‑climbing** – a final polishing phase that transfers tiny
   amounts of mass between coordinates only when an improvement is obtained.

All steps respect the simplex constraint (non‑negative entries, sum = 1) and
are stopped just before the overall 1000 s budget expires.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) when a lies on the simplex.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate sequences
# ----------------------------------------------------------------------


def beta_shape(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density sampled on n points and projected to the simplex."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # log‑domain to avoid overflow for α<1
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    a = np.exp(log_a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def exponential_shape(rate: float, n: int) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-rate * i)
    right = left[::-1]
    a = left + right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def powerlaw_shape(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (-p)
    right = left[::-1]
    a = left + right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of the max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c       # ≤ 0
    exp_shift = np.exp(shifted / lam)  # safe, ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z                 # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient slice that corresponds to a (indices n‑1 … n‑1+n‑1)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous λ.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient on the true max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exact exponentiated‑gradient descent on the hard max‑convolution.
    The update a ← a·exp(−η·g) automatically stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size schedule – decays like 1/√t
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing (tiny mass transfers)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a modest random fraction of mass from one coordinate
    to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects a 1000 s budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Starting point (use historic solution if available)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # interpolate to required length
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
            else:
                a0 = old.copy()
                a0 = np.maximum(a0, 0.0)
                a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # --------------------------------------------------------------
    # Stage 0 – explore a few parametric families (tiny time budget)
    # --------------------------------------------------------------
    families = []

    # Uniform
    families.append(np.full(n, 1.0 / n))

    # Symmetric Beta
    for alpha in np.linspace(0.25, 1.0, 10):
        families.append(beta_shape(alpha, n))

    # Symmetric exponential
    for rate in np.linspace(0.0005, 0.005, 8):
        families.append(exponential_shape(rate, n))

    # Symmetric power‑law
    for p in np.linspace(0.2, 2.5, 9):
        families.append(powerlaw_shape(p, n))

    # Random initialisations (few)
    for _ in range(5):
        families.append(rng.random(n))

    # Evaluate each candidate and keep the best few
    candidate_pool = []
    for cand in families:
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        score = fast_score(cand, fft_len_conv, n)
        candidate_pool.append((score, cand))
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            print(f"[family] new best fast_score={best_score:.6f}")

    # keep the top K candidates for deeper optimisation
    candidate_pool.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in candidate_pool[:5]]
    if not top_candidates:
        top_candidates = [best_a]

    # --------------------------------------------------------------
    # Stage 1 – heavy L‑BFGS refinement (majority of the budget)
    # --------------------------------------------------------------
    # λ‑schedule reaching a very small smoothing factor
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
        0.000005, 0.000002
    ]
    maxiter_per_lam = 3000  # a bit larger than in the reference code

    # allocate ~80 % of the total budget to this phase
    phase1_deadline = start_time + 0.80 * budget_s

    while time.time() < phase1_deadline:
        # pick a starting point: either a perturbed best solution
        # or one of the top candidates with a small random jitter
        if rng.random() < 0.5 and best_a is not None:
            a_start = add_noise(best_a, sigma=rng.uniform(0.0005, 0.003), rng=rng)
        else:
            a_start = rng.choice(top_candidates)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

        # occasionally inject a fresh random start to avoid stagnation
        if time.time() < phase1_deadline and rng.random() < 0.1:
            a_start = rng.random(n)
            a_start = a_start / a_start.sum()
            a_opt, score_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                phase1_deadline,
                fft_len_conv,
                fft_len_g,
                n,
            )
            if score_opt < best_score:
                best_score = score_opt
                best_a = a_opt.copy()
                print(f"[L‑BFGS‑rand] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – exact exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final greedy hill‑climbing polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=50000,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # Provided by the environment
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
