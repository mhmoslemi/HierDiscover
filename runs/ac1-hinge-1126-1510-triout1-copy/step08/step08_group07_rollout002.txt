"""
Propose a low‑score step‑function (non‑negative sequence) for the
`evaluate_sequence` benchmark.

The optimiser works in four stages:

1️⃣ Heavy L‑BFGS on a smoothed max‑convolution (log‑sum‑exp) with a
   long schedule of decreasing smoothing parameters.  Randomised
   restarts (including small Gaussian perturbations of the current best)
   keep the search diverse.

2️⃣ ADAM‑style projected sub‑gradient descent on the *true* max‑convolution.
   The sub‑gradient is obtained from the active set of indices where the
   convolution attains (or almost attains) its maximum.
   After each ADAM step the iterate is projected onto the probability simplex
   (non‑negative, sum = 1).

3️⃣ Deterministic hill‑climbing: repeatedly move a small fraction of the mass
   from a “high‑gradient” index to a “low‑gradient” one and keep the move
   only if it strictly reduces the objective.  A fine list of fractions
   (down to 0.001) is tried so the search can perform very small refinements.

4️⃣ Ultra‑fine L‑BFGS polish with an almost‑zero smoothing factor.

All stages respect the simplex constraints, so the final score is
scale‑invariant.  The time budget (default 1000 s) is split automatically
between the phases; progress and intermediate scores are printed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """
    Fast evaluation of the benchmark objective for a simplex‑normalised a
    (∑a = 1):
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a with a = softmax(y).
    Returns (objective value, gradient w.r.t. y).
    """
    # --- softmax (stable) ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth_max w.r.t. convolution entries ---
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # --- back‑propagation through the convolution ---
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to the derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --- chain rule through softmax ---
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (multiple smoothing levels)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    # initialise in the log‑softmax space
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best (λ={lam:.1e}) score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient of the (unsmoothed) max‑convolution
# ----------------------------------------------------------------------


def subgradient_max_conv(a: np.ndarray, n: int, fft_len_g: int) -> np.ndarray:
    """
    Compute a sub‑gradient g of  max_i (a*a)_i  w.r.t. a.
    g has shape (n,) and is non‑negative.
    """
    # convolution a*a
    fa = np.fft.fft(a, fft_len_g)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # back‑propagation through the convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return g


# ----------------------------------------------------------------------
# ADAM‑style projected sub‑gradient descent
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    ADAM on the sub‑gradient of the max‑convolution, with Euclidean
    projection onto the simplex after each step.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 1

    # learning rate schedule – start relatively large and decay
    base_lr = 0.6

    while time.time() < deadline:
        g = subgradient_max_conv(a, n, fft_len_g)  # sub‑gradient of max

        # ADAM moment updates
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = base_lr / np.sqrt(t)                 # simple decay

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # move opposite to the gradient (we want to *decrease* the max)
        a_new = a - step
        a_new = simplex_projection(a_new)

        cur_score = fast_score(a_new, fft_len_conv, n)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
            # occasional progress report
            if t % 200 == 0:
                print(f"[ADAM] iter={t} best_score={best_score:.6f}")

        a = a_new
        t += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb with line‑search over several fractions
# ----------------------------------------------------------------------


def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb that transfers a fraction of the mass from a
    high‑gradient index to a low‑gradient one.  Multiple fractions are tried
    and only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    # fractions spanning from very small to moderate – this gives fine
    # granularity near convergence.
    fractions = [0.001, 0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.10]

    while time.time() < deadline and it < max_iter:
        # current convolution and active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (same as in ADAM)
        best_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        f_rev = np.fft.fft(best_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(f_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source / target based on gradient percentiles
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        improved = False
        best_local_score = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta

            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local_score - 1e-12:
                best_local_score = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local_score

        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                      # length of the sequence
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Starting point (reuse previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to the current length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    phase1_end = start_time + 0.38 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # ADAM
    phase3_end = start_time + 0.95 * budget_s   # hill‑climb
    # remaining time: final ultra‑fine L‑BFGS

    # --------------------------------------------------------------
    # Phase 1 – robust L‑BFGS continuation with decreasing smoothing
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 1e-06,
        5e-07, 2e-07, 1e-07,
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # diversify starts – occasionally perturb the current best
        if rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.018)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1 L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – ADAM projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Phase‑2 ADAM] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – fine‑grained gradient‑guided hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = targeted_hill_climb_line_search(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iter=300000,
        )
        print(f"[Phase‑3 hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – ultra‑fine L‑BFGS polish with a near‑zero smoothing factor
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_final, fine_score = lbfgs_continuation(
            best_a,
            [1e-08],
            maxiter_per_lam=500,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score:
            best_score = fine_score
            best_a = a_final.copy()
            print(f"[Phase‑4 final L‑BFGS] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final external evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
