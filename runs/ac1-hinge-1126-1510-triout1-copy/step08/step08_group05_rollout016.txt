        x_old = np.linspace(0, 1, old.size)
        x_new = np.linspace(0, 1, n)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start = time.time()
    deadline = start + budget_s - 0.2

    # number of random restarts
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        # generate random initial
        a_start = random_initial(n, rng)
        # mix with best with some weight for diversity
        if rng.random() < 0.3:
            a_start = 0.5*best_a + 0.5*a_start
            a_start /= a_start.sum()
        # run Adam subgradient for a fixed iteration budget
        a_opt, score_opt = subgradient_adam(a_start, deadline, fft_len_conv, fft_len_g, n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[restart {r}] new best after Adam: {best_score:.6f}")
        # hill climb
        a_opt2, score_opt2 = targeted_hill_climb_aggressive(best_a, best_score,
                                                             fft_len_conv, fft_len_g,
                                                             n, deadline,
                                                             max_iter=200000)
        if score_opt2 < best_score:
            best_score, best_a = score_opt2, a_opt2.copy()
            print(f"[restart {r}] new best after hill: {best_score:.6f}")

    # final fine L-BFGS with very small λ schedule
    if time.time() < deadline:
        fine_lam_schedule = [5e-06, 2e-06, 1e-06, 5e-07, 2e-07]
        fine_maxiter = 400
        best_a, fine_score = lbfgs_continuation(best_a, fine_lam_schedule,
                                                fine_maxiter, deadline,
                                                fft_len_conv, fft_len_g, n)
        if fine_score < best_score:

    final_seq = best_a.tolist()
    # use external evaluation to produce final metric (optional)
    val = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence result = {val:.6f}")

    return final_seq

"""
Optimisation driver for the “step‑function” challenge.

The objective is  

    f(a) = 2·n·max_k (a * a)[k] / (∑ a_i)² .

Because the objective is homogeneous we may (and do) enforce ∑ a_i = 1,
so that   f(a) = 2·n·max_k (a*a)[k] .
The algorithm works on a single simplex‑constrained vector `a` :

1.  A collection of diverse random starts is generated (uniform,
    exponential, power‑law, edge‑heavy, bimodal, …).  If a previous
    solution ``height_sequence_1`` exists it is interpolated to the
    current size and used as an additional start.

2.  For every start we run a *projected Adam* optimiser that uses the
    exact sub‑gradient of the max‑convolution.  The step‑size decays as
    1/√t and the iterate is projected onto the simplex after each
    update.

3.  The best iterate from the Adam phase is refined by an aggressive
    **gradient‑guided hill‑climb**: the sub‑gradient identifies a
    “source’’ index (large gradient) and a “target’’ index (small
    gradient).  A small fraction of the source mass is transferred to
    the target; several fractions are tried and the move is kept only
    if it strictly lowers the objective.  This simple greedy
    neighbourhood search is very cheap and often finds improvements
    that the smooth optimiser misses.

4.  A short continuation of L‑BFGS on a Log‑Sum‑Exp smoothed version
    of the max‑convolution (decreasing smoothing parameters) polishes
    the final candidate.

The whole routine respects the 1000 s wall‑time limit, repeatedly
checking the deadline.  The returned list is already normalised
(∑a = 1) and therefore can be fed directly to the supplied
``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation when a is already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diversified random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "linear"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  #   linear decreasing (a_0 largest, a_{n-1}=0)
        a = np.arange(n, 0, -1, dtype=float)
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) + its gradient (w.r.t. logits)
# ------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # d smooth_max / d conv_i
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.  Returns the best a and
    its fast_score.
    """
    y = np.log(a_start + 1e-12)          # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ------------------------------------------------------------
# Projected Adam using the true max‑convolution
# ------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (adaptive moments) using the exact sub‑gradient of
    max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # mask of active (max) indices (allow a tiny tolerance)
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the max (see derivation in the description)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.8 / np.sqrt(it)            # a slightly larger step than before
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to jump out of plateaus
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.005)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ------------------------------------------------------------
# Aggressive gradient‑guided hill‑climb
# ------------------------------------------------------------


def targeted_hill_climb_aggressive(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 100000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that uses the sub‑gradient to move mass from
    high‑gradient indices to low‑gradient ones.  Several candidate
    transfer fractions are tested and only strictly improving moves
    are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # current convolution & active max mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        f_rev = np.fft.fft(rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(f_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source (large gradient) and target (small gradient)
        high_thr = np.percentile(g, 90)
        low_thr = np.percentile(g, 10)
        high_idx = np.where(g >= high_thr)[0]
        low_idx = np.where(g <= low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = int(rng.choice(high_idx))
            j = int(rng.choice(low_idx))

        if i == j:
            it += 1
            continue

        improved = False
        # try a few transfer fractions in increasing order
        for frac in (0.02, 0.05, 0.10, 0.15):
            delta = best[i] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            new_score = fast_score(cand, fft_len_conv, n)
            if new_score < best_score:
                best = cand
                best_score = new_score
                improved = True
                break  # accept first improving fraction

        if not improved:
            # occasional random perturbation to avoid stagnation
            if it % 2500 == 0:
                sigma = rng.uniform(0.001, 0.004)
                best = best + rng.normal(0.0, sigma, size=n)
                best = np.maximum(best, 0.0)
                best = best / best.sum()

        it += 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 8000                     # 2 × previous size, still fast enough

    if "height_sequence_1" in globals():

        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a
    fft_len_g = next_pow2(3 * n - 2)     # length needed for gradient steps

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2   # a small safety margin

    # Allocate time slices
    phase1_end = start + 0.30 * (budget_s - 0.2)   # heavy Adam + restarts
    phase2_end = start + 0.70 * (budget_s - 0.2)   # hill‑climb
    # remaining time → fine L‑BFGS

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    restarts = 7

        a_start = random_initial(n, rng)

        a_opt, score_opt = subgradient_adam(

        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[restart {r}] new best after Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – aggressive gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = targeted_hill_climb_aggressive(

            fft_len_conv,
            fft_len_g,
            n,
            phase2_end,
            max_iter=150000,
        )
        print(f"[hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short fine‑tuned L‑BFGS with tiny λ values
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [1e-05, 5e-06, 2e-06, 1e-06, 5e-07, 2e-07]
        fine_maxiter = 350
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,

    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence result = {final_eval:.6f}")

    return final_seq
