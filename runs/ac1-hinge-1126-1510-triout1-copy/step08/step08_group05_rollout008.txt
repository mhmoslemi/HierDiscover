"""
Propose a step‑function that (hopefully) beats the previous best score
(~1.5065).  The algorithm is a multi‑phase continuous optimisation on the
simplex (∑a = 1, a_i ≥ 0).

Phases
-------
1. **Many short L‑BFGS runs** on a smooth log‑sum‑exp approximation of the
   max‑convolution.  A very aggressive λ‑schedule and tiny max‑iter per
   λ allow a large number of independent restarts, which is crucial for
   escaping poor local minima.

2. **Projected Adam** (a sub‑gradient method) works directly on the exact
   max‑convolution.  Adam’s adaptive step‑size together with a Euclidean
   projection onto the simplex makes fast progress even after the smooth
   stage.

3. **Deterministic hill‑climbing**: the exact sub‑gradient identifies
   “source’’ indices (large gradient) and “target’’ indices (small
   gradient).  A small amount of mass is transferred from source to target
   using a line‑search over several fractions; only strictly improving
   moves are kept.

4. **Fine L‑BFGS** with an ultra‑small λ (≈10⁻⁸) to polish the final
   candidate.

The whole routine respects a 1000 s wall‑clock limit and returns a list
of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a simplex‑normalised vector (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "double_power"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        # a few heavy entries at the two ends
        edge_frac = rng.uniform(0.005, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = rng.uniform(0.5, 2.0) / m_edge
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # double_power – symmetric power‑law decreasing toward the centre
        p = rng.uniform(0.5, 2.0)
        left = (np.arange(n) + 1) ** (-p)
        right = (np.arange(n, 0, -1) + 1) ** (-p)
        a = left + right
    a /= a.sum()
    return a


def interpolate_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from an old vector to a new length."""
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    return new if s == 0 else new / s


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Return (smooth objective, gradient) for a = softmax(y).
    The smooth max of the convolution is approximated by a log‑sum‑exp
    with a λ proportional to the current max.
    """
    # softmax, stable
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weight of each convolution entry in the smooth max
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑prop through convolution:
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient of max w.r.t. a (length n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy or fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run a sequence of L‑BFGS optimisations, one for every λ in
    ``lam_schedule``.  The result of the previous λ is used as the start
    for the next.  Returns the best vector (on the simplex) and its
    exact ``fast_score``.
    """
    y = np.log(a_start + 1e-12)  # logits, keep positivity implicitly
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # any numerical hiccup – just skip this λ
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        # back‑to the simplex
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that uses the exact sub‑gradient of the max‑convolution.
    After each step the vector is projected back onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- exact convolution and max mask -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # ----- sub‑gradient of the max (via convolution with mask) -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size slowly decays (base = 0.3 works well for n≈8000)
        step = 0.3 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the simplex -----
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.008)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Aggressive gradient‑guided hill‑climb (deterministic moves)
# ----------------------------------------------------------------------


def targeted_hill_climb_aggressive(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 80000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly transfers a small fraction of mass
    from a “source’’ index (large sub‑gradient) to a “target’’ index (small
    sub‑gradient).  Several candidate fractions are examined; the first
    improving move is kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        # convolution and mask for the current max
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        rev_f = np.fft.fft(rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(rev_f * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # high‑gradient (source) and low‑gradient (target) indices
        high_thr = np.percentile(g, 90)
        low_thr = np.percentile(g, 10)
        high_idx = np.where(g >= high_thr)[0]
        low_idx = np.where(g <= low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j:
            it += 1
            continue

        # try a few fractions – adaptively search for a beneficial delta
        improved = False
        for frac in (0.02, 0.05, 0.09, 0.15):
            delta = best[i] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
            new_score = fast_score(cand, fft_len_conv, n)
            if new_score < best_score:
                best = cand
                best_score = new_score
                improved = True
                break
        if not improved:
            # tiny random perturbation occasionally to shake the system
            if it % 5000 == 0:
                sigma = rng.uniform(0.0005, 0.003)
                best = best + rng.normal(0.0, sigma, size=n)
                best = np.maximum(best, 0.0)
                best = simplex_projection(best)

        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser that returns a list of non‑negative floats.
    The routine respects the 1000 s time budget and prints intermediate
    progress.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n often yields a smaller constant.
    # --------------------------------------------------------------
    n = 8000

    # initialise from a previous solution if available, otherwise random
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            a0 = interpolate_sequence(old, n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.15         # small safety margin

    phase1_end = start + 0.38 * (budget_s - 0.15)    # many short L‑BFGS runs
    phase2_end = start + 0.68 * (budget_s - 0.15)    # Adam
    phase3_end = start + 0.92 * (budget_s - 0.15)    # aggressive hill‑climb
    # remainder → fine L‑BFGS

    # --------------------------------------------------------------
    # Phase 1 – many short L‑BFGS runs with a broad λ‑schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.16, 0.10, 0.06,
        0.035, 0.020, 0.012, 0.007, 0.004,
        0.0025, 0.0015, 0.0009, 0.0005, 0.0003,
        0.00015, 0.00007, 0.00003,
    ]
    maxiter_per_lam = 350   # enough for a quick descent, many restarts

    while time.time() < phase1_end:
        # occasionally perturb the current best to diversify
        if rng.random() < 0.30:
            sigma = rng.uniform(0.0005, 0.015)
            a_start = best_a + rng.normal(0.0, sigma, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start = simplex_projection(a_start)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Projected Adam on the exact objective
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = targeted_hill_climb_aggressive(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iter=120000,
        )
        print(f"[gradient hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – very fine L‑BFGS polishing
    # --------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [1e-05, 5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07]
        fine_maxiter = 300
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation (using the external checker)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
