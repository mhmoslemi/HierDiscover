def directed_hill_climb(a, curr_score, n, fft_len, deadline, rng,
                         max_fraction=0.05, recompute_every=1000):
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)  # full conv, length 2n-1
    accepted = 0
    it = 0
    while time.time() < deadline:
        # compute gradient at current best_a
        # conv already computed
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient computation via FFT
        a_rev = best_a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad[n-1:n-1+n]
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if best_a[src] <= 0.0:
            # maybe all zero at src, break
            break
        # move fraction
        delta = best_a[src] * max_fraction * rng.random()
        if delta <= 0:
            continue
        # incremental update conv
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        if new_score < best_score:
            # accept
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1
        # periodic recomputation to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)
        it += 1
    return best_a, best_score

"""
Step‑function optimizer for the challenge.

The search is broken into three phases (under a 1000 s budget):

1. **Exponentiated‑gradient descent** on the true L∞ objective.
   The update works on the probability simplex (mirrored descent)
   and runs until 60 % of the time budget is exhausted.

2. **Directed gradient hill‑climb**.
   At every iteration the sub‑gradient of the current maximal
   convolution entry is computed (via FFT).  Mass is moved from the
   index with the largest sub‑gradient to the one with the smallest,
   accepting only moves that lower the maximal convolution value.
   This phase consumes the next 25 % of the budget.

3. **Simulated‑annealing hill‑climb**.
   Random source/destination pairs are tried; a move that worsens the
   score may still be accepted with a temperature‑dependent probability.
   The remaining time is spent here.

All phases keep the sequence non‑negative and normalised
(`sum(a)=1`).  The final sequence is evaluated with the provided
`evaluate_sequence` function and returned as a plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two  ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) – the quantity used by `evaluate_sequence` when sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A diverse random start on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exponentiated_gradient(a_start: np.ndarray,
                           deadline: float,
                           n: int,
                           fft_len: int,
                           rng: np.random.Generator,
                           eta0: float = 20.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the simplex using the sub‑gradient of max(a*a)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]          # size n

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:                # numerical safety‑net
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – directed gradient hill‑climb
# ----------------------------------------------------------------------


def directed_hill_climb(a: np.ndarray,
                       curr_score: float,
                       n: int,
                       fft_len: int,
                       deadline: float,
                       rng: np.random.Generator,
                       max_fraction: float = 0.05,
                       recompute_every: int = 1500) -> tuple[np.ndarray, float]:
    """
    Greedy deterministic hill‑climb.
    At each step the sub‑gradient of the current maximal convolution entry
    is computed; mass is moved from the index with the largest gradient to
    the index with the smallest gradient.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    diff = np.zeros_like(conv)                  # reused buffer
    accepted = 0
    it = 0

    while time.time() < deadline:
        # sub‑gradient of the current maximal entry
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best_a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or best_a[src] <= 0.0:
            break

        # transfer a modest fraction of mass
        delta = best_a[src] * max_fraction * rng.random()
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept the move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # periodic exact recomputation to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing hill‑climb (random moves)
# ----------------------------------------------------------------------


def annealing_hill_climb(a: np.ndarray,
                         curr_score: float,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_fraction: float = 0.08,
                         recompute_every: int = 2000,
                         init_temp: float = 1e-2,
                         final_temp: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Random hill‑climb with a temperature schedule.
    Even moves that increase the score can be accepted with a
    decreasing probability, allowing the search to escape shallow
    local minima.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)        # exact full convolution
    diff = np.zeros_like(conv)
    accepted = 0
    start = time.time()
    total = deadline - start

    while time.time() < deadline:
        # linear temperature decay
        elapsed = time.time() - start
        temp = init_temp * (1.0 - elapsed / total) + final_temp * (elapsed / total)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from the larger entry to the smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] <= 0.0:
            continue

        delta = best_a[src] * max_fraction * rng.random()
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        improve = new_score < best_score
        accept = improve or (rng.random() < math.exp(-(new_score - best_score) / max(temp, 1e-12)))

        if accept:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            accepted += 1
            if new_score < best_score:
                best_score = new_score

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

    return best_a, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The function respects a ~1000 s time budget and returns the best
    sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 points gives a good speed/quality trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previously found construction if available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    # Phase 1 – exponentiated‑gradient (≈60 % of budget)
    phase1_deadline = start_time + 0.60 * budget_s
    if time.time() < phase1_deadline:
        best_a, best_score = exponentiated_gradient(
            best_a, phase1_deadline, n, fft_len, rng, eta0=20.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # Phase 2 – directed gradient hill‑climb (≈25 % of budget)
    phase2_deadline = start_time + 0.85 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = directed_hill_climb(
            best_a, best_score, n, fft_len, phase2_deadline, rng,
            max_fraction=0.05, recompute_every=1500
        )
        print(f"[direct‑hill] fast_score={best_score:.6f}")

    # Phase 3 – simulated‑annealing hill‑climb (remaining time)
    if time.time() < deadline:
        best_a, best_score = annealing_hill_climb(
            best_a, best_score, n, fft_len, deadline, rng,
            max_fraction=0.08, recompute_every=2000,
            init_temp=1e-2, final_temp=1e-6
        )
        print(f"[anneal‑hill] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
