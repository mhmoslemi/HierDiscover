def local_annealing(a, deadline, rng):
    n = len(a)
    a = a.copy()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_val = conv.max()
    # store best_score = 2*n*best_val / sum_a^2 (but sum_a = 1)
    # Temperature schedule: linear cooling; T0 = 0.05 * best_val? Actually maybe start with fraction of best_val.
    T0 = 0.02 * best_val
    start = time.time()
    total_time = deadline - start
    while time.time() < deadline:
        # update temperature linearly
        remain = deadline - time.time()
        T = T0 * (remain / total_time)
        # choose move
        i = rng.integers(n)
        # skip if a[i]==0
        if a[i] <= 1e-12:
            continue
        # choose j distinct
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        # decide delta
        max_delta = a[i]
        frac = rng.random() * 0.2  # up to 20% of mass at i
        delta = max_delta * frac
        if delta == 0:
            continue
        # Save old values
        a_i_old = a[i]
        a_j_old = a[j]
        # compute conv update using old a (we need a_old = a before move)
        a_old = a.copy()
        # apply move to a
        a[i] -= delta
        a[j] += delta
        # update conv
        conv[i:i+n] -= 2*delta*a_old
        conv[j:j+n] += 2*delta*a_old
        conv[2*i] += delta*delta
        conv[2*j] += delta*delta
        conv[i+j] -= 2*delta*delta
        # compute new_max
        new_max = conv.max()
        # acceptance
        diff = new_max - best_val
        if diff < 0 or rng.random() < np.exp(-diff / (T+1e-12)):
            # accept change
            if new_max < best_val:
                best_val = new_max
                best_a = a.copy()
        else:
            # revert
            # revert a
            a[i] = a_i_old
            a[j] = a_j_old
            # revert conv using inverse updates (with same delta and a_old)
            conv[i:i+n] += 2*delta*a_old
            conv[j:j+n] -= 2*delta*a_old
            conv[2*i] -= delta*delta
            conv[2*j] -= delta*delta
            conv[i+j] += 2*delta*delta
        # loop
    # final best_score
    final_score = 2 * len(a) * best_val # sum=1
    return best_a, final_score

"""
Propose a low‑evaluation step sequence.

The algorithm consists of three stages:

1. **Cheap parametric sweep** – generate many random families
   (power‑law, exponential, Gaussian, cosine, bimodal, random)
   and keep the best normalised vector.

2. **Random‑restart L‑BFGS** – optimise a smoothed version of the
   max‑convolution using the soft‑max parametrisation.
   Several restarts with small jitter around the current best are
   performed while respecting the simplex constraint.

3. **Exact local annealing** – starting from the best vector found so
   far, repeatedly move a tiny fraction of mass from one index to another.
   The convolution is updated *exactly* with O(n) arithmetic using the
   analytical formula for a rank‑2 update, which allows millions of
   cheap trials.  A simulated‑annealing schedule accepts uphill moves,
   guaranteeing exploration of the neighbourhood and further reduction
   of the maximum convolution value.

The best sequence found before the deadline (default 1000 s) is returned.
All numbers are non‑negative; the vector is normalised to unit L1‑norm
so that the objective equals `2·n·max(conv)`, which is invariant under
scaling.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) for a that already satisfies ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each smoothing parameter in `lam_schedule`,
    starting each stage from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Random parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    if family == "power":
        p = rng.uniform(-5.0, 5.0)  # broadened range
        inc = rng.choice([True, False])
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** p if inc else idx[::-1] ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(150.0, 2500.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, max(200.0, n / 2.0))
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(80.0, 600.0)
        sigma2 = rng.uniform(80.0, 600.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:  # pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Perform a cheap sweep over many random families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + 0.15 * budget_s

    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact local annealing (O(n) update of the convolution)
# ----------------------------------------------------------------------


def local_annealing(
    a_start: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    move_frac: float = 0.05,
    temperature_factor: float = 0.02,
    recompute_interval: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb that updates the self‑convolution
    exactly after each elementary mass transfer.  The algorithm works
    on the simplex (total mass = 1) and keeps the best vector seen.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)           # length 2n‑1
    best_a = a.copy()
    best_max = conv.max()
    cur_max = best_max

    # temperature schedule – initial temperature is a small fraction of the score
    T0 = temperature_factor * best_max
    start_time = time.time()
    total_time = max(deadline - start_time, 1e-6)
    iteration = 0

    while time.time() < deadline:
        # linear cooling
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        if a[i] <= 1e-14:
            continue
        # choose a distinct j
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # move a random fraction of the mass at i to j
        delta = a[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue

        # store copies for a cheap revert if needed
        a_old = a.copy()
        conv_old = conv.copy()

        # apply move
        a[i] -= delta
        a[j] += delta

        # exact O(n) convolution update using the analytical formula
        # (based on conv_new = conv_old + 2·conv(a_old, Δ) + conv(Δ,Δ))
        conv[i : i + n] -= 2.0 * delta * a_old
        conv[j : j + n] += 2.0 * delta * a_old
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        new_max = conv.max()
        diff = new_max - cur_max

        accept = False
        if diff < 0:
            accept = True
        else:
            # uphill move accepted with probability exp(-diff/T)
            if T > 0:
                accept = rng.random() < np.exp(-diff / (T + 1e-12))

        if accept:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
        else:
            # revert to previous state
            a = a_old
            conv = conv_old
            # cur_max unchanged

        # occasional full recomputation to avoid numerical drift
        if iteration % recompute_interval == 0 and iteration > 0:
            conv = np.convolve(a, a)
            cur_max = conv.max()
        iteration += 1

    return best_a, best_max

    # --------------------------------------------------------------
    # Phase 2 – intensive exact local annealing (remaining time)
    # --------------------------------------------------------------
    a_final, max_conv = local_annealing(best_a, deadline, rng)
    final_score = fast_score(a_final, fft_len_conv, n)
    print(f"[local‑anneal] final fast_score={final_score:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and evaluation
    # ------------------------------------------------------------------
    a_final = np.maximum(a_final, 0.0)
    total = a_final.sum()
    if total == 0.0:
        a_final = np.full(n, 1.0 / n)
    else:
        a_final /= total

    final_seq = a_final.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the host environment
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
