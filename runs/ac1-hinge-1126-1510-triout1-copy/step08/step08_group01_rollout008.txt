"""
Propose a step‐function sequence that minimizes the evaluation metric.

Strategy
--------
1. Work on a probability vector a (∑a = 1).  The score to minimise is
       score(a) = 2·n·max(a∗a)               (fast_score)
   where a∗a is the linear convolution of a with itself.
2. Search a low‑dimensional parametric family that is known to flatten the
   auto‑convolution:
      * symmetric exponential decay from the two ends,
      * symmetric power‑law decay,
      * the discrete arcsine law (≈ 1/√(i·(n‑1‑i))).
   For each family a dense grid of parameters is tried and the best
   candidate (according to fast_score) is kept.
3. Starting from the best parametric candidate run an *incremental greedy
   hill‑climb* that moves a small fraction of the mass from one index to
   another.  The convolution is updated in O(n) per move, so many moves
   can be performed within the remaining time.
4. Return the best sequence found before the deadline.  The final
   sequence is checked with the official ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute the score 2·n·max(a*a) for a probability vector a.
    This is exactly the value returned by ``evaluate_sequence`` when
    ``sum(a) == 1`` (the function is scale‑invariant).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Families of candidate shapes
# ----------------------------------------------------------------------


def make_exponential(n: int, lam: float, eps: float = 0.0) -> np.ndarray:
    """
    Symmetric exponential decay from both ends.
    a[i] ∝ exp(-lam*i) + exp(-lam*(n-1-i)) + eps
    """
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    if eps > 0.0:
        a += eps
    a /= a.sum()
    return a


def make_power(n: int, p: float, eps: float = 0.0) -> np.ndarray:
    """
    Symmetric power‑law decay.
    a[i] ∝ (i+1)^(-p) + (n-i)^(-p) + eps
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (idx ** (-p)) + ((n + 1 - idx) ** (-p))
    if eps > 0.0:
        a += eps
    a /= a.sum()
    return a


def make_arcsine(n: int, eps: float = 0.0) -> np.ndarray:
    """
    Discrete arcsine law: a[i] ∝ 1/√(i·(n‑1‑i)).
    The law is known to flatten the self‑convolution.
    """
    idx = np.arange(1, n + 1, dtype=np.float64) - 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    if eps > 0.0:
        a += eps
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Incremental greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           start_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.12,
                           recompute_every: int = 1500) -> tuple[np.ndarray, float]:
    """
    Greedy Hill‑Climb:
      * move a random fraction of mass from a source index to a destination index,
      * update the full convolution incrementally (O(n) per move),
      * accept the move only if it strictly decreases the score.
    """
    best_a = a.copy()
    best_score = start_score
    conv = np.convolve(best_a, best_a)               # exact full convolution (length 2n‑1)
    diff = np.zeros_like(conv)                       # reusable temporary buffer
    accepted = 0
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] <= 0.0:
            continue

        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # Incremental update of the convolution.
        # (a*a) new = old + 2·delta·(e_dst – e_src) * a   where e_k is the basis vector.
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

            # Occasionally recompute the convolution from scratch to avoid drift.
            if accepted % recompute_every == 0:
                conv = np.convolve(best_a, best_a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 points gives a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial uniform distribution (∑a = 1).
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Parameterised families (≈ 20 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.20 * budget_s
    # -------- exponential family ----------
    lambdas = np.logspace(-4, 0, num=30)                 # 1e‑4 … 1
    epsilons = [0.0, 1e-5, 1e-4, 5e-4]
    for lam in lambdas:
        if time.time() >= phase1_deadline:
            break
        for eps in epsilons:
            a = make_exponential(n, lam, eps)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[exp] λ={lam:.5g} eps={eps:g} score={sc:.6f}")

    # -------- power‑law family ----------
    powers = np.linspace(0.2, 3.0, 30)
    for p in powers:
        if time.time() >= phase1_deadline:
            break
        for eps in epsilons:
            a = make_power(n, p, eps)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[power] p={p:.3f} eps={eps:g} score={sc:.6f}")

    # -------- arcsine family ----------
    for eps in epsilons:
        if time.time() >= phase1_deadline:
            break
        a = make_arcsine(n, eps)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[arcsine] eps={eps:g} score={sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Intensive hill‑climb on the best candidate (rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.12,
            recompute_every=1500,
        )
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
