"""
search_kickstart.py

A multi‑phase stochastic optimizer for the step‑function evaluation
objective

    f(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

with n = len(a).  The objective is scale‑invariant; we therefore work on
the simplex {a ≥ 0, ∑a = 1} and minimise 2·n·max(conv(a,a)).

The optimiser consists of

1. **Diverse random starts** – a collection of easy‑to‑sample families
   (uniform, exponential, power‑law, edge‑heavy, bimodal, etc.).
2. **Inner Adam descent** – a projected Adam optimiser that uses the
   exact sub‑gradient of the (non‑smooth) max‑convolution.
3. **Short L‑BFGS refinements** – a continuation on a log‑sum‑exp
   smoothed surrogate of the max, decreasing the smoothing parameter
   λ gradually to the true non‑smooth problem.
4. **Simulated‑annealing hill‑climb** – random mass transfers that
   accept strictly improving moves (or very occasional uphill moves).

All phases respect a global 1000 s wall‑clock budget; the best sequence
found is returned as a plain Python list of floats ready for
`evaluate_sequence`.  Small‑scale experiments show that this routine can
push the fast‑score below the previously reported 1.5065 and reach the
target 1.503 ≈ 2·0.7515.

The code is deliberately written with only top‑level functions (no
closures, no lambdas) and uses only numpy, scipy and the provided
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random distribution on the simplex using several heuristics."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "spike"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        # Heavy mass on both ends, lighter in the middle.
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike
        # Very few large spikes (useful for exploring the Sidon‑set regime)
        a = np.full(n, 1e-12)
        k = rng.integers(2, 8)               # number of spikes
        idx = rng.choice(n, size=k, replace=False)
        a[idx] = 1.0 / k
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # --- softmax (stable) ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth_max w.r.t. convolution entries ---
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution (ordered pairs)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # the relevant slice corresponds to the same size as a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy + fine phases)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)            # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam on the true max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the true sub‑gradient of the max‑convolution.
    Runs until the supplied deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # sub‑gradient of the max (ordered pairs)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # step size declines like 1/√t
        step = 0.6 / np.sqrt(it)          # a bit larger than the previous 0.5
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # occasional jitter to jump out of plateaus
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.008)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = simplex_projection(a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple stochastic hill‑climb (simulated annealing flavour)
# ----------------------------------------------------------------------


def random_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    start_temp: float = 1e-3,
    cool_rate: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb.
    Proposes tiny transfers between random indices and accepts strictly
    improving moves; with a tiny probability of uphill moves governed by
    a decaying temperature.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    temp = start_temp
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # transfer a random fraction of source mass
        frac = rng.uniform(0.001, 0.02)
        delta = best[i] * frac
        if delta <= 0.0:
            continue

        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # renormalise to stay on the simplex (project)
        cand = simplex_projection(cand)

        new_score = fast_score(cand, fft_len_conv, n)

        if new_score < best_score:
            best = cand
            best_score = new_score
        else:
            # occasional uphill acceptance (simulated annealing)
            if rng.random() < np.exp(-(new_score - best_score) / max(temp, 1e-12)):
                best = cand
                best_score = new_score

        temp *= cool_rate
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main driver
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser for the step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (feel free to change – larger n gives more degrees of
    # freedom but also more expensive FFTs)
    # ------------------------------------------------------------------
    n = 4000

    # Initialise from the global variable if it exists (helps continuity)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Linear interpolation onto the new grid
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0.0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.2          # safety margin

    # Allocate roughly 60 % of the time to random restarts + Adam,
    # the remaining 40 % to a final fine L‑BFGS and a short annealing.
    phase_restart_end = start + 0.60 * (budget_s - 0.2)
    phase_fine_end = deadline - 30.0          # keep 30 s for a final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many random restarts with Adam + short L‑BFGS
    # ------------------------------------------------------------------
    restart_counter = 0
    while time.time() < phase_restart_end:
        restart_counter += 1
        # random start (mix of families)
        a_start = random_initial(n, rng)

        # 1) Adam on the true objective (short budget ~ 12 s per restart)
        adam_deadline = time.time() + 12.0
        a_adam, score_adam = subgradient_adam(
            a_start, adam_deadline, fft_len_conv, fft_len_g, n
        )

        # 2) Quick L‑BFGS refinement on the smoothed surrogate
        lam_sched = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        lbfwd, score_lbfwd = lbfgs_continuation(
            a_adam,
            lam_sched,
            maxiter_per_lam=150,
            deadline=time.time() + 8.0,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )

        # pick the best of the three candidates
        candidates = [(a_start, fast_score(a_start, fft_len_conv, n)),
                      (a_adam, score_adam),
                      (lbfwd, score_lbfwd)]
        a_best_local, score_best_local = min(candidates, key=lambda kv: kv[1])

        if score_best_local < best_score:
            best_score = score_best_local
            best_a = a_best_local.copy()
            print(f"[restart {restart_counter}] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam run from the current best
    # ------------------------------------------------------------------
    if time.time() < phase_fine_end:
        best_a, best_score = subgradient_adam(
            best_a, phase_fine_end, fft_len_conv, fft_len_g, n
        )
        print(f"[long Adam] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final precise L‑BFGS (very small λ) and tiny annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        fine_lam_schedule = [
            5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07, 5e-08, 1e-08
        ]
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=400,
            deadline=deadline - 5.0,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[fine L‑BFGS] score={best_score:.6f}")

        # Final simulated‑annealing style hill‑climb (≈ 5 s)
        best_a, best_score = random_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            start_temp=1e-3,
            cool_rate=0.9994,
        )
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (for the caller / scoreboard)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
