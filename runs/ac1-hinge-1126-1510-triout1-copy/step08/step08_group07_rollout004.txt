        # linear cooling

        # choose source i with positive mass > small epsilon

"""
Propose a low‑score step‑function sequence.

The optimizer combines several complementary phases:

1️⃣  Many random restarts of an L‑BFGS continuation on a log‑sum‑exp
    smoothed version of the maximum convolution.  The smoothing
    parameter λ is driven down geometrically, allowing the optimizer
    to approach the true (non‑smooth) objective.

2️⃣  A projected exponentiated‑gradient (mirror‑descent) stage that
    follows the exact sub‑gradient of the max‑convolution on the
    probability simplex.

3️⃣  A stochastic annealing stage that performs random mass‑transfer
    moves with Metropolis acceptance; this helps to escape shallow
    local minima that L‑BFGS may miss.

4️⃣  A deterministic gradient‑guided hill‑climb with line‑search on a
    small set of fractions (the same routine used in the reference).

5️⃣  A final ultra‑low‑λ L‑BFGS polish (several λ values, many iterations)
    to tighten the remaining kinks.

All phases respect non‑negativity and the simplex constraint; the
evaluation function is scale‑invariant, so we always keep the vector
normalised to sum = 1.  The total run‑time is limited to the supplied
budget (≈ 1000 s).  The best sequence found is returned as a list of
floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation when a is normalised (∑a = 1):
        2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. convolution entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ----- back‑propagation through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    # log‑softmax space
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex using the true sub‑gradient
    of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        # ----- convolution a*a -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the max‑convolution (size n) -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ----- exponentiated‑gradient update -----
        eta = 0.20 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def targeted_hill_climb_line_search(
    a: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iter: int = 150000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic gradient‑guided hill‑climb.
    For each (source, target) pair several mass‑transfer fractions are tried.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    fractions = [0.005, 0.01, 0.02, 0.04, 0.07, 0.10, 0.15]

    while time.time() < deadline and it < max_iter:
        # current convolution & active mask
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-9 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # choose source / target based on gradient percentiles
        high_thr = np.percentile(g, 85)
        low_thr = np.percentile(g, 15)
        high_idx = np.where(g > high_thr)[0]
        low_idx = np.where(g < low_thr)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
        else:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)

        if i == j or best[i] <= 0.0:
            it += 1
            continue

        improved = False
        best_local_score = best_score
        best_delta = 0.0

        for f in fractions:
            delta = best[i] * f
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta

            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_local_score - 1e-12:
                best_local_score = cand_score
                best_delta = delta
                improved = True

        if improved:
            best[i] -= best_delta
            best[j] += best_delta
            best_score = best_local_score

        it += 1

    return best, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random distribution on the simplex.
    """
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "cosine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # cosine bump
        x = np.linspace(0, np.pi, n)
        a = np.cos(x) ** 2 + 0.001
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and renormalise onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def annealing_optimize(
    a0: np.ndarray,
    init_score: float,
    deadline: float,
    fft_len_conv: int,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.
    Randomly transfers a fraction of the mass from a source index to a
    target index.  Moves are accepted if they improve the score or with a
    Metropolis probability that decays with a linear cooling schedule.
    """
    a = a0.copy()
    best = a.copy()
    best_score = init_score
    cur_score = init_score

    start = time.time()
    T0 = max(1e-12, 0.05 * init_score)          # initial temperature

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = max(1e-12, T0 * (1.0 - frac))       # linear cooling

        # pick a source with non‑zero mass
        pos = np.where(a > 1e-12)[0]
        if pos.size == 0:
            break
        i = rng.choice(pos)
        j = rng.integers(0, n)
        if i == j:
            continue

        # fractional transfer of the source mass
        f = rng.uniform(0.001, 0.30)
        delta = a[i] * f
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        # normalise (scale does not matter for the objective)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)
        dE = cand_score - cur_score

        if dE < -1e-12 or rng.random() < np.exp(-dE / T):
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best = cand.copy()

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a previous best if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # Interpolate onto current length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.75 * budget_s
    phase3_end = start_time + 0.90 * budget_s
    phase4_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS continuation with decreasing λ
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.2, 0.1, 0.05,
        0.02, 0.01, 0.005, 0.002,
        0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06,
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Randomly perturb the current best or generate a fresh start
        if rng.random() < 0.7:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[exp‑gradient] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic annealing (random mass transfers)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_optimize(
            best_a,
            best_score,
            phase3_end,
            fft_len_conv,
            n,
            rng,
        )
        print(f"[annealing] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic gradient‑guided hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = targeted_hill_climb_line_search(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase4_end,
            max_iter=250000,
        )
        print(f"[hill‑climb] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final ultra‑low‑λ L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_final, fine_score = lbfgs_continuation(
            best_a,
            [1e-07, 5e-08, 1e-08],
            maxiter_per_lam=800,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if fine_score < best_score:
            best_score = fine_score
            best_a = a_final.copy()
            print(f"[final L‑BFGS] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided external function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
