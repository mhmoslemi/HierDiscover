def adam_optimize(a_start, start_lambda, lam_end, deadline, fft_len_conv, fft_len_g, n):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    while time.time() < deadline:
        # schedule lambda factor decreasing
        lam_factor = max(lam_end, start_lambda * (0.9 ** (t // 100))) etc.
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = 0.1 / (np.sqrt(t) + 1e-8)  # decaying steps
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)
    a = softmax(y)
    return a

"""
propose_candidate - fast, multi‑phase optimizer for the max‑convolution
objective used by the competition.

The algorithm proceeds through several time‑controlled stages:

  0️⃣   Cheap parametric sweep (beta, power‑law, exponential, Gaussian,
      sine‑power, and mixtures with the uniform distribution).  Thousands
      of candidates are evaluated by an O(n log n) FFT based “fast_score”.

  1️⃣   Long L‑BFGS continuation on a smooth (log‑sum‑exp) approximation of
      the max‑convolution.  The smoothing parameter λ is reduced
      progressively (λ = factor·max(conv)) and a large number of iterations
      (≈1500 per λ) are performed.  Random restarts and small Gaussian
      jitter around the current best keep the search diverse.

  2️⃣   Adam (mirror‑descent) on the same smooth objective, starting from
      the best L‑BFGS point.  The learning‑rate decays with √t and λ is
      driven to a very small value (≈5·10⁻⁶) so that the optimisation
      approaches the true max‑convolution.

  3️⃣   Deterministic gradient‑guided hill‑climb.  At each step the most
      harmful coordinate (largest sub‑gradient) transfers a fraction of its
      mass to the most helpful one (smallest sub‑gradient).  Several
      candidate fractions are tested and the best improving move is kept.

All phases respect the global budget (default 1000 s).  The final sequence
is returned as a plain list of floats; it satisfies the non‑negativity
requirement and is normalised to sum = 1 (the scaling cancels out in the
evaluation function).  The best score found during the run is printed.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a probability vector `a`
    (∑a = 1).  O(n log n) via FFT.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Softmax utilities (log‑sum‑exp smooth max of convolution)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max convolution where a = softmax(y).
    Returns the smooth objective and its gradient w.r.t. y.
    """
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ---------- smooth max ----------
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12  # smoothing width
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient wrt convolution entries ----------
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    # back‑propagation through convolution:
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---------- chain rule through softmax ----------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    start_factor: float,
    min_factor: float,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam (mirror‑descent on the simplex through softmax logits) on the
    smoothed max objective.  The smoothing factor lam_factor decays from
    start_factor to min_factor during the run.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    # linear schedule for lam_factor
    total_time = deadline - time.time()
    while time.time() < deadline:
        # progress based λ schedule
        elapsed = time.time() - (deadline - total_time)
        lam_factor = max(min_factor, start_factor * (1.0 - elapsed / total_time))

        f, grad = objective_and_grad(
            y, n, fft_len_conv, fft_len_g, lam_factor
        )
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning rate decays like 1/√t
        lr = 0.12 / (np.sqrt(t) + 1.0)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def gradient_guided_hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 40000,
) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb.  Repeatedly moves a fraction of mass from the
    coordinate with the largest sub‑gradient (most harmful) to the one with
    the smallest sub‑gradient (most helpful).  Several candidate fractions
    are examined and the best improving move is kept.
    """
    best = a.copy()
    best_score = current_score
    it = 0
    # candidate fractions to try (ascending)
    fracs = [0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.45, 0.60]

    while time.time() < deadline and it < max_iter:
        # compute sub‑gradient of the true max‑convolution
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if i_hi == i_lo or best[i_hi] <= 0.0:
            break

        improved = False
        for frac in fracs:
            delta = best[i_hi] * frac
            if delta <= 0.0:
                continue
            cand = best.copy()
            cand[i_hi] -= delta
            cand[i_lo] += delta
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                best = cand
                best_score = cand_score
                improved = True
                break  # accept first (largest) improvement

        if not improved:
            # if none of the fractions improve, stop
            break

        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a random Dirichlet vector with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    x /= x.sum()
    return x


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random distribution on the simplex (mixture of families)."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "beta", "gauss", "sine", "dirichlet"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "beta":
        alpha = rng.uniform(0.2, 2.0)
        i = np.arange(1, n + 1, dtype=float)
        a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    elif mode == "gauss":
        sigma = rng.uniform(0.1, 0.3) * n
        mid = (n - 1) / 2.0
        a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    elif mode == "sine":
        p = rng.choice([0.5, 1.0, 2.0, 4.0])
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    else:  # dirichlet
        alpha = rng.uniform(0.01, 2.0)
        a = random_dirichlet(n, alpha, rng)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Candidate families for the cheap sweep
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def try_update_candidate(a: np.ndarray, best_a_ref, best_score_ref, fft_len, n):
    """Utility used in the parametric sweep."""
    sc = fast_score(a, fft_len, n)
    if sc < best_score_ref[0]:
        best_score_ref[0] = sc
        best_a_ref[0] = a.copy()
        print(f"[sweep] improved fast_score={sc:.6f}")
    return

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (feel free to change; 4000 is a good compromise)
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise – use an existing good sequence if provided
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, hs)
            else:
                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] start fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2     # 0.2 s safety margin

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (including mixtures)
    # --------------------------------------------------------------
    best_a_ref = [best_a]
    best_score_ref = [best_score]

    # power‑law (incl. decreasing), exponential, Gaussian, beta, sine‑power
    for p in np.linspace(-2.0, 5.0, 17):
        try_update_candidate(candidate_power_law(n, p), best_a_ref, best_score_ref,
                             fft_len_conv, n)
        try_update_candidate(candidate_rev_power_law(n, p), best_a_ref,
                             best_score_ref, fft_len_conv, n)

    for lam in np.linspace(0.0005, 0.02, 14):
        try_update_candidate(candidate_exponential(n, lam, False), best_a_ref,
                             best_score_ref, fft_len_conv, n)
        try_update_candidate(candidate_exponential(n, lam, True), best_a_ref,
                             best_score_ref, fft_len_conv, n)

    for sigma_frac in np.linspace(0.12, 0.45, 9):
        sigma = sigma_frac * n
        try_update_candidate(candidate_gaussian(n, sigma), best_a_ref,
                             best_score_ref, fft_len_conv, n)

    for alpha in np.linspace(0.2, 5.0, 17):
        try_update_candidate(candidate_beta_symmetric(n, alpha), best_a_ref,
                             best_score_ref, fft_len_conv, n)

    for p in [0.5, 1.0, 2.0, 4.0, 6.0]:
        try_update_candidate(candidate_sine_power(n, p), best_a_ref,
                             best_score_ref, fft_len_conv, n)

    # Mixtures with the uniform distribution
    uniform = np.full(n, 1.0 / n)
    for alpha in np.linspace(0.2, 5.0, 11):
        beta_seq = candidate_beta_symmetric(n, alpha)
        for w in np.linspace(0.0, 1.0, 9):
            mix = w * beta_seq + (1 - w) * uniform
            mix /= mix.sum()
            try_update_candidate(mix, best_a_ref, best_score_ref,
                                 fft_len_conv, n)

    # Random restarts for a short while (≈10 % of budget)
    while time.time() < start_time + 0.10 * budget_s:
        try_update_candidate(random_initial(n, rng), best_a_ref,
                             best_score_ref, fft_len_conv, n)

    # --------------------------------------------------------------
    # Timings for the remaining phases
    # --------------------------------------------------------------
    phase1_end = start_time + 0.65 * budget_s      # L‑BFGS
    phase2_end = start_time + 0.90 * budget_s      # Adam
    # remaining time → deterministic hill‑climb

    best_a = best_a_ref[0]
    best_score = best_score_ref[0]

    # --------------------------------------------------------------
    # Phase 1 – Random‑restart L‑BFGS continuation
    # --------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.06,
        0.04,
        0.025,
        0.015,
        0.008,
        0.004,
        0.002,
        0.001,
        0.0005,
        0.0002,
        0.0001,
        0.00005,
        0.00002,
    ]
    maxiter_per_lam = 1500   # more thorough than the baseline

    while time.time() < phase1_end:
        # choose a start point (jittered version of the current best
        # with prob 0.25, otherwise a fresh random draw)
        if rng.random() < 0.25:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.008), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the smooth max‑convolution
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a,
            start_factor=0.5,
            min_factor=5e-6,
            deadline=phase2_end,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        print(f"[Adam] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Deterministic gradient‑guided hill‑climb
    # --------------------------------------------------------------
    best_a, best_score = gradient_guided_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_iter=60000,
    )
    print(f"[final hill‑climb] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
