"""
Propose a step sequence that minimizes the evaluation function.

Algorithm
---------
1. **Problem size** – work with a moderate length (n≈4000) which keeps FFTs
   cheap while still giving enough degrees of freedom.

2. **Fast utilities** – convolution via FFT, a fast “max‑score” that assumes
   ∑a = 1, Euclidean projection onto the simplex and a numerically stable
   soft‑max.

3. **Large parametric sweep** – generate many candidate shapes:
   * power‑law, reversed power‑law,
   * exponential / reversed,
   * Gaussian,
   * symmetric & asymmetric Beta (i.e. i^{α‑1}(n‑i)^{β‑1}),
   * sine‑power, triangular,
   * standard signal‑processing windows (Hann, Hamming, Blackman,
     Kaiser with several β‑values).
   The best candidate from this sweep seeds the next phases.

4. **Smooth optimisation** – minimise a log‑sum‑exp smoothing of the
   max‑convolution using L‑BFGS on the soft‑max (log‑it) parametrisation.
   A decreasing schedule of the smoothing factor (λ) refines the solution.

5. **Projected sub‑gradient descent** – optimisation directly on the true
   max‑convolution with momentum.  The iterate is always projected onto the
   probability simplex.

6. **Exponentiated‑gradient descent** – a mirror‑descent step that respects
   the simplex and often escapes shallow plateaus.

7. **Simulated‑annealing hill‑climb** – randomly move a small fraction of
   mass between two coordinates; accept improving moves and, with a
   temperature that cools linearly, occasionally accept worsening moves.
   This step can find improvements that deterministic moves miss.

8. **Final evaluation** – the best sequence found (still on the simplex)
   is handed to the supplied `evaluate_sequence` and returned.

All phases respect a global 1000 s budget; the program prints progress
messages and the current best “fast_score” (2·n·max(conv) for ∑a=1).  The
returned list consists of non‑negative floats whose sum is 1, guaranteeing
that the final evaluation equals the printed fast_score.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy import signal

# ----------------------------------------------------------------------
# Basic FFT‑based utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv) for a sequence that already satisfies
    ∑a = 1.  O(n log n) by FFT.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by sub‑gradient / EG steps
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Soft‑max parametrisation utilities (logits → probability simplex)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a * Δ = conv; Δ = w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best `a` found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.5f} improved fast_score={best_score:.6f}")
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # decreasing step size ~ 1/√t
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            # print occasional improvement
            if it % 500 == 0:
                print(f"[sub‑grad] it={it} improved fast_score={best_score:.6f}")
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(a0: np.ndarray,
                          deadline: float,
                          fft_len_conv: int,
                          fft_len_g: int,
                          n: int):
    """
    Exponentiated‑gradient descent (mirror descent) on the simplex using the true
    sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[exp‑grad] it={it} improved fast_score={best_score:.6f}")
        it += 1
    return best_a, best_score


def simulated_annealing_hill_climb(a0: np.ndarray,
                                   start_score: float,
                                   fft_len: int,
                                   n: int,
                                   deadline: float,
                                   max_iter: int = 40000):
    """
    Simple simulated‑annealing style hill‑climb.
    Randomly move a fraction of mass from one coordinate to another.
    Occasionally accept a worsening move depending on a linearly cooling
    temperature.
    """
    best = a0.copy()
    best_score = start_score
    cur = a0.copy()
    cur_score = start_score
    rng = np.random.default_rng()
    it = 0

    # initial temperature – a small fraction of the current score
    T0 = max(1e-4, (best_score - 1.0) * 0.05)

    while time.time() < deadline and it < max_iter:
        # fraction and pair selection
        frac = rng.random() * 0.25          # up to 25 % of the donor mass
        i_hi = rng.integers(n)              # donor
        i_lo = rng.integers(n)              # receiver
        if i_hi == i_lo or cur[i_hi] <= 1e-14:
            it += 1
            continue
        delta = cur[i_hi] * frac
        new = cur.copy()
        new[i_hi] -= delta
        new[i_lo] += delta

        new_score = fast_score(new, fft_len, n)

        # acceptance rule
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # linear cooling
            progress = it / max_iter
            T = T0 * (1.0 - progress)
            if T > 0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
                print(f"[SA] it={it} improved fast_score={best_score:.6f}")

        it += 1
    return best, best_score

# Candidate families (params → probability simplex)
# ----------------------------------------------------------------------


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_beta(n: int, alpha: float, beta_: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta_ - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


def candidate_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape (rise then fall)."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(float)
    a /= a.sum()
    return a


def candidate_window(name: str, n: int, **kw) -> np.ndarray:
    """Wrap common windows from scipy.signal."""
    w = getattr(signal.windows, name)(n, **kw)
    w = np.maximum(w, 0.0)
    w /= w.sum()
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (a bit larger than the previous run to give more freedom)
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial point – uniform simplex
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin for final eval

    # ------------------------------------------------------------------
    # Phase 0 – extensive parametric sweep
    # ------------------------------------------------------------------
    def try_update(a: np.ndarray, tag: str = ""):
        nonlocal best_a, best_score
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sweep]{tag} improved fast_score={best_score:.6f}")

    # uniform (already done)
    try_update(best_a, " (uniform)")

    # Power‑law families
    for p in np.linspace(-2.0, 5.0, 20):
        try_update(candidate_power_law(n, p), f" power p={p:.2f}")
        try_update(candidate_rev_power_law(n, p), f" rev_power p={p:.2f}")

    # Exponential families
    for lam in np.linspace(0.0005, 0.02, 15):
        try_update(candidate_exponential(n, lam, reverse=False), f" exp lam={lam:.4f}")
        try_update(candidate_exponential(n, lam, reverse=True), f" rev_exp lam={lam:.4f}")

    # Gaussian windows
    for sigma_frac in np.linspace(0.12, 0.45, 10):
        sigma = sigma_frac * n
        try_update(candidate_gaussian(n, sigma), f" gauss σ={sigma_frac:.2f}")

    # Symmetric beta‑like shapes
    for alpha in np.linspace(0.2, 5.0, 15):
        try_update(candidate_beta_symmetric(n, alpha), f" beta_sym α={alpha:.2f}")

    # Asymmetric beta families (grid)
    alphas = np.linspace(0.2, 5.0, 8)
    betas = np.linspace(0.2, 5.0, 8)
    for a_ in alphas:
        for b_ in betas:
            try_update(candidate_beta(n, a_, b_), f" beta α={a_:.2f} β={b_:.2f}")

    # Sine‑power windows
    for p in [0.5, 1.0, 2.0, 4.0, 6.0, 8.0]:
        try_update(candidate_sine_power(n, p), f" sine p={p}")

    # Triangular shape
    try_update(candidate_triangle(n), " triangle")

    # Standard windows from scipy.signal
    for win in ["hann", "hamming", "blackman"]:
        try_update(candidate_window(win, n), f" win {win}")

    # Kaiser windows with a variety of β
    for beta_val in [2, 4, 6, 8, 12, 14]:
        try_update(candidate_window("kaiser", n, beta=beta_val), f" kaiser β={beta_val}")

    # Quick random initialisations (short burst)
    while time.time() < start_time + 0.07 * budget_s:
        try_update(random_initial(n, rng), " rand")

    # ------------------------------------------------------------------
    # Allocate time for the remaining phases
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS (or Adam)
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.85 * budget_s   # exponentiated‑gradient
    # remaining time → simulated‑annealing hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation on a smoothed max
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # occasional jitter around the best point
        if rng.random() < 0.25:
            a_start = best_a * (0.9 + 0.2 * rng.random(n))
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (global exploration)
    # ------------------------------------------------------------------
    best_a, best_score = simulated_annealing_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_iter=80000,
    )
    print(f"[final SA] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
