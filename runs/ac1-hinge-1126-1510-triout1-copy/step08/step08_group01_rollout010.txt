def alternating_lp_optimize(a_start, deadline, n, fft_len, rng, t=0.3, max_iter=10):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it = 0
    while time.time() < deadline and it < max_iter:
        # compute conv and max
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
        M = conv.max()
        # solve linear program
        A = build_conv_matrix(a)  # sparse
        b = np.full(2*n-1, M)
        c = -np.ones(n)   # maximize sum, minimize -sum
        res = linprog(c=c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            # fallback: break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g  # normalize to sum=1
        # mix
        a = (1 - t) * a + t * g
        # normalize again
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score
```

"""Step‑function optimizer with alternating LP refinement.

The search proceeds in four stages:
1. Light L‑BFGS on a smooth max‑convolution surrogate (several random restarts)
   to obtain a good initial distribution.
2. Alternating linear‑programming refinement: for the current distribution f,
   solve   max Σ g_j   s.t.   f * g ≤ M   (M = max(f*f)), g ≥ 0.
   The resulting g (renormalised) is mixed with f, reducing the peak of the
   autocorrelation.
3. Exponentiated‑gradient descent on the true L∞ objective for further
   improvement.
4. Fast greedy hill‑climb with incremental convolution updates for fine‑grained
   optimisation.

The whole routine respects the 1000 s time budget and returns the best
non‑negative sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(a*a) for a probability vector a."""
    conv = full_conv(a, fft_len)
    return 2.0 * a.shape[0] * conv.max()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# 1. L‑BFGS on a smoothed approximation of the max‑convolution
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                       lam_factor: float = 0.30):
    """Smoothed objective based on log‑sum‑exp of the convolution."""
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# 2. Alternating LP refinement (f * g ≤ M)
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray):
    """Build the (2n‑1)×n convolution matrix B where (B@g)=f*g."""
    n = f.shape[0]
    data = []
    offsets = []
    # only non‑negative offsets produce non‑zero entries
    for d in range(n):
        length = 2 * n - 1 - d        # number of rows for this diagonal
        data.append(np.full(length, f[d]))
        offsets.append(d)
    return diags(data, offsets, shape=(2 * n - 1, n), format='csc')


def alternating_lp_optimize(a_start: np.ndarray,
                            deadline: float,
                            n: int,
                            fft_len: int,
                            rng: np.random.Generator,
                            t: float = 0.20,
                            max_iter: int = 10):
    """
    Repeatedly:
        • compute M = max(a*a)
        • solve LP: maximise Σ g_j   s.t.   a*g ≤ M , g ≥ 0
        • renormalise g to sum 1 and mix a ← (1‑t)a + t g
    Returns the best distribution encountered.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iter:
        # current bound
        conv = full_conv(a, fft_len)
        M = conv.max()

        # build the linear constraints a*g ≤ M
        A = build_conv_matrix(a)                     # sparse (2n‑1)×n
        b = np.full(2 * n - 1, M)

        # linprog solves   min c·x   subject to A_ub·x ≤ b
        # we maximise Σ g_j  ⇔ minimise -Σ g_j
        c = -np.ones(n)

        res = linprog(c=c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        if not res.success:
            # LP failed – give up the LP phase
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g                     # renormalise to sum‑1

        # mix and re‑normalise
        a = (1.0 - t) * a + t * g
        a = a / a.sum()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # shrink step size a little when we make progress
            t = max(0.05, t * 0.95)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# 3. Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the L∞‑objective.
    Uses step size η = η₀ / √t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# 4. Fast greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.12,
                           recompute_every: int = 1500):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    index to a destination index.  Convolution is updated incrementally
    (O(n) per move) and periodically recomputed exactly.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    diff = np.zeros_like(conv)
    it = 0
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # move from larger to smaller
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        it += 1
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.

    """
    rng = np.random.default_rng(seed)

            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ----- time budgeting ------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.20 * budget_s       # L‑BFGS
    phase2_end = start_time + 0.60 * budget_s       # LP refinement
    phase3_end = start_time + 0.85 * budget_s       # exponentiated‑gradient
    # remaining time → hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – lightweight L‑BFGS with a few random restarts
    # ------------------------------------------------------------------
    lam_schedule = [

    # ------------------------------------------------------------------
    # Phase 2 – alternating linear program refinement
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = alternating_lp_optimize(

    # ------------------------------------------------------------------
    # Phase 3 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=10.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – intensive greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:

            max_fraction=0.12,

    # ----- final evaluation ---------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
