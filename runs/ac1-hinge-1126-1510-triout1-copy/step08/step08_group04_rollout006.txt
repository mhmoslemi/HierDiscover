"""
Optimiser for the step‑function evaluation problem.

The algorithm proceeds in three stages:

1.  **Projected gradient descent with back‑tracking line search**.
    Starting from many random initialisations (including several
    handcrafted shapes) we repeatedly minimise the quantity
    `max(a∗a)` while keeping `a` on the probability simplex.
    The sub‑gradient of the max‑convolution is obtained by FFT
    and a simple back‑tracking ensures a strict decrease of the
    objective at every accepted step.

2.  **Simulated annealing** – a stochastic polishing phase that
    makes occasional large Gaussian perturbations and local
    mass‑transfer moves, accepting uphill moves according to a
    linearly cooling schedule.

3.  **Greedy hill‑climbing** – a final deterministic sweep that
    moves a small fraction of mass between random coordinates
    only if the move strictly improves the score.

All stages respect the non‑negativity and normalisation constraints.
The routine is time‑bounded (default 1000 s) and returns the best
sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) when Σa = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Random / heuristic initialisation utilities
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        [
            'uniform', 'random', 'exp', 'power', 'edge',
            'bimodal', 'sparse', 'logit_normal', 'beta'
        ]
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'logit_normal':
        logits = rng.normal(0.0, 1.0, size=n)
        a = np.exp(logits - np.max(logits))
    else:   # beta distribution with random shape parameters
        p = rng.uniform(0.3, 0.9)
        q = p   # symmetric beta
        x = (np.arange(n) + 0.5) / n
        a = (x ** (p - 1)) * ((1 - x) ** (q - 1))
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Gradient of the max‑convolution
# ----------------------------------------------------------------------


def compute_grad_max(a: np.ndarray, fft_len_g: int):
    """
    Compute max of a∗a and a sub‑gradient of the max‑convolution.
    Returns (max_val, gradient, full_convolution).
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len_g)[: 2 * n - 1]
    max_val = conv.max()
    # tolerance for identifying active constraints
    eps = max_val * 1e-12 + 1e-15
    mask = np.zeros_like(conv)
    mask[conv >= max_val - eps] = 1.0

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    grad_full = np.real(np.fft.ifft(fm * fa_rev))
    grad = 2.0 * grad_full[(n - 1):(n - 1 + n)]
    return max_val, grad, conv


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def gradient_descent_iter(
    a_start: np.ndarray,
    max_iters: int,
    deadline: float,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    rng: np.random.Generator,
):
    """
    Projected gradient descent on max(a*a) with back‑tracking line search.
    Returns the best vector found during the allotted iterations.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)

    for _ in range(max_iters):
        if time.time() >= deadline:
            break

        _, grad, _ = compute_grad_max(a, fft_len_g)

        step = 1.0
        improved = False
        # back‑track until a strict improvement is observed
        while step > 1e-6:
            a_new = simplex_projection(a - step * grad)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-12:
                improved = True
                break
            step *= 0.5

        if improved:
            a = a_new
            best = a.copy()
            best_score = new_score
        else:
            # occasional stochastic kick to escape plateaus
            sigma = rng.uniform(0.0005, 0.01)
            a = add_noise(a, sigma, rng)

    return best, best_score


def simulated_annealing(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    max_local_transfer: float = 0.5,
):
    """
    Simple simulated‑annealing on the simplex.
    Proposes two kinds of moves:
        * a mass transfer between two coordinates,
        * a global Gaussian perturbation (add_noise).
    Accepts moves according to a linearly decreasing temperature.
    """
    a_cur = a_start.copy()
    cur_score = fast_score(a_cur, fft_len, n)
    best_a = a_cur.copy()
    best_score = cur_score

    start = time.time()
    init_temp = cur_score * 0.02 + 1e-9

    while time.time() < deadline:
        frac = (time.time() - start) / (deadline - start)
        temp = init_temp * (1.0 - frac)

        a_new = a_cur.copy()
        if rng.random() < 0.6:
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * max_local_transfer
                a_new[i] -= delta
                a_new[j] += delta
        else:
            sigma = rng.uniform(0.0005, 0.015)
            a_new = add_noise(a_new, sigma, rng)

        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len, n)
        delta = new_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20000,
):
    """
    Greedy hill‑climb: move a modest random fraction of mass from
    one coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1

    return best, best_score
