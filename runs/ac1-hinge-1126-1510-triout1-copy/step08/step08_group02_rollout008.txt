def simulated_annealing(a, start_score, deadline, n, fft_len):
    a_current = a.copy()
    score_current = start_score
    best = a_current.copy()
    best_score = start_score
    rng = np.random.default_rng()
    T0 = 0.01 * start_score  # initial temperature relative to score.
    # schedule: T = T0 * exp(-decay * t)
    t = 0
    while time.time() < deadline:
        t += 1
        # schedule
        T = T0 * np.exp(-t / 50000) # adapt
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose delta
        delta = rng.uniform(0, min(0.2 * a_current[i], 0.2 * a_current[i]))
        if delta <= 0:
            continue
        a_new = a_current.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure non-negative (i side may become negative)
        if a_new[i] < 0:
            continue
        # Normalize sum = 1 automatically (mass moved)
        score_new = fast_score(a_new, fft_len, n)
        if score_new < score_current:
            a_current = a_new
            score_current = score_new
            if score_new < best_score:
                best_score = score_new
                best = a_new.copy()
        else:
            # accept with temperature
            if rng.random() < np.exp(-(score_new - score_current)/T):
                a_current = a_new
                score_current = score_new
    return best, best_score
```

def fast_score(a, fft_len):
    # assumes sum(a)=1
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*len(a)-1]
    return float(2.0 * len(a) * conv.max())

def subgradient_descent(a, deadline):
    n = len(a)
    fft_len = next_pow2(2*n - 1)
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it = 0
    step0 = 0.3
    while time.time() < deadline:
        # compute conv and mask (active constraints)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2*n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)

        score = fast_score(a, fft_len)
        if score < best_score:

            best_a = a.copy()
        it += 1
    return best_a, best_score

def hill_climb_refine(a, current_score, deadline):

    best = a.copy()
    best_score = current_score

        else:

"""
Propose a low‑score step function.
---------------------------------
The search merges three ideas that complement each other:

1.  A few L‑BFGS refinements of a smoothed log‑sum‑exp version of the
    max‑convolution (large vector, n≈4000).  This quickly reaches the
    state of the previous best (~1.5065).

2.  An LP‑based mixing scheme on *smaller* vectors (n≈1500‑2200).
    For a given f we solve
        max   Σ g_j
        s.t.  f * g ≤ max(f * f)      (convolution ≤ current peak)
              g ≥ 0
    The solution pushes mass into positions where f is small.
    Mixing f←(1‑t)f + t·g and re‑normalising steadily reduces the
    maximal autocorrelation.  The LP is solved with `scipy.optimize.linprog`
    (HiGHS) on a sparse Toeplitz constraint matrix.

3.  A fast greedy hill‑climb that moves a tiny random fraction of the
    mass from one coordinate to another, accepting only strictly
    improving moves.

All phases respect a 1000 s deadline and the best sequence found
(whatever its length) is returned.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) for a already normalised to Σa=1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len: int,
                       lam_factor: float):
    """
    Smoothed max of the self‑convolution (log‑sum‑exp) and its gradient
    w.r.t. logits y (where a = softmax(y)).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # back‑propagate through the convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


def subgradient_descent(a_start: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Keeps the sum = 1 via simplex projection.
    """
    a = a_start.copy()
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    it = 0
    step0 = 0.3
    while time.time() < deadline:
        # self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the hard max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # direction that reduces the max while preserving the sum
        mean_g = np.mean(g)
        d = -(g - mean_g)                     # Σd = 0 by construction

        # diminishing step size and projection onto the simplex
        alpha = step0 / np.sqrt(it + 1.0)
        a = simplex_projection(a + alpha * d)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def build_constraint_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Return the (2·n‑1)×n sparse matrix C with
        C[i, j] = f[i‑j]   (0 ≤ i‑j < n, else 0).
    This encodes the linear mapping g ↦ f * g.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    diag_data = []
    offsets = []
    for d in range(n):
        length = rows - d
        diag_data.append(np.full(length, f[d], dtype=np.float64))
        offsets.append(d)
    C = sp.diags(diag_data, offsets, shape=(rows, n), format='csr')
    return C


def lp_mixing_optimize(n: int,
                       time_budget: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    LP‑based mixing for a given size n.
    Repeatedly:
        * solve max Σg  s.t.  (f * g) ≤ max(f * f), g ≥ 0
        * mix f ← (1‑t) f + t g   and renormalise
    Returns the best vector (Σ=1) seen and its score 2·n·max(conv).
    """
    start = time.time()
    deadline = start + time_budget
    # initial distribution – Dirichlet‑like random point
    a = rng.random(n)
    a /= a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * np.convolve(a, a).max()

    it = 0
    step0 = 0.5
    max_iter = 200                     # safety‑cap

    while time.time() < deadline and it < max_iter:
        # current peak of the self‑convolution
        M = np.convolve(a, a).max()

        # build the Toeplitz constraint matrix for the current a
        C = build_constraint_matrix(a)

        # linear program: maximise Σg  ⇔ minimise −Σg
        c = -np.ones(n, dtype=np.float64)
        ub = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # HiGHS respects a time limit; we pass the remaining time
        opts = {"presolve": True,
                "time_limit": max(0.0, deadline - time.time())}
        res = opt.linprog(c,
                          A_ub=C,
                          b_ub=ub,
                          bounds=bounds,
                          method='highs',
                          options=opts)

        if not res.success:
            # If the LP fails, abort this size early.
            break

        g = res.x
        # mixing step (t shrinks with the iteration count)
        t = step0 / np.sqrt(it + 1.0)
        a = (1.0 - t) * a + t * g
        total = a.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= total

        score = 2.0 * n * np.convolve(a, a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def hill_climb_refine(a: np.ndarray,
                      current_score: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a small random fraction of mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 10 % of the source mass
        delta = best[i] * rng.random() * 0.10
        if delta == 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
