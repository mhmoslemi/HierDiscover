def exp_gradient_optimize(a_start, sub_deadline, n, fft_len, rng):

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < sub_deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]
        # exponentiated gradient step
        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        sum_a = a.sum()
        if sum_a <= 0.0:
            a = best_a.copy()
        else:
            a = a / sum_a
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
        # occasional re-normalization? Already done.
        # Optionally add small noise every 1000 iterations to escape.
        if it % 2000 == 0:
            # add small Gaussian noise
            sigma = 0.001 * (1.0 - it / 1e6)  # decreasing?
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum
    return best_a, best_score

def random_local_search(a, best_score, fft_len, n, deadline, rng, max_trials=100000):

    while time.time()<deadline:
        # choose i, j where a[i] > 0
        i = rng.integers(0, n)
        if a[i]==0:
            continue
        j = rng.integers(0, n)
        if i==j:
            continue
        delta_frac = rng.uniform(0.001, 0.01) # fraction of a[i]
        delta = a[i] * delta_frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

"""
Hybrid search for a low‑evaluation step‑function.

The algorithm works in three stages (within the 1000 s budget):

1. **Heavy L‑BFGS** – repeated random restarts on a log‑sum‑exp
   smoothing of the max‑convolution.  The smoothing parameter λ is
   driven down to 10⁻⁶ to approach the true max.
2. **Long exponentiated‑gradient descent** – exact max‑convolution
   gradient with a decreasing stepsize η = 1/√t, keeping the vector
   on the probability simplex automatically.
3. **Refined deterministic hill‑climb** – repeatedly transfers a
   small fraction of mass from the coordinate that contributes most
   to the current maximal convolution entry to a set of low‑partner
   coordinates (trying several destinations per iteration).  The
   step size is halved when no improvement is found.

All stages keep the vector normalised (∑a = 1).  The best sequence
found is finally evaluated with the supplied ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) when a is normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse families of random points on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Initialise via a square‑rooted uniform spectrum (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    uniform = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    padded = np.zeros(L)
    padded[: 2 * n - 1] = uniform
    S = np.fft.fft(padded)
    sqrt_amp = np.sqrt(np.abs(S))
    sqrt_phase = np.angle(S) / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Gaussian perturbation followed by simplex projection."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max‑convolution (log‑sum‑exp) and gradient w.r.t. logits y.
    a = softmax(y) ensures the simplex constraint.
    """
    a = softmax(y)

    # -------- convolution a*a (full) ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # -------- gradient of smooth max ----------
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # gradient w.r.t. a (size n) – extracts the relevant slice
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in ``lam_schedule``, starting each stage from
    the result of the previous stage.  Returns the best a and its score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # rarely happens – just skip this λ
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} new best score={best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # ---- exact max‑convolution ----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ---- gradient of (hard) max ----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # ---- stepsize schedule ----
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # occasional aggressive perturbation after a good improvement
            if it % 500 == 0:
                a = add_noise(a, sigma=0.002 * (deadline - time.time()) / (deadline - start_time), rng)

        it += 1
        # stop early if improvements stall for a long time
        if it % 5000 == 0 and time.time() > deadline - 30:
            # In the last seconds we just keep the current best
            break
    return best_a, best_score


def refined_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.10,
                      min_step: float = 1e-5,
                      tries: int = 7) -> tuple[np.ndarray, float]:
    """
    Deterministic improvement: move a fraction of the mass from the
    coordinate that contributes most to the current maximal convolution
    entry to a few low‑partner coordinates (trying several destinations
    per iteration).  The step size is shrunk when no improvement is
    found.
    """
    step = init_step
    while time.time() < deadline:
        # ----- convolution and critical index -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        max_idx = np.argmax(conv)

        # ----- partner values for the chosen shift -----
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # ----- contributions to the critical entry -----
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # try a few destinations with the smallest partner value
        dest_candidates = np.argsort(partner)[:tries]
        improved = False

        for i_min in dest_candidates:
            if i_min == i_max:
                continue
            delta = a[i_max] * step
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i_max] -= delta
            a_new[i_min] += delta
            # renormalise (numerical safety)
            s = a_new.sum()
            if s <= 0.0:
                continue
            a_new /= s
            new_score = fast_score(a_new, fft_len, n)
            if new_score < score:
                a = a_new
                score = new_score
                improved = True
                break   # accept the first improving move

        if not improved:
            step = max(step * 0.7, min_step)   # shrink step gracefully

    return a, score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a modest increase to give a finer discretisation
    # --------------------------------------------------------------
    n = 8000                     # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise candidates (interpolate previous solution if present)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 1:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # other diverse starts
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)

    candidates = [a_start, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    phase1_deadline = start_time + 0.40 * budget_s          # L‑BFGS
    phase2_deadline = start_time + 0.85 * budget_s          # EG descent
    phase3_deadline = start_time + 0.97 * budget_s          # refined hill‑climb
    # -----------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS with many restarts
    # -----------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_deadline:
        # choose a diverse start for each L‑BFGS run
        r = rng.random()
        if r < 0.45:
            a0 = random_initial(n, rng)
        elif r < 0.9:
            sigma = rng.uniform(0.001, 0.010)
            a0 = add_noise(best_a, sigma, rng)
        else:
            a0 = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a0,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # -----------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[exp‑grad] best score after EG={best_score:.6f}")

    # -----------------------------------------------------------------
    # Phase 3 – refined deterministic hill‑climb
    # -----------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = refined_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_deadline,
            init_step=0.12,
            min_step=1e-5,
            tries=9,
        )
        print(f"[refined hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
```

This implementation builds upon the earlier hybrid optimiser, enlarges the discretisation (n = 8000), runs a more aggressive L‑BFGS phase with a finer λ‑schedule, follows it with a long exponentiated‑gradient descent, and finishes with a refined hill‑climb that tests several destination indices per iteration. The search respects the 1000 s budget and returns the best sequence it discovers.
