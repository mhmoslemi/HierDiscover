def projected_gradient_descent(a_start, deadline, n, fft_len, rng, step0: float = 0.1):

    a = a_start.copy()

    best_a = a.copy()

    best_score = fast_score(best_a, fft_len)

    it = 0

    while time.time() < deadline:

        # compute conv and mask

        fa = np.fft.fft(a, fft_len)

        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12

        mask = (conv >= max_val - eps).astype(np.float64)

        # compute gradient

        a_rev = a[::-1]

        fm = np.fft.fft(mask, fft_len)

        fa_rev = np.fft.fft(a_rev, fft_len)

        grad_conv = np.real(np.fft.ifft(fm * fa_rev))

        g = 2.0 * grad_conv[n-1: n-1 + n]

        # step size

        step = step0 / np.sqrt(it + 1)

        # update

        a_temp = a - step * g

        # ensure nonneg and project onto simplex

        a_temp = np.maximum(a_temp, 0.0)

        a_temp = simplex_projection(a_temp)

        a = a_temp

        score = fast_score(a, fft_len)

        if score < best_score:

            best_score = score

            best_a = a.copy()

        it += 1

    return best_a, best_score

def hill_climb_simulated_annealing(a, curr_score, fft_len, n, deadline, rng, max_fraction=0.12, temp0=0.005, cool_factor=0.99997, recompute_every=2000):

    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    diff = np.zeros_like(conv)
    accepted = 0
    T = temp0

    while time.time() < deadline:
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0:
            continue

        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        acceptance = False
        if new_score < best_score:
            acceptance = True
        else:
            # temperature may be zero at later stage; if very small, acceptance rarely.
            prob = np.exp((best_score - new_score) / (T + 1e-12))
            if rng.random() < prob:
                acceptance = True

        if acceptance:
            best_a[src] -= delta

        T *= cool_factor

    return best_a, best_score

    # Uniform

    # Triangular peak in middle
    idx = np.arange(n)
    tri = np.minimum(idx + 1, n - idx)
    candidates.append(tri.astype(float))
    # Quadratic concave shape
    quad = (idx + 1) * (n - idx)
    candidates.append(quad.astype(float))
    # Cosine shape (full period)
    cos = 1 - np.cos(2 * np.pi * idx / (n - 1))
    candidates.append(cos)
    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
    candidates.append(hann)
    # Sine window
    sine = np.sin(np.pi * (idx + 0.5) / n)
    candidates.append(sine)
    # Exponential decays
    for rate in np.linspace(0.001, 0.02, 10):
        exp = np.exp(-rate * idx)
        candidates.append(exp)
    # Power law
    for p in np.linspace(0.2, 2.5, 10):
        pow_arr = (idx + 1) ** (-p)
        candidates.append(pow_arr)
    # Bimodal: two Gaussian-like peaks
    for center in [int(n*0.25), int(n*0.75)]:
        width = int(0.05 * n)
        x = np.arange(n)
        peak = np.exp(- ((x - center) ** 2) / (2 * width ** 2))
        candidates.append(peak)
    # Edge heavy (like random_initial 'edge')
    edge_frac = rng.uniform(0.006, 0.03)
    m_edge = max(1, int(n * edge_frac))
    edge = np.full(n, 1.0 / n)
    heavy = np.sqrt(n / 2.0) / n
    edge[:m_edge] = heavy
    edge[-m_edge:] = heavy
    candidates.append(edge)
    # Sparse random
    m = rng.integers(int(0.02 * n), int(0.07 * n))
    sparse = np.zeros(n)
    idxs = rng.choice(n, size=m, replace=False)
    vals = rng.random(m) + 1e-6
    sparse[idxs] = vals
    candidates.append(sparse)

    # Normalize each candidate
    normalized = []
    for a in candidates:

        if a.sum() == 0:

        a = a / a.sum()
        normalized.append(a)

"""
Propose a step‐function (non‑negative sequence) with a low evaluation score.

Algorithm Overview
------------------
1. **Initialisation**
   * Use the previously found sequence if available, otherwise start from a
     uniform distribution (sum = 1).
   * Scan a small library of analytically‑motivated shapes (triangular,
     exponential, power‑law, Hann, etc.) and keep the best one.

2. **Phase 1 – L‑BFGS on a smoothed max‑convolution**
   * Optimize a log‑sum‑exp surrogate of `max(conv)` in the log‑softmax
     (log‑probability) space.  Several random restarts are performed.
   * The best result is kept for the next phase.

3. **Phase 2 – Exponentiated‑gradient (mirror) descent**
   * Perform a long EG descent directly on the true `max(conv)` objective.
   * Step size ηₜ = η₀/√t with η₀ tuned automatically; the iterate is kept
     on the simplex using an exponential update.

4. **Phase 3 – Simulated‑annealing greedy hill‑climb**
   * Move a random fraction of mass from a source index to a destination
     index, updating the convolution incrementally (O(n) per move).
   * Accept improving moves always, and accept worsening moves with a
     temperature‑controlled probability; temperature decays slowly.
   * Periodic exact recomputation avoids drift.

5. **Final polish**
   * If time permits, run a short L‑BFGS refinement.

The function respects the 1000 s budget, prints progress, and returns the
best sequence found (as a plain Python list).  The official scorer
`evaluate_sequence` is called only once at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a probability vector a (sum = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smooth surrogate and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax → stay on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full) via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the L∞‑objective.
    Step size η = η₀ / √t .
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing greedy hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def hill_climb_simulated_annealing(a: np.ndarray,
                                   curr_score: float,
                                   fft_len: int,
                                   n: int,
                                   deadline: float,
                                   rng: np.random.Generator,
                                   max_fraction: float = 0.12,
                                   temp0: float = 0.005,
                                   cool_factor: float = 0.99997,
                                   recompute_every: int = 2000):
    """
    Greedy hill‑climb that moves a random fraction of mass from a source
    to a destination index.  Accepted moves may increase the score with a
    probability depending on a slowly decreasing temperature.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)                       # exact full conv
    diff = np.zeros_like(conv)                               # reusable buffer
    accepted = 0
    T = temp0

    while time.time() < deadline:
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if best_a[src] == 0.0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        accept = False
        if new_score < best_score:
            accept = True
        else:
            prob = np.exp((best_score - new_score) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        T *= cool_factor

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Analytic families for quick seeding
# ----------------------------------------------------------------------


def generate_candidate_families(n: int, rng: np.random.Generator):
    """
    Produce a list of analytically‑motivated distributions (not normalised).
    They are later normalised to sum() == 1.
    """
    candidates = []

    idx = np.arange(n, dtype=float)

    # Uniform
    candidates.append(np.full(n, 1.0))

    # Simple triangular (peak in the centre)
    tri = np.minimum(idx + 1, n - idx)
    candidates.append(tri)

    # Quadratic concave shape
    quad = (idx + 1) * (n - idx)
    candidates.append(quad)

    # Cosine (full period)
    cos = 1.0 - np.cos(2.0 * np.pi * idx / (n - 1))
    candidates.append(cos)

    # Hann window
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * idx / (n - 1)))
    candidates.append(hann)

    # Sine window
    sine = np.sin(np.pi * (idx + 0.5) / n)
    candidates.append(sine)

    # Exponential decays
    for rate in np.linspace(0.001, 0.02, 10):
        candidates.append(np.exp(-rate * idx))

    # Power‑law decays
    for p in np.linspace(0.2, 2.5, 10):
        candidates.append((idx + 1) ** (-p))

    # Two Gaussian‑like bumps (bimodal)
    for centre in [int(n * 0.25), int(n * 0.75)]:
        width = max(1, int(0.05 * n))
        bump = np.exp(-((idx - centre) ** 2) / (2.0 * width * width))
        candidates.append(bump)

    # Edge heavy (similar to the 'edge' mode of random_initial)
    edge_frac = rng.uniform(0.006, 0.03)
    m_edge = max(1, int(n * edge_frac))
    edge = np.full(n, 1.0)
    heavy = np.sqrt(n / 2.0)
    edge[:m_edge] = heavy
    edge[-m_edge:] = heavy
    candidates.append(edge)

    # Very sparse random
    m = rng.integers(int(0.02 * n), int(0.07 * n))
    sparse = np.zeros(n)
    ids = rng.choice(n, size=m, replace=False)
    sparse[ids] = rng.random(m) + 1e-6
    candidates.append(sparse)

    # Normalise all candidates to sum == 1 and filter degenerate cases
    normalised = []
    for a in candidates:
        a = np.asarray(a, dtype=float)
        s = a.sum()
        if s <= 0.0:
            continue
        a = a / s
        normalised.append(a)
    return normalised


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimension – 4000 points offers a good trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from the global best if available, otherwise uniform.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05    # safety margin

    phase0_end = start_time + 0.07 * budget_s   # analytic families
    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.60 * budget_s   # EG descent
    phase3_end = start_time + 0.90 * budget_s   # hill‑climb (annealing)

    # ------------------------------------------------------------------
    # Phase 0 – quick scan of analytic families
    # ------------------------------------------------------------------
    families = generate_candidate_families(n, rng)
    for cand in families:
        # evaluate only if we still have time
        if time.time() >= phase0_end:
            break
        score = fast_score(cand, fft_len)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
            print(f"[family] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with multiple random restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 800
    restarts = 0

    while time.time() < phase1_end:
        # Occasionally perturb the current best, otherwise fresh start
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        restarts += 1
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS #{restarts}] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=12.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = hill_climb_simulated_annealing(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            max_fraction=0.12,
            temp0=0.006,
            cool_factor=0.99998,
            recompute_every=1500,
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final short L‑BFGS polish (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        # a quick finish with a tighter lambda schedule and fewer iterations
        fine_lam = [0.00002, 0.00001, 5e-6, 2e-6, 1e-6]
        best_a, best_score = lbfgs_continuation(
            best_a,
            fine_lam,
            maxiter_per_lam=400,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        print(f"[final‑L‑BFGS] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
