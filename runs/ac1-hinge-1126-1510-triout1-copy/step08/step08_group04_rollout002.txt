def gradient_flatten_descent(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # early break if optimum below target?
        # compute gradient of max conv
        k = int(np.argmax(conv))
        # gradient of conv_k w.r.t a
        # compute g_i = 2*a_{k-i} if i in [max(0,k-(n-1)), min(n-1,k)], else 0
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # vectorized:
        idx = np.arange(i_min, i_max+1)
        g[idx] = 2.0 * a[k - idx]
        # step size adaptively: try eta that reduces max_val
        eta = 0.5
        improved = False
        for _ in range(10):
            a_candidate = a - eta * g
            # project onto simplex
            a_candidate = simplex_projection(a_candidate)
            # compute new max
            fa_c = np.fft.fft(a_candidate, fft_len)
            conv_c = np.real(np.fft.ifft(fa_c * fa_c))[:2*n-1]
            max_c = conv_c.max()
            if max_c < max_val - 1e-12:
                a = a_candidate
                max_val = max_c
                improved = True
                break
            eta *= 0.5
        # update best
        cur_score = 2.0 * n * max_val  # fast_score = 2*n*max_val
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # if not improved, maybe add random perturbation to escape

            # small random transfer
            i, j = rng.integers(0, n, size=2)
            if a[i] > 0:
                delta = a[i] * rng.random() * 0.1
                a[i] -= delta
                a[j] += delta
                a = simplex_projection(a)
    return best, best_score

"""
Propose a low‑score step‑function sequence.

The algorithm follows four stages:

1.  Heavy L‑BFGS on a log‑sum‑exp (smoothed‑max) objective,
    using many random restarts and a long λ‑schedule.

2.  Exact exponentiated‑gradient descent on the true max‑convolution
    (sub‑gradient of the ‑max).  The step‑size is larger at the start
    and decays slowly to speed up convergence.

3.  Simulated annealing on the simplex (mass transfers + Gaussian
    perturbations) to escape shallow basins.

4.  A dedicated projected‑gradient refinement that repeatedly reduces
    the current maximal convolution entry by moving mass opposite to its
    sub‑gradient.  This phase typically squeezes the final score a few
    thousandths lower than the previous pipeline.

All phases respect the non‑negative simplex constraint
∑ a_i = 1 (scaling does not affect the official objective).  The
routine stops a few hundredths of a second before the 1000 s budget
expires and returns the best list of floats found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the official objective 2·n·max(a*a)
    (the denominator Σa² cancels because a is kept on the simplex).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random distribution on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "logit_normal",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # logit_normal
        logits = rng.normal(0.0, 1.0, size=n)
        e = np.exp(logits - np.max(logits))
        a = e / e.sum()
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed version of the max‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax mapping
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # ≤ 0
    exp_shift = np.exp(shifted / lam)          # safe, ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # slice that corresponds to a (indices n‑1 … n‑1 + n‑1)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous λ.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exact exponentiated‑gradient descent on the hard max‑convolution.
    The step‑size starts relatively large and decays like 1/√t.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1) : (n - 1 + n)]

        # step size – larger at the beginning, then 1/√t
        eta = max(0.5, 1.0 / np.sqrt(it + 1.0))

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = simplex_projection(a)

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        max_local_transfer: float = 0.5):
    """
    Simple simulated‑annealing on the simplex.
    Proposes either a small mass transfer between two coordinates
    or a global Gaussian perturbation (add_noise).  Temperature
    decays linearly.
    """
    a_cur = a_start.copy()
    cur_score = fast_score(a_cur, fft_len, n)
    best_a = a_cur.copy()
    best_score = cur_score

    start = time.time()
    init_temp = cur_score * 0.02 + 1e-9

    while time.time() < deadline:
        frac = (time.time() - start) / (deadline - start)
        temp = init_temp * (1.0 - frac)

        a_new = a_cur.copy()
        if rng.random() < 0.6:
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 0.0:
                delta = a_new[i] * rng.random() * max_local_transfer
                a_new[i] -= delta
                a_new[j] += delta
        else:
            sigma = rng.uniform(0.0005, 0.015)
            a_new = add_noise(a_new, sigma, rng)

        a_new = simplex_projection(a_new)
        new_score = fast_score(a_new, fft_len, n)
        delta = new_score - cur_score

        if delta < 0.0 or rng.random() < np.exp(-delta / (temp + 1e-12)):
            a_cur = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


def gradient_flatten_descent(a_start: np.ndarray,
                             deadline: float,
                             n: int,
                             fft_len: int,
                             rng: np.random.Generator,
                             max_no_improve: int = 300):
    """
    Projected sub‑gradient descent that explicitly reduces the current
    maximal convolution entry.  At each iteration we move opposite to
    the gradient of the active convolution entry, perform a line‑search,
    and project back onto the simplex.  Occasional random transfers help
    escape tiny plateaus.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    no_improve_counter = 0
    while time.time() < deadline:
        # compute self‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        k = int(np.argmax(conv))

        # build gradient of conv_k
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        g[idx] = 2.0 * a[k - idx]

        # simple back‑tracking line‑search
        eta = 0.5
        improved = False
        for _ in range(12):
            a_trial = a - eta * g
            a_trial = simplex_projection(a_trial)
            ft = np.fft.fft(a_trial, fft_len)
            conv_trial = np.real(np.fft.ifft(ft * ft))[: 2 * n - 1]
            if conv_trial.max() < max_val - 1e-14:
                a = a_trial
                max_val = conv_trial.max()
                improved = True
                break
            eta *= 0.5

        # occasional random perturbation if stuck
        if not improved:
            i, j = rng.integers(0, n, size=2)
            if a[i] > 0.0:
                delta = a[i] * rng.random() * 0.07
                a[i] -= delta
                a[j] += delta
                a = simplex_projection(a)

        cur_score = 2.0 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            no_improve_counter = 0
        else:
            no_improve_counter += 1
            if no_improve_counter >= max_no_improve:
                # restart from the best seen so far with a tiny noise
                a = add_noise(best_a, 0.0008, rng)
                a = simplex_projection(a)
                no_improve_counter = 0

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    The routine respects a 1000 s budget and returns the best list found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial sequence
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # Use a previously stored construction if available
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = simplex_projection(a0)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    # allocate 35 % → L‑BFGS, 30 % → exp‑grad, 20 % → SA,
    # final 15 % → gradient flattening
    phase1_end = start_time + 0.35 * budget_s
    phase2_end = start_time + 0.65 * budget_s
    phase3_end = start_time + 0.85 * budget_s
    phase4_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
        5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 1800

    while time.time() < phase1_end:
        # occasional use of the current best as a warm start
        if best_score < np.inf and rng.random() < 0.25:
            sigma = rng.uniform(0.0008, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        a_opt, score_opt = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[exp‑grad] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        a_opt, score_opt = simulated_annealing(
            best_a,
            phase3_end,
            n,
            fft_len_conv,
            rng,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[SA] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected gradient flattening (final polishing)
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        a_opt, score_opt = gradient_flatten_descent(
            best_a,
            phase4_end,
            n,
            fft_len_conv,
            rng,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[grad‑flat] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Symmetrisation (the problem is symmetric; mirroring cannot increase the max)
    # ------------------------------------------------------------------
    sym_a = (best_a + best_a[::-1]) / 2.0
    sym_a = simplex_projection(sym_a)
    sym_score = fast_score(sym_a, fft_len_conv, n)
    if sym_score < best_score:
        best_score = sym_score
        best_a = sym_a
        print(f"[symmetrize] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # noqa: F821 – provided by the environment
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
