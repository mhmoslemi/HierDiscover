"""
Hybrid optimizer for the step‑function bound.

The routine builds a probability vector a (sum = 1, a_i ≥ 0) of length n.
Its quality is measured by
        score = 2·n·max(conv(a,a)),
which is exactly the quantity returned by evaluate_sequence after the
normalisation performed in that function.  The algorithm proceeds in four
phases, each respecting a global time budget:

1.  L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution.
    A continuation in the smoothing parameter λ yields a very good
    neighbourhood of the optimum.

2.  Exponentiated‑gradient descent on the *exact* max‑convolution.
    The update a ← a·exp(−η·g) stays on the simplex automatically.

3.  Projected sub‑gradient descent (hard max) – an alternative descent
    direction that often escapes the plateaus left by the exponentiated
    gradient.

4.  Stochastic hill‑climb: repeatedly transfer a tiny fraction of mass
    from the “most harmful’’ coordinate (largest contribution to the
    maximal convolution entry) to a “least harmful’’ one, accepting all
    improving moves.  Random jitter is added to avoid getting stuck.

All phases keep the best vector found so far.  The final vector is
converted to a plain Python list and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a has been normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z      # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w) in Fourier domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extracts the part that corresponds to gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits for the soft‑max

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # any numerical trouble – just skip this λ
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len_conv: int,
                                   fft_len_g: int,
                                   rng: np.random.Generator):
    """
    Classic projected sub‑gradient descent on the *hard* max‑convolution.
    Keeps iterates on the simplex via Euclidean projection.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max entry
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # diminishing step size (AdaGrad‑like scaling)
        eta = 1.0 / np.sqrt(it + 1.0)

        a_new = a - eta * g
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score


def guided_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      init_step: float = 0.12,
                      min_step: float = 0.001) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb: move a fraction of mass from the coordinate
    that contributes most to the current maximal convolution entry to a
    coordinate with the smallest partner value.  Step size shrinks on
    rejections.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside the support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # choose a destination with the smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()          # protect against round‑off drift

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.8, min_step)
    return a, score


def random_local_search(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb: repeatedly transfer a tiny random fraction of mass
    from a high‑contributing index to a low‑contributing one.  Only strict
    improvements are kept.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    while time.time() < deadline:
        # recompute the convolution and the critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner as in the deterministic climb
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_high = np.argmax(contrib)
        i_low = np.argmin(partner)         # tends to be a “light’’ coordinate

        # a random fraction; start with 1 % and shrink slowly
        progress = (time.time() - start_time) / (deadline - start_time)
        step = max(0.01 * (1.0 - progress), 0.001)

        delta = a[i_high] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_high] -= delta
        a_new[i_low] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
        else:
            # occasionally try a random destination to escape flat plateaus
            if rng.random() < 0.02:
                j = rng.integers(n)
                if j != i_high:
                    a_alt = a.copy()
                    a_alt[i_high] -= delta
                    a_alt[j] += delta
                    a_alt /= a_alt.sum()
                    alt_score = fast_score(a_alt, fft_len, n)
                    if alt_score < best_score:
                        best_score = alt_score
                        best_a = a_alt.copy()
                        a = a_alt
        # end of one iteration
    return best_a, best_score

    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:                     # beta distribution (asymmetric shapes)
        a = rng.beta(a=rng.uniform(0.5, 2.0),
                     b=rng.uniform(0.5, 2.0), size=n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Initialise via a spectral construction and project onto the simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000               # length of the step function (feel free to change)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a previously found construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # a few diverse starters
    a0 = a_start.copy()
    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.40 * budget_s   # L‑BFGS
    phase2_deadline = start_time + 0.80 * budget_s   # exp‑grad
    phase3_deadline = start_time + 0.95 * budget_s   # projected sub‑grad
    phase4_deadline = deadline                      # stochastic hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with a finer λ schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.001, 0.0005,
        0.0002, 0.0001, 5e-05, 1e-05
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_deadline:
        r = rng.random()
        if r < 0.45:
            a_start = random_initial(n, rng)
        elif r < 0.90:
            a_start = add_noise(best_a,
                                sigma=rng.uniform(0.001, 0.012),
                                rng=rng)
        else:
            a_start = spectral_init(n)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[exp‑grad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (hard max)
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = projected_subgradient_optimize(
            best_a,
            phase3_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[proj‑subgrad] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – stochastic hill‑climb (fine‑grained local search)
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = random_local_search(
            best_a,
            phase4_deadline,
            n,
            fft_len_conv,
            rng,
        )
        print(f"[random‑hill‑climb] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
