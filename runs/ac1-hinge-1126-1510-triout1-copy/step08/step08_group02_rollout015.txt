"""
Conditional‑gradient (Frank‑Wolfe) + exponentiated‑gradient optimizer.

* Phase 1 – Frank‑Wolfe style linear‑programming updates.
  For the current step‑function a (∑a=1) we compute M = max(a*a).
  On the active set of largest convolution entries we solve
  a linear program that maximises ∑b subject to (a*b)≤M and b≥0.
  The optimiser b (renormalised to a simplex) is a promising
  direction; we move a partially toward b and repeat.

* Phase 2 – exponentiated‑gradient (mirror descent with KL‑projection)
  on the exact max‑convolution.  A decreasing step‑size η≈1/√t
  is used and a is kept on the probability simplex via normalisation.

* Phase 3 – a short L‑BFGS polish on a smooth log‑sum‑exp
  surrogate of the max‑convolution.

* Phase 4 – greedy hill‑climb for a final polish.

The routine respects the 1000 s budget and returns the best
non‑negative sequence found (a simple Python list of floats).
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) assuming a sums to 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def full_conv(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Full (zero‑padded) self‑convolution of a using FFTs."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of sqrt of the FFT of a uniform target, projected onto simplex."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L, dtype=np.float64)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(amp)
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random simplex vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – Frank‑Wolfe style LP step
# ----------------------------------------------------------------------


def solve_lp_active(a: np.ndarray, active_idx: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max ∑b   s.t.  (a*b)[k] ≤ M for all k∈active_idx,  b≥0.
    Returns the solution b (not normalised).
    """
    n = a.shape[0]
    m = active_idx.shape[0]
    # Build dense A_ub (m × n)
    A = np.zeros((m, n), dtype=np.float64)
    for row, s in enumerate(active_idx):
        k_min = max(0, s - (n - 1))
        k_max = min(s, n - 1)
        ks = np.arange(k_min, k_max + 1, dtype=np.intp)
        coeffs = a[s - ks]
        A[row, ks] = coeffs
    # Objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    b_ub = M * np.ones(m, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback: return the current a (a feasible point)
        return a.copy()


def frank_wolfe_step(a: np.ndarray,
                     fft_len: int,
                     n: int,
                     active_k: int = 30,
                     step_candidates: tuple = (0.5, 0.3, 0.2, 0.1, 0.05, 0.02)):
    """
    Perform one Frank‑Wolfe iteration:
    * compute active set,
    * solve LP for direction g,
    * try a few convex‑combination step sizes,
    * return the best new a and its score.
    """
    conv = full_conv(a, fft_len, n)
    M = conv.max()
    # pick the K largest indices (active constraints)
    if active_k >= conv.size:
        active = np.arange(conv.size, dtype=np.intp)
    else:
        active = np.argpartition(-conv, active_k - 1)[:active_k]
    b = solve_lp_active(a, active, M)
    sum_b = b.sum()
    if sum_b <= 0.0:
        g = a.copy()
    else:
        g = b / sum_b                      # feasible direction on simplex

    best_local = a.copy()
    best_score = fast_score(a, fft_len, n)

    for t in step_candidates:
        cand = (1.0 - t) * a + t * g
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s <= 0.0:
            continue
        cand /= s
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_local = cand.copy()
    return best_local, best_score


# ----------------------------------------------------------------------
# Phase 2 – Exponentiated gradient on exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """Mirror‑descent (exponentiated‑gradient) on the exact max‑conv."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        conv = full_conv(a, fft_len, n)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
            s = a.sum()
        a /= s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Smoothed L‑BFGS refinement
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """Log‑sum‑exp smooth approximation of max(a*a) and its gradient."""
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                                   # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_refine(a_start: np.ndarray,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """Run a short L‑BFGS on the smoothed objective."""
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    # a decreasing schedule for the smoothing parameter
    lam_schedule = [0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002]
    maxiter = 500
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # rare numerical issues
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 15000):
    """Simple hill‑climb: move a small random mass between coordinates."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                        # trade‑off between quality and speed
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Prior construction (if available)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2️⃣  Spectral guess
    candidates.append(spectral_init(n))

    # 3️⃣  Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # 4️⃣  Random initial
    candidates.append(random_initial(n, rng))

    # pick the best of the initial pool
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # Phase 1 – conditional‑gradient / LP iterations
    phase1_end = start_time + 0.30 * budget_s
    while time.time() < phase1_end:
        new_a, new_sc = frank_wolfe_step(best_a, fft_len, n,
                                         active_k=30,
                                         step_candidates=(0.5, 0.3, 0.2,
                                                          0.1, 0.05, 0.02))
        if new_sc < best_score:
            best_score = new_sc
            best_a = new_a.copy()
            print(f"[FW] improved fast_score={best_score:.6f}")

    # Phase 2 – exponentiated‑gradient descent
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # Phase 3 – L‑BFGS polishing on a smooth surrogate
    if time.time() < deadline:
        best_a, best_score = lbfgs_refine(best_a, deadline, fft_len, n)
        print(f"[lbfgs] fast_score={best_score:.6f}")

    # Phase 4 – final greedy hill‑climb
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=25000)
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
