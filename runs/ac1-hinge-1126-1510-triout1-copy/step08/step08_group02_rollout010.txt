def simplex_projection(v):
    """Euclidean projection of vector onto probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all entries are zero, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
```

    fa = np.fft.rfft(a, fft_len)

def spectral_init(n):
    """Initialize a using square‑root of uniform spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)
    return a

def random_initial(n, rng):
    """Generate a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

    """
    Smoothed max of self‑convolution via log‑sum‑exp.
    Returns (objective, gradient) with respect to logits y (a = softmax(y)).
    """
    a = softmax(y)

    # Full convolution a * a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt conv entries
    w = exp_shift / Z   # d smooth_max / d conv_i

    # Backpropagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # Extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y
```

def lbfgs_continuation(a_start, lam_schedule, maxiter_per_lam,
                       deadline, fft_len_conv, fft_len_g, n):
    """
    Run L‑BFGS on a smoothed max objective for each lambda in lam_schedule.
    Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Rare failures – skip this lambda
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def adam_optimize(a_start, lam_schedule, iters_per_lam,
                  deadline, n, fft_len, rng):
    """
    Mirror‑descent (Adam) on the smoothed max objective.
    Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            # compute gradient of smoothed max
            f, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam_factor=lam)
            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = 0.5  # base learning rate; can be tuned
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

def subgradient_descent_ls(a_start, deadline, n, fft_len):
    """
    Projected sub‑gradient descent with back‑tracking line search.
    Returns best a and best score.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score
    it = 0
    step0 = 1.0

    while time.time() < deadline:
        # compute convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction that respects sum constraint
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # back‑tracking line search
        alpha = step0 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(10):
            a_new = simplex_projection(a + alpha * d)
            new_score = fast_score(a_new, fft_len, n)
            if new_score < cur_score - 1e-12:
                a = a_new
                cur_score = new_score
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                break
            alpha *= 0.5
        if not improved:
            # accept a minimal move to avoid stagnation
            a = simplex_projection(a + alpha * d)
            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        it += 1
    return best_a, best_score

def hill_climb(a, current_score, fft_len, n, deadline, max_iter=20000):
    """
    Simple greedy hill‑climb that transfers a random fraction of mass.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

"""
Propose a low‑score step function (search for the best non‑negative height
sequence).  The algorithm combines several complementary optimisation
phases within the allotted time budget:

1.  Initialise a diverse set of candidates (interpolation of a previous
    solution, a spectral “square‑root of uniform’’ guess, and a random
    Dirichlet‑style draw).

2.  Short L‑BFGS polishings on a log‑sum‑exp smoothed version of the
    max‑convolution objective.  Multiple random restarts explore the
    landscape.

3.  Mirror‑descent (Adam) on the same smoothed objective with a
    continuation on the smoothing factor.  This efficiently reduces the
    true max‑convolution while automatically keeping the vector on the
    probability simplex.

4.  Projected sub‑gradient descent with a back‑tracking line search.
    The step size is larger than in the reference code and is
    continuously adapted, allowing the method to escape shallow plateaus.

5.  A final greedy hill‑climb that transfers a small random fragment of
    mass between two coordinates and accepts only strictly improving
    moves.

All phases are timed so that the total execution never exceeds the
budget (default 1000 s).  The returned list of floats is ready to be fed
to `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def spectral_init(n: int) -> np.ndarray:
    """Initialise a by taking the inverse FFT of the square‑root of the
    FFT of a uniform target (length 2·n‑1)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0.0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random probability vector on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # Full convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z          # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  iters_per_lam,
                  deadline,
                  n,
                  fft_len,
                  rng: np.random.Generator):
    """
    Mirror‑descent (Adam) on the smoothed max‑convolution objective.
    Returns the best a encountered and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            # gradient of the smoothed objective
            _, grad_y = objective_and_grad(y, n, fft_len, fft_len,
                                          lam_factor=lam)

            # Adam update on the logits
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            lr = 0.5                     # base learning rate
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


def subgradient_descent_ls(a_start: np.ndarray,
                           deadline,
                           n,
                           fft_len):
    """
    Projected sub‑gradient descent with a simple back‑tracking line search.
    Returns the best a and its fast_score.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score
    it = 0
    step0 = 1.0                      # initial step scale

    while time.time() < deadline:
        # --- convolution and mask of active constraints ---
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # --- sub‑gradient of the max ---
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # --- descent direction (sum zero) ---
        mean_g = np.mean(g)
        d = -(g - mean_g)

        # --- line search ---
        alpha = step0 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(10):
            a_new = simplex_projection(a + alpha * d)
            new_score = fast_score(a_new, fft_len, n)
            if new_score < cur_score - 1e-12:
                a = a_new
                cur_score = new_score
                improved = True
                if new_score < best_score:
                    best_score = new_score
                    best_a = a_new.copy()
                break
            alpha *= 0.5

        if not improved:
            # take a tiny step to keep moving
            a = simplex_projection(a + alpha * d)
            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        it += 1
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 30000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Uses a mix of L‑BFGS, Adam‑style mirror descent, sub‑gradient descent,
    and a final hill‑climb.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (larger n gives more flexibility while keeping FFT cheap)
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial candidates
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0.0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    a_spec = spectral_init(n)
    a_rand = random_initial(n, rng)

    candidates = [a0, a_spec, a_rand]
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Split the time budget
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    phase1_deadline = start_time + 0.15 * budget_s
    phase2_deadline = start_time + 0.45 * budget_s
    phase3_deadline = start_time + 0.85 * budget_s

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing from random starts
    # ------------------------------------------------------------------
    lam_schedule_lbfgs = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                         0.05, 0.03, 0.018, 0.01, 0.006,
                         0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 300
    while time.time() < phase1_deadline:
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule_lbfgs,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len,
                                           fft_len,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam mirror descent on smoothed max
    # ------------------------------------------------------------------
    lam_schedule_adam = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
                         0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    iters_per_lam = 2000
    if time.time() < phase2_deadline:
        a_adam, sc_adam = adam_optimize(best_a,
                                        lam_schedule_adam,
                                        iters_per_lam,
                                        phase2_deadline,
                                        n,
                                        fft_len,
                                        rng)
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam.copy()
            print(f"[Adam] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent with line search
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        a_sub, sc_sub = subgradient_descent_ls(best_a,
                                               phase3_deadline,
                                               n,
                                               fft_len)
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub.copy()
            print(f"[Subgrad] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Final greedy hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=30000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
