"""
Propose a low‑score step‑function sequence.

The search proceeds in four stages:

1.  Initialise from the previous best (if any) and from a small
    parametric sweep (exponential / power “U‑shaped” families) that
    already yields a fairly flat self‑convolution.

2.  Run many short runs of exponentiated‑gradient descent from random
    starts (different “modes” – sparse, edge‑heavy, random, …).
    The EG update keeps the vector on the probability simplex and
    directly reduces the true max‑convolution objective.

3.  Keep the best vector found so far and polish it with a simple
    hill‑climb that transfers a small fraction of mass between two
    coordinates only if the score improves.

4.  Return the final list; the official ``evaluate_sequence`` function
    is called once for reporting purposes.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, conv_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:conv_len]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random distribution on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "logit_normal",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m = max(1, int(edge_frac * n))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m] = heavy
        a[-m:] = heavy
    elif mode == "bimodal":
        centre = rng.integers(n // 3, 2 * n // 3)
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = rng.uniform(0.5, 1.5) / n
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # logit_normal
        logits = rng.normal(0.0, 1.0, size=n)
        # stable soft‑max
        mlog = np.max(logits)
        e = np.exp(logits - mlog)
        a = e / e.sum()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution)
# ----------------------------------------------------------------------


def exp_gradient_one_run(
    a_start: np.ndarray,
    fft_len: int,
    conv_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Run exponentiated‑gradient descent from ``a_start`` until ``deadline``.
    Returns the best vector and its fast_score.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, conv_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # linear convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:conv_len]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # 0/1 mask

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # step‑size schedule (1/√t)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best.copy()
        else:
            a = a / s

        cur_score = fast_score(a, fft_len, conv_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

    return best, best_score


# ----------------------------------------------------------------------
# Simple hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    conv_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a small fraction of mass from a
    “heavy’’ coordinate to a lighter one whenever the move improves
    the score.
    """
    rng = np.random.default_rng()
    best = a.copy()
    best_score = cur_score

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, conv_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score
