def objective_and_grad(y, n, fft_len, lam_factor=0.30):
    a = softmax(y)
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_rev_full = np.fft.irfft(fa_rev * fw, n=fft_len)
    g_vec = conv_rev_full[(n-1):(n-1 + n)]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def objective_and_grad_pnorm(y, n, fft_len, p):
    a = softmax(y)
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    # compute sum of p-th powers
    conv_pow = np.power(conv_full, p, dtype=np.float64)
    s = conv_pow.sum()
    if s <= 0:
        smooth_max = 0.0
        w = np.zeros_like(conv_full)
    else:
        smooth_max = s ** (1.0/p)
        # w_i = conv_i^{p-1} / smooth_max^{p-1}
        w = np.power(conv_full, p-1, where=conv_full > 0, out=np.zeros_like(conv_full))
        if smooth_max != 0:
            w = w / (smooth_max ** (p-1))
        else:
            w = np.zeros_like(conv_full)
    f = 2.0 * n * smooth_max
    # gradient wrt a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_rev_full = np.fft.irfft(fa_rev * fw, n=fft_len)
    g_vec = conv_rev_full[(n-1):(n-1 + n)]
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def simulated_annealing(a_start, score_start, deadline):
    a = a_start.copy()
    best = a.copy()
    best_score = score_start
    score = score_start
    T0 = 0.01 * score_start  # initial temperature
    T_end = 1e-7
    # compute number of iterations approx
    # We'll use exponential cooling
    it = 0
    # initial temperature scaling factor per iteration:
    # compute cooling factor automatically based on remaining time and desired iterations.
    # We'll just use factor = 0.99995
    cooling = 0.99995
    T = T0
    while time.time() < deadline:
        grad, max_val = compute_subgradient(a, fft_len)
        # choose i_hi, i_lo
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if a[i_hi] <= 0:
            # random perturb
            i_hi = rng.integers(0, n)
            i_lo = rng.integers(0, n)
            continue
        frac = rng.random() * 0.2  # up to 20%
        delta = a[i_hi] * frac
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i_hi] -= delta
        a_new[i_lo] += delta
        new_score = fast_score(a_new, fft_len, n)
        delta_score = new_score - score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):
            # accept
            a = a_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
        # cooling
        T = max(T_end, T * cooling)
        it += 1
        # occasional random restart or noise injection
        if it % 2000 == 0:
            # maybe add small noise if stuck
            if rng.random() < 0.5:
                a = add_noise(a, sigma=0.005, rng=rng)
                # reproject onto simplex
                a = simplex_projection(a)
                score = fast_score(a, fft_len, n)
        # optionally print
        if it % 5000 == 0:
            print(f"[SA] it={it} best_score={best_score:.6f} T={T:.6e}")
    return best, best_score
```

def lbfgs_optimize(a_start, schedule, max_iter_per_stage, deadline, fft_len, n, objective_func):
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)
    for param in schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_func,
                x0=y,
                args=(n, fft_len, param),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": max_iter_per_stage, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def subgradient_optimize_momentum(a0, deadline, fft_len, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        # compute conv and max
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # gradient
        a_rev = a[::-1]
        fw = np.fft.rfft(mask, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv_g_full = np.fft.irfft(fa_rev * fw, n=fft_len)
        g = 2.0 * conv_g_full[(n-1):(n-1+n)]
        # momentum
        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)
        cand_score = fast_score(a_candidate, fft_len, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score
```

def exp_gradient_optimize(a0, deadline, fft_len, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # compute conv and mask
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        fw = np.fft.rfft(mask, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv_g_full = np.fft.irfft(fa_rev * fw, n=fft_len)
        g = 2.0 * conv_g_full[(n-1):(n-1+n)]
        eta = 0.25 / np.sqrt(it + 1.0)
        # EG update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1
    return best_a, best_score

    n = a.shape[0]
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]

    fa_rev = np.fft.rfft(a_rev, n=fft_len)

def simulated_annealing(a_start, score_start, fft_len, n, deadline, rng):
    a = a_start.copy()
    best = a.copy()
    best_score = score_start
    score = score_start
    T0 = 0.02 * score_start   # a bit larger to accept uphill moves
    T_end = 1e-8
    cooling = 0.99995
    T = T0
    it = 0
    while time.time() < deadline:
        grad, max_val = compute_subgradient(a, fft_len)
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if a[i_hi] <= 0:
            # random indices fallback
            i_hi = rng.integers(0, n)
            i_lo = rng.integers(0, n)
        frac = rng.random() * 0.2   # up to 20%
        delta = a[i_hi] * frac
        if delta <= 0:
            continue
        a_new = a.copy()
        a_new[i_hi] -= delta
        a_new[i_lo] += delta
        new_score = fast_score(a_new, fft_len, n)
        delta_score = new_score - score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):
            a = a_new
            score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()
        T = max(T_end, T * cooling)
        it += 1
        # occasional injection of noise to escape plateaus
        if it % 5000 == 0:
            a = add_noise(a, sigma=0.005, rng=rng)
            a = simplex_projection(a)
            score = fast_score(a, fft_len, n)
            if score < best_score:
                best_score = score
                best = a.copy()
        if it % 2000 == 0:
            print(f"[SA] it={it}, best_score={best_score:.6f}, T={T:.2e}")
    return best, best_score
```

"""
Search for a non‑negative step sequence minimizing

    2·n·max(conv(a, a))

where `conv` is the linear convolution of the sequence with itself.
The algorithm combines several complementary phases that respect the
overall 1000 s time budget:

1. Cheap parametric sweep over a variety of analytic families.
2. Continuation L‑BFGS on a smoothed log‑sum‑exp approximation of the
   max‑convolution.
3. Continuation L‑BFGS on high‑order p‑norm approximations (p → ∞)
   of the max‑convolution.
4. Projected sub‑gradient descent with momentum on the true max‑convolution.
5. Exponentiated‑gradient descent on the simplex.
6. A simulated‑annealing hill‑climber that repeatedly moves mass from the
   most harmful coordinate (largest sub‑gradient) to the most helpful
   one, occasionally accepting uphill moves to escape local minima.

All phases keep the sequence on the probability simplex (sum = 1,
non‑negative).  The best sequence found is finally evaluated with the
provided `evaluate_sequence` and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    return conv_full[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a sequence that already satisfies ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed objectives
# ----------------------------------------------------------------------


def objective_logsumexp(y: np.ndarray, n: int, fft_len: int, lam_factor: float):
    """Smooth max via log‑sum‑exp. Returns (objective, grad w.r.t. y)."""
    a = softmax(y)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)

    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, n=fft_len)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def objective_pnorm(y: np.ndarray, n: int, fft_len: int, p: float):
    """Smooth max via the p‑norm (p → ∞). Returns (objective, grad w.r.t. y)."""
    a = softmax(y)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

    conv = np.maximum(conv, 0.0)          # numerical safety
    conv_pow = np.power(conv, p, dtype=np.float64)
    s = conv_pow.sum()
    smooth_max = s ** (1.0 / p)
    # weight for each convolution entry
    if smooth_max == 0.0:
        w = np.zeros_like(conv)
    else:
        w = np.power(conv, p - 1, where=conv > 0, out=np.zeros_like(conv))
        w = w / (smooth_max ** (p - 1))

    f = 2.0 * n * smooth_max
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, n=fft_len)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def lbfgs_optimize(a_start: np.ndarray,
                   schedule,
                   maxiter_per_stage: int,
                   deadline: float,
                   fft_len: int,
                   n: int,
                   objective_func):
    """
    Continuation L‑BFGS: run L‑BFGS for each parameter in ``schedule``,
    starting each stage from the result of the previous stage.
    Returns the best sequence found and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for param in schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_func,
                x0=y,
                args=(n, fft_len, param),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len: int,
                                  n: int):
    """Projected sub‑gradient descent with momentum on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fw = np.fft.rfft(mask, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv_g = np.fft.irfft(fa_rev * fw, n=fft_len)
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(a0: np.ndarray,
                          deadline: float,
                          fft_len: int,
                          n: int):
    """Exponentiated‑gradient descent on the simplex using the true sub‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fw = np.fft.rfft(mask, n=fft_len)
        fa_rev = np.fft.rfft(a_rev, n=fft_len)
        conv_g = np.fft.irfft(fa_rev * fw, n=fft_len)
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.25 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def compute_subgradient(a: np.ndarray, fft_len: int):
    """Return (gradient of max‑convolution w.r.t. a, max value)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fw = np.fft.rfft(mask, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_g = np.fft.irfft(fa_rev * fw, n=fft_len)
    grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return grad, max_val


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Gradient‑guided simulated annealing.
    Moves a random fraction of mass from the index with the largest
    sub‑gradient to the one with the smallest, occasionally accepting
    uphill moves.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = score_start
    cur_score = score_start

    T0 = 0.02 * score_start          # initial temperature
    T_end = 1e-8
    cooling = 0.99995
    T = T0
    it = 0

    while time.time() < deadline:
        grad, _ = compute_subgradient(a, fft_len)
        i_hi = int(np.argmax(grad))
        i_lo = int(np.argmin(grad))
        if a[i_hi] <= 0.0:
            i_hi = rng.integers(0, n)
            i_lo = rng.integers(0, n)

        frac = rng.random() * 0.20          # up to 20 % of the mass at i_hi
        delta = a[i_hi] * frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i_hi] -= delta
        a_new[i_lo] += delta

        new_score = fast_score(a_new, fft_len, n)
        delta_score = new_score - cur_score

        if delta_score < 0.0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()

        T = max(T_end, T * cooling)
        it += 1

        # occasional noise injection to escape plateaus
        if it % 5000 == 0:
            a = add_noise(a, sigma=0.005, rng=rng)
            a = simplex_projection(a)
            cur_score = fast_score(a, fft_len, n)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()

        if it % 2000 == 0:
            print(f"[SA] it={it:,} best_score={best_score:.6f} T={T:.2e}")

    return best, best_score

def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def candidate_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_rev_power_law(n: int, p: float) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float) ** p)
    a /= a.sum()
    return a


def candidate_exponential(n: int, lam: float, reverse: bool = False) -> np.ndarray:
    if reverse:
        a = np.exp(-lam * np.arange(n)[::-1])
    else:
        a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def candidate_gaussian(n: int, sigma: float) -> np.ndarray:
    mid = (n - 1) / 2.0
    a = np.exp(-((np.arange(n) - mid) ** 2) / (2.0 * sigma * sigma))
    a /= a.sum()
    return a


def candidate_beta_symmetric(n: int, alpha: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    a = (i ** (alpha - 1)) * ((n - i + 1) ** (alpha - 1))
    a /= a.sum()
    return a


def candidate_sine_power(n: int, p: float) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        if hs.ndim == 1 and hs.size > 0:
            if hs.size != n:
                x_old = np.linspace(0, 1, hs.size)

                a0 = hs.copy()
            a0 = np.maximum(a0, 0.0)

    best_a = a0.copy()

    while time.time() < phase1_end:

    # ------------------------------------------------------------------

        if rng.random() < 0.25:

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.75 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase3_end, fft_len, n

        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.85 * budget_s

            best_a, phase4_end, fft_len, n

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
