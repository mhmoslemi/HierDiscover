"""
Propose a low‑score step‑function sequence.

Algorithm outline
-----------------
1. **Initialisation** – start from the previously discovered best sequence
   (if available) or from a uniform distribution.
2. **Random parametric search** – generate thousands of simple analytical
   families (exponential, power‑law, cosine‑bump, mixtures with the uniform)
   and keep the best one according to a fast FFT‑based score.
3. **Adam optimisation on a smoothed max‑objective** – work in the log‑softmax
   (log‑it) space, minimise a log‑sum‑exp approximation of the max‑convolution.
   A schedule of decreasing smoothing factors (lam) drives the optimisation
   towards the true max.
4. **Projected sub‑gradient descent on the exact max** – compute the exact
   sub‑gradient of the `max(a*a)` functional (via a mask of the active
   entries) and perform a decreasing‑step projected gradient descent on the
   simplex.
5. **Optional final L‑BFGS polish** – a single low‑lam L‑BFGS run for a final
   touch‑up.
6. **Return** the best sequence found; it will be evaluated with
   `evaluate_sequence` after the search finishes.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two >= x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when `a` is already normalised (∑a = 1):
        2·n·max(a*a)
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and renormalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    The smoothing parameter is lam = lam_factor * max(conv).
    """
    # --- softmax (stable) ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution a*a ---
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth_max w.r.t. convolution entries ---
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # --- back‑propagation through the convolution (a ← softmax(y)) ---
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to the derivative w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --- chain rule through softmax ---
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS continuation (heavy smoothing → almost true max)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a and its fast_score.
    """
    # work in log‑softmax space
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as exc:
            # if something goes wrong just skip this lam
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc

# Main entry point
# ----------------------------------------------------------------------

    while time.time() < phase1_end:
        shape = rng.choice(["exp", "power", "cosine", "uniform"])
        mix_alpha = rng.random()               # weight of the pure uniform part

        if shape == "uniform":
            cand = np.full(n, 1.0 / n)
        elif shape == "exp":
            lam = rng.uniform(5e-4, 5e-2)       # exponential decay rate
            base = np.exp(-lam * np.arange(n)) + np.exp(-lam * np.arange(n - 1, -1, -1))
            cand = (1 - mix_alpha) * base + mix_alpha * np.ones(n)
        elif shape == "power":
            p = rng.uniform(0.2, 3.0)           # power‑law exponent
            idx = np.arange(1, n + 1, dtype=float)
            base = (idx ** (-p) + (idx[::-1]) ** (-p))
            cand = (1 - mix_alpha) * base + mix_alpha * np.ones(n)
        else:  # cosine bump
            x = np.linspace(0, np.pi, n)
            base = np.cos(x) ** 2 + 0.001
            cand = (1 - mix_alpha) * base + mix_alpha * np.ones(n)

        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()

        # occasional tiny noise to increase diversity
        if rng.random() < 0.08:
            cand = add_noise(cand, sigma=rng.uniform(0.001, 0.008), rng=rng)

        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[random] new best {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on log‑sum‑exp smoothed max
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        # work in logits (log‑softmax) space
        y = np.log(best_a + 1e-12)

        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        it = 0

        # decreasing smoothing factors (lam = fraction of current max)
        lam_schedule = [
            0.5, 0.2, 0.1, 0.05, 0.02, 0.01,
            0.005, 0.002, 0.001, 5e-04, 2e-04,
            1e-04, 5e-05, 2e-05, 1e-05,
        ]

        for lam in lam_schedule:
            if time.time() >= phase2_end:
                break
            # run a few thousand Adam steps for each lam (bounded by remaining time)
            while time.time() < phase2_end:
                it += 1
                f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
                # Adam update on logits
                m = beta1 * m + (1 - beta1) * grad_y
                v = beta2 * v + (1 - beta2) * (grad_y ** 2)
                m_hat = m / (1 - beta1 ** it)
                v_hat = v / (1 - beta2 ** it)
                step = 0.12 * m_hat / (np.sqrt(v_hat) + eps)
                y = y - step

                # convert to probabilities and evaluate the *true* max‑score
                a = np.exp(y - np.max(y))
                a = a / a.sum()
                cur_sc = fast_score(a, fft_len_conv, n)

                if cur_sc < best_score:
                    best_score = cur_sc
                    best_a = a.copy()
                    print(f"[adam] lam={lam:.1e}, iter={it}, new best={best_score:.6f}")

                # early break if the objective stagnates for a while
                if it % 2000 == 0 and time.time() > phase2_end:
                    break

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the exact max
    # --------------------------------------------------------------
    a = best_a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < phase3_end:
        # ---- convolution and current maximum ----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # ---- mask of (near) active entries ----
        eps_mask = M * 1e-12 + 1e-12
        mask = (conv >= M - eps_mask).astype(np.float64)

        # ---- sub‑gradient of max w.r.t. a (size n) ----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # ---- step size (decreasing) ----
        it += 1
        step = 0.6 / np.sqrt(it)

        # ---- projected gradient step ----
        a_next = a - step * g
        a_next = simplex_projection(a_next)

        sc_next = fast_score(a_next, fft_len_conv, n)
        if sc_next < best_score - 1e-12:
            best_score = sc_next
            best_a = a_next.copy()
            a = a_next
            if it % 100 == 0:
                print(f"[subgrad] iter={it}, best={best_score:.6f}")
        else:
            # still move a bit – prevents stagnation
            a = a_next

    # --------------------------------------------------------------
    # Optional final L‑BFGS polish (very low smoothing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        lam_final = [1e-06]
        a_final, sc_final = lbfgs_continuation(
            best_a,
            lam_final,
            maxiter_per_lam=500,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_score:
            best_score = sc_final
            best_a = a_final.copy()
            print(f"[final L‑BFGS] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the external check
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
