def generate_shuffle(n, rng):
    vals = rng.exponential(scale=1.0, size=n)
    vals.sort()
    vals = vals[::-1]  # descending
    stride = rng.integers(1, n)
    # Ensure stride coprime to n for full cycle
    if np.gcd(stride, n) != 1:
        # adjust to nearest coprime
        for offset in range(1, n):
            cand = (stride + offset) % n
            if np.gcd(cand, n) == 1:
                stride = cand
                break
    perm = [(i * stride) % n for i in range(n)]
    a = np.empty_like(vals)
    a[perm] = vals
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= total
    return a
```

"""
Improved search for a low‑evaluation step sequence.

The evaluation score is (2·n·max(a*a)) / (∑a)².
Since the problem is scale‑invariant we fix ∑a = 1 and minimise
2·n·max(a*a).  The algorithm consists of

1.   A cheap parametric sweep (including a new “shuffle” family that
     spreads large values uniformly across the index range).

2.   A continuation of L‑BFGS on a log‑sum‑exp smooth approximation of
     the max‑convolution.  The smoothing parameter λ is halved many
     times, reaching a very sharp approximation.

3.   Projected (momentum) sub‑gradient descent on the true max‑
     convolution.  The step size decays as 1/√t and we keep the best
     iterate.

4.   A final stochastic annealing that moves a random fraction of mass
     between two positions; uphill moves are accepted with a
     temperature that vanishes linearly.

All phases respect a global 1000 s budget (default).  The best sequence
found at any time is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    out = e / e.sum()
    return out


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(a*a) for a already on the simplex ∑a=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, ∑x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def smooth_max_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of max(a*a) where a = softmax(y).
    Returns (objective, gradient wrt y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp trick
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    obj = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_k

    # back‑propagation through convolution (a appears twice)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # the relevant part corresponds to the gradient w.r.t. a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return obj, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, using the previous result as start.
    """
    y = np.log(a_start + 1e-12)  # logits for simplex

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_max_and_grad,
                x0=y,
                args=(a_start.shape[0], fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best fast_score={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # exact convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑conv
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum
        v = beta * v + (1.0 - beta) * g

        step = 0.9 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv)
        if sc < best_sc:
            best_sc, best_a = sc, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_sc: float,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a random fraction of mass.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_sc
    cur_sc = start_sc

    n = a.shape[0]
    # initial temperature is a small fraction of the starting score
    T0 = 0.02 * start_sc
    t0 = time.time()
    total = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total)  # linear cooling

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue

        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_sc = fast_score(a_new, fft_len)
        diff = new_sc - cur_sc

        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Parametric families (including a new “shuffle” family)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector from a parametric family, normalised to sum = 1."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "shuffle":
        # draw exponential weights, sort, then distribute via a coprime stride
        vals = rng.exponential(scale=1.0, size=n)
        vals.sort()
        vals = vals[::-1]                     # descending
        # choose a stride coprime with n (ensures good spreading)
        stride = rng.integers(1, n)
        if np.gcd(stride, n) != 1:
            # find the nearest coprime stride
            for off in range(1, n):
                cand = (stride + off) % n
                if np.gcd(cand, n) == 1:
                    stride = cand
                    break
        perm = [(i * stride) % n for i in range(n)]
        a = np.empty_like(vals)
        a[perm] = vals
    else:
        # fallback: pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Cheap sweep over many random families, keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len)
    deadline = time.time() + budget_s * 0.15

    families = ["power", "exp", "gauss", "cos", "bimodal", "shuffle", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
    return best_a, best_sc

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (slightly larger than the previous 4000 to give more freedom)
    # ------------------------------------------------------------------
    n = 5000                                 # you may increase, but 5 k keeps FFT cheap
    fft_len_conv = next_pow2(2 * n - 1)       # length for a*a
    fft_len_g = next_pow2(3 * n - 2)          # length needed for back‑prop steps

    # ------------------------------------------------------------------
    # Start from the uniform distribution (guaranteed feasible)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len_conv)
    print(f"[init] uniform fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # If a previous construction exists, try to reuse it (interpolated)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv)
            if sc1 < best_sc:
                best_a, best_sc = a1, sc1
                print(f"[init] previous best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈15 % of the time budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_sc:
        best_a, best_sc = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Global timing
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # Allocate phase end‑times (percentages of total budget)
    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.75 * budget_s   # sub‑gradient (majority)
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation on smoothed max
    # ------------------------------------------------------------------
    lam_schedule = [
        0.50, 0.30, 0.20, 0.15, 0.10, 0.07, 0.05,
        0.03, 0.02, 0.01, 0.005, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 2e-05, 1e-05,
    ]
    maxiter_per_lam = 1200   # more inner iterations per λ

    while time.time() < phase1_end:
        # Occasionally jitter the current best; otherwise start from a fresh random family
        if rng.random() < 0.30:
            a0 = best_a.copy()
            a0 = a0 * rng.normal(1.0, 0.015, size=n)   # small Gaussian jitter
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = generate_family("shuffle", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a0,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent with momentum (major chunk)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = subgradient_descent(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[sub‑gradient] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing final refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            fft_len_conv,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
