"""
Multi‑parameter shape search for the step‑function sequence.

The search explores two one‑parameter families of non‑negative sequences
that are analytically normalised:

    1. Power‑law shape   a_i = ((p+1)/p) * (1 - (i/(n‑1))**p) / n   (p>0)
    2. Exponential shape a_i = ((1‑r) * r**i) / (1‑r**n)           (0<r<1)

For each candidate the fast FFT‑based convolution is used to obtain the
exact objective 2·n·max(a*a) (the denominator Σa = 1 after normalisation).
The best candidate is further refined by a short random hill‑climb.
The final sequence is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute the evaluation score for a normalised sequence a (Σa = 1)
    using an FFT‑based convolution.
    Returns 2·n·max(a*a).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises ``evaluate_sequence``.
    The routine tries a few analytic families, picks the best by the fast
    FFT score and finally performs a lightweight random hill‑climb.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.2      # safety margin

    # ------------------------------------------------------------------
    # Problem size (use existing length if a previous solution is present)
    # ------------------------------------------------------------------
    n = 4000
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            n = old.size

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate (normalise then fast_score)
    # ------------------------------------------------------------------
    def eval_candidate(seq: np.ndarray) -> float:
        s = seq.sum()
        if s <= 0.0:
            return float("inf")
        seq = seq / s
        return fast_score(seq, fft_len, n)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1. Power‑law family  (p > 0)
    # ------------------------------------------------------------------
    p_vals = [0.25, 0.33, 0.4, 0.5, 0.6, 0.8, 1.0, 1.5, 2.0, 3.0]
    idx = np.arange(n, dtype=np.float64)
    x = idx / (n - 1)

    for p in p_vals:
        if time.time() >= deadline:
            break
        # a_i = ((p+1)/p) * (1 - x**p) / n
        coeff = (p + 1.0) / p
        seq = coeff * (1.0 - np.power(x, p)) / n
        seq = np.maximum(seq, 0.0)
        val = eval_candidate(seq)
        print(f"[power p={p:.2f}] fast_score={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 2. Exponential family  (0 < r < 1)
    # ------------------------------------------------------------------
    r_vals = [0.999, 0.998, 0.996, 0.995, 0.99, 0.985, 0.98]
    for r in r_vals:
        if time.time() >= deadline:
            break
        powers = np.power(r, idx)                     # r**i
        # normalise: sum = (1‑r**n)  →  a_i = (1‑r) r**i / (1‑r**n)
        seq = (1.0 - r) * powers / (1.0 - r**n)
        seq = np.maximum(seq, 0.0)
        val = eval_candidate(seq)
        print(f"[exp r={r:.3f}] fast_score={val:.6f}")
        if val < best_val:

            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 3. Keep a previous global solution if it is better
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old_seq = np.array(globals()["height_sequence_1"], dtype=float)
        old_seq = np.maximum(old_seq, 0.0)
        if old_seq.sum() > 0:
            old_seq = old_seq / old_seq.sum()
        old_val = eval_candidate(old_seq)
        print(f"[old] fast_score={old_val:.6f}")
        if old_val < best_val:
            best_val = old_val
            best_seq = old_seq.copy()

    # 4. Light random hill‑climb to polish the best candidate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        rng = np.random.default_rng(seed + 1234)
        cur_seq = best_seq.copy()
        cur_val = best_val
        it = 0

    # ------------------------------------------------------------------
