"""
Propose a low‑score step‑function.

This implementation builds on the previous baseline and adds a
linear‑programming refinement step (the “LP‑mix” step) that iteratively
tightens the self‑convolution bound.  The optimisation proceeds as:

1.  Generate several diverse start points (uniform, exponential,
    random Dirichlet‑like, and a copy of the previous best if present).
2.  Perform a medium‑length exponentiated‑gradient descent on the exact
    max‑convolution (the same EG routine used in the baseline).
3.  Repeatedly solve a linear program

        max  Σ g_j
        s.t. conv(a,g) ≤ M   (M = max(conv(a,a)))
             g_j ≥ 0

   and mix the solution back into the current vector.
   Each successful mix strictly reduces the evaluation score.
4.  Finish with a greedy hill‑climb for polishing.

All heavy work (convolution, FFT, LP) respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize
from scipy import sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def compute_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (several families)."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge', 'bimodal', 'dirichlet'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # dirichlet
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha, 1.0)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def decreasing_exponential(n: int, rate: float = 0.001) -> np.ndarray:
    """Monotonically decreasing exponential shape, normalised."""
    a = np.exp(-rate * np.arange(n))
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        conv = compute_conv(a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (LP‑mix)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the (2·n‑1) × n matrix A where (A·g)[k] = (a*g)[k]
    (linear convolution of a with variable g).  Sparse CSR format.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    # use LIL for easy columnwise construction, then convert to CSR
    A = sparse.lil_matrix((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a  # column j contributes a shifted copy of a
    return A.tocsr()


def lp_mix_refine(a: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  max_outer_iters: int = 30):
    """
    Repeatedly solve a linear program to find a vector g that maximises
    Σ g_j under the constraint conv(a,g) ≤ M (where M = max(conv(a,a))).
    The solution g is normalised and mixed back into a.
    The mixing factor t is adapted; we only keep the new a if the score
    strictly improves.
    """
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_outer_iters:
        # current bound
        conv = compute_conv(best_a, fft_len)
        M = conv.max()

        # build constraint matrix (this dominates the cost)
        A = build_conv_matrix(best_a)
        b = np.full(2 * n - 1, M)

        # linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)

        # solve
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})

        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # normalise to a probability vector
        g_norm = g / sum_g

        # try a few mixing factors, keep the first improvement
        improved = False
        for t in (0.5, 0.25, 0.125, 0.0625):
            a_candidate = (1.0 - t) * best_a + t * g_norm
            # re‑normalise just in case of numerical drift
            a_candidate = a_candidate / a_candidate.sum()
            score_candidate = fast_score(a_candidate, fft_len, n)
            if score_candidate < best_score - 1e-12:
                best_a = a_candidate
                best_score = score_candidate
                improved = True
                print(f"[LP] iter {it:02d}, t={t:.3f}, new score={best_score:.6f}")
                break
        if not improved:
            # no improvement with any mixing factor → stop LP refinement
            break

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 12000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another.  Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # 1️⃣  Exponentiated‑gradient descent (medium‑length)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.35 * budget_s
    best_a, best_score = exp_gradient_optimize(best_a,
                                               phase1_deadline,
                                               n, fft_len, rng)
    print(f"[exp‑grad] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  LP‑mix refinement (tighten the max‑convolution bound)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.80 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = lp_mix_refine(best_a,
                                           phase2_deadline,
                                           n, fft_len, rng,
                                           max_outer_iters=40)
        print(f"[LP‑mix] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Final greedy hill‑climb (polish)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=25000)
        print(f"[final hill‑climb] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
