"""
Improved candidate generator for the step‑function evaluation problem.

Algorithm overview
------------------
* The evaluation metric is scale‑invariant; we therefore keep the L1‑norm
  of the sequence normalised to sqrt(2·n) (the same normalisation used in
  the reference implementation).
* The uniform (constant) sequence is a known stationary point with a score
  of exactly 2.  To explore the search space we perform a simulated‑annealing
  style local optimisation:
    – Randomly pick two positions i, j and transfer a small amount of mass
      from one to the other while preserving the total sum.
    – After each perturbation we re‑normalise the sequence.
    – The new sequence is accepted if it strictly improves the objective
      (the maximal convolution value) or with a probability depending on
      a temperature schedule (to escape shallow local minima).
* Convolution is computed by FFT which is O(n log n) and fast enough for
  n≈8000.  The maximal convolution entry is the objective (because the
  normalisation makes the full evaluation equal to this value).
* The routine runs for the allotted time (default 1000 s), keeping the
  best sequence seen.  If no improvement is found it simply returns the
  uniform sequence (which is already optimal for many instances).

The function `propose_candidate` follows the required signature and
returns a list of non‑negative floats ready for `evaluate_sequence`.
"""

import time
import numpy as np
from scipy import optimize  # imported for compatibility, not used directly

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _fft_convolution_max(seq: np.ndarray) -> float:
    """
    Return max_{k} (seq * seq)[k] using an FFT based convolution.
    The result is a real non‑negative number.
    """
    n = seq.size
    # Length needed for linear convolution
    conv_len = 2 * n - 1
    # Choose a power‑of‑two size for speed
    size = 1 << ((conv_len - 1).bit_length())
    # Forward FFT (real → complex)
    fft = np.fft.rfft(seq, n=size)
    # Element‑wise product corresponds to linear convolution
    conv = np.fft.irfft(fft * fft, n=size)[:conv_len]
    # Numerical noise may introduce tiny imaginary parts – drop them
    return float(conv.real.max())


def _normalise_to_target(seq: np.ndarray, target_norm: float) -> np.ndarray:
    """
    Scale the sequence so that its L1‑norm equals target_norm.
    The sequence is assumed to be non‑negative.
    """
    s = seq.sum()
    if s == 0.0:
        # Avoid division by zero – put a tiny uniform mass
        seq = np.full_like(seq, target_norm / seq.size)
    else:
        seq = seq * (target_norm / s)
    return seq


def _random_perturb(seq: np.ndarray, step: float, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a new candidate by moving a small amount of mass between two
    randomly chosen positions while keeping the L1‑norm (approximately)
    constant.  `step` is the typical magnitude of the transferred mass.
    """
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq.copy()

    # Proposed transfer (can be positive or negative)
    delta = (rng.random() - 0.5) * step

    new_seq = seq.copy()
    new_seq[i] = max(0.0, new_seq[i] + delta)
    # Adjust the other position to keep the total mass (approximately) unchanged
    new_seq[j] = max(0.0, new_seq[j] - delta)

    # Re‑normalise to the fixed L1‑norm required by the problem
    target_norm = np.sqrt(2.0 * n)
    new_seq = _normalise_to_target(new_seq, target_norm)
    return new_seq

# ----------------------------------------------------------------------

    """

    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)           # type: ignore[name-defined]

        n = 2000

    # Target L1‑norm (the same scaling used by the reference code)
    target_norm = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Start from the uniform (constant) sequence – this already yields a
    # score of exactly 2.0 and is a sensible baseline.
    # ------------------------------------------------------------------
    uniform_val = target_norm / n
    current = np.full(n, uniform_val, dtype=np.float64)

    best_score = _fft_convolution_max(best_seq)
