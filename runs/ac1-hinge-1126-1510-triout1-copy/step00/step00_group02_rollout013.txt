def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n) # maximize sum b
    # Build A_ub and b_ub
    # We'll use list of rows; but better to allocate numpy array.

    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for i in range(i_min, i_max+1):
        #   j = k - i
        #   row[j] = f_seq[i]
        # we can assign more efficiently using slice.
        # Loop within normal
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    # Add non-negativity constraints: -I <= 0
    A_ub = np.vstack(rows + [-np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Solve
    # Use method='highs'
    result = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if result.success:
        return result.x
    else:
        return None

"""
Propose a candidate step‑function sequence.

The algorithm mixes three complementary heuristics:
1.  **LP‑direction descent** – exactly the strategy described in the
    statement: solve a linear program that maximises the total mass
    while keeping the convolution by the current sequence bounded.
    A short line‑search over several mixing weights (t) is performed
    and the best improvement is kept.
2.  **Parametric families** – generate simple one‑parameter families
    (power‑law, exponential, linear) that concentrate mass near the
    beginning of the array.  After normalisation they are evaluated
    and kept when they improve the score.
3.  **Random Dirichlet perturbations** – occasionally create a sparse
    random vector (Dirichlet with a small concentration) and keep it
    if it yields a better bound.

The three heuristics are run repeatedly until the time budget expires.
The best sequence found is returned.  All vectors are kept non‑negative,
their sum is never allowed to drop below 0.01 (the evaluator’s guard)
and all arithmetic is performed with Python floats for compatibility
with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy import optimize

# -------------------------------------------------------------
# Helper: Linear programme finding a direction with larger sum
# -------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """
    Solve:
        maximise   sum(b)
        subject to  conv(f_seq, b)[k] <= rhs   for all k
                    b_i >= 0
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # Objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # Build convolution constraints (2n‑1 rows)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # fill row[j] = f_seq[i] where j = k-i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.array(rows, dtype=np.float64)          # shape (2n‑1, n)
    b_ub = np.full(2 * n - 1, rhs, dtype=np.float64)

    # Bounds for b_i : 0 <= b_i  (no upper bound)
    bounds = [(0.0, None)] * n

    # Use the HiGHS solver (fast and reliable)
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 2.0},
        )
    except Exception:          # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


# -------------------------------------------------------------
# Helper: obtain a better direction using the LP
# -------------------------------------------------------------
def lp_improve(seq, step_sizes):
    """
    Given a current sequence `seq` (list of floats), compute a
    new candidate by solving the LP and mixing with the original
    using the provided `step_sizes`.  The best improvement is
    returned, otherwise None.
    """
    n = len(seq)
    sum_seq = float(np.sum(seq))
    if sum_seq < 1e-12:
        return None

    # Normalise to the canonical sum sqrt(2n) used by the LP formulation
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / sum_seq
    f_norm = np.array(seq, dtype=np.float64) * scale

    # bound on the convolution of the normalised sequence
    rhs = float(np.max(np.convolve(f_norm, f_norm)))

    # Solve the linear programme
    g = solve_convolution_lp(f_norm, rhs)
    if g is None:
        return None

    # Scale g back to the same canonical sum (so that
    #   conv(f_norm,g) <= rhs  still holds)
    sum_g = float(np.sum(g))
    if sum_g == 0.0:
        return None
    g_norm = g * (target_sum / sum_g)

    # Try several mixing coefficients and keep the best.
    best_local = None
    best_score = float("inf")
    for t in step_sizes:
        candidate = (1.0 - t) * np.array(seq) + t * g_norm
        # evaluate – any exception or inf is ignored
        try:
            sc = evaluate_sequence(candidate.tolist())
        except Exception:          # pragma: no cover
            continue
        if sc < best_score:
            best_score = sc
            best_local = candidate.tolist()

    # also try the pure direction (t = 1.0) – occasionally useful
    try:
        sc_full = evaluate_sequence(g_norm.tolist())
        if sc_full < best_score:
            best_score = sc_full
            best_local = g_norm.tolist()
    except Exception:              # pragma: no cover
        pass

    return best_local, best_score


# -------------------------------------------------------------
# Helper: generate simple parametric families
# -------------------------------------------------------------
def generate_power_law(n, exponent):
    """a_i ∝ (n‑i)^{exponent}, normalised to sum 1."""
    if exponent == 0:
        vals = np.ones(n, dtype=np.float64)
    else:
        idx = np.arange(n, 0, -1, dtype=np.float64)      # n, n‑1, … ,1
        vals = np.power(idx, exponent)
    vals /= np.sum(vals)
    return vals.tolist()


def generate_exponential(n, lam):
    """a_i ∝ exp(-lam * i), normalised."""
    idx = np.arange(n, dtype=np.float64)
    vals = np.exp(-lam * idx)
    vals /= np.sum(vals)
    return vals.tolist()


def generate_linear_falloff(n, slope):
    """
    a_i = max(0, 1 - slope * i/(n‑1)), normalised.
    slope ∈ [0,1] gives a decreasing linear profile.
    """
    idx = np.arange(n, dtype=np.float64) / (n - 1)
    vals = np.maximum(0.0, 1.0 - slope * idx)
    if np.sum(vals) == 0.0:
        vals = np.ones(n, dtype=np.float64)
    vals /= np.sum(vals)
    return vals.tolist()
