def project_simplex(v, target_sum=1.0):
    n = len(v)
    # ensure all non-negative possible
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - target_sum
    ind = np.arange(1, n+1)
    cond = v_sorted - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

1. **Greedy mass transfer**
   – Compute the full auto‑convolution `b = a * a`.
   – Find the shift `k_max` where `b` is maximal.
   – For that shift compute the contribution `c_i = a_i·a_{k_max-i}`
     (zero if the partner index is out of range).
   – Transfer a fraction of the mass from the index with the largest
     contribution (`src`) to the index with the smallest contribution
     (`dst`).  The move is accepted if it improves the score (or with a
     tiny probability to escape plateaus).  Afterwards the vector is
     projected onto the simplex (non‑negative, unit sum).

2. **Soft‑max gradient step**
   – Approximate the `max` by a smooth soft‑max:
        L(a) ≈ (1/β)·log Σ_k exp(β·b_k) .
   – Its gradient w.r.t. `a_i` is
        g_i = 2·Σ_k w_k·a_{k-i}   where w_k = soft‑max weights.
   – Move a small step opposite to the gradient and project back onto
     the simplex.

Occasionally a completely random restart is performed.  The best
sequence seen so far is returned.  All operations respect the
non‑negative constraint and the runtime stays well within the 1000 s
budget.

The score evaluated by `evaluate_sequence` is equivalent to
`2·n·max(b)/sum(a)²`; because the algorithm keeps `sum(a)=1` we can
compute it directly without the extra security wrapper.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray, target_sum: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i ≥ 0, Σ x_i = target_sum }.
    Uses the O(n log n) algorithm of Duchi et al.
    """
    if target_sum < 0:
        raise ValueError("target_sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(seq: np.ndarray, n: int) -> float:
    """
    Compute the evaluation score for a given sequence.
    The sequence is assumed to be non‑negative.
    """
    # direct convolution (numpy uses an efficient implementation)
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    s = np.sum(seq)
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    Returns a list of non‑negative heights.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # a safety margin

    # ------------------------------------------------------------------
    # 1. Determine length and initialise the sequence.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)          # safety
        n = base.size
        total = base.sum()
        if total > 0:
            seq = base * (1.0 / total)       # normalise to ℓ₁‑norm = 1
        else:
            seq = np.full(n, 1.0 / len(base))
    else:
        n = 2000
        seq = np.full(n, 1.0 / n, dtype=np.float64)

    # tiny random perturbation to break perfect symmetry
    seq += np.random.uniform(0.0, 1e-6, size=n)
    seq = _project_simplex(seq, 1.0)

    best_seq = seq.copy()
    best_sc = _score(best_seq, n)

    # ------------------------------------------------------------------
    # 2. Hyper‑parameters
    # ------------------------------------------------------------------
    transfer_frac = 0.05        # fraction of source mass moved in greedy step
    beta = 10.0                # softness for the soft‑max approximation
    lr = 0.1                   # learning rate for gradient step
    iter_cnt = 0
    log_interval = 50

    # ------------------------------------------------------------------
    # 3. Main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iter_cnt += 1

        # ----- compute current convolution and score -------------------
        b = np.convolve(seq, seq)
        cur_sc = float(2.0 * n * np.max(b))   # sum(seq) == 1

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_seq = seq.copy()
            if iter_cnt % log_interval == 0:
                print(f"[{iter_cnt}] new best score = {best_sc:.6f}")

        # Choose a move type
        r = np.random.rand()

        # --------------------------------------------------------------
        # 3a. Greedy mass transfer (majority of the time)
        # --------------------------------------------------------------
        if r < 0.60:
            k_max = int(np.argmax(b))

            # contribution of each index to the maximal shift
            idx = np.arange(n)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[mask] = seq[idx[mask]] * seq[partner[mask]]

            src = int(np.argmax(contrib))
            dst = int(np.argmin(contrib))

            if src != dst and seq[src] > 1e-12:
                delta = min(seq[src] * transfer_frac, seq[src])
                new_seq = seq.copy()
                new_seq[src] -= delta
                new_seq[dst] += delta
                # re‑project onto the simplex (removes tiny rounding errors)
                new_seq = _project_simplex(new_seq, 1.0)

                new_sc = _score(new_seq, n)
                # deterministic improvement, tiny random acceptance to jump out
                if new_sc < cur_sc or np.random.rand() < 0.001:
                    seq = new_seq
                    continue

        # --------------------------------------------------------------
        # 3b. Soft‑max gradient descent (occasional)
        # --------------------------------------------------------------
        elif r < 0.85:
            # soft‑max over the convolution values
            max_b = np.max(b)
            exp_weights = np.exp(beta * (b - max_b))
            w = exp_weights / exp_weights.sum()

            # gradient g_i = 2 * Σ_k w_k * a_{k-i}
            grad = np.zeros(n, dtype=np.float64)
            # vectorised implementation using convolution of w with seq
            # Pad seq with zeros on both sides to align indices
            # This loop is O(n²) but n is moderate (≈2000)
            for i in range(n):
                lo = max(i, 0)
                hi = min(i + n - 1, 2 * n - 2)
                ks = np.arange(lo, hi + 1, dtype=np.int64)
                partner = ks - i
                grad[i] = 2.0 * np.sum(w[ks] * seq[partner])
            new_seq = seq - lr * grad
            new_seq = _project_simplex(new_seq, 1.0)

            new_sc = _score(new_seq, n)
            if new_sc < cur_sc:
                seq = new_seq
                continue

        # --------------------------------------------------------------
        # 3c. Small random perturbation (fallback)
        # --------------------------------------------------------------
        else:
            i, j = np.random.choice(n, size=2, replace=False)
            if seq[i] > 1e-12:
                delta = seq[i] * 0.02
                new_seq = seq.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
                new_seq = _project_simplex(new_seq, 1.0)
                new_sc = _score(new_seq, n)
                if new_sc < cur_sc:
                    seq = new_seq
                    continue

        # --------------------------------------------------------------
        # 4. Rare random restart to diversify search
        # --------------------------------------------------------------
        if iter_cnt % 5000 == 0 and np.random.rand() < 0.12:
            rnd = np.random.rand(n)
            rnd = _project_simplex(rnd, 1.0)
            rnd_sc = _score(rnd, n)
            if rnd_sc < best_sc:
                best_sc = rnd_sc
                best_seq = rnd.copy()
                seq = rnd.copy()
                print(f"[{iter_cnt}] random restart improved best to {best_sc:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered
    # ------------------------------------------------------------------
    print(f"Finished after {iter_cnt} iterations. Best score = {best_sc:.6f}")
    return best_seq.tolist()
