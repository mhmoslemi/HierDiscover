import time
import numpy as np
from scipy.optimize import linprog

def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    c = -np.ones(n)      # maximise sum b_j
    # Build inequality matrix for each convolution index k
    A = []
    b = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)
    # Non‑negativity constraints b_j >= 0  →  -b_j <= 0
    A.append(-np.eye(n))
    b.append(np.zeros(n))
    A = np.vstack(A)
    b = np.hstack(b)
    # use a rather tight time limit per LP (a few seconds)
    res = linprog(c, A_ub=A, b_ub=b, method='highs',
                  options={'time_limit': 5.0, 'disp': False})
    if res.success:
        return res.x
    return None

def get_direction(seq, rhs):
    """
    Solve the LP that gives a direction g with larger L1 norm
    while keeping the convolution below the current bound ``rhs``.
    """
    n = len(seq)
    s = np.sum(seq)
    if s == 0:
        return None
    # normalise only for numerical stability – the bound is linear in the
    # variables, so any scaling works.
    norm_seq = seq * (np.sqrt(2 * n) / s)
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    # re‑scale g back to the original scale so that the mixing step does not
    # change the overall scale of the sequence.
    g = g * (s / np.sum(g)) if np.sum(g) > 0 else g
    return g

def random_edge_shape(n, alpha):
    """Return a deterministic edge‑biased shape a_i = ((i+1)*(n-i))**alpha."""
    idx = np.arange(n, dtype=float)
    vals = ((idx + 1) * (n - idx)) ** alpha
    # keep the scale moderate – any global factor is irrelevant for the score
    vals = vals / np.max(vals) * 5.0   # keep numbers comfortably below 1000
    return vals.tolist()

"""
Search algorithm for a non‑negative step function that minimises
`evaluate_sequence`.

The algorithm works with three ideas:

1. **Initial seeds** – a few deterministic families (uniform, a
   parametrised edge‑biased shape) together with the previously discovered
   best sequence (if available) and a few random vectors.

2. **LP direction step** – exactly the method described in the state‑of‑the‑art
   code: given the current peak `M = max(conv(f,f))` we solve a linear program

        maximise   Σ b_j
        subject to conv(f, b) ≤ M ,   b ≥ 0

   The solution `g` can be added to the current sequence without increasing
   the peak.  We mix it with a step size `t` that slowly decays from 0.2 to
   0.01.

3. **Stochastic local move** – pick an index that participates in the
   currently maximal convolution entry and move a tiny amount of weight from
   that index to a random other index.  The move is accepted if the objective
   value does not increase (simulated‑annealing style).

Both phases are interleaved while a time budget (default 1000 s) is left.
The best sequence encountered is returned.  The implementation uses only
NumPy, SciPy and the provided `evaluate_sequence` and respects all
environment constraints.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: fast evaluation (identical to evaluate_sequence but without the
# extra safety checks – we apply the same clamping beforehand).
# ----------------------------------------------------------------------
def _fast_score(arr):
    """Return the value of evaluate_sequence for a pre‑clamped array."""
    n = len(arr)
    conv = np.convolve(arr, arr)
    max_b = np.max(conv)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Linear programme used by the “direction” step.
# ----------------------------------------------------------------------
def _solve_convolution_lp(f_seq, rhs):
    """
    Solve  max Σ b_j  s.t.  conv(f_seq, b) ≤ rhs , b ≥ 0.
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise Σ b_j
    a_rows = []
    b_vals = []

    # one inequality per convolution index (2n‑1 of them)
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_rows.append(row)
        b_vals.append(rhs)

    # non‑negativity constraints: -b ≤ 0
    a_rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))

    A = np.vstack(a_rows)
    b = np.hstack(b_vals)

    # use the HiGHS interior‑point solver (fast for these modest sizes)
    res = linprog(c, A_ub=A, b_ub=b, method='highs',
                  options={'time_limit': 5.0, 'disp': False})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Obtain a direction g that can increase the L1 norm while keeping the
# current convolution peak unchanged.
# ----------------------------------------------------------------------
def _get_direction(seq, current_peak):
    """
    Return a vector g (as a NumPy array) that satisfies
    conv(seq, g) ≤ current_peak and has larger L1 norm than seq.
    Returns None if the LP fails.
    """
    n = len(seq)
    total = np.sum(seq)
    if total == 0.0:
        return None
    # normalise only for numerical stability – the bound is linear,
    # any scaling works.
    norm_seq = seq * (np.sqrt(2.0 * n) / total)
    g = _solve_convolution_lp(norm_seq, current_peak)
    if g is None:
        return None
    # rescale g back to the original total mass of seq
    g_sum = np.sum(g)
    if g_sum == 0.0:
        return None
    g = g * (total / g_sum)
    return g


# ----------------------------------------------------------------------
# Deterministic edge‑biased family of seeds:
#     a_i = ((i+1) * (n-i)) ** alpha   (alpha can be negative)
# ----------------------------------------------------------------------
def _edge_shape(n, alpha):
    """Create a deterministic shape with parameter alpha."""
    idx = np.arange(n, dtype=float)
    vals = ((idx + 1.0) * (n - idx)) ** alpha
    # keep the absolute size modest; scaling does not affect the score.
    vals = vals / np.max(vals) * 5.0
    return vals.tolist()


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.
    Returns a list of non‑negative floats (the best sequence found
    before the time budget expires).
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Determine the length to work with – reuse the historic best if
    # it exists, otherwise fall back to a reasonable default.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # pylint: disable=undefined-variable
    except Exception:
        n = 1500

    # small safety margin so we finish cleanly
    deadline = time.time() + budget_s - 5.0

    # --------------------------------------------------------------
    # Assemble a collection of diverse starting vectors.
    # --------------------------------------------------------------
    seeds = []

    # a) already known construction (if present)
    try:
        seeds.append(list(height_sequence_1))    # pylint: disable=undefined-variable
    except Exception:
        pass

    # b) pure uniform vector – the analytical baseline (score = 2.0)
    seeds.append([1.0] * n)

    # c) edge‑biased shapes with a few alphas
    for a in (-0.5, -0.3, -0.2, -0.1, 0.0, 0.1):
        seeds.append(_edge_shape(n, a))

    # d) a handful of random vectors (scaled to keep numbers modest)
    for _ in range(3):
        rnd = np.random.rand(n)
        rnd = rnd / np.sum(rnd) * 5.0
        seeds.append(rnd.tolist())

    # --------------------------------------------------------------
    # Evaluate all seeds and remember the best one.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    for s in seeds:
        arr = np.maximum(0.0, np.minimum(1000.0, np.array(s, dtype=float)))
        sc = _fast_score(arr)
        if sc < best_score:
            best_score = sc
            best_seq = arr.tolist()
    print(f"[init] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Core optimisation loop.
    # --------------------------------------------------------------
    cur = np.array(best_seq, dtype=float)
    cur_score = best_score

    # mixing parameter – linearly decays from 0.2 → 0.01 over the whole run
    def _mix_coeff(elapsed):
        total = budget_s - 5.0
        return 0.2 - 0.19 * min(elapsed / total, 1.0)

    # restrict the number of costly LP solves
    lp_calls = 0
    max_lp_calls = 30

    while time.time() < deadline:
        elapsed = time.time() - (deadline - (budget_s - 5.0))

        # ----- current convolution and maximal index -----
        conv = np.convolve(cur, cur)
        peak = np.max(conv)
        peak_idx = int(np.argmax(conv))

        # ------------------------------------------------------
        # 1) Occasionally request an LP direction (probability 0.2,
        #    limited by max_lp_calls).
        # ------------------------------------------------------
        if lp_calls < max_lp_calls and random.random() < 0.2:
            direction = _get_direction(cur, peak)
            if direction is not None:
                lp_calls += 1
                t = _mix_coeff(elapsed)
                cand = (1.0 - t) * cur + t * direction
                cand = np.maximum(0.0, np.minimum(1000.0, cand))
                cand_score = _fast_score(cand)
                if cand_score < cur_score:
                    cur = cand
                    cur_score = cand_score
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand.tolist()
                    print(f"[LP] t={t:.3f}  iter≈{elapsed:.1f}s  new best {best_score:.6f}")
                continue   # skip the simple local move this iteration

        # ------------------------------------------------------
        # 2) Stochastic local transfer of a tiny amount of mass.
        # ------------------------------------------------------
        # Find a random index that participates in the current peak.
        candidates = [i for i in range(n) if 0 <= peak_idx - i < n]
        if not candidates:
            continue
        i = random.choice(candidates)
        j = peak_idx - i

        # choose a destination different from i and j
        dest = random.randrange(n)
        while dest == i or dest == j:
            dest = random.randrange(n)

        # amount to move – 1 % of the source weight (but never exceeding it)
        eps = 0.01 * cur[i]
        if eps <= 0.0:
            continue
        cand = cur.copy()
        transfer = min(eps, cand[i])
        cand[i] -= transfer
        cand[dest] += transfer
        cand = np.maximum(0.0, np.minimum(1000.0, cand))
        cand_score = _fast_score(cand)

        # accept if we do not get worse (simple simulated‑annealing)
        if cand_score <= cur_score:
            cur = cand
            cur_score = cand_score
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.tolist()
            print(f"[local] iter≈{elapsed:.1f}s  new best {best_score:.6f}")

    print(f"[done] best score = {best_score:.6f}")
    # return a plain Python list (evaluate_sequence expects a list)
    return best_seq
