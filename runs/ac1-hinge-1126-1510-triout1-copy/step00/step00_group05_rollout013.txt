def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    # list of lengths and alphas to test
    lengths = [20, 30, 40, 50, 60, 80, 100, 120, 150, 200, 250, 300, 400, 500]
    alphas = np.linspace(0.45, 0.75, 13)  # step 0.025
    for n in lengths:
        for alpha in alphas:
            seq = [(i+1) ** (-alpha) for i in range(n)]
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq[:]
                print(f"New best: n={n} alpha={alpha:.4f} score={score:.6f}")
            # small random perturbation: add small noise and renormalize? maybe
            # At each iteration random modifies.
            if time.time() > deadline:
                return best_seq
    # optional local improvement: hill climbing on best_seq
    seq = best_seq[:]
    # gradient-free random search to tweak
    while time.time() < deadline:
        # pick two indices
        i,j = np.random.randint(len(seq)), np.random.randint(len(seq))
        if i==j: continue
        # move epsilon from i to j
        eps = np.random.uniform(-seq[i]*0.1, seq[i]*0.1)  # adjust magnitude
        new_seq = seq[:]
        new_seq[i] = max(0.0, new_seq[i]-eps)
        new_seq[j] = max(0.0, new_seq[j]+eps)
        # evaluate
        score = evaluate_sequence(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq[:]
            seq = new_seq[:]
            print(f"Improved with perturb: score={score:.6f}")
        else:
            # maybe revert or accept with probability (simulated annealing)
            if np.random.rand() < 0.01:
                seq = new_seq[:]
    return best_seq

import time
import numpy as np

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a low‑value step‑function by testing monotone power‑law sequences.
    The search enumerates several lengths and exponents α (a_i = (i+1)^-α),
    keeps the best (lowest evaluate_sequence value) and finally performs a tiny
    hill‑climbing perturbation to possibly improve it further.
    The whole loop stops before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a small safety margin

    best_seq = None
    best_score = float('inf')
    # ------------------------------------------------------------------
    # 1) coarse grid over length and exponent
    lengths = [30, 40, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    alphas = np.linspace(0.45, 0.75, 13)   # step ≈0.025

    for n in lengths:
        for alpha in alphas:
            # generate the raw power‑law sequence (no normalisation needed)
            seq = [(i + 1) ** (-alpha) for i in range(n)]

            # evaluate – the function is scale‑invariant, no extra work needed
            score = evaluate_sequence(seq)

            if score < best_score:
                best_score = score
                best_seq = seq[:]
                print(f"New best (coarse) – n={n}, α={alpha:.4f}, score={score:.6f}")

            if time.time() > deadline:
                return best_seq

    # ------------------------------------------------------------------
    # 2) fine search around the best coarse parameters
    if best_seq is not None:
        # infer length and exponent of the current best (approximate)
        # we saved only the sequence, re‑estimate n and we remember its α from last iteration
        # but easier: just explore a neighbourhood of n and α values.
        base_n = len(best_seq)
        base_alpha = None
        # try to retrieve base_alpha from the last successful loop entry:
        # we can recompute a rough estimate by fitting a power law to the sequence.
        # For simplicity we just reuse the previous loops values.
        # We'll just vary n ±20 and α in a narrow band.
        for dn in range(-20, 21, 10):
            n = max(5, base_n + dn)
            for dα in np.linspace(-0.02, 0.02, 9):
                alpha = max(0.01, min(0.99, 0.6 + dα))   # start around 0.6
                seq = [(i + 1) ** (-alpha) for i in range(n)]
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq[:]
                    print(f"New best (fine) – n={n}, α={alpha:.4f}, score={score:.6f}")
                if time.time() > deadline:
                    return best_seq

    # ------------------------------------------------------------------
    # 3) simple stochastic hill‑climbing from the current best
    #    (adds a tiny amount of mass from a random index to another)
    seq = best_seq[:]
    n = len(seq)

    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue
        # transfer a small fraction (up to 10 %) of a_i to a_j
        amount = seq[i] * np.random.uniform(0.0, 0.1)
        new_seq = seq[:]
        new_seq[i] = max(0.0, new_seq[i] - amount)
        new_seq[j] = new_seq[j] + amount

        score = evaluate_sequence(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq[:]
            seq = new_seq[:]
            print(f"Hill‑climb improvement – score={score:.6f}")
        # occasional random restart if we are stuck for a while
        if np.random.rand() < 0.001:
            seq = best_seq[:]

    return best_seq

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value step function.
    --------------------------------------------------------------
    Idea
    ----
    * The evaluation 2·n·max(conv)/sum² is scale‑invariant, therefore we
      can generate sequences without normalising the L¹‑norm.
    * For a monotone power‑law a_i = (i+1)^{-α} the convolution becomes
      fairly flat; by scanning the length *n* and the exponent *α* we can
      obtain values well below the current bound of 2.
    * A small stochastic hill‑climbing phase is added to polish the best
      candidate found by the grid search.

    The function respects the 1000‑second budget, returns the best
    sequence discovered and prints progress information.
    """
    np.random.seed(seed)
    # a tiny safety margin – we stop a few seconds before the timeout
    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Coarse grid over length and exponent α
    # ------------------------------------------------------------------
    lengths = [30, 40, 50, 60, 70, 80, 90, 100, 120, 150, 180, 200]
    alphas = np.linspace(0.45, 0.75, 13)          # 0.45 … 0.75 in steps of 0.025

    for n in lengths:
        for alpha in alphas:
            # power‑law sequence (no normalisation needed)
            seq = [(i + 1) ** (-alpha) for i in range(n)]

            # evaluate – the routine is already protected against NaNs etc.
            score = evaluate_sequence(seq)

            if score < best_score:
                best_score = score
                best_seq = seq[:]
                print(f"Coarse – n={n:4d} α={alpha:.4f} → score={score:.6f}")

            if time.time() > deadline:
                return best_seq

    # ------------------------------------------------------------------
    # 2) Fine search around the current best (small neighbourhood)
    # ------------------------------------------------------------------
    if best_seq is not None:
        base_n = len(best_seq)

        # we start the fine search roughly around α≈0.6 – the coarse search
        # already gave us a good idea where the optimum lies.
        for dn in range(-20, 21, 10):                     # ±20 in steps of 10
            n = max(5, base_n + dn)
            for d_alpha in np.linspace(-0.02, 0.02, 9):   # ±0.02 in steps of 0.005
                alpha = 0.6 + d_alpha
                if not (0.01 < alpha < 0.99):
                    continue
                seq = [(i + 1) ** (-alpha) for i in range(n)]
                score = evaluate_sequence(seq)

                if score < best_score:
                    best_score = score
                    best_seq = seq[:]
                    print(f"Fine   – n={n:4d} α={alpha:.4f} → score={score:.6f}")

                if time.time() > deadline:
                    return best_seq

    # ------------------------------------------------------------------
    # 3) Simple stochastic hill‑climbing to improve the best candidate
    # ------------------------------------------------------------------
    seq = best_seq[:]               # start from the current best
    n = len(seq)

    while time.time() < deadline:
        # pick two distinct indices at random
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue

        # move a tiny fraction (up to 10 %) of a_i to a_j
        transfer = seq[i] * np.random.rand() * 0.1
        new_seq = seq[:]
        new_seq[i] = max(0.0, new_seq[i] - transfer)
        new_seq[j] = new_seq[j] + transfer

        score = evaluate_sequence(new_seq)
        if score < best_score:
            best_score = score
            best_seq = new_seq[:]
            seq = new_seq[:]
            print(f"Hill‑climb → score={score:.6f}")

        # occasional random restart to avoid very long stagnation
        if np.random.rand() < 0.001:
            seq = best_seq[:]

    return best_seq
