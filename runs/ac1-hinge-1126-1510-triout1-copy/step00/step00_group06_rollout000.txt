"""
Propose a very low‑score sequence for the given evaluation function.

The evaluation is
    score = 2 * n * max(conv(a, a)) / (sum(a)^2)

For a non‑negative sequence `a` the maximum of the linear convolution
occurs either at the first element (a[0]**2) or at the centre (where
the whole sequence overlaps with its reverse).  If we choose a
geometrically decreasing sequence

    a_i = r**i ,      i = 0 … n‑1  ,   0 < r < 1

then the centre value is

    conv_center = n * r**(n‑1)

and the first element value is `a_0**2 = 1`.  By selecting r so that
`n * r**(n‑1) = 1` we make the two possible maxima equal to 1, i.e.
`max(conv) = 1`.  With this choice the score simplifies to

    score = 2 * n / (sum(a)^2)  ≈  2 (ln n)^2 / n

which tends to 0 as n grows.  For a modest length (e.g. n≈8000) the
score is already < 0.03, far below the target 1.503.

The function below builds exactly such a sequence.  The length `n` is
chosen as large as is comfortable for a single convolution in the
available time budget, but can be increased if more runtime is
available.
"""

import time
import numpy as np

# The evaluation function is provided in the execution environment.
# from <environment> import evaluate_sequence


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Generate a geometric decreasing sequence whose convolution max is 1.
    The length is chosen to give a very small score while staying well
    within the time budget.

    Parameters
    ----------
    seed : int
        Random seed (unused, kept for compatibility).
    budget_s : float
        Time budget in seconds (the function respects this trivially).

    Returns
    -------
    list[float]
        The candidate sequence.
    """
    # The algorithm is deterministic – the seed is kept only for the
    # required signature.
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Choose length n.
    # A few thousand entries already give a score ~0.02.
    # We use 8000 which comfortably fits into memory and finishes the
    # convolution in far less than a second.
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Compute the decay factor r so that the centre of the convolution
    # equals 1:   n * r**(n-1) = 1   =>   r = exp(-log(n) / (n-1))
    # ------------------------------------------------------------------
    r = np.exp(-np.log(n) / (n - 1))

    # Build the sequence a_i = r**i.
    seq = [r ** i for i in range(n)]

    # ------------------------------------------------------------------
    # (Optional) sanity check – compute the score once for logging.
    # ------------------------------------------------------------------
    try:
        score = evaluate_sequence(seq)
        print(
            f"[propose_candidate] n={n}, r={r:.6f}, "
            f"max(conv)=~{max(np.convolve(seq, seq)):.6f}, score={score:.6f}"
        )
    except Exception as e:
        # In the unlikely event something goes wrong we still return the
        # sequence; the outer harness will evaluate it.
        print(f"[propose_candidate] Evaluation failed: {e}")

    return seq
