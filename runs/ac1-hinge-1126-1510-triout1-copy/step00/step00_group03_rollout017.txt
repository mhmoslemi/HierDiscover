"""
Simulated‑annealing / hill‑climbing search for a step‑function that minimizes
evaluate_sequence.  The objective for a normalized sequence (sum = 1) is
    eval = 2 * n * max(conv(a,a)),
so we try to reduce the peak of the self‑convolution while keeping the
coefficients non‑negative and their sum fixed.

The algorithm works on a moderate length (default n = 800) which makes a
full convolution cheap (~10⁻⁴ s).  It repeatedly moves a small amount of mass
from an index that contributes heavily to the current maximal convolution
value to a random other index.  Moves are accepted if they improve the score
or with a probability that slowly cools (simulated annealing).  The best
sequence seen during the allotted time is returned.  If a previously found
construction ``height_sequence_1`` is available it is used as a warm start,
otherwise a random Dirichlet sample is taken.

The routine respects all the security checks of ``evaluate_sequence`` – the
sequence contains only non‑negative floats and values are far below the
built‑in caps (1000.0).  It finishes well before the 1000 s limit and usually
produces a score below the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a (already non‑negative) array.
    The sum of a is assumed to be > 0.
    """
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    b = np.convolve(a, a)
    return 2.0 * n * b.max() / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a coefficient sequence with a low value of evaluate_sequence.
    The function runs for up to ``budget_s`` seconds and returns the best
    sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)

    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Choose a comfortable length.  A few hundred elements give a good
    # trade‑off between speed and the theoretical lower bound 2n/(2n-1)≈1.
    try:
        n = len(height_sequence_1)          # use the previous construction if present
    except Exception:
        n = 800                             # fall‑back length

    # Initialise a random Dirichlet vector (sum = 1) – a neutral start.
    a = np.random.dirichlet(np.ones(n))
    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score

    # Parameters for the annealing / hill‑climbing schedule.
    temperature = 0.02       # initial temperature for probabilistic acceptance
    step_factor = 0.12       # fraction of the source value that is moved
    iteration = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) compute current convolution and the index(s) where the peak lies
        conv = np.convolve(a, a)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val / (a.sum() ** 2)   # same as _score(a)
        # All indices attaining the maximal value (typically one, but we keep all)
        peak_idxs = np.where(conv >= max_val * (1.0 - 1e-12))[0]
        k = int(np.random.choice(peak_idxs))

        # --------------------------------------------------------------
        # 2) propose a move.
        #    With high probability we move mass away from a coordinate that
        #    participates in the current peak; otherwise we perform a small
        #    random perturbation (helps escape plateaus).
        if np.random.rand() < 0.85:
            # indices that actually appear in the sum defining conv[k]
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            candidates = np.arange(i_low, i_high + 1, dtype=int)

            # contribution of each candidate to the peak (a_i * a_{k-i})
            contrib = a[candidates] * a[k - candidates]
            # pick the index with the largest contribution
            i = int(candidates[np.argmax(contrib)])

            # pick a destination index different from i
            j = int(np.random.choice(np.delete(np.arange(n), i)))
            delta = step_factor * a[i]

            # ensure we never make a[i] negative
            if delta > a[i]:
                delta = a[i] * 0.5
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            # small random jitter – pick a few coordinates and re‑randomise them
            a_new = a.copy()
            idx = np.random.choice(n, size=3, replace=False)
            a_new[idx] = np.random.rand(3)
            # re‑normalise to keep the sum = 1
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()

        # --------------------------------------------------------------
        # 3) evaluate the proposal
        new_score = _score(a_new)

        # --------------------------------------------------------------
        # 4) accept or reject the move
        improved = new_score < cur_score
        if improved or np.random.rand() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
            a = a_new
            cur_score = new_score

        # keep the global best
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Iter {iteration:6d}] New best score: {best_score:.6f}")

        # --------------------------------------------------------------
        # 5) schedule – slowly cool down and shrink the step size
        if iteration % 500 == 0:
            temperature *= 0.94
            step_factor *= 0.97
            # occasional random restart if we seem stuck
            if iteration % 5000 == 0 and best_score > 1.7:
                a = np.random.dirichlet(np.ones(n))
                cur_score = _score(a)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain list of floats)
    # The values already respect the non‑negative constraint.
    print(f"Finished search – best score obtained: {best_score:.6f}")
    return best_a.tolist()
