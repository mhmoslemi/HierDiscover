def generate_sidon_set(N):
    chosen = []
    sums = set()
    for p in range(N):
        ok = True
        for q in chosen:
            s = p+q
            if s in sums:
                ok = False
                break
        # also check double sum p+p
        if p+p in sums:
            ok = False
        if ok:
            for q in chosen:
                sums.add(p+q)
                sums.add(q+p)  # same
            sums.add(p+p)
            chosen.append(p)
        if len(chosen) >= int(np.sqrt(N))+5: # stop early maybe
            break
    return chosen

"""
Propose a step‑function (non‑negative list of heights) that minimises
evaluate_sequence().

The idea is to combine a few very large values (set to the allowed maximum
1000) with many small values.  The large values are placed on a *Sidon*
set of indices – a set whose pairwise sums are all distinct – so that any
convolution entry receives at most one contribution that is 1000².
All remaining indices receive a modest constant ε (e.g. 5).  This makes
the L∞‑norm of the self‑convolution essentially the square of the large
value, while the total L¹‑mass is boosted by the many small entries,
driving the evaluation function
    2·n·max(conv) / (sum)²
well below the previous bound 2.0.

The routine explores a few choices of:
    * total length n (based on the known construction length)
    * the large value L (≤1000)
    * the small value ε  (>0)
and keeps the best sequence found within the time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a Sidon set (all pairwise sums distinct)
# ----------------------------------------------------------------------
def generate_sidon_set(N, rng):
    """
    Greedy construction of a Sidon set of indices in {0,…,N‑1}.
    The order of candidates is shuffled using the supplied rng to obtain
    different sets on different runs.
    """
    candidates = list(range(N))
    rng.shuffle(candidates)

    chosen = []
    sums = set()                # all i+j for i,j in chosen (including i=j)

    for i in candidates:
        # check whether i+i collides
        if i + i in sums:
            continue
        # check collisions with already chosen indices
        bad = False
        for p in chosen:
            if i + p in sums:
                bad = True
                break
        if bad:
            continue

        # accept i
        for p in chosen:
            sums.add(i + p)
            sums.add(p + i)     # redundant but explicit
        sums.add(i + i)
        chosen.append(i)

    return chosen
