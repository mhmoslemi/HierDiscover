"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm exploits the known optimal continuous shape: the *arcsine*
distribution.  For a discrete sequence of length *n* we use

    a_i ∝ 1 / sqrt(t_i * (1-t_i)),   t_i = (i+0.5) / n .

This makes the self‑convolution almost constant, i.e. the numerator is close
to its average value, which drives the evaluation score close to
2·n/(2·n‑1) ≈ 1 for large *n*.  

A small stochastic search around this family (random lengths and
exponents of a Beta density) is performed while the time budget permits.
The best sequence found is returned.
"""

import time
import random
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Return a discretised Beta(α,β) density on n points.
    Points are taken at the centre of each bin to avoid infinities at the
    interval boundaries.
    The raw values are *not* normalised; they are only used up to a common
    scaling factor (the evaluation function is scale‑invariant).
    """
    # centre points of the bins (0,1)
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    # compute log‑pdf for numerical stability
    log_pdf = (alpha - 1) * np.log(t) + (beta - 1) * np.log1p(-t)
    seq = np.exp(log_pdf)
    # guard against possible under‑/overflow (replace NaNs by 0)
    seq = np.where(np.isfinite(seq), seq, 0.0)
    return seq


def arcsine_sequence(n: int) -> np.ndarray:
    """
    Special case of the Beta family with α = β = 0.5 (the arcsine law).
    a_i ∝ 1 / sqrt(t_i * (1-t_i)).
    """
    t = (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = 1.0 / np.sqrt(t * (1.0 - t))
    return seq


def scaled_to_unit(seq: np.ndarray, target_max: float = 0.9) -> np.ndarray:
    """
    Scale the whole sequence so that its maximum equals *target_max*.
    This avoids the clipping performed inside ``evaluate_sequence``.
    """
    max_val = np.max(seq)
    if max_val == 0:
        return seq
    return seq * (target_max / max_val)


def evaluate(seq: np.ndarray) -> float:
    """
    Small wrapper around the provided ``evaluate_sequence`` that accepts a
    NumPy array.
    """
    # ``evaluate_sequence`` works on plain Python lists
    return evaluate_sequence(seq.tolist())

            b1 = max(0.1, b0 * (1.0 + random.uniform(-0.2, 0.2)))
            n1 = max(50, min(8000, int(n * (1.0 + random.uniform(-0.1, 0.1)))))
            seq_ref = beta_sequence(n1, a1, b1)
            seq_ref = scaled_to_unit(seq_ref)
            try:
                score_ref = evaluate(seq_ref)
                if score_ref < best_score:
                    best_score = score_ref
                    best_seq = seq_ref.tolist()
                    best_params = (n1, a1, b1)
                    print(f"[Refine] n={n1} α={a1:.3f} β={b1:.3f}  score={score_ref:.6f}")
            except Exception:
                pass

    # ------------------------------------------------------------------
    # Final reporting
    # ------------------------------------------------------------------
    if best_seq is None:
        # Fallback: a simple uniform sequence (never reached in practice)
        n = 1000
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print("[Fallback] uniform sequence used")

    print(f"[Result] best score = {best_score:.6f}  (length={len(best_seq)})")
    return best_seq
