def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # choose length
    n = 500   # can increase later

    S = np.sqrt(2 * n)
    # initial sequence - random Dirichlet
    seq = np.random.dirichlet(np.ones(n)) * S
    # optionally use uniform as baseline
    seq = np.full(n, S / n)   # uniform

    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    curr_seq = seq.copy()
    curr_score = best_score

    # SA parameters
    T0 = 1.0
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 0.5)
        T = T0 * (1 - progress) + 0.001  # linear decay

        # propose move: choose two indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random delta fraction of the larger value
        if curr_seq[i] > 0:
            delta = (np.random.rand() - 0.5) * 0.5 * curr_seq[i]  # [-0.25,0.25] * a_i
        else:
            continue
        # ensure non-negative after move
        new_i = curr_seq[i] - delta
        new_j = curr_seq[j] + delta
        if new_i < 0:
            continue
        # create new sequence
        new_seq = curr_seq.copy()
        new_seq[i] = new_i
        new_seq[j] = new_j

        new_score = evaluate_sequence(new_seq.tolist())
        # accept if better or with probability
        if new_score < curr_score or np.random.rand() < np.exp((curr_score - new_score) / T):
            curr_seq = new_seq
            curr_score = new_score
            if new_score < best_score:
                best_seq = new_seq.copy()
                best_score = new_score
                print(f'[{time.time()-start:.1f}s] New best {best_score}')
    # return best_seq as list
    return best_seq.tolist()

    a = np.array(seq, dtype=float)
    n = len(a)

    # normalize sum
    a *= S_target / np.sum(a)
    best_a = a.copy()
    best_score = evaluate_sequence(a.tolist())
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        k = np.argmax(conv)
        max_val = conv[k]
        # compute gradient
        g = np.zeros_like(a)
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            j = k - i
            if i == j:
                g[i] += a[i]  # derivative of a_i*a_i
            else:
                g[i] += a[j]
                g[j] += a[i]
        # step size
        # compute max possible step to keep nonnegative
        positive = g > 1e-12
        if np.any(positive):
            max_step = np.min(a[positive] / g[positive])
        else:
            max_step = 1.0
        step = 0.5 * max_step  # use half
        # apply
        a_new = a - step * g
        # ensure non-negative (should be)
        a_new = np.maximum(a_new, 0.0)
        # renormalize sum to S_target
        sum_a = np.sum(a_new)
        if sum_a < 1e-12:
            continue
        a_new *= S_target / sum_a
        new_score = evaluate_sequence(a_new.tolist())
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        # accept if improves; else maybe keep original
        if new_score <= evaluate_sequence(a.tolist()):
            a = a_new
        # optional decreasing step size etc.
    return best_a

    for beta in betas:

        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f'Beta {beta:.2f} new best {score:.6f}')
        if time.time() > deadline: break
    # exponentials
    rs = np.linspace(0.90, 0.99, 21)
    for r in rs:
        a = (r ** np.arange(n))
        a = a / np.sum(a) * S
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f'r {r:.3f} new best {score:.6f}')
        if time.time() > deadline: break
    # also random Dirichlet
    for _ in range(20):
        a = np.random.dirichlet(np.ones(n))
        a = a * S
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f'Random start new best {score:.6f}')
        if time.time() > deadline: break
    # local optimization (hill climbing)
    if best_seq is None:
        raise RuntimeError("Failed to generate initial sequence")
    # hill climbing loops
    a_curr = best_seq.copy()
    curr_score = best_score
    while time.time() < deadline:
        # propose a move
        i, j = np.random.randint(n, size=2)
        if i == j: continue
        # amount to transfer
        delta = (np.random.rand() - 0.5) * 0.02 * a_curr[i]  # up to 1% of a_i
        new_i = a_curr[i] - delta
        new_j = a_curr[j] + delta
        if new_i < 0: continue
        a_new = a_curr.copy()
        a_new[i] = new_i
        a_new[j] = new_j
        # optional renormalization to keep sum = S
        s = np.sum(a_new)
        if s <= 0: continue
        a_new *= (S / s)
        new_score = evaluate_sequence(a_new.tolist())
        if new_score < curr_score or np.random.rand() < 0.01:
            a_curr = a_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                print(f'Improved during hill {best_score:.6f} at {time.time()-start:.1f}s')
    return best_seq.tolist()

"""
Propose a good step‑function (non‑negative sequence) that minimizes
the evaluation function

    2 * n * max(convolve(a, a)) / (sum(a) ** 2)

The algorithm proceeds in two phases:

1. **Parameter sweep** – generate a handful of deterministic families
   (power‑law and exponential decays) and a few random Dirichlet
   vectors.  Each candidate is normalised so that its ℓ¹‑norm equals
   sqrt(2·n); the best one (lowest score) is kept.

2. **Simple stochastic hill‑climbing** – starting from the best
   candidate, repeatedly move a small amount of mass from one index to
   another (preserving non‑negativity).  After each move the sequence
   is re‑normalised to keep the sum fixed, the new score is evaluated,
   and the move is accepted if it improves the score (or with a tiny
   probability to escape plateaus).  The best sequence ever seen is
   returned.

The search runs for the allotted wall‑time (default 1000 s) and prints
progress updates.  No external files or network access are used.
"""

import time
import numpy as np

def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that yields a low
    evaluation score.  The function respects the time budget; when the
    budget is exhausted it returns the best sequence found so far.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Total time budget in seconds (default 1000 seconds).

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    # -------------------------------------------------
    # 0. basic setup
    # -------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin so we finish cleanly
    deadline = start_time + budget_s - 0.5

    # problem size – a few hundred entries give a good speed/quality trade‑off
    n = 800
    target_sum = np.sqrt(2.0 * n)          # we keep the ℓ¹‑norm fixed to this value

    best_seq = None
    best_score = float("inf")

    # -------------------------------------------------
    # 1. parameter sweep (deterministic families + few random starts)
    # -------------------------------------------------
    # ---- power‑law families a_i ∝ (i+1)^(-beta) -----------------
    betas = np.linspace(0.0, 2.0, 21)      # 0.0 (uniform) … 2.0
    for beta in betas:
        if time.time() >= deadline:
            break
        idx = np.arange(1, n + 1, dtype=float)
        a = idx ** (-beta)
        a = a * (target_sum / np.sum(a))   # normalise ℓ¹‑norm
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}s] beta={beta:.2f} → new best {score:.6f}")

    # ---- exponential families a_i ∝ r^i -------------------------
    rs = np.linspace(0.90, 0.99, 21)
    for r in rs:
        if time.time() >= deadline:
            break
        a = (r ** np.arange(n, dtype=float))
        a = a * (target_sum / np.sum(a))
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}s] r={r:.3f} → new best {score:.6f}")

    # ---- a few random Dirichlet vectors -------------------------
    for _ in range(30):
        if time.time() >= deadline:
            break
        a = np.random.dirichlet(np.ones(n))
        a = a * target_sum
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[{time.time() - start_time:.1f}s] random start → new best {score:.6f}")

    # safety: if nothing was generated (should never happen) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print("[fallback] using uniform sequence")

    # -------------------------------------------------
    # 2. stochastic hill‑climbing from the best candidate
    # -------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score

    # small helper: re‑normalise a sequence to the target ℓ¹‑norm
    def renorm(x):
        s = np.sum(x)
        if s == 0:
            return x
        return x * (target_sum / s)

    # main optimisation loop
    while time.time() < deadline:
        # pick two distinct indices at random
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # propose moving a tiny fraction of mass from i to j
        # the fraction is chosen uniformly in [-0.01, +0.01] of a_i
        delta = (np.random.rand() - 0.5) * 0.02 * cur_seq[i]
        new_i = cur_seq[i] - delta
        new_j = cur_seq[j] + delta
        if new_i < 0:
            continue          # reject moves that would make an entry negative

        # build the candidate sequence
        cand = cur_seq.copy()
        cand[i] = new_i
        cand[j] = new_j
        cand = renorm(cand)

        # evaluate
        cand_score = evaluate_sequence(cand.tolist())
        # accept if better, otherwise accept with a tiny probability
        if cand_score < cur_score or np.random.rand() < 0.005:
            cur_seq = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{time.time() - start_time:.1f}s] improved to {best_score:.6f}")

    # -------------------------------------------------
    # Return the best sequence found
    # -------------------------------------------------
    return best_seq.tolist()
