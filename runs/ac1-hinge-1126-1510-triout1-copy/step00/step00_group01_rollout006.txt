"""
Propose a candidate step‑function sequence.

The objective is to minimise
    2 * n * max(conv(a,a)) / (sum(a)^2)

where `conv` is the linear convolution of the non‑negative sequence `a`.
A uniform sequence gives the value 2, but random Dirichlet draws already
produce much smaller values because the convolution values concentrate
around their mean (≈ (sum a)^2/(2n)).  The algorithm therefore consists of
two phases:

1. **Random sampling** – draw many Dirichlet vectors, normalise them so
   that `sum(a) = sqrt(2n)`, and keep the best one.
2. **Local refinement** – starting from the best sample, run a simple
   simulated‑annealing / hill‑climbing loop that moves a small amount of
   mass from one index to another.  Moves that reduce the evaluation are
   always accepted; worse moves are accepted with a temperature‑dependent
   probability to escape local minima.

All calculations are done with a fast hand‑written evaluator that mirrors
`evaluate_sequence`.  The final sequence is returned as a plain Python
list of floats; the external grader will call `evaluate_sequence` on it.
"""

import time
import numpy as np

def _fast_evaluation(a: np.ndarray, n: int) -> float:
    """Return the same quantity that evaluate_sequence computes (without checks)."""
    conv = np.convolve(a, a)
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is guaranteed > 0.01 by the construction below.
    return 2.0 * n * max_b / (sum_a * sum_a)


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Generate a sequence of non‑negative heights that (hopefully) beats the
    current bound of 2.0.  The function respects the 1000 s time limit
    and returns the best sequence found.

    Args:
        seed: random seed for reproducibility.
        budget_s: total time budget in seconds.
        **kwargs: unused – kept for compatibility with the original template.

    Returns:
        A list of floats representing the heights of the step function.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length of the sequence.
    # Prefer the length of the previously discovered construction if
    # it exists; otherwise fall back to a reasonably large default.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)          # type: ignore
    else:
        n = 1500

    # The sum is fixed to sqrt(2n); scaling does not change the score.
    sum_target = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Phase 1 – Random Dirichlet sampling.
    # ------------------------------------------------------------------
    best_a = np.full(n, sum_target / n, dtype=np.float64)
    best_score = _fast_evaluation(best_a, n)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0            # keep a safety margin

    # Allocate roughly half of the time to pure random sampling.
    random_phase_deadline = start_time + 0.45 * (deadline - start_time)

    while time.time() < random_phase_deadline:
        # Choose a concentration parameter for the Dirichlet.  Small values
        # produce sparse vectors (few large entries), large values give
        # nearly uniform vectors.  Sampling across a range explores both
        # extremes.
        alpha = rng.uniform(0.2, 2.0)

        # Gamma -> Dirichlet sampling (vectorised, fast).
        g = rng.gamma(shape=alpha, scale=1.0, size=n)
        if g.sum() == 0.0:               # extremely unlikely, but guard against div‑by‑zero
            continue
        a = g * (sum_target / g.sum())

        score = _fast_evaluation(a, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[random] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Simple simulated‑annealing refinement.
    # ------------------------------------------------------------------
    # Start the local search from the best random sample.
    a = best_a.copy()
    current_score = best_score

    # Temperature schedule: linear from T_start to T_end.
    T_start = 1.0
    T_end = 1e-4

    iter_count = 0
    while time.time() < deadline:
        # Linear temperature decay
        frac = (time.time() - start_time) / (deadline - start_time)
        temperature = T_start * (1.0 - frac) + T_end * frac

        # ------------------------------------------------------------------
        # Propose a move: transfer a fraction of mass from i to j.
        # ------------------------------------------------------------------
        i = rng.integers(n)
        if a[i] <= 0.0:
            continue                     # cannot take mass from a zero entry

        # ensure j != i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # Transfer at most half of a[i]; the fraction itself is random.
        frac_move = rng.random() * 0.5
        delta = a[i] * frac_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # Numerical drift may slightly change the total sum – renormalise.
        total = a_new.sum()
        if total > 0.0:
            a_new *= (sum_target / total)

        new_score = _fast_evaluation(a_new, n)

        # Acceptance test (always accept if better; otherwise Metropolis)
        accept = False
        if new_score < current_score:
            accept = True
        else:
            # probability of accepting a worse move
            accept_prob = np.exp(-(new_score - current_score) / max(temperature, 1e-12))
            if rng.random() < accept_prob:
                accept = True

        if accept:
            a = a_new
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"[SA] iter {iter_count:>7d} improved best to {best_score:.6f}")

        # Occasionally inject a tiny random perturbation to avoid stagnation.
        if rng.random() < 0.001:
            noise = rng.normal(scale=0.01 * sum_target / n, size=n)
            a = np.maximum(0.0, a + noise)
            a_sum = a.sum()
            if a_sum > 0.0:
                a = a * (sum_target / a_sum)
            current_score = _fast_evaluation(a, n)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
                print(f"[noise] iter {iter_count:>7d} new best {best_score:.6f}")

        iter_count += 1

    # ------------------------------------------------------------------
    # Final post‑processing: enforce the limits imposed by `evaluate_sequence`.
    # ------------------------------------------------------------------
    final_sum = best_a.sum()
    if final_sum > 0.0:
        best_a = best_a * (sum_target / final_sum)

    # Clamp to the interval [0, 1000] (the evaluator does this anyway).
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Search finished after {iter_count} iterations – best score ≈ {best_score:.6f}")
    return best_a.tolist()
