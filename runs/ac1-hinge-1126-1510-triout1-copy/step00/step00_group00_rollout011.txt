```
def project_simplex(v):
    # v can be any real vector
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Propose a candidate sequence of non‑negative heights that (heuristically) lowers
the value returned by ``evaluate_sequence``.

The algorithm works in two phases:

1. **Random exploration** – many short random vectors (length 2‑250) are sampled.
   The best‑scoring vector found so far is kept.  For n=2 the analytic optimum
   (2/3 , 1/3) is also tried explicitly.

2. **Local refinement** – starting from the best vector a simple “flatten”
   move is applied: mass is taken from indices that contribute to the current
   maximal convolution entry and moved to a random non‑contributing index.
   Occasionally a small random perturbation or a direction obtained from a
   linear program (maximising the sum under the same convolution bound) is
   used.  The sequence is always re‑normalised to keep the total sum equal to
   one, which makes the objective scale‑invariant.

The code respects the time budget supplied by the caller.  It only relies on
NumPy and SciPy (for the optional LP step) and never performs I/O.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _project_to_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex:
    {w | w_i >= 0, sum w_i = 1}
    """
    if np.all(v <= 0):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximise   sum(b)
        subject to  (f * b)[k] <= rhs   for all k
                     b_j >= 0
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    a_ub = []
    b_ub = []

    # convolution constraints
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = f_i where i + j = k  => i = k - j
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints (b >= 0) as -b <= 0
    a_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    a_ub = np.vstack(a_ub)
    b_ub = np.hstack(b_ub)

    # use the HiGHS solver (default) – no time‑limit flag needed for moderate sizes
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


def _lp_direction(sequence, mix_factor):
    """
    Produce a new sequence by moving a small fraction (mix_factor) towards a
    direction obtained from the linear programme.
    Returns a list of floats or None if the LP fails.
    """
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:
        return None

    # Normalise to the conventional scaling used in the reference paper
    normal = [x * np.sqrt(2.0 * n) / sum_seq for x in sequence]

    # the current worst convolution value (bounding rhs for the LP)
    rhs = float(np.max(np.convolve(normal, normal)))

    g = _solve_convolution_lp(normal, rhs)
    if g is None:
        return None

    # re‑scale the LP solution to the same normalisation as the reference
    sum_g = np.sum(g)
    if sum_g == 0:
        return None
    g_norm = [x * np.sqrt(2.0 * n) / sum_g for x in g]

    # convex combination – the mix_factor is a small step size (e.g. 0.05)
    new_seq = [(1.0 - mix_factor) * x + mix_factor * y for x, y in zip(sequence, g_norm)]
    return new_seq


def _flatten_step(seq, step_factor=0.05):
    """
    Take a tiny amount of probability mass from indices that contribute to the
    current maximal convolution entry and move it to a random non‑contributing
    index.  This tends to reduce the peak of the convolution.
    The sequence is returned as a NumPy array (still summing to 1).
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # indices that actually partake in the max entry
    contributing = [i for i in range(n) if 0 <= k_max - i < n]

    if not contributing:
        return seq

    # choose a donor index among the contributors
    i = np.random.choice(contributing)
    # choose a recipient that does NOT contribute (if any)
    non_contrib = [j for j in range(n) if j not in contributing]
    if non_contrib:
        j = np.random.choice(non_contrib)
    else:
        # fall back to any other index
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)

    # amount to shift – a small fraction of the donor’s weight
    delta = min(step_factor * seq[i], seq[i])
    if delta <= 0:
        return seq

    seq[i] -= delta
    seq[j] += delta
    # numerical safety – project back onto the simplex
    seq = _project_to_simplex(seq)
    return seq


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The function respects the supplied time budget and
    returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence located within the time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # small safety margin

    # -------------------------------------------------
    # Phase 1 – random sampling across many small lengths
    # -------------------------------------------------
    best_seq = None
    best_score = float('inf')

    # explicit optimal 2‑element candidate (2/3 , 1/3) – known to beat uniform
    two_opt = np.array([2.0 / 3.0, 1.0 / 3.0])
    two_opt_score = evaluate_sequence(two_opt.tolist())
    best_seq = two_opt.copy()
    best_score = two_opt_score
    print(f"[Init] n=2 explicit optimum score = {best_score:.6f}")

    while time.time() < deadline - 30.0:   # reserve ~30 s for refinement
        n = np.random.randint(2, 250)      # lengths 2 … 249
        seq = np.random.rand(n)
        seq /= seq.sum()                    # normalise to sum = 1
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[Random] n={n} new best score = {best_score:.6f}")

    # -------------------------------------------------
    # Phase 2 – local refinement (flatten + LP steps)
    # -------------------------------------------------
    cur_seq = best_seq.copy()
    mix_factor = 0.05          # step size for the LP direction
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1

        # occasionally try the LP direction (costly, so not every iteration)
        if np.random.rand() < 0.1:
            new_seq = _lp_direction(cur_seq.tolist(), mix_factor)
            if new_seq is not None:
                cur_seq = np.array(new_seq)
        else:
            # main move: a small flattening step
            cur_seq = _flatten_step(cur_seq, step_factor=0.03)

        # occasional random mass‑swap to keep exploration alive
        if np.random.rand() < 0.02:
            i, j = np.random.choice(len(cur_seq), 2, replace=False)
            delta = cur_seq[i] * 0.1 * np.random.rand()
            delta = min(delta, cur_seq[i])
            cur_seq[i] -= delta
            cur_seq[j] += delta
            cur_seq = _project_to_simplex(cur_seq)

        # evaluate current candidate
        cur_score = evaluate_sequence(cur_seq.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[Iter {iter_count}] improved best score = {best_score:.6f}")

        # slowly shrink the LP mixing factor (helps convergence)
        if iter_count % 1000 == 0:
            mix_factor = max(0.001, mix_factor * 0.9)

    # Return the best sequence discovered (as plain Python floats)
    return [float(x) for x in best_seq]
