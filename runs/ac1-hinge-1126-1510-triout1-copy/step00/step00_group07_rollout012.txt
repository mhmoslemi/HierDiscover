def subgradient(a, max_indices):
    n = len(a)
    g = np.zeros_like(a)
    for k in max_indices:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i_vals = np.arange(start, end + 1)
        g[i_vals] += 2 * a[k - i_vals]
    return g

**project onto simplex**:

```
def simplex_projection(v, sum_target=1.0):
    # similar to algorithm by Wang & Carreira-Perpinan
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - sum_target
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Algorithm:
- Maintain best sequence found.
- For a set of lengths (powers of two up to ~5000), generate random Dirichlet vectors (sum=1).
- Perform a few projected gradient steps that try to reduce the max convolution.
- Keep best.
- Return the best sequence.
"""

import time
import numpy as np

def simplex_projection(v, s=1.0):
    """Project vector v onto the simplex {x >= 0, sum(x) = s}."""
    if np.sum(v) == s and np.all(v >= 0):
        return v
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find rho
    rho_arr = np.arange(1, n + 1)
    cond = u - cssv / rho_arr > 0
    if not np.any(cond):
        # Should not happen, fallback to equal distribution
        return np.full(n, s / n)
    rho = rho_arr[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def max_convolution(a):
    """Return max value of linear convolution of a with itself."""
    conv = np.convolve(a, a)
    return conv.max(), conv

def projected_gradient_step(a, eta):
    """One subgradient step to reduce max(conv(a,a))."""
    n = a.shape[0]
    # compute convolution and active indices
    conv = np.convolve(a, a)
    max_val = conv.max()
    # tolerance for active set
    tol = max_val * 1e-12 + 1e-15
    active = np.flatnonzero(conv >= max_val - tol)
    # compute subgradient
    g = np.zeros_like(a)
    for k in active:
        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i = np.arange(start, end + 1)
        g[i] += 2.0 * a[k - i]
    # gradient descent step and projection
    a_new = a - eta * g
    a_new = simplex_projection(a_new, s=1.0)
    return a_new

"""
Search for a non‑negative step‑function that minimizes the value returned by
`evaluate_sequence`.  The algorithm proceeds as follows:

1.  Keep the best sequence seen so far.
2.  Iterate over a small set of lengths (from a few hundred up to several
    thousand).  For each length a random Dirichlet vector (i.e. a random point
    on the probability simplex) is generated – this guarantees a non‑negative
    sequence with sum = 1.
3.  Starting from that random point a few projected sub‑gradient steps are
    performed.  The sub‑gradient is obtained from the active indices of the
    self‑convolution (the ones attaining the current maximum).  After each step
    the iterate is projected back onto the simplex to preserve feasibility.
4.  Occasionally a tiny random jitter is added to help escape plateaus.
5.  The candidate is evaluated; if it improves the best score it is stored.
6.  The loop continues until the allotted budget (≈ 1000 s) is exhausted.
   If no improvement over the trivial uniform sequence is found, the uniform
   sequence (which attains the known theoretical optimum of 2.0) is returned.
"""

import time
import numpy as np

def simplex_projection(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Project a vector ``v`` onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm from:
    Wang & Carreira‑Perpiñán, “Projection onto the probability simplex:
    An efficient algorithm with a simple proof, and an application”.
    """
    if np.isclose(v.sum(), s) and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted in descending order
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                         # numerical safety
        return np.full(n, s / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

def projected_gradient_step(a: np.ndarray, eta: float) -> np.ndarray:

    ``max(conv(a, a))``.

        start = max(0, k - (n - 1))
        end = min(n - 1, k)
        i = np.arange(start, end + 1)
        g[i] += 2.0 * a[k - i]

    a_new = a - eta * g
    a_new = simplex_projection(a_new, s=1.0)
    return a_new

        try:
            cur_score = evaluate_sequence(a.tolist())
        except Exception:
            continue

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.tolist()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.1f}s] New best score {best_score:.6f} (length={n})")

        # The theoretical minimum of the objective (in exact arithmetic) is 2.
        # If we already hit that value we can stop early.
        if best_score <= 2.000001:
            break

    # -----------------------------------------------------------------
    # Fallback – should never be reached, but guards against an empty result.
    # -----------------------------------------------------------------
    if best_seq is None:
        best_seq = [0.5, 0.5]

    return best_seq
