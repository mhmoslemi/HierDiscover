"""
Propose a sequence that gives a low value of `evaluate_sequence`.

The construction is based on a classic Sidon (B₂) set:
* Choose a prime p and form a set S = {i·p + i² mod p | i = 0,…,p‑1}.
  S has size p, its elements lie in [0, p²‑1] and every sum i+j (i,j∈S)
  is unique.  Consequently the self‑convolution of a sequence that is
  non‑zero only on S has at most one contribution per index.
* Fill the whole array (length N = p²) with a small value ε and set the
  positions in S to 1.0.  With ε ≈ 1/p the contributions from
  large–large, large–small and small–small terms are all of the same
  order, which makes the maximal convolution coefficient close to the
  average and drives the evaluation function towards the theoretical
  lower bound ≈ 1.5.
* A tiny local search (random perturbations) is then performed for the
  remaining time budget to possibly improve the result further.

The algorithm tries a few primes (≤70), a few multiples of 1/p for ε,
keeps the best sequence seen and finally refines it with hill climbing.
"""

import time
import math
import numpy as np

def _is_prime(n: int) -> bool:
    """Simple deterministic primality test for n < 200."""
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    r = int(math.isqrt(n))
    for d in range(3, r + 1, 2):
        if n % d == 0:
            return False
    return True


def _sidon_set(p: int) -> np.ndarray:
    """Return a Sidon (B₂) set of size p inside [0, p²‑1]."""
    # Singer construction: i·p + (i² mod p)
    return np.array([i * p + (i * i) % p for i in range(p)], dtype=int)


def _initial_candidate(p: int, eps_factor: float) -> list[float]:
    """
    Build a candidate sequence for a given prime p and a factor for ε.
    The large entries are 1.0, the background value is ε = eps_factor / p.
    """
    N = p * p
    eps = eps_factor / p
    seq = np.full(N, eps, dtype=float)
    idx = _sidon_set(p)
    seq[idx] = 1.0
    return seq.tolist()


def _local_improve(seq: list[float], time_limit: float) -> list[float]:
    """
    Simple hill‑climbing: repeatedly perturb a random coordinate.
    Keep the change if the score improves.
    Runs while there is time left (time_limit seconds).
    """
    if time_limit <= 0.0:
        return seq
    arr = np.array(seq, dtype=float)
    best = arr.copy()
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = float("inf")
    start = time.time()
    n = len(arr)
    while time.time() - start < time_limit:
        i = np.random.randint(n)
        old = arr[i]
        # Gaussian perturbation with a scale relative to the current value
        scale = max(old, 0.01) * 0.05
        arr[i] = max(0.0, old + np.random.randn() * scale)
        try:
            score = evaluate_sequence(arr.tolist())
        except Exception:
            score = float("inf")
        if score < best_score:
            best_score = score
            best = arr.copy()
        else:
            # revert most of the time; occasionally keep a worsening move
            if np.random.rand() < 0.01:
                best_score = score
                best = arr.copy()
            else:
                arr[i] = old
    return best.tolist()

    """

    # start with the known construction if it exists
    best_seq = None

        best_seq = None

    # -----------------------------------------------------------------
    # 1) Build Sidon‑based candidates for several small primes
    # -----------------------------------------------------------------
    primes = [p for p in range(2, 80) if _is_prime(p)]
    max_len = 5000  # keep the length manageable for the time budget
    eps_factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]  # multiply 1/p

    for p in primes:
        N = p * p
        if N > max_len:

        for factor in eps_factors:
            cand = _initial_candidate(p, factor)

            if score < best_score:
