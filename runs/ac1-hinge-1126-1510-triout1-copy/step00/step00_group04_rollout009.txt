def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import numpy as np, time, math, random
    from scipy.special import gamma
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0  # early stop

    best_seq = None
    best_score = np.inf

    # define lengths to try
    lengths = [512, 1024, 2048, 3072, 4096]  # within reasonable range
    # ensure lengths are <= maybe 4000 to speed up
    for n in lengths:
        target_sum = math.sqrt(2.0 * n)
        # sample exponents p
        for p in np.linspace(0.52, 0.68, 17):  # from slightly above 0.5
            # few shift values
            for shift in [0.0, 0.5, 1.0, 2.0]:
                if time.time() > deadline:
                    break
                # generate raw sequence
                idx = np.arange(1, n+1, dtype=np.float64) + shift
                raw = idx ** (-p)
                # avoid all zeros, sum positive
                sum_raw = np.sum(raw)
                if sum_raw == 0:
                    continue
                seq = raw * (target_sum / sum_raw)
                # evaluate
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.tolist()
                    print(f"New best n={n}, p={p:.4f}, shift={shift}, score={score:.6f}")
        if time.time() > deadline:
            break

    # Optional: local refinement on best_seq using small perturbations
    if best_seq is not None:
        seq = np.array(best_seq, dtype=np.float64)
        n = len(seq)
        target_sum = math.sqrt(2.0*n)
        # simple hill climb
        while time.time() < deadline:
            # pick a random index to tweak
            i = np.random.randint(n)
            # propose small random change delta (both directions)
            delta = np.random.randn() * 0.01 * seq[i] if seq[i] > 0 else np.random.rand()*0.01
            new_seq = seq.copy()
            new_seq[i] = max(0.0, new_seq[i] + delta)  # ensure non-negative
            # renormalize sum to target_sum
            new_seq = new_seq * (target_sum / new_seq.sum())
            try:
                score = evaluate_sequence(new_seq.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = new_seq.tolist()
                    seq = new_seq
                    print(f"Refine: improved score={score:.6f}")
            except Exception:
                continue

    return best_seq if best_seq is not None else [1.0]

The search is based on the analytic observation that the continuous
function f(x)=x^{-p} (p>0.5) has a self‑convolution that is
approximately constant on the interior of its support.
After normalising the ℓ₁‑norm to √(2n) this yields a convolution

    M(p,n) = 2·(1-p)²·max( n^{2p‑1} ,  Beta(1-p,1-p) ),

normalises it to the required sum √(2n) and evaluates it.
The best candidate is kept, then a cheap hill‑climbing phase performs
tiny random perturbations while keeping the ℓ₁‑norm fixed.  All
operations respect the security checks in ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# The evaluation function is provided by the competition environment.
# from your_module import evaluate_sequence   # not needed, already in globals.

def build_sequence(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """
    Construct a non‑negative sequence of length ``n`` of the form
        a_i = (i + shift)^{-p}
    and scale it so that sum(a) = sqrt(2*n).
    """
    # i starts at 1 to avoid a zero denominator.
    i = np.arange(1, n + 1, dtype=np.float64) + shift
    raw = i ** (-p)
    # Normalise to the required ℓ₁‑norm.
    target = math.sqrt(2.0 * n)
    scale = target / raw.sum()
    return raw * scale


def hill_climb(seq: np.ndarray, time_deadline: float) -> np.ndarray:
    """
    Very cheap local search: repeatedly pick a random index,
    add a tiny perturbation, renormalise the sum and keep the change
    if it improves the evaluation score.
    """
    n = len(seq)
    target = math.sqrt(2.0 * n)

    best = seq.copy()
    try:
        best_score = evaluate_sequence(best.tolist())
    except Exception:
        best_score = np.inf

    while time.time() < time_deadline:
        # choose an index and propose a small change
        idx = np.random.randint(n)
        delta = np.random.randn() * 0.02 * best[idx]  # relative step
        cand = best.copy()
        cand[idx] = max(0.0, cand[idx] + delta)

        # renormalise to keep the ℓ₁‑norm fixed
        cand *= target / cand.sum()

        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        if score < best_score:
            best_score = score
            best = cand
            print(f"hill‑climb improvement: score={score:.6f}")
    return best


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation value.
    The method scans a small grid of (n, p, shift) parameters that
    implement the analytically motivated power‑law shape, then refines
    the best result with a simple hill‑climbing loop.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a safety margin for the final hill‑climb
    deadline = start_time + budget_s - 10.0

    best_seq = None
    best_score = np.inf

    # lengths to try – a few values give a good trade‑off between quality
    # and evaluation cost.
    lengths = [512, 1024, 2048, 3072, 4096]

    # exponent grid (p>0.5 is needed for the interior constant behaviour)
    p_grid = np.linspace(0.52, 0.68, 17)   # 0.52, 0.53, … , 0.68

    # small additive shifts to smooth the singularity at the first index
    shifts = [0.0, 0.5, 1.0, 2.0]

    for n in lengths:
        if time.time() >= deadline:
            break
        for p in p_grid:
            if time.time() >= deadline:
                break
            for sh in shifts:
                if time.time() >= deadline:
                    break
                seq = build_sequence(n, p, sh)
                # safety: enforce the upper bound of 1000 (the evaluator will clamp anyway)
                if seq.max() > 1000.0:
                    continue
                try:
                    score = evaluate_sequence(seq.tolist())
                except Exception:
                    continue
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"new best → n={n}, p={p:.4f}, shift={sh}, score={score:.6f}")

    # If the grid search did not find anything (unlikely), fall back to a uniform sequence.
    if best_seq is None:
        n = 256
        uniform = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)
        best_seq = uniform
        best_score = evaluate_sequence(best_seq.tolist())
        print("fallback to uniform sequence, score=", best_score)

    # -----------------------------------------------------------------
    # Local optimisation (tiny random moves, sum‑preserving)
    # -----------------------------------------------------------------
    best_seq = hill_climb(best_seq, deadline)

    # Convert to plain list for the competition harness.
    return best_seq.tolist()
