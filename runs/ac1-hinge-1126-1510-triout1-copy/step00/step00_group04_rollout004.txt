"""
Projected gradient descent on an Lp‐approximation of the L∞ norm of the
self‑convolution.  The objective is to minimise

    2 * n * max_k (a * a)_k / (sum a)^2

which is invariant to scaling of ``a``.  We therefore keep the vector on the
probability simplex (non‑negative, sum = 1).  At each iteration we compute the
self‑convolution, construct an Lp‑norm (p > 1) which concentrates the
gradient on the largest convolution entries, take a gradient step and project
back onto the simplex.  The algorithm is started from the known construction
``height_sequence_1`` (if available) or from a random Dirichlet vector and
runs until the time budget expires, keeping the best sequence found.

The code uses only NumPy and standard library modules and respects all
submission rules.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Euclidean projection onto the probability simplex)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def _flatten_convolution(a: np.ndarray,
                         time_deadline: float,
                         p: int = 8,
                         lr0: float = 0.7,
                         restart_interval: int = 3000) -> np.ndarray:
    """
    Perform projected gradient descent on the Lp‑norm of the self‑convolution
    of ``a``.  Returns the best vector (on the simplex) seen before the
    deadline.
    """
    n = a.shape[0]

    # initial best
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    lr = lr0
    it = 0
    last_improve = 0

    while time.time() < time_deadline:
        # self‑convolution
        conv = np.convolve(a, a)               # length 2n‑1, all non‑negative
        # Lp‑norm and its gradient coefficients
        conv_p = conv ** p
        S = conv_p.sum()
        if S == 0.0:            # numerical safety – should never happen
            break
        # Derivative of ( Σ conv^p )^{1/p}
        weight = conv ** (p - 1) / (S ** ((p - 1) / p))

        # gradient of the Lp‑norm w.r.t. a:
        #   ∂/∂a_i Σ_k weight_k * Σ_j a_j a_{k-j}
        #   = 2 * (weight * a_rev)[i]   where a_rev = a reversed
        grad = 2.0 * np.convolve(weight, a[::-1])[n - 1:n - 1 + n]

        # gradient step
        a = a - lr * grad
        # project back onto the probability simplex
        a = _project_simplex(a)

        # evaluate current sequence
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = it
            # occasional progress report
            print(f"[Iter {it}] new best score = {best_score:.6f}")

        # simple learning‑rate schedule
        if it % 200 == 0 and it > 0:
            lr *= 0.8

        # if we have stagnated for a while, inject a small random perturbation
        if it - last_improve > restart_interval:
            perturb = np.random.randn(n) * 0.005
            a = best_a + perturb
            a = np.maximum(a, 0.0)
            a = _project_simplex(a)
            last_improve = it
            # reset learning rate a little
            lr = lr0

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Public entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises the
    evaluation function.  The routine runs a projected gradient descent on an
    Lp‑approximation of the max convolution term and returns the best vector
    found before the time budget expires.
    """
    np.random.seed(seed)
    # reserve a few seconds for clean‑up / finalisation
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # initialise a starting point
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided globally in the harness
        init = np.asarray(height_sequence_1, dtype=float)
        n = init.size
        # normalise to sum = 1 (the evaluation is scale‑invariant)
        a0 = init / init.sum()
        print(f"Initialising from height_sequence_1 (n={n})")
    except Exception:
        # fall back to a random Dirichlet vector of moderate size
        n = 2500
        a0 = np.random.rand(n)
        a0 = a0 / a0.sum()
        print(f"Initialising randomly (n={n})")

    # ------------------------------------------------------------------
    # run the optimisation
    # ------------------------------------------------------------------
    best_vec = _flatten_convolution(a0, deadline, p=8, lr0=0.6,

    # final score (optional reporting)
    final_score = evaluate_sequence(best_vec.tolist())
    print(f"Finished optimisation – best score = {final_score:.6f}")

    # return as plain Python list
    return best_vec.tolist()
