def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # initial candidate
    if np.random.rand() < 0.7 and height_sequence_1 is not None:
        curr = list(height_sequence_1)
    else:
        # random sparse sequence
        n = np.random.randint(200, 800)
        # Choose number of nonzero supports randomly, up to sqrt(2n)
        m = np.random.randint(10, int(np.sqrt(2*n)*2)+1)
        idx = np.random.choice(n, size=m, replace=False)
        values = np.random.rand(m) + 0.1
        seq = np.zeros(n)
        seq[idx] = values
        curr = seq.tolist()
    best_seq = curr.copy()
    best_score = evaluate_sequence(best_seq)
    n = len(curr)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # decide to perform LP step or random perturbation
        if np.random.rand() < 0.8:
            # LP step
            new_seq = lp_improve(curr)
        else:
            new_seq = random_perturb(curr)
        # evaluate
        try:
            score = evaluate_sequence(new_seq)
        except Exception:
            continue
        if score < best_score - 1e-12: # improvement
            best_score = score
            best_seq = new_seq.copy()
            # logging
            print(f"Iter {iteration}: improved score {best_score:.6f}, len {len(best_seq)}")
        # incorporate occasional restart: if no improvement for many steps, random restart
        if iteration % 200 == 0:
            # maybe keep best further, else random restart if stuck
            if score >= best_score * 0.999:
                # random restart near best: jitter
                curr = jitter(best_seq, scale=0.05)
            else:
                curr = new_seq.copy()
        else:
            curr = new_seq.copy()
    return best_seq

def random_perturb(seq):
    new = seq.copy()
    n = len(new)
    # choose a few indices to modify randomly
    for _ in range(np.random.randint(1,5)):
        idx = np.random.randint(n)
        # add Gaussian noise scaled by current avg magnitude
        scale = max(1e-2, np.mean(new) * 0.1)
        new[idx] = max(0.0, new[idx] + np.random.randn() * scale)
    # optionally adjust total sum by a small scaling factor to avoid sum going to zero
    if np.random.rand() < 0.2:
        factor = 1 + np.random.randn() * 0.01
        new = [max(0.0, x*factor) for x in new]

1. **Linear‑programming direction** –  For the current sequence `f` we
   compute the maximal value `M = max(conv(f,f))`.  Solving an LP
   (maximise Σb   s.t. conv(f,b) ≤ M, b ≥ 0) yields a new non‑negative
   sequence `b` that has a (strictly) larger ℓ₁‑norm while respecting the
   same convolution bound.  The difference `d = b‑f` is a promising descent
   direction.

2. **Line‑search on the direction** –  Rather than using a fixed tiny
   mixing weight (as in the reference code), we evaluate a small grid of
   mixing parameters `t ∈ (0,1]`.  For each `t` we form the candidate
   `f_t = max(0, f + t·d)` (clipping negatives to zero) and compute its
   score with `evaluate_sequence`.  The best `t` is kept, which yields a
   much larger decrease per LP step.

3. **Stochastic perturbations & occasional restarts** –  When the LP
   fails (or does not improve the score) we apply cheap random
   perturbations to a few coordinates.  If the algorithm has not improved
   for many iterations we jitter the current best solution to escape a
   possible local minimum.

The search runs until the allotted time (`budget_s`) expires, constantly
keeping the best sequence seen so far.  The sequence length is capped at
`MAX_LEN` (default 1200) to keep the LP tractable; a longer historic
solution `height_sequence_1` is truncated if necessary.

All operations respect the non‑negative requirement and the score is
strictly decreasing – the returned sequence is the best found within
the time limit.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def solve_convolution_lp(f_seq, rhs):
    """
    Solve  max Σ b_j   s.t.   (f_seq * b)_k ≤ rhs  for all k,
    with b_j ≥ 0.
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # Objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the Toeplitz‑like matrix for the convolution constraints.
    # Row k (0‑based) corresponds to (f * b)[k] = Σ_i f[i] * b[k-i].
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    for k in range(rows):
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        # indices j = low … high → i = k‑j
        A[k, low:high + 1] = f_seq[k - np.arange(low, high + 1)]

    b = np.full(rows, rhs, dtype=np.float64)

    # Non‑negativity handled via bounds (faster than extra rows)
    bounds = [(0.0, None) for _ in range(n)]

    # Use the default HiGHS solver – fast and reliable.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def lp_improve(seq):
    """
    Perform one LP‑based improvement step on `seq`.
    Returns a new sequence (list of floats) or None if the LP fails.
    """
    n = len(seq)
    seq_arr = np.asarray(seq, dtype=np.float64)

    # Current convolution bound M
    conv_ff = np.convolve(seq_arr, seq_arr)
    M = np.max(conv_ff)

    # Solve the LP for a direction with larger ℓ₁‑norm
    b_opt = solve_convolution_lp(seq_arr, M)
    if b_opt is None:
        return None

    # Direction d = b - f
    d = b_opt - seq_arr

    # If direction is (almost) zero – nothing to do
    if np.linalg.norm(d) < 1e-12:
        return None

    # ----- line search on t ------------------------------------------------
    # Evaluate a modest grid of mixing parameters.
    best_candidate = None
    best_score = float("inf")
    # Include the current sequence as a fallback
    try:
        cur_score = evaluate_sequence(seq)
        best_score = cur_score
        best_candidate = seq
    except Exception:
        pass

    # Grid of candidate t values
    t_grid = np.linspace(0.01, 1.0, 20)
    for t in t_grid:
        cand = np.maximum(seq_arr + t * d, 0.0)
        try:
            score = evaluate_sequence(cand.tolist())
        except Exception:
            continue
        if score < best_score:
            best_score = score
            best_candidate = cand.tolist()

    # If none of the t‑values improved, fall back to a tiny random
    # perturbation (handled outside).
    return best_candidate


def random_perturb(seq, scale_factor=0.1):
    """
    Apply a cheap random perturbation to a few coordinates of `seq`.
    Returns a new list.
    """
    arr = np.array(seq, dtype=np.float64)
    n = len(arr)
    # number of coordinates to touch (1‑5)
    k = np.random.randint(1, min(5, n) + 1)
    for _ in range(k):
        idx = np.random.randint(n)
        # magnitude of noise proportional to the typical entry size
        avg = np.mean(arr) if np.mean(arr) > 0 else 1.0
        sigma = max(1e-4, avg * scale_factor)
        arr[idx] = max(0.0, arr[idx] + np.random.randn() * sigma)
    # occasional global scaling (keeps the shape but changes magnitude)
    if np.random.rand() < 0.15:
        factor = 1.0 + np.random.randn() * 0.01
        arr = np.maximum(arr * factor, 0.0)
    return arr.tolist()


def jitter(seq, sigma=0.05):
    """Add small i.i.d. Gaussian noise to every entry."""
    arr = np.array(seq, dtype=np.float64)
    avg = np.mean(arr) if np.mean(arr) > 0 else 1.0
    noise = np.random.randn(*arr.shape) * avg * sigma
    return np.maximum(arr + noise, 0.0).tolist()


def symmetrize(seq):
    """Force mirror symmetry: a_i = a_{n‑1‑i} (average of the pair)."""
    arr = np.asarray(seq, dtype=np.float64)
    n = len(arr)
    for i in range(n // 2):
        avg = (arr[i] + arr[n - 1 - i]) * 0.5
        arr[i] = avg
        arr[n - 1 - i] = avg
    if n % 2 == 1:
        arr[n // 2] = arr[n // 2]  # middle stays unchanged
    return arr.tolist()

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The function runs for up to `budget_s` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # reserve a little margin

    # ------------------------------------------------------------------
    # Initialisation ----------------------------------------------------
    # ------------------------------------------------------------------
    MAX_LEN = 1200           # keep LP tractable
    # Prefer the historic SOTA solution if it exists.
    try:
        init_seq = [float(x) for x in height_sequence_1]
        # Drop tiny trailing entries (treated as zero)
        while init_seq and abs(init_seq[-1]) < 1e-12:
            init_seq.pop()
        if len(init_seq) > MAX_LEN:
            init_seq = init_seq[:MAX_LEN]
    except Exception:
        # Fallback: random sparse start
        n = np.random.randint(200, MAX_LEN + 1)
        init_seq = np.zeros(n, dtype=np.float64).tolist()
        # sprinkle a few random positive entries
        for _ in range(np.random.randint(5, 15)):
            idx = np.random.randint(n)
            init_seq[idx] = np.random.rand() + 0.1

    # Ensure at least one positive entry
    if np.all(np.array(init_seq) == 0):
        init_seq[0] = 1.0

    curr_seq = init_seq.copy()
    best_seq = init_seq.copy()
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_score = float("inf")

    # ------------------------------------------------------------------
    # Search loop -------------------------------------------------------
    # ------------------------------------------------------------------
    iter_no = 0
    last_improve_iter = 0

    while time.time() < deadline:
        iter_no += 1

        # Choose a move type
        if np.random.rand() < 0.45 and len(curr_seq) <= MAX_LEN:
            cand_seq = lp_improve(curr_seq)
        else:
            cand_seq = None

        if cand_seq is None:
            # cheap random move
            cand_seq = random_perturb(curr_seq, scale_factor=0.15)

        # Occasionally enforce symmetry (helps flatten the convolution)
        if np.random.rand() < 0.10:
            cand_seq = symmetrize(cand_seq)

        # Evaluate candidate
        try:
            cand_score = evaluate_sequence(cand_seq)
        except Exception:
            continue

        # Keep the best ever
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_seq = cand_seq.copy()
            last_improve_iter = iter_no
            print(f"[{iter_no:5d}] New best score: {best_score:.6f} (len={len(best_seq)})")

        # Prepare for next iteration – accept the candidate if it improves
        # the current sequence (a simple greedy acceptance).
        cur_score = None
        try:
            cur_score = evaluate_sequence(curr_seq)
        except Exception:
            cur_score = float("inf")
        if cand_score < cur_score:
            curr_seq = cand_seq
        else:
            # occasional random jump to avoid stagnation
            if iter_no - last_improve_iter > 200:
                curr_seq = jitter(best_seq, sigma=0.08)
                last_improve_iter = iter_no
                print(f"[{iter_no:5d}] Stagnation – jittering around best solution.")

        # Early exit if a sufficiently good bound is reached
        if best_score < 1.51:
            print(f"[{iter_no:5d}] Target achieved (score {best_score:.6f}). Stopping early.")
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered
    # ------------------------------------------------------------------
    return best_seq
