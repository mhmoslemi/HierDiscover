def improve_sequence(seq, steps, max_time):
    best = seq.copy()
    best_score = evaluate_sequence(best)
    start = time.time()
    n = len(seq)
    step_size = 0.1  # fraction of max entry?
    while time.time() - start < max_time:
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        max_idx = int(conv.argmax())
        # find pair i that maximizes a[i]*a[max_idx-i] (subject to indices)
        candidates = []
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                candidates.append((i, seq[i]*seq[j]))
        i_max, prod_max = max(candidates, key=lambda x: x[1])
        # Determine partner j_max
        j_max = max_idx - i_max
        # Choose a random index k not equal to i_max or j_max
        k = np.random.randint(n)
        while k == i_max or k == j_max:
            k = np.random.randint(n)
        # move epsilon from the larger of i_max or j_max to k
        src = i_max if seq[i_max] > seq[j_max] else j_max
        eps = step_size * seq[src]  # fraction
        if eps <= 1e-12:
            continue
        new_seq = seq.copy()
        new_seq[src] -= eps
        new_seq[k] += eps
        # ensure nonnegative
        if new_seq[src] < 0: continue
        # evaluate
        new_score = evaluate_sequence(new_seq)
        if new_score < best_score:
            best_score = new_score
            best = new_seq.copy()
        # accept if improvement
        if new_score < evaluate_sequence(seq) or np.random.rand() < 0.01:
            seq = new_seq
    return best, best_score

import cvxpy as cp
def optimize_length(n):
    x = cp.Variable(n)
    # define expressions
    exprs = []
    for k in range(2*n -1):
        terms = []
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                terms.append(x[i] + x[j])
        exprs.append(cp.log_sum_exp(cp.hstack(terms)))
    max_expr = cp.max(cp.hstack(exprs))
    sum_exp = cp.sum(cp.exp(x))
    objective = max_expr - 2 * cp.log(sum_exp)

"""
Propose a high‑quality step‑function using convex optimisation.

The evaluation to be minimised is

    E(a) = 2·n·max_k (a * a)[k] / (∑ a_i)²

For a fixed length n the optimisation problem can be written as

    minimise   log max_k Σ_{i+j=k} a_i a_j  –  2·log Σ_i a_i

with a_i ≥ 0, which is equivalent (after the change a_i = exp(x_i))
to the convex programme

    minimise   max_k log_sum_exp( x_i + x_j  : i+j=k ) – 2·log Σ_i exp(x_i)

The objective is convex because `log_sum_exp` is convex,
`max` of convex functions is convex and `-log` is convex.
Hence any DCP‑compatible solver (ECOS, SCS, …) will find the global optimum.

The function iterates over a list of candidate lengths, solves the convex
problem for each length (up to the remaining time budget) and keeps the
sequence with the smallest evaluation value.  If CVXPY is unavailable or a
solve fails, a simple 2‑element construction `[1/3, 2/3]` (the known optimal
for n=2) is used as a safe fallback.
"""

import time
import numpy as np

def _solve_optimal_sequence(n: int, time_left: float):
    """Solve the convex problem for a given length n.

    Returns a tuple (sequence, score) or (None, inf) on failure.
    """
    try:
        import cvxpy as cp
    except Exception as e:  # CVXPY not present
        return None, float("inf")

    # Decision variable: log‑amplitudes x_i (no non‑negativity constraint needed)
    x = cp.Variable(n)

    # Σ exp(x_i)  – the L1 norm of the (positive) amplitudes
    sum_exp = cp.sum(cp.exp(x))

    # Build the list of log‑sum‑exp expressions for each convolution index k
    conv_terms = []
    for k in range(2 * n - 1):
        pairs = []
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                pairs.append(x[i] + x[j])
        # log‑sum‑exp of all ordered pairs that contribute to index k
        conv_terms.append(cp.log_sum_exp(cp.hstack(pairs)))

    # Max over all k (convex)
    max_conv = cp.max(cp.hstack(conv_terms))

    # Objective = max_conv − 2·log(∑ a_i)  (convex)
    objective = max_conv - 2 * cp.log(sum_exp)

    prob = cp.Problem(cp.Minimize(objective))
    # ECOS handles the exponential cone required by log_sum_exp
    try:
        prob.solve(solver="ECOS", warm_start=True, max_iters=5000, verbose=False)
    except Exception:
        return None, float("inf")

    if x.value is None:
        return None, float("inf")

    # Recover a_i = exp(x_i) and normalise to sum 1 (scale does not matter for the score)
    a = np.exp(x.value)
    a = a / a.sum()

    # Evaluate the sequence using the provided routine
    from __main__ import evaluate_sequence  # the evaluator is injected in the environment
    score = evaluate_sequence(a.tolist())
    return a, score

    """Return a sequence of non‑negative heights with a low evaluation score.

    The routine tries several lengths (2 … 100) and solves a convex programme
    for each length.  The best sequence found within the time budget is
    returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5  # keep a small safety margin

    best_seq = None
    best_score = float("inf")

    # A modest list of lengths – larger lengths give a bigger search space
    # but also cost more time.  The list can be extended if time permits.
    lengths = [2, 3, 4, 5, 6, 7, 8, 10, 12, 15, 20, 30, 40, 50, 80, 100]
