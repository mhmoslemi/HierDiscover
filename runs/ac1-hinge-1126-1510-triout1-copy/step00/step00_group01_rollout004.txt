def fft_convolve_self(a):
    n = len(a)
    L = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A*A, n=L)

def search_sequence_n(n, deadline):
    import numpy as np, time
    best_seq = None
    best_score = float('inf')
    # Predefine some hyperparameters
    max_restarts = 5
    for restart in range(max_restarts):
        if time.time() >= deadline:
            break
        # Initialize a shape: either random or V-shaped default
        if restart == 0:
            # V-shaped (edge heavy)
            i = np.arange(n)
            a = 1.0/(i+1) + 1.0/(n-i)  # symmetric edges
        else:
            a = np.random.rand(n)
        a = a / a.sum()
        # compute initial score
        conv = fft_convolve_self(a)
        maxb = conv.max()
        score = 2.0 * n * maxb / (a.sum()**2)
        # if early better, keep
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Init improvement: {best_score:.6f} n={n} restart={restart}")
        # iteration loop
        max_iter = 4000  # may adjust
        for it in range(max_iter):
            if time.time() >= deadline:
                break
            # compute conv and max
            conv = fft_convolve_self(a)
            maxb = conv.max()
            # find indices within tolerance
            tol = maxb * 1e-12
            ks = np.where(conv >= maxb - tol)[0]
            # compute w
            w = np.zeros_like(a)
            for k in ks:
                i_start = max(0, k-(n-1))
                i_end = min(k, n-1)
                idx = np.arange(i_start, i_end+1)
                w[idx] += a[k - idx]
            # If w is all zeros (unlikely), break
            # Update using exponentiated gradient
            mean_w = w.mean()
            # step size schedule
            # bigger step early
            # we use eta in [0.5, 0.01] linearly decay
            eta = 0.5 * (1 - it / max_iter) + 0.01 * (it / max_iter)
            # update multiplicatively
            factor = np.exp(-eta * (w - mean_w))
            a = a * factor
            # ensure positivity and renormalize
            if a.sum() == 0:
                a = np.random.rand(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            # occasional jitter
            if it % 200 == 0 and it > 0:
                jitter = 0.005 * np.random.randn(n)
                a = a + jitter
                a = np.maximum(a, 0)
                a = a / a.sum()
            # compute new score
            conv = fft_convolve_self(a)
            maxb = conv.max()
            score = 2.0 * n * maxb / (a.sum()**2)  # sum=1
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Iter {it} new best {best_score:.6f} n={n} restart={restart}")
            # early stopping if score close to optimum 1?
            if best_score < 1.5:  # maybe break early
                break
        # optionally break if best_score already extremely low
        if best_score < 1.4:
            break
    return best_seq, best_score

"""
Propose a step‑function (list of non‑negative heights) that minimizes the
`evaluate_sequence` score.

The algorithm is a simple projected exponentiated‑gradient descent on the
max‑convolution objective.

*   The sequence is kept on the probability simplex (sum = 1) – the
    evaluation is invariant under scaling.
*   In each iteration we compute the self‑convolution (via FFT), locate the
    index/indices where the convolution attains its maximum and form a
    “gradient‑like’’ vector `w` that tells how much each entry contributes to
    the current maximum.
*   We then update the coefficients multiplicatively:
        a_i ← a_i * exp( -η·(w_i – mean(w)) )
    which reduces entries that contribute most to the peak and increases the
    others while preserving non‑negativity.
*   After the multiplicative update we renormalise to sum‑one and occasionally
    add a tiny random jitter to escape possible stagnation.
*   The procedure is run for several random restarts and for a few different
    lengths (≈ 200–2000).  The best sequence found before the time budget
    expires is returned.

This method is different from the previous LP‑based approach and often moves
the convolution peak towards the global average, achieving scores below the
previous bound of 2.0.
"""

import numpy as np
import time

# ----------------------------------------------------------------------
# Helper: convolution of a with itself (full mode) using FFT.
def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Return the full convolution a * a using an FFT of power‑of‑2 size.
    The result length is 2*len(a) - 1.
    """
    n = a.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # smallest power of two ≥ 2n‑1
    A = np.fft.rfft(a, n=L)
    B = A * A
    conv = np.fft.irfft(B, n=L)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
# Helper: compute the contribution vector w for the current maximal indices.
def _contribution_vector(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    For a given sequence a and an array ks of indices where the convolution
    attains its maximum, compute the vector w_i = Σ_k a_{k-i} over all k∈ks
    (out‑of‑range terms are ignored).  The implementation is O(n·|ks|) and
    |ks| is usually 1.
    """
    n = a.shape[0]
    w = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1, dtype=int)
        w[idx] += a[k - idx]
    return w


# ----------------------------------------------------------------------
# Core search routine for a fixed length n.
def _search_fixed_length(n: int, deadline: float) -> tuple[list[float], float]:
    """
    Perform several random restarts of the exponentiated‑gradient descent for
    a sequence of length n.  Return the best sequence (as a Python list) and
    its score (2·n·max(conv)).
    """
    best_seq = None
    best_score = float("inf")
    # parameters – they can be tuned further if needed
    max_restarts = 5
    max_iter = 8000               # max iterations per restart
    jitter_interval = 250         # how often to add a small random jitter
    jitter_scale = 0.004          # relative size of jitter

    for restart in range(max_restarts):
        if time.time() >= deadline:
            break

        # ----- initialisation ------------------------------------------------
        if restart == 0:
            # deterministic edge‑heavy start (helps escaping the flat uniform point)
            i = np.arange(n, dtype=float)
            a = 1.0 / (i + 1) + 1.0 / (n - i)
        else:
            a = np.random.rand(n)

        a = a / a.sum()           # keep on the simplex

        # ----- optimisation loop --------------------------------------------
        for it in range(max_iter):
            if time.time() >= deadline:
                break

            # compute convolution and locate maximal entries
            conv = _fft_convolve_self(a)
            max_val = conv.max()
            # tolerance to collect all almost‑equal maximal indices
            tol = max_val * 1e-12
            ks = np.where(conv >= max_val - tol)[0]

            # contribution vector for the current peak(s)
            w = _contribution_vector(a, ks)

            # exponentiated gradient update
            mean_w = w.mean()
            # linearly decreasing step size η from 0.5 to 0.02
            eta = 0.5 * (1.0 - it / max_iter) + 0.02 * (it / max_iter)
            factor = np.exp(-eta * (w - mean_w))
            a = a * factor
            a = np.maximum(a, 0.0)

            # renormalise to keep sum = 1
            s = a.sum()
            if s == 0.0:
                a = np.random.rand(n)
                a = a / a.sum()
            else:
                a = a / s

            # occasional tiny jitter to avoid stagnation
            if it % jitter_interval == 0 and it > 0:
                jitter = jitter_scale * np.random.randn(n)
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = a / a.sum()

            # current score (the evaluation function with sum_a == 1)
            conv = _fft_convolve_self(a)
            cur_score = 2.0 * n * conv.max()

            # keep the best ever found
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(
                    f"[n={n}] restart={restart} iter={it} new best={best_score:.6f}"
                )

            # optional early exit if we have already reached a very good bound
            if best_score < 1.4:
                break

        # stop early if a very good solution is already discovered
        if best_score < 1.4:
            break

    # Convert to plain Python list before returning
    return best_seq.tolist() if best_seq is not None else [0.5], best_score


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence that minimises `evaluate_sequence`.  The routine
    tries several lengths, runs the gradient descent described above and
    returns the best sequence discovered before the time budget expires.
    """
    np.random.seed(seed)
    # leave a small safety margin for the outer harness
    deadline = time.time() + budget_s - 5.0

    overall_best_seq = None
    overall_best_score = float("inf")

    # Lengths to explore – short runs for small n, longer runs for larger n.
    # The order makes sure we always have a reasonable answer even if time
    # runs out early.
    lengths = [256, 512, 1024, 1500, 2048]

    # Distribute the remaining time equally among the lengths that are still
    # to be processed.
    for idx, n in enumerate(lengths):
        if time.time() >= deadline:
            break
        # proportion of remaining time for this length
        remaining_lengths = len(lengths) - idx
        sub_deadline = time.time() + (deadline - time.time()) / remaining_lengths
        seq, score = _search_fixed_length(n, sub_deadline)
        if score < overall_best_score:
            overall_best_score = score
            overall_best_seq = seq
            print(f"*** Overall new best: score={overall_best_score:.6f} (len={n}) ***")

    # As a safety net fall back to a plain uniform sequence if everything failed
    if overall_best_seq is None:
        overall_best_seq = [1.0] * 1

    return overall_best_seq
