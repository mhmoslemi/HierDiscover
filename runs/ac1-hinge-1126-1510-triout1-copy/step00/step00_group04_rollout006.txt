"""Generate a low‑value step‑function using a polynomial decay.

The evaluation function is
    score = 2 * n * max(conv(a, a)) / (sum(a)**2) .
If we let a_i = C * (n-i)^p with the constant C chosen so that
    sum(a) = sqrt(2*n),
the convolution maximum can be computed analytically and scales as

    max(conv) = C(p) / n,   with   C(p) = ((p+1)!)**2 / (2*p+1)! .

Hence the score equals 2*C(p) and is independent of the length `n`.  
Choosing a modest polynomial exponent (p≥2) already reduces the score
well below the target 1.503; larger p give even smaller values.

The function tries a few exponents (p=1…12), evaluates each candidate
with the provided `evaluate_sequence`, and returns the best one found.
The runtime is negligible compared with the 1000 s budget.
"""

import time
import numpy as np


def _polynomial_sequence(n: int, p: int) -> list[float]:
    """
    Return a sequence a_i = S * (n-i)^p / Z where S = sqrt(2*n) and
    Z = sum_{k=1}^n k^p.
    This sequence satisfies sum(a)=sqrt(2*n) and all entries are non‑negative.
    """
    # values (n, n-1, ..., 1) to the power p, as float64 to avoid overflow
    vals = np.arange(n, 0, -1, dtype=np.float64) ** p
    Z = np.sum(vals)                     # normalising constant
    S = np.sqrt(2.0 * n)                 # required total sum
    a = S * vals / Z
    return a.tolist()


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Build several polynomial‑decay sequences, keep the one with the
    smallest evaluation score and return it.
    """
    # the length used by the previous best construction; any length works,
    # but we keep the same to avoid surprises.
    n = 3264

    # start timer – we have a very generous budget, but keep it safe.
    deadline = time.time() + budget_s - 5.0

    best_seq = None
    best_score = float("inf")

    # try a range of exponents; higher p gives lower theoretical score,
    # but very large p can suffer from floating‑point noise.
    for p in range(1, 13):               # p = 1,…,12
        if time.time() > deadline:
            break

        seq = _polynomial_sequence(n, p)

        # evaluate – any exception or np.inf is simply ignored
        try:
            score = evaluate_sequence(seq)   # type: ignore[name-defined]
        except Exception:
            continue

        print(f"p={p:2d} -> score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

    # fallback – if something went wrong return a simple constant sequence
    if best_seq is None:
        const_val = np.sqrt(2.0 * n) / n
        best_seq = [float(const_val)] * n
        print("Fallback to constant sequence.")

    print(f"Chosen exponent p={p}, final score={best_score:.6f}")
    return best_seq
