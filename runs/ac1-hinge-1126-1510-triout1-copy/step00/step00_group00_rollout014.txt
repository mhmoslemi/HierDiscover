def convolve_self(a):
    n = a.shape[0]
    l = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, n=l)
    B = A * A
    conv_full = np.fft.irfft(B, n=l)[:2*n-1]

"""
Propose a candidate sequence for the “max self‑convolution / sum²” problem.

The algorithm is a simple simulated‑annealing / hill‑climbing search that
operates directly on the objective

    F(a) = 2·n·max_k (a * a)[k]       (the sum of a is kept equal to 1)

because the evaluation function `evaluate_sequence` is scale‑invariant.
The procedure is:

1. Choose a length `n`.  If the global variable `height_sequence_1`
   exists we reuse its length, otherwise we fall back to 2000.
2. Initialise a feasible sequence `a` (non‑negative, sum = 1) either from
   `height_sequence_1` or as a random uniform vector.
3. Repeatedly:
   * Compute the full self‑convolution with an FFT (O(n log n)).
   * Locate the index `k*` where the convolution attains its maximum.
   * Among the pairs `(i, k*‑i)` that contribute to this maximum,
     pick the pair with the largest product a_i·a_{k*‑i}.
   * Transfer a small amount `ε` from both members of this pair to a
     third, randomly chosen index (preserving the total sum).
   * Evaluate the new objective.  Accept the move if it improves the
     best value seen; otherwise accept it with a temperature‑controlled
     probability (simulated annealing) or revert to the current best.
   * Slowly decrease the step size and the temperature.
   * Occasionally add a tiny Gaussian jitter to avoid stagnation.

The search runs until the supplied time budget is almost exhausted and
returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: self‑convolution using a real FFT (fast for many calls)
def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a * a (full linear convolution) using FFT of length `fft_len`."""
    A = np.fft.rfft(a, n=fft_len)
    B = A * A
    conv_full = np.fft.irfft(B, n=fft_len)[: 2 * a.size - 1]
    return conv_full


# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises the evaluation function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds.
    **kwargs : dict
        Ignored – present for compatibility with the framework.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)

    # --------------------------------------------------------
    # Determine problem size -------------------------------------------------
    try:
        # Use the length of the provided construction if possible.
        init_seq = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        n = max(500, len(init_seq))                 # ensure a reasonable size
    except Exception:
        n = 2000                                     # fallback size

    # --------------------------------------------------------
    # Initialise two starting points and keep the better one ----------
    def _random_sequence():
        a = np.random.rand(n)
        a /= a.sum()
        return a

    # start from the known construction if it exists
    if 'height_sequence_1' in globals():
        a0 = np.array(height_sequence_1, dtype=float)  # type: ignore[name-defined]
        if a0.size != n:
            # truncate or pad with zeros
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.concatenate([a0, np.zeros(n - a0.size)])
        a0 = np.maximum(a0, 0.0)
        if a0.sum() == 0:
            a0 = _random_sequence()
        else:
            a0 /= a0.sum()
    else:
        a0 = _random_sequence()

    a1 = _random_sequence()

    # helper to compute the objective (sum is 1, so 2·n·max(b))
    fft_len = 1 << ((2 * n - 1).bit_length())

    def _objective(vec):
        conv = _self_convolution(vec, fft_len)
        return float(2.0 * n * np.max(conv))

    best_a = a0.copy()
    best_obj = _objective(best_a)

    obj1 = _objective(a1)
    if obj1 < best_obj:
        best_a = a1.copy()
        best_obj = obj1

    # current working sequence
    a = best_a.copy()

    # --------------------------------------------------------
    # Hyper‑parameters for the search
    step = 0.10                 # initial fraction of mass moved each iteration
    temperature = 0.01          # initial temperature for simulated annealing
    decay = 0.9995              # exponential cooling factor
    min_step = 1e-5             # lower bound for step size
    min_temp = 1e-6             # lower bound for temperature

    deadline = time.time() + budget_s - 2.0  # leave a tiny safety margin
    it = 0
    last_improve = 0

    while time.time() < deadline:
        # ----------------------------------------------------
        # 1) compute current self‑convolution and locate the peak
        conv = _self_convolution(a, fft_len)
        k_star = int(np.argmax(conv))

        # 2) find the pair (i, j = k_star‑i) with the largest product
        lo = k_star - (n - 1)
        if lo < 0:
            lo = 0
        hi = k_star if k_star < n else n - 1
        idx = np.arange(lo, hi + 1)
        prod = a[idx] * a[k_star - idx]
        i_max = idx[np.argmax(prod)]
        j_max = k_star - i_max

        # 3) propose a move: take a small ε from both i_max and j_max
        #    and place it (twice) on a third random index.
        eps = step * min(a[i_max], a[j_max])
        if eps > 0:
            # pick a destination index different from the two sources
            r = np.random.randint(n)
            while r == i_max or r == j_max:
                r = np.random.randint(n)

            a[i_max] -= eps
            a[j_max] -= eps
            a[r] += 2.0 * eps

            # numerical safety
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = _random_sequence()
            else:
                a /= a_sum

        # 4) evaluate the new candidate
        new_obj = _objective(a)

        # 5) acceptance rule
        if new_obj < best_obj:
            best_obj = new_obj
            best_a = a.copy()
            last_improve = it
        else:
            # simulated‑annealing acceptance of worse solutions
            if temperature > min_temp and np.random.rand() < math.exp((best_obj - new_obj) / temperature):
                # keep the worse state (exploration)
                pass
            else:
                # revert to the best known state
                a = best_a.copy()

        # ----------------------------------------------------
        # 6) schedule updates
        temperature = max(min_temp, temperature * decay)
        step = max(min_step, step * 0.9999)

        it += 1

        # occasional random jitter to shake the system out of a plateau
        if it % 1024 == 0:
            a += np.random.randn(n) * 1e-4
            a = np.maximum(a, 0.0)
            a /= a.sum()

        # progress report (optional, printed at most every few seconds)
        if it % 5000 == 0:
            print(f"Iter {it:6d} | best objective {best_obj:.6f} | step {step:.6f}")

        # if we have made no improvement for a long while, restart from a random point
        if it - last_improve > 20000:
            a = _random_sequence()
            last_improve = it

    # --------------------------------------------------------
    # Return the best sequence discovered
    return best_a.tolist()
