"""
Propose a candidate step‑function sequence that improves the bound.

The algorithm starts from a theoretically good shape a_i ≈ 1/√(i+1) whose
self‑convolution is nearly constant, giving a score ≈ π/2 ≈ 1.57 (better than
the uniform value 2).  From this seed we perform a few iterations of the
LP‑based refinement described in the problem statement:

1.  Normalise the current sequence so that its ℓ₁‑norm is sqrt(2·n).  Its score
    is then exactly the maximal entry of the convolution.
2.  Solve a linear programme that maximises the total mass of a new sequence
    g under the constraint conv(f,g) ≤ current_max (elementwise) and g ≥ 0.
3.  Rescale g to the same ℓ₁‑norm as f.
4.  Mix the two sequences.  The convolution of the mixture is a quadratic
    function of the mixing coefficient t; we evaluate this function on a dense
    grid and pick the t that gives the smallest maximal convolution entry.
5.  Accept the mixture if it improves the score and repeat while time permits.

The whole search is performed on a modest length (≤1000) to keep the LP
matrices small, but the final sequence is returned in the original length
(the first part of the seed) – longer sequences are not required for a better
bound.  Simple random perturbations are used if the LP fails.

The function `propose_candidate` follows the required signature and prints
progress information.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog

# ----------------------------------------------------------------------
# Helper: solve the LP  max 1·b  s.t. conv(f,b) ≤ rhs,  b ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """
    Solve:
        maximize   sum(b_i)
        subject to conv(f_seq, b)_k ≤ rhs   for all k
                    b_i ≥ 0
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f_seq)
    # Objective: maximize sum(b)  -> minimize -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.
    # For each k = 0 .. 2n-2 we have Σ_i f[i] * b[k-i] ≤ rhs,
    # where indices outside [0,n) are ignored.
    rows = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # indices j = k - i run from i_max..i_min (reverse order)
        idx = np.arange(i_min, i_max + 1)
        rows[k, idx] = f_seq[k - idx]

    # non‑negativity constraints: -I·b ≤ 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Solve with HiGHS (fast, no explicit time limit needed)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)   # safety clipping
    return None


# ----------------------------------------------------------------------
# Helper: line‑search for the optimal mixing coefficient t ∈ [0,1]
# ----------------------------------------------------------------------
def optimal_mix(f, g, sum_target):
    """
    Given two sequences f and g having the same ℓ₁‑norm (sum_target),
    find t∈[0,1] that minimises the maximal entry of the self‑convolution of
    (1−t)·f + t·g.
    Returns the mixed sequence and its score (max convolution entry).
    """
    n = len(f)
    # Pre‑compute the three convolutions needed for the quadratic form.
    A = np.convolve(f, f)
    B = np.convolve(f, g)
    C = np.convolve(g, g)

    # Because sum_target = sqrt(2·n), the evaluation score is just max(conv).
    # Sample t on a grid and then refine locally around the best point.
    ts = np.linspace(0.0, 1.0, 61)          # coarse grid
    max_vals = np.empty_like(ts)
    for i, t in enumerate(ts):
        combo = (1 - t) ** 2 * A + 2 * t * (1 - t) * B + t ** 2 * C
        max_vals[i] = combo.max()
    best_idx = int(np.argmin(max_vals))
    t_best = ts[best_idx]

    # Local refinement (simple ternary search on a small interval)
    left = max(0.0, t_best - 0.02)
    right = min(1.0, t_best + 0.02)
    for _ in range(12):
        t1 = left + (right - left) / 3.0
        t2 = right - (right - left) / 3.0
        combo1 = (1 - t1) ** 2 * A + 2 * t1 * (1 - t1) * B + t1 ** 2 * C
        combo2 = (1 - t2) ** 2 * A + 2 * t2 * (1 - t2) * B + t2 ** 2 * C
        m1 = combo1.max()
        m2 = combo2.max()
        if m1 < m2:
            right = t2
        else:
            left = t1

    t_opt = (left + right) / 2.0
    mixed = (1 - t_opt) * f + t_opt * g
    final_score = np.convolve(mixed, mixed).max()
    return mixed, final_score


# ----------------------------------------------------------------------
# Main candidate construction
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generate a step‑function height sequence with a low evaluation score.
    The method:
        * initialise with a 1/√(i+1) shape (theoretical near‑optimal);
        * repeatedly apply the LP‑based direction improvement and an optimal
          mixing step;
        * keep the best sequence found while respecting the time budget.
    Returns the best sequence as a plain Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Determine a modest length for the optimisation (≤1000 to keep LP cheap).
    if 'height_sequence_1' in globals():
        base_len = len(height_sequence_1)
    else:
        base_len = 800
    n_opt = min(1000, base_len)   # length used for the LP iterations

    # ------------------------------------------------------------------
    # Step 1: analytical seed – a_i = 1 / sqrt(i+1)
    # ------------------------------------------------------------------
    raw_seed = np.array([1.0 / np.sqrt(i + 1) for i in range(n_opt)], dtype=float)
    # Normalise to the required ℓ₁‑norm sqrt(2·n_opt)
    sum_target = np.sqrt(2.0 * n_opt)
    seed_seq = raw_seed * (sum_target / raw_seed.sum())

    best_seq = seed_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n_opt}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline and iteration < 30:
        iteration += 1
        # Current bound (max convolution)
        conv_f = np.convolve(best_seq, best_seq)
        M = conv_f.max()

        # Solve LP for a better direction g
        g_raw = solve_convolution_lp(best_seq, M)
        if g_raw is None:
            # LP failed – fall back to a random local perturbation
            idx = np.random.randint(n_opt)
            delta = (np.random.rand() - 0.5) * best_seq[idx] * 0.2
            new_seq = best_seq.copy()
            new_seq[idx] = max(0.0, new_seq[idx] + delta)
            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < best_score:
                best_seq, best_score = new_seq, new_score
                print(f"[perturb] iter={iteration}, new best={best_score:.6f}")
            continue

        # Normalise g to the same ℓ₁‑norm
        sum_g = g_raw.sum()
        if sum_g == 0:
            break
        g_norm = g_raw * (sum_target / sum_g)

        # Compute the optimal mixture of f and g
        mixed_seq, mixed_score = optimal_mix(best_seq, g_norm, sum_target)

        # Accept if we improve
        if mixed_score < best_score:
            best_seq, best_score = mixed_seq, mixed_score
            print(f"[mix] iter={iteration}, improved score={best_score:.6f}")
        else:
            # No improvement – perform a tiny random jiggle to escape plateaus
            idx = np.random.randint(n_opt)
            delta = (np.random.rand() - 0.5) * best_seq[idx] * 0.05
            jiggle = best_seq.copy()
            jiggle[idx] = max(0.0, jiggle[idx] + delta)
            jiggle_score = evaluate_sequence(jiggle.tolist())
            if jiggle_score < best_score:
                best_seq, best_score = jiggle, jiggle_score
                print(f"[jiggle] iter={iteration}, improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If the original problem expects a longer sequence, repeat the best
    # prefix to reach the original length (zero‑padding does not worsen the
    # bound because added zero entries leave the convolution unchanged).
    # ------------------------------------------------------------------
    if base_len > n_opt:
        # Pad with zeros to the original length
        padded = np.zeros(base_len, dtype=float)
        padded[:n_opt] = best_seq
        best_seq = padded

    print(f"[final] best score={best_score:.6f}, length={len(best_seq)}")
    return best_seq.tolist()
