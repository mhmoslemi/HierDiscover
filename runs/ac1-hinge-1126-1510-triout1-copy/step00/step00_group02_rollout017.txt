"""
Fast search for a low‑evaluation step‑function.

The evaluation metric is
    E(a) = 2 * n * max(a*a) / (sum(a)^2)

where `max(a*a)` is the peak of the linear convolution of the
non‑negative vector `a`.  The metric is scale invariant, therefore only
the *shape* of the sequence matters.

A classic family of shapes that makes the convolution almost flat is
the power‑law

        a_i = 1 / sqrt(i+1)        (i = 0 … n‑1)

which gives an asymptotic value E≈π/2≈1.571.  Starting from this
analytical template we perform two complementary optimisation phases
inside the allotted time (≈1000 s):

1. **LP‑guided improvement** – a tiny linear‑programming step that
   finds a direction `g` increasing the L1‑mass while keeping the
   current convolution peak unchanged.  A short convex‑mix with the
   current vector reduces the peak.

2. **Stochastic hill‑climbing** – random “mass‑transfer’’ moves between
   two indices (or a small perturbation of a single entry) are
   accepted only when they lower the metric.  The convolution is
   evaluated with an FFT for speed (O(n log n) per evaluation).

The algorithm keeps the best vector found and finally returns it as a
plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _fast_evaluate(seq: np.ndarray) -> float:
    """Return the same value as ``evaluate_sequence`` but much faster.

    The function applies the same clamps (≥0, ≤1000) and then computes
    the full linear convolution using an FFT.
    """
    # protect against negative / too large values – the optimizer never
    # produces them, but we follow the same rules as ``evaluate_sequence``.
    seq = np.clip(seq, 0.0, 1000.0)
    n = seq.size

    # size for a zero‑padded convolution (next power of two)
    size = 1 << ((2 * n - 1).bit_length())
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]

    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _solve_convolution_lp(f_seq: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve   maximise   Σ b_j
    subject to Σ_i f_i * b_{k-i} ≤ rhs   for all k,
               b_j ≥ 0

    The LP is small enough to be solved with SciPy's high‑level interface.
    If the solver fails or runs out of time ``None`` is returned.
    """
    n = f_seq.size
    # maximise Σ b  → minimise −Σ b
    c = -np.ones(n)

    # Build the matrix for the convolution constraints.
    # Row k contains f_i at column j = k-i whenever 0≤i,j<n.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    # Non‑negativity constraints  –  handled via bounds.
    A_ub = np.vstack(rows)
    b_ub = np.full(2 * n - 1, rhs)

    # Use the HiGHS interior‑point solver; give it a tiny time budget.
    try:
        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=(0, None),
                      method='highs',
                      options={'time_limit': 2.0})
    except Exception:          # pragma: no cover – safety net
        return None

    if res.success:
        return res.x
    return None


def _lp_step(seq: np.ndarray, t: float = 0.01) -> np.ndarray | None:
    """
    Perform one LP‑based improvement step.

    The current sequence is normalised to have l2‑norm √(2·n) (the same
    normalisation used in the reference algorithm).  The LP finds a
    feasible direction `g` that increases the total mass without raising
    the current convolution peak.  The new vector is a convex mixture
    (1‑t)·seq + t·g_norm.
    """
    n = seq.size
    sum_seq = seq.sum()
    if sum_seq == 0:
        return None

    # Normalise to the standard "√(2 n)" scale – the scaling does not
    # affect the metric, but the LP works better with numbers of order 1.
    norm_seq = seq * np.sqrt(2.0 * n) / sum_seq

    # Current peak of the convolution (with the normalised vector).
    rhs = _fast_evaluate(norm_seq) * (sum_seq * sum_seq) / (2.0 * n)

    g = _solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None

    # Rescale g back to the same l2‑norm as the original sequence.
    sum_g = g.sum()
    if sum_g == 0:
        return None
    g_norm = g * np.sqrt(2.0 * n) / sum_g

    # Convex combination – a small step to keep the process stable.
    new_seq = (1.0 - t) * seq + t * g_norm
    return new_seq


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence with a low evaluation value.

    The function respects the 1000 s time budget.  It returns a list of
    non‑negative floats (the step‑function heights).  The returned object
    is ready to be passed directly to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # leave a small safety margin

    # Length of the candidate.  1500–2000 is a good trade‑off between
    # asymptotic quality (the analytical shape improves with n) and
    # computational cost of the FFT.
    n = 1600

    # -----------------------------------------------------------------
    # 1️⃣  Initialise with the analytically good power‑law shape.
    seq = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=float))

    best_seq = seq.copy()
    best_score = _fast_evaluate(best_seq)
    print(f"initial score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 2️⃣  A few LP‑driven refinement steps (they are expensive, so we
    #     limit the number).  They quickly flatten the convolution.
    for lp_iter in range(12):
        if time.time() >= deadline:
            break
        cand = _lp_step(seq, t=0.02)
        if cand is None:
            continue
        cand_score = _fast_evaluate(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand                      # continue from the improved point
            print(f"LP step {lp_iter + 1:2d}: improved score = {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3️⃣  Stochastic hill‑climbing / simulated‑annealing phase.
    #     Moves consist of transferring a small amount of mass from one
    #     index to another (keeps the total sum unchanged, which is
    #     convenient because the metric is scale‑invariant).
    it = 0
    while time.time() < deadline:
        it += 1
        # ----- random move -----
        if np.random.rand() < 0.7:
            # transfer a tiny fraction from i to j
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            amount = np.random.rand() * 0.02 * seq[i]   # ≤2 % of source
            if amount <= 0:
                continue
            cand = seq.copy()
            cand[i] = max(0.0, cand[i] - amount)
            cand[j] = cand[j] + amount
        else:
            # single‑entry perturbation
            i = np.random.randint(0, n)
            delta = (np.random.rand() - 0.5) * 0.02 * seq[i]   # ±2 %
            cand = seq.copy()
            cand[i] = max(0.0, cand[i] + delta)

        # occasional smoothing operation (helps to escape shallow local minima)
        if np.random.rand() < 0.04:
            k = np.random.randint(1, n - 1)
            cand[k] = 0.5 * (cand[k - 1] + cand[k + 1])

        cand_score = _fast_evaluate(cand)

        # ----- acceptance -----
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand
            print(f"iter {it:6d}: new best = {best_score:.6f}")
        else:
            # rare Metropolis‑type acceptance to keep diversity
            if np.random.rand() < 0.001:
                seq = cand

    print(f"finished – best score = {best_score:.6f}")
    return best_seq.tolist()
