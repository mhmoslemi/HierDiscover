```
def local_search(seq, deadline):
    n = len(seq)
    target_sum = np.sqrt(2.0 * n)
    # parameters
    max_iter = int(1e6)
    # initial temperature
    T0 = 0.1 * np.max(seq)  # maybe
    T = T0
    decay = 0.99995
    best_seq = seq.copy()
    best_val = evaluate_fast(seq)
    cur_seq = seq.copy()
    cur_val = best_val
    iter = 0
    while time.time() < deadline and iter < max_iter:
        # compute convolution using FFT
        conv = convolution_fft(cur_seq)
        k_max = np.argmax(conv)
        # compute gradient via matched indices
        # build vector of gradient magnitude for each i
        # gradient_i = 2 * cur_seq[k_max - i] if within range else 0
        grad = np.zeros(n)
        low = max(0, k_max - (n-1))
        high = min(n-1, k_max)
        # i ranges low..high inclusive
        # For i in range(low, high+1): counterpart = k_max - i
        # assign grad[i] = 2 * cur_seq[counterpart]
        idx = np.arange(low, high+1)
        counterpart = k_max - idx
        grad[idx] = 2.0 * cur_seq[counterpart]
        # compute probabilities
        prob_up = grad / (grad.sum() + 1e-12)  # for selecting source index (high gradient)
        # source index i where moving mass reduces conv
        i = rng.choice(n, p=prob_up)  # can be zero-prob for many i
        # select destination j where gradient low (prefer low contribution)
        prob_down = (grad.max() - grad + 1e-12)
        prob_down = prob_down / prob_down.sum()
        j = rng.choice(n, p=prob_down)
        if i == j:
            continue
        # transfer amount delta
        max_transfer = cur_seq[i] * 0.5  # at most half of source
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.2  # up to 10% maybe
        # apply move
        cur_seq[i] -= delta
        cur_seq[j] += delta
        # ensure sum preserved
        # compute new value
        new_val = evaluate_fast(cur_seq)
        # acceptance rule
        if new_val < cur_val or rng.random() < np.exp((cur_val - new_val) / max(T, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = cur_seq.copy()
        else:
            # revert move
            cur_seq[i] += delta
            cur_seq[j] -= delta
        # Update temperature
        T *= decay
        iter += 1
    return best_seq, best_val
```

```
def propose_candidate(seed=0, budget_s=1000):
    import numpy as np
    import time

    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5

    # choose length
    n = len(globals().get('height_sequence_1', []))
    if n == 0:
        n = 1800  # default
    target_sum = np.sqrt(2.0 * n)

    # helper functions
    def evaluate_fast(seq):
        # compute max convolution using FFT for speed
        conv = np.fft.irfft(np.fft.rfft(seq, n=2*n) * np.fft.rfft(seq, n=2*n), n=2*n)
        max_b = conv[:2*n-1].max()
        sum_a = seq.sum()
        return 2.0 * n * max_b / (sum_a * sum_a)

    def conv_full_fft(seq):
        # returns full linear convolution (real)
        fft_len = 2 * n
        fa = np.fft.rfft(seq, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)
        return conv[:2*n-1]

    best_seq = None
    best_score = np.inf

    # number of independent runs (restarts)
    restart = 0
    while time.time() < deadline:
        restart += 1
        # init sequence
        if best_seq is not None and rng.random() < 0.3:
            seq = best_seq.copy()
        else:
            # random init
            seq = rng.random(n)
            seq = seq / seq.sum() * target_sum

        # local search
        cur_seq = seq
        cur_score = evaluate_fast(cur_seq)
        # parameters
        T = cur_score * 0.5  # initial temperature proportionally to score
        T_min = 1e-8
        decay = 0.9995
        # inner loop
        while time.time() < deadline:
            # compute full convolution and peak index
            conv = conv_full_fft(cur_seq)
            k_max = int(np.argmax(conv))
            # compute gradient vector (weights of contribution)
            grad = np.zeros_like(cur_seq)
            low = max(0, k_max - (n-1))
            high = min(n-1, k_max)
            idx = np.arange(low, high+1)
            counterpart = k_max - idx
            grad[idx] = 2.0 * cur_seq[counterpart]  # derivative magnitude

            # choose source index proportionally to gradient magnitude
            grad_sum = grad.sum()
            if grad_sum == 0:
                # all zero? fallback to random
                i = rng.integers(n)
                j = rng.integers(n)
            else:
                prob_src = grad / grad_sum
                i = rng.choice(n, p=prob_src)
                # choose destination with low gradient (inverse probability)
                inv = grad.max() - grad + 1e-12

                j = rng.choice(n, p=prob_dst)
```

```python
"""Propose candidate sequence via simulated annealing on the auto-convolution.
The algorithm keeps the total sum fixed (sqrt(2n)), moves mass between
coordinates to reduce the peak value of the linear convolution
`a * a`.  Gradient information of the current peak is used to bias the
source of the transfer, while the destination is chosen where the
gradient is small.  A simple exponential cooling schedule and occasional
random restarts provide exploration within the given time budget.
"""

import time
import numpy as np

def _evaluate_fast(seq: np.ndarray) -> float:
    # Fast evaluation based on the same formula as evaluate_sequence.
    n = seq.shape[0]
    # Compute full linear convolution via FFT (real input).
    # Pad to length 2*n for simple FFT.
    nfft = 2 * n
    spec = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(spec * spec, n=nfft)
    conv = conv[: 2 * n - 1]  # drop the extra zero at the end
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))


def _full_conv(seq: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of `seq` with itself (length 2n‑1)."""
    n = seq.shape[0]
    nfft = 2 * n
    spec = np.fft.rfft(seq, n=nfft)
    conv = np.fft.irfft(spec * spec, n=nfft)
    return conv[: 2 * n - 1]


def _make_initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random non‑negative sequence of length n with the required sum."""
    target_sum = np.sqrt(2.0 * n)
    seq = rng.random(n)  # uniform on [0,1)
    seq = seq / seq.sum() * target_sum
    return seq


def _improve_sequence(seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Run a simulated‑annealing style local search until time expires."""
    n = seq.shape[0]
    target_sum = np.sqrt(2.0 * n)

    cur = seq.copy()
    cur_score = _evaluate_fast(cur)

    best_seq = cur.copy()
    best_score = cur_score

    # Initial temperature proportional to the current score.
    T = cur_score * 0.5 if cur_score > 0 else 1.0
    T_min = 1e-8
    decay = 0.9995

    # Minimum step factor – we move at most this fraction of the source entry.
    max_frac = 0.2

    while time.time() < deadline:
        # Compute convolution and the index of the current maximum.
        conv = _full_conv(cur)
        k_max = int(np.argmax(conv))

        # Gradient of the peak entry w.r.t. each coordinate (up to a constant factor).
        grad = np.zeros_like(cur)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=int)
        counterpart = k_max - idx
        # derivative ∂conv[k_max]/∂a_i = 2 * a_{k_max-i}
        grad[idx] = 2.0 * cur[counterpart]

        # Choose a source index where the gradient is large (i.e. contributes heavily).
        grad_sum = grad.sum()
        if grad_sum == 0.0:
            src = rng.integers(n)
        else:
            prob_src = grad / grad_sum
            src = rng.choice(n, p=prob_src)

        # Choose a destination where the gradient is small (so the move hurts the peak little).
        inv = grad.max() - grad + 1e-12
        prob_dst = inv / inv.sum()
        dst = rng.choice(n, p=prob_dst)

        if src == dst:
            continue

        # Amount to transfer – a fraction of the source entry.
        max_transfer = cur[src] * max_frac
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random()

        # Apply the move.
        cur[src] -= delta
        cur[dst] += delta

        # Numerical safety.
        if cur[src] < 0.0:
            cur[src] = 0.0

        new_score = _evaluate_fast(cur)

        # Metropolis acceptance.
        if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = cur.copy()
                print(f"[{time.time():.2f}] New best {best_score:.6f}")
        else:
            # Revert.
            cur[src] += delta
            cur[dst] -= delta

        # Temperature cooling.
        T = max(T * decay, T_min)

        # Occasionally perform a tiny random perturbation to escape plateaus.
        if rng.random() < 0.001:
            i, j = rng.choice(n, size=2, replace=False)
            cur[i], cur[j] = cur[j], cur[i]

    # Return the best sequence found in this run and its score.
    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine uses a simple simulated‑annealing scheme
    that repeatedly moves mass between coordinates while keeping the total
    sum fixed (√(2 n)).  Gradient information of the current convolution peak
    guides the source of the transfer, and the destination is chosen where the
    peak is insensitive.  Multiple random restarts are performed until the
    time budget expires.  The best sequence found is returned as a plain
    Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # reserve a small safety margin

    # Determine a sensible length.
    # Prefer the length of the previously discovered construction if available.
    if 'height_sequence_1' in globals():
        n = len(globals()['height_sequence_1'])
        # Ensure we have a decent size; fall back if it is too short.
        if n < 500:
            n = 1800
    else:
        n = 1800  # a round number that works well for the current target

    target_sum = np.sqrt(2.0 * n)

    best_overall_seq = np.full(n, target_sum / n)  # uniform as a safe fallback
    best_overall_score = _evaluate_fast(best_overall_seq)

    # Include the known construction (if any) as one possible start.
    if 'height_sequence_1' in globals():
        try:
            start = np.array(height_sequence_1, dtype=float)
            # Rescale to the canonical sum without breaking the cap.
            start = start / start.sum() * target_sum
            score_start = _evaluate_fast(start)
            if score_start < best_overall_score:
                best_overall_seq = start.copy()
                best_overall_score = score_start
                print(f"Initial known sequence score {best_overall_score:.6f}")
        except Exception:
            pass

    # Main optimisation loop – perform independent restarts while we have time.
    restart = 0
    while time.time() < deadline:
        restart += 1
        # Build a fresh start (uniform, random or copy of the current best).
        if rng.random() < 0.3:
            seq0 = best_overall_seq.copy()
        else:
            seq0 = _make_initial_sequence(n, rng)

        # Allocate a slice of the remaining time for this run.
        run_deadline = min(deadline, time.time() + 0.25 * (deadline - time.time()))
        seq_improved, score_improved = _improve_sequence(seq0, run_deadline)

        if score_improved < best_overall_score:
            best_overall_score = score_improved
            best_overall_seq = seq_improved.copy()
            print(f"[{time.time():.2f}] Restart {restart}: new global best {best_overall_score:.6f}")

        # Early exit if we have already reached the target bound (with a tiny cushion).
        if best_overall_score < 1.51:
            break

    # Convert to plain Python list for the external evaluator.
    return best_overall_seq.tolist()
```
