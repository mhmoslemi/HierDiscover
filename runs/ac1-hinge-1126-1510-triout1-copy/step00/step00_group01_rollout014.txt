def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, random, math

    deadline = time.time() + budget_s - 5  # safety margin

    best_seq = None
    best_score = float('inf')
    # Option 1: geometric shapes.
    for n in range(50, 5001, 50):
        if time.time() > deadline:
            break
        r = 1.0 - 1.0/n  # parameter
        seq = [r**i for i in range(n)]
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
    # Additionally test a finer grid around n values maybe 100..2000.
    # Option 2: power law exponent.
    for n in [200,400,800,1600,3200]:
        if time.time() > deadline:
            break
        for p in np.linspace(0.1, 2.0, 20):
            seq = [1.0/(i+1)**p for i in range(n)]
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq

    # Option 3: random initializations maybe combine with simple hill climbing.
    # Try random perturbations of best seq for a few iterations.
    for _ in range(20):
        if time.time() > deadline:
            break
        if best_seq is None:
            break
        seq = best_seq.copy()
        # mutate: multiply each entry by (1 + epsilon) where epsilon small random.
        epsilon = 0.05
        seq = [max(0.0, x * (1 + (np.random.rand()-0.5)*epsilon)) for x in seq]
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq

    # fallback: return uniform if none.
    if best_seq is None:
        best_seq = [1.0]*100
        best_score = evaluate_sequence(best_seq)

    print(f"Best score: {best_score:.6f} with length {len(best_seq)}")
    return best_seq

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    import numpy as np, time, random, math

    deadline = time.time() + budget_s - 5
    random.seed(seed)
    np.random.seed(seed)
    best_seq = None
    best_score = float('inf')

    # Keep a list of n values to try
    n_candidates = [10, 20, 30, 40, 50, 60, 80, 100, 150, 200, 300, 500, 800, 1200]
    # but ensure not too large because evaluate cost ~n log n.

    for n in n_candidates:
        # quick check time
        if time.time() > deadline:
            break
        # generate initial sequence random
        seq = np.random.rand(n)
        # normalize sum to 1: scale to sum 1 (optional)
        seq = seq / seq.sum()
        current_score = evaluate_sequence(seq.tolist())
        if current_score < best_score:
            best_score = current_score
            best_seq = seq.tolist()
        # hill climbing
        # adapt step size
        step = 0.1
        for iter_ in range(2000):
            if time.time() > deadline:
                break
            i, j = random.sample(range(n), 2)
            # choose delta as proportion of seq[i]
            delta = step * (np.random.rand() - 0.5) * seq[i]  # could be positive or negative moving from i to j
            # Actually we want transfer from i to j: new_i = seq[i] - delta, new_j = seq[j] + delta
            # delta positive when moving from i to j, i reduces
            if delta > seq[i]:
                delta = seq[i]  # cannot go negative
            new_i = seq[i] - delta
            new_j = seq[j] + delta
            # Ensure non-negative
            if new_i < 0:
                continue
            # build new sequence copy
            new_seq = seq.copy()
            new_seq[i] = new_i
            new_seq[j] = new_j
            # evaluate
            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < current_score:
                seq = new_seq
                current_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
            # optionally reduce step gradually
            step *= 0.9999
    # fallback to simple pattern if nothing found
    if best_seq is None:
        best_seq = [1.0]*100
        best_score = evaluate_sequence(best_seq)
    print(f"Best score: {best_score:.6f}, length {len(best_seq)}")

"""
Propose a better step‑function sequence.

The evaluation metric is
 
    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

with n = len(a).
A lower value is better.  The trick is to increase the L1‑mass (the sum)
while keeping the maximal convolution value (the “peak”) as low as possible.
If we have a candidate f, we can look for another non‑negative vector g
with a larger L1‑mass such that conv(f,g) stays below the current peak.
That is a linear program:

    maximise   sum(g)
    subject to   conv(f,g) <= M
                  g >= 0

where M = max(conv(f,f)).  After solving, we scale g so that its
L2‑mass matches the normalisation used in the original research
(`sum(g) = sqrt(2n)`) and replace f by the scaled g.  Repeating this
quickly drives the bound down.

The implementation runs the LP for several different lengths (n up to
~500) and keeps the best sequence discovered within the allowed time.
A small amount of random perturbation is added when the LP fails, so the
search does not get stuck.
"""

import time
import random
import numpy as np
from scipy import optimize

# The evaluation function `evaluate_sequence` is assumed to be already
# imported in the execution environment.

def _conv_max(seq: np.ndarray) -> float:
    """Return the maximum value of the discrete convolution of seq with itself."""
    return np.max(np.convolve(seq, seq))


def _solve_lp(f_seq: np.ndarray, M: float):
    """
    Solve the linear programme:
        maximise    sum(b)
        subject to   conv(f_seq, b) <= M
                     b >= 0

    Returns a NumPy array `b` (the optimal vector) or None if the LP fails.
    """
    n = f_seq.size
    # Objective: maximise sum(b) => minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints.  For each possible shift k,
    # the linear combination of b coefficients equals conv(f,b)[k].
    A = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv(f,b)[k] = sum_i f[i] * b[k-i]
        # only terms where 0 <= i < n and 0 <= k-i < n contribute.
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] = f_seq[i]
        A.append(row)
        b.append(M)

    # Non‑negativity constraints: -b <= 0
    A.append(-np.eye(n))
    b.append(np.zeros(n))

    A = np.vstack(A)
    b = np.hstack(b)

    # Use the high‑performance HiGHS solver; give it a modest time budget.
    opt = optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if opt.success:
        return opt.x
    return None

    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function runs until the time budget (minus a
    small safety margin) is exhausted and returns the best sequence found.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 5.0   # keep a safety buffer

    # start from the known construction if it exists, otherwise uniform.
    try:
        best_seq = list(height_sequence_1)   # type: ignore[name-defined]
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)

    print(f"Initial score: {best_score:.6f} (len={len(best_seq)})")

    # try several problem sizes; smaller sizes give faster LP solves.
    for n in [80, 120, 180, 250, 350, 500]:
        if time.time() > deadline:
            break

        # initialise with a mildly decreasing geometric shape (helps LP)
        r = 0.9
        seq = np.array([r ** i for i in range(n)], dtype=float)
        seq = seq / seq.sum() * np.sqrt(2 * n)   # normalise as in the paper
        # main optimisation loop for this length
        while time.time() < deadline:
            M = _conv_max(seq)
            # attempt a linear‑programming step
            g = _solve_lp(seq, M)
            if g is not None:
                s = g.sum()
                if s < 1e-12:
                    # degenerate solution – break out of the inner loop
                    break
                # rescale to keep the normalisation used by the original method
                scale = np.sqrt(2 * n) / s
                new_seq = g * scale
                new_score = evaluate_sequence(new_seq.tolist())

                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(
                        f"Improved to {best_score:.6f} (len={n}) "
                        f"after LP step, M={M:.4f}"
                    )

                # replace the current sequence (full update, not a tiny mix)
                seq = new_seq
                # stop early if we already beat the target comfortably
                if best_score < 1.6:
                    break
            else:
                # LP failed – perform a tiny random perturbation to shake things up
                i, j = random.sample(range(n), 2)
                eps = 0.02 * seq[i]   # 2 % of the chosen component
                delta = (np.random.rand() - 0.5) * eps
                seq[i] = max(0.0, seq[i] - delta)
                seq[j] = max(0.0, seq[j] + delta)
                # re‑normalise to keep the same overall L2 magnitude
                norm = np.linalg.norm(seq)
                if norm > 0:
                    seq = seq * np.sqrt(2 * n) / norm

        # early exit if we have already reached a very good bound
        if best_score < 1.51:
            break

    # final sanity check – ensure we never return an empty or invalid sequence
    if not best_seq:
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)

    print(f"Finished search. Best score: {best_score:.6f} (len={len(best_seq)})")
    return best_seq
