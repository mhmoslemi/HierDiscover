"""
Propose a low‑score step‑function using a smooth “arcsine‑type” profile.
The construction is based on the known near‑optimal shape a[i] ≈ 1/√(i+δ).
A small projected‑gradient loop is applied to reduce the maximum of the
autoconvolution while keeping the ℓ1‑norm (and non‑negativity) fixed.
The routine searches a few lengths and offsets, keeps the best seed and
finishes with a few thousand cheap updates.  It respects the 1000 s budget
and never writes to the filesystem.
"""

import time
import numpy as np
from scipy import optimize  # fallback LP used by the original code


def _initial_sequence(length: int, exponent: float = 0.5,
                      offset: float = 0.0) -> np.ndarray:
    """
    Return a non‑negative sequence of the given length with the shape
    a[i] = (i + 1 + offset)^{-exponent}.  The values are already clipped
    to the [0, 1000] range required by ``evaluate_sequence``.
    """
    idx = np.arange(length, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -exponent)
    # safeguard against the 1000‑clamp (never triggered for these values)
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _project_onto_simplex(x: np.ndarray) -> np.ndarray:
    """
    Project non‑negative vector x onto the affine hyperplane sum(x)=S,
    where S is the original ℓ1‑norm of x.
    This is equivalent to scaling after setting negatives to zero.
    """
    s = np.sum(x)
    if s == 0.0:
        return x
    return x * (np.sum(x) / s)  # keep the original total mass


def _gradient_of_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution entry (index k) w.r.t. a.
    For ordered convolution the derivative is 2*a[k‑j] whenever
    0 ≤ k‑j < len(a); otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # j ranges where k‑j lies in [0, n‑1]
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if lo <= hi:
        j = np.arange(lo, hi + 1, dtype=np.int64)
        i = k - j
        grad[j] = 2.0 * a[i]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises
        2 * n * max(a*a) / (sum(a)**2)

    The algorithm:
      * try a handful of lengths (500‑4000) and small offset values;
        keep the best‑scoring seed;
      * run a cheap projected‑gradient descent that moves mass away
        from the indices that contribute most to the current maximal
        convolution value;
      * occasionally inject a tiny random perturbation to escape
        plateaus;
      * stop when the time budget is exhausted and return the best
        sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a safety margin

    # --------------------------------------------------------------------
    # 1) coarse grid search over length / offset
    # --------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    lengths = [800, 1200, 2000, 3200, 5000]   # keep them moderate for speed
    offsets = [0.0, 0.2, 0.5, 1.0]

    print("=== coarse grid search ===")
    for n in lengths:
        for off in offsets:
            seq = _initial_sequence(n, exponent=0.5, offset=off)
            try:
                score = evaluate_sequence(seq.tolist())
                print(f"len={n:4d} offset={off:3.1f} → score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
            except Exception:
                continue

    if best_seq is None:
        # fallback – a simple uniform vector (guaranteed to be valid)
        best_seq = np.ones(1000, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"Initial best: len={len(best_seq)} score={best_score:.6f}")

    # --------------------------------------------------------------------
    # 2) fine‑grained improvement with projected gradient descent
    # --------------------------------------------------------------------
    a = best_seq.copy()
    n = a.shape[0]

    # keep the ℓ1‑norm constant throughout the optimisation
    total_mass = float(np.sum(a))

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution and the index of the maximal entry
        conv = np.convolve(a, a)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # compute sub‑gradient for the active entry
        grad = _gradient_of_max_conv(a, k_max)

        # project gradient onto the hyperplane Σa = total_mass
        grad -= grad.mean()

        # simple diminishing step size
        step = 0.02 / (1.0 + iteration * 0.001)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)                     # enforce non‑negativity
        # re‑scale to keep the same ℓ1‑norm (scale invariance of the objective)
        cur_sum = np.sum(a_new)
        if cur_sum == 0.0:
            a_new = a.copy()
        else:
            a_new *= (total_mass / cur_sum)

        # occasional random jitter to escape flat regions
        if iteration % 200 == 0:
            jitter = np.random.randn(n) * 1e-3
            a_new = np.maximum(a_new + jitter, 0.0)
            a_new *= (total_mass / np.sum(a_new))

        # evaluate the new candidate
        try:
            score = evaluate_sequence(a_new.tolist())
        except Exception:
            # any numerical problem – reject this move
            a_new = a.copy()
            continue

        if score < best_score:
            best_score = score
            best_seq = a_new.copy()
            print(f"iter {iteration:5d}: improved score={best_score:.6f}")

        # accept the move if it does not worsen the objective by much;
        # this simple “accept‑if‑better‑or‑almost‑equal” rule keeps the
        # optimisation stable.
        if score <= best_score * 1.001:
            a = a_new
        else:
            # occasional rollback to the best known vector
            if np.random.rand() < 0.05:
                a = best_seq.copy()

    print("=== finished ===")
    print(f"final best score: {best_score:.6f}  length: {len(best_seq)}")
    return best_seq.tolist()
