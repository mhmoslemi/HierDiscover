"""
Propose a good step‑function sequence.

The algorithm consists of two stages:

1. Random search.
   • Choose a random length `n` (≈ 800–3000).
   • Choose a random support size `m` (≈ 35 %–80 % of `n`).
   • Put equal weight `1/m` on the `m` chosen positions, zero elsewhere.
   • Evaluate the sequence (the evaluation function is scale‑invariant, so the
     sum is forced to be exactly 1).
   • Keep the best sequence seen.

   Random supports make the self‑convolution almost flat: the maximal
   convolution value is roughly `1/(2 n)`, giving an evaluation close to 1,
   well below the target 1.503.

2. Light refinement (optional).
   • Starting from the best sequence, run a few steps of the LP‑based
     “direction” update described in the reference (solve a linear program
     that finds a direction `g` with larger total mass while keeping the
     convolution bounded, then move a small step towards `g`).
   • Because the LP is expensive we perform only a few steps, but they
     often improve the bound further.

The whole routine respects the supplied time budget (`budget_s`) and
returns the best sequence found.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the linear program:
        maximise   sum(b)
        subject to conv(f_seq, b) <= rhs   (for all convolution indices)
                     b >= 0
    Returns the optimal b as a 1‑D NumPy array, or None on failure.
    """
    n = len(f_seq)
    # Objective: maximise sum(b)  -> minimize -sum(b)
    c = -np.ones(n, dtype=float)

    # Build A_ub, b_ub for the convolution constraints.
    # For each k in [0, 2n-2] we have   sum_i f[i] * b[k-i] <= rhs,
    # where indices outside [0,n-1] are ignored.
    a_ub = np.zeros((2 * n - 1, n), dtype=float)

    for k in range(2 * n - 1):
        # i ranges over valid positions where both i and k-i are inside [0,n-1]
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # For each i, the coefficient of b_{k-i} is f[i]
        # So we fill the row accordingly.
        for i in range(i_min, i_max + 1):
            j = k - i
            a_ub[k, j] = f_seq[i]

    # Non‑negativity constraints: -b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n, dtype=float)])
    b_ub = np.hstack([np.full(2 * n - 1, rhs, dtype=float), np.zeros(n, dtype=float)])

    # Solve with HiGHS (default). 5‑second time limit to avoid getting stuck.
    try:
        res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs', options={'time_limit': 5})
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(seq):
    """
    Given a sequence `seq`, compute a better direction using the LP described
    in the reference.  Returns a new sequence (list of floats) or None if the
    LP fails.
    """
    n = len(seq)
    sum_seq = float(np.sum(seq))
    if sum_seq < 1e-12:
        return None

    # Normalise to the standard sum sqrt(2n) used by the reference.
    scale = math.sqrt(2.0 * n) / sum_seq
    f = [x * scale for x in seq]

    # Upper bound for the convolution (rhs of the LP).
    rhs = float(np.max(np.convolve(f, f)))

    # Solve the LP.
    g = solve_convolution_lp(f, rhs)
    if g is None:
        return None

    sum_g = float(np.sum(g))
    if sum_g < 1e-12:
        return None

    # Re‑normalise g in the same way.
    scale_g = math.sqrt(2.0 * n) / sum_g
    g_norm = [x * scale_g for x in g]

    # Small convex combination step.
    t = 0.02
    new_seq = [(1.0 - t) * a + t * b for a, b in zip(seq, g_norm)]
    return new_seq


def random_support_sequence(rng, n_min=800, n_max=3000,
                           prop_min=0.35, prop_max=0.80):
    """
    Produce a random non‑negative sequence:
        - length `n` uniformly drawn from [n_min, n_max]
        - exactly `m = floor(p * n)` entries set to 1/m, the rest zero,
          where `p` is uniformly drawn from [prop_min, prop_max].
    The sum of the sequence is exactly 1.
    Returns a plain Python list.
    """
    n = int(rng.integers(n_min, n_max + 1))
    p = float(rng.random() * (prop_max - prop_min) + prop_min)
    m = max(2, int(p * n))
    if m > n:
        m = n
    # Random support.
    idx = rng.choice(n, m, replace=False)
    seq = np.zeros(n, dtype=float)
    seq[idx] = 1.0 / m
    return seq.tolist()


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function sequence that minimises `evaluate_sequence`.
    The function runs for at most `budget_s` seconds (leaving a tiny safety
    margin) and returns the best sequence found.
    """
    # Random generator – deterministic given the seed.
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # Reserve a small amount of time for final return / cleanup.
    deadline = start_time + budget_s - 0.2

    # Allocate ~80 % of the budget to the cheap random search.
    phase1_deadline = start_time + 0.80 * (budget_s - 0.2)

    best_seq = None
    best_score = float('inf')
    iteration = 0

    # -------------------------------------------------
    # Phase 1 – fast random sampling of support patterns
    # -------------------------------------------------
    while time.time() < phase1_deadline:
        cand = random_support_sequence(rng)
        try:
            score = evaluate_sequence(cand)
        except Exception:  # pragma: no cover
            continue

        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[Random] iter {iteration:06d} → new best = {best_score:.6f} "
                  f"(len={len(cand)})")
        iteration += 1

        # Early exit if we are already comfortably below the target.
        if best_score <= 1.5:
            # Still allow a few more iterations to possibly improve further.
            if time.time() > phase1_deadline - 5.0:
                break

    # -------------------------------------------------
    # Phase 2 – optional LP‑based refinement (few steps)
    # -------------------------------------------------
    if best_seq is None:
        # Extremely unlikely – fall back to a tiny uniform sequence.
        best_seq = [1.0]
        best_score = evaluate_sequence(best_seq)

    # Run a limited number of refinement steps while time permits.
    refine_steps = 0
    while time.time() < deadline and refine_steps < 25:
        new_seq = get_good_direction_to_move_into(best_seq)
        if new_seq is None:
            break
        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:  # pragma: no cover
            break

        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
            print(f"[Refine] step {refine_steps:02d} → improved = {best_score:.6f}")

        # Prepare for the next iteration.
        best_seq = new_seq
        refine_steps += 1

    print(f"Finished search: best score = {best_score:.6f}, length = {len(best_seq)}")
    # Return as a plain list of floats (evaluate_sequence accepts that directly).
    return best_seq
