"""
Hill‑climbing optimizer for the step‑function bound.

The algorithm works on a fixed length `n` (chosen empirically) and a
fixed ℓ¹‑norm  ‖a‖₁ = √(2 n).  Starting from the uniform vector (which
gives a value of 2.0) we repeatedly:
  1. Compute the full autocorrelation `b = a * a` and locate the index
     `k_max` where `b` attains its maximum.
  2. Identify the pair of entries that contribute most to this peak:
     the donor index `i` maximising a[i]·a[k_max‑i].
  3. Move a modest amount of mass from that donor to a “safe” recipient
     – an index that does not increase the current peak (or, if none
     exists, the index with the smallest contribution to the peak).
  4. Accept the move only if the new max‑value (and therefore the
     evaluation score) does not increase.
The process is repeated until the time budget expires, keeping the best
sequence seen over all runs.  Random restarts and occasional
perturbations help escape local minima.

The returned sequence consists of non‑negative floats and can be any
length; the implementation tries several lengths (2000–6000) and picks
the best result obtained within the allotted time.
"""

import time
import numpy as np

def _update_convolution_on_delta(a, b, idx, delta):
    """
    Update the convolution `b = conv(a,a)` in‑place after changing a[idx] by `delta`.
    `a` and `b` are modified directly.
    """
    n = a.shape[0]
    old = a[idx]
    a[idx] = old + delta
    # Vectorised update for all j
    # b[idx + j] receives 2*delta*a[j] for every j.
    # For the diagonal term (j == idx) we have added 2*delta*new_a[idx];
    # the correct addition is 2*old*delta + delta**2, i.e. we added
    # an extra delta**2 which we subtract below.
    j = np.arange(n, dtype=int)
    b[idx + j] += 2.0 * delta * a  # a already contains the updated value at idx
    # fix the diagonal over‑addition
    b[2 * idx] -= delta * delta
    return a, b


def _convolve(a):
    """Return the linear convolution of a with itself."""
    return np.convolve(a, a)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    The function respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a small safety margin

    # Global best across all restarts
    best_seq = None
    best_score = float("inf")

    # Target ℓ¹‑norm for a given length n
    def target_sum(n):
        return np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Main search loop – several restarts with different lengths
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Randomly pick a length (the search works for any length >= 1000)
        n = int(np.random.choice([2000, 3000, 4000, 5000, 6000]))
        S = target_sum(n)

        # Uniform starting point (value = 2.0)
        a = np.full(n, S / n, dtype=np.float64)
        cur_score = evaluate_sequence(a.tolist())

        # Keep a copy as the currently accepted state
        cur_a = a.copy()
        cur_max = np.max(_convolve(cur_a))

        # Counter for stagnation – a long streak without improvement triggers a random restart
        no_imp = 0

        # ------------------------------------------------------------------
        # Hill‑climbing iterations for this restart
        # ------------------------------------------------------------------
        while time.time() < deadline:
            # Full convolution and current peak
            b = _convolve(cur_a)
            k_max = int(np.argmax(b))
            max_val = b[k_max]

            # ----- 1. choose donor (largest contribution to the peak) -----
            low_i = max(0, k_max - (n - 1))
            high_i = min(k_max, n - 1)
            i_range = np.arange(low_i, high_i + 1, dtype=int)
            partners = k_max - i_range

            # contributions of each possible donor to the current peak
            contrib = cur_a[i_range] * cur_a[partners]
            donor_idx = int(i_range[np.argmax(contrib)])

            # ----- 2. choose a safe recipient (ideally does not touch the peak) -----
            if k_max < n:                               # there exist indices > k_max
                safe = np.arange(k_max + 1, n, dtype=int)
            else:                                        # k_max >= n
                limit = k_max - (n - 1) - 1
                safe = np.arange(0, limit + 1, dtype=int) if limit >= 0 else np.array([], dtype=int)

            if safe.size > 0:
                rec_idx = int(np.random.choice(safe))
            else:
                # no completely safe position – pick the index with the *smallest* contribution
                all_i = np.arange(n, dtype=int)
                all_partners = k_max - all_i
                valid = (all_partners >= 0) & (all_partners < n)
                prod = np.zeros(n, dtype=np.float64)
                prod[valid] = cur_a[all_i[valid]] * cur_a[all_partners[valid]]
                prod[donor_idx] = np.inf                # never move mass back to the donor
                rec_idx = int(np.argmin(prod))

            # ----- 3. propose a move -------------------------------------------------
            # Move a fraction of the donor's mass; we limit the step size so the
            # neighbourhood is explored smoothly.
            eps = 0.10 * cur_a[donor_idx]                # 10 % of donor's value
            if eps <= 0.0:
                break
            a_candidate = cur_a.copy()
            a_candidate[donor_idx] -= eps
            a_candidate[rec_idx]   += eps

            # evaluate the candidate (scale invariant – we keep the ℓ¹ norm exactly S)
            cand_score = evaluate_sequence(a_candidate.tolist())

            # ----- 4. accept if we made progress ------------------------------------
            if cand_score < best_score:
                best_score = cand_score
                best_seq = a_candidate.copy()
                print(f"[{time.time() - start_time:6.2f}s] New global best: {best_score:.6f}")

            if cand_score < cur_score:
                cur_a = a_candidate
                cur_score = cand_score
                cur_max = np.max(_convolve(cur_a))
                no_imp = 0
                # early exit if we have already reached the target bound
                if cur_score < 1.503:
                    print("Target reached – exiting early.")
                    return cur_a.tolist()
                continue

            # occasional uphill acceptance to escape shallow local minima
            if np.random.rand() < 0.001:
                cur_a = a_candidate
                cur_score = cand_score
                cur_max = np.max(_convolve(cur_a))
                no_imp = 0
                continue

            # ----- 5. stagnation handling -------------------------------------------
            no_imp += 1
            if no_imp > 2000:
                # Random perturbation around the current state
                perturb = np.random.randn(n) * (S / n) * 0.02
                a_pert = cur_a + perturb
                a_pert = np.clip(a_pert, 0.0, None)
                # renormalise to keep the ℓ¹‑norm exactly S
                s = a_pert.sum()
                if s > 0:
                    a_pert *= (S / s)
                cur_a = a_pert
                cur_score = evaluate_sequence(cur_a.tolist())
                cur_max = np.max(_convolve(cur_a))
                no_imp = 0

        # --------------------------------------------------------------
        # End of one restart – try another length if time permits
        # --------------------------------------------------------------

    # ----------------------------------------------------------------------
    # Fallback: if we never improved or the global variable is available,
    # return it (the caller guarantees that a valid sequence exists).
    # ----------------------------------------------------------------------
    if best_seq is None:
        try:
            # `height_sequence_1` is supplied by the competition environment
            best_seq = np.array(height_sequence_1, dtype=np.float64)
        except Exception:                 # pragma: no cover
            # As a last resort return a tiny uniform vector (any valid sequence)
            n = 1000
            S = target_sum(n)
            best_seq = np.full(n, S / n, dtype=np.float64)

    return best_seq.tolist()
