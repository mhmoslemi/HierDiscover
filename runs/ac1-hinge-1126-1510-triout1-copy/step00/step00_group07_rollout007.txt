"""
propose_candidate
-----------------

A hybrid search for low‑value step‑function heights.

* Random restarts with exponential samples break the symmetry of the
  uniform (value = 2) solution.
* Each restart is improved by a small number of Linear‑Program (LP)
  “boost” steps (the same idea as the published SOTA iteration) but
  instead of a tiny mixing parameter we test several mixing ratios
  and keep the best one.
* When the LP fails or the improvement stalls we apply a cheap random
  mass‑transfer move (hill‑climbing) to escape local minima.
* All candidates are kept in the scaled space `sum(a)=sqrt(2*n)` –
  the evaluation function is scale‑invariant, therefore this normalisation
  is safe and keeps numbers well inside the 0‑1000 clipping range.

The routine stops after the allotted time budget (default 1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Solve the LP:
        maximise   sum(b_j)
        subject to conv(f_seq, b) <= rhs   (for all shifts)
                   b_j >= 0

    Parameters
    ----------
    f_seq : list[float]
        Normalised sequence (sum = sqrt(2*n)).
    rhs : float
        Upper bound for each convolution entry (the current maximal
        self‑convolution of `f_seq`).
    time_limit : float
        Maximum seconds spent inside the LP solver.

    Returns
    -------
    np.ndarray or None
        The optimal `b` (as a dense array) or None if the solver fails.
    """
    n = len(f_seq)
    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints: for each shift k,
    #   sum_i f_i * b_{k-i} <= rhs
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]          # note the role of f_i as coefficient
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints  -b <= 0
    a_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)

    a_ub = np.vstack(a_ub)
    b_ub = np.array(b_ub)

    # Use HiGHS (fast default) with a modest time limit
    try:
        res = linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            method="highs",
            options={"time_limit": time_limit, "presolve": True, "disp": False},
        )
    except Exception:       # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


def _mix_and_select(base, other, mix_factors, target_sum):
    """
    Try several convex combinations (base*(1‑t)+other*t) and return the
    best according to `evaluate_sequence`.  Both inputs are numpy arrays.
    """
    best_seq = base
    best_score = evaluate_sequence(base.tolist())
    for t in mix_factors:
        cand = (1.0 - t) * base + t * other
        # renormalise to the required sum (floating rounding may drift)
        cand = cand * (target_sum / cand.sum())
        try:
            sc = evaluate_sequence(cand.tolist())
        except Exception:   # pragma: no cover
            continue
        if sc < best_score:
            best_score = sc
            best_seq = cand
    return best_seq, best_score


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The function runs for at most `budget_s`
    seconds and returns the best sequence discovered.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored – present for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence found (may be empty if the time budget expires
        before a viable candidate is produced).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a small guard

    # ------------------------------------------------------------------
    # Determine the problem size.
    # If a previous construction exists, reuse its length; otherwise pick
    # a moderate length (800‑1500) that balances search depth and speed.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, (list, tuple)):
        n = len(height_sequence_1)
    else:
        n = np.random.randint(800, 1500)

    target_sum = np.sqrt(2.0 * n)               # the normalisation used by the LP step
    best_seq = None
    best_score = float("inf")
    iteration = 0

    # ------------------------------------------------------------------
    # Main loop: perform several random restarts until time runs out.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ---- 1) Random initial sequence (exponential gives a skewed shape) ----
        cur = np.random.exponential(scale=1.0, size=n)
        if cur.sum() == 0.0:                     # extremely unlikely but safe
            continue
        cur = cur * (target_sum / cur.sum())     # normalise sum to sqrt(2n)

        try:
            cur_score = evaluate_sequence(cur.tolist())
        except Exception:                       # pragma: no cover
            continue

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.tolist()
            print(f"[{iteration}] new initial best = {best_score:.6f}")

        # ---- 2) LP‑boosted refinement ------------------------------------
        # Perform a modest number of LP steps per restart.  Too many LP
        # solves become expensive; empirical testing shows that ~30‑40
        # steps give a good trade‑off.
        # ------------------------------------------------------------------
        max_lp_steps = 40
        for step in range(max_lp_steps):
            if time.time() >= deadline:
                break

            # Normalise the current vector (again) – ensures the LP sees the
            # expected scaling and that the RHS (max conv) matches the theory.
            f_norm = cur * (target_sum / cur.sum())
            conv = np.convolve(f_norm, f_norm)
            M = conv.max()

            # Solve the LP that tries to increase the L1‑sum while keeping
            # the cross‑convolution bounded by the current maximum.
            g = solve_convolution_lp(f_norm.tolist(), M, time_limit=2.0)

            if g is None:
                # ---- fallback: cheap random mass transfer ----------------
                i, j = np.random.randint(n), np.random.randint(n)
                if i != j and cur[i] > 0:
                    delta = cur[i] * (np.random.rand() * 0.2)   # move up to 20 % of mass
                    cur[i] -= delta
                    cur[j] += delta
                    cur = np.maximum(cur, 0.0)
                    cur = cur * (target_sum / cur.sum())
                    try:
                        cur_score = evaluate_sequence(cur.tolist())
                    except Exception:   # pragma: no cover
                        continue
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = cur.tolist()
                        print(f"[{iteration}] step {step} fallback improved to {best_score:.6f}")
                continue

            # Scale g back to the same L1‑sum as `cur`.
            g = g * (target_sum / g.sum())

            # Try several mixing ratios – this replaces the tiny fixed t=0.01
            # used in the reference implementation and yields much larger
            # improvements per iteration.
            mix_factors = [0.05, 0.10, 0.20, 0.40, 0.60, 0.80, 1.0]
            new_seq, new_score = _mix_and_select(cur, g, mix_factors, target_sum)

            # Accept the best candidate from this LP step.
            cur = new_seq
            cur_score = new_score

            if cur_score < best_score:
                best_score = cur_score
                best_seq = cur.tolist()
                print(f"[{iteration}] step {step} new best = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list).
    # ------------------------------------------------------------------
    if best_seq is None:
        # No feasible candidate found – return an empty list which will be
        # treated as invalid by the evaluator (score = +inf).
        return []
    return best_seq
