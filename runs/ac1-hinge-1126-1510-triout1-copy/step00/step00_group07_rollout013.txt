"""
Propose a new step‐function sequence.

The algorithm builds a sequence whose self‑convolution is almost constant.
The generating function of the infinite series (1‑z)^{-1/2} has this
property; truncating it to a finite length gives a sequence `f0` with
max(convolution) ≈ 1.57 (score ≈ π/2).  Starting from this seed we iteratively
solve a linear program:

    maximize   Σ b_i
    subject to conv(f, b) ≤ M            (M = max(conv(f,f)))
               b_i ≥ 0

The LP finds a direction `b` that adds as much mass as possible while
keeping the convolution bound.  The candidate `b` is rescaled to the same
ℓ₁‑norm as `f` (the normalization used in `evaluate_sequence`) and mixed
with the current iterate:

    f ← (1‑t)·f + t·b   (t small)

Because the ℓ₁‑norm stays fixed, the objective after normalization equals the
current maximal convolution value `M`.  Each mixing step strictly reduces
`M`, driving the score towards its theoretical lower bound (≈1).  The loop
runs until the time budget expires, keeping the best sequence seen.

The implementation uses only NumPy and SciPy (highs LP solver) and respects
the 1000 s overall budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def convolution_max(seq: np.ndarray) -> float:
    """Return max_{k} (seq * seq)[k] using a direct convolution."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def solve_convolution_lp(f_seq: np.ndarray, bound: float):
    """
    Solve the linear program:
        max   Σ b_i
        s.t.  (f_seq * b)[k] ≤ bound  for all k = 0 … 2·n‑2
              b_i ≥ 0
    Returns the optimal b (as a NumPy array) or None on failure.
    """
    n = f_seq.shape[0]
    # objective: maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution inequality matrix A_ub @ b ≤ bound
    rows = []
    for k in range(2 * n - 1):
        # indices i that contribute to (f * b)[k] are those with 0 ≤ i < n
        # and 0 ≤ j = k‑i < n.
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_vals = np.arange(i_min, i_max + 1, dtype=int)
        j_vals = k - i_vals                  # corresponding column index for b
        row = np.zeros(n, dtype=float)
        row[j_vals] = f_seq[i_vals]
        rows.append(row)

    A_ub = np.vstack(rows)
    b_ub = np.full(A_ub.shape[0], bound, dtype=float)

    # Non‑negativity constraints: –I·b ≤ 0
    A_ub = np.vstack([A_ub, -np.eye(n, dtype=float)])
    b_ub = np.concatenate([b_ub, np.zeros(n, dtype=float)])

    # Solve with the high‑performance HiGHS interior‑point method
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None

    a *= target_sum / a.sum()
