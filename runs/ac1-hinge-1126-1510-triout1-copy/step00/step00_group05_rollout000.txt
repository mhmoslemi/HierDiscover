    n = 2000

    # Ratio of the geometric progression.  Any r < 1 works; r = 0.5
    # provides a very rapid decay while keeping the early terms well
    # above floating‑point underflow.
    r = 0.5

    # Build the sequence a_k = r**k.
    # For k > ~1023 the value underflows to 0, which is harmless.
    sequence = [r ** k for k in range(n)]

    # The sum of the sequence is ~2, safely above the 0.01 cutoff.
    # No additional scaling is required because the score is
    # homogeneous of degree zero.
    return sequence
