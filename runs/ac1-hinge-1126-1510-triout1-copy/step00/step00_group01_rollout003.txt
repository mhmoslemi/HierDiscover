def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Generates a monotonic decreasing step function (linear or power-law) and finds the best
    exponent that minimizes `evaluate_sequence`.  The algorithm enumerates a small grid of
    exponents, keeps the best sequence, and returns it.  The search runs within the given
    time budget.
    """
    import numpy as np, math, time
    # Determine length n from existing global sequence if present
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 1526  # fallback
    # target normalization constant
    target_norm = math.sqrt(2 * n)
    best_seq = None
    best_score = float('inf')
    start = time.time()
    deadline = start + budget_s - 5.0   # a safety margin
    # grid for power exponent p
    # We'll scan from 0.1 to 4.0 with some steps, but adapt based on time
    p_vals = list(np.linspace(0.1, 4.0, 80))  # 80 values initially
    # also include exponential lambda values
    lam_vals = list(np.linspace(0.0001, 0.01, 80))
    # random order to reduce bias
    np.random.shuffle(p_vals)
    np.random.shuffle(lam_vals)
    # evaluate power shapes
    for p in p_vals:
        if time.time() > deadline:
            break
        raw = np.arange(1, n+1)[::-1] ** p  # decreasing sequence
        sum_raw = raw.sum()
        if sum_raw == 0:
            continue
        scale = target_norm / sum_raw
        seq = raw * scale
        # ensure within [0,1000] - scaling should guarantee
        # convert to Python list
        seq_list = seq.tolist()
        score = evaluate_sequence(seq_list)
        if score < best_score:
            best_score = score
            best_seq = seq_list
            print(f"New best p={p:.4f} score={score:.6f}")
    # evaluate exponential shapes
    for lam in lam_vals:
        if time.time() > deadline:
            break
        raw = np.exp(-lam * np.arange(n))
        sum_raw = raw.sum()
        if sum_raw == 0:
            continue
        scale = target_norm / sum_raw
        seq = raw * scale
        seq_list = seq.tolist()
        score = evaluate_sequence(seq_list)
        if score < best_score:
            best_score = score
            best_seq = seq_list
            print(f"New best lam={lam:.6f} score={score:.6f}")
    # Additional refinement: small random perturbations around best seq
    # Use simple hill climbing for remaining time
    while time.time() < deadline:
        # pick a random index and perturb its value a bit (preserving sum via renorm)
        seq = np.array(best_seq)
        idx = np.random.randint(n)
        # generate a small random delta
        delta = np.random.randn() * seq[idx] * 0.1  # up to 10%
        seq[idx] = max(0.0, seq[idx] + delta)
        # renormalize to target sum
        s = seq.sum()
        if s == 0:
            continue
        seq = seq * (target_norm / s)
        seq_list = seq.tolist()
        score = evaluate_sequence(seq_list)
        if score < best_score:
            best_score = score
            best_seq = seq_list
            print(f"perturb improvement score={score:.6f}")
    print(f"Final best score={best_score:.6f}")
    return best_seq

Propose a candidate step‑function sequence that minimizes
`evaluate_sequence`.  The algorithm searches over simple
parameterised families (power‑law and exponential decay) and
keeps the best sequence found within the allocated time budget.
No external files or network calls are used.
"""

import time
import math
import numpy as np

def _evaluate(seq: list[float]) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # The function `evaluate_sequence` is assumed to be already imported.
    return evaluate_sequence(seq)  # type: ignore[name-defined]


def _make_power_sequence(n: int, p: float) -> list[float]:
    """
    Create a monotone decreasing power‑law sequence of length n,
    scaled so that its L1 norm equals sqrt(2*n) (the normalisation
    used in the evaluation function).
    """
    # raw[i] = (n-i)^p  for i = 0..n-1  (largest value first)
    raw = (np.arange(1, n + 1)[::-1].astype(np.float64)) ** p
    sum_raw = raw.sum()
    if sum_raw == 0.0:
        # should never happen for p >= 0
        return [0.0] * n
    scale = math.sqrt(2.0 * n) / sum_raw
    seq = raw * scale
    return seq.tolist()


def _make_exp_sequence(n: int, lam: float) -> list[float]:
    """
    Create a decreasing exponential sequence a_i = exp(-lam*i),
    scaled to the required L1 norm.
    """
    raw = np.exp(-lam * np.arange(n, dtype=np.float64))
    sum_raw = raw.sum()
    if sum_raw == 0.0:
        return [0.0] * n
    scale = math.sqrt(2.0 * n) / sum_raw
    seq = raw * scale
    return seq.tolist()
