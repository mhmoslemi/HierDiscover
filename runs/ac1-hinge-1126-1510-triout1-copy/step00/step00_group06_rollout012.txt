def project_to_simplex(v, z=1):
    # v: array
    # project onto simplex {x : sum x = z, x >= 0}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def solve_convolution_lp(f, rhs):
    n = len(f)
    c = -np.ones(n)  # maximize sum(b)
    # Build A_ub and b_ub
    # A_ub rows: each convolution inequality sum_i f_i * b_{k-i} <= rhs
    # for each k in 0..2n-2
    rows = []
    b = np.full(2 * n - 1, rhs)
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i from max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # for i in [i_start, i_end], j = k - i
        # row[j] = f[i]
        # Equivalent: assign for j in [i_start, i_end]: row[j] = f[k-j]
        # Simpler: loop.
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([b, np.zeros(n)])
    # Solve
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 5.0})
    if result.success:
        return result.x
    return None

def project_to_simplex(v, target_sum):
    # Ensure non-negative after projection; algorithm from Duchi et al.
    # v: np.ndarray
    if target_sum <= 0:
        return np.zeros_like(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero
        theta = 0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def random_perturb(seq, eps=0.01):
    a = np.array(seq)
    n = len(a)
    i, j = np.random.randint(n), np.random.randint(n)
    delta = eps * np.mean(a)  # small value
    if a[i] > delta:
        a[i] -= delta
        a[j] += delta
    else:
        # transfer all from i to j
        a[j] += a[i]
        a[i] = 0.0
    # ensure nonneg, then project to simplex to keep sum same
    a = np.maximum(a, 0)
    # Not necessary to renormalize because we kept sum same

"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm
---------
* Work on several moderate lengths (n in the few hundreds).  The evaluation
  after normalising the sum to sqrt(2·n) is exactly the maximum of the
  self‑convolution, so we minimise that maximum.
* For a current sequence `f` (already normalised) compute the bound
  M = max(conv(f,f)).
* Solve a linear programme:
      maximise   Σ b_j
      subject to   conv(f,b) ≤ M   (2·n‑1 inequalities)
                   b_j ≥ 0
  The LP finds a direction `g` that can increase the total mass without
  increasing the current maximal convolution value.
* Normalise `g` to the same sum as `f` and try a few convex combinations
  `h = (1‑t)·f + t·g` (t in {0.2,0.5,0.8,1.0}).  The best feasible combination
  becomes the new iterate.
* If the LP fails, a small random mass transfer between two coordinates is
  performed (a simple perturbation).
* After each accepted move the sequence is projected onto the simplex
  `{x ≥ 0, Σx = sqrt(2·n)}` to keep the normalisation exact.
* The procedure is repeated under a global time budget (default 1000 s).
  The best sequence encountered over all lengths is returned.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog

# --------------------------------------------------------------------------- #
# Helper utilities (top‑level, no closures)
# --------------------------------------------------------------------------- #

def project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector ``v`` onto the simplex
    {x ≥ 0, Σ x = target_sum}.
    """
    if target_sum <= 0:
        return np.zeros_like(v)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def solve_convolution_lp(f: np.ndarray, rhs: float):
    """
    Solve the LP:
        maximise   Σ b_j
        subject to   f * b  ≤  rhs   (convolution bound for each shift)
                     b ≥ 0
    Returns the optimal vector ``b`` or ``None`` if the solver fails.
    """
    n = len(f)
    # objective: maximise Σ b_j  → minimise -Σ b_j
    c = -np.ones(n)

    # Build the convolution constraints A_ub @ b ≤ rhs
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)

    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])

    # 'highs' is fast and reliable for medium sized LPs
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=(0, None), method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
    except Exception:
        return None

    if res.success:
        return np.array(res.x, copy=True)
    return None


def mix_sequences(a: np.ndarray, b: np.ndarray, t: float) -> np.ndarray:
    """Convex combination (1‑t)·a + t·b."""
    return (1.0 - t) * a + t * b


def random_perturb(seq: np.ndarray, eps: float = 0.02) -> np.ndarray:
    """
    Transfer a small amount of mass between two random indices.
    The total sum is unchanged.
    """
    n = len(seq)
    i, j = np.random.randint(n), np.random.randint(n)
    # amount proportional to the current average value
    delta = eps * np.mean(seq)
    delta = min(delta, seq[i])  # cannot go negative
    if delta > 0:
        seq[i] -= delta
        seq[j] += delta
    return seq


# --------------------------------------------------------------------------- #
# Main entry point expected by the grader
# --------------------------------------------------------------------------- #

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the given
    evaluation function.  The routine follows the LP‑based improvement scheme
    described above and respects the 1000 s time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 0.1  # a little safety margin

    best_sequence = None
    best_score = float('inf')
    TARGET = 1.503  # stop early if we beat this bound

    # try a few different lengths (all are moderate so the LP stays fast)
    lengths = [120, 180, 240, 320, 400, 500, 640]
    random.shuffle(lengths)

    # mixing parameters we will test for each LP step
    T_VALUES = [0.2, 0.5, 0.8, 1.0]

    for n in lengths:
        if time.time() >= deadline:
            break

        # initialise a random feasible vector (sum = sqrt(2·n))
        target_sum = np.sqrt(2.0 * n)
        cur = np.random.rand(n)
        cur = cur / np.sum(cur) * target_sum
        cur = np.maximum(cur, 0.0)

        cur_score = evaluate_sequence(cur.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_sequence = cur.tolist()
            print(f"[init] n={n}, score={cur_score:.6f}")

        # early‑stop counters
        stagnant = 0
        MAX_STAGNANT = 30

        while time.time() < deadline and stagnant < MAX_STAGNANT:
            # current vector is already normalised (sum == target_sum)
            f_norm = cur
            M = np.max(np.convolve(f_norm, f_norm))

            # LP step – try to find a direction that can increase the total mass
            g = solve_convolution_lp(f_norm, M)

            improved_this_round = False

            if g is not None:
                # bring g to the same ℓ₁‑norm as f (sqrt(2·n))
                g_norm = g * (target_sum / np.sum(g))

                # try several mixings
                for t in T_VALUES:
                    cand = mix_sequences(f_norm, g_norm, t)
                    cand = np.maximum(cand, 0.0)
                    cand = project_to_simplex(cand, target_sum)

                    cand_score = evaluate_sequence(cand.tolist())
                    if cand_score < best_score - 1e-7:
                        best_score = cand_score
                        best_sequence = cand.tolist()
                        print(f"[update] n={n}, t={t:.2f}, new best={cand_score:.6f}")

                    if cand_score + 1e-9 < cur_score:   # improvement for the current walk
                        cur = cand
                        cur_score = cand_score
                        improved_this_round = True
                        break   # accept the first improving mix

            else:
                # LP failed – apply a cheap random perturbation
                cur = random_perturb(cur.copy(), eps=0.02)
                cur = project_to_simplex(cur, target_sum)
                cur_score = evaluate_sequence(cur.tolist())
                improved_this_round = True   # we treat it as a move

            if not improved_this_round:
                stagnant += 1
            else:
                stagnant = 0

            if best_score <= TARGET:
                print(f"[target] Achieved score {best_score:.6f} (≤{TARGET})")
                return best_sequence

    # fallback – if nothing was found, return a simple constant sequence
    if best_sequence is None:
        n = 200
        const = np.full(n, np.sqrt(2.0 * n) / n)
        best_sequence = const.tolist()
    return best_sequence
