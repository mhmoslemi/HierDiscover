def project_simplex(v):
    # Ensure v is numpy array
    n = len(v)
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    best_local_a = a.copy()
    best_local_score = evaluate_sequence(a.tolist())
    for it in range(max_iters):
        if time.time() >= deadline: break
        # compute conv via FFT
        conv = np.real(np.fft.ifft(np.fft.fft(a, L) * np.fft.fft(a, L)))
        max_val = conv.max()

def project_to_simplex(v):
    # v: 1d numpy array
    # returns w: projected onto simplex {w >= 0, sum w = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho = max i such that u_i - cssv_i / i > 0
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        # all negative, return uniform?
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

propose_candidate
-----------------

This implementation searches for a step‑function (a non‑negative list of heights)
that gives a low value of the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The strategy consists of three phases:

1. **Parametric sweep** –   create candidates of the form
       a[i] ∝ (i+1) ** p   (p∈[0,2])
   (and the reversed version).  For each exponent we normalise the vector so
   that Σa = sqrt(2·n) (the normalisation used by the reference code) and
   keep the best candidate.

2. **Simplex‑projected sub‑gradient descent** – starting from the best
   candidate, minimise the supremum norm of the self‑convolution.
   At each iteration the sub‑gradient of the current maximum entry is
   computed by a correlation (via FFT) and the vector is moved opposite
   to that gradient.  The step is projected onto the probability simplex
   (non‑negative, Σa = 1) which guarantees feasibility; after the loop the
   vector is scaled back to Σa = sqrt(2·n).

3. **Result** – the best sequence found (scaled to the required sum) is
   returned as a plain Python list.  Progress and intermediate scores are
   printed so that the optimisation process can be observed.

The code respects the time budget (default 1000 s) and uses only NumPy
(and the already‑imported `evaluate_sequence`) – no filesystem or network
access is required.
"""

import time
import math

# Helper: project a vector onto the probability simplex {x ≥ 0, Σx = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of v onto the simplex (non‑negative, sum = 1). """
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # Find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(seq: np.ndarray) -> float:

    max_b = conv.max()
    sum_a = seq.sum()
    return 2.0 * n * max_b / (sum_a * sum_a)
