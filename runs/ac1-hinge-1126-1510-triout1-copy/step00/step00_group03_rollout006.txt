def project_to_simplex(v):
    n = len(v)
    if v.sum() == 1 and (v>=0).all(): return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def _project_simplex(v):
    """
    Projects a vector onto the probability simplex (non-negative, sum=1).
    """
    n = v.shape[0]
    # shift v to ensure sum >0 etc.
    # algorithm from https://arxiv.org/abs/1309.1541
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _convolve(p):
    return np.convolve(p, p)

def _score(p):
    # assume p sum == 1
    b = _convolve(p)
    return 2 * len(p) * b.max()  # sum=1

def _optimize_one(n, max_iters, time_deadline):
    # simple gradient descent with random restarts
    best_p = None
    best_sc = np.inf
    it = 0
    while time.time() < time_deadline and it < max_iters:
        # random init Dirichlet alpha=1
        p = np.random.rand(n)
        p = p / p.sum()
        # enforce symmetry
        p = (p + p[::-1]) / 2
        
        # evaluate
        cur_sc = _score(p)
        # early improvement check
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_p = p.copy()
        # gradient descent loop for this start
        for step_i in range(200):
            if time.time() >= time_deadline:
                break
            b = _convolve(p)
            max_b = b.max()
            max_idxs = np.where(b >= max_b - 1e-12)[0]
            grad = np.zeros_like(p)
            # compute subgradient sum over all max indices
            for k in max_idxs:
                low = max(0, k - (n-1))
                high = min(k, n-1)
                i_vals = np.arange(low, high+1)
                j_vals = k - i_vals
                grad[i_vals] += 2 * p[j_vals]
            # step size diminishing
            eta = 0.2 / np.sqrt(step_i + 1)
            p_new = p - eta * grad
            # projection to simplex
            p_new = np.maximum(p_new, 0)
            if p_new.sum() == 0:
                p_new = np.random.rand(n)
            p_new = _project_simplex(p_new)
            # enforce symmetry
            p_new = (p_new + p_new[::-1]) / 2
            new_sc = _score(p_new)
            if new_sc < cur_sc:
                p = p_new
                cur_sc = new_sc
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_p = p.copy()
            else:
                # occasional acceptance to escape local minima
                if np.random.rand() < 0.01:
                    p = p_new
                    cur_sc = new_sc
        it += 1
    return best_p, best_sc

"""
Optimisation of a non‑negative step‑function using a smooth
loss that pushes the self‑convolution towards a flat distribution.
The loss is  L = Σ_k ( (f*f)_k – 1/(2n‑1) )² .
Its gradient can be written as a convolution, which allows very cheap
updates even for a few thousand coefficients.
A simple projected gradient descent with a decreasing step size,
periodic random perturbations and up‑sampling of the solution is used.
The routine returns the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Simple (fast) projection of a non‑negative vector onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_prob(p: np.ndarray) -> float:
    """
    Compute the evaluation score of a probability vector p
    (i.e. with sum(p) == 1).  The score is exactly what
    evaluate_sequence returns for such a vector:
        2 * n * max(p * p)
    """
    n = len(p)
    conv = np.convolve(p, p)
    return 2.0 * n * conv.max()


def _gradient_flatten(p: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k ( (p*p)_k – 1/(2n‑1) )² w.r.t. p.
    The formula reduces to a convolution:
        grad_i = 4 * ( (conv(p,p) – avg) * p_rev )[n‑1+i]
    where p_rev = p[::-1] and avg = 1/(2n‑1) (because sum(p)=1).
    """
    n = len(p)
    conv = np.convolve(p, p)                     # length 2n‑1
    avg = 1.0 / conv.size                         # = 1/(2n‑1)
    diff = conv - avg
    # convolution of diff with reversed p
    conv_grad = np.convolve(diff, p[::-1])        # length 3n‑2
    grad = 4.0 * conv_grad[n - 1: 2 * n - 1]       # take the n components we need
    return grad


def _upsample(p: np.ndarray, target_len: int) -> np.ndarray:
    """Duplicate entries to reach a larger length while keeping total mass = 1."""
    if len(p) == target_len:
        return p.copy()
    factor = target_len // len(p)
    if target_len % len(p) != 0 or factor < 1:
        # simple linear interpolation for non‑exact multiples
        x_old = np.linspace(0, 1, len(p))
        x_new = np.linspace(0, 1, target_len)
        return np.interp(x_new, x_old, p)
    # exact integer factor – repeat and split the mass equally
    up = np.repeat(p, factor) / factor
    # numerical drift correction
    up = up / up.sum()
    return up


# ----------------------------------------------------------------------
# Core optimisation for a single length
# ----------------------------------------------------------------------


def _optimise_length(p_init: np.ndarray,
                     time_limit: float,
                     max_iter: int = 10_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the flatten‑loss for a fixed length.
    Returns the best probability vector (sum = 1) and its score.
    """
    start = time.time()
    p = p_init.copy()
    n = len(p)

    # make sure we start on the simplex and (optionally) symmetric
    p = _project_to_simplex(p)
    p = (p + p[::-1]) / 2.0                     # enforce mirror symmetry
    p = _project_to_simplex(p)

    best_p = p.copy()
    best_sc = _score_from_prob(p)

    it = 0
    while time.time() - start < time_limit and it < max_iter:
        # gradient of the flat‑loss
        grad = _gradient_flatten(p)

        # step size – slowly decreasing
        eta = 0.3 / np.sqrt(it + 1)

        # gradient step + projection onto the simplex
        p_new = p - eta * grad
        p_new = np.maximum(p_new, 0.0)
        if p_new.sum() == 0.0:
            p_new = np.random.rand(n)
        p_new = _project_to_simplex(p_new)

        # small symmetry enforcement (helps the search space)
        p_new = (p_new + p_new[::-1]) / 2.0
        p_new = _project_to_simplex(p_new)

        # evaluate the true objective (max of the convolution)
        cur_sc = _score_from_prob(p)
        new_sc = _score_from_prob(p_new)

        # accept if we improve; otherwise keep the old point,
        # but occasionally accept a worsening move to escape saddles
        if new_sc < cur_sc:
            p = p_new
            cur_sc = new_sc
        elif np.random.rand() < 0.001:          # tiny stochastic acceptance
            p = p_new
            cur_sc = new_sc

        # keep the best ever seen
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_p = p.copy()
            print(f"  length {n} – iter {it:5d} – new best score {best_sc:.6f}")

        # occasional random jitter to avoid stagnation
        if it % 500 == 0 and np.random.rand() < 0.2:
            jitter = np.random.randn(n) * 0.02
            p = best_p + jitter
            p = np.maximum(p, 0.0)
            p = _project_to_simplex(p)
            p = (p + p[::-1]) / 2.0
            p = _project_to_simplex(p)

        it += 1

    return best_p, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    The algorithm works in stages:
      1. optimise a short vector (≈128 entries) from a random start,
      2. iteratively up‑sample the current best solution (×2 each stage)
         and continue the optimisation on the larger vector.
    A simple projected gradient descent on a loss that penalises
    deviations of the self‑convolution from a flat distribution is used.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Stage 0 – initialise a small random vector
    # ------------------------------------------------------------------
    lengths = [128, 256, 512, 1024]                  # progressively larger sizes
    # If a historic construction exists we can seed from it (down‑sampled)
    if 'height_sequence_1' in globals():
        try:
            old = np.array(height_sequence_1, dtype=float)
            # normalise and optionally cut/resize later
            old = old / max(old.sum(), 1e-12)
            # start from a down‑sampled version if it is not ridiculously large
            if len(old) > 128:
                p_cur = _upsample(old, 128)
            else:
                # pad with small random noise to reach 128
                p_cur = np.pad(old, (0, 128 - len(old)), 'constant')
                p_cur += np.random.rand(128) * 1e-6
                p_cur = _project_to_simplex(p_cur)
        except Exception:
            p_cur = np.random.rand(128)
            p_cur = _project_to_simplex(p_cur)
    else:
        p_cur = np.random.rand(128)
        p_cur = _project_to_simplex(p_cur)

    best_overall_p = None
    best_overall_score = np.inf

    # ------------------------------------------------------------------
    # Iterate over increasing resolutions
    # ------------------------------------------------------------------
    for target_len in lengths:
        now = time.time()
        if now >= deadline:
            break

        # time we allocate to this stage (roughly proportional to length)
        remaining = deadline - now
        # give at most 30% of the remaining budget to the next stage,
        # but never less than ~5 seconds.
        stage_time = max(5.0, min(remaining * 0.3, remaining - 5.0))

        # up‑sample current vector to the target size
        p_start = _upsample(p_cur, target_len)

        print(f"Starting optimisation for length {target_len} (time budget {stage_time:.1f}s)")
        p_opt, sc_opt = _optimise_length(p_start, stage_time)

        # keep the best overall candidate
        if sc_opt < best_overall_score:
            best_overall_score = sc_opt
            best_overall_p = p_opt.copy()
            print(f"*** New overall best: score {best_overall_score:.6f} (len={target_len})")

        # prepare for the next stage
        p_cur = p_opt.copy()

    # ------------------------------------------------------------------
    # Final sanity check – if nothing succeeded fall back to uniform
    # ------------------------------------------------------------------
    if best_overall_p is None or best_overall_p.size == 0:
        n_fallback = 256
        best_overall_p = np.full(n_fallback, 1.0 / n_fallback)

    # Ensure the sequence respects the security checks of `evaluate_sequence`
    best_overall_p = np.maximum(best_overall_p, 0.0)      # non‑negative
    best_overall_p = np.minimum(best_overall_p, 1000)    # sensible upper bound

    return best_overall_p.tolist()
