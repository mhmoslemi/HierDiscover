def propose_candidate(seed=42,budget_s=1000, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # length from previous best's length if exists, else set
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    # initialize
    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())
    # scaling: keep sum=1
    # schedule
    iter = 0
    lambda_min = 1.0
    lambda_max = 30.0
    lr0 = 0.1
    while time.time() < deadline:
        # compute lambda schedule: increase over iterations
        t = (time.time() - (deadline - budget_s + 10)) / (budget_s - 10)  # approx progress
        lam = lambda_min + (lambda_max - lambda_min) * t
        # compute L as power of 2 >= 3*n-2
        L = 1 << ((3*n-2).bit_length())
        # FFT of a (computed each iteration)
        fft_a = np.fft.rfft(a, n=L)
        # convolution to get b
        conv_full = np.fft.irfft(fft_a * fft_a, n=L)
        b = conv_full[:2*n-1]
        max_b = b.max()
        # compute weighted exp (softmax)
        # shift for numerical stability
        shift = max_b
        exp_term = np.exp(lam * (b - shift))
        w = exp_term / exp_term.sum()
        # correlation gradient via FFT
        fft_w = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(fft_w * np.conj(fft_a), n=L)
        grad = 2 * lam * corr_full[:n]
        # learning rate schedule
        lr = lr0 / math.sqrt(iter + 1)
        # update
        a = a - lr * grad
        # project
        a = np.maximum(a, 0)
        total = a.sum()
        if total == 0 or np.isnan(total):
            a = np.random.rand(n)
            a = a / a.sum()
        else:
            a = a / total  # renorm sum=1
        # evaluate
        if iter % 10 == 0:  # occasionally compute
            score = evaluate_sequence(a.tolist())

"""
Propose a better step‐function sequence.

The algorithm treats the problem as a smooth approximation of the
∞‑norm of the self‑convolution.  For a fixed sum of the coefficients the
objective is

    score = 2·n·max_k (a * a)_k .

We minimise a smooth surrogate  L(a)=log‑sum‑exp(λ·(a*a)) (λ>0) whose
gradient can be written analytically:

    ∂L/∂a_i = 2·λ·∑_k w_k·a_{k‑i},
    w_k = exp(λ·b_k)/∑_j exp(λ·b_j),   b = a*a .

The sums are correlations; they are evaluated efficiently with FFTs.
A simple momentum (heavy‑ball) update together with a projection onto the
probability simplex (non‑negative entries, unit sum) is used.
The smoothing parameter λ is increased linearly from a small value to a
larger one, so the optimisation first explores the landscape and then
focuses on the true max‑term.

Multiple random restarts, occasional small perturbations and an
adaptive learning rate help to escape plateaus.  The routine runs for at
most ``budget_s`` seconds and returns the best sequence encountered.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper utilities (all at top‑level, no closures)
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects a non‑negative vector onto the simplex {x≥0, sum(x)=1}.
    Simple renormalisation works because all components stay ≥0.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


def softmax_weights(b: np.ndarray, lam: float) -> np.ndarray:
    """
    Returns a probability distribution w_k = exp(lam·b_k) / Σ exp(lam·b_j).
    The subtraction of max(b) is used for numerical stability.
    """
    shift = b.max()
    exp_term = np.exp(lam * (b - shift))
    w = exp_term / exp_term.sum()
    return w


def convolution_fft(a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution a*a (full, length 2·len(a)-1) using FFT of length L.
    """
    fft_a = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * a.size - 1]
    return conv


def correlation_fft(w: np.ndarray, a: np.ndarray, L: int) -> np.ndarray:
    """
    Linear correlation (w ⋆ a) with full length len(w)+len(a)-1,
    computed via FFT of length L.
    Result length = len(w)+len(a)-1.
    """
    fft_w = np.fft.rfft(w, n=L)
    fft_a = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(fft_w * np.conj(fft_a), n=L)[: len(w) + len(a) - 1]
    return corr

# Main entry point
# ------------------------------------------------------------

    Search for a non‑negative coefficient sequence that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # tiny safety margin

    # ------------------------------------------------------------------
    # Determine the length of the sequence.
    # Use the existing construction if it exists, otherwise fall back
    # to a moderate size (2000).
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))  # type: ignore[name-defined]
    except Exception:
        n = 2000

    # ------------------------------------------------------------------
    # Initialisation helpers
    # ------------------------------------------------------------------
    def geometric_initial(alpha: float) -> np.ndarray:
        """
        Geometric profile a_i ∝ r^i with r = exp(-alpha/n).
        The distribution is normalised so that Σ a_i = 1.
        """
        r = math.exp(-alpha / n)
        # Normalisation factor for a truncated geometric series
        denom = 1.0 - r ** n
        if denom == 0.0:  # fallback to uniform
            return np.full(n, 1.0 / n)
        coeff = (1.0 - r) / denom
        return coeff * (r ** np.arange(n))

    def random_initial() -> np.ndarray:
        """Uniform Dirichlet random start."""
        x = np.random.rand(n)
        return x / x.sum()

    # Mix several initialisation strategies to give the optimiser variety.
    initialisers = [
        lambda: random_initial(),
        lambda: geometric_initial(alpha=np.random.uniform(0.5, 3.0)),
        lambda: geometric_initial(alpha=np.random.uniform(3.0, 6.0)),
        lambda: geometric_initial(alpha=np.random.uniform(6.0, 12.0)),

    # ------------------------------------------------------------

    L = next_pow2(3 * n - 2)        # FFT length sufficient for all convolutions
    beta = 0.9                      # momentum coefficient
    m = np.zeros_like(a)            # momentum accumulator
    iteration = 0

    # λ evolves from a small to a larger value (focuses on the true max)
    lam_min, lam_max = 0.5, 30.0

    lr_base = 0.7

    while time.time() < deadline:

        corr_full = correlation_fft(w, a, L)
        grad = 2.0 * lam * corr_full[:n]          # only the first n entries are needed

        # ------------------------------------------------------------------
        # Momentum update, learning‑rate schedule and projection
        # ------------------------------------------------------------------
        m = beta * m + (1.0 - beta) * grad
        lr = lr_base / (1.0 + iteration * 0.001)  # slow decay
        a = a - lr * m

        # Enforce the feasibility constraints (non‑negative, sum = 1)
        a = project_to_simplex(a)

        # ------------------------------------------------------------------
        # Occasionally evaluate the true objective and keep the best
        # ------------------------------------------------------------------
        if iteration % 20 == 0:

                if cur_score < best_score:

                    best_a = a.copy()
                    print(f"[iter {iteration:5d}] new best = {best_score:.6f}")
            except Exception:

                pass

        # ------------------------------------------------------------------
        # Small random perturbations to avoid stagnation
        # ------------------------------------------------------------------
        if iteration % 250 == 0 and iteration > 0:
            # Add a low‑amplitude Gaussian jitter, then re‑normalise
            a += 0.002 * np.random.randn(n)
            a = project_to_simplex(a)

        iteration += 1

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"[done] iterations={iteration}, best_score={best_score:.6f}")
    return best_a.tolist()
