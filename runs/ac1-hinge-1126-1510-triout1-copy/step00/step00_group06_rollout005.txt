"""
Propose a candidate sequence of non‑negative heights that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a refinement of the published “LP‑direction + small step’’
method:

1. Start from a uniform (or the supplied `height_sequence_1`) sequence.
2. At each iteration compute the current self‑convolution and its peak M.
3. Solve a *sparse* linear programme that maximises the total mass of a
   new sequence `g` under the constraints  conv(f, g) ≤ M  (elementwise)
   and  g ≥ 0.  Only the most‑binding convolution constraints
   (those with value close to M) are kept, keeping the LP small.
4. The mixture `f' = (1‑t) f + t g` is examined.  For a given t the
   self‑convolution expands to

        B(t) = (1‑t)²·(f*f) + 2 t (1‑t)·(f*g) + t²·(g*g)

   The function max_k B_k(t) is convex in t, so a bounded scalar
   optimisation (Brent) quickly finds the optimal t∈[0,1].
5. Replace `f` by the optimal mixture, clip to the allowed range
   [0, 1000] (the same clipping performed by `evaluate_sequence`),
   and keep the best sequence seen so far.
6. If the LP fails a small random perturbation is performed.
   The loop runs until the allotted time (≈1000 s) expires.

The combination of a direction that does not increase the peak
convolution and an optimal line‑search typically drives the bound well
below the naïve value 2.0 within the given budget.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize_scalar
from scipy import sparse

# ---------------------------------------------------------

def _solve_convolution_lp_sparse(f_seq: np.ndarray,
                                 rhs: float,
                                 active_ratio: float = 0.95,
                                 max_constraints: int = 300) -> np.ndarray | None:
    """
    Solve the LP:
        maximise   Σ b_j
        subject to conv(f, b) ≤ rhs  (elementwise)
                   b ≥ 0
    Only the most‑binding convolution constraints are kept to keep the
    problem small.  Returns the optimal vector `b` (as a NumPy array) or
    None if the LP did not succeed.
    """
    n = f_seq.shape[0]

    # Full self‑convolution to decide which constraints are tight
    conv_f = np.convolve(f_seq, f_seq)
    # Indices where the bound is almost active
    active_idxs = np.where(conv_f >= rhs * active_ratio)[0]

    # If we have too many constraints, keep only the largest ones
    if active_idxs.size > max_constraints:
        # sort by value descending and keep the top `max_constraints`
        order = np.argsort(conv_f[active_idxs])[::-1][:max_constraints]
        active_idxs = active_idxs[order]

    n_active = active_idxs.size
    # Build the sparse matrix for the active convolution constraints
    rows = []
    cols = []
    data = []

    for row, k in enumerate(active_idxs):
        # j runs over indices where both f_i and b_j appear in conv at position k
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1, dtype=int)
        coeffs = f_seq[k - js]          # f_i where i = k - j
        rows.extend([row] * len(js))
        cols.extend(js.tolist())
        data.extend(coeffs.tolist())

    # Non‑negativity constraints: -I * b ≤ 0
    rows.extend(range(n_active, n_active + n))
    cols.extend(range(n))
    data.extend([-1.0] * n)

    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(n_active + n, n))
    b_ub = np.concatenate([np.full(n_active, rhs), np.zeros(n)])

    # Objective: maximise Σ b_j  → minimise -Σ b_j
    c = -np.ones(n)

    # Use the HiGHS solver (fast & accepts sparse matrices)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


def _optimal_mixing_t(f, g, conv_f, conv_g, conv_fg):
    """
    For the mixture  f' = (1‑t) f + t g  compute the t∈[0,1] that
    minimises the maximal entry of the self‑convolution.
    The objective is convex, so a bounded scalar optimisation is safe.
    """
    def max_at_t(t):
        # quadratic combination of the three convolutions
        mixed = ((1 - t) ** 2) * conv_f + (2 * t * (1 - t)) * conv_fg + (t ** 2) * conv_g
        return mixed.max()

    # Brent’s method on a bounded interval; tolerance ~1e‑3 is enough
    opt = minimize_scalar(max_at_t, bounds=(0.0, 1.0), method='bounded',
                          options={'xatol': 1e-3, 'maxiter': 40})
    return float(opt.x) if opt.success else 0.0


def propose_candidate(seed: int = 42, budget_s: int = 970, **_):
    """
    Main entry point used by the evaluation harness.

    Returns a list of non‑negative floats (the heights) that gives the
    best bound discovered within the allocated time.
    """
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Initialise the sequence
    # ----------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        f = np.array(height_sequence_1, dtype=float)   # may already be good
    else:
        # fallback: a uniform sequence of a moderate length
        n = 2000
        f = np.full(n, np.sqrt(2 * n) / n, dtype=float)

    n = f.shape[0]
    # Guarantee a reasonable starting sum (the evaluation is scale‑invariant)
    if f.sum() == 0:
        f[:] = np.sqrt(2 * n) / n

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    print(f"Initial length {n}, initial score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2. Main optimisation loop
    # ----------------------------------------------------------------------
    deadline = time.time() + budget_s - 5   # a few seconds safety margin
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        # current self‑convolution and its peak
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # ------------------------------------------------------------------
        # 2a. Obtain a direction `g` from a sparse LP
        # ------------------------------------------------------------------
        g = _solve_convolution_lp_sparse(f, M,
                                         active_ratio=0.95,
                                         max_constraints=250)
        if g is None:
            # LP failed – perform a tiny random redistribution
            i, j = np.random.choice(n, size=2, replace=False)
            delta = min(f[i], f[i] * 0.01)
            f[i] -= delta
            f[j] += delta
            continue

        # ------------------------------------------------------------------
        # 2b. Compute the convolutions needed for the line search
        # ------------------------------------------------------------------
        conv_g = np.convolve(g, g)
        conv_fg = np.convolve(f, g)

        # ------------------------------------------------------------------
        # 2c. Find the optimal mixing coefficient t
        # ------------------------------------------------------------------
        t_opt = _optimal_mixing_t(f, g, conv_f, conv_g, conv_fg)

        # If the optimiser returns an extreme value we fall back to a small step
        if not (0.0 < t_opt < 1.0):
            t_opt = 0.03

        # ------------------------------------------------------------------
        # 2d. Form the new candidate and clip to the allowed range
        # ------------------------------------------------------------------
        f_new = (1.0 - t_opt) * f + t_opt * g
        f_new = np.clip(f_new, 0.0, 1000.0)    # mirrors evaluate_sequence’s clipping

        # ------------------------------------------------------------------
        # 2e. Evaluate and keep the best result seen so far
        # ------------------------------------------------------------------
        try:
            score = evaluate_sequence(f_new.tolist())
        except Exception:
            # Defensive – should never happen
            score = np.inf

        if score < best_score:
            best_score = score
            best_seq = f_new.copy()
            no_improve = 0
            print(f"Iter {iteration:4d}: improved to {best_score:.6f} (t={t_opt:.3f})")
        else:
            no_improve += 1

        # --------------------------------------------------------------
        # 2f. Prepare for the next iteration
        # --------------------------------------------------------------
        f = f_new
        iteration += 1

        # Occasionally inject a small random perturbation to escape stagnation
        if no_improve > 150 and np.random.rand() < 0.5:
            perturb = np.random.randn(n) * 0.005 * f.mean()
            f = np.clip(f + perturb, 0.0, 1000.0)
            no_improve = 0
            print(f"Iter {iteration:4d}: random perturbation injected")

    # ----------------------------------------------------------------------
    # 3. Return the best sequence found
    # ----------------------------------------------------------------------
    print(f"Finished after {iteration} iterations, best score {best_score:.6f}")
    return best_seq.tolist()
