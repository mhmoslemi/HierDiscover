def solve_lp_g(f_seq, rhs):
    # solve LP maximize sum(b) s.t. conv(f,b) <= rhs, b >= 0
    n = len(f_seq)
    c = -np.ones(n)  # maximize sum(b)
    # Build A_ub matrix rows for each conv index
    # For each k=0..2n-2, row[j] = f_seq[i] where i = k - j and 0 <= i < n
    # Equivalent to Toeplitz.
    A = []
    b = []
    for k in range(2*n - 1):
        row = np.zeros(n)
        # i loops over possible indices where 0 <= i < n and 0 <= k-i < n
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # we want row[j] = f_seq[i] where j = k - i => i = k - j
        # so for each i in range i_start..i_end:
        # j = k - i
        # row[j] = f_seq[i]
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f_seq[i]
        A.append(row)
        b.append(rhs)
    # Non-negativity constraints: -I x <= 0
    A = np.vstack([A, -np.eye(n)])
    b = np.hstack([b, np.zeros(n)])
    # Solve LP using high-performance method.
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 5})
    if res.success:
        return np.array(res.x)
    else:
        return None

def optimal_mix_parameter(f, g):
    # compute conv arrays
    Cf = np.convolve(f, f)
    Cg = np.convolve(g, g)
    Cfg = np.convolve(f, g)
    # define function to compute max conv for t
    def max_conv(t):
        combo = (1 - t) * (1 - t) * Cf + t * t * Cg + 2 * t * (1 - t) * Cfg
        return float(np.max(combo))
    # ternary search on [0,1]
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        if max_conv(t1) < max_conv(t2):
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0

"""
Propose a step‐function (list of non‑negative heights) that minimises
evaluate_sequence.

The algorithm is an iterative improvement loop consisting of three parts:

1.  **Linear‑programming direction** – given the current sequence f,
    solve a LP that finds a non‑negative sequence g with a larger ℓ₁‑norm
    while keeping the convolution (f∗g) bounded by the current max
    convolution M = max(f∗f).  Afterwards g is rescaled to have exactly the
    same total sum as f.

2.  **Optimal mixing** – for any 0 ≤ t ≤ 1 the mixture
    h(t) = (1‑t)·f + t·g satisfies the same ℓ₁‑norm.  Its self‑convolution
    can be written analytically:
        conv_h(t) = (1‑t)²·(f∗f) + t²·(g∗g) + 2t(1‑t)·(f∗g) .
    The worst (maximum) entry of this convolution is a univariate function
    of t.  We minimise it with a cheap ternary search, obtaining the best
    mixing coefficient t_opt.

3.  **Exploration** – if the LP fails or the mixing does not improve the
    objective we perform a tiny random perturbation of the current
    sequence.  Occasionally a larger random jitter is injected to escape
    shallow local minima.

The loop runs until the allocated time budget is exhausted (≈ 1000 s)
and the best sequence seen so far is returned.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _solve_lp_for_g(f_seq: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the linear program

        maximise   sum(b)
        subject to conv(f_seq, b) ≤ rhs  (for every convolution index)
                   b ≥ 0

    The variable b has the same length as f_seq.  Returns the optimal b
    (as a NumPy array) or None if the solver fails.
    """
    n = len(f_seq)
    # objective: maximise sum(b) → minimise -sum(b)
    c = -np.ones(n)

    # Build the inequality matrix A_ub * b ≤ b_ub.
    # There are (2n‑1) convolution constraints.
    rows = []
    b_ub = np.full(2 * n - 1, rhs, dtype=float)

    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        # i runs over indices of f that contribute to (f∗b)[k]
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i                     # the index of b that pairs with f[i]
            row[j] = f_seq[i]
        rows.append(row)

    A_ub = np.vstack(rows)

    # Use bounds (0, ∞) for non‑negativity – no need to add explicit rows.
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"disp": False, "time_limit": 8.0},
        )
    except Exception:  # pragma: no cover – safety
        return None

    if res.success:
        return np.array(res.x, dtype=float)
    return None


def _optimal_mix_parameter(
    f: np.ndarray, g: np.ndarray, max_iters: int = 30
) -> float:
    """
    Find t∈[0,1] minimising max(conv_h(t)) where
        h(t) = (1‑t)·f + t·g
    and conv_h(t) = (1‑t)²·(f∗f) + t²·(g∗g) + 2t(1‑t)·(f∗g).

    A simple ternary search (unimodal assumption) is sufficient because
    the envelope of quadratic functions is convex‑like.
    """
    cf = np.convolve(f, f)
    cg = np.convolve(g, g)
    cfg = np.convolve(f, g)

    def _max_at(t: float) -> float:
        mix = (1.0 - t) * (1.0 - t) * cf + t * t * cg + 2.0 * t * (1.0 - t) * cfg
        return float(np.max(mix))

    lo, hi = 0.0, 1.0
    for _ in range(max_iters):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        if _max_at(t1) < _max_at(t2):
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights with a small value of
    evaluate_sequence.  The method combines LP‑based direction finding,
    an analytically optimal mixing step, and occasional random perturbations.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0  # spare a couple of seconds for cleanup

    # ------------------------------------------------------------------
    # 1️⃣  Initial sequence ------------------------------------------------
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the surrounding harness
        init_seq = list(height_sequence_1)  # type: ignore[name-defined]
        # keep a reasonable size (the LP cost grows with n)
        if len(init_seq) > 1800:
            # randomly crop a contiguous block of reasonable length
            start = np.random.randint(0, len(init_seq) - 1500)
            init_seq = init_seq[start : start + 1500]
    except Exception:
        # fallback: random start of moderate length
        n0 = np.random.randint(300, 1200)
        init_seq = np.random.rand(n0).tolist()

    # Ensure valid numeric, non‑negative values
    init_seq = [float(max(0.0, x)) for x in init_seq]

    best_seq = init_seq.copy()
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:  # pragma: no cover
        best_score = float("inf")

    curr_seq = best_seq.copy()
    iteration = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # 2️⃣  Main optimisation loop -----------------------------------------
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1
        f = np.array(curr_seq, dtype=float)

        # Current bound M = max(f∗f)
        conv_f = np.convolve(f, f)
        M = float(np.max(conv_f))

        # ---------- LP step ------------------------------------------------
        g_raw = _solve_lp_for_g(f, M)
        if g_raw is None:
            # LP failed – apply a tiny random jitter and continue
            idx = np.random.randint(len(curr_seq))
            delta = np.random.randn() * 0.02 * max(1.0, curr_seq[idx])
            curr_seq[idx] = max(0.0, curr_seq[idx] + delta)
            continue

        # Rescale g so that it has the same ℓ₁‑norm as f
        sum_f = float(np.sum(f))
        sum_g = float(np.sum(g_raw))
        if sum_g <= 0.0:
            # degenerate solution – skip
            continue
        scale = sum_f / sum_g
        g = (g_raw * scale).astype(float)

        # ---------- Optimal mixing -----------------------------------------
        t_opt = _optimal_mix_parameter(f, g)

        # Mixture h = (1‑t)·f + t·g
        new_seq = [(1.0 - t_opt) * a + t_opt * b for a, b in zip(curr_seq, g)]

        # Occasionally inject a small extra jitter to avoid stagnation
        if iteration % 23 == 0 and np.random.rand() < 0.15:
            for _ in range(3):
                j = np.random.randint(len(new_seq))
                new_seq[j] = max(0.0, new_seq[j] + np.random.randn() * 0.01)

        # ---------- Evaluate ------------------------------------------------
        try:
            score = evaluate_sequence(new_seq)
        except Exception:
            # Should not happen because we already guarded the values
            continue

        # Keep the best solution seen so far
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            no_improve = 0
            print(
                f"Iter {iteration:4d}: new best score = {best_score:.6f} "
                f"(mix t={t_opt:.4f})"
            )
        else:
            no_improve += 1

        # Decide the next current sequence
        # If the mixing reduced the maximal convolution we keep it,
        # otherwise we fall back to a mild random perturbation.
        if score <= best_score + 1e-12:
            curr_seq = new_seq
        else:
            # deteriorated – random walk around the current best
            idx = np.random.randint(len(curr_seq))
            delta = np.random.randn() * 0.02 * max(1.0, curr_seq[idx])
            curr_seq[idx] = max(0.0, curr_seq[idx] + delta)

        # If we have been stuck for many iterations, re‑initialise a bit
        if no_improve >= 150:
            # start from a slightly perturbed copy of the best sequence
            curr_seq = [
                max(0.0, x + np.random.randn() * 0.01) for x in best_seq
            ]
            no_improve = 0

    # ------------------------------------------------------------------
    # 3️⃣  Return the best sequence found ----------------------------------
    # ------------------------------------------------------------------
    return best_seq
