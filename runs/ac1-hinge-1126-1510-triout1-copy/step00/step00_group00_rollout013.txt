"""
Algorithm: start from the exact Catalan (central binomial) coefficients,
which are known to give an asymptotically optimal self‑convolution
(L∞) ratio π/2 ≈ 1.571.  The sequence is scaled to unit ℓ¹–norm.
A cheap stochastic local search then transfers a small amount of mass
from an early index to a later one (preserving non‑negativity and the
total sum) whenever this reduces the evaluation score.
The search runs until the time budget expires and returns the best
sequence found.
"""

import numpy as np, time

def _catalan_sequence(n: int) -> np.ndarray:
    """Return the first n terms of binom(2k,k)/4**k as float64."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for k in range(1, n):
        # recurrence a[k] = a[k‑1] * (2k)*(2k‑1) / (4*k*k)
        a[k] = a[k-1] * (2.0 * k) * (2.0 * k - 1.0) / (4.0 * k * k)

    fx = np.fft.rfft(x, n=nfft)
