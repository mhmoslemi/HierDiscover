def gradient_step(seq):
    n = len(seq)
    conv = convolution(seq)
    k_max = np.argmax(conv)
    # get rev vector
    rev = np.zeros(n)
    lo = max(0, k_max - (n-1))
    hi = min(n-1, k_max)
    rev[lo:hi+1] = seq[k_max - np.arange(lo, hi+1)]
    # direction: -rev
    step = 0.05 * np.mean(seq)  # maybe small
    delta = step * rev
    new = seq - delta
    # ensure nonnegative
    new = np.maximum(new, 0)
    if np.sum(new) == 0:
        return seq
    # renormalize
    new = new * (np.sqrt(2*n) / np.sum(new))
    return new

"""
Propose a step‑function that improves the evaluation score.
The algorithm mixes three simple ideas:

1. **Gradient‑descent on the current peak** –  the convolution peak
   occurs at index k_max.  Its derivative w.r.t. a_i is
   2·a_{k_max‑i} (zero if the index is out of range).  Moving a small
   amount opposite to this gradient reduces the peak while keeping the
   total sum (≈√(2·n)) fixed.

2. **LP‑direction (the published SOTA move)** –  solve the linear program
   “maximise Σ b_j  subject to  (f * b) ≤ M,  b ≥ 0”, where M is the
   current maximum of f * f.  The solution g can be mixed with the
   current sequence.  A short line‑search over the mixing weight t
   (0 ≤ t ≤ 1) is performed and the best mix is kept.

3. **Random perturbations / restarts** –  occasionally a small random
   jitter is added or a fresh random start is taken to escape local
   minima.

The main loop runs until the allocated time (≈ 1000 s) is exhausted,
always keeping the best sequence seen so far.  All heavy work is done
with NumPy and SciPy (FFT for fast convolution, ``linprog`` for the LP).

The returned list is directly usable by the supplied ``evaluate_sequence``.
"""

import time
import random
import math
import numpy as np
from scipy import optimize

# ---------- helper utilities -------------------------------------------------

def next_pow_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast version of evaluate_sequence (used internally for decisions).
    The function applies the same clipping that the official evaluator does.
    """
    seq = np.clip(seq, 0.0, 1000.0)
    total = np.sum(seq)
    if total < 0.01:
        return float("inf")
    n = seq.shape[0]
    size = next_pow_two(2 * n - 1)
    # convolution via FFT (real -> real)
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (total * total)


def rev_for_peak(seq: np.ndarray, k_max: int) -> np.ndarray:
    """
    Return the vector r where r[i] = a_{k_max-i} if that index exists,
    otherwise 0.  This is the gradient of the peak b_{k_max} w.r.t. a.
    """
    n = seq.shape[0]
    rev = np.zeros_like(seq)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=int)
    rev[idx] = seq[k_max - idx]
    return rev


def gradient_step(seq: np.ndarray) -> np.ndarray:
    """
    Perform a few trial gradient steps (moving opposite to the peak
    gradient) and keep the best candidate.
    """
    n = seq.shape[0]
    # current convolution peak
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))
    rev = rev_for_peak(seq, k_max)

    best = seq
    best_val = fast_score(seq)

    # try several step sizes
    for factor in (0.02, 0.05, 0.10, 0.20):
        cand = seq - factor * rev
        cand = np.maximum(cand, 0.0)
        s = np.sum(cand)
        if s == 0.0:
            continue
        # renormalise to the canonical sum √(2n)
        cand = cand * (math.sqrt(2.0 * n) / s)
        val = fast_score(cand)
        if val < best_val:
            best_val = val
            best = cand
    return best


def solve_convolution_lp(f_seq, rhs):
    """
    Linear programme:
        maximise Σ b_j
        s.t. (f * b)(k) ≤ rhs  for all k = 0 … 2n‑2
             b_j ≥ 0
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise Σ b  → minimise –Σ b
    # Build the (2n‑1) convolution constraints.
    A = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A.append(row)
    A = np.vstack(A)
    b = np.full(2 * n - 1, rhs, dtype=float)

    # bounds enforce b ≥ 0
    bounds = [(0.0, None)] * n

    # ``highs`` is the default modern solver – fast and reliable.
    res = optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return np.array(res.x, dtype=float)
    return None


def lp_mix_step(seq: np.ndarray):
    """
    One LP‑based improvement step:
        * solve the LP for the current normalised sequence,
        * mix the obtained g with the current sequence using a short
          line‑search over t ∈ {0.1,0.25,0.5,0.75,0.9},
        * return the best mixed candidate (or the original if nothing
          improves).
    """
    n = seq.shape[0]
    # normalise to the canonical sum √(2n) – the LP works with this scale.
    norm_seq = seq * (math.sqrt(2.0 * n) / np.sum(seq))
    # current bound M
    M = np.max(np.convolve(norm_seq, norm_seq))

    g_raw = solve_convolution_lp(norm_seq.tolist(), M)
    if g_raw is None:
        # LP failed – fall back to a simple gradient move.
        return gradient_step(seq)

    # renormalise g to the same total as the canonical one
    g = g_raw * (math.sqrt(2.0 * n) / np.sum(g_raw))

    # line‑search on the mixing weight t
    best_seq = seq
    best_val = fast_score(seq)
    for t in (0.10, 0.25, 0.50, 0.75, 0.90):
        cand = (1.0 - t) * seq + t * g
        # keep the canonical sum
        cand = cand * (math.sqrt(2.0 * n) / np.sum(cand))
        val = fast_score(cand)
        if val < best_val:
            best_val = val
            best_seq = cand
    return best_seq

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The routine runs for roughly *budget_s* seconds, performs a mixture
    of gradient, LP and random moves and finally returns the best
    sequence discovered (as a plain Python list of floats).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ---- configuration ----------------------------------------------------
    # length of the sequence – a compromise between expressivity and speed.
    n = 800

    # choose a start point: either a truncated SOTA solution or a fresh random one.
    if "height_sequence_1" in globals() and np.random.rand() < 0.5:
        base = np.asarray(height_sequence_1, dtype=float)

    best_score = evaluate_sequence(best_seq.tolist())  # the authoritative score

    deadline = time.time() + budget_s - 5.0   # keep a small safety margin
    iteration = 0

    while time.time() < deadline:
        iteration += 1

        # decide which elementary move to try
        if np.random.rand() < 0.6:
            # primary move: gradient descent on the current peak
            cand_seq = gradient_step(seq)
        else:
            # LP‑based move with line‑search

        # keep the canonical sum after any modification
        cand_seq = cand_seq * (math.sqrt(2.0 * n) / np.sum(cand_seq))

        cand_score = evaluate_sequence(cand_seq.tolist())

        if cand_score < best_score:
            best_score = cand_score

            print(

                f"time left {time_left:.1f}s"
            )

    print(f"Finished – best score found: {best_score:.6f}")
    # Return as plain list (required by the outer harness)
    return best_seq.tolist()
