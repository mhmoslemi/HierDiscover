def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np
    # set random seed maybe not used
    np.random.seed(seed)
    start = time.time()
    # list of lengths to try
    lengths = [200, 500, 1000, 2000, 5000]
    best_seq = None
    best_score = float('inf')
    for n in lengths:
        # arithmetic progression 1..n
        seq = np.arange(1, n+1, dtype=float)
        # Scale to respect max 1000
        scale = 1000.0 / n
        seq = seq * scale
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
            print(f"Length {n}: score {score}")
    # Optionally local random improvement
    # Do limited hill climbing using gaussians
    while time.time() - start < budget_s - 1:
        # small random perturbations
        seq = best_seq.copy()
        # Choose random number of modifications
        for _ in range(3):
            idx = np.random.randint(len(seq))
            # modify by up to 10% of max value
            delta = (np.random.rand() - 0.5) * 0.2 * 1000.0
            seq[idx] = max(0.0, min(1000.0, seq[idx] + delta))
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"Improved score {best_score} after perturbation")
    print(f"Returning best score {best_score}, length {len(best_seq)}")
    return best_seq

"""
Propose a high‑quality step function for the evaluate_sequence objective.

The evaluation metric is
    2 * n * max(conv(a, a)) / (sum(a) ** 2)

which is invariant under an overall scaling of the vector.
For a polynomial profile a_i = (i+1)^p (p > 0) the theoretical limit of the
metric as the length n → ∞ is

    R(p) = 2 * (p+1)^2 * Gamma(p+1)^2 / Gamma(2p+2),

which drops below 1 as soon as p ≥ 2 and tends rapidly to 0 for larger p.
Therefore a simple power‑law sequence (scaled so that the largest entry is
exactly 1000, to avoid the clipping performed inside *evaluate_sequence*)
produces a score far better than the target 1.503.

The function below tries a handful of lengths and exponents, keeps the best
candidate, and then performs a very cheap random‑perturbation hill‑climb
until the allocated time expires.  No external I/O is used.
"""

import time
import random
import numpy as np

def _power_sequence(n: int, p: float) -> list[float]:
    """
    Return a length‑n non‑negative sequence a_i = (i+1)^p,
    scaled so that max(a) = 1000.
    """
    base = np.arange(1, n + 1, dtype=float) ** p
    scale = 1000.0 / base[-1]          # bring the largest entry to 1000
    seq = base * scale
    return seq.tolist()


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The search space consists of power‑law profiles with several lengths
    and exponents, followed by a tiny stochastic hill‑climb.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin for the final return
    deadline = start_time + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # ---- coarse grid of candidates ---------------------------------
    lengths = [500, 1000, 2000, 3000]          # moderate sizes (fast conv)
    exponents = [1.0, 1.5, 2.0, 3.0, 4.0]     # p = 1 is uniform, higher are better

    for n in lengths:
        for p in exponents:
            if time.time() >= deadline:
                break
            seq = _power_sequence(n, p)
            # evaluate_sequence expects a plain Python list
            try:
                score = evaluate_sequence(seq)
            except Exception:
                # any unexpected failure – treat as infinite score
                score = float("inf")
            if score < best_score:
                best_score = score
                best_seq = seq
                print(f"New best | n={n}, p={p} -> score={score:.6f}")

        if time.time() >= deadline:
            break

    # Fallback – should never happen, but keep the system robust
    if best_seq is None:
        best_seq = [1.0]
        best_score = evaluate_sequence(best_seq)

    # ---- cheap stochastic local improvement -------------------------
    # Only a few perturbations – cheap enough to stay inside the time budget
    while time.time() < deadline:
        # create a mutable copy
        seq_arr = np.array(best_seq, copy=True)

        # apply a handful of tiny random tweaks
        for _ in range(3):
            idx = np.random.randint(len(seq_arr))
            # perturb by up to ±2.5 % of the full height range (±25)
            delta = (np.random.rand() - 0.5) * 0.05 * 1000.0
            seq_arr[idx] = np.clip(seq_arr[idx] + delta, 0.0, 1000.0)

        # evaluate the perturbed candidate
        try:
            new_score = evaluate_sequence(seq_arr.tolist())
        except Exception:
            continue

        # accept only improvements (strict descent)
        if new_score < best_score:
            best_score = new_score
            best_seq = seq_arr.tolist()
            print(f"Improved by perturbation -> score={new_score:.6f}")

    print(f"Finished. Best score = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
