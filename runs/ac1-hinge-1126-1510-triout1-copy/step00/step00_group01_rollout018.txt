def flatten_sequence(seq, budget):
    start_time = time.time()
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq)
    n = len(seq)
    while time.time() - start_time < budget:
        # compute convolution
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        # indices where conv near max:
        tol = max_val * 1e-12 # might need relative tolerance
        max_idxs = np.where(conv >= max_val - tol)[0]
        # compute contributions per index
        contrib = np.zeros(n)
        for k in max_idxs:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                contrib[i] += seq[j]
        # choose indices
        i_max = np.argmax(contrib)
        i_min = np.argmin(contrib)
        if i_max == i_min:
            # maybe random perturbation
            i_max = np.random.randint(n)
            i_min = np.random.randint(n)
            while i_min == i_max:
                i_min = np.random.randint(n)
        # move mass
        delta = 0.01 * seq[i_max]  # small step
        if delta <= 0:
            # random small perturb:
            delta = 1e-4
        seq[i_max] = max(0.0, seq[i_max] - delta)
        seq[i_min] = seq[i_min] + delta
        # evaluate
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            # optional print
            print(f"New best {best_score:.6f} at time {time.time()-start_time:.1f}s")
        # occasionally random perturb to avoid stagnation
        if np.random.rand() < 0.001:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * delta)
    return best_seq

        # random sample parameters

        seq = (indices + d) ** (-beta)
        # optionally ensure max element not exceed 1000, safe
        # evaluate

        if score < best_score:
            best_score = score
            best_seq = seq.tolist()

        # Occasionally try symmetrical variant

                best_score = score2

"""
propose_candidate:  Search for a sequence that minimizes the evaluation function.

The search uses a simple parametric family inspired by the analytical optimum:
    a_i = (i + offset) ** (-exponent)

For exponent > 0.5 the convolution peak is bounded by the first element,
while the sum grows like n^(1‑exponent).  By tuning the exponent, the offset
and the length we can drive the score well below the current bound 2.0.
Random samples of (length, exponent, offset) are generated until the time
budget expires.  The best sequence found is returned.
"""

import numpy as np
import time

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Returns a list of non‑negative heights (the step function) with a low
    evaluation score.  The routine runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0       # a few seconds safety margin

    best_seq = None
    best_score = float('inf')
    best_params = None

    # ----- helper to build a sequence from parameters -----
    def build_seq(length, exponent, offset):
        """Create a_i = (i + offset) ** (-exponent) for i=0..length-1."""
        if offset <= 0.0:
            # avoid 0**negative
            idx = np.arange(1, length + 1, dtype=float)
        else:
            idx = np.arange(length, dtype=float) + offset
        seq = np.power(idx, -exponent)
        # enforce the limits used by evaluate_sequence (the function itself
        # will also clip, but we stay well below the 1000 ceiling)
        return seq.tolist()

    # ----- initial candidates (including the previous SOTA) -----
    # use the known construction as a fallback
    if 'height_sequence_1' in globals():
        try:
            init_seq = list(height_sequence_1)
            init_score = evaluate_sequence(init_seq)
            if init_score < best_score:
                best_score = init_score
                best_seq = init_seq
                best_params = ('SOTA', None, None)
                print(f'Initial SOTA score {best_score:.6f}')
        except Exception:
            pass

    # random search over the parametric family
    while time.time() < deadline:
        # choose a random length (not too small, not too large)
        length = np.random.randint(400, 2000)          # 400‑2000 items

        # exponent > 0.5 is where the interior convolution peak drops below a_0**2
        exponent = np.random.uniform(0.55, 0.70)

        # offset shifts the whole curve to the right and reduces the first term
        offset = np.random.uniform(0.0, 5.0)

        # build and evaluate
        seq = build_seq(length, exponent, offset)
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # any numerical issue – skip this candidate
            continue

        if score < best_score:
            best_score = score
            best_seq = seq
            best_params = (length, exponent, offset)
            print(f'New best {best_score:.6f}  '
                  f'len={length}  exp={exponent:.4f}  off={offset:.4f}')

        # occasional local refinement around the current best parameters
        if best_params is not None and np.random.rand() < 0.02:
            l0, e0, o0 = best_params
            # small random perturbations
            l1 = int(max(50, min(2500, l0 + np.random.randint(-30, 31))))
            e1 = max(0.51, min(0.80, e0 + np.random.normal(scale=0.01)))
            o1 = max(0.0,  o0 + np.random.normal(scale=0.2))
            seq = build_seq(l1, e1, o1)
            try:

            except Exception:

            if score < best_score:

                best_seq = seq
