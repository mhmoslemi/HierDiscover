def internal_score(a):
    N = len(a)
    X = np.fft.fft(a, 2*N)
    conv = np.fft.ifft(X * X).real[:2*N-1]
    max_b = conv.max()
    S = a.sum()
    return 2 * N * max_b / (S ** 2)

def gradient_step(a, step):
    N = len(a)
    X = np.fft.fft(a, 2*N)
    conv = np.fft.ifft(X * X).real[:2*N-1]
    max_idx = np.argmax(conv)
    # compute gradient vector
    idx = max_idx - np.arange(N)
    mask = (idx >= 0) & (idx < N)
    grad = np.zeros_like(a)
    grad[mask] = 2 * a[idx[mask]]
    # update
    a_new = a - step * grad
    a_new = np.clip(a_new, 0, None)
    # renormalize sum to 1
    sum_new = a_new.sum()
    if sum_new == 0:
        a_new = np.random.rand(N)
        a_new = a_new / a_new.sum()
    else:
        a_new = a_new / sum_new
    return a_new

"""
Propose a step‐function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

The method is a two‑stage continuous optimisation:

1.  **Exponential smoothing descent** – minimise a smooth
    approximation of the L∞‑norm,  fβ(a)=log∑exp(β·(a∗a)).
    The gradient of fβ is a weighted sum of the gradients of every
    convolution entry; the weights concentrate on the largest values.
    The update is a projected gradient step onto the probability simplex
    (non‑negative entries with unit sum).  β is increased gradually so
    that the algorithm first smooths the landscape and then focuses on
    the true maximum.

2.  **Plain max‑gradient descent** – after the smoothing phase we
    directly reduce the current maximal convolution entry using its exact
    sub‑gradient (g_i = 2·a_{k−i} for the shift k where the maximum occurs).
    Again we project onto the simplex.

Both phases use fast FFT based convolutions (O(n log n)) and are repeated
with several random restarts until the allotted time (~1000 s) expires.
The best sequence found is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top level (no closures, no lambdas)
# ----------------------------------------------------------------------


def linear_conv(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution using FFT.
    Returns a real vector of length 2·len(a)‑1.
    """
    n = a.shape[0]
    # FFT length 2*n guarantees no circular wrap‑around
    F = np.fft.rfft(a, 2 * n)
    conv = np.fft.irfft(F * F, 2 * n)[: 2 * n - 1]
    return conv


def score_sequence(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective (identical to evaluate_sequence
    up to the clipping performed there).  The sequence is assumed
    non‑negative; sum(a) is guaranteed >0.
    """
    n = a.shape[0]
    conv = linear_conv(a)
    max_b = conv.max()
    s = a.sum()
    # Very small sums are rejected by the official checker – we avoid them.
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (s * s))


def gradient_of_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal convolution entry.
    For the shift `max_idx` the gradient is
        g_i = 2·a_{max_idx‑i}   if 0 ≤ max_idx‑i < n, else 0
    """
    n = a.shape[0]
    idx = max_idx - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    return g


def weighted_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed log‑sum‑exp objective.
    w has length 2·n‑1 and is already normalised (∑w = 1).
    The gradient is 2·(a ⋆ w), where ⋆ denotes the convolution
    with w but taken at the positions i = 0 … n‑1.
    """
    n = a.shape[0]
    # reverse w for the proper cross‑correlation
    w_rev = w[::-1]
    size = 1
    target_len = n + len(w_rev) - 1  # = 3·n‑2
    while size < target_len:
        size <<= 1

    A = np.fft.rfft(a, size)
    B = np.fft.rfft(w_rev, size)
    conv_rev = np.fft.irfft(A * B, size)[:target_len]

    offset = len(w) - 1          # = 2·n‑2
    grad = 2.0 * conv_rev[offset - np.arange(n)]
    return grad


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a non‑negative vector onto the simplex
    {x ≥ 0, sum(x) = 1}.  The implementation works in O(n log n) time.
    """
    if v.sum() == 0:
        # uniform fallback – never happens during normal use
        return np.ones_like(v) / v.shape[0]

    # shift to make the vector non‑negative (already true in our use‑case)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not cond.any():
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond][ -1 ] / rho
    w = np.maximum(v - theta, 0.0)
    # normalise explicitly (numerical safety)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(w) / w.shape[0]
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Main entry point – the function required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑value step‑function.
    The routine runs several random restarts, each consisting of:
        * an exponential‑smoothing descent (log‑sum‑exp of the convolution)
        * a short plain max‑gradient descent
    The best sequence (as a plain Python list) found before the time
    budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1  # small safety margin

    best_seq = None
    best_val = np.inf

    # Parameters that control the two optimisation phases
    SMOOTH_ITERS = 250          # smoothing iterations per restart
    GD_ITERS = 3000             # max gradient‑descent steps per restart
    BETA_START = 5.0
    BETA_MAX = 50.0
    BETA_FACTOR = 1.12
    STEP_SM = 0.15               # step size in the smoothing phase
    STEP_GD = 0.5                # initial step size for the max‑gradient phase

    # ------------------------------------------------------------------
    # Main optimisation loop (many short restarts until time runs out)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --- choose a random length -------------------------------------------------
        n = np.random.randint(250, 1200)          # moderate size – large enough for good bounds
        a = np.random.rand(n).astype(np.float64)  # positive start
        a = a / a.sum()                           # place on the simplex

        # ------------------------------------------------------------------
        # Phase 1 – exponential smoothing descent (log‑sum‑exp of the convolution)
        # ------------------------------------------------------------------
        beta = BETA_START
        for it in range(SMOOTH_ITERS):
            if time.time() >= deadline:
                break

            conv = linear_conv(a)                     # length 2n‑1
            c_max = conv.max()
            # numerically stable soft‑max weights
            w = np.exp(beta * (conv - c_max))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.ones_like(w) / w.shape[0]
            else:
                w = w / w_sum

            # gradient of the smooth approximation
            g = weighted_gradient(a, w)

            # simple projected gradient step
            a = a - STEP_SM * g
            a = np.clip(a, 0.0, None)
            a = project_onto_simplex(a)

            # slowly raise β to concentrate more on the true maximum
            beta = min(BETA_MAX, beta * BETA_FACTOR)

            # occasional score check
            if it % 25 == 0:
                cur_val = score_sequence(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[{time.time():.2f}] New best after smoothing: n={n}, val={best_val:.6f}")

        # ------------------------------------------------------------------
        # Phase 2 – plain max‑gradient descent (sub‑gradient of the L∞ term)
        # ------------------------------------------------------------------
        step = STEP_GD
        last_improve = 0
        for it in range(GD_ITERS):
            if time.time() >= deadline:
                break

            conv = linear_conv(a)
            max_idx = int(conv.argmax())
            max_val = conv[max_idx]

            # sub‑gradient of the maximal entry
            g = gradient_of_max(a, max_idx)

            # projected gradient step
            a_new = a - step * g
            a_new = np.clip(a_new, 0.0, None)
            a_new = project_onto_simplex(a_new)

            a = a_new

            # evaluate every few iterations
            if it % 20 == 0:
                cur_val = score_sequence(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = it
                    step = min(step * 1.2, 2.0)          # accelerate if improving
                    print(f"[{time.time():.2f}] New best after GD: n={n}, val={best_val:.6f}")
                else:
                    # reduce step if we have been stuck for a while
                    if it - last_improve > 200:
                        step = max(step * 0.5, 1e-4)
                        last_improve = it

            # early stop if we have already beaten the target bound
            if best_val < 1.503:
                break

        # stop all restarts once the target is reached
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        # a tiny 2‑element sequence that is already better than uniform
        best_seq = np.array([2.0 / 3.0, 1.0 / 3.0])
    # final safety clamp (the official evaluator also does this)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # make sure we do not return an empty list
    if best_seq.size == 0:
        best_seq = np.array([0.5, 0.5])

    return best_seq.tolist()
