    return 2 * len(a) * max_b / (sum_a**2)

def project_simplex(v):
    # ensure sum(v) = 1 and v>=0 (simplex)
    # Works for any vector.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Simulated‑annealing style optimizer for the autocorrelation bound.

The evaluator is scale‑invariant, so we work with a probability vector
(sum = 1).  The objective is   R(a) = 2·n·max(conv(a,a)) .
The algorithm repeatedly moves a small amount of mass from the index that
contributes most to the current maximal convolution entry towards the
currently smallest entry.  Moves that improve the bound are always kept;
worse moves are accepted with a temperature‑dependent probability to
escape local minima.  The temperature is cooled geometrically.
The search is performed on a single length (default 200) but can be
restarted with different lengths if time permits.  The best sequence
found before the time budget expires is returned.
"""

import time
import numpy as np

def _ratio(seq: np.ndarray) -> float:
    """Return the evaluation function value for a given sequence."""
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    s = np.sum(seq)
    if s < 1e-12:               # should never happen – safeguard
        return np.inf
    return 2.0 * len(seq) * max_b / (s * s)


def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """
    Search for a step‑function (non‑negative list) with a low value of the
    provided ``evaluate_sequence`` function.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds.

    Returns
    -------
    list[float]
        The best sequence discovered before the time limit.
    """
    np.random.seed(seed)
    # reserve a few seconds for finalisation/printing
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1.  Choose a length.  200 is a good compromise between expressivity
    #     and speed.  The loop below can be extended to other lengths if
    #     time permits.
    # ------------------------------------------------------------------
    n = 200
    # start from a uniform distribution (sum == 1)
    a = np.full(n, 1.0 / n, dtype=float)

    # current statistics
    cur_ratio = _ratio(a)
    best_a = a.copy()
    best_ratio = cur_ratio

    # SA parameters
    T = 0.1                     # initial temperature
    cooling = 0.9995            # geometric cooling factor
    min_T = 1e-6                # floor for the temperature
    it = 0

    print(f"[init] length={n}, start ratio={cur_ratio:.6f}")

    while time.time() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 2.  Choose a move.
        #     With high probability we perform a *greedy* move that tries
        #     to reduce the current maximal convolution entry.
        # ------------------------------------------------------------------
        if np.random.rand() < 0.70:
            # Compute the full convolution only once per iteration.
            conv_full = np.convolve(a, a)
            k_max = int(np.argmax(conv_full))

            # contribution of each index to the maximal entry
            idx = np.arange(n)
            j = k_max - idx
            mask = (j >= 0) & (j < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[idx[mask]] * a[j[mask]]

            src = int(np.argmax(contrib))          # index that hurts the most
            dst = int(np.argmin(a))                # index that has the least mass
            if src == dst:
                # fall back to a random destination
                dst = (dst + 1) % n

            # move a modest fraction of the source mass
            eps = a[src] * 0.05                     # 5 % of the source weight
            if eps <= 0.0:
                continue
            a_new = a.copy()
            a_new[src] -= eps
            a_new[dst] += eps
        else:
            # ------------------------------------------------------------------
            # 3.  Simple random redistribution move (helps to explore).
            # ------------------------------------------------------------------
            i, j = np.random.choice(n, size=2, replace=False)
            eps = a[i] * 0.05
            if eps <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps

        # Numerical safety – discard infeasible candidates immediately.
        if np.any(a_new < -1e-12):
            continue
        a_new = np.clip(a_new, 0.0, None)   # enforce non‑negativity

        # Re‑normalise to keep the sum exactly 1 (tiny drift may accumulate).
        s = np.sum(a_new)
        if s == 0.0:
            continue
        a_new /= s

        new_ratio = _ratio(a_new)

        # ------------------------------------------------------------------
        # 4.  Acceptance test (Metropolis criterion).
        # ------------------------------------------------------------------
        delta = new_ratio - cur_ratio
        accept = False
        if delta < 0:
            accept = True
        else:
            # probability of accepting a worsening move
            if T > min_T:
                prob = np.exp(-delta / T)
                accept = np.random.rand() < prob

        if accept:
            a = a_new
            cur_ratio = new_ratio
            # record the best found so far
            if new_ratio < best_ratio:
                best_ratio = new_ratio
                best_a = a_new.copy()
                print(f"[{it:8d}] new best ratio = {best_ratio:.6f}")

        # ------------------------------------------------------------------
        # 5.  Temperature schedule.
        # ------------------------------------------------------------------
        T = max(T * cooling, min_T)

        # occasional re‑heating to avoid complete stagnation
        if it % 5000 == 0:
            T = max(T, 0.01)

    # ----------------------------------------------------------------------
    # 6.  Return the best sequence.  The evaluator caps values at 1000, so
    #     we are safely within that range.
    # ----------------------------------------------------------------------
    print(f"[final] iterations={it}, best ratio={best_ratio:.6f}")
    return best_a.tolist()
