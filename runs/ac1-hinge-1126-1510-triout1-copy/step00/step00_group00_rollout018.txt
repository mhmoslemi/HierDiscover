"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

where *conv* is the discrete convolution.  The score is independent of an
overall scale of the vector, so we keep the total sum equal to one.  The
algorithm is a simulated‑annealing / stochastic‑gradient search that works
directly on the full‑length vector (the current best construction has
≈6400 entries, which is still cheap for FFT‑based convolution).

The main ideas
~~~~~~~~~~~~~~

* Initialise the sequence with a random probability vector (sum = 1).
* At each step pick a random ordered pair of positions (i,j) and transfer a
  small amount of mass from *i* to *j* while keeping the total sum unchanged.
* The new score is obtained with a fast FFT based convolution; the score is
  ``2 * n * max_conv`` because the sum is forced to 1.
* The move is always accepted if it improves the score, otherwise it is
  accepted with a Metropolis probability that slowly cools down.
* Several independent restarts are performed until the global time budget
  (≈1000 s) is exhausted; the best sequence seen so far is returned.

The routine respects all security checks performed by ``evaluate_sequence``:
all entries stay non‑negative, never exceed 1000, and the total sum stays
well above the 0.01 threshold.

The code is completely self‑contained, uses only NumPy and the standard
library and should finish well within the allowed time limit.
"""

import time
import math
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """Return a list of heights that (hopefully) beats the current bound.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.  The function stops a few seconds before this
        limit to avoid a forced termination.
    **kwargs
        Ignored – present only for compatibility with the surrounding test
        harness.

    Returns
    -------
    List[float]
        The best step‑function found within the time budget.
    """
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Determine the problem size.  If a previous construction is
    # available (global variable ``height_sequence_1``) we reuse its
    # length, otherwise we fall back to the default of 6402.
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 6402

    # --------------------------------------------------------------
    # Helper: compute the maximum entry of the (linear) convolution
    # using an FFT of size a power of two (fast and numerically stable).
    # --------------------------------------------------------------
    # Length needed for a linear convolution of two vectors of length n.
    conv_len = 2 * n - 1
    # Next power of two for efficient FFT.
    fft_len = 1 << (conv_len - 1).bit_length()

    def max_convolution(v: np.ndarray) -> float:
        """Return max(conv(v, v)) using an FFT of length ``fft_len``."""
        # rfft works on real input and returns the non‑redundant half.
        fv = np.fft.rfft(v, n=fft_len)
        conv = np.fft.irfft(fv * fv, n=fft_len)[:conv_len]
        return float(conv.max())

    # --------------------------------------------------------------
    # Simulated annealing / stochastic hill‑climbing parameters.
    # --------------------------------------------------------------
    # Evolutionary parameters.
    #   – start temperature (as a raw score, not normalised).
    #   – exponential cooling factor.
    #   – initial magnitude of a transfer (as a fraction of the source
    #     entry).  This shrinks together with the temperature.
    # --------------------------------------------------------------
    start_temp = None          # will be set after the first evaluation
    cooling = 0.99995          # temperature multiplier per iteration
    frac_step_start = 0.4      # up to 40 % of the donor mass may be moved
    frac_step_end = 0.001      # final allowed relative step size

    # --------------------------------------------------------------
    # Book‑keeping for the best solution seen.
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Main search loop – we repeatedly restart with a fresh random
    # probability vector and perform a local walk until the time runs
    # out.
    # --------------------------------------------------------------
    deadline = time.time() + budget_s - 1.0   # a safety margin

    iteration = 0
    while time.time() < deadline:
        # ---- random initial vector (probability distribution) ----
        seq = np.random.rand(n).astype(np.float64)
        seq /= seq.sum()                     # normalise to sum = 1

        # ---- evaluate the initial score (fast, no call to evaluate_sequence) ----
        cur_max = max_convolution(seq)
        cur_score = 2.0 * n * cur_max        # = evaluate_sequence(seq) for sum=1

        # Initialise temperature based on the first score we see.
        if start_temp is None:
            start_temp = cur_score * 0.05   # 5 % of the current score
        temp = start_temp

        # linear interpolation for the step size factor
        # (large at the beginning, tiny near the end of the inner walk)
        step_factor = frac_step_start

        # ---- inner stochastic walk ----
        while time.time() < deadline:
            # Randomly pick two distinct indices.
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

            # Propose a transfer from i to j.
            # The transferred amount is a random fraction of the donor entry.
            # It is never larger than the donor value, guaranteeing non‑negativity.
            # As the walk proceeds the fraction shrinks (see ``step_factor``).
            delta = (np.random.rand() - 0.5) * step_factor * seq[i] * 2.0
            # Clamp delta so that we stay non‑negative.
            delta = max(-seq[i], min(delta, np.inf))

            if abs(delta) < 1e-15:      # negligible move – skip.
                continue

            new_seq = seq.copy()
            new_seq[i] -= delta
            new_seq[j] += delta

            # Normalise again – this leaves the total sum exactly 1 and
            # removes any tiny numerical drift.
            new_seq /= new_seq.sum()

            # Evaluate the new candidate.
            new_max = max_convolution(new_seq)
            new_score = 2.0 * n * new_max

            # Acceptance test (Metropolis criterion).
            accept = False
            if new_score < cur_score:

            else:
                # probability = exp( -Δ / T )
                if temp > 0:

                    if np.random.rand() < prob:

            if accept:

                cur_max = new_max

            temp *= cooling
            # Linear decay of step size from ``frac_step_start`` to ``frac_step_end``.
            step_factor = max(
                frac_step_end,
                step_factor * cooling,
            )

            iteration += 1

            # A cheap exit condition – if the temperature is already vanishing
            # we stop the inner walk and restart.
            if temp < 1e-12:
                break

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list).  All
    # entries are non‑negative and sum to 1, which respects the
    # security checks inside ``evaluate_sequence``.
    # --------------------------------------------------------------
    if best_seq is None:
        # As a last resort fall back to a uniform sequence.
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    return best_seq.tolist()
