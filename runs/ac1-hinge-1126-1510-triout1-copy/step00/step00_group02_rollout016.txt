"""
Gradient descent (projected) on probability simplex to minimise
the peak value of the self‑convolution a*a.
The objective is equivalent to evaluating the given evaluation
function with the sum of the coefficients normalised to one.
We start from a power‑law (∝ i^{-0.5}) shape which already gives
an evaluation around π/2 ≈ 1.57, then iteratively move probability
mass away from indices that contribute most to the current peak.
A simple sub‑gradient step with diminishing step size and projection
onto the simplex is used. Multiple random restarts are performed
until the time budget expires and the best sequence found is returned.
"""

import time
import numpy as np

def _simplex_projection(v):
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    # Implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    # works for arbitrary length v.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gradient_of_convolution(p, k):
    """Return ∇_p (p*p)_k for a given index k."""
    n = p.shape[0]
    grad = np.zeros_like(p)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    # both i and j are guaranteed to be in [0, n-1]
    grad[i] += p[i] + p[j]
    return grad

def _objective(p):
    """Return the exact evaluation value for a probability vector p."""
    # sum(p) = 1 → evaluate = 2*n*max(p*p)
    n = p.shape[0]
    b = np.convolve(p, p)
    return 2.0 * n * np.max(b)

def propose_candidate(seed=0, budget_s=990, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    provided ``evaluate_sequence`` function.
    The method works on a probability vector (sum = 1) because
    the evaluation formula simplifies to ``2*n*max(p*p)``.
    A projected sub‑gradient descent on the simplex, started from a
    power‑law initialisation, is used together with several random
    restarts. The best sequence found before the time limit is returned.
    """
    np.random.seed(seed)
    DEADLINE = time.time() + budget_s - 5.0  # keep a small safety margin

    # length of the candidate – we reuse the length of the current best
    # construction (1542) but allow a small variation if needed.
    N = 1542

    best_seq = None
    best_score = np.inf

    # ---------- helpers ----------
    def evaluate_and_update(p_vec):
        nonlocal best_seq, best_score
        cur_score = evaluate_sequence(p_vec.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = p_vec.copy()
            print(f"[{time.time():.2f}] New best: {best_score:.6f}")
        return cur_score

    # pre‑allocate temporary arrays once (helps a little with speed)
    while time.time() < DEADLINE:
        # ----- initialise -------------------------------------------------------------------
        if np.random.rand() < 0.6:
            # power‑law (≈ 1/√i) – already close to the theoretical optimum π/2
            idx = np.arange(N, dtype=np.float64) + 1.0
            p = 1.0 / np.sqrt(idx)
        else:
            # random Dirichlet start – helps escaping possible local minima
            p = np.random.rand(N)
        # normalise to the simplex
        p = p / p.sum()

        # ----- inner optimisation loop -------------------------------------------------------
        it = 0
        # we run a modest number of iterations per restart, stopping earlier
        # if the global budget runs out.
        while time.time() < DEADLINE and it < 3000:
            # compute self‑convolution and the current peak
            b = np.convolve(p, p)
            max_val = b.max()

            # pick a few indices that are (almost) maximal – this smooths the sub‑gradient
            # ``np.argpartition`` is O(n) and gives the indices of the largest values.
            top_k = np.argpartition(b, -5)[-5:]  # 5 indices with largest convolution values

            # aggregate the sub‑gradient of the selected peaks
            grad = np.zeros_like(p)
            for k in top_k:
                grad += _gradient_of_convolution(p, k)

            # diminishing step size – the 0.5 factor works well in practice
            lr = 0.5 / np.sqrt(it + 1)

            # gradient step and projection back onto the simplex
            p = p - lr * grad
            p = _simplex_projection(p)

            # occasional evaluation (every 10 iterations) – cheap compared to the time budget
            if it % 10 == 0:
                evaluate_and_update(p)

            it += 1

        # occasionally try a tiny random perturbation on the best vector seen so far
        # – this can help breaking symmetry after a long stagnation period.
        if best_seq is not None and time.time() < DEADLINE:
            perturb = best_seq.copy()
            # pick a random index and move a small amount of probability mass
            i = np.random.randint(N)
            delta = (np.random.rand() - 0.5) * 0.01  # up to ±0.5 % of the total mass
            perturb[i] = max(0.0, perturb[i] + delta)
            perturb = _simplex_projection(perturb)
            evaluate_and_update(perturb)

    # If nothing was found (unlikely) fall back to a uniform sequence.
    if best_seq is None:
        best_seq = np.full(N, 1.0 / N)

    return best_seq.tolist()

"""
search for a low‑value step‑function using a simple power‑law family.
The evaluation function simplifies for a probability vector p (∑p=1) to

    E(p) = 2·n·max(p * p)

where * denotes the discrete convolution.
For p_i ∝ (i+1)^(-α) the peak of p*p is analytically close to
the theoretical optimum when α≈0.6, giving E≈1.38 (< 1.503).

The optimizer therefore:
  * sweeps a small set of α values,
  * keeps the best power‑law vector,
  * optionally performs a few projected sub‑gradient steps to
    reduce the current peak further,
  * returns the best vector found before the time budget expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x≥0, sum x = 1}
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex using the O(n log n) algorithm."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the largest rho where u_i > (cssv_i / i)
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # all entries are non‑positive
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Gradient of a single convolution entry (p * p)_k w.r.t. p
def _grad_convolution(p: np.ndarray, k: int) -> np.ndarray:
    """
    Return ∇_p (p*p)_k.
    For each i that can participate in the sum i + j = k we have
        ∂/∂p_i (p_i p_{k-i} + p_{k-i} p_i) = p_{k-i} + p_i .
    """
    n = p.shape[0]
    grad = np.zeros_like(p)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    i = np.arange(lo, hi + 1, dtype=int)
    j = k - i
    grad[i] += p[i] + p[j]
    return grad

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 990, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) beats the
    current upper bound 2.0.  The method is a quick exploration of the
    power‑law family a_i = (i+1)^(-α) together with a few
    projected sub‑gradient steps.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0      # a small safety margin

    # length of the final construction – we reuse the known good length
    N = 1542

    best_seq = None
    best_score = np.inf
    best_alpha = None

    # ------------------------------------------------------------------
    # 1) coarse sweep over the exponent α
    alphas = np.linspace(0.40, 0.80, 41)          # step = 0.01
    for alpha in alphas:
        if time.time() >= deadline:
            break
        idx = np.arange(N, dtype=np.float64) + 1.0
        seq = idx ** (-alpha)
        seq = seq / seq.sum()                     # turn into a probability vector
        score = evaluate_sequence(seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            best_alpha = alpha
            print(f"[{time.time():.2f}] α={alpha:.4f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 2) a tiny refinement around the best α found (optional but cheap)
    if best_alpha is not None:
        fine_alphas = np.linspace(max(0.35, best_alpha - 0.03),
                                 min(0.85, best_alpha + 0.03), 31)
        for alpha in fine_alphas:
            if time.time() >= deadline:
                break
            idx = np.arange(N, dtype=np.float64) + 1.0
            seq = idx ** (-alpha)
            seq = seq / seq.sum()
            score = evaluate_sequence(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                best_alpha = alpha
                print(f"[{time.time():.2f}] fine α={alpha:.5f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # 3) a few projected sub‑gradient steps starting from the best power‑law
    if best_seq is not None:
        p = best_seq.copy()
        it = 0
        while time.time() < deadline and it < 3000:
            b = np.convolve(p, p)
            max_val = b.max()
            # take the several indices that currently attain the peak
            top_k = np.argpartition(b, -5)[-5:]   # 5 indices (unsorted)
            grad = np.zeros_like(p)
            for k in top_k:

            p = p - lr * grad
            p = _simplex_projection(p)

            if it % 10 == 0:
                score = evaluate_sequence(p.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = p.copy()
                    print(f"[{time.time():.2f}] sub‑grad it={it:4d} → score={score:.6f}")

            it += 1

    # ------------------------------------------------------------------
    # fallback – should never happen, but keep the function total
    if best_seq is None:
        best_seq = np.full(N, 1.0 / N)

    print(f"final best α≈{best_alpha:.5f}, score={best_score:.6f}")
    return best_seq.tolist()
```
