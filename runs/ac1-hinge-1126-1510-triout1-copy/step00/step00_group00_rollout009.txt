Implementation details for `project_simplex`:

```
def project_simplex(v):
    # v: numpy array, will project onto simplex sum=1, v_i >= 0
    # based on algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    a = np.random.rand(n)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a) # 2*n*max(conv)
    for t in range(iters):
        conv = np.convolve(a, a)
        max_conv = conv.max()
        # compute softmax weights
        # subtract max for stability
        exps = np.exp(alpha * (conv - max_conv))
        w = exps / exps.sum()
        # gradient
        grad_full = np.correlate(a, w, mode='full')
        grad = 2 * grad_full[2*n-2: 2*n-2 + n]
        # step
        lr = lr0 / np.sqrt(t+1)   # decaying
        a = a - lr * grad
        # projection onto simplex
        a = project_simplex(a)
        # Compute score occasionally
        if (t % 10) == 0:
            score = 2 * n * np.max(np.convolve(a,a))
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score

def optimal_mix_t(conv_f, conv_g, cross):
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3
        t2 = hi - (hi - lo) / 3
        max1 = np.max((1-t1)**2 * conv_f + 2*t1*(1-t1) * cross + t1**2 * conv_g)
        max2 = np.max((1-t2)**2 * conv_f + 2*t2*(1-t2) * cross + t2**2 * conv_g)
        if max1 < max2:
            hi = t2
        else:
            lo = t1
    t_opt = (lo + hi) / 2
    return t_opt

    best_seq = a.copy()

    # Precompute: define iteration budgets with mixture of gradient and LP
    while time.time() < deadline:
        # gradient descent steps
        n = len(a)
        # Determine parameters based on remaining time
        # use small number of gradient steps each loop
        grad_iters = 200
        lr0 = 0.5
        alpha0 = 2.0
        # run gradient descent
        for it in range(grad_iters):

            conv = np.convolve(a, a)
            max_conv = conv.max()
            # weight for gradient: softmax around max
            w = np.exp(alpha0 * (conv - max_conv))
            w = w / np.sum(w)
            # gradient via correlation
            grad_full = np.correlate(a, w, mode='full')
            grad = 2.0 * grad_full[2*n-2:2*n-2 + n]

            lr = lr0 / np.sqrt(it + 1)
            a = a - lr * grad
            # enforce nonnegativity
            a = np.maximum(a, 0)
            # project onto simplex to keep sum 1
            a = project_simplex(a)
        # evaluate after gradient batch
        score = evaluate_sequence(a.tolist())
        if score < best_score:

            best_seq = a.copy()

        conv_f = np.convolve(a, a)

"""
Hybrid optimizer for the step‑function evaluation.

The score is  S = 2·n·max(conv(a,a)) / (sum a)^2 .
Since scaling does not affect the value we fix sum(a)=1.
The aim is to flatten the auto‑convolution so that its peak
is as close as possible to the average value 1/(2n‑1).

The algorithm combines two complementary ideas:

1. Gradient descent on a smooth approximation of the
   max‑operator.  For a given sequence a we compute the
   convolution c = a*a, build a soft‑max weight
   w_k ∝ exp(α·(c_k‑max(c))) and take the gradient
   of the weighted sum w·c.  The update
   a←a‑η·∇ (projected onto the probability simplex)
   pushes mass away from indices that contribute most
   to the current peak.

2. An occasional linear‑programming step (the same LP
   used in the published SOTA).  Given the current f we
   solve for a g with larger ℓ₁‑norm while keeping
   conv(f,g) bounded by the current peak M.  Mixing
   f and g with the optimal scalar t (found by a tiny
   ternary search on the convex function
   max_k[(1‑t)²·conv(f,f)+2t(1‑t)·conv(f,g)+t²·conv(g,g)])
   yields a guaranteed non‑increasing peak.

Both mechanisms are cheap (O(n) per iteration) for
moderate length n (≈500‑1500) and together drive the
objective well below the uniform bound of 2.0.

The function runs for the supplied time budget (≤1000 s)
and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i >= 0, sum x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # find rho
    rho = ind[u - cssv / ind > 0].max()
    theta = cssv[rho - 1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def solve_convolution_lp(f_seq, rhs):
    """
    Solve the LP:
        maximize   sum b_j
        subject to   conv(f_seq, b) <= rhs
                     b >= 0
    Returns the optimal b (as a NumPy array) or None on failure.
    """
    n = len(f_seq)
    # Linear objective (maximise sum b -> minimise -sum b)
    c = -np.ones(n)

    # Build convolution inequalities
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints: -b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # Use the default high‑performance solver (HiGHS)
    res = linprog(c, A_ub=a_ub, b_ub=b_ub,
                  method='highs', bounds=(0, None),
                  options={'time_limit': 5.0, 'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def optimal_mix_t(conv_f, conv_g, conv_fg):
    """
    For a given pair of sequences f,g (length n) we have
        h(t)_k = (1‑t)^2·conv_f[k] + 2t(1‑t)·conv_fg[k] + t^2·conv_g[k]
    The function max_k h(t)_k is convex in t∈[0,1].
    This routine locates t that minimises that maximum using a short
    ternary search (30 iterations → sub‑microsecond precision).
    """
    lo, hi = 0.0, 1.0
    for _ in range(30):
        t1 = lo + (hi - lo) / 3.0
        t2 = hi - (hi - lo) / 3.0
        h1 = (1 - t1) ** 2 * conv_f + 2 * t1 * (1 - t1) * conv_fg + t1 ** 2 * conv_g
        h2 = (1 - t2) ** 2 * conv_f + 2 * t2 * (1 - t2) * conv_fg + t2 ** 2 * conv_g
        if h1.max() < h2.max():
            hi = t2
        else:
            lo = t1
    return (lo + hi) / 2.0


def gradient_step(a, lr, alpha):
    """
    Perform one projected gradient step that reduces a soft‑max
    approximation of the convolution peak.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    peak = conv.max()

    # soft‑max weights (more weight on large entries)
    w = np.exp(alpha * (conv - peak))
    w /= w.sum()

    # gradient of weighted sum: 2 * correlation(a, w)
    grad_full = np.correlate(a, w, mode='full')
    grad = 2.0 * grad_full[2 * n - 2: 2 * n - 2 + n]

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = project_simplex(a_new)
    return a_new

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    # ------------------------------------------------------------------

    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:

    a = a / a.sum()                     # enforce sum = 1

    # keep the best sequence seen so far
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------

    min_n = 300                # smallest length we ever handle
    max_n = 800                # largest length we keep
    grad_iters_per_batch = 200
    lr_start = 0.8
    alpha_start = 1.0

    while time.time() < deadline:
        n = a.shape[0]

            # time check inside the inner loop for safety

            frac = (it + 1) / grad_iters_per_batch

            alpha = alpha_start + (alpha_end - alpha_start) * frac
            a = gradient_step(a, lr, alpha)

        # quick evaluation (no heavy evaluate_sequence call)

    print(f"Finished. Best score = {final_score:.6f} (length {len(best_seq)})")
