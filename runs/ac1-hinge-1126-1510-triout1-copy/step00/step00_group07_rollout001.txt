"""
Propose a step‑function that beats the uniform bound 2.0.

The evaluation metric is
    2·n·max(conv(a,a)) / (sum(a)^2)

For a non‑negative linear ramp  a_i = i  (i = 0 … n‑1) one can compute
    max(conv) = n·(n‑1)·(n‑2) / 6        (the centre of the convolution)
    sum(a)    = n·(n‑1) / 2

which gives the closed form
    score = 4·(n‑2) / (3·(n‑1))

and tends to 4/3 ≈ 1.333 as n grows.  This is well below the current
upper bound 2.0 and the target 1.503.

The algorithm simply tries a few candidate lengths, builds the ramp,
scales it to stay inside the evaluator’s clipping range (the scale cancels
out of the formula), and keeps the best one.  A short random‑perturbation
phase is run while time permits, in case a tiny improvement can be found.

All numbers are non‑negative and the function returns the best sequence
found before the time budget expires.
"""

import time
import numpy as np

def _linear_ramp(length: int) -> list[float]:
    """Return a ramp a_i = i (i = 0 … length‑1) scaled to max ≤ 1000."""
    if length <= 1:
        return [0.0]
    # raw ramp: 0,1,2,…,length‑1
    ramp = np.arange(length, dtype=float)
    # scale so the largest entry is exactly 1000 (the evaluator clips at 1000)
    scale = 1000.0 / (length - 1)
    ramp *= scale
    return ramp.tolist()


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> list[float]:
    """Search for a low‑score step function within the given time budget.

    The search consists of:
      1. Evaluating a few linear ramps of different lengths.
      2. Keeping the best ramp found.
      3. Performing tiny random perturbations for the remaining time.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds.  The function returns before the budget is
        exhausted (a small safety margin is kept).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    # initialise random generator and time limit
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # keep a small safety margin

    # ------------------------------------------------------------------
    # 1. Try a handful of ramp lengths and keep the best one.
    # ------------------------------------------------------------------
    candidate_lengths = [50, 100, 250, 500, 1000, 2000, 5000]
    best_seq: list[float] | None = None
    best_score = float("inf")

    for n in candidate_lengths:
        if time.time() >= deadline:
            break

        seq = _linear_ramp(n)
        score = evaluate_sequence(seq)               # evaluate_sequence is provided globally
        print(f"[init] length={n:5d} → score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

        # a reversed ramp is just a mirror image; keep it for completeness
        seq_rev = list(reversed(seq))
        score_rev = evaluate_sequence(seq_rev)
        if score_rev < best_score:
            best_score = score_rev
            best_seq = seq_rev

    # If something went wrong and we have no sequence, fall back to a tiny ramp.
    if best_seq is None:
        best_seq = _linear_ramp(100)

    # ------------------------------------------------------------------
    # 2. Small stochastic refinement while time permits.
    # ------------------------------------------------------------------
    curr_seq = best_seq[:]                     # work on a copy
    while time.time() < deadline:
        # pick a random index and perturb it by a few percent
        idx = np.random.randint(len(curr_seq))
        old_val = curr_seq[idx]

        # relative Gaussian perturbation (10 % of the current value)
        perturb = np.random.normal(scale=0.1) * old_val
        new_val = max(0.0, old_val + perturb)   # keep non‑negative

        curr_seq[idx] = new_val
        new_score = evaluate_sequence(curr_seq)

        if new_score < best_score:
            best_score = new_score
            best_seq = curr_seq[:]

        else:
            # revert the change
            curr_seq[idx] = old_val

    print(f"Final best score={best_score:.6f} with length={len(best_seq)}")
    return best_seq
