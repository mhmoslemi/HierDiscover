def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random
    from scipy.optimize import linprog

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5   # safety

    # initialize from SOTA if available
    try:
        init_seq = np.array(height_sequence_1, dtype=float)
        if init_seq.size == 0:
            raise Exception
    except Exception:
        init_seq = np.random.rand(np.random.randint(500, 1500))

    # normalize to sum=1
    if init_seq.sum() == 0:
        init_seq = np.ones_like(init_seq) / len(init_seq)
    else:
        init_seq = init_seq / init_seq.sum()

    best_seq = init_seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    f = init_seq.copy()

    iteration = 0
    while time.time() < deadline:
        n = len(f)
        # compute max convolution M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # solve LP
        # Variables: b (length n)
        # maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n)
        # Build A_ub matrix
        # We'll precompute rows for each k 0..2n-2
        rows = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            # for i in range(max(0, k-(n-1)), min(k, n-1)+1):
            #   rows[k, k-i] += f[i] # because variable index j = k - i
            i_start = max(0, k-(n-1))
            i_end = min(k, n-1)
            # Loop vectorized: set rows[k, k - i_range] = f[i_range]
            idx = np.arange(i_start, i_end+1)
            rows[k, k - idx] = f[idx]
        # add nonnegativity as -I * b <= 0
        A_ub = np.vstack([rows, -np.eye(n)])
        b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])

        # Solve
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', bounds=(0, None), options={'presolve': True})
        except Exception as e:
            res = None

        if res is not None and res.success:
            g = res.x
            if g.sum() == 0:
                # fallback random
                pass
            else:
                # normalize g to sum=1
                g = g / g.sum()
                # line search for mixing parameter
                # try several t values
                best_local_seq = f
                best_local_score = best_score
                for t in [0.01, 0.05, 0.1, 0.2, 0.4, 0.6]:
                    new_seq = (1 - t) * f + t * g
                    new_seq = new_seq / new_seq.sum()  # ensure sum=1 (should already)
                    score = evaluate_sequence(new_seq.tolist())
                    if score < best_local_score:
                        best_local_score = score
                        best_local_seq = new_seq.copy()
                # accept best among them
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_seq = best_local_seq.copy()
                # update f to the best local anyway (to continue iteration)
                f = best_local_seq.copy()
        else:
            # LP failed; do random local perturbation
            # Pick random index i with weight > 0
            i = np.random.randint(n)
            # compute delta as small fraction of its value
            delta = f[i] * 0.05
            if delta > 0:
                f[i] -= delta
                # add delta to random other index
                j = np.random.randint(n-1)
                if j >= i:
                    j += 1
                f[j] += delta
                # renormalize
                f = f / f.sum()
        iteration += 1
        if iteration % 10 == 0:
            print(f"Iter {iteration}, best_score {best_score:.6f}, n={len(f)}")
    # return best_seq as python list
    return best_seq.tolist()

"""
Propose a step‑function sequence that minimizes the evaluation metric.

Algorithm
---------
The metric (for a non‑zero sum) is   2·n·max(conv(a,a)) / (sum(a))².
Because the function is homogeneous we may enforce sum(a)=1.
The goal then becomes to minimise the ℓ∞‑norm of the convolution
f * a, where f is the current sequence.

We iteratively solve a linear programme:

    minimise   t
    subject to conv(f, a) ≤ t          (2·n‑1 linear constraints)
               sum(a) = 1
               a ≥ 0

All constraints are linear in the unknown a (the convolution is a
Toeplitz‑matrix times a).  The LP returns a new sequence a that has a
smaller maximum convolution value with the current f while keeping the
total mass fixed.  Replacing f by this a and repeating the procedure
flattens the auto‑convolution and drives the objective towards its
theoretical lower bound ~ 2·n/(2·n‑1) ≈ 1.

The routine runs as many LP iterations as possible within the allotted
time (default 1000 s).  It keeps the best sequence found and finally
returns it as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: build the inequality matrix for a given left‑hand sequence 'f'.
# ----------------------------------------------------------------------
def _build_A_ub(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A_ub for the LP:
        conv(f, a) - t <= 0    (one row per convolution index)
    The variable vector is x = [a_0, …, a_{n-1}, t].
    """
    n = f.shape[0]
    m = 2 * n - 1                     # length of full convolution
    A = np.zeros((m, n + 1), dtype=np.float64)

    # Fill each convolution constraint.
    for k in range(m):
        # indices i where f_i contributes to (f*a)_k
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals                     # corresponding a‑indices
        A[k, j_vals] = f[i_vals]                # coefficient f_i for a_j
        A[k, -1] = -1.0                          # coefficient of the slack variable t
    return A

            # If the LP fails (rare), perform a tiny random perturbation
            # and continue.
            i, j = np.random.choice(n, 2, replace=False)
            delta = cur[i] * 0.05
            cur[i] = max(cur[i] - delta, 0.0)
            cur[j] += delta
            cur /= cur.sum()
            iteration += 1
            continue

        # Extract the new candidate (ignore the auxiliary t).
        new_seq = res.x[:n]
        # Numerical errors may deviate the sum slightly – renormalise.
        new_seq = np.maximum(new_seq, 0.0)
        new_seq /= new_seq.sum()

        # ----------------------------------------------------------------
        # Evaluate the candidate and keep the best one.
        # ----------------------------------------------------------------
        try:
            score = evaluate_sequence(new_seq.tolist())
        except Exception:            # safety net – should never happen
            score = float('inf')

        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f'Iter {iteration:5d} – new best score: {best_score:.6f}')

        # Update the current sequence for the next iteration.
        cur = new_seq

        iteration += 1

        # Occasionally (every ~30 iterations) print a short status line.
        if iteration % 30 == 0:
            elapsed = time.time() - start_time
            print(f'Iter {iteration:5d} – elapsed {elapsed:.1f}s – '
                  f'best {best_score:.6f}')

    # Return the best discovered sequence as a plain Python list.
    return best_seq.tolist()
```
