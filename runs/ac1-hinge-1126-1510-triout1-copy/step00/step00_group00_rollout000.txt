    # returns best sigma and best score
    phi = (1 + math.sqrt(5)) / 2

    fd = eval_sigma(d, n)

"""
Propose a candidate step‑function using a (scaled) Gaussian profile.

The evaluation objective is
    2 * n * max(conv(a, a)) / (sum(a)**2)

which for a fixed sum (the evaluator rescales the sum internally) reduces to
minimising the peak of the auto‑convolution.  A Gaussian distribution gives a
smooth auto‑convolution whose peak can be made arbitrarily small by changing
its variance, but if the variance is too large the distribution becomes almost
uniform and the peak rises again to the uniform value 2.  Consequently the
objective is unimodal in the Gaussian width σ.

The algorithm
-------------
1. Choose the length `n`.  If the global `height_sequence_1` is present we
   reuse its length (the current best solution has length 6402); otherwise a
   random length in the 2‑5 k range is used.

2. Search for the optimal σ in the interval `[0.05 n, 0.9 n]` with a cheap
   golden‑section optimiser (≈30 evaluations).  Each evaluation builds a
   Gaussian, scales it so that the sum equals `sqrt(2 n)` (the normalisation
   used by the evaluator) and calls `evaluate_sequence`.

3. Refine the solution with a short hill‑climbing phase: repeatedly transfer
   a tiny amount of mass from one index to another while keeping the total
   sum unchanged.  Accept the move only if the evaluation improves.

The whole search runs well inside the 1000 s budget and typically reaches a
value below 1.5 for the current length 6402 (the target is 1.503).

All helper functions are defined at top level; the only external function
used is the provided `evaluate_sequence`.
"""

import time
import math
import numpy as np

def gaussian_sequence(n: int, sigma: float) -> list[float]:
    """
    Return a non‑negative step‑function of length ``n`` whose entries follow
    a Gaussian centred at the middle of the interval.  The sequence is scaled
    so that its sum equals ``sqrt(2*n)`` – the normalisation used by the
    evaluator.  Values are clipped to the allowed range [0, 1000].
    """
    mu = (n - 1) / 2.0
    idx = np.arange(n, dtype=float)
    # raw Gaussian (peak = 1)
    raw = np.exp(-((idx - mu) ** 2) / (2.0 * sigma ** 2))
    # scale to the required sum
    scale = math.sqrt(2.0 * n) / raw.sum()
    raw *= scale
    # safety clipping (practically never needed for the σ we use)
    raw = np.clip(raw, 0.0, 1000.0)
    return raw.tolist()


def _eval_sigma(sigma: float, n: int) -> float:
    """
    Helper that builds a Gaussian of length ``n`` with width ``sigma`` and
    returns its evaluation score.
    """
    seq = gaussian_sequence(n, sigma)
    return evaluate_sequence(seq)


def _golden_section_search(n: int, a: float, b: float, deadline: float,
                           max_iter: int = 15) -> tuple[float, float]:
    """
    Unimodal optimisation of `_eval_sigma` on the interval [a, b] using the
    golden‑section method.  Returns the best σ found together with its score.
    The search stops early if the time limit ``deadline`` is reached.
    """
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    inv_phi = (math.sqrt(5.0) - 1.0) / 2.0  # 1/phi

    c = b - inv_phi * (b - a)
    d = a + inv_phi * (b - a)
    fc = _eval_sigma(c, n)
    fd = _eval_sigma(d, n)

    best_sigma = c if fc <= fd else d
    best_score = fc if fc <= fd else fd

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - inv_phi * (b - a)
            fc = _eval_sigma(c, n)
        else:
            a = c
            c = d
            fc = fd
            d = a + inv_phi * (b - a)
            fd = _eval_sigma(d, n)

        # keep the best seen so far
        if fc < best_score:
            best_score = fc
            best_sigma = c
        if fd < best_score:
            best_score = fd
            best_sigma = d

        # stop when interval is very small
        if b - a < 1e-3:
            break

    return best_sigma, best_score


def _hill_climb(seq: list[float], score: float, deadline: float) -> tuple[list[float], float]:
    """
    Very cheap local search: repeatedly move a tiny amount of mass from one
    index to another while preserving the total sum.  The move is accepted
    only if it improves the evaluation.  The routine stops when the time
    budget expires.
    """
    n = len(seq)
    best_seq = seq
    best_score = score
    # pre‑computed step size factor (relative to the current value)
    step_factor = 0.01

    while time.time() < deadline:
        i, j = np.random.randint(n), np.random.randint(n)
        if i == j:
            continue

        # amount to move from i to j (always non‑negative)
        max_move = step_factor * best_seq[i]
        if max_move <= 0.0:
            continue
        delta = np.random.rand() * max_move

        # generate the candidate
        cand = best_seq.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = evaluate_sequence(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand

    return best_seq, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function sequence (list of non‑negative floats) that
    attempts to minimise `evaluate_sequence`.  The method starts from a
    Gaussian profile, optimises its width with a golden‑section search and
    finishes with a tiny hill‑climbing phase.  The function respects the
    supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small margin for finalisation
    deadline = start_time + budget_s - 0.5

    # ---------------------------------------------------------
    # 1. Determine the length of the construction
    # ---------------------------------------------------------
    try:
        n = len(height_sequence_1)  # type: ignore[name-defined]
        print(f"Using length from existing construction: n = {n}")
    except Exception:
        # fallback to a random reasonable length
        n = np.random.randint(2000, 8000)
        print(f"height_sequence_1 not available – using random length: n = {n}")

    # ---------------------------------------------------------
    # 2. Coarse bounds for the Gaussian width σ
    # ---------------------------------------------------------
    sigma_min = max(0.05 * n, 1.0)               # avoid too narrow spikes
    sigma_max = max(0.9 * n, sigma_min + 1.0)   # stay away from the uniform limit
    print(f"Searching σ in [{sigma_min:.2f}, {sigma_max:.2f}]")

    # ---------------------------------------------------------
    # 3. Golden‑section search for the optimal σ
    # ---------------------------------------------------------
    best_sigma, best_score = _golden_section_search(
        n=n,
        a=sigma_min,
        b=sigma_max,
        deadline=deadline,
        max_iter=15,
    )
    print(f"Golden‑section finished: σ* = {best_sigma:.2f}, score = {best_score:.6f}")

    # Build the best Gaussian found so far
    best_seq = gaussian_sequence(n, best_sigma)

    # ---------------------------------------------------------
    # 4. Optional local improvement (hill climbing)
    # ---------------------------------------------------------
    if time.time() < deadline:
        print("Starting hill‑climbing refinement...")
        best_seq, best_score = _hill_climb(best_seq, best_score, deadline)
        print(f"Hill‑climb finished: final score = {best_score:.6f}")

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq
