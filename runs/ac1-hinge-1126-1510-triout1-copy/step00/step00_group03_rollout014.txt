def project_simplex(v):
    # ensure nonnegative sum to 1
    # from https://github.com/inducer/pythontools.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Simulated Annealing + Subgradient Descent optimizer for minmax convolution ratio.
The algorithm searches over sequence length n, random Fourier-shaped sequences,
and refines promising candidates using subgradient descent.
"""

import time
import math
import numpy as np

def project_simplex(v):
    """Project vector v onto the probability simplex (non-negative & sum=1)."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_self(a):
    """Return full convolution of a with itself using real FFT (fast)."""
    n = a.shape[0]
    # choose next power of two for speed
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)[:2 * n - 1]
    return conv_full

def compute_score(a):
    """Compute the objective value 2*n*max(conv)/sum(a)^2.
       a is a 1‑D numpy array of non‑negative numbers."""
    conv = conv_self(a)
    max_conv = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return float(2.0 * n * max_conv / (s * s))

def subgradient_step(a, step):
    """Perform one subgradient step for minimizing max(conv)."""
    n = a.shape[0]
    conv = conv_self(a)
    max_val = conv.max()
    # indices where conv is within a tiny tolerance of the max
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]

    # compute subgradient: g_i = 2 * sum_{k in active} a_{k-i} (if in range)
    grad = np.zeros_like(a)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # slice of a needed: a[k - i] where i runs from i_min to i_max
        idx = np.arange(i_min, i_max + 1)
        grad[i_min:i_max + 1] += a[k - idx]
    grad *= 2.0

    # take a step opposite to the subgradient
    a_new = a - step * grad
    # projection onto simplex (non‑negative & sum 1)
    a_proj = project_simplex(a_new)
    return a_proj

def local_refine(a, time_limit):
    """Refine a candidate using subgradient descent until time expires."""
    start = time.time()
    best_a = a.copy()
    best_score = compute_score(best_a)
    n = a.shape[0]
    step0 = 1.0
    iteration = 0
    while time.time() - start < time_limit:
        iteration += 1
        step = step0 / math.sqrt(iteration)
        a = subgradient_step(a, step)
        # occasional random jitter to escape flat plateaus
        if iteration % 100 == 0 and np.random.rand() < 0.3:
            jitter = np.random.randn(n) * 0.01
            a = project_simplex(a + jitter)
        scr = compute_score(a)
        if scr < best_score:
            best_score = scr
            best_a = a.copy()
        # early exit if already very good
        if best_score < 1.4:  # safe guard – cannot be much lower than 1
            break
    return best_a, best_score

def random_fourier_sequence(rng, n, max_freq=4):
    """Generate a non‑negative sequence as a sum of a few cosine waves."""
    i = np.arange(n, dtype=float)
    # start with a flat baseline of 1
    a = np.ones(n, dtype=float)
    num_terms = rng.integers(1, max_freq + 1)
    for _ in range(num_terms):
        freq = rng.integers(1, n // 2 + 1)
        amp = rng.uniform(0.1, 0.9)  # positive amplitude shifts mass toward edges
        a += amp * np.cos(2.0 * np.pi * freq * i / (n - 1))
    # ensure non‑negativity by a simple offset
    min_val = a.min()
    if min_val < 0:
        a += -min_val + 1e-8
    # normalise to sum 1 (the objective is scale‑invariant, but normalisation
    # makes the projection step easier)
    a /= a.sum()
    return a

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function height sequence that minimizes
    2 * n * max(conv(a,a)) / (sum(a)**2).

    The routine mixes three simple strategies:
    * random Fourier‑style sequences,
    * local sub‑gradient descent,
    * occasional simulated‑annealing style jitter.

    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

"""
Improved optimizer for the step‑function “max‑convolution” bound.

The objective to minimise is

    2 * n * max(conv(a, a)) / (sum(a) ** 2),

with `a` a non‑negative vector (the heights of the step function).
The ratio is scale‑invariant, therefore we fix sum(a)=1 and minimise
`2 * n * max(conv(a, a))`.

The algorithm mixes three cheap, complementary strategies:

1. **Random Fourier‑style candidates** – a flat baseline plus a few
   cosine waves.  By varying amplitude and frequency we obtain a
   diverse pool of shapes while keeping the vector non‑negative.

2. **Sub‑gradient descent** – starting from a candidate we repeatedly
   move mass away from the indices that currently realise the maximal
   convolution value.  After each step the vector is projected onto the
   probability simplex (non‑negative, sum = 1).  This directly reduces
   the `max(conv)` and often finds flatter convolutions than the uniform
   distribution (which gives the trivial bound 2).

3. **Small simulated‑annealing perturbations** of the best sequence so
   far – occasionally transfer a tiny amount of mass between two
   positions and keep the change if the bound improves.

All three searches run in parallel until the time budget expires.
The best sequence discovered is returned as a plain list of Python
floats, ready to be fed to `evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full convolution a * a using a real FFT (fast and accurate)."""
    n = a.shape[0]
    # length for zero‑padding: next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)[: 2 * n - 1]
    return conv_full


def compute_score(a: np.ndarray) -> float:
    """Return 2*n*max(conv(a,a)) / (sum(a)**2).  (sum(a) is assumed >0.)"""
    conv = conv_self(a)
    max_conv = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))


def subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    One sub‑gradient descent step for minimising max(conv(a,a)).
    The gradient of the active (maximal) convolution entries is taken,
    a step is performed and the result is projected onto the simplex.
    """
    n = a.shape[0]
    conv = conv_self(a)
    max_val = conv.max()
    eps = 1e-12
    active = np.where(conv >= max_val - eps)[0]

    grad = np.zeros_like(a)
    for k in active:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        # contribution a_{k-i}
        grad[i_min : i_max + 1] += a[k - idx]
    grad *= 2.0

    a_new = a - step * grad
    return project_simplex(a_new)


def local_refine(a0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Starting from `a0` perform sub‑gradient descent for as long as
    `time_limit` seconds remain.  Returns the best vector seen and its
    score.
    """
    start = time.time()
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(best_a)

    step0 = 1.0
    it = 0
    while time.time() - start < time_limit:
        it += 1
        step = step0 / math.sqrt(it)
        a = subgradient_step(a, step)

        # occasional jitter to escape possible plateaus
        if it % 100 == 0 and np.random.rand() < 0.3:
            jitter = np.random.randn(a.shape[0]) * 0.02
            a = project_simplex(a + jitter)

        scr = compute_score(a)
        if scr < best_score:
            best_score = scr
            best_a = a.copy()
            # a very good solution is unlikely to improve much further,
            # we can break early
            if best_score < 1.3:
                break
    return best_a, best_score


def random_fourier_sequence(rng: np.random.Generator, n: int, max_terms: int = 4) -> np.ndarray:
    """
    Build a non‑negative sequence as a flat baseline plus a few cosine waves.
    The amplitudes are chosen so that the final vector stays non‑negative.
    """
    i = np.arange(n, dtype=float)
    a = np.ones(n, dtype=float)  # flat baseline
    n_terms = rng.integers(1, max_terms + 1)
    for _ in range(n_terms):
        # random frequency (at most half the length to avoid aliasing)
        freq = rng.integers(1, max(2, n // 2))
        # positive amplitude pushes mass towards the ends; we keep it ≤0.9
        amp = rng.uniform(0.05, 0.9)
        a += amp * np.cos(2.0 * np.pi * freq * i / (n - 1))
    # shift upward if needed, then normalise to sum=1
    min_val = a.min()
    if min_val < 0.0:
        a += -min_val + 1e-10
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function height vector that lowers the evaluation
    bound.  Three complementary heuristics are employed (see module docstring).
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # spare a little margin

    best_seq = None
    best_score = float("inf")

    # a few useful lengths – longer vectors give more freedom but cost more
    candidate_lengths = [200, 400, 800, 1200, 1600, 2000]

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1. cheap random Fourier‑style candidate
        # ------------------------------------------------------------------
        n = int(rng.choice(candidate_lengths))
        a = random_fourier_sequence(rng, n)

        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            print(f"[{time.time():.1f}] New cheap candidate – n={n}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 2. occasional deeper refinement of a random Dirichlet vector
        # ------------------------------------------------------------------
        if rng.random() < 0.08 and time.time() < deadline - 5.0:
            # start from a genuinely random probability vector
            a0 = rng.random(n)
            a0 /= a0.sum()
            # give the refinement routine a few seconds
            remaining = deadline - time.time() - 0.2
            a_ref, scr_ref = local_refine(a0, time_limit=remaining)
            if scr_ref < best_score:
                best_score = scr_ref
                best_seq = a_ref.tolist()
                print(f"[{time.time():.1f}] Refined Dirichlet – n={n}, score={best_score:.6f}")

        # ------------------------------------------------------------------
        # 3. small simulated‑annealing perturbation of the current best
        # ------------------------------------------------------------------
        if best_seq is not None and rng.random() < 0.07 and time.time() < deadline - 1.0:
            cur = np.array(best_seq, dtype=float)
            ncur = cur.shape[0]
            i, j = rng.choice(ncur, size=2, replace=False)
            # move a few percent of mass from i to j
            delta = (rng.random() - 0.5) * 0.02 * cur[i]
            cur[i] = max(cur[i] - delta, 0.0)
            cur[j] = max(cur[j] + delta, 0.0)
            cur = project_simplex(cur)
            scr = compute_score(cur)
            if scr < best_score:
                best_score = scr
                best_seq = cur.tolist()
                print(f"[{time.time():.1f}] SA perturbation – n={ncur}, score={best_score:.6f}")

    # fallback – should never happen, but keep the code robust
    if best_seq is None:
        n_fallback = 500
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = compute_score(np.array(best_seq, dtype=float))
        print("Fallback to uniform sequence.")

    # ensure we return plain Python floats (no NumPy scalars)
    best_seq = [float(x) for x in best_seq]
    print(f"Search finished – best score = {best_score:.6f} (length {len(best_seq)})")
    return best_seq
