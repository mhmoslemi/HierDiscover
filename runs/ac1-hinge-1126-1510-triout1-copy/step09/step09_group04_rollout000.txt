def adam_optimize(a_start, deadline, n, fft_len):
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    lam_factor = 0.3  # starting; we can adapt scaling with iteration.
    while time.time() < deadline:
        t += 1
        lam = lam_factor  # could adapt as max_c * lam_factor but we need compute max_c inside objective_and_grad anyway.
        # compute obj and grad
        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor=lam_factor)
        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1 - beta1**t)
        v_hat = v/(1 - beta2**t)
        y -= 0.01 * m_hat / (np.sqrt(v_hat) + eps)
        # Optionally adjust lam_factor over time: gradually decrease.
        if t % 200 == 0:  # dec lam_factor
            lam_factor = max(1e-4, lam_factor * 0.9)
        # compute current a & fast_score

        # track best

            best_a_local = a.copy()

def subgradient_descent(a_start, deadline, n, fft_len):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a,...)
    it = 0
    while time.time() < deadline:
        it += 1
        # compute convolution and mask
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
        max_val = conv.max()
        # mask of max (approx)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        # compute subgradient: g = 2 * reverse(a) * mask via conv
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = 2 * np.real(np.fft.ifft(fa_rev * fm))[n-1:n-1+n] # correct indices
        # step size
        eta = 0.5 / np.sqrt(it)  # tune
        # gradient descent step
        a_new = a - eta * grad
        # project onto simplex to maintain nonnegative and sum=1
        a_new = simplex_projection(a_new)
        # evaluate
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

"""
Propose a step‐function (non‑negative sequence) that minimises

    2·n·max(a*a) / (∑a)²

using a mix of:

* random initialisations (uniform, edge‑heavy, sinusoidal, previous best)
* exponentiated‑gradient descent on the true max‑convolution (sub‑gradient)
* projected Euclidean sub‑gradient descent (Adam‑free)
* a deterministic hill‑climb that shifts mass from the most harmful
  coordinate to the least harmful one.

The search is time‑bounded (default 1000 s).  The code keeps the best
sequence found before the deadline and finally evaluates it with the
provided `evaluate_sequence` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    a = rng.random(n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """A few heavy edges, rest uniform."""
    a = np.full(n, 1.0 / n)
    edge_frac = rng.uniform(0.005, 0.02)
    m_edge = max(1, int(n * edge_frac))
    heavy = np.sqrt(n / 2.0) / n
    a[:m_edge] = heavy
    a[-m_edge:] = heavy
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def sinusoidal_initial(n: int) -> np.ndarray:
    """Sinusoidal bump – always non‑negative."""
    t = np.arange(n, dtype=float)
    # one full wave, shifted up to stay non‑negative
    s = np.sin(2.0 * np.pi * t / n) + 1.0
    s = np.maximum(s, 0.0)
    s /= s.sum()
    return s

# Exponentiated‑gradient descent on the true max‑convolution

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Multiplicative (exponentiated) sub‑gradient descent.
    Keeps the vector on the probability simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full self‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[n - 1:n - 1 + n]

        # step size – classic 1/√t schedule
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated update (with clipping for stability)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = best_a.copy()

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score

def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int):
    """
    Projected sub‑gradient descent on the (non‑smoothed) max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_full[n - 1:n - 1 + n]

        # step size (decreasing like 1/√t)
        eta = 0.5 / np.sqrt(it)

        # Euclidean gradient step followed by simplex projection
        a_new = a - eta * g
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

# Deterministic fine‑grained hill‑climb

def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  The step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        # convolution and index of its maximum
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: index with smallest partner value (ignoring i_max)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # renormalise (protect against round‑off)
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep current step size for the next iteration
        else:
            # reject – make the move finer
            step = max(step * 0.8, min_step)
    return a, score

            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase‑2] new best after sub‑grad restart {r} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = fine_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            phase3_deadline,
            init_step=0.2,
            min_step=1e-6,
        )
        print(f"[Phase‑3] hill‑climb best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and evaluation
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s > 0.0:
        best_a = best_a / s
    else:
        best_a = np.full(n, 1.0 / n)

    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returns {final_eval:.6f}")
    except Exception as exc:
        print("[final] evaluation raised an exception:", exc)

    return final_seq
