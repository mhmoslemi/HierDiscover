def variance_descent(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    # compute avg = 1/(2*n-1) (since sum=1)
    avg = 1.0/(2*n-1)
    best_a = a.copy()
    conv = np.convolve(a, a)
    best_score = fast_score(best_a, fft_len)
    it = 0
    while time.time() < deadline:
        conv = np.convolve(a, a)  # maybe compute each iteration
        diff = conv - avg
        # gradient via FFT:
        a_rev = a[::-1]
        fdiff = np.fft.fft(diff, fft_len)
        farev = np.fft.fft(a_rev, fft_len)
        grad_fft = fdiff * farev
        grad = np.real(np.fft.ifft(grad_fft))[n-1:n-1+n] * 4.0
        # step size schedule
        eta = 0.5 / np.sqrt(it+1)
        # update with mirror descent (exponential) or simple projected gradient
        a_new = a - eta * grad
        # projection onto simplex to ensure non-negative sum==1
        a_new = simplex_projection(a_new)   # ensures sum=1 and non-negativity
        # Compute fast_score
        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it+=1
    return best_a, best_score

def variance_gradient(a, n, fft_len):
    avg = 1.0/(2*n-1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

"""
Improved step‑function optimizer (≈ 1 000 s budget)

The optimiser works in five phases:

1️⃣  **Smoothed L‑BFGS** – many random restarts on a log‑sum‑exp
    surrogate of the max‑convolution.  The λ‑schedule goes down to 1e‑6,
    and each restart receives up to 1200 L‑BFGS iterations.

2️⃣  **Exponentiated‑gradient descent** – sub‑gradient descent on the
    true max‑convolution (the objective 2·n·max(conv)).  The stepsize
    ηₜ = η₀/√t with η₀=12.

3️⃣  **Variance‑reduction descent** – we minimise the variance of the
    self‑convolution (which automatically reduces its peak).  The exact
    gradient  ∇ Var(conv) = 4·((conv−avg) * a_rev) is obtained by a
    single FFT.  After each step we project back onto the probability
    simplex.

4️⃣  **Deterministic greedy descent** – the safe mass‑transfer move from
    the index with the largest gradient to the one with the smallest,
    using the analytical bound that never raises the peak.

5️⃣  **Simulated‑annealing hill‑climb** – random mass moves accepted
    with a Metropolis rule while the temperature is linearly cooled.

The algorithm works with a larger problem size (n=8000) – more degrees
of freedom give a lower theoretical bound (≈ 1).  All phases keep the
current best sequence and return it after the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv) assuming a sums to 1 (fast via FFT)."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "beta",
            "arcsine",
            "cosine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "beta":
        # Beta(0.8,0.8) – mild U‑shape
        a = (np.arange(n) + 0.5) / n
        a = np.power(a, -0.2) * np.power(1 - a, -0.2)
    elif mode == "arcsine":
        # Discretised arcsine (Beta(0.5,0.5))
        i = np.arange(n, dtype=float)
        a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + 1e-12)
    else:  # cosine hump
        theta = np.pi * np.arange(n) / n
        a = 0.5 * (1 + np.cos(theta))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Phase‑1 : smoothed L‑BFGS (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Input y are logits such that a = softmax(y).  Returns (value, grad_y).
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """Run a cascade of L‑BFGS optimisations for the lam‑schedule."""
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exponentiated‑gradient descent on the hard max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
):
    """Mirrored (exponentiated) SGD on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑3 : variance‑reduction gradient descent (flatten convolution)
# ----------------------------------------------------------------------


def variance_gradient(a: np.ndarray, n: int, fft_len: int):
    """
    Gradient of the variance of the self‑convolution w.r.t. a.
    Returns (grad, conv) where conv is the full convolution a*a.
    """
    # average of convolution under sum(a)=1
    avg = 1.0 / (2 * n - 1)

    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    diff = conv - avg
    a_rev = a[::-1]

    fdiff = np.fft.fft(diff, fft_len)
    farev = np.fft.fft(a_rev, fft_len)
    grad_full = np.real(np.fft.ifft(fdiff * farev))

    # Extract the n relevant components and multiply by 4 (see derivation)
    grad = 4.0 * grad_full[n - 1 : n - 1 + n]
    return grad, conv


def variance_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """Projected gradient descent on Var(conv) with simplex projection."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        grad, conv = variance_gradient(a, n, fft_len)

        # Simple decaying step size; tuned empirically
        eta = 0.5 / np.sqrt(it + 1.0)

        a_new = a - eta * grad
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # occasional full recomputation to avoid drift
        if rng.random() < 0.001:
            a_new = simplex_projection(a_new)  # re‑project (safe)

        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑4 : deterministic greedy descent (analytic safe step)
# ----------------------------------------------------------------------


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """Move mass from the index with largest gradient to the smallest
    with a maximal safe step that never raises the peak."""
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        # admissible δ
        delta_src = a[src]

        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.92  # safety factor
        if delta <= 0.0:
            break

        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑5 : simulated‑annealing hill‑climb (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted with Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.15
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        dE = new_score - curr_score

        accept = False
        if dE < 0.0:
            accept = True
        elif rng.random() < np.exp(-dE / temp):
            accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # --------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previously found construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split (roughly)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.10 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.50 * budget_s   # exp‑gradient
    phase3_end = start_time + 0.78 * budget_s   # variance descent
    phase4_end = start_time + 0.93 * budget_s   # deterministic greedy
    # remaining → simulated annealing

    # --------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS (many restarts, finer λ‑schedule)
    # --------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        0.00005,
        0.00002,
        0.00001,
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Occasionally perturb the current best; otherwise fresh random start
        if rng.random() < 0.30:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=12.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – variance‑reduction gradient descent (flatten conv)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = variance_descent(
            best_a, phase3_end, n, fft_len, rng
        )
        print(f"[var‑desc] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – deterministic greedy descent (analytic safe steps)
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = deterministic_descent(
            best_a, phase4_end, n, fft_len
        )
        print(f"[det‑desc] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq

    return final_seq
