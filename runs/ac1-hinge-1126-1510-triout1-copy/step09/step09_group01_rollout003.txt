"""
Propose a low‑score step sequence.

Algorithm
---------
* Use a simple stochastic construction: pick a random subset of the
  indices, put equal probability mass on the chosen positions and zero
  elsewhere.  For a subset of size *k* the auto‑convolution of the
  resulting vector has an expected maximum close to the average
  (≈1/(2n‑1)).  By sampling many subsets we can quickly obtain a vector
  whose convolution peak is far below the uniform‑vector value (≈2).
* The search runs until the global deadline (default 1000 s).  Each
  candidate is evaluated with a fast FFT‑based `fast_score` that
  computes 2·n·max(conv) for a vector already normalised to Σa = 1.
* When a new best score is found it is printed.  Finally the best
  vector is returned; the provided `evaluate_sequence` function is called
  once for reporting the true score.
* The implementation is self‑contained, uses only NumPy and the
  standard library, and works for any time budget ≤ 1000 s.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a vector already normalised to Σa = 1.
    The convolution is performed via FFT for speed.
    """
    # FFT of the padded vector
    fa = np.fft.rfft(a, fft_len)
    # element‑wise square (convolution in time domain)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

# Main entry point
# ----------------------------------------------------------------------

    print(f"[init] uniform fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Random‑subset search.
    #   – pick a random subset of size k (k = fraction·n)
    #   – assign uniform mass 1/k on the selected indices
    #   – evaluate the fast score
    # The subset size is varied; fractions in (0.15, 0.85) give good diversity.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # Fractions of n to try for the support size.
    fractions = np.array([0.15, 0.20, 0.25, 0.30, 0.35,
                          0.40, 0.45, 0.50, 0.55, 0.60,
                          0.65, 0.70, 0.75, 0.80, 0.85])

    # Main random‑subset loop
    iteration = 0
    while time.time() < deadline:
        # Choose a support size k
        frac = float(rng.choice(fractions))
        k = max(2, int(frac * n))

        # Sample a subset without replacement
        # `rng.permutation` is faster than many calls to `choice`.
        idx = rng.permutation(n)[:k]

        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / k      # uniform mass on the chosen positions

        sc = fast_score(a, fft_len, n)

        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[iter {iteration:06d}] new best fast_score = {best_score:.6f} (k={k})")

        iteration += 1

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
