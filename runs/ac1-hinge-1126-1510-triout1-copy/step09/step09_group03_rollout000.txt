"""
step‑function optimizer (≈ 1 000 s budget)

Goal: minimise `2·n·max(conv(a,a))/(sum(a)**2)`.  The quantity is
scale‑invariant, therefore we keep `a` normalised (`∑a = 1`) and
optimise the simpler objective `2·n·max(conv(a,a))`.

Strategy
--------
1. **Multi‑scale start** – if a previous solution exists, interpolate it
   to the current length; otherwise start from a uniform vector.

2. **Smoothed Adam optimisation** – minimise a log‑sum‑exp
   approximation of `max(conv)` (the same objective used for the
   L‑BFGS phase) but with the Adam optimiser.  The smoothing parameter
   (`lam`) is decreased geometrically during the run, turning the
   smooth surrogate into the true max.  Several random restarts are
   performed; the best vector is kept.

3. **Exponentiated‑gradient sub‑gradient descent** – work on the exact
   `max` (hard) objective with a decreasing stepsize, equipped with
   the multiplicative update that automatically respects non‑negativity
   and the simplex constraint.

4. **Deterministic greedy descent** – repeatedly move a small amount of
   mass from the index whose gradient most harms the peak to the index
   that helps the most.  An analytically derived safe step size guarantees
   that the peak never increases.

5. **Simulated‑annealing hill‑climb** – final fine‑tuning with random
   mass transfers accepted according to a Metropolis rule while the
   temperature is linearly cooled.

All phases share a global “best” vector and stop as soon as the time
budget is exhausted.  The final vector is evaluated with
`evaluate_sequence` and stored in the global `height_sequence_1` for
future runs.
"""

import time

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a vector that already satisfies
    sum(a)=1.  This is the quantity that `evaluate_sequence` would
    return after cancelling the normalisation factor.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Phase 1 – Smoothed Adam optimisation (log‑sum‑exp surrogate)
# --------------------------------------------------------------

def adam_smooth_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         lr: float = 0.05,
                         beta1: float = 0.9,
                         beta2: float = 0.999,
                         eps: float = 1e-8,
                         lam0: float = 0.5,
                         lam_min: float = 1e-4,
                         lam_decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth max (log‑sum‑exp) of the self‑convolution.
    Returns the best vector encountered (projected onto the simplex) and its
    fast_score.
    """
    # work in log‑space and use softmax to enforce positivity & sum=1
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = None
    best_score = np.inf
    it = 0

    while time.time() < deadline:
        it += 1
        lam = max(lam0 * (lam_decay ** it), lam_min)

        # softmax
        max_y = np.max(y)
        e = np.exp(y - max_y)
        a = e / e.sum()

        # convolution via FFT
        fa = np.fft.fft(a, fft_len)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_c = conv_full.max()
        shifted = conv_full - max_c
        exp_shift = np.exp(shifted / lam)
        Z = exp_shift.sum()
        # smooth max (not needed for gradient magnitude, but kept for completeness)
        # smooth_max = max_c + lam * np.log(Z)

        # gradient of smooth max w.r.t. convolution entries
        w = exp_shift / Z                     # d smooth_max / d conv_i

        # back‑propagation through convolution (a_rev * w)
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        g_vec = conv_rev[n - 1: n - 1 + n]   # gradient w.r.t. a (pre‑softmax)

        # chain rule for softmax
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        # Adam update on y
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate true objective
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # occasional log
            if it % 500 == 0:
                print(f"[Adam] iter={it}, lam={lam:.2e}, best_score={best_score:.6f}")

    # if never improved (should not happen) fallback to start
    if best_a is None:
        best_a = a_start.copy()
        best_score = fast_score(best_a, fft_len)

    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated‑gradient) on the hard max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # shape 2n‑1

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # adaptive stepsize
        eta = eta0 / sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

def deterministic_descent(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int) -> tuple[np.ndarray, float]:
    """Move mass from the most harmful index to the most helpful one."""
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))   # mass to remove
        dst = int(np.argmin(grad))   # where to add
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a

        # admissible δ:
        delta_src = a[src]

        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.92   # safety margin
        if delta <= 0.0:
            break

        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to keep numerical drift small
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score

def simulated_annealing(a_start: np.ndarray,

                        deadline: float,
                        n: int,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:

    a = a_start.copy()

    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.05, 1e-6)

    max_fraction = 0.30          # allow relatively large moves early on
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with minimal evaluation score.
    The function runs for at most ``budget_s`` seconds and returns the best
    list of floats found.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------
    # Problem size – a larger n gives a better theoretical bound.
    # ----------------------------------------------------------
    n = 8000                     # try a longer sequence (still fast enough)
    fft_len = next_pow2(2 * n - 1)

    # ----------------------------------------------------------
    # Initialise from a previous construction if present.
    # ----------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Time budget split
    # ----------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # safety margin

    phase1_end = start_time + 0.20 * budget_s   # Adam (smooth)
    phase2_end = start_time + 0.60 * budget_s   # exp‑gradient
    phase3_end = start_time + 0.85 * budget_s   # deterministic descent
    # remaining → annealing

    # ----------------------------------------------------------
    # Phase 1 – Smoothed Adam with several random restarts
    # ----------------------------------------------------------
    while time.time() < phase1_end:
        # decide start point: perturbed best or fresh random
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = adam_smooth_optimize(
            a_start,
            phase1_end,
            n,
            fft_len,
            rng,
            lr=0.07,
            lam0=0.4,
            lam_min=5e-5,
            lam_decay=0.9992,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient descent (hard max)
    # ----------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=10.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 3 – Deterministic greedy descent
    # ----------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_descent(
            best_a,
            phase3_end,
            n,
            fft_len,
        )
        print(f"[det‑desc] fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill‑climb
    # ----------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Final evaluation with the official scorer
    # ----------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # store for potential reuse
    globals()['height_sequence_1'] = final_seq

    return final_seq
