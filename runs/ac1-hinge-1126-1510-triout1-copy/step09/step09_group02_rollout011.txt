def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    n = 6000 # increased size
    fft_len = next_pow2(2*n-1)

    # initial a: use random_initial as weighted mix and maybe add some smoothness
    a0 = random_initial(n, rng)

    # gradient descent in log-space with mask of top 10% of conv entries (or dynamic)
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    start = time.time()
    deadline = start + budget_s - 0.1

    # log-parameter y
    eps = 1e-12
    y = np.log(best_a + eps)

    it = 0
    while time.time() < deadline - 30: # leave 30s for hill climb
        # compute a from logits
        a = np.exp(y - np.max(y))
        a /= a.sum()
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_c = conv.max()
        # compute gradient of max (subgradient)
        eps_mask = max_c * 1e-12 + 1e-12
        mask = (conv >= max_c - eps_mask).astype(float)

        # gradient of sum(mask*conv) w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fm * fa_rev))
        # gradient for each a_i (derivative of conv index contributions)
        g = 2.0 * grad_full[n-1:n-1+n]  # shape (n,)

        # step size schedule
        eta = 15.0 / np.sqrt(it + 1.0)  # start around 15
        # update logits
        y -= eta * g

        # evaluate score
        score = 2.0 * n * max_c
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # print progress occasionally
            if it % 1000 == 0:
                print(f"[expgd] it={it}, score={best_score:.6f}")

        # occasionally add small noise to logits to escape plateaus
        if it % 5000 == 0 and it > 0:
            y += rng.normal(scale=0.01, size=n)

        it += 1

    # after loop, set a = best_a (or recompute from best y)
    a = best_a.copy()

    # Hill-climbing phase
    curr_score = best_score
    conv = np.convolve(a, a)
    # incremental hill climb similar as earlier but maybe more targeted.
    # We'll run for remaining time.
    max_fraction = 0.15
    recompute_every = 2000
    accepted = 0
    it = 0
    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i==j:
            continue
        # move mass from larger to smaller
        src, dst = (i,j) if a[i] > a[j] else (j,i)  # src > dst
        if a[src] == 0:
            continue
        delta = a[src] * rng.random() * max_fraction
        # compute diff updates
        diff = np.zeros(2*n-1)
        diff[src:src+n] -= a
        diff[dst:dst+n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        if new_score < curr_score:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        it += 1

    # final sequence to return
    return a.tolist()

def exp_gradient_flatten(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_c = conv.max()
        avg = 1.0/(2*n-1)  # because sum conv =1
        # choose threshold between avg and max
        # let beta decrease over time: start 0.9, reduce to 0.1
        beta = max(0.1, 0.9 * (1 - it/500000))  # decreasing
        thresh = avg + beta * (max_c - avg)
        mask = (conv > thresh).astype(float)
        # if mask sum zero (unlikely) break
        if mask.sum() == 0:
            break
        # compute gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad[n-1:n-1+n]

        # step size schedule
        eta = 5.0 / np.sqrt(it + 1.0)

        score = fast_score(a, fft_len)
        if score < best_score:

            best_a = a.copy()

        if it % 10000 == 0:

def hill_climb_incremental(a, curr_score, fft_len, n, deadline, rng, max_fraction=0.15):
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)
    diff = np.zeros(2*n-1)
    accepted = 0
    moves = 0
    while time.time() < deadline:
        src = rng.integers(n)
        dst = rng.integers(n)
        if src == dst:
            continue
        if best_a[src] < best_a[dst]:
            src, dst = dst, src
        if best_a[src] == 0:
            continue
        delta = best_a[src] * rng.random() * max_fraction
        # compute diff updates
        diff.fill(0.0)
        diff[src:src+n] -= best_a
        diff[dst:dst+n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1
        moves += 1
        # adapt step size based on acceptance
        if accepted and accepted % 5000 == 0:
            max_fraction *= 0.9  # reduce moves
        # recompute conv occasionally for numerical stability
        if moves % 4000 == 0:
            conv = np.convolve(best_a, best_a)
    return best_a, best_score
```

"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

with sum(a)=1 the score is simply 2·n·max(conv).
The algorithm works in three stages:

1. Initialise a probability vector on the simplex (various random patterns,
   possibly interpolating a previous solution).

2. Long “flattening” phase: repeated exponentiated‑gradient updates on the
   true max‑convolution.  Instead of using only the exact max entry,
   a mask of the largest convolution values (a moving threshold between the
   current maximum and the global average) is used, giving a smoother
   sub‑gradient and faster convergence.  The step‑size decays as 1/√t.

3. Intensive greedy hill‑climb: move a random fraction of mass from a large
   entry to a small one, accepting only moves that lower the score.
   The move size shrinks as improvements are found.

The whole routine respects the 1000 s budget, keeps every value non‑negative,
and returns a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:                      # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient “flattening” phase
# ----------------------------------------------------------------------


def exp_gradient_flatten(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent that flattens the convolution.
    Uses a dynamic mask that contains the largest convolution entries,
    i.e. values > avg + beta·(max-avg) with a decreasing beta.
    Returns the best vector and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    eps0 = 1e-12
    # theoretical average of the convolution (sum(conv)=1)
    avg = 1.0 / (2 * n - 1)

    while time.time() < deadline:
        # --- convolution and mask -------------------------------------------------
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_c = conv.max()

        # decreasing beta: start large (focus on the very top) and shrink
        beta = max(0.05, 0.9 * (1.0 - it / 400000.0))
        thresh = avg + beta * (max_c - avg)
        mask = (conv > thresh).astype(float)

        if mask.sum() == 0:
            # nothing above threshold – we are already flat enough
            break

        # --- gradient of sum(mask * conv) w.r.t. a -------------------------------
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_full[n - 1:n - 1 + n]                    # (n,)

        # --- multiplicative (exponentiated) update -------------------------------
        eta = eta0 / np.sqrt(it + 1.0)          # decaying step size
        # operate in log‑space for stability
        y = np.log(a + eps0) - eta * g
        a = np.exp(y - np.max(y))
        a /= a.sum()

        # --- keep the best --------------------------------------------------------
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # occasional progress report
            if it % 5000 == 0:
                print(f"[exp‑grad] it={it} score={best_score:.6f} max={max_c:.8f}")

        # tiny random jitter to escape possible plateaus
        if it % 20000 == 0 and it > 0:
            y += rng.normal(scale=0.001, size=n)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (incremental convolution updates)
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.18,
                           recompute_every: int = 1200) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a random fraction of mass from a larger index
    to a smaller one.  Convolution updates are performed incrementally.
    The move size shrinks whenever a batch of improvements is accepted.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)   # exact linear convolution
    diff = np.zeros(2 * n - 1, dtype=float)
    accepted = 0
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue

        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linearised)
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # occasional full recomputation to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

        # shrink the move size slowly as we make progress
        if accepted and accepted % (recompute_every * 2) == 0:
            max_fraction = max(0.02, max_fraction * 0.9)

        it += 1

    return best_a, best_score
