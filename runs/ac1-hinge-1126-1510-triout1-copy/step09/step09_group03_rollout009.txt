"""
Step‑function optimizer – 1000 s budget

We look for a non‑negative vector `a` (∑a = 1) that minimises

    2·n·max(conv(a,a))

where `conv` is the linear convolution.  The programme consists of four
phases:

1. **Smoothed L‑BFGS** – many random restarts on a log‑sum‑exp
   surrogate of the max‑convolution, with a decreasing smoothing
   parameter (`lam`).  The result provides a good basin.

2. **Adam on the smoothed objective** – a momentum based first‑order
   method (Adam) on the same surrogate, using many more iterations.

3. **Adaptive deterministic descent** – moves mass from the index with
   the largest gradient to the smallest one, computes the exact safe
   step size (including the quadratic term of the convolution update)
   and performs a line‑search.

4. **Simulated‑annealing hill‑climb** – random transfers of mass are
   accepted with a Metropolis rule while a temperature is slowly
   cooled.  This phase explores the neighbourhood of the best point
   found so far.

All phases keep the best sequence seen and return it after the
deadline.  The final sequence is evaluated with the official
`evaluate_sequence` function and stored in the global variable
`height_sequence_1` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) assuming a sums to 1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "halfheavy",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # halfheavy – many entries heavy on one half, light on the other
        a = np.full(n, 1.0 / n)
        half = n // 2
        heavy = rng.uniform(0.5, 1.0) * np.sqrt(n / (2.0 * half)) / n
        a[:half] = heavy
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """Smooth approximation of 2·n·max(conv) and its gradient.

    The optimisation variables are the logits `y`; the actual
    probability vector is obtained by softmax.
    """
    # softmax → a
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]

    # chain rule for the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_restart(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """Run a cascade of L‑BFGS optimisation passes over decreasing λ."""
    y = np.log(a_start + 1e-12)  # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def adam_optimize(a_start: np.ndarray, deadline: float, n: int, fft_len: int):
    """Adam on the log‑sum‑exp surrogate (very low λ)."""
    # hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    lam = 1e-4  # very tight smoothing (almost true max)

    # work in logits
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)
    t = 0

    while time.time() < deadline:
        t += 1
        f, grad = objective_and_grad(y, n, fft_len, lam_factor=lam / (np.max(np.exp(y - np.max(y))) + 1e-12))
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # obtain probability vector
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)

        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def deterministic_descent_adaptive(a_start: np.ndarray, deadline: float, n: int, fft_len: int):
    """Move mass from the index with the biggest gradient to the smallest,
    using an exact safe δ (including the quadratic term of the convolution
    update) and a back‑tracking line‑search.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact full convolution, length = 2*n-1
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        # mask of the maximal entries (sub‑gradient support)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # prepare linear part of the convolution update
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        # quadratic part (u*u where u = e_dst - e_src)
        second = np.zeros_like(conv)
        second[2 * src] += 1.0
        second[2 * dst] += 1.0
        second[src + dst] -= 2.0

        # maximal admissible δ from source positivity
        delta_src = a[src]

        # bound from positions where diff > 0 (linear increase)
        pos = diff > 1e-15
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound_lin = np.min(bound_vals)
        else:
            delta_bound_lin = np.inf

        # a very crude bound from the quadratic term (ensure it never exceeds max)
        # For indices where second>0 we solve the quadratic inequality
        quad_pos = second > 0
        if np.any(quad_pos):
            # solve: second*δ² + 2*diff*δ + (conv - max_val) ≤ 0
            a_q = second[quad_pos]
            b_q = 2.0 * diff[quad_pos]
            c_q = conv[quad_pos] - max_val
            # discriminant
            disc = b_q ** 2 - 4.0 * a_q * c_q
            # positive root (the one limiting the step)
            # Only keep roots with a_q > 0
            root = (-b_q + np.sqrt(disc)) / (2.0 * a_q)
            delta_bound_quad = np.min(root[root > 0])
        else:
            delta_bound_quad = np.inf

        delta_max = min(delta_src, delta_bound_lin, delta_bound_quad) * 0.99
        if delta_max <= 0.0:
            break

        # back‑tracking line search – try the full step, otherwise halve
        delta = delta_max
        for _ in range(8):
            # exact convolution update with quadratic term
            conv_new = conv + 2.0 * delta * diff + (delta ** 2) * second
            new_max = conv_new.max()
            if new_max <= max_val + 1e-14:  # non‑increase (tiny tolerance)
                break
            delta *= 0.5
        if delta <= 0.0:
            break

        # accept the move
        conv = conv + 2.0 * delta * diff + (delta ** 2) * second
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional full recompute to keep numerical drift in check
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (enhanced)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """Hill‑climbing with Metropolis acceptance and a slowly cooling schedule."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # start temperature as a modest fraction of the current score
    temp0 = max(best_score * 0.03, 1e-6)
    start_time = time.time()
    max_frac = 0.25   # maximal fraction of source mass we may move
    recompute_every = 4000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update (same formula as deterministic descent)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        second = np.zeros_like(conv)
        second[2 * src] += 1.0
        second[2 * dst] += 1.0
        second[src + dst] -= 2.0

        conv_new = conv + 2.0 * delta * diff + (delta ** 2) * second
        new_score = 2.0 * n * conv_new.max()
        dE = new_score - curr_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            # accept
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 works well and keeps the runtime reasonable
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise – reuse a previous construction if present, otherwise
    # randomise a diverse starting point.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the current length
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.20 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.55 * budget_s   # Adam
    phase3_end = start_time + 0.80 * budget_s   # deterministic descent
    # remaining → simulated annealing

    # --------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS with many restarts
    # --------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        0.00005,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # random perturbation of the current best with probability 0.3,
        # otherwise a fresh diverse start
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_restart(
            a_start, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed surrogate
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a, phase2_end, n, fft_len)
        print(f"[Adam] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic greedy descent with exact line search
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_descent_adaptive(
            best_a, phase3_end, n, fft_len
        )
        print(f"[det‑desc] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (more aggressive)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the result for possible reuse in a subsequent run
    globals()["height_sequence_1"] = final_seq

    return final_seq
