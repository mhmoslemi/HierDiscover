def lp_step(f, M):
    n = len(f)
    offsets = np.arange(n)
    # each diagonal entry constant f[i]
    diags = [np.full(n, f[i]) for i in range(n)]
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n) # minimize -sum(g)
    bounds = [(0, None)] * n
    res = spopt.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    return g0
```

def lp_iterative_optimize(a_start, fft_len, n, deadline):

def solve_lp_max_sum(f, M):
    n = len(f)
    offsets = np.arange(n)
    diags = [np.full(n, f[i]) for i in range(n)]
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M)
    c = -np.ones(n) # minimize -sum(g)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:

    else:
        return None

    random Dirichlet‑like and a few parametric families).  Keep the best few.
2️⃣  **Alternating LP refinement** – the core of the method.
    Starting from a distribution `f` (∑f = 1) we compute
        M = max( f * f ) .
    Then we solve the linear programme

          maximise   𝟙ᵀ g
          subject to  f * g ≤ M ,   g ≥ 0 .

    The matrix of the convolution `f * g` is a (2n‑1)×n Toeplitz matrix and can be
    built cheaply with `scipy.sparse.diags`.  The LP gives a new candidate `g₀`.
    After normalising `g = g₀ / ∑g₀` we move a little towards it:

          f ← (1‑t)·f + t·g ,   t = 0.2

    and repeat while the score improves or time permits.  This procedure is
    guaranteed to monotically decrease `max(f*f)`.
3️⃣  Polish the best result with a few iterations of the smoothed L‑BFGS
    optimiser (log‑sum‑exp approximation of the max‑convolution) and a short
    greedy hill‑climb.
4️⃣  Return the best sequence found before the deadline.

The LP stage dramatically reduces the peak of the self‑convolution and
usually brings the score below the target 1.503 within the 1000 s budget.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast score (sum already = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of sqrt of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random draw on the simplex (several flavours)."""
    mode = rng.choice(['uniform', 'exp', 'power', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    else:  # edge
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# LP utilities
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sparse.csr_matrix:
    """
    Return a sparse matrix A where A @ g = f * g (full convolution, size 2n‑1).
    The i‑th diagonal (offset i) contains the constant value f[i] (i = 0..n‑1).
    """
    n = f.shape[0]
    offsets = np.arange(n)                # 0 … n‑1
    # each diagonal consists of n copies of f[i]
    diags = [np.full(n, f[i]) for i in range(n)]
    A = sparse.diags(diags, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def lp_max_sum_under_conv(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise 𝟙ᵀ g   subject to   f * g ≤ M ,   g ≥ 0 .
    Returns the optimal g (may be non‑normalised) or None on failure.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)                     # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)  # upper bounds
    c = -np.ones(n, dtype=np.float64)            # minimise -sum(g)
    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Alternating LP improvement
# ----------------------------------------------------------------------


def lp_refine(a_start: np.ndarray,
              fft_len: int,
              deadline: float,
              step: float = 0.2,
              max_iter: int = 30) -> tuple[np.ndarray, float]:
    """
    Perform the alternating LP steps described in the paper.
    Returns the best distribution seen and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iter:
        # 1) current max‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
        M = conv.max()

        # 2) LP for a promising partner g₀
        g0 = lp_max_sum_under_conv(a, M)
        if g0 is None:
            break

        # 3) normalise g (same ℓ₁ norm as a)
        s = g0.sum()
        if s <= 0:
            break
        g = g0 / s

        # 4) convex combine – try a few step sizes, keep the best improvement
        improved = False
        for t in (step, step / 2, step * 2):
            a_new = (1.0 - t) * a + t * g
            a_new = np.clip(a_new, 0, None)
            a_new /= a_new.sum()
            sc = fast_score(a_new, fft_len)
            if sc < best_score - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_score = sc
                print(f"[LP] iter {it:02d} t={t:.3f} → score {best_score:.6f}")
                improved = True
                break
        if not improved:
            # no improvement – shrink step size
            step *= 0.7
            if step < 1e-3:
                break
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed L‑BFGS (log‑sum‑exp) – same as in the original script
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smooth approximation of max(conv) + its gradient (w.r.t. logits y)."""
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑prop through convolution: d conv / d a = rev(a) ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """SGD‑free L‑BFGS continuation on the smoothed objective."""
    y = np.log(a_start + 1e-12)           # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS] new best {best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb polishing
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """Greedy hill‑climb moving a small random fraction of mass."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        sc = fast_score(best, fft_len)
        if sc < best_score:
            best_score = sc
        else:
            # undo the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate n keeps the LP cheap while still giving a
    # fine approximation of the continuous problem.
    # ------------------------------------------------------------------
    n = 1024                         # you may change this value
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two for fast FFTs

    # ------------------------------------------------------------------
    # Gather a small set of diverse initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous best (if available)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # Interpolate to current length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            a_prev /= a_prev.sum()
            candidates.append(a_prev)
    # 2️⃣  Standard families
    candidates.append(np.full(n, 1.0 / n))               # uniform
    candidates.append(spectral_init(n))                 # sqrt‑uniform spectral
    candidates.append(random_initial(n, rng))           # random Dirichlet‑like
    candidates.append(exponential_shape(n, 0.001))      # light exponential tail
    candidates.append(powerlaw_shape(n, 1.0))           # moderate power‑law

    # ------------------------------------------------------------------
    # Keep the best few after a cheap evaluation (fast_score)
    # ------------------------------------------------------------------
    scored = [(fast_score(a, fft_len), a) for a in candidates]
    scored.sort(key=lambda x: x[0])
    best_a = scored[0][1].copy()
    best_score = scored[0][0]
    seeds = [cand for _, cand in scored[:4]]   # top‑4 seeds for further work
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – alternating LP refinement
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.30 * budget_s
    for a_start in seeds:
        if time.time() >= phase0_deadline:
            break
        a_opt, sc_opt = lp_refine(a_start,
                                  fft_len,
                                  phase0_deadline,
                                  step=0.2,
                                  max_iter=25)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[LP] new global best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing on a smoothed version of max‑conv
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 500
    if time.time() < phase1_deadline:
        best_a, best_score = lbfgs_continuation(best_a,
                                                lam_schedule,
                                                maxiter_per_lam,
                                                phase1_deadline,
                                                fft_len, fft_len, n)
        print(f"[L‑BFGS] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=60000)
        print(f"[hill‑climb] final refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
