def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    n = 4000
    fft_len = next_pow2(2*n-1)
    # Setup start sequences
    def init_from_global():
        if 'height_sequence_1' in globals(): ...
    # produce base a0
    a_best = a0
    best_score = fast_score(a_best, fft_len)
    start_time = time.time()
    deadline = start_time + budget_s - 0.05

    # Phase 0: generate candidate sequences by param families and evaluate quick
    # Use 0.1 budget for random families
    phase0_end = start_time + 0.10*budget_s
    while time.time() < phase0_end:
        # choose family and param random
        family = rng.choice(['uniform','exp','power','edge','bimodal','sparse','tri','logistic','doubleexp'])
        if family == 'uniform':
            a = np.full(n, 1/n)
        elif family == 'exp':
            r = rng.uniform(0.96, 0.999)  # high r to not too peaked
            a = (1 - r) * r**np.arange(n)
            # For finite n, need to normalise
        elif family == 'power':
            p = rng.uniform(0.2, 2.0)
            a = (np.arange(1,n+1, dtype=float)**(-p))
        elif family == 'edge':
            edge_frac = rng.uniform(0.005, 0.025)
            m_edge = max(1, int(edge_frac * n))
            a = np.full(n, 1/n)
            heavy = np.sqrt(n / 2.0) / n
            a[:m_edge] = heavy
            a[-m_edge:] = heavy
        elif family == 'bimodal':
            centre = n // 2
            width = rng.integers(int(0.015*n), int(0.06*n))
            peak_factor = rng.uniform(0.5, 1.5)
            a = np.full(n, 1/n)
            left = max(0, centre - width)
            right = min(n, centre + width)
            a[left:right] = peak_factor / n
        elif family == 'sparse':
            m = rng.integers(int(0.01*n), int(0.1*n))
            idx = rng.choice(n, size=m, replace=False)
            vals = rng.random(m) + 1e-6
            a = np.zeros(n)
            a[idx] = vals
        elif family == 'tri':
            # triangular shape peaked at center
            a = np.concatenate([np.arange(1, n//2+1), np.arange(n//2,0,-1)])[:n]
        elif family == 'logistic':
            x = np.linspace(-6, 6, n)
            a = 1/(1+np.exp(x))
        elif family == 'doubleexp':
            # two exponentials from each side meeting in middle
            r1 = rng.uniform(0.96,0.999)
            r2 = rng.uniform(0.96,0.999)
            left = (1 - r1) * r1**np.arange(n)
            right = (1 - r2) * r2**np.arange(n)[::-1]
            a = np.maximum(left, right)  # or sum? We'll sum.
            a = left + right
        else:
            a = np.full(n, 1/n)
        a = np.maximum(a, 0)
        if a.sum() == 0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    # Phase 1: L-BFGS restarts (like original, but more sec)
    phase1_end = start_time + 0.25*budget_s
    # L-BFGS parameters
    lam_schedule = [0.5,0.35,0.25,0.18,0.12,0.08,0.05,0.03,0.018,0.01,0.006,0.003,0.0015,0.0008,0.0004,0.0002,0.0001,0.00005]
    maxiter_per_lam = 800
    while time.time() < phase1_end:
        # start from best with noise, occasionally fresh random
        if rng.random() < 0.3:
            a_start = add_noise(best_a, rng.uniform(0.0005, 0.01), rng)
        else:
            a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] New best fast_score={best_score:.6f}")
    # Phase 2: exponentiated gradient descent (maybe longer)
    phase2_end = start_time + 0.65*budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a, phase2_end, n, fft_len, rng, eta0=30.0)
        print(f"[exp‑grad] fast_score={best_score:.6f}")
    # Phase 3: deterministic descent (longer)
    phase3_end = start_time + 0.90*budget_s
    if time.time() < phase3_end:
        best_a, best_score = deterministic_descent(best_a, phase3_end, n, fft_len)
        print(f"[det‑desc] fast_score={best_score:.6f}")
    # Phase 4: final polishing L-BFGS on smoothed objective small lam
    if time.time() < deadline:
        # Do a final L-BFGS with small lam schedule for refinement
        final_a, final_score = lbfgs_continuation(best_a, [0.00005, 0.00002, 0.00001], 400, deadline, fft_len, n)
        if final_score < best_score:
            best_score = final_score
            best_a = final_a.copy()
            print(f"[L‑BFGS‑final] fast_score={best_score:.6f}")
    # Phase 5: simulated annealing fine-tuning
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a, best_score, deadline, n, rng)
        print(f"[anneal] fast_score={best_score:.6f}")
    # End: evaluate
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    globals()['height_sequence_1'] = final_seq
    return final_seq

"""
Step‑function optimizer (≈ 1 000 s budget)

Goal: minimise   2·n·max(conv(a,a)) / (sum(a)^2)
          = 2·n·max(conv)  (we keep ∑a = 1).

Approach
--------
* Work on a probability simplex – the vector is always projected onto the
  simplex after every update.
* Four complementary search phases, each using the remaining time budget:
  1. **Diverse random starts** – hundreds of cheap candidates drawn from a
     collection of parametric families (uniform, exponential, power‑law,
     edge‑heavy, bimodal, sparse, triangular, logistic, double‑exponential).
  2. **Smoothed L‑BFGS** – minimises a log‑sum‑exp surrogate of the max‑convolution
     (softmax‑parameterised a = softmax(y)).  Several restarts with a schedule of
     decreasing smoothness.
  3. **Exponentiated‑gradient descent** – sub‑gradient descent on the true
     (non‑smoothed) max‑convolution using an AdaGrad‑style stepsize.
  4. **Deterministic greedy descent** – each iteration moves a safe amount of
     mass from the index with the largest gradient to the one with the smallest.
  5. **Final polish** – a short L‑BFGS run on the exact objective followed by a
     short simulated‑annealing hill‑climb.

All phases keep the current best sequence and the best observed “fast_score”
(2·n·max(conv) for ∑a = 1).  The function returns the best sequence found before
the deadline and stores it globally for warm‑starting future runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) assuming ∑a = 1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "tri",
            "logistic",
            "doubleexp",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        r = rng.uniform(0.96, 0.999)
        a = (1 - r) * r ** np.arange(n)
    elif mode == "power":
        p = rng.uniform(0.2, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.005, 0.025)
        m_edge = max(1, int(edge_frac * n))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.015 * n), int(0.06 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.01 * n), int(0.1 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "tri":
        # simple linear triangle (peak in the centre)
        a = np.concatenate([np.arange(1, n // 2 + 1), np.arange(n // 2, 0, -1)])[:n]
    elif mode == "logistic":
        x = np.linspace(-6, 6, n)
        a = 1.0 / (1.0 + np.exp(x))
    elif mode == "doubleexp":
        r1 = rng.uniform(0.96, 0.999)
        r2 = rng.uniform(0.96, 0.999)
        left = (1 - r1) * r1 ** np.arange(n)
        right = (1 - r2) * r2 ** np.arange(n)[::-1]
        a = left + right
    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """Run a cascade of L‑BFGS optimisations for the lam‑schedule."""
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Phase 2 – Exponentiated‑gradient descent (sub‑gradient)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 30.0,
):
    """Exponentiated‑gradient descent with AdaGrad‑style stepsizes."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    grad_sq_acc = np.zeros_like(a)  # for AdaGrad
    it = 0

    while time.time() < deadline:
        # convolution and active mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # size n

        # AdaGrad stepsize
        grad_sq_acc += g ** 2
        eta = eta0 / (np.sqrt(grad_sq_acc) + 1e-8)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

# Phase 3 – Deterministic greedy descent with analytical safe step
# ----------------------------------------------------------------------


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """Move mass from the worst‑gradient index to the best‑gradient index."""
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max‑convolution (same derivation as in phase‑2)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))   # where decreasing a helps most
        dst = int(np.argmin(grad))   # where adding a helps most
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        # source positivity bound
        delta_src = a[src]

        # bound from entries that would increase
        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.92  # safety factor
        if delta <= 0.0:
            break

        # apply move
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to avoid drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – Simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted with Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # start temperature: a small fraction of the best score
    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.12   # maximal fraction of src mass we may move
    recompute_every = 2500
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good balance between expressivity
    # and runtime.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previously found construction if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase0_end = start_time + 0.05 * budget_s   # random families
    phase1_end = start_time + 0.25 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.60 * budget_s   # exp‑gradient
    phase3_end = start_time + 0.85 * budget_s   # deterministic descent
    # remaining → simulated annealing + final polish

    # --------------------------------------------------------------
    # Phase 0 – quick exploration of parametric families
    # --------------------------------------------------------------
    while time.time() < phase0_end:
        a = random_initial(n, rng)
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[rand‑fam] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        0.00005,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        if rng.random() < 0.3:
            a_start = add_noise(best_a, rng.uniform(0.0005, 0.01), rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam, phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (AdaGrad)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=30.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic greedy descent
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_descent(
            best_a, phase3_end, n, fft_len
        )
        print(f"[det‑desc] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing + final L‑BFGS polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        # simulated annealing
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

        # final short L‑BFGS on a very smooth surrogate (helps polishing)
        final_a, final_score = lbfgs_continuation(
            best_a,
            [0.00002, 0.00001, 0.000005],
            400,
            deadline,
            fft_len,
            n,
        )
        if final_score < best_score:
            best_score = final_score
            best_a = final_a.copy()
            print(f"[L‑BFGS‑final] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq

    return final_seq
