"""
Propose a step‑function (non‑negative vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

within the 1000 s budget.

Strategy
--------
1. Choose a moderately large problem size (n≈6000).  Larger n gives a
   tighter theoretical lower bound (≈1.0) while still keeping FFTs cheap.
2. Initialise the search from several diverse candidates:
   * a pure “arcsine” shape (U‑shaped, a_i ∝ 1/√[(i+½)(n‑½‑i)]) – known to
     flatten the self‑convolution,
   * uniform, exponential, power‑law, edge‑heavy, and a random simplex draw.
3. **Phase 1 – heavy L‑BFGS** on a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The smoothing parameter λ is decreased gradually;
   each L‑BFGS run starts from a fresh random start (sometimes the arcsine
   shape) and is limited to a modest number of iterations per λ.
4. **Phase 2 – projected Adam** using the exact sub‑gradient of the true
   max‑convolution.  Adam’s adaptive step‑size together with a simplex
   projection drives the iterate towards a local optimum.
5. **Phase 3 – simulated‑annealing hill‑climb**.  Mass is transferred from
   indices with large sub‑gradient to those with small sub‑gradient; a
   simple Metropolis rule allows occasional uphill moves.
6. **Phase 4 – fine L‑BFGS** with very small λ values to polish the final
   vector.
7. The best vector found is finally passed to the external
   ``evaluate_sequence`` routine; it is also stored globally as
   ``height_sequence_1`` for warm‑starts in later runs.

All helper functions are defined at module level; no closures or file I/O
are used.  The code respects the 1000 s limit by budgeting each phase
against the wall‑clock.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """2·n·max(conv(a, a)) assuming a is already normalised (sum = 1)."""
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_shape(n: int) -> np.ndarray:
    """U‑shaped density a_i ∝ 1/√[(i+½)(n‑½‑i)] (discrete arcsine)."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-14
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    w = np.maximum(w, eps)
    w /= w.sum()
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random point on the simplex using a variety of patterns."""
    mode = rng.choice(
        ["uniform", "exp", "power", "edge", "bimodal", "arcsine", "random"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "arcsine":
        a = arcsine_shape(n)
    else:  # random simplex point via exponentials
        a = rng.exponential(1.0, size=n)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(conv) where a = softmax(y).
    The max is approximated by log‑sum‑exp with a temperature λ that
    depends on the current maximal convolution entry.
    """
    # ----- softmax (stable) -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    fval = float(2.0 * n * smooth_max)

    # ----- gradient of smoothed max w.r.t. conv entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_k

    # ----- back‑prop through the convolution -----
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient for each a_i (size n) lives in the central window of conv_rev
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through the softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return fval, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run a short L‑BFGS optimisation for each λ in lam_schedule.
    The optimisation is warm‑started from the previous stage.
    """
    y = np.log(a_start + 1e-12)  # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Compute (max_conv, subgradient) for the exact max‑convolution a*a.
    The sub‑gradient corresponds to the sum of gradients of all
    entries that attain the maximum (within a tiny tolerance).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient = 2 * (a_rev * mask)  (convolution)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g

# Projected Adam on the exact (nonsmooth) objective
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected Adam using the exact sub‑gradient of max(conv)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule: slowly decaying
        step = 1.0 / (np.sqrt(it) + 1.0)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it:,} best={best_score:.6f}")

        # occasional random jitter to escape plateaus
        if it % 4000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb phase
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb with a classic Metropolis acceptance rule.
    Transfers a fraction of mass from high‑gradient indices to low‑gradient
    ones and occasionally swaps two positions.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9996
    min_temp = 1e-9
    it = 0

    while time.time() < deadline and it < max_iters:
        # exact max‑convolution and sub‑gradient for the current vector
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # high‑gradient / low‑gradient index pools
        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i != j:
                frac = rng.uniform(0.04, 0.18)   # transfer 4‑18 % of source mass
                delta = cur_a[i] * frac
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            # simple random swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # re‑normalize (project onto simplex)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it:,} best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser that returns a list of non‑negative floats
    (sum = 1) minimising the external ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ---- problem size --------------------------------------------------
    n = 6000  # a compromise between flexibility and speed

    # ---- initial vector ------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate old solution to the new size
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial score={best_score:.6f}")

    # ---- time budgeting ------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2  # tiny safety margin

    phase1_end = start + 0.38 * (budget_s - 0.2)   # heavy L‑BFGS
    phase2_end = start + 0.62 * (budget_s - 0.2)   # Adam
    phase3_end = start + 0.90 * (budget_s - 0.2)   # annealing
    # remaining time → fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS continuation (many random restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.15, 0.09, 0.06,
        0.04, 0.025, 0.015, 0.009, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 5e-05, 2e-05, 1e-05,
        5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
    ]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        # 30% chance to start from the dedicated arcsine shape
        if rng.random() < 0.30:
            a_start = arcsine_shape(n)
        else:
            a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS heavy] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the exact (nonsmooth) objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing / aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iters=500_000,
        )
        print(f"[Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine‑tuned L‑BFGS with very small λ values
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08, 1e-08]
        fine_maxiter = 800
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (external)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally
    globals()["height_sequence_1"] = final_seq

    return final_seq
