    candidates.append(random_initial(n, rng))
    candidates.append(spectral_init(n))
    candidates.append(np.exp(-np.arange(n)/ (n/10)))  # exponential decay
    # Normalize all candidates

    best_a = None
    best_score = float('inf')

        score = fast_score(a, L, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # Example schedule for step size
    eta0 = 0.5
    iteration = 0
    # We'll keep current a = best_a (copy)
    a = best_a.copy()
    # store best found
    best_a_overall = a.copy()
    best_score_overall = best_score

    while time.time() < deadline:
        # maybe random restart if stuck
        if iteration % 2000 == 0 and iteration > 0:
            # decide if restart
            if rng.random() < 0.3:
                # perturb current best
                a = add_noise(best_a_overall, sigma=rng.uniform(0.001,0.01), rng=rng)
                # ensure normalized
                a /= a.sum()
        # compute conv and gradient
        fa = np.fft.fft(a, L)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)
        M = conv[max_idx]

        diff = np.full_like(a, -M)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)

            idx = np.arange(i_start, i_end + 1)
            diff[idx] = a[max_idx - idx] - M

        else:
            a = a / s
        # evaluate score
        score = 2.0 * n * M / (s * s)   # since sum is s should be 1 approx; compute directly

        fa = np.fft.fft(a, L)
        conv = np.real(np.fft.ifft(fa * fa))[:2 * n - 1]
        max_val = conv.max()
        score = 2.0 * n * max_val / (a.sum() ** 2)
        if score < best_score_overall:
            best_score_overall = score
            best_a_overall = a.copy()
            print(f"[iter {iteration}] new best {best_score_overall:.6f}")

    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluate error:", e)
    return final_seq

"""
Propose a low‑evaluation step‑function using a fast exponentiated‑gradient
descent on the *exact* objective 2·n·max(a*a)/(∑a)².
The algorithm works on the probability simplex (∑a = 1) where the
objective simplifies to 2·n·max(conv).  At each iteration we:
  • compute the full auto‑convolution by FFT,
  • locate the index k where the convolution attains its maximum,
  • form the sub‑gradient of the max‑convolution (g_i = 2·a_{k‑i}),
  • add the denominator term (−2·M) to obtain the gradient of the true
    objective,
  • take an exponentiated‑gradient step a_i ← a_i·exp(−η·gradient_i),
  • renormalise onto the simplex.
The step size η decays as η₀/√t.  Occasional random perturbations
prevent stagnation.  After the time budget a short deterministic hill‑climb
finetunes the result.  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) assuming ∑a = 1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and project back onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb: move a small fraction of mass from the index
    that contributes most to the maximal convolution entry towards the index
    with the smallest partner value.
    """
    step = init_step
    while time.time() < deadline:
        # convolution and the critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: smallest partner value (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta

        # re‑normalise (numerical drift)
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.8, min_step)
    return a, score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep it moderate (4000 proved effective)
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise a pool of diverse starting points
    # ------------------------------------------------------------------
    candidates = []

    # 1) previous best (if available)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:

            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() > 0.0:
                a0 = a0 / a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # 2) simple uniform
    candidates.append(np.full(n, 1.0 / n))

    # 3) random initial

    # 4) spectral start
    candidates.append(spectral_init(n))

    # 5) exponential decay (tuned)
    exp_decay = np.exp(-np.arange(n) / (n / 8.0))
    exp_decay = np.maximum(exp_decay, 0.0)
    exp_decay /= exp_decay.sum()
    candidates.append(exp_decay)

    # Pick the best among the pool via a cheap score
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for a in candidates[1:]:
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] n={n}, initial best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Exponentiated‑gradient descent on the *exact* objective
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2      # safety margin
    iteration = 0
    a = best_a.copy()
    best_a_overall = a.copy()
    best_score_overall = fast_score(a, fft_len, n)

        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))
        M = float(conv[max_idx])                # current maximal convolution value

        # ---- sub‑gradient of the *full* objective ----
        # diff_i = (a_{k-i} - M)   for i inside the support, else -M
        diff = np.full_like(a, -M)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            diff[idx] = a[max_idx - idx] - M

        # gradient of the actual objective is 2·n·(2·a_{k-i}−2·M) = (2·n)·2·diff
        scale = 2.0 * n
        grad = diff * scale               # may be large, we'll clip later

        # ---- exponentiated‑gradient step ----

        exp_arg = -eta * grad
        exp_arg = np.clip(exp_arg, -50, 50)   # protect against overflow
        a = a * np.exp(exp_arg)

        # enforce a tiny lower bound before normalising to avoid exact zeros
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        new_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        if new_score < best_score_overall:
            best_score_overall = new_score
            best_a_overall = a.copy()
            patience = 0
            print(f"[iter {iteration:06d}] new best fast_score={best_score_overall:.6f}")

    # Deterministic fine‑grained hill‑climb
    # ------------------------------------------------------------------

    best_a_overall, best_score_overall = fine_hill_climb(
        best_a_overall,

        n,
