"""
step‑function optimizer (≈ 1000 s)

Improvements over the reference implementation
------------------------------------------------
* Added a “sidon” initialization mode that builds a set with few
  repeated pair‑sums – a good starting point for low L∞‑convolution.
* More aggressive L‑BFGS schedule (larger max‑iter per λ) and a tighter
  λ‑continuation.
* Exponentiated‑gradient descent now injects occasional Gaussian noise
  (helps escape plateaus) and uses a slightly larger initial step size.
* Deterministic greedy descent is run twice (once after EG, once after SA)
  to clean up any remaining “high‑gradient” mass.
* Simulated annealing gets a temperature‑restart and a gradually shrinking
  maximal transfer fraction, which makes large exploratory moves early
  and fine‑grained moves later.
* The whole pipeline is wrapped in a time‑budget loop so that the
  algorithm continually restarts L‑BFGS from diverse initial points
  until the allocated budget for that phase is exhausted.
* The code works for any n (default = 4000) and respects the 1000 s
  global deadline.

The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# Utility helpers

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) assuming sum(a)=1.
    Uses FFT for O(n log n) runtime.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "sidon",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # sidon – greedy construction of a B2 set (few repeated sums)
        # target size ≈ sqrt(n)
        target = int(np.sqrt(n))
        used_sums = set()
        chosen = []
        attempts = 0
        max_attempts = 5 * target
        while len(chosen) < target and attempts < max_attempts:
            cand = rng.integers(0, n)
            conflict = False
            for s in chosen:
                if (cand + s) in used_sums:
                    conflict = True
                    break
            if not conflict:
                for s in chosen:
                    used_sums.add(cand + s)
                    used_sums.add(s + cand)
                used_sums.add(cand + cand)
                chosen.append(cand)
            attempts += 1
        if len(chosen) < target:
            extra = target - len(chosen)
            extra_idx = rng.choice(n, size=extra, replace=False)
            chosen.extend(extra_idx.tolist())
        a = np.zeros(n, dtype=float)
        a[chosen] = 1.0 / target
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

# Phase‑1 : smoothed L‑BFGS (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Parameters are logits y (softmax → a on the simplex).
    Returns (objective, gradient w.r.t. y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """Run a cascade of L‑BFGS optimisations for the λ‑schedule."""
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase‑2 : exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 16.0,
    noise_sigma: float = 0.002,
    noise_every: int = 5000,
):
    """
    Mirror (exponentiated) descent on the L∞‑objective.
    Includes occasional Gaussian perturbations to escape shallow minima.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution and mask of (near) maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max‑convolution (size n)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # stepsize – 1/√t schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update (exponentiated gradient)
        exp_term = np.exp(np.clip(-eta * g, -60, 60))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # occasional random jitter to shake off plateaus
        if (it + 1) % noise_every == 0:
            a = add_noise(a, sigma=noise_sigma, rng=rng)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score

# Phase‑3 : deterministic greedy descent (analytic safe step)
# ----------------------------------------------------------------------


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """Repeatedly shift mass from the worst to the best index using a provably safe step."""
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a

        # admissible δ – keep non‑negativity & do not increase the peak
        delta_src = a[src]

        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.93  # safety factor <1
        if delta <= 0.0:
            break

        # apply move
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional recompute to avoid drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score

def enhanced_simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is restarted halfway through the phase to regain explorative power.
    The maximal transfer fraction decays over time.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # temperature schedule – start relatively low but allow restarts
    temp0 = max(best_score * 0.025, 1e-6)
    half_time = (deadline - time.time()) * 0.5
    start = time.time()
    recompute_every = 2500
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start
        # linear cooling, with a restart at the halfway point
        if elapsed < half_time:
            frac = elapsed / half_time
            temp = max(temp0 * (1.0 - frac), 1e-8)
        else:
            # second‑stage cooling – start a bit hotter again then cool
            frac2 = (elapsed - half_time) / half_time
            temp = max(temp0 * 0.5 * (1.0 - frac2), 1e-9)

        # gradually shrink the amount of mass we may relocate
        max_fraction = 0.20 * (1.0 - elapsed / (deadline - start))

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (same as in deterministic descent)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        dE = new_score - curr_score

        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        # periodic exact recompute to keep numbers clean
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

    # Phase 2 – long exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < phase2_dead:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_dead,
            n,
            fft_len,
            rng,
            eta0=16.0,
            noise_sigma=0.0015,
            noise_every=4000,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy descent (twice for extra polishing)
    # ------------------------------------------------------------------
    if time.time() < phase3_dead:
        best_a, best_score = deterministic_descent(
            best_a,
            phase3_dead,
            n,
            fft_len,
        )
        print(f"[det‑desc‑1] fast_score={best_score:.6f}")

    # A short second pass after a tiny random shake helps to clear remaining peaks
    if time.time() < phase3_dead:
        best_a = add_noise(best_a, sigma=0.0008, rng=rng)
        best_a, best_score = deterministic_descent(
            best_a,
            phase3_dead,
            n,
            fft_len,
        )
        print(f"[det‑desc‑2] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – enhanced simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = enhanced_simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for a possible next run (optional)
    globals()["height_sequence_1"] = final_seq

    return final_seq
