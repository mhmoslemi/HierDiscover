def evolutionary_search(n, deadline, rng):
    pop_size = 30
    # initialize population
    pop = [random_initial(n, rng) for _ in range(pop_size)]
    scores = [fast_score(p, fft_len) for p in pop]
    best_idx = np.argmin(scores)
    best = pop[best_idx].copy()
    best_score = scores[best_idx]
    while time.time() < deadline:
        # select parents via tournament (random)
        i = rng.integers(pop_size)
        j = rng.integers(pop_size)
        parent1 = pop[i] if scores[i] < scores[j] else pop[j]

        i = rng.integers(pop_size)
        j = rng.integers(pop_size)
        parent2 = pop[i] if scores[i] < scores[j] else pop[j]

        # crossover and mutation
        child = (parent1 + parent2) / 2.0
        # add gaussian noise
        sigma = 0.01 # maybe relative
        child = add_noise(child, sigma, rng)
        # optionally apply mutation of a few positions.
        # ensure simplex
        child = simplex_projection(child)

        child_score = fast_score(child, fft_len)
        # replace worst if better
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx]:
            pop[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:

def adam_projected_descent(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    t = 0
    while time.time() < deadline:
        # compute gradient g (subgradient)
        grad = compute_subgradient(a, n, fft_len)
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        # use step size alpha decaying: maybe 0.5
        alpha = 0.5 / (np.sqrt(t))
        # update in log domain? We'll update via a - alpha * m_hat / (np.sqrt(v_hat)+epsilon)
        step = alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        a = a - step
        # Simplex projection
        a = simplex_projection(a)
        # Evaluate
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

```
def compute_subgradient(a, n, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]
    return g
```

"""
Step‑function optimizer – multi‑phase search.

The search proceeds on a probability vector `a` (∑a = 1, a_i ≥ 0) of length `n`.
The evaluation to minimise is  

    score = 2·n·max(conv(a,a))

where `conv` is the linear convolution of `a` with itself.
The optimiser consists of the following phases (time‑controlled):

1. **Evolutionary search** – a small population is evolved by
   tournament selection, averaging and Gaussian mutation, keeping the best
   candidates (fast evaluation via FFT).

2. **Smoothed L‑BFGS** – a few random restarts on a log‑sum‑exp smoothed
   version of the max‑convolution, using a decreasing smoothing schedule.

3. **Exponentiated‑gradient descent** – sub‑gradient descent on the exact
   L∞‑objective with a √‑decaying stepsize (mirrored SGD).

4. **Adam‑projected gradient descent** – an ADAM‑style mirror descent on
   the true sub‑gradient, projected onto the probability simplex.

5. **Deterministic greedy descent** – moves a maximal safe mass from the
   index with largest gradient to the one with smallest gradient (analytic
   step size that never raises the peak).

6. **Simulated‑annealing hill‑climb** – random mass transfers accepted
   with a Metropolis rule while cooling.

All phases keep the global best vector and return it after the budget
expires.  The final sequence is evaluated with the official scorer
(`evaluate_sequence`).  The best vector is stored in the global variable
`height_sequence_1` for possible warm‑starts in later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming a sums to 1."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector of size n."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def compute_subgradient(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """
    Sub‑gradient of 2·n·max(conv(a,a)) w.r.t. a.
    Returns a vector of length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]
    return g


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Works on logits y where a = softmax(y).
    Returns (objective, gradient) with respect to y.
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """Run a cascade of L‑BFGS optimisations for the lam‑schedule."""
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def evolutionary_search(a_start, start_score, n, fft_len, deadline, rng):
    """
    Simple (μ+λ) evolutionary search.
    Population size is modest; the best individual is continuously updated.
    """
    pop_size = 30
    pop = [random_initial(n, rng) for _ in range(pop_size)]
    pop[0] = a_start.copy()                 # ensure we keep the current best
    scores = [fast_score(p, fft_len) for p in pop]

    best_idx = int(np.argmin(scores))
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    while time.time() < deadline:
        # tournament selection for two parents
        i1, i2 = rng.integers(pop_size, size=2)
        parent1 = pop[i1] if scores[i1] < scores[i2] else pop[i2]

        i3, i4 = rng.integers(pop_size, size=2)
        parent2 = pop[i3] if scores[i3] < scores[i4] else pop[i4]

        child = (parent1 + parent2) / 2.0
        sigma = rng.uniform(0.001, 0.02)
        child = add_noise(child, sigma, rng)
        child = simplex_projection(child)

        child_score = fast_score(child, fft_len)

        # replace the worst individual if the child is better
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            pop[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best = child.copy()
    return best, best_score


def phase_lbfgs(a_start, start_score, n, fft_len, deadline, rng):
    """
    Short smoothed L‑BFGS with several random restarts.
    """
    best_a = a_start.copy()
    best_score = start_score
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        0.00005,
        2e-5,
        1e-5,
    ]
    maxiter_per_lam = 600

    while time.time() < deadline:
        # occasionally perturb the current best; otherwise fresh random start
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.01)
            a0 = add_noise(best_a, sigma, rng)
        else:
            a0 = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a0, lam_schedule, maxiter_per_lam, deadline, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
):
    """Exponentiated‑gradient descent (mirrored SGD) on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def adam_projected_descent(a_start, deadline, n, fft_len, rng):
    """
    ADAM‑style projected gradient descent on the true L∞‑objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    while time.time() < deadline:
        g = compute_subgradient(a, n, fft_len)

        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # stepsize decreasing like 1/√t
        alpha = 0.5 / np.sqrt(t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = simplex_projection(a)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def deterministic_descent(a_start: np.ndarray, deadline: float, n: int, fft_len: int):
    """Move mass from the index with largest gradient to the smallest,
    choosing the maximal safe δ that never raises the peak."""
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact full convolution
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))
        dst = int(np.argmin(grad))
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        # admissible δ to keep non‑negativity and not increase the peak
        delta_src = a[src]
        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.92
        if delta <= 0.0:
            break

        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional recompute to avoid drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted with a Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.25  # allow larger moves than the original 0.15
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 offers a reasonable speed/quality trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Warm‑start from a previous construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # allocate portions of the budget (adjustable)
    phase_e_end = start_time + 0.30 * budget_s  # evolutionary search
    phase_lbfgs_end = start_time + 0.45 * budget_s
    phase_exp_end = start_time + 0.70 * budget_s
    phase_adam_end = start_time + 0.90 * budget_s
    # the remaining time will be used by deterministic + annealing

    # ------------------------------------------------------------------
    # Phase E – evolutionary search (fast FFT‑based evaluation)
    # ------------------------------------------------------------------
    best_a, best_score = evolutionary_search(
        best_a, best_score, n, fft_len, phase_e_end, rng
    )
    print(f"[evo] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase B – short smoothed L‑BFGS runs
    # ------------------------------------------------------------------
    best_a, best_score = phase_lbfgs(
        best_a, best_score, n, fft_len, phase_lbfgs_end, rng
    )
    print(f"[lbfgs] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – exponentiated‑gradient descent (sub‑gradient)
    # ------------------------------------------------------------------
    best_a, best_score = exp_gradient_optimize(
        best_a, phase_exp_end, n, fft_len, rng, eta0=12.0
    )
    print(f"[exp‑grad] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – Adam‑style projected gradient descent
    # ------------------------------------------------------------------
    best_a, best_score = adam_projected_descent(
        best_a, phase_adam_end, n, fft_len, rng
    )
    print(f"[adam] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy descent (analytic safe step)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = deterministic_descent(
            best_a, deadline, n, fft_len
        )
        print(f"[det‑desc] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq

    return final_seq
