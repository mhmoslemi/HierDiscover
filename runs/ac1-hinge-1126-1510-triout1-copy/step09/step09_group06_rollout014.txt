def random_initial(n, rng):
    mode = rng.choice(['dirichlet', 'exp', 'uniform'])
    if mode == 'dirichlet':
        alpha = rng.random(n) * 2 + 0.1
        a = rng.gamma(alpha)
        a = a / a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        a = a / a.sum()
    else:
        a = np.full(n, 1.0/n)
    return a
```

def upsample_sequence(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, seq)
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() == 0:
        a_new = np.full(new_n, 1.0/new_n)
    else:
        a_new = a_new / a_new.sum()
    return a_new
```

"""
Propose a low‑score step function.

Algorithm
---------
1. Choose a problem size (default n=8000).  Larger n gives a
   smaller theoretical lower bound 2·n·(1/(2n‑1)) ≈ 1,
   but still runs quickly with FFTs of size ≈ 2·n.

2. Build a diverse pool of initial candidates:
   * up‑sampled previous best (if present)
   * a few random Dirichlet draws
   * exponential / power‑law families
   * a cheap spectral initialiser

3. For each candidate run a short Adam optimisation on the
   smoothed log‑sum‑exp surrogate of the max‑convolution.
   The smoothing factor (lam_factor) is decreased on a schedule
   (0.5 → 1e‑4).  Adam’s adaptive steps and a momentum‑like
   schedule give a stronger descent than the plain sub‑gradient
   method used previously.

4. Keep the best vector found (sum = 1, non‑negative).

5. Use the remaining time for a greedy hill‑climb:
   repeatedly move a random fraction (≤10 %) of the mass from
   one index to another, accepting only strictly improving
   moves.  This simple polishing often pushes the score a bit
   further.

All heavy work is performed with FFTs (O(n·log n)) and the
routine respects the 1000 s global budget by checking the
deadline after every iteration.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a∗a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lam_factors):
    """
    Adam optimiser on the smooth log‑sum‑exp surrogate.
    The smoothing factor (lam_factor) follows the supplied schedule.
    Returns (best_a, best_score).
    """
    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    lam_idx = 0
    iters_per_lam = 2000   # fixed work per smoothing level

    while time.time() < deadline:
        lam_factor = lam_factors[lam_idx]

        # gradient of the smoothed objective
        f, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam_factor)

        it += 1
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple learning‑rate schedule (decreases slowly)
        lr = 0.12 / (np.sqrt(it) + 1.0)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # move to the next smoothing level after a fixed number of steps
        if it % iters_per_lam == 0 and lam_idx < len(lam_factors) - 1:
            lam_idx += 1
        # safety break if we already processed the finest level
        if lam_idx == len(lam_factors) - 1 and it % iters_per_lam == 0:
            # continue at the finest level until time runs out
            pass
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 25000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move at most 10 % of the source mass
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Initialisation families

def spectral_init(n: int) -> np.ndarray:
    """A crude spectral initialiser – inverse FFT of sqrt of uniform spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet‑like)."""
    mode = rng.choice(['dirichlet', 'exp', 'uniform'])
    if mode == 'dirichlet':
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
    else:  # uniform
        a = np.full(n, 1.0 / n)
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def upsample_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a previous solution to a larger size."""
    old_n = old_seq.size
    if old_n == new_n:
        a = old_seq.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n)
        x_new = np.linspace(0.0, 1.0, new_n)
        a = np.interp(x_new, x_old, old_seq)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else np.full(new_n, 1.0 / new_n)

def propose_candidate(seed: int = 42,

    """
    Search for a non‑negative step sequence with a low evaluation score.

    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a lower theoretical bound.
    # ------------------------------------------------------------------
    n = 8000                       # you may increase this a little if time permits
    fft_len = next_pow2(2 * n - 1)  # power‑of‑two for fast FFTs

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Upsampled previous best (if any)

    else:

    # 2️⃣  Spectral and random initialisations
    candidates.append(spectral_init(n))

    # 3️⃣  Parametric families

    # ------------------------------------------------------------------
    # Keep the best seed according to the cheap fast_score
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
