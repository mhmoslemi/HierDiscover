def random_initial(n, rng):
    mode = rng.choice(["uniform", "random", "exp", "power", "edge", "bimodal", "arcsine"])
    if mode == "uniform":
        a = np.full(n, 1.0/n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.0)
        a = (np.arange(1, n+1) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.01, 0.05)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0/n)
        heavy = np.sqrt(n/2.0)/n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = rng.integers(n)
        width = rng.integers(int(0.02*n), int(0.07*n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0/n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # arcsine
        i = np.arange(n)
        x = (i + 0.5) / n
        eps = 1e-12
        a = 1.0 / np.sqrt(np.maximum(x * (1 - x), eps))
        a = np.maximum(a, 0.0)
    a /= a.sum()
    return a
```

"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

The implementation follows a 4‑phase scheme (still ≤ 1000 s):
1.  Many short L‑BFGS runs on a log‑sum‑exp smooth surrogate of the
    max‑convolution.  A diverse set of random starts (uniform, exponential,
    power‑law, edge‑heavy, bimodal, arcsine) is used – the best result is
    kept as a warm‑start.
2.  Projected Adam on the *exact* sub‑gradient of the max‑convolution
    refines the solution.
3.  An aggressive simulated‑annealing / hill‑climb moves mass from
    high‑gradient indices to low‑gradient ones (and swaps).  Temperature
    decays exponentially.
4.  If time permits a short fine‑grained L‑BFGS continuation with tiny
    smoothing parameters finishes the optimisation.

All heavy work (convolution, gradient) is done with FFTs, thus the
algorithm scales well up to n≈10 000.  The final sequence is normalised
(the objective is scale‑invariant) and returned as a plain Python list.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) assuming sum(a) == 1.
    Uses FFT for a fast linear convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = v.astype(np.float64, copy=False)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "arcsine"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.01, 0.05)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = rng.integers(0, n)
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # arcsine
        i = np.arange(n)
        x = (i + 0.5) / n
        eps = 1e-12
        a = 1.0 / np.sqrt(np.maximum(x * (1.0 - x), eps))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate for max‑convolution (log‑sum‑exp) – L‑BFGS helper
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of 2·n·max(conv) where a = softmax(y).
    Returns (value, gradient w.r.t. y).
    """
    # softmax with numerical stability
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (full, length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. each convolution entry
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑prop through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run a short L‑BFGS optimisation for each λ in lam_schedule,
    starting each stage from the result of the previous stage.
    Returns the best vector (normalised to sum = 1) and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # any numerical issue – skip this λ
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and its sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (max_val, subgradient) for the exact max‑convolution of a with itself.
    The sub‑gradient is the sum of the two “active’’ gradients for all shifts
    attaining the maximal value (within a tiny tolerance).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the *exact* sub‑gradient of the max‑convolution.
    Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)  # diminishing step size
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the probability simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # jitter every few thousand iterations to escape plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / hill‑climb phase
# ----------------------------------------------------------------------


def annealing_hill_climb(
    a_start: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb with a simple simulated‑annealing acceptance rule.
    Moves mass from high‑gradient indices to low‑gradient ones and occasionally
    swaps two entries.  Returns the best vector encountered.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    best_a = cur_a.copy()
    best_score = cur_score

    # temperature schedule – starts proportionally to the current score
    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9995
    min_temp = 1e-9

    it = 0
    while time.time() < deadline and it < max_iters:
        # exact max & sub‑gradient for the current vector
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # pick source (high gradient) and target (low gradient) indices
        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i == j:
                continue
            frac = rng.uniform(0.05, 0.20)          # 5‑20 % of source mass
            delta = cur_a[i] * frac
            if delta <= 0.0:
                continue
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # simple swap – helps the algorithm explore permutations
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # feasibility (non‑negative, sum = 1)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        # Metropolis acceptance
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 123, budget_s: int = 1000):
    """
    Multi‑phase optimiser for the step‑function sequence.
    Returns a list of non‑negative floats (sum ≈ 1) that minimises the
    external ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives a smaller theoretical lower bound.
    # ------------------------------------------------------------------
    n = 8000  # 8 k points – still cheap for FFT‑based operations

    # ------------------------------------------------------------------
    # Warm‑start: reuse previous construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting (4 phases)
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2  # safety margin

    phase1_deadline = start + 0.35 * (budget_s - 0.2)   # many L‑BFGS restarts
    phase2_deadline = start + 0.65 * (budget_s - 0.2)   # Adam refinement
    # remaining time → annealing + optional fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS runs with diverse starts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.4, 0.2, 0.1, 0.05, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-04, 2e-04, 1e-04,
        5e-05,
    ]
    maxiter_per_lam = 300

    while time.time() < phase1_deadline:
        # 50 % chance to start from a fresh random vector
        if rng.random() < 0.5:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected Adam on the exact objective
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = subgradient_adam(
            best_a, phase2_deadline, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb + optional fine L‑BFGS
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            total_deadline,
            max_iters=400_000,
        )
        print(f"[Anneal] best_score={best_score:.6f}")

        # If a few seconds are left, run a very fine L‑BFGS continuation
        if time.time() < total_deadline - 2.0:
            fine_lam = [5e-05, 2e-05, 1e-05, 5e-06, 2e-06]
            best_a, fine_score = lbfgs_continuation(
                best_a,
                fine_lam,
                maxiter_per_lam=200,
                deadline=total_deadline,
                fft_len_conv=fft_len_conv,
                fft_len_g=fft_len_g,
                n=n,
            )
            if fine_score < best_score:
                best_score = fine_score
                print(f"[L‑BFGS fine] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (the only call to the protected routine)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally for possible warm‑starts later
    globals()["height_sequence_1"] = final_seq
    return final_seq
