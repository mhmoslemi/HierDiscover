def adam_optimize(y_start, n, fft_len, init_lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8, max_iters, lambda_schedule, deadline):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_a, best_score = None, np.inf
    for lam in lambda_schedule:
        for _ in range(iters_per_lambda):
            if time.time() >= deadline:
                break
            t += 1
            a = softmax(y)
            # compute conv
            fa = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
            max_c = conv.max()
            # smoothed max
            shifted = conv - max_c
            exp_shift = np.exp(shifted / lam)
            Z = exp_shift.sum()
            smooth_max = max_c + lam * np.log(Z)
            f = 2 * n * smooth_max

            # compute gradient of smooth max wrt conv
            w = exp_shift / Z
            # gradient wrt a
            a_rev = a[::-1]
            fa_rev = np.fft.rfft(a_rev, fft_len)
            fw = np.fft.rfft(w, fft_len)
            conv_rev = np.fft.irfft(fa_rev * fw, fft_len)
            g_vec = conv_rev[(n - 1):(n - 1 + n)]
            # factor 2?
            g_vec = 2.0 * g_vec
            # gradient wrt logits
            mean_g = np.dot(a, g_vec)
            grad_y = a * (g_vec - mean_g)

            # Adam update
            lr_t = init_lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y**2)
            y -= lr_t * m / (np.sqrt(v) + eps)

            # evaluate exact fast_score
            a_cur = softmax(y)
            score = fast_score(a_cur, fft_len, n)
            if score < best_score:
                best_score = score
                best_a = a_cur.copy()
    return best_a, best_score
```

def hill_climb(a, score, fft_len, n, deadline, max_total_moves):
    best = a.copy()
    best_score = score
    rng = np.random.default_rng()
    while time.time() < deadline and max_total_moves > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        # choose delta = min(best[i], best[i] * 0.05 * rng.random()) may give small moves.
        # but we could try multiple delta: e.g., random fraction up to 0.2 of source mass.
        delta = best[i] * rng.random() * 0.2
        if delta <= 0: continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        max_total_moves -= 1
    return best, best_score
```

"""
Propose a low‑score step function.

Algorithm
---------
1. Build a diverse pool of initial candidates (previous best, spectral,
   random Dirichlet draws, and several parametric families).
2. Keep the best seed and polish it with a short L‑BFGS continuation
   on a smoothed log‑sum‑exp surrogate of the max‑convolution.
3. Perform a long Adam‑style optimising phase on the same surrogate,
   gradually decreasing the smoothing parameter λ.
4. Finish with an aggressive greedy hill‑climb that moves a random
   fraction of mass between two coordinates and keeps improving moves.
All phases respect a global 1000 s budget and the best sequence found
is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed surrogate of max‑convolution and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len: int,
                       lam: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    # smooth max
    shifted = conv_full - max_c
    # avoid overflow
    exp_shift = np.exp(np.clip(shifted / lam, -50, 50))
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient wrt a (size n)
    g_vec = 2.0 * conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Short L‑BFGS polishing over a schedule of λ values.
    Returns the best a found together with its fast_score.
    """
    y = np.log(a_start + 1e-12)           # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Long Adam‑style optimisation on the smoothed max‑convolution.
    The smoothing λ is geometrically decayed from λ₀ to λₘᵢₙ.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    # schedule parameters
    lam0 = 0.5
    lam_min = 0.001
    target_iters = 20000
    decay_factor = (lam_min / lam0) ** (1.0 / target_iters)
    lam = lam0

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.13                      # base learning rate

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()

        # smooth max weights
        shifted = conv - max_c
        exp_shift = np.exp(np.clip(shifted / lam, -50, 50))
        Z = exp_shift.sum()
        w = exp_shift / Z

        # gradient wrt a (2 * rev_a ⋆ w)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        conv_rev = np.fft.irfft(fa_rev * fw, fft_len)
        g_vec = 2.0 * conv_rev[(n - 1):(n - 1 + n)]

        # gradient wrt logits
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        # Adam update on logits
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # exact evaluation
        a_cur = softmax(y)
        sc = fast_score(a_cur, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()

        # decay λ
        lam = max(lam_min, lam * decay_factor)

        if time.time() >= deadline:
            break
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 50000):
    """
    Greedy hill‑climb: relocate a random fraction of mass from i → j,
    only accepting strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i_src, i_dst = rng.integers(0, n, size=2)
        if i_src == i_dst:
            i += 1
            continue
        # move up to 15 % of source mass
        delta = best[i_src] * rng.random() * 0.15
        if delta <= 0.0:
            i += 1
            continue
        best[i_src] -= delta
        best[i_dst] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta
        i += 1
    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random probability vector sampled from several families."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def sweep_parameter_families(n: int, fft_len: int, rng: np.random.Generator,
                            keep: int = 8):
    """Explore a few parametric families and keep the best `keep` candidates."""
    candidates = []

    # Exponential family
    for c in np.logspace(-4, -0.5, num=12):
        a = exponential_shape(n, c)
        candidates.append((fast_score(a, fft_len, n), a))

    # Power‑law family
    for p in np.linspace(0.1, 4.0, num=15):
        a = powerlaw_shape(n, p)
        candidates.append((fast_score(a, fft_len, n), a))

    # Edge‑heavy families (both interior non‑zero and zero)
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            a1 = edge_shape(n, ef, r, interior_zero=False)
            candidates.append((fast_score(a1, fft_len, n), a1))
            a2 = edge_shape(n, ef, r, interior_zero=True)
            candidates.append((fast_score(a2, fft_len, n), a2))

    # Random Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)
        candidates.append((fast_score(a, fft_len, n), a))

    # keep the best few, dedup by rounded score
    candidates.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in candidates:
        h = round(sc, 7)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                         # length of the step function
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two for fast FFTs

    # ------------------------------------------------------------------
    # Gather initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # 2️⃣  Spectral and random initialisations
    candidates.append(spectral_init(n))
    candidates.append(random_initial(n, rng))

    # 3️⃣  Parametric families sweep
    family_best = sweep_parameter_families(n, fft_len, rng, keep=6)
    candidates.extend(family_best)

    # ------------------------------------------------------------------
    # Keep the best seed from the initial pool
    # ------------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin
    phase1_deadline = start_time + 0.15 * budget_s
    phase2_deadline = start_time + 0.80 * budget_s

    # ------------------------------------------------------------------
    # Phase 1 – brief L‑BFGS polishing from a handful of seeds
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 800

    # seed set: best so far + a few random starts
    seeds = [best_a]
    for _ in range(2):
        seeds.append(random_initial(n, rng))

    for a_start in seeds:
        if time.time() >= phase1_deadline:
            break
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phase1_deadline,
                                            fft_len,
                                            n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_deadline,
                                           n,
                                           fft_len,
                                           rng)
        print(f"[Adam] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climb polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=50000)
        print(f"[Hill‑climb] final refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print(f"[final] evaluation failed: {e}")

    return final_seq
