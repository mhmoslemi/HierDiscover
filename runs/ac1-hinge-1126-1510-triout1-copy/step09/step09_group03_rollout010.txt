"""
High‑precision step‑function optimizer.

Goal
----
Minimise

    2·n·max(conv(a,a)) / (sum(a)**2)

for a non‑negative vector a (len = n).  The ratio is scale‑invariant, so the
search keeps the vector normalised (∑a = 1) and only the shape matters.

Method
------
The optimiser runs four complementary phases inside the allotted time
(≈1000 s).  All phases work with the same normalised vector and keep the
currently best solution.

1. **Smoothed L‑BFGS (continuation)**
   Optimise a log‑sum‑exp smooth approximation of the max‑convolution.
   The smoothing parameter λ is decreased in a schedule; each stage is
   solved with L‑BFGS in the soft‑max (logit) parametrisation.
   Multiple random restarts are performed.

2. **Exponentiated‑gradient descent on the true max**
   The sub‑gradient of the hard max is obtained by convolving the mask of
   maximal entries with the reversed vector.  An entropic mirror‑descent
   update (`a ← a·exp(−η·g)`) is applied with a step‑size η that decays as
   1/√t.  Occasionally the current best is perturbed to escape flat
   plateaus.

3. **Deterministic safe mass transfer**
   In each iteration the index with the largest gradient (most harmful)
   donates mass to the index with the smallest gradient.  The maximal
   admissible transfer δ is computed analytically so that no other entry
   of the convolution exceeds the current peak.  This yields a provably
   non‑increasing objective.

4. **Simulated‑annealing fine‑tuning**
   Random source→dest moves are accepted with a Metropolis rule while a
   linear temperature schedule cools the system.  Incremental updates of
   the convolution keep the cost O(n) per move.

A final short L‑BFGS pass with a very small λ polishes the result.
The best vector found is returned as a plain list of Python floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming ∑a = 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector of length n."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:       # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    if noisy.sum() == 0.0:
        return a.copy()
    return noisy / noisy.sum()


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smooth approximation of the max‑convolution.
    Parameters
    ----------
    y : logits (softmax will be applied, thus a lives on the simplex)
    n : length of a
    fft_len : length of zero‑padded FFT
    lam_factor : multiplies the current max to obtain the smoothing λ
    Returns
    -------
    (f, grad_y) where f is the smooth objective and grad_y the gradient w.r.t. y.
    """
    # softmax → a on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a * a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smoothing
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d(smooth_max)/d conv_i
    # back‑propagation through convolution: a_rev * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_sched,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for a decreasing λ‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # log‑space (softmax)
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑2 : exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0,
                          stagnation_patience: int = 15000):
    """
    Mirror‑descent (exponentiated‑gradient) on the true L∞‑objective.
    Restarts with a little noise after a long period without improvement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)

    it = 0
    last_improve = 0

    while time.time() < deadline:
        # --- convolution and mask of maximal entries ---
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # --- sub‑gradient of the max conv ---
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]        # size n

        # --- step size ---
        eta = eta0 / np.sqrt(it + 1.0)

        # --- exponentiated‑gradient update (clip to avoid overflow) ---
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # --- record improvement ---
        sc = fast_score(a, fft_len)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            last_improve = it

        # --- occasional restart if stuck ---
        if it - last_improve > stagnation_patience:
            a = add_noise(best_a, sigma=0.002, rng=rng)
            last_improve = it

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑3 : deterministic greedy descent (analytical safe step)
# ----------------------------------------------------------------------


def deterministic_descent(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int):
    """
    Move mass from the index with the largest gradient to the one with the
    smallest gradient, using the maximal safe transfer δ that never raises
    the peak of the convolution.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)                # exact full convolution
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max‑convolution (same as in phase‑2)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1: n - 1 + n]

        src = int(np.argmax(grad))      # donate from here
        dst = int(np.argmin(grad))      # receive here
        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a

        # bound from source positivity
        delta_src = a[src]

        # bound from other entries that would exceed the current peak
        pos = diff > 0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * 0.96      # a safety factor <1
        if delta <= 0.0:
            break

        # apply move (exact update of the convolution)
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cur_sc = 2.0 * n * conv.max()
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()

        # occasional recompute to avoid drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Phase‑4 : simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator):
    """Random mass transfers accepted with a Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    cur_sc = start_score
    best_a = a.copy()
    best_sc = cur_sc

    # linear temperature schedule
    temp0 = max(best_sc * 0.02, 1e-6)
    t0 = time.time()

    max_fraction = 0.15   # maximal fraction of src mass we may move
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        # cooling factor
        elapsed = time.time() - t0
        frac = elapsed / (deadline - t0)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_sc = 2.0 * n * conv_new.max()

        dE = new_sc - cur_sc
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            cur_sc = new_sc
            accepted += 1
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 6000 provides a good balance between quality and speed
    # ------------------------------------------------------------------
    n = 6000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previously found construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_sc = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.25 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.80 * budget_s   # EG descent
    phase3_end = start_time + 0.95 * budget_s   # deterministic descent
    # remaining → final annealing & polishing

    # ------------------------------------------------------------------
    # Phase 1 – smoothed L‑BFGS (several restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002, 0.00001,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # with modest probability perturb the incumbent, otherwise fresh start
        if rng.random() < 0.3:
            a_start = add_noise(best_a, sigma=rng.uniform(0.001, 0.015), rng=rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=14.0,
            stagnation_patience=12000,
        )
        print(f"[exp‑grad] fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic greedy descent (analytic safe step)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = deterministic_descent(
            best_a,
            phase3_end,
            n,
            fft_len,
        )
        print(f"[det‑desc] fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with a very low‑λ L‑BFGS run (optional)
    # ------------------------------------------------------------------
    final_deadline = time.time() + 15.0   # a few seconds for polishing
    if time.time() < final_deadline:
        try:
            a_final, sc_final = lbfgs_continuation(
                best_a,
                lam_sched=[1e-5, 5e-6, 2e-6, 1e-6],
                maxiter_per_lam=400,
                deadline=final_deadline,
                fft_len=fft_len,
                n=n,
            )
            if sc_final < best_sc:
                best_a, best_sc = a_final, sc_final
                print(f"[final‑L‑BFGS] fast_score={best_sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for possible subsequent runs
    globals()['height_sequence_1'] = final_seq

    return final_seq
