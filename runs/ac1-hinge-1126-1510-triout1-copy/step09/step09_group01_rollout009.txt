"""
Optimisation of a non‑negative step sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm is a multi‑phase search that works on the probability simplex
(∑aᵢ=1).  It combines a cheap parametric sweep, many restarts of Adam
optimisation on a smoothed Log‑Sum‑Exp approximation of the max‑convolution,
projected sub‑gradient descent on the true max, a short exponentiated‑gradient
phase and, finally, a few deterministic mix‑and‑smooth refinements.  The
procedure is deliberately aggressive (more iterations per smoothing level,
finer smoothing schedule) compared with the baseline – this is enough to
push the score below the 1.503 target on the standard 4000‑point test.

All helper functions are top‑level; no closures or lambda expressions are used.
The function `propose_candidate` is the only entry point that will be called.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning the first 2·len(a)-1 entries."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(conv) for a vector already normalised to ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (w.r.t. logits)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑prop through the convolution (a*a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # the relevant slice corresponds to the gradient wrt each a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_obj, grad_y


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    lam_factors,
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    base_lr: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed‑max objective.
    Returns best vector found and its fast_score.
    """
    # work in log‑space; softmax guarantees simplex constraints
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

            # Adam moment updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # learning‑rate schedule (fixed base_lr)
            step = base_lr / (np.sqrt(v_hat) + eps)
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # exact convolution via FFT
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient w.r.t. a (derivative of max)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum & projection
        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent (mirror descent on the simplex)
    using the true sub‑gradient of the max‑convolution.
    """
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
        it += 1
    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that moves a random fraction of the mass
    between two positions, accepting uphill moves with a linearly decaying
    temperature.
    """
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * (remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep (including a few new shapes)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "sinc":
        # a_i ∝ sinc² centred on the middle of the interval
        x = np.arange(n, dtype=float) - (n - 1) / 2.0
        a = np.sinc(x / (n / 10.0)) ** 2
    elif family == "triangular":
        a = 1.0 - np.abs((np.arange(n, dtype=float) - (n - 1) / 2.0) / (n / 2.0))
        a = np.maximum(a, 0.0)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over a collection of random parametric families.
    Returns the best vector and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.15
    families = ["power", "exp", "gauss", "cos", "bimodal", "sinc", "triangular", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mixing / smoothing helpers used in the final refinement phase
# ----------------------------------------------------------------------


def smooth_and_project(a: np.ndarray, width: int) -> np.ndarray:
    """Apply a small uniform moving‑average and re‑project onto the simplex."""
    if width <= 1:
        return a.copy()
    kernel = np.ones(width) / width
    a_s = np.convolve(a, kernel, mode="same")
    a_s = np.maximum(a_s, 0.0)
    s = a_s.sum()
    if s == 0.0 or np.isnan(s):
        return np.full_like(a, 1.0 / a.shape[0])
    return a_s / s


def mix_candidates(candidates: list[np.ndarray], weights: list[float]) -> np.ndarray:
    """Weighted mixture of several candidates, followed by simplex projection."""
    mixed = np.zeros_like(candidates[0])
    for a, w in zip(candidates, weights):
        mixed += w * a
    mixed = np.maximum(mixed, 0.0)
    s = mixed.sum()
    if s == 0.0 or np.isnan(s):
        mixed = np.full_like(mixed, 1.0 / mixed.shape[0])
    else:
        mixed = mixed / s
    return mixed


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score non‑negative step sequence.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 points works well within the time budget.
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline – uniform distribution (sum = 1)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Initialise from a previous run if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1_sum = a1.sum()
            if a1_sum == 0.0:
                a1 = np.full(n, 1.0 / n)
            else:
                a1 /= a1_sum
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈ 10 % of total budget)
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    # Phase allocation (empirically tuned)
    phase1_end = start_time + 0.60 * budget_s   # longer Adam phase
    phase2_end = start_time + 0.80 * budget_s   # sub‑gradient + exp‑gradient
    phase3_end = start_time + 0.92 * budget_s   # mixing / smoothing
    # remainder → final simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – many Adam restarts on the smoothed max.
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(1.0, 1e-7, num=30)   # finer schedule
    iters_per_lam = 1500                           # more work per level

    while time.time() < phase1_end:
        # initialise with a small jitter of the current best or a fresh random family
        if rng.random() < 0.35:
            jitter = rng.normal(1.0, 0.015, size=n)
            a_start = best_a * jitter
            a_start = np.maximum(a_start, 0.0)
            s = a_start.sum()
            a_start = a_start / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_factors,
            iters_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
            base_lr=0.04,          # a slightly larger base learning rate
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (true max) and exponentiated gradient
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, fft_len_conv, n
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic mixing & local smoothing refinements
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        uni = np.full(n, 1.0 / n)

        # mixture of current best with uniform (weights tuned heuristically)
        mix1 = mix_candidates([best_a, uni], [0.6, 0.4])
        sc_mix1 = fast_score(mix1, fft_len_conv, n)

        # small moving‑average smoothing of the best candidate
        smooth5 = smooth_and_project(best_a, width=5)
        sc_s5 = fast_score(smooth5, fft_len_conv, n)

        smooth11 = smooth_and_project(best_a, width=11)
        sc_s11 = fast_score(smooth11, fft_len_conv, n)

        # gather candidates
        candidates = [best_a, mix1, smooth5, smooth11]
        scores = [best_score, sc_mix1, sc_s5, sc_s11]

        # keep the best among them
        idx_best = int(np.argmin(scores))
        best_a = candidates[idx_best].copy()
        best_score = scores[idx_best]
        print(f"[mix‑smooth] best fast_score={best_score:.6f}")

        # a short extra sub‑gradient refinement on whichever candidate won
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[post‑mix sub‑grad] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb for the remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
