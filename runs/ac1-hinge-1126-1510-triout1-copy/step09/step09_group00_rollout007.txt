"""
Propose a low‑score step function.

The search consists of five stages that together respect the global
deadline (default 1000 s):

1. **Rich parametric sweep** – a large catalogue of families (uniform,
   power, exponential, Gaussian, mixtures, triangular, sinusoid,
   sparse binary, random) is sampled.  The best vector is kept as a warm
   start for the later stages.

2. **Continuation L‑BFGS on a log‑sum‑exp smooth max** – a smoothed
   version of  max(a*a) is minimised.  The smoothing parameter λ is
   decreased geometrically; each stage starts from the result of the
   previous one and runs many more LBFGS iterations than the reference
   implementation.

3. **Projected sub‑gradient descent with Nesterov momentum** – the true
   non‑smooth objective is tackled directly.  The step‑size follows a
   1/√t schedule and Nesterov momentum improves convergence.

4. **Exponentiated‑gradient descent (mirror descent on the simplex)** –
   another first‑order method that often escapes shallow basins.

5. **Aggressive simulated‑annealing hill‑climb** – mass is moved
   between three randomly chosen indices; the temperature follows a
   square‑root schedule and uphill moves are accepted probabilistically.
   Several random shifts of the best candidate are also examined.

All stages record the best sequence seen so far (according to the
fast‑score  2·n·max(a*a)  ) and return it in the format required by
``evaluate_sequence``.  The algorithm works on the simplex
`∑a = 1`, which is scale‑invariant for the true objective.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.shape[0])
    return e / s


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    c = np.real(np.fft.ifft(fa * fa))
    return c[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # log‑sum‑exp smooth maximum
    max_c = conv_full.max()
    shift = conv_full - max_c
    exp_shift = np.exp(shift / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # gradient w.r.t. the convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "uniform":
        a = np.ones_like(i)

    elif family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        base = i + 1 if inc else n - i
        a = base ** p

    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)

    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)

    elif family == "triangular":
        # linear ramp up then down
        a = np.minimum(i, n - 1 - i) + 1.0

    elif family == "cosine":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)

    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2

    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))

    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)

    elif family == "mixture_gauss":
        # mixture of 2–4 Gaussians with random weights
        k = rng.integers(2, 5)
        a = np.zeros_like(i)
        weights = rng.random(k)
        weights /= weights.sum()
        for w in weights:
            mu = rng.uniform(0.0, n - 1)
            sigma = rng.uniform(200.0, 1200.0)
            a += w * np.exp(-0.5 * ((i - mu) / sigma) ** 2)

    else:
        # fallback: pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    time_budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Explore a large number of random families and keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    families = [
        "uniform",
        "power",
        "exp",
        "gauss",
        "triangular",
        "cosine",
        "bimodal",
        "logistic",
        "binary_sparse",
        "mixture_gauss",
        "random",
    ]

    deadline = time.time() + time_budget
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_sc:.6f} (family={fam})")
    return best_a, best_sc

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS for each λ in the schedule."""
    y = np.log(a_start + 1e-12)          # logits, stay on the simplex

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_obj_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:               # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)

        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] improved fast_score={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with Nesterov momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # velocity for Nesterov
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true max‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Nesterov update
        v = beta * v + (1.0 - beta) * g
        step = 0.8 / (np.sqrt(it + 1.0))    # slower decay
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a_candidate.copy()
            print(f"[subgrad] new best fast_score={best_sc:.6f}")

        a = a_candidate
        it += 1

    return best_a, best_sc


def exp_gradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent (mirror descent on simplex)."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.5 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[exp‑grad] new best fast_score={best_sc:.6f}")

        it += 1

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Aggressive square‑root cooling SA that transfers mass among three
    indices each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    # initial temperature – a modest fraction of the current score
    T0 = max(0.08 * start_score, 1e-6)
    t0 = time.time()
    total = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total)

        # choose three distinct indices
        i, j, k = rng.choice(n, size=3, replace=False)
        if a[i] <= 0.0:
            continue

        # move a random fraction from i to j and k (split)
        delta = a[i] * rng.random() * 0.25   # ≤25 % of source
        alpha = rng.random()                 # fraction to j
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta * alpha
        a_new[k] += delta * (1.0 - alpha)

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()
        # occasional tiny speed‑up: try a random cyclic shift
        if rng.random() < 0.003:
            shift = rng.integers(1, n)
            a_shifted = np.zeros_like(a)
            a_shifted[shift:] = a[:-shift]
            sc_shift = fast_score(a_shifted, fft_len, n)
            if sc_shift < best_sc:
                best_sc, best_a = sc_shift, a_shifted.copy()
                a = a_shifted
                cur_sc = sc_shift

    return best_a, best_sc

            sc1 = fast_score(a1, fft_len_conv, n)

    # --------------------------------------------------------------
    if time.time() < deadline:
        # try a few random shifts (zero‑pad shift, not circular)
        for _ in range(15):
            offset = rng.integers(1, n)
            a_shift = np.zeros_like(best_a)
            a_shift[offset:] = best_a[:-offset]
            sc_shift = fast_score(a_shift, fft_len_conv, n)
            if sc_shift < best_sc:
                best_sc, best_a = sc_shift, a_shift.copy()
                print(f"[final‑shift] improved fast_score={best_sc:.6f}")

        # final jitter
        jitter = best_a * rng.normal(1.0, 0.008, size=n)
        jitter = np.maximum(jitter, 0.0)
        jitter /= jitter.sum()
        sc_jit = fast_score(jitter, fft_len_conv, n)
        if sc_jit < best_sc:
            best_sc, best_a = sc_jit, jitter.copy()
            print(f"[final‑jitter] improved fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
