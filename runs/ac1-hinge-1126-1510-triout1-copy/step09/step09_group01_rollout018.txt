"""
Improved search for a step sequence that minimizes
2·n·max(conv(a,a)) / (sum(a)²).  The algorithm consists of
a cheap parametric sweep, a smoothed‑max Adam optimisation,
a projected sub‑gradient descent, an exponentiated‑gradient
phase and a final simulated‑annealing refinement.  The
hyper‑parameters (smoothing schedule, learning‑rates,
iteration budgets) are tuned to exploit the whole
1000 s time budget more aggressively than the reference
implementation.
"""

import time
import numpy as np

# Utility functions
def next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT; return length 2·len(a)-1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Returned value 2·n·max(a*a) for a already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Parametric families
# ----------------------------------------------------------------------
def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector sampled from a simple family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "beta":
        # shape parameters α,β are drawn in a range that keeps the vector well‑behaved
        alpha = rng.uniform(-0.5, 5.0)
        beta = rng.uniform(-0.5, 5.0)
        i = np.arange(1, n + 1, dtype=float)
        a = (i ** alpha) * ((n - i + 1) ** beta)
    elif family == "sine":
        a = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    elif family == "linear":
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float)
        else:
            a = np.arange(n, 0, -1, dtype=float)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def parametric_sweep(n: int, fft_len: int, time_budget: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run a cheap random sweep over a collection of families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + time_budget
    families = ["power", "exp", "gauss", "cos", "bimodal", "beta", "sine", "linear", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score

# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) objective and Adam optimiser
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int, lam_factor: float) -> tuple[float, np.ndarray]:
    """Smoothed max based on log‑sum‑exp; returns (objective, gradient w.r.t. y)."""
    a = softmax(y)
    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z    # d smooth_max / d conv_i

    # back‑propagation through convolution (same pattern as sub‑gradient)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f_obj, grad_y

def adam_optimize(a_start: np.ndarray,
                  lam_factors: np.ndarray,

                  n: int) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed‑max objective."""
    y = np.log(a_start + 1e-12)        # logits – guarantees positivity after softmax
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)

                best_score, best_a = sc, a.copy()
    return best_a, best_score

"""
Improved search for a step sequence that minimizes
2·n·max(conv(a,a)) / (sum(a)²).  The algorithm consists of
a cheap parametric sweep, a smoothed‑max Adam optimisation,
a projected sub‑gradient descent, an exponentiated‑gradient
phase and a final simulated‑annealing refinement.  The
hyper‑parameters (smoothing schedule, learning‑rates,
iteration budgets) are tuned to exploit the whole

# Utility functions

def next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT; return length 2·len(a)-1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Returned value 2·n·max(a*a) for a already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector sampled from a simple family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "beta":
        # Beta‑like shape: (i)^{α}·(n‑i+1)^{β}
        alpha = rng.uniform(-0.5, 5.0)
        beta = rng.uniform(-0.5, 5.0)
        i = np.arange(1, n + 1, dtype=float)
        a = (i ** alpha) * ((n - i + 1) ** beta)
    elif family == "sine":
        a = np.sin(np.pi * (np.arange(n, dtype=float) + 0.5) / n)
    elif family == "linear":
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float)
        else:
            a = np.arange(n, 0, -1, dtype=float)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(n: int, fft_len: int, time_budget: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Run a cheap random sweep over a collection of families."""

# Smoothed‑max (log‑sum‑exp) objective and Adam optimiser

def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max based on log‑sum‑exp; returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z       # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f_obj, grad_y


def adam_optimize(a_start: np.ndarray,
                  lam_factors: np.ndarray,
                  iters_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,

                  n: int) -> tuple[np.ndarray, float]:

    y = np.log(a_start + 1e-12)          # logits – guarantees positivity after softmax
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break

            # Adam update

            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            sc = fast_score(a, fft_len_conv, n)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises the
    evaluation metric.  The routine follows a multi‑phase strategy
    (parametric sweep → Adam on a smoothed maximum → true‑max sub‑gradient
    → exponentiated‑gradient → simulated annealing).  Hyper‑parameters have
    been tuned to use the whole 1000 s budget more aggressively than the
    reference implementation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – feel free to change n if you have spare time
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Baseline: uniform distribution (∑a = 1)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0.0, 1.0, hs.size)
            x_new = np.linspace(0.0, 1.0, n)
            a_prev = np.interp(x_new, x_old, hs)
            a_prev = np.maximum(a_prev, 0.0)
            a_prev /= a_prev.sum()
            sc_prev = fast_score(a_prev, fft_len_conv, n)
            if sc_prev < best_score:
                best_a, best_score = a_prev, sc_prev
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈ 10 % of total budget)
    # ------------------------------------------------------------------
    sweep_budget = 0.10 * budget_s
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, sweep_budget, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    phase1_end = start_time + 0.45 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.90 * budget_s
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (log‑sum‑exp)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 2e-6, num=35)   # finer schedule, tighter final smoothing
    iters_per_lam = 400

    while time.time() < phase1_end:
        # initialise either by jittering the current best or from a fresh random family
        if rng.random() < 0.25:
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_factors,
            iters_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum (true max)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (mirror descent)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            fft_len_conv,
            n,
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing hill‑climb (stochastic refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
