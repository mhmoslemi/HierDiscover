"""
Propose a low‑score step function.

Algorithm
---------
1. Build a pool of diverse starting distributions:
   - previous best (if any)
   - spectral / random initialisations
   - many random draws from parametric families (edge‑heavy,
     exponential, power‑law, cosine, polynomial, mixtures).

2. Keep the best few (by the cheap “fast_score” = 2·n·max(conv) with
   sum(a)=1) and run an extensive L‑BFGS polishing on a smoothed
   log‑sum‑exp approximation of the max‑convolution.  The smoothing
   parameter λ is annealed from 0.5 down to 1e‑5 and each stage may
   perform up to 1500 LBFGS iterations.

3. Refine the best shape with a long exponentiated‑gradient descent on
   the exact max‑convolution (sub‑gradient of the ℓ∞‑norm).

4. Finish with a simulated‑annealing hill‑climb that moves a random
   fraction of mass between two coordinates, accepting worsening
   moves with a temperature schedule.

All phases obey the global 1000 s budget and the best sequence found
is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum = 1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(self‑convolution) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage. Returns the best a found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)           # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    # initial step‑size factor (tuned empirically)
    eta0 = 2.0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule (larger early)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def anneal_climb(a: np.ndarray,
                 current_score: float,
                 fft_len: int,
                 n: int,
                 deadline: float,
                 max_move_frac: float = 0.15,
                 init_temp_factor: float = 0.02,
                 cooling_rate: float = 0.9995):
    """
    Simulated‑annealing hill climbing.
    Randomly transfers a fraction of mass from one index to another.
    Accepts worsening moves with probability exp(‑Δ/temperature).
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    # temperature proportional to current score
    T = max(best_score * init_temp_factor, 1e-6)

    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most max_move_frac of source mass
        max_delta = best[i] * max_move_frac
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        # apply move
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        diff = new_score - best_score
        if diff < 0 or rng.random() < np.exp(-diff / max(T, 1e-12)):
            # accept
            if new_score < best_score:
                best_score = new_score
        else:
            # reject – revert
            best[i] += delta
            best[j] -= delta

        # temperature decay
        T *= cooling_rate
        it += 1

    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def cosine_shape(n: int, power: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n) ^ power (symmetric)."""
    x = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.power(x, power)
    a[a < 0] = 0.0
    return a / a.sum()


def polynomial_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^α · (n-i)^β (both ends can be tuned)."""
    i = np.arange(n, dtype=float) + 1.0
    rev = n - np.arange(n, dtype=float)
    a = np.power(i, alpha) * np.power(rev, beta)
    a[a < 0] = 0.0
    return a / a.sum()


def mix_two_shapes(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random convex combination of two random parametric families."""
    # pick two families at random
    families = [
        lambda: exponential_shape(n, rng.uniform(1e-4, 1e-0)),
        lambda: powerlaw_shape(n, rng.uniform(0.1, 4.0)),
        lambda: edge_shape(n, rng.uniform(0.001, 0.05), rng.uniform(2, 100), interior_zero=False),
        lambda: cosine_shape(n, rng.uniform(0.5, 5.0)),
        lambda: polynomial_shape(n, rng.uniform(-2, 2), rng.uniform(-2, 2)),
    ]
    f1, f2 = rng.choice(families, size=2, replace=False)
    a1 = f1()
    a2 = f2()
    w = rng.random()
    a = w * a1 + (1 - w) * a2
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)

            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent (long)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = anneal_climb(best_a,
                                          best_score,
                                          fft_len,
                                          n,
                                          deadline,
                                          max_move_frac=0.12,
                                          init_temp_factor=0.02,
                                          cooling_rate=0.9994)
        print(f"[anneal] final refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
