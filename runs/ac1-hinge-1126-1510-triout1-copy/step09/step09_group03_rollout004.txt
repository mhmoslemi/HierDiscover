"""
step‑function optimizer – aggressive continuation + exact greedy descent

The optimiser works inside the given time budget (default 1000 s) in four
stages:

1.  **Diverse initialisation** – a large pool of random probability vectors
    (uniform, power‑law, beta‑distributions, logistic, etc.) is sampled and
    the best one (lowest fast_score) is kept.

2.  **Continuation L‑BFGS on a log‑sum‑exp smooth max** – a cascade of
    L‑BFGS runs with decreasing smoothing parameter `lam`.  Each run starts
    from the current best vector (with a little Gaussian noise) or from a
    fresh random start.  The schedule goes down to 5·10⁻⁶, which allows the
    optimiser to approach the true non‑smooth objective.

3.  **Mirror (exponentiated) gradient descent** – a sub‑gradient step on the
    true `max(conv)` with a step size ηₜ = η₀ / √t.  The update is performed
    in the simplex using the exponentiated‑gradient rule.

4.  **Exact greedy mass transfer** – each iteration moves the maximal
    possible amount of probability mass from the index with the largest
    sub‑gradient (most harmful for the peak) to the index with the smallest
    sub‑gradient, using the analytically exact safe step size that never
    raises the current peak.  No safety‑factor is needed; a tiny 1‑e‑12
    factor guards against rounding.

The best vector found is finally evaluated with the provided
`evaluate_sequence` function, stored in the global variable
`height_sequence_1` (so that a later run can start from it) and returned
as a plain Python list.

The whole routine respects the 1000 s deadline; after that it returns the
best sequence found so far.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the objective when sum(a) == 1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "beta",
            "logistic",
            "sine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "beta":
        # Beta(α,β) on [0,1]; discretise
        alpha = rng.uniform(0.2, 4.0)
        beta = rng.uniform(0.2, 4.0)
        x = np.linspace(0, 1, n)
        a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
        a = np.maximum(a, 1e-16)
    elif mode == "logistic":
        k = rng.uniform(0.005, 0.05)  # steepness
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + np.exp(k * (x - 0.5)))
    elif mode == "sine":
        x = np.linspace(0, np.pi, n)
        a = np.sin(x) ** 2
    else:
        a = np.full(n, 1.0 / n)

    a += 1e-12                     # avoid exact zeros that may lock the optimiser
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Input:
        y   – logits (softmax parametrisation)
    Returns:
        f       – smoothed objective value
        grad_y  – gradient w.r.t. y
    """
    # softmax → a (on simplex)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z               # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations for the lam‑schedule.
    The routine returns the best vector found during this phase.
    """
    y = np.log(a_start + 1e-12)  # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 14.0,
):
    """
    Mirror SGD (exponentiated‑gradient) on the true L∞‑objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of hard max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact greedy descent – move as much mass as possible without raising the peak
# ----------------------------------------------------------------------


def deterministic_descent(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
):
    """
    Exact greedy descent: each iteration transfers the maximum safe amount
    δ from the index with the largest sub‑gradient to the index with the
    smallest sub‑gradient (the direction that reduces the peak).  The safe
    step size is computed analytically, so no safety factor is required.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact full convolution (length 2n‑1)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n - 1 : n - 1 + n]

        src = int(np.argmax(grad))   # mass to remove (largest sub‑gradient)
        dst = int(np.argmin(grad))   # mass to add (most beneficial)

        if src == dst or a[src] <= 0.0:
            break

        # incremental effect of moving δ from src to dst:
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        # 1) source positivity bound
        delta_src = a[src]

        # 2) bound from entries that would increase the peak
        pos = diff > 0.0
        if np.any(pos):
            bound_vals = (max_val - conv[pos]) / (2.0 * diff[pos])
            delta_bound = np.min(bound_vals)
        else:
            delta_bound = np.inf

        delta = min(delta_src, delta_bound) * (1.0 - 1e-12)  # exact safe step
        if delta <= 0.0:
            break

        # apply move (conv changes linearly with δ)
        conv = conv + 2.0 * delta * diff
        a[src] -= delta
        a[dst] += delta

        cand_score = 2.0 * n * conv.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a.copy()

        # occasional exact recompute to avoid drift
        if np.random.random() < 0.001:
            conv = np.convolve(a, a)

    return best_a, best_score

def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted with a Metropolis rule while cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.15
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-8)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        elif rng.random() < np.exp(-dE / temp):
            accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score
