"""
Propose a step‑function sequence that (approximately) minimizes

       2·n·max(a*a) / (sum(a)**2)

under the constraints of ``evaluate_sequence``.
The algorithm mixes several heuristics:

1. Cheap parametric sweep (10 % of the budget).
2. Random‑restart L‑BFGS on a log‑sum‑exp smooth approximation of the max‑convolution.
3. Projected sub‑gradient descent with momentum.
4. Exponentiated‑gradient descent.
5. A tiny “LP‑refinement” step: solve a linear programme
   max ∑g_j  subject to  a∗g ≤ M  and  g ≥ 0,
   then mix the solution back into the current iterate.
6. Slow simulated‑annealing hill‑climb for the remaining budget.

All phases respect the global deadline (default 1000 s).  The best
candidate seen at any moment is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")
    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum on the true max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g

        step = 0.6 / (it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the simplex using the true sub‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.5 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[exp‑grad] new best fast_score={best_score:.6f}")

        it += 1
    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Very simple stochastic hill‑climb with square‑root cooling."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.05 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "sinusoid":
        freq = rng.integers(1, 10)
        a = np.maximum(np.sin(2 * np.pi * freq * i / n), 0.0) + 0.01
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Cheap random sweep over a collection of families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10          # 10 % of total budget

    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "sinusoid",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")
    return best_a, best_score


# ----------------------------------------------------------------------
# LP‑based refinement step
# ----------------------------------------------------------------------


def lp_refinement_step(a: np.ndarray, deadline: float, rng: np.random.Generator) -> np.ndarray:
    """
    Solve  max ∑g_j   s.t.   a ∗ g ≤ M,   g ≥ 0,
    then mix the solution back into `a`.
    Returns a new vector (still on the simplex) or the original `a` if
    the LP fails or gives no improvement.
    """
    n = a.shape[0]
    # current max convolution
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    M = conv.max()

    # Build convolution matrix A (shape (2n-1, n))
    # Row k = sum_i a_i * g_{k-i}  ↔   diagonal offset i = a_i
    offsets = np.arange(n)
    diagonals = [np.full(n, a[i]) for i in range(n)]
    A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr')

    # Linear programme: maximise sum(g)  ⇔ minimise -sum(g)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    # Time left for the LP (a few seconds at most)
    lp_time = max(0.1, (deadline - time.time()) * 0.9)
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": lp_time},
        )
    except Exception as e:  # pragma: no cover
        print("[lp] linprog exception:", e)
        return a

    if not res.success or res.x is None:
        # LP failed – just return the input
        return a

    g = np.maximum(res.x, 0.0)
    s = g.sum()
    if s <= 0.0 or np.isnan(s):
        return a
    g_norm = g / s

    # Mix a and g_norm; the mixing coefficient is random but stays away
    # from 0 and 1 so that the update is neither too timid nor too aggressive.
    t = 0.3 + 0.4 * rng.random()
    a_new = (1.0 - t) * a + t * g_norm
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Re‑use a previous construction if one is present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] parametric sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2     # safety margin

    # Allocate time slices (tuned for the 1000 s budget)
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.85 * budget_s   # exponentiated‑gradient
    phase4_end = start_time + 0.95 * budget_s   # LP refinement
    # remaining → simulated annealing

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS with a finer lambda schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # occasional jitter of the current best
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] sub‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase3] exp‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – LP‑based refinement (a few heavy steps)
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        lp_iters = 0
        while time.time() < phase4_end and lp_iters < 5:
            a_candidate = lp_refinement_step(best_a, phase4_end, rng)
            sc_candidate = fast_score(a_candidate, fft_len_conv, n)
            if sc_candidate < best_score:
                best_score, best_a = sc_candidate, a_candidate.copy()
                print(f"[lp] improved fast_score={best_score:.6f}")
            else:
                # if no improvement, break early
                break
            lp_iters += 1

    # --------------------------------------------------------------
    # Phase 5 – long simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[phase5] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
