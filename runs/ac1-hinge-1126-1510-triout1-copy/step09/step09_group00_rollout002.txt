- Clip to [0,1000] and non-negative.

- sum_a = np.sum(sequence)

- Return 2 * n * max_b / (sum_a**2).

"""
Propose a high‑quality step sequence for ``evaluate_sequence``.

The algorithm consists of four successive phases (all share a global 1000 s
budget):

1. **Parametric sweep** – generate many random vectors from a large
   collection of simple families (power, exponential, Gaussian, cosine,
   triangular, Hamming, Kaiser, binary‑sparse, etc.).  The best candidate
   from this cheap search seeds the later phases.

2. **Smoothed‑max Adam optimisation** – minimise a smooth approximation
   of the max‑convolution using Adam on the log‑softmax (logits) variables.
   A decreasing schedule of smoothing parameters (the ``lam`` factor) is
   used, ending at a very small value so the final objective is almost the
   true max.

3. **Projected sub‑gradient descent with momentum** – true max‑convolution
   sub‑gradients are used to further improve the solution on the simplex.

4. **Logit‑space simulated annealing** – perturb the logits by a Gaussian
   noise whose variance is slowly cooled; uphill moves are occasionally
   accepted.  This stochastic hill‑climb can escape shallow local minima.

The best vector seen at any moment is returned and finally evaluated with
the official ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = np.sum(e)
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    2·n·max(a*a) for a vector already normalised (∑a = 1).
    Fast (FFT‑based) version used during the search.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# --------------------------------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # ------- convolution a*a -------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing parameter
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ------- gradient w.r.t. convolution -------
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ------- back‑prop through softmax -------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# --------------------------------------------------------------
# Optimisation primitives
# --------------------------------------------------------------

def adam_optimize(
    y0: np.ndarray,
    lam_schedule,
    max_iter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the logits y for the smoothed max objective.
    Returns the best probability vector (softmax(y)) seen and its fast_score.
    """
    y = y0.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        for _ in range(max_iter_per_lam):
            if time.time() >= deadline:
                break
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)

            # learning rate schedule – larger for early (high lam) stages
            alpha = 0.05 * (lam + 1e-3) ** 0.5

            y -= alpha * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                # occasional log
                print(f"[adam] lam={lam:.2e} new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true max‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 0.6 / (it + 1.0)

        # simple projection to the simplex (still sums to 1)
        a_candidate = a - step * v
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s > 0.0:
            a_candidate /= s
        else:
            a_candidate = np.full_like(a, 1.0 / n)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[subgrad] iteration {it} new best fast_score={best_score:.6f}")

        a = a_candidate
        it += 1

    return best_a, best_score


def logit_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing in the logit space.
    A Gaussian perturbation is added to the logits; the temperature
    controls the perturbation magnitude and the Metropolis acceptance.
    """
    # initialise logits from the current best probability vector
    y = np.log(a_start + 1e-12)
    best_y = y.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.05 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # square‑root cooling – slower than linear
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        # perturb logits – variance proportional to temperature
        sigma = 0.2 * (T / T0 + 1e-3)
        y_new = y + rng.normal(scale=sigma, size=n)

        a_new = softmax(y_new)
        new_score = fast_score(a_new, fft_len_conv, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            y = y_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_y = y_new.copy()

    best_a = softmax(best_y)
    return best_a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector sampled from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p

    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)

    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)

    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)

    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2

    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))

    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)

    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))

    elif family == "linear_inc":
        a = i + 1.0
        a = a ** rng.uniform(0.2, 2.0)          # optional power shape

    elif family == "linear_dec":
        a = (n - i)
        a = a ** rng.uniform(0.2, 2.0)

    elif family == "triangular":
        a = np.minimum(i + 1, n - i)
        a = a.astype(float)

    elif family == "trapezoid":
        w = rng.uniform(0.1, 0.4) * n
        left = int((n - w) // 2)
        right = left + int(w)
        a = np.zeros(n, dtype=float)
        a[left:right] = 1.0
        a = np.cumsum(a)          # optional smoothing later – keep simple

    elif family == "hamming":
        a = np.hamming(n)

    elif family == "kaiser":
        beta = rng.uniform(0.0, 14.0)
        a = np.kaiser(n, beta)

    else:       # pure random (Dirichlet‑like)
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap random search over many simple families.
    Returns the best vector found and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10   # 10 % of total budget

    families = [
        "power", "exp", "gauss", "cos", "bimodal", "logistic",
        "binary_sparse", "double_exp", "linear_inc", "linear_dec",
        "triangular", "hamming", "kaiser", "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (can be tweaked – larger n may give a slightly lower bound)
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)    # length for a*a
    fft_len_g = next_pow2(3 * n - 2)       # length for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a previous construction is present, reuse it (interpolated)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] parametric sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # allocate time slices (tuned for this version)
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = start_time + 0.95 * budget_s
    # remaining time → final annealing

    # --------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (many lambda stages)
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06, 5e-07,
    ]
    maxiter_per_lam = 800   # a few hundred iterations per lambda

    while time.time() < phase1_end:
        # jitter the current best to stay in its basin (5 % Gaussian noise)
        y0 = np.log(best_a + 1e-12)
        jitter = rng.normal(scale=0.02, size=n)
        y0 += jitter

        a_opt, sc_opt = adam_optimize(
            y0,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] sub‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – logit‑space simulated annealing (slow cooling)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = logit_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] logit‑annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official ``evaluate_sequence``
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
