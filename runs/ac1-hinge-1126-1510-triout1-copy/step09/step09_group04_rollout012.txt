def adam_optimize(y0, n, fft_len_conv, fft_len_g, lam_start, lam_end, deadline):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0
    best_a = softmax(y)
    best_score = fast_score(best_a,...)
    while time.time()<deadline:
        t+=1
        lam = max(lam_end, lam_start * (lam_decay**t))
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        y = y - lr * m_hat / (np.sqrt(v_hat)+eps)
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score
```

"""
Propose a low‑score step‑function.

Algorithm outline (≈1000 s total):
  1.  Generate a diverse set of candidate vectors on the probability
      simplex using a few parametric families (uniform, random,
      edge‑power, edge‑exponential, blended edge/uniform).
      Keep the best few according to the fast 2·n·max(a*a) score.
  2.  Refine the best candidate with a custom Adam optimiser applied
      to the smooth‑max surrogate (log‑sum‑exp) of the self‑convolution.
      The smoothing factor (lam_factor) is decreased continuously,
      i.e. a continuation method.
  3.  Perform exact exponentiated‑gradient descent on the true max‑convolution
      (sub‑gradient of the hard max) for a medium‑length phase.
  4.  Deterministic fine‑grained hill‑climb that repeatedly moves a tiny
      fraction of mass from the “most harmful’’ coordinate to the
      “least harmful’’ one.
  5.  A short simulated‑annealing pass to escape any remaining local
      minima.
  6.  Return the best vector found; it is automatically clipped to
      non‑negative values by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators (all vectors are normalised to sum‑1)
# ----------------------------------------------------------------------


def uniform_init(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a /= a.sum()
    return a


def edge_power_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Heavy mass near the two edges: a_i ∝ (i+1)^‑p + (n‑i)^‑p."""
    p = rng.uniform(0.1, 3.0)
    i = np.arange(n, dtype=float)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def edge_exp_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Edge‑exponential decay: a_i ∝ exp(-rate·i) + exp(-rate·(n‑1‑i))."""
    rate = rng.uniform(0.001, 0.05)
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a


def blended_edge_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Blend uniform mass with an edge‑heavy component."""
    eps = rng.uniform(0.2, 0.8)           # weight of edge component
    base = uniform_init(n)
    edge = edge_power_init(n, rng)
    a = (1.0 - eps) * base + eps * edge
    a /= a.sum()
    return a


def generate_initial_pool(n: int, rng: np.random.Generator, pool_size: int = 30):
    """Return a list of diverse initial probability vectors."""
    pool = []
    pool.append(uniform_init(n))
    pool.append(random_init(n, rng))
    for _ in range(pool_size // 5):
        pool.append(edge_power_init(n, rng))
        pool.append(edge_exp_init(n, rng))
        pool.append(blended_edge_init(n, rng))
    # add a few noisy perturbations of the best uniform candidate
    for _ in range(pool_size - len(pool)):
        base = uniform_init(n)
        noisy = base + rng.normal(0.0, 0.01, size=n)
        noisy = np.maximum(noisy, 0.0)
        noisy /= noisy.sum()
        pool.append(noisy)
    return pool


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. y where a = softmax(y).
    """
    a = softmax(y)

    # Full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # Gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # Back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # Extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor_start: float = 0.5,
                  lam_factor_end: float = 0.0005,
                  lr: float = 0.15):
    """
    Adam optimisation of the smooth max surrogate.
    Returns the best vector and its fast_score seen before the deadline.
    """
    y = np.log(a_start + 1e-12)                # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    iteration = 0
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    lam_factor = lam_factor_start
    lam_decay = 0.9995  # geometric decay per iteration

    while time.time() < deadline:
        iteration += 1
        # schedule smoothing factor
        lam_factor = max(lam_factor_end, lam_factor * lam_decay)

        # objective and gradient for current lam_factor
        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate score every few iterations
        if iteration % 25 == 0:
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                # optional: tighten learning rate if improvement appears
                # (kept simple to stay deterministic)

    # final evaluation
    a = softmax(y)
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent via exponentiated‑gradient (hard max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    Keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size (classic 1/√t schedule)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic fine‑grained hill‑climb
# ----------------------------------------------------------------------


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with
    the smallest partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        # convolution and the critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: smallest partner value (avoid picking i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # renormalise (should stay 1 but guard against round‑off)
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep current step size for next iteration
        else:
            step = max(step * 0.8, min_step)
    return a, score

    best_score = cur_score

    T = cur_score * 0.02          # initial temperature (2 % of score)
    decay = 0.99995               # very slow cooling

    while time.time() < deadline:
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of mass from i to j
        eps = cur_a[i] * 0.05 * rng.random()
        if eps <= 0.0:
            continue
        new_a = cur_a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # (sum stays 1) evaluate
        new_score = fast_score(new_a, fft_len, n)
        delta = new_score - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_a = new_a
            cur_score = new_score
            if cur_score < best_score:
                best_a, best_score = cur_a.copy(), cur_score

        T = max(T * decay, 1e-9)   # avoid exact zero

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (tunable)
    # ------------------------------------------------------------------
    n = 4000                         # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 0 – generate a pool of diverse initial vectors
    # ------------------------------------------------------------------
    init_pool = generate_initial_pool(n, rng, pool_size=30)
    best_a = None
    best_score = np.inf
    for cand in init_pool:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
    print(f"[Phase‑0] best initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Allocate proportionally
    phase1_deadline = start_time + 0.45 * budget_s
    phase2_deadline = start_time + 0.70 * budget_s
    phase3_deadline = start_time + 0.85 * budget_s
    phase4_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase1_deadline:
        best_a, best_score = adam_optimize(
            best_a,
            n,
            fft_len_conv,
            fft_len_g,
            phase1_deadline,
            lam_factor_start=0.5,
            lam_factor_end=0.0003,
            lr=0.12,
        )
        print(f"[Phase‑1] Adam best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient descent (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_deadline,
            n,
            fft_len_conv,
            fft_len_g,
        )
        print(f"[Phase‑2] exp‑gradient best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – deterministic hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = fine_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_deadline,
            init_step=0.2,
            min_step=1e-6,
        )
        print(f"[Phase‑3] hill‑climb best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing for final polishing
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_deadline,
        )
        print(f"[Phase‑4] simulated‑annealing best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final validation using the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    return final_seq
