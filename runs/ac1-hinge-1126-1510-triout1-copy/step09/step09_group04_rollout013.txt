"""
Hybrid optimizer for the step‑function evaluation problem.

Algorithm (≈1000 s total)
  1. Initialise a diverse set of probability vectors (uniform,
     spectral, random, and a possible previous solution).
  2. Perform a short L‑BFGS run on a log‑softmax parametrisation
     with a decreasing log‑sum‑exp smoothing schedule – this gives a
     good coarse starting point.
  3. Run a very long exponentiated‑gradient (mirror‑descent) loop on the
     true max‑convolution objective.  Each iteration needs only one
     FFT‑based convolution, so we can afford many thousands of steps.
  4. Periodically inject a little Gaussian noise to escape plateaus.
  5. Finish with a deterministic hill‑climb that repeatedly transfers a
     small fraction of mass from the “most harmful’’ coordinate (largest
     a_i·partner_i) to a “most helpful’’ one (smallest partner_i) while
     shrinking the step size.
  6. Return the best sequence found; a final call to ``evaluate_sequence``
     checks the true score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast compute 2·n·max(a*a) assuming ∑a = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns objective value and gradient w.r.t. logits y (a = softmax(y)).
    """
    a = softmax(y)                     # probability vector

    # Full convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # Gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                    # d smooth_max / d conv_i

    # Back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # Extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Fine deterministic hill‑climb
# ----------------------------------------------------------------------


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  Step size shrinks geometrically until min_step.
    """
    step = init_step
    while time.time() < deadline:
        # convolution and the critical index
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: smallest partner value (avoid picking i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # renormalise (should stay 1 but guard against round‑off)
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep current step size for next iteration
        else:
            # reject – make the move finer
            step = max(step * 0.8, min_step)
    return a, score

    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    # Phase 1 – short L‑BFGS run (coarse)
    # --------------------------------------------------------------
    lam_schedule_coarse = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]
    maxiter_coarse = 1500

    y0 = np.log(best_a + 1e-12)
    for ls in lam_schedule_coarse:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y0,
                           args=(n, fft_len_conv, fft_len_g, ls),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_coarse,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[L‑BFGS] exception:", e)
            continue
        y0 = res.x
        a_tmp = softmax(y0)
        score_tmp = fast_score(a_tmp, fft_len_conv, n)
        if score_tmp < best_score:
            best_score = score_tmp
            best_a = a_tmp.copy()
            print(f"[L‑BFGS] new best after lam={ls:.5f}  score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent (sub‑gradient)
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.rfft(best_a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        # mask of entries attaining (or very close to) the max
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = best_a[::-1]
        fm = np.fft.rfft(mask, fft_len_g)
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # classic 1/√t step size
        eta = 1.0 / np.sqrt(iteration + 1.0)

        # exponentiated update (mirrored descent on the simplex)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a_new = best_a * exp_term
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = best_a.copy()

        score_new = fast_score(a_new, fft_len_conv, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            # occasionally print progress
            if iteration % 1000 == 0:
                print(f"[exp‑grad] it={iteration}   new best score={best_score:.6f}")

        # occasional random perturbation to escape shallow plateaus
        if iteration % 5000 == 0 and iteration > 0:
            cand = add_noise(best_a, sigma=0.01, rng=rng)
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[exp‑grad] it={iteration}   noise‑improved score={best_score:.6f}")

        iteration += 1
        # safety break if we are about to exceed the deadline
        if time.time() + 0.1 >= deadline:
            break

    print(f"[phase‑exp‑grad] finished after {iteration} iterations, best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – fine deterministic hill‑climb
    # --------------------------------------------------------------
    best_a, best_score = fine_hill_climb(best_a, best_score,
                                         fft_len_conv, n, deadline,
                                         init_step=0.3,
                                         min_step=1e-7)
    print(f"[fine‑hill‑climb] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final verification with the provided security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)   # type: ignore
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as ex:
        print("[final] evaluation raised an exception:", ex)
        final_eval = None

    return final_seq
