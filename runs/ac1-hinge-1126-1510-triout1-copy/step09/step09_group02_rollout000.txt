def refined_exp_gradient(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    it = 0
    eta = 10.0

    while time.time() < deadline:
        # compute gradient of max conv (subgradient)
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = 2 * np.real(np.fft.ifft(fa_rev * fm))
        g = grad_conv[n-1 : n-1 + n]   # as earlier

        # try a few candidate etas
        best_local_a = a
        best_local_score = best_score
        for factor in [0.5, 1.0, 2.0]:
            eta_try = eta * factor
            exp_term = np.exp(np.clip(-eta_try * g, -50, 50))
            a_candidate = a * exp_term
            s = a_candidate.sum()
            if s == 0:
                continue
            a_candidate /= s
            score = fast_score(a_candidate, fft_len)
            if score < best_local_score:
                best_local_score = score
                best_local_a = a_candidate

        if best_local_score < best_score - 1e-9:
            a = best_local_a
            best_score = best_local_score
            best_a = a.copy()
            # optionally reduce eta if improvement small
        else:
            # no improvement, maybe reduce eta
            eta *= 0.9

        # gradually decrease eta overall:
        eta = max(0.01, eta * 0.9995)

        it += 1
        # Possibly break if improvement small
        if it % 500 == 0:
            print(f"[refined_exp] it={it} score={best_score:.6f}")

"""
Proposes a step‑function (non‑negative sequence) minimizing

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines several optimisation phases inside a 1000 s budget:

1. **L‑BFGS** on a log‑sum‑exp smoothed max‑convolution, several random
   restarts, decreasing smoothing λ.
2. **Adam** on the same smoothed objective for refined descent.
3. **Refined exponentiated‑gradient** on the true max‑convolution,
   with a tiny line‑search over the learning rate.
4. **Adaptive hill‑climbing** that moves a fraction of mass from a
   high‑mass index to a low‑mass index; the move size shrinks
   progressively and the full convolution is recomputed periodically.

The best sequence found is returned as a plain Python list of floats.
All intermediate steps keep the vector on the probability simplex,
hence the official evaluation reduces to `2·n·max(conv)`.

"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.  The optimisation variable is y (logits); the
    actual distribution is a = softmax(y).
    Returns (objective_value, gradient_wrt_y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a (FFT)
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]       # gradient w.r.t a_i (before softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result. Returns the best (a,score).
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best (a,score) observed.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if steps % 200 == 0:
                    print(f"[Adam] lam={lam:.5f} step={steps} score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

# Refined exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def refined_exp_gradient_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 eta0: float = 12.0,
                                 min_eta: float = 0.001):
    """
    Exponentiated‑gradient descent on the actual max‑convolution.
    At every iteration a tiny line‑search over the learning‑rate
    (0.5×, 1×, 2×) is performed to guarantee monotone improvement.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    eta = eta0
    it = 0
    while time.time() < deadline:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # try three candidate step‑sizes
        improved = False
        best_local_a = a
        best_local_score = best_score

        for factor in (0.5, 1.0, 2.0):
            eta_try = max(min_eta, eta * factor)
            exp_term = np.exp(np.clip(-eta_try * g, -50, 50))
            cand = a * exp_term
            s = cand.sum()
            if s == 0.0:
                continue
            cand /= s
            sc = fast_score(cand, fft_len)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = cand
                improved = True
                # keep the factor giving improvement for next iteration
                eta = eta_try

        if improved:
            a = best_local_a
            best_score = best_local_score
            best_a = a.copy()
        else:
            # no improvement – shrink step size gently
            eta = max(min_eta, eta * 0.95)

        # gradual decay of eta to keep exploration limited
        eta = max(min_eta, eta * 0.9995)

        it += 1
        if it % 500 == 0:
            print(f"[Refined‑Exp‑Grad] it={it} score={best_score:.6f}")

    return best_a, best_score

def hill_climb_adaptive(a_start: np.ndarray,
                        curr_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        start_frac: float = 0.20,
                        end_frac: float = 0.02,
                        recompute_every: int = 2000):
    """
    Greedy hill‑climb that moves a random fraction of mass from a heavy
    index to a lighter one.  The allowed fraction shrinks linearly with
    time, and the full convolution is recomputed every `recompute_every`
    successful moves to curb numerical drift.
    """
    best_a = a_start.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    diff = np.zeros_like(conv)
    accepted = 0
    start_time = time.time()

    while time.time() < deadline:
        # linearly decay the allowed fraction
        progress = (time.time() - start_time) / (deadline - start_time)
        max_frac = start_frac * (1 - progress) + end_frac * progress

        # pick source (biased towards larger mass) and destination
        src = rng.choice(n, p=best_a)               # likely a heavy index
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if best_a[src] < best_a[dst]:
            src, dst = dst, src
        if best_a[src] <= 0.0:
            continue

        delta = best_a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            # recompute from scratch to avoid drift
            conv = np.convolve(best_a, best_a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best sequence found before the deadline as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ---- problem size ---------------------------------------------------
    n = 5000                     # a bit larger than the previous 4000
    fft_len = next_pow2(2 * n - 1)

    # ---- initialise -----------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ---- time budget split ------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.15 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.45 * budget_s   # Adam
    phase3_end = start_time + 0.78 * budget_s   # Refined exp‑grad
    # rest → hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with several restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005
    ]
    maxiter_per_lam = 1000

    while time.time() < phase1_end:
        if rng.random() < 0.25:
            sigma = rng.uniform(0.001, 0.012)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        lam_schedule_adam = [
            0.4, 0.25, 0.15, 0.08, 0.04,
            0.02, 0.01, 0.005, 0.002, 0.001
        ]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam, max_steps_per_lam=2000,
            deadline=phase2_end, n=n, fft_len=fft_len, lr0=0.13,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Refined exponentiated gradient on true max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = refined_exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng,
            eta0=15.0, min_eta=1e-4,
        )
        print(f"[Refined‑Exp‑Grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adaptive hill‑climbing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_adaptive(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            start_frac=0.18,
            end_frac=0.01,
            recompute_every=1500,
        )
        print(f"[Hill‑Climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
