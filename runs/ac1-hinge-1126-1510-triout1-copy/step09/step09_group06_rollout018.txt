def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)

"""Step‑function optimizer for the geometric‐mean convolution bound.

The implementation builds a diverse pool of initial candidates (uniform,
spectral, random Dirichlet, exponential, power‑law, edge‑heavy and a
new *sparse* family).  The best few seeds are polished with a sequence of
L‑BFGS runs on a log‑sum‑exp smoothing of the max‑convolution, then a long
exponentiated‑gradient descent on the exact max, finally a short simulated‑
annealing mass‑transfer phase.  All phases respect a 1000 s budget and
periodically keep the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Objective and fast score

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation for ∑a = 1:
        score = 2·n·max(conv(a,a))

    Parameters
    ----------
    a : np.ndarray
        non‑negative vector with sum = 1.
    fft_len : int
        power‑of‑two length used for the FFT.
    n : int
        length of the step function.

    Returns
    -------
    float
        2·n·max convolution.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum (log‑sum‑exp) of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).

    Parameters
    ----------
    y : np.ndarray
        logits (unconstrained variables).
    n : int
        length of the step function.
    fft_len_conv, fft_len_g : int
        FFT lengths (identical in our use‑case).
    lam_factor : float
        smoothing factor; actual λ = max(conv)·lam_factor.

    Returns
    -------
    float, np.ndarray
        objective value and gradient w.r.t. y.
    """
    a = softmax(y)

    # full self‑convolution via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # Extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Standard random families (uniform, Dirichlet, exp, power, edge, bimodal)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()

    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n)

def weighted_sparse_initial(n: int, rng: np.random.Generator,

    candidates = []

    # exponential family (symmetric)
    for c in np.logspace(-4, -0.5, num=12):
        a = exponential_shape(n, c)
        candidates.append((fast_score(a, fft_len, n), a))

    # power‑law family (symmetric)
    for p in np.linspace(0.1, 4.0, num=15):
        a = powerlaw_shape(n, p)
        candidates.append((fast_score(a, fft_len, n), a))

    # edge‑heavy families
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            a1 = edge_shape(n, ef, r, interior_zero=False)
            candidates.append((fast_score(a1, fft_len, n), a1))
            a2 = edge_shape(n, ef, r, interior_zero=True)
            candidates.append((fast_score(a2, fft_len, n), a2))

    # random Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)
        candidates.append((fast_score(a, fft_len, n), a))

    # sparse families (new)
    for _ in range(20):
        a = random_sparse_initial(n, rng)
        candidates.append((fast_score(a, fft_len, n), a))
    for _ in range(20):
        a = weighted_sparse_initial(n, rng)
        candidates.append((fast_score(a, fft_len, n), a))

    # keep the best distinct candidates
    candidates.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in candidates:
        h = round(sc, 8)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

def lbfgs_refine(a_start: np.ndarray,
                 lam_schedule,
                 maxiter_per_lam: int,
                 deadline: float,
                 fft_len: int,
                 n: int):
    """
    Run L‑BFGS for each lam_factor in `lam_schedule`.  Starts each stage
    from the result of the previous stage.  Returns the best sequence
    encountered and its fast_score.
    """
    y = np.log(a_start + 1e-12)           # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # exact convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_idx = int(np.argmax(conv))

        # gradient of the (hard) max‑convolution (single active index)
        # derivative w.r.t. a_i: 2 * a_{max_idx-i} (if in range), except when i==max_idx-i
        g = np.zeros_like(a)
        lo = max_idx - (n - 1)
        hi = min(max_idx, n - 1)
        # i ranges where j = max_idx - i is inside [0, n-1]
        i_vals = np.arange(lo, hi + 1)
        j_vals = max_idx - i_vals
        # contributions
        g[i_vals] = 2.0 * a[j_vals]
        # correct the diagonal case (i == j)
        diag = (i_vals == j_vals)
        g[i_vals[diag]] = a[i_vals[diag]]

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        eta = 1.0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float):
    """
    Greedy mass‑transfer annealing.
    Randomly moves a small fraction of mass from one coordinate to another,
    accepting moves that improve the score or with a temperature‑dependent
    probability.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = start_score
    rng = np.random.default_rng()
    start_time = time.time()

    while time.time() < deadline:
        # linear progress in [0,1]
        prog = (time.time() - start_time) / (deadline - start_time + 1e-12)
        # temperature decays exponentially from 0.02 to 1e-6
        T = 0.02 * (1e-6 / 0.02) ** prog

        i, j = rng.integers(0, n, size=2)
        if i == j or a[i] == 0.0:
            continue
        # move at most 5 % of the source mass
        delta = a[i] * rng.random() * 0.05
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta          # sum unchanged ⇒ stays on simplex

        new_score = fast_score(a_new, fft_len, n)
        dE = new_score - best_score
        if dE < 0.0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best = a_new.copy()

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    n = 4000                         # length of the step function
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two for fast FFTs

    # --------------------------------------------------------------
    # Gather a diverse initial pool
    # --------------------------------------------------------------
    init_candidates = []

    # previous best if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    init_candidates.append(a_prev)

    # deterministic families
    init_candidates.append(spectral_init(n))
    init_candidates.append(random_initial(n, rng))

    # parametric sweep (including sparse families)
    family_best = sweep_parameter_families(n, fft_len, rng, keep=8)
    init_candidates.extend(family_best)

    # --------------------------------------------------------------
    # Evaluate the pool and retain the most promising seeds
    # --------------------------------------------------------------
    best_a = init_candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in init_candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # keep a handful of seeds for intensive optimisation
    seeds = [best_a]
    # add a few random Dirichlet and sparse starters
    for _ in range(3):
        seeds.append(random_initial(n, rng))
        seeds.append(random_sparse_initial(n, rng))
        seeds.append(weighted_sparse_initial(n, rng))

    # keep only the best 5 distinct seeds
    seeds_scored = [(fast_score(s, fft_len, n), s) for s in seeds]
    seeds_scored.sort(key=lambda x: x[0])
    seeds = [s for _, s in seeds_scored[:5]]

    # --------------------------------------------------------------
    # Timing management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing (coarse → fine smoothing)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001,
        5e-05, 2e-05, 1e-05, 5e-06
    ]
    maxiter_per_lam = 800

    for a_start in seeds:
        if time.time() >= phase1_deadline:
            break
        a_opt, sc_opt = lbfgs_refine(a_start,
                                     lam_schedule,
                                     maxiter_per_lam,
                                     phase1_deadline,
                                     fft_len,
                                     n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent on the exact max
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing mass‑transfer polishing
    # --------------------------------------------------------------
    phase3_deadline = deadline
    if time.time() < phase3_deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len,
                                                 n,
                                                 phase3_deadline)
        print(f"[anneal] final refined score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Return the final sequence
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
