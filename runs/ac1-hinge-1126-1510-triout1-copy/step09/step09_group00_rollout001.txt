def compute_subgradient(a):
    n = len(a)
    fft_len = next_pow2(2*n -1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(float)
    # compute gradient w.r.t a
    a_rev = a[::-1]
    fft_len_g = next_pow2(3*n-2)  # for convolution of mask and reversed a
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * conv_g[n-1 : n-1 + n]   # length n
    return g
```

def adam_optimize(a0, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    it=0
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    while time.time() < deadline:
        g = compute_subgradient(a)
        it += 1
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*(g**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        step = 0.05 * m_hat/(np.sqrt(v_hat)+eps)    # lr schedule -> 0.05 may be okay
        a = a - step
        # ensure nonnegativity and simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a = a / s
        # symmetrize optionally
        a = (a + a[::-1]) / 2.0
        # normalize again
        a = a / a.sum()
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] new best fast_score={best_score:.6f}")
    return best_a, best_score

def simulated_annealing_v2(a_start, start_score, fft_len, n, deadline, rng):
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score
    T0 = max(0.1 * start_score, 1e-6) # set initial temperature high relative
    t0 = time.time()
    total = max(deadline - t0, 1e-6)
    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        fraction = remain / total
        T = T0 * (fraction ** 0.4)   # slower decay (exponent < 1)
        # number of source indices to take from: random 1-3
        k_src = rng.integers(1, 4) # 1-3
        src_idxs = rng.choice(n, size=k_src, replace=False)
        dst_idxs = rng.choice(n, size=k_src, replace=False)
        # compute total mass to move from src
        tot_mass = sum(a[i] for i in src_idxs)
        if tot_mass <= 0:
            continue
        # fraction to move (0-0.5)
        frac = rng.random() * 0.5
        moved = tot_mass * frac
        if moved <= 0:
            continue
        # distribute moved mass equally to dests
        delta_src = moved / k_src
        delta_dst = moved / k_src
        a_new = a.copy()
        for i in src_idxs:
            a_new[i] = max(a_new[i] - delta_src, 0.0)
        for j in dst_idxs:
            a_new[j] += delta_dst
        # project to ensure sum unchanged (should already)
        # symmetrize
        #a_new = (a_new + a_new[::-1]) / 2.0
        # Normalize (should be close)
        s = a_new.sum()
        if s == 0:
            continue
        a_new = a_new / s
        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
        # occasional re-symmetrize
        if rng.random() < 0.01:
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()
    return best_a, best_score

"""
Height sequence optimizer for the `evaluate_sequence` metric.

The algorithm works in several stages, always respecting a global time
budget (default 1000 s):

1. **Cheap parametric sweep** – samples many simple families (power,
   exponential, Gaussian, beta, sinusoidal, etc.) and keeps the best
   candidate.

2. **L‑BFGS on a smoothed objective** – the non‑smooth max of the
   auto‑convolution is approximated by a log‑sum‑exp (controlled by a
   schedule of smoothing‑parameters).  The vector is represented by
   soft‑max logits, guaranteeing positivity and unit‑sum.

3. **Adam‑style sub‑gradient descent** – uses the true sub‑gradient of
   the max‑convolution together with an Adam schedule and projects onto
   the simplex after each step.  The vector is symmetrised after each
   update (a[i] = a[n‑1‑i]) which is known to be near‑optimal for this
   problem.

4. **Slow‑cooling simulated annealing** – a stochastic hill‑climb that
   transfers a random fraction of mass between a few indices and accepts
   uphill moves with a square‑root temperature decay.

5. **Final evaluation** – the best vector is returned as a plain list
   of floats; `evaluate_sequence` is called once to obtain the final
   score.

All phases share the global deadline; the best solution seen at any
moment is returned.  The implementation uses only NumPy and SciPy and
runs comfortably within the 1000 s limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑ x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation for a normalised vector (∑a = 1):
    returns 2·n·max(a*a) which equals `evaluate_sequence`.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fm = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fm))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits for the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        # re‑symmetrise (symmetry is known to be beneficial)
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def compute_subgradient(a: np.ndarray, fft_len_conv: int, fft_len_g: int) -> np.ndarray:
    """
    True sub‑gradient of max(a*a) w.r.t. a (using a mask of all
    maximising positions). Returns a vector of length n.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(float)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return g


def adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation with true sub‑gradient and projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0

    while time.time() < deadline:
        g = compute_subgradient(a, fft_len_conv, fft_len_g)

        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule (decays like 1/√it)
        lr = 0.05 / np.sqrt(it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # project onto simplex and re‑symmetrise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # symmetry helps to keep the shape regular
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[adam] new best fast_score={best_score:.6f}")

    return best_a, best_score


def simulated_annealing_v2(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a random fraction of mass between a
    few indices; temperature decays with a square‑root exponent < 1
    (slow cooling).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature: a noticeable fraction of the current score
    T0 = max(0.08 * start_score, 1e-6)
    t0 = time.time()
    total = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        # exponent 0.4 gives slower than sqrt decay
        T = T0 * (remain / total) ** 0.4

        # choose a random number of source/destination indices (1‑3)
        k_src = rng.integers(1, 4)
        src = rng.choice(n, size=k_src, replace=False)
        dst = rng.choice(n, size=k_src, replace=False)

        src_mass = a[src].sum()
        if src_mass <= 0.0:
            continue

        # random fraction of the total source mass to move (up to 40 %)
        frac = rng.random() * 0.40
        moved = src_mass * frac
        if moved <= 0.0:
            continue

        delta_src = moved / k_src
        delta_dst = moved / k_src

        a_new = a.copy()
        a_new[src] -= delta_src
        a_new[src] = np.maximum(a_new[src], 0.0)
        a_new[dst] += delta_dst

        # re‑normalise and re‑symmetrise
        a_new = a_new / a_new.sum()
        a_new = (a_new + a_new[::-1]) / 2.0
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        # Metropolis acceptance
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        # occasional extra symmetrisation (helps numeric stability)
        if rng.random() < 0.01:
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector sampled from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "beta":
        alpha = rng.uniform(0.5, 5.0)
        beta = rng.uniform(0.5, 5.0)
        x = (i + 0.5) / n
        a = x ** (alpha - 1) * (1.0 - x) ** (beta - 1)
    elif family == "sin_power":
        p = rng.uniform(0.5, 8.0)
        a = np.sin(np.pi * (i + 0.5) / n) ** p
    elif family == "triangular":
        a = 1.0 - np.abs((i - (n - 1) / 2.0) / ((n - 1) / 2.0))
        a = np.maximum(a, 0.0)
    elif family == "raised_cos":
        width = rng.uniform(0.1, 0.5) * n
        a = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / width))
        a = np.maximum(a, 0.0)
    else:  # fallback: pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0 or np.isnan(total):
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Cheap sweep over many random parametric families.
    Returns the best vector and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10  # 10 % of total budget

    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "beta",
        "sin_power",
        "triangular",
        "raised_cos",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Use a previously found construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a_old = np.interp(x_new, x_old, hs)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            score_old = fast_score(a_old, fft_len_conv, n)
            if score_old < best_score:
                best_a, best_score = a_old, score_old
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] parametric sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # allocate time slices
    phase1_end = start_time + 0.40 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.95 * budget_s
    # remaining → final fine‑tuning (none in this version)

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS with finer schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01,
        0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
        5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # start from a jittered version of the current best or a random vector
        if rng.random() < 0.5:
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam‑style sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] Adam best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – slow‑cooling simulated annealing
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing_v2(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final symmetry enforcement and normalisation
    # --------------------------------------------------------------
    best_a = (best_a + best_a[::-1]) / 2.0
    best_a = best_a / best_a.sum()

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
