def l2_objective_and_grad(y, n, fft_len_conv, fft_len_g):
    Compute a via softmax(y).
    conv = real(ifft(fft(a)*fft(a)))[:2*n-1]
    target = 1.0/(2*n-1)
    diff = conv - target
    loss = np.sum(diff**2)
    # gradient of loss w.r.t a: 2 * conv(a_rev, diff)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fd = np.fft.fft(diff, fft_len_g)
    grad_a = 2 * np.real(np.fft.ifft(fa_rev * fd))[(n-1):(n-1 + n)]

    # chain rule through softmax
    mean = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean)
    return loss, grad_y

def l2_objective_and_grad(y, n, fft_len_conv, fft_len_g):
    # compute a via softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()
    # conv
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    target = 1.0 / (2*n - 1)
    diff = conv - target
    loss = np.sum(diff * diff)
    # gradient: 2 * convolution of reversed a with diff
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fd = np.fft.fft(diff, fft_len_g)
    grad_a = 2.0 * np.real(np.fft.ifft(fa_rev * fd))[(n-1):(n-1 + n)]
    # backprop through softmax

"""
Improved multi‑phase optimizer for the step‑function problem.

The search proceeds in four stages (all respecting the 1000 s budget):

1️⃣  Random (and analytic) restarts + heavy L‑BFGS on a smooth log‑sum‑exp
    surrogate of   max(conv(a,a)).  The continuation over a decreasing
    λ‑schedule drives the solution towards the true nonsmooth objective.

2️⃣  Projected Adam (sub‑gradient) refinement on the exact max‑convolution,
    followed by a short L2‑flattening L‑BFGS phase that minimises the
    squared deviation of the whole convolution from the uniform target
    (≈ 1/(2n‑1)).  This helps “spread” the mass and reduces the peak.

3️⃣  Aggressive simulated‑annealing / hill‑climb with many iterations.
    Moves transfer a fraction of mass from high‑gradient to low‑gradient
    indices and occasionally swap entries.  The temperature decays
    slowly, allowing a large number of evaluations within the time budget.

4️⃣  Fine‑tuned L‑BFGS with an ultra‑small λ‑schedule that polishes the
    final vector.

All vectors are kept on the probability simplex (non‑negative,
sum = 1) via a softmax parameterisation during the gradient‑based phases
and a Euclidean simplex projection for the Adam / annealing steps.

The best sequence found is stored globally as ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) assuming sum(a)=1.
    Full linear convolution via FFT, then the max entry.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal"], replace=False
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def analytic_initial(n: int) -> np.ndarray:
    """Return the best among a handful of hand‑crafted shapes."""
    x = np.arange(n, dtype=float)
    cand = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n))

    # 2. half‑sine (positive on the whole interval)
    s = np.sin(np.pi * (x + 0.5) / n)
    cand.append(s)

    # 3. raised‑cosine (a bell shape)
    r = (1 - np.cos(2.0 * np.pi * x / (n - 1))) / 2.0
    cand.append(r)

    # 4. quadratic decay (more mass near the centre)
    q = 1.0 / np.sqrt(x + 1.0)
    cand.append(q)

    # 5. power‑law with exponent 0.7
    p = (x + 1.0) ** -0.7
    cand.append(p)

    # keep non‑negative, normalise and pick the best according to fast_score
    best = None
    best_score = np.inf
    # placeholder FFT lengths – will be recomputed later when n is known
    fft_len = next_pow2(2 * n - 1)
    for a in cand:
        a = np.maximum(a, 0.0)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    return best


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------


def smooth_max_obj_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                            fft_len_g: int, lam_factor: float = 0.30):
    """
    Smooth approximation of 2·n·max(conv) where a = softmax(y).
    Returns (objective, gradient wrt y).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=smooth_max_obj_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(a: np.ndarray,
                        fft_len_conv: int,
                        fft_len_g: int):
    """
    Returns (max_conv, subgradient) for the exact max‑convolution.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Projected Adam on the exact non‑smooth objective
# ----------------------------------------------------------------------


def subgradient_adam(a0: np.ndarray,
                     deadline: float,
                     fft_len_conv: int,
                     fft_len_g: int,
                     n: int):
    """
    Projected Adam on the true sub‑gradient of max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        _, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)          # diminishing step
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum

    return best_a, best_score

# L2‑flattening phase (minimise deviation from uniform convolution)
# ----------------------------------------------------------------------


def l2_obj_and_grad(y: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int):
    """
    Objective = Σ (conv - 1/(2n‑1))²  (L2 distance to the uniform target).
    Returns (loss, gradient w.r.t. logits y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    target = 1.0 / (2 * n - 1)
    diff = conv - target
    loss = np.sum(diff * diff)

    # gradient w.r.t. a : 2 * (a_rev * diff)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fd = np.fft.fft(diff, fft_len_g)
    grad_a = 2.0 * np.real(np.fft.ifft(fa_rev * fd))[(n - 1) : (n - 1 + n)]

    # back‑prop through softmax
    mean = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean)

    return loss, grad_y


def l2_lbfgs_optimize(a_start: np.ndarray,

def annealing_hill_climb(a_start: np.ndarray,
                         current_score: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000):
    """
    Aggressive hill‑climb with simple simulated‑annealing acceptance.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9995
    min_temp = 1e-9
    it = 0

    while time.time() < deadline and it < max_iters:
        # exact max and sub‑gradient for the current vector
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # pick source / target sets based on gradient percentiles
        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.6 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i != j:
                frac = rng.uniform(0.03, 0.15)           # finer transfers as we cool
                delta = cur_a[i] * frac
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # keep feasibility
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        # acceptance rule
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser returning a list of non‑negative floats (sum≈1)
    that minimizes the external ``evaluate_sequence`` objective.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a compromise between resolution and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # FFT lengths (padded to powers of two for speed).
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start: use previously found solution if present, otherwise
    # the best analytic shape.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            a0 = analytic_initial(n)
    else:
        a0 = analytic_initial(n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.1   # tiny safety margin

    # allocate phases (empirically tuned)
    phase1_end = start + 0.20 * (budget_s)           # random restarts + L‑BFGS
    phase2_end = start + 0.45 * (budget_s)           # Adam
    phase3_end = start + 0.80 * (budget_s)           # Annealing
    # remainder → fine polishing

    # ------------------------------------------------------------------
    # Phase 1 – many random (and analytic) restarts, heavy L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-05, 1e-05,
    ]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        # pick a diverse starter
        starter_type = rng.choice(["random", "analytic"])
        if starter_type == "analytic":
            a_start = analytic_initial(n)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[Phase1] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[Phase2] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2b – L2 flattening (minimise deviation from the uniform target)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, score_l2 = l2_lbfgs_optimize(
            best_a,
            maxiter=500,
            deadline=phase3_end,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if score_l2 < best_score:
            best_score = score_l2
            print(f"[Phase2b L2] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing / hill‑climb (large iteration budget)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iters=2_000_000,
        )
        print(f"[Phase3 Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final polishing with a very small λ‑schedule
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [1e-05, 5e-06, 2e-06, 1e-06, 5e-07, 2e-07]
        fine_maxiter = 400
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[Phase4 fine L‑BFGS] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally for possible warm‑starts later
    globals()["height_sequence_1"] = final_seq

    return final_seq
