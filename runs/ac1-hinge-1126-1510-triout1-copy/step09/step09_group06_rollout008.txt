    # conv
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]

    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)

"""
Multi‑phase optimizer for the max‑self‑convolution objective.

The algorithm builds a diverse pool of initial step functions,
polishes them with a short L‑BFGS run on a smoothed max objective,
then performs a long Adam‑style optimisation on the exact smoothed
max, and finally applies an aggressive greedy hill‑climb.
All phases respect a global time budget (default 1000 s) and keep the
best sequence found.  The output sequence is normalised to sum = 1,
so evaluate_sequence() returns the true objective value.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    y = np.asarray(y, dtype=np.float64)
    m = np.max(y)
    e = np.exp(y - m)
    s = np.sum(e)
    if s == 0:
        # fallback uniform
        return np.full_like(y, 1.0 / y.shape[0])
    return e / s


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum(a)=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (linear, not circular)
    fa = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len)
    fw = np.fft.rfft(w, fft_len)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f_val, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  n: int,
                  fft_len: int,
                  deadline: float,
                  rng: np.random.Generator):
    """
    Adam‑style optimisation on the smoothed max‑convolution.
    Works in the log‑softmax space (logits y) to keep the simplex constraint.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline:
        # decreasing smoothing factor
        lam_factor = max(0.001, 0.5 / np.sqrt(it + 1))

        f_val, grad_y = objective_and_grad(y, n, fft_len, lam_factor)

        # Adam moments update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        # evaluate
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

        # occasional random jitter to help escape flat regions
        if it % 500 == 0:
            jitter = rng.normal(scale=0.02, size=y.shape)
            y = y + jitter

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move at most 12 % of the source mass
        delta = best[i] * rng.random() * 0.12
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L, dtype=np.complex128)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (several families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.005, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def sweep_parameter_families(n: int, fft_len: int, rng: np.random.Generator,
                            keep: int = 8):
    """
    Explore a few parametric families and return the best `keep` candidates.
    """
    candidates = []

    # Exponential family
    for c in np.logspace(-4, -0.4, num=15):
        a = exponential_shape(n, c)
        score = fast_score(a, fft_len, n)
        candidates.append((score, a))

    # Power‑law family
    for p in np.linspace(0.1, 4.5, num=20):
        a = powerlaw_shape(n, p)
        score = fast_score(a, fft_len, n)
        candidates.append((score, a))

    # Edge‑heavy families (both interior‑zero and non‑zero)
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100, 200]
    for ef in edge_fracs:
        for r in ratios:
            a1 = edge_shape(n, ef, r, interior_zero=False)
            candidates.append((fast_score(a1, fft_len, n), a1))
            a2 = edge_shape(n, ef, r, interior_zero=True)
            candidates.append((fast_score(a2, fft_len, n), a2))

    # Dirichlet‑type random draws
    for _ in range(40):
        alpha = rng.uniform(0.05, 2.0, size=n)
        a = rng.gamma(alpha)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full(n, 1.0 / n)
        candidates.append((fast_score(a, fft_len, n), a))

    # Select the best distinct candidates
    candidates.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in candidates:
        h = round(sc, 8)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    # ------------------------------------------------------------------
    # RNG and problem size
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Length of the step function – a larger value gives more freedom.
    n = 6000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build initial pool of candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # Resize by linear interpolation
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # 2️⃣  Spectral and random initialisations
    candidates.append(spectral_init(n))
    candidates.append(random_initial(n, rng))

    # 3️⃣  Parameter sweep for additional diversity
    family_best = sweep_parameter_families(n, fft_len, rng, keep=6)
    candidates.extend(family_best)

    # ------------------------------------------------------------------
    # Keep the best few seeds for the intensive phases
    # ------------------------------------------------------------------
    scored = [(fast_score(c, fft_len, n), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    best_seeds = [c for _, c in scored[:5]]
    best_a = best_seeds[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS polishing (≈15 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.15 * budget_s
    lam_schedule = np.geomspace(0.5, 0.001, num=18)  # decreasing smoothness
    maxiter_per_lam = 500

    # run L‑BFGS on each seed while time permits
    for a_start in best_seeds:
        if time.time() >= phase1_deadline:
            break
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phase1_deadline,
                                            fft_len,
                                            n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈60 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.75 * budget_s
    # We perform several short Adam runs from perturbed versions of the current best.
    while time.time() < phase2_deadline:
        # generate a start point: either the current best or a jittered version
        if rng.random() < 0.3:
            start_a = best_a.copy()
        else:
            jitter = rng.normal(scale=0.02, size=best_a.shape)
            start_a = softmax(np.log(best_a + 1e-12) + jitter)
        sub_deadline = min(time.time() + 6.0, phase2_deadline)  # ≤6 s per Adam run
        a_opt, sc_opt = adam_optimize(start_a, n, fft_len, sub_deadline, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Adam] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        max_iter=200000)
        print(f"[hill‑climb] final refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # type: ignore  # a provided global
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose the sequence for a possible next run
    globals()['height_sequence_1'] = final_seq

    return final_seq
