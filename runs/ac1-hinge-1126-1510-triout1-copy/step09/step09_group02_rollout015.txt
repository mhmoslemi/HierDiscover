"""
Optimiser for the step‑function challenge.

Approach
--------
1. **Parametric seed (geometric progression)** –  
   A distribution  a_i ∝ r^i  (r close to 1) is a very good
   starting point for the anti‑concentration problem.
   We scan the decay parameter α = n·(1‑r) on a fine grid
   and keep the best sequence (fast evaluation by FFT).

2. **Local refinement – exponentiated gradient** –  
   Using the exact sub‑gradient of the true max‑convolution we
   perform an exponentiated‑gradient descent with a decreasing
   learning‑rate η∝1/√t.  The iterates stay on the probability
   simplex automatically.

3. **Fine‑grained hill‑climbing** –  
   Random mass moves (source → destination) are accepted only
   if they strictly improve the objective.  Convolution updates
   are done incrementally for speed.

The whole pipeline respects the 1000 s time budget and returns the
best sequence found.  The final evaluation is performed with the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Primitive building blocks
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of a with itself via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    For a probability vector (sum(a)=1) the evaluation reduces to
    2·n·max(conv).  This function returns exactly that quantity.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


# ----------------------------------------------------------------------
# Geometric‑progression seed
# ----------------------------------------------------------------------


def geometric_sequence(n: int, r: float) -> np.ndarray:
    """Return a normalized geometric progression a_i ∝ r**i (r∈(0,1])."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r))
    a /= a.sum()
    return a


def search_geometric(n: int, fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan α = n·(1‑r) on a coarse grid, then refine by golden‑section
    search.  Returns the best sequence and its fast_score.
    """
    i = np.arange(n, dtype=np.float64)
    best_a = None
    best_score = np.inf
    best_alpha = None

    # -------- coarse grid -------------------------------------------------
    for alpha in np.linspace(0.5, 6.0, 111):          # step ≈0.05
        if time.time() >= deadline:
            break
        r = 1.0 - alpha / n
        if r <= 0.0:
            continue
        a = np.exp(i * np.log(r))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_alpha = alpha

    # -------- golden‑section refinement ------------------------------------
    if best_alpha is None:
        # fallback – uniform distribution
        a = np.full(n, 1.0 / n)
        return a, fast_score(a, fft_len)

    left = max(0.1, best_alpha - 0.5)
    right = best_alpha + 0.5
    phi = (np.sqrt(5) - 1) / 2          # 0.618...

    for _ in range(30):
        if time.time() >= deadline:
            break
        m1 = right - phi * (right - left)
        m2 = left + phi * (right - left)

        r1 = 1.0 - m1 / n
        r2 = 1.0 - m2 / n
        if r1 <= 0 or r2 <= 0:
            break

        a1 = np.exp(i * np.log(r1))
        a1 /= a1.sum()
        s1 = fast_score(a1, fft_len)

        a2 = np.exp(i * np.log(r2))
        a2 /= a2.sum()
        s2 = fast_score(a2, fft_len)

        if s1 < s2:
            right, best_alpha, best_score, best_a = m2, m1, s1, a1
        else:
            left, best_alpha, best_score, best_a = m1, m2, s2, a2

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient of max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution using exponentiated
    updates (multiplicative weights).  The iterates stay on the simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative‑weights update and re‑normalisation
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climbing with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a: np.ndarray,
                           curr_score: float,
                           fft_len: int,
                           n: int,
                           deadline: float,
                           rng: np.random.Generator,
                           max_fraction: float = 0.18,
                           recompute_every: int = 1200) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass from a larger entry to a
    smaller one.  The convolution is updated incrementally, avoiding
    an O(n log n) recomputation for each trial.
    """
    best_a = a.copy()
    best_score = curr_score
    conv = np.convolve(best_a, best_a)          # exact full convolution
    diff = np.zeros_like(conv)
    accepted = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # always move from the larger entry to the smaller entry
        if best_a[i] < best_a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if best_a[src] == 0.0:
            continue

        delta = best_a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= best_a
        diff[dst:dst + n] += best_a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best_a[src] -= delta
            best_a[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(best_a, best_a)

    return best_a, best_score
