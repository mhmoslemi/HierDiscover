def simulated_annealing_optimize(a_start, deadline, n, fft_len, rng):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    cur_score = best_score
    start = time.time()
    max_move = 0.2
    # define temperature based on best_score
    T0 = best_score * 0.05   # 5% of best score
    while time.time() < deadline:
        # compute elapsed fraction
        elapsed = time.time() - start
        remaining = deadline - time.time()
        # progress fraction
        frac = elapsed / (deadline - start)
        T = T0 * max(1.0 - frac, 0.0) + 1e-12
        
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        source_mass = a[i]
        if source_mass <= 0:
            continue
        # delta limited to max_move fraction of source mass
        delta = source_mass * rng.random() * max_move
        # propose new a
        a[i] -= delta
        a[j] += delta
        
        new_score = fast_score(a, fft_len, n)
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta
    return best_a, best_score

  1️⃣  Build a diverse pool of initial candidates (previous best,
      spectral, random Dirichlet‑like, and parametric families).

  2️⃣  Short L‑BFGS polishing from a handful of seeds with a long
      λ‑schedule (smooth approximation of the max‑convolution).

  3️⃣  Long exponentiated‑gradient descent on the exact max‑convolution.

  4️⃣  Projected sub‑gradient descent with a larger initial step size.

  5️⃣  Simulated‑annealing hill‑climb (tiny mass transfers, accepting
      uphill moves) for a fine‑grained search.

  6️⃣  Final ultra‑short L‑BFGS run with extremely small λ to polish
      the result.

All phases respect the global deadline and keep the best sequence
found.  The objective is invariant under scaling, therefore the
distribution is normalised to sum = 1 during the optimisation; the
final list is returned unchanged – the exact same values are passed to
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (sum=1 ⇒ 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)           # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step‑size schedule (larger initial step)
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clip exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def subgrad_optimize(a_start: np.ndarray,
                     deadline: float,
                     n: int,
                     fft_len: int,
                     step0: float = 0.5):
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Uses a larger initial step size than the previous implementation.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction that preserves the sum
        d = -(g - np.mean(g))

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 20000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def simulated_annealing_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator):
    """
    Simulated‑annealing style local search.
    Small random mass transfers are accepted with a Metropolis criterion.
    Intended as a very fine polishing step after gradient methods.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    cur_score = best_score

    start = time.time()
    max_move_frac = 0.20               # up to 20 % of source mass per move
    T0 = best_score * 0.05            # initial temperature (5 % of the current score)

    while time.time() < deadline:
        # linear cooling
        progress = (time.time() - start) / (deadline - start)
        T = max(T0 * (1.0 - progress), 1e-12)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        src = a[i]
        if src <= 0.0:
            continue

        delta = src * rng.random() * max_move_frac
        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / T):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # undo
            a[i] += delta
            a[j] -= delta

    return best_a, best_score


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex (various families)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    return a / a.sum()


def sweep_parameter_families(n: int, fft_len: int, rng: np.random.Generator,
                            keep: int = 8):
    """
    Explore a few parametric families and return the best `keep` candidates.
    """
    candidates = []

    # Exponential family
    for c in np.logspace(-4, -0.5, num=12):
        a = exponential_shape(n, c)
        candidates.append((fast_score(a, fft_len, n), a))

    # Power‑law family
    for p in np.linspace(0.1, 4.0, num=15):
        a = powerlaw_shape(n, p)
        candidates.append((fast_score(a, fft_len, n), a))

    # Edge‑heavy families
    edge_fracs = [0.001, 0.002, 0.004, 0.008, 0.012, 0.02, 0.03, 0.05]
    ratios = [2, 5, 10, 20, 50, 100]
    for ef in edge_fracs:
        for r in ratios:
            a1 = edge_shape(n, ef, r, interior_zero=False)
            candidates.append((fast_score(a1, fft_len, n), a1))
            a2 = edge_shape(n, ef, r, interior_zero=True)
            candidates.append((fast_score(a2, fft_len, n), a2))

    # Random Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0, size=n)
        a = rng.gamma(alpha)
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.full(n, 1.0 / n)
        candidates.append((fast_score(a, fft_len, n), a))

    # keep distinct best candidates
    candidates.sort(key=lambda x: x[0])
    best = []
    seen = set()
    for sc, a in candidates:
        h = round(sc, 8)
        if h in seen:
            continue
        seen.add(h)
        best.append((sc, a))
        if len(best) >= keep:
            break
    return [a for _, a in best]

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                         # length of the step function
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two for fast FFTs

    # ------------------------------------------------------------------
    # Gather initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # 2️⃣  Spectral and random initialisations
    candidates.append(spectral_init(n))
    candidates.append(random_initial(n, rng))

    # 3️⃣  Sweep a few parametric families
    family_best = sweep_parameter_families(n, fft_len, rng, keep=6)
    candidates.extend(family_best)

    # 4️⃣  Additional random draws (cheap, many)
    random_draw_deadline = time.time() + 0.08 * budget_s
    while time.time() < random_draw_deadline:
        a = random_initial(n, rng)
        candidates.append(a)

    # ------------------------------------------------------------------
    # Keep the best few seeds for the expensive phases
    # ------------------------------------------------------------------
    candidates_scored = [(fast_score(a, fft_len, n), a) for a in candidates]
    candidates_scored.sort(key=lambda x: x[0])
    best_a = candidates_scored[0][1].copy()
    best_score = candidates_scored[0][0]
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # Keep the top 5 distinct seeds
    seeds = [cand for _, cand in candidates_scored[:5]]

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS polishing from a handful of diverse seeds
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.25 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 5e-05, 2e-05, 1e-05,
        5e-06, 2e-06, 1e-06
    ]
    maxiter_per_lam = 800

    for a_start in seeds:
        if time.time() >= phase1_deadline:
            break
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phase1_deadline,
                                            fft_len,
                                            fft_len,
                                            n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.55 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_deadline,
                                                   n,
                                                   fft_len,
                                                   rng)
        print(f"[exp‑grad] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (larger step0)
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.75 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = subgrad_optimize(best_a,
                                              phase3_deadline,
                                              n,
                                              fft_len,
                                              step0=0.8)
        print(f"[subgrad] best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated‑annealing fine polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_optimize(best_a,
                                                          deadline,
                                                          n,
                                                          fft_len,
                                                          rng)
        print(f"[SA] best score after annealing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final ultra‑short L‑BFGS run with tiny λ (finetune)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        tiny_lam_schedule = [5e-05, 2e-05, 1e-05, 5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07]
        best_a, best_score = lbfgs_continuation(best_a,
                                                tiny_lam_schedule,
                                                maxiter_per_lam=400,
                                                deadline=deadline,
                                                fft_len_conv=fft_len,
                                                fft_len_g=fft_len,
                                                n=n)
        print(f"[final L‑BFGS] refined score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the result
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
