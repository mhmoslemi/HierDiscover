"""
Search for a non‑negative step‑function (vector on the simplex) that
minimises 2·n·max(conv(a,a)) / (sum a)^2.
The optimiser combines three complementary phases:

1. **Broad random exploration** – many Dirichlet samples are generated
   and the best few are kept.
2. **Smooth L‑BFGS continuation** – a log‑sum‑exp surrogate of the
   max‑convolution is minimised with decreasing smoothing
   parameters, using the L‑BFGS‑B optimiser.
3. **Exponentiated‑gradient (EG) sub‑gradient descent** – the exact
   max‑convex sub‑gradient is computed (via FFT) and the iterate is
   updated by a KL‑projected (multiplicative) step, which respects the
   simplex constraints naturally.  An adaptive step‑size schedule
   yields monotone improvement.
4. **Fine L‑BFGS polishing** – a final run with extremely small
   smoothing parameters refines the solution.

All phases respect the global 1000 s budget; the best vector found is
returned as a plain list.
"""
import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())

def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) under the assumption sum(a)=1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    return float(2.0 * n * conv.max())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet distribution)."""
    x = rng.gamma(alpha, 1.0, size=n)
    x_sum = x.sum()
    if x_sum == 0:
        return np.full(n, 1.0 / n)
    return x / x_sum

def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int, lam: float):
    """Smooth log‑sum‑exp surrogate of 2·n·max(conv)."""
    # softmax safely
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()
    # convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa)[: 2 * n - 1]
    max_c = conv_full.max()
    # smoothed max
    shift = conv_full - max_c
    exp_shift = np.exp(shift / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max
    # gradient wrt conv entries
    w = exp_shift / Z
    # back‑prop through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw)[: 2 * n - 1]
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]
    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def lbfgs_continuation(a_start: np.ndarray, lam_schedule, maxiter_per_lam: int,
                       deadline: float, fft_len_conv: int, fft_len_g: int):
    """Run L‑BFGS for a list of smoothing parameters."""
    n = a_start.shape[0]
    # initialise logits
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv)
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] λ={lam:.2e} new best={best_score:.6f}")
    return best_a, best_score

def max_and_subgradient(a: np.ndarray, fft_len_conv: int, fft_len_g: int):
    """Exact max‑convolution value and one sub‑gradient."""
    n = a.shape[0]
    # full convolution via FFT
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa)[: 2 * n - 1]
    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)
    # sub‑gradient = 2 * (a_rev * mask)
    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_g = np.fft.irfft(fa_rev * fm)[: 2 * n - 1]
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g

def exponentiated_gradient(a0: np.ndarray, deadline: float,
                          fft_len_conv: int, fft_len_g: int):
    """KL‑projected sub‑gradient descent (multiplicative updates)."""
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv)
    t = 0
    while time.time() < deadline:
        t += 1
        _, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)
        # adaptive step size

"""
Search for a non‑negative step‑function (vector on the simplex) that
minimises

    2·n·max(conv(a, a)) / (sum(a) ** 2)

as required by ``evaluate_sequence``.  The optimiser consists of four
phases that are carefully scheduled inside the 1000 s budget:

1. **Random Dirichlet exploration** – many random simplex points are
   generated (with a range of concentration parameters) and the best
   few are retained.

2. **Heavy L‑BFGS continuation** – a smooth log‑sum‑exp surrogate of the
   max‑convolution is minimised with a decreasing schedule of smoothing
   parameters.  This quickly moves the iterate into a good basin.

3. **Exponentiated‑gradient (EG) sub‑gradient descent** – the exact
   sub‑gradient of the (nonsmooth) max‑convolution is computed via FFT;
   a KL‑projected multiplicative update moves mass away from high‑gradient
   coordinates while staying on the probability simplex.

4. **Fine L‑BFGS polishing** – very small smoothing parameters are used
   to polish the final vector.

All phases respect a global time limit; the best vector found is
returned as a plain Python list (its entries sum to ≈ 1).  The final
vector is stored globally as ``height_sequence_1`` for warm‑starts in
later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) assuming sum(a)=1.
    Uses FFT for speed.
    """
    n = a.shape[0]
    # real FFT – length fft_len (≥ 2*n-1) is required
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a random point on the simplex (Dirichlet distribution)."""
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0:
        return np.full(n, 1.0 / n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
):
    """
    Smooth approximation of 2·n·max(conv) where a = softmax(y).
    Returns function value and gradient w.r.t. y.
    """
    # --- softmax (stable) ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution a*a ---
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa)[: 2 * n - 1]

    max_c = conv_full.max()

    # --- smoothed max via log‑sum‑exp ---
    shift = conv_full - max_c
    exp_shift = np.exp(shift / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- gradient of smooth max w.r.t. convolution entries ---
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # --- back‑propagation through the convolution (a_rev * w) ---
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw)[: 2 * n - 1]
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --- chain rule through the softmax ---
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    n = a_start.shape[0]
    y = np.log(a_start + 1e-12)  # initialise logits safely
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] λ={lam:.2e} new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and its sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(a: np.ndarray, fft_len_conv: int, fft_len_g: int):
    """
    Returns (max_conv, subgradient) for the exact max‑convolution of a
    with itself.  The sub‑gradient corresponds to the sum of contributions
    from all shifts attaining the maximal value (within numerical tolerance).
    """
    n = a.shape[0]

    # full convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa)[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_g = np.fft.irfft(fa_rev * fm)[: 2 * n - 1]
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact nonsmooth objective
# ----------------------------------------------------------------------


def exponentiated_gradient(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[np.ndarray, float]:
    """
    KL‑projected (multiplicative) sub‑gradient descent.
    Updates are a_i ← a_i · exp(‑η·g_i) followed by renormalisation.
    Returns the best vector observed before the deadline.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv)

    it = 0
    while time.time() < deadline:
        it += 1
        _, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # adaptive step size (standard η_t = η₀ / √t)
        eta = 0.5 / np.sqrt(it)

        # multiplicative update (KL‑projection)
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        cur_score = fast_score(a, fft_len_conv)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[EG] it={it} new best={best_score:.6f}")

    return best_a, best_score

        fft_len_g=fft_len_g,
    )
    if fine_score < best_score:
        best_score = fine_score
        best_a = best_a.copy()
    print(f"[phase 3] final best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final external evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # make the best sequence globally available for warm‑starts
    globals()["height_sequence_1"] = final_seq
    return final_seq
