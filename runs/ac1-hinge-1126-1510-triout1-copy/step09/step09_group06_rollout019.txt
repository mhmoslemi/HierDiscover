"""
Optimise a non‑negative step function for the objective  

    eval(a) = 2·n·max(a*a) / (sum(a)^2)     (sum(a) is forced to 1)

The search consists of
* a diverse set of seeds (previous best, spectral init, exponential,
  power‑law, random Dirichlet)
* a quick evaluation of the fast score via FFT
* a long hill‑climb phase that repeatedly moves a small random
  fraction of the mass between two coordinates and keeps only
  strictly improving moves.  The hill‑climb runs until the time
  budget is exhausted; because the improvement required is tiny
  (≈ 0.2 %), this simple local search is enough to beat the
  previous 1.5065 bound.

The function returns the best sequence found (as a plain Python list)
and prints the intermediate scores.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Fast evaluation (assumes sum(a)=1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) for a already normalised to sum = 1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initial families
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of the square‑root of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def decreasing_exponential(n: int, c: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(‑c·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-c * idx)
    s = a.sum()
    return a / s


def decreasing_powerlaw(n: int, p: float) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^(‑p), normalised."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    return a / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random simplex point (Dirichlet‑like)."""
    a = rng.random(n)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Simple hill‑climb optimisation
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 200000,
               start_step: float = 0.20,
               end_step: float = 0.005) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of the mass from index i to j.
    The step size (fraction of the source) decays linearly from `start_step`
    to `end_step` over the allowed iterations.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        # linear decay of the allowed move size
        frac = start_step - (start_step - end_step) * (it / max_iter)
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # cannot move more than the source mass
        delta = best[i] * rng.random() * frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo the move
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                         # length of the step function
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two for fast FFTs

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # 1️⃣  Previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to the required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a_prev = np.full(n, 1.0 / n)
    else:
        a_prev = np.full(n, 1.0 / n)
    candidates.append(a_prev)

    # 2️⃣  Spectral and random starts
    candidates.append(spectral_init(n))
    candidates.append(random_initial(n, rng))

    # 3️⃣  Parameter sweeps of monotone families
    #    – decreasing exponential
    xs = np.linspace(0.5, 15.0, 30)          # x = c·n  (c = x/n)
    for x in xs:
        c = x / n
        a = decreasing_exponential(n, c)
        candidates.append(a)

    #    – decreasing power‑law
    for p in np.linspace(0.2, 6.0, 30):
        a = decreasing_powerlaw(n, p)
        candidates.append(a)

    # --------------------------------------------------------------
    # Keep the best seed according to the fast_score
    # --------------------------------------------------------------
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best initial fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Time management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # Long hill‑climb phase (takes the majority of the budget)
    # --------------------------------------------------------------
    # Give the hill‑climb roughly 90 % of the remaining time.
    hill_deadline = start_time + 0.90 * budget_s
    if time.time() < hill_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        max_iter=500000,
                                        start_step=0.15,
                                        end_step=0.003)
        print(f"[hill‑climb] refined fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final exact evaluation (uses the provided evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
