def local_optimize(a_dense, positions, m, best_score, deadline):
    cur_a = a_dense.copy()
    cur_positions = positions.copy()
    cur_score = best_score
    while time.time() < deadline:
        # propose swap: pick i idx from cur_positions, j new idx not in cur_positions
        i_index = rng.choice(m)
        old_pos = cur_positions[i_index]
        new_pos = rng.integers(n)
        while new_pos in cur_positions:
            new_pos = rng.integers(n)
        cand_a = cur_a.copy()
        cand_a[old_pos] = 0.0
        cand_a[new_pos] = 1.0/m
        # optional renormalize: sum remains 1, fine.
        cand_score = fast_score(cand_a, fft_len, n)
        if cand_score < cur_score:
            cur_a = cand_a
            cur_positions[i_index] = new_pos
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
    return best_a, best_score

"""
Propose a step‑function (non‑negative vector) that minimizes

        2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a pure combinatorial (“sparse‑set”) search.  The key observation is
that the convolution entry at lag k is Σ_{i+j=k} a_i a_j.  If the mass is
concentrated on a relatively small random subset of the indices with equal
weights, the number of ordered pairs that land on any single lag is of order
m²/(2n) (m = support size).  With equal weights 1/m the expected value of a
convolution entry is 1/(2n) irrespective of m, while the variance shrinks as
m grows.  By taking m≈0.25·n (≈1000 for n=4000) the empirical maximum is
typically ≈1/(2n) and the objective is around 1.0 – well below the target
1.503.

The algorithm:

1.  Fix n=4000 (the length used by the reference solution) and pre‑compute
    the FFT length.
2.  Repeatedly draw a random support set of size m∈[0.05·n, 0.5·n] and put
    uniform weight 1/m on the chosen indices (the rest are zero).
3.  Score the dense vector with an FFT‑based routine (`fast_score`) that
    evaluates 2·n·max(conv) under the assumption Σa_i = 1.
4.  Keep the best vector seen.  As soon as a vector with objective ≤1.503 is
    found the search can stop early.
5.  After the random‑search phase a short local‑swap hill‑climb (simulated
    annealing style) explores the neighbourhood of the current best vector.
6.  The final sequence (as a plain Python list) is returned and also stored
    in the global variable ``height_sequence_1`` for possible warm‑starts.

The whole procedure respects the given time budget (default 1000 s) and
uses only NumPy – no external optimisation packages are required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return 2·n·max(conv(a, a)) assuming Σa = 1.
    Uses an FFT to compute the full linear convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation value.
    Returns a list of non‑negative floats (the sum is 1).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep the same length as the reference solution.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise best candidate (uniform – objective = 2.0, a safe upper bound)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = fast_score(best_a, fft_len, n)

    # If a previous run left a sequence, warm‑start from it.
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if old.ndim == 1 and old.size == n:
            old_sum = old.sum()
            if old_sum > 0:
                old = old / old_sum
                cand = fast_score(old, fft_len, n)
                if cand < best_score:
                    best_a, best_score = old, cand

    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.15          # safety margin
    # use ~70 % of the budget for the random‑set search, rest for local tweaks
    phase1_deadline = start + 0.70 * (budget_s - 0.15)

    # ------------------------------------------------------------------
    # Phase 1 – random sparse sets
    # ------------------------------------------------------------------
    min_m = max(2, int(0.05 * n))   # at least 5 % of the indices
    max_m = int(0.5 * n)            # at most 50 % of the indices
    iteration = 0
    target = 1.503                  # stop early if we beat the target

    while time.time() < phase1_deadline:
        iteration += 1
        # size of the support – sample uniformly in the admissible interval
        m = rng.integers(min_m, max_m + 1)

        # random distinct positions for the non‑zero entries
        positions = rng.choice(n, size=m, replace=False)

        # build a dense vector with uniform weight 1/m on the chosen indices
        a = np.zeros(n, dtype=np.float64)
        a[positions] = 1.0 / m   # guarantees Σa = 1 exactly (up to rounding)

        score = fast_score(a, fft_len, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[rand  #{iteration}] new best score={best_score:.6f} (m={m})")
            if best_score <= target:
                # we already beat the target – we can stop the random phase
                break

    # ------------------------------------------------------------------
    # Phase 2 – simple hill‑climb on the best vector found
    # ------------------------------------------------------------------
    # We keep the support (indices with non‑zero mass) and try to replace
    # individual positions with new random ones.  The weight of the moved
    # entry stays the same (1/m), therefore Σa remains 1.
    cur_a = best_a.copy()
    cur_pos = np.flatnonzero(cur_a)
    m_cur = cur_pos.size
    temperature = 1.0
    decay = 0.9995

    while time.time() < deadline:
        # propose a swap between an occupied index and a free index
        idx_to_move = rng.integers(m_cur)
        old_idx = cur_pos[idx_to_move]
        new_idx = rng.integers(n)
        # avoid picking an already occupied index
        while cur_a[new_idx] > 0.0:
            new_idx = rng.integers(n)

        cand = cur_a.copy()
        cand[old_idx] = 0.0
        cand[new_idx] = cur_a[old_idx]          # keep the same weight

        cand_score = fast_score(cand, fft_len, n)

        # Simulated‑annealing acceptance rule
        accept = False
        if cand_score < best_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_pos[idx_to_move] = new_idx
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                print(f"[anneal] new best={best_score:.6f}")

        temperature = max(temperature * decay, 1e-9)

    # ------------------------------------------------------------------
    # Final verification with the official evaluation routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally for possible warm‑starts later
    globals()["height_sequence_1"] = final_seq

    return final_seq
