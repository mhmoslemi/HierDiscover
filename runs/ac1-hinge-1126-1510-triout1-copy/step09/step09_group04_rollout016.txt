def alternating_projection(init_a, target_fft_amp, L, n, iters):
    a = init_a.copy()
    for t in range(iters):
        # FFT
        A = np.fft.fft(a, L)
        # enforce magnitude
        A_mag = np.abs(A)
        # avoid divide by zero
        mask = A_mag > 0
        A[mask] = target_fft_amp[mask] * A[mask] / A_mag[mask]
        # else if magnitude zero, set to target amplitude with random phase?
        # set phase random maybe
        A[~mask] = target_fft_amp[~mask] * np.exp(1j * np.random.uniform(0, 2*np.pi, size=np.sum(~mask)))
        # inverse FFT
        a = np.real(np.fft.ifft(A))
        # enforce nonnegative and sum 1
        a[a < 0] = 0.0
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = init_a.copy()
    return a

"""
Propose a low‑score step‑function using a mixture of spectral
initialisation, alternating‑projection “phase‑retrieval”, L‑BFGS on a
smooth max‑convolution, exponentiated‑gradient sub‑gradient descent and a
final deterministic hill‑climb (with a short simulated‑annealing phase).

The algorithm works on the simplex (∑a = 1, a_i ≥ 0).  The evaluation
objective is
    2·n·max(conv(a,a)) / (∑a)²  = 2·n·max(conv(a,a))
so the sum constraint can be ignored after normalisation.
The code respects the 1000 s budget and prints intermediate scores.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

# Initialiser families
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector with several possible shapes."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Square‑root of a flat spectrum (exact solution of a*a = uniform, then clipped)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s > 0.0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a


def phase_retrieval_initial(
    n: int,
    rng: np.random.Generator,
    iters: int = 2000,
    fft_len: int | None = None,
) -> np.ndarray:
    """
    Alternating‑projection “phase‑retrieval” initialiser.
    The goal is to enforce a*a ≈ uniform while keeping a ≥ 0 and ∑a = 1.
    """
    if fft_len is None:
        fft_len = next_pow2(2 * n - 1)

    # target uniform convolution
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(fft_len)
    u_pad[:target_len] = uniform
    target_spec = np.fft.rfft(u_pad)               # length fft_len//2+1
    target_mag = np.sqrt(np.abs(target_spec))     # desired magnitude of A

    # start from a random probability vector
    a = random_initial(n, rng)

    for _ in range(iters):
        # FFT of zero‑padded a
        a_pad = np.zeros(fft_len)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        # enforce magnitude = target_mag, keep current phase
        mag = np.abs(A)
        zero = mag < 1e-12
        phase = np.angle(A)
        # for zero magnitude set a random phase (does not matter much)
        phase[zero] = rng.uniform(0.0, 2.0 * np.pi, size=np.sum(zero))
        A = target_mag * np.exp(1j * phase)

        # back‑to time domain
        a_new_pad = np.fft.irfft(A, fft_len)
        a_new = a_new_pad[:n]

        # project onto the simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a = a_new / s
        else:
            # degenerate – keep old a
            pass
    return a

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth approximation of max(conv(a,a)) where a = softmax(y).
    Returns value and gradient w.r.t. y.
    """
    a = softmax(y)

    # full convolution a*a via FFT (length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑prop to a (a_rev * w) using FFT
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation (coarse / refined)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS through a sequence of smoothing parameters.
    Returns the best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent via exponentiated gradient
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    a stays on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # classic 1/√t step‑size
        eta = 1.0 / np.sqrt(it + 1.0)

        # exponentiated update
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = best_a.copy()

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic fine‑grained hill‑climb (plus a tiny annealing phase)
# ----------------------------------------------------------------------


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    """
    Move a small fraction of mass from the “most harmful’’ coordinate
    to a coordinate with the smallest partner value.
    Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] where defined
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution to the maximal entry
        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: index with smallest partner value (skip i_max)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # renormalise (should stay 1 but guard against rounding)
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.8, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.8, min_step)
    return a, score


def simulated_annealing(a_start: np.ndarray,
                        init_score: float,
                        time_budget: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator):
    """
    Very lightweight annealing: random mass transfers with Metropolis
    acceptance.  Used only for a few seconds at the very end.
    """
    a = a_start.copy()
    score = init_score
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # move a fraction of mass from i to j
        delta = a[i] * 0.05  # 5 % of source mass
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum unchanged, no renorm needed
        new_score = fast_score(a_new, fft_len, n)
        # simple temperature schedule
        T = max(0.01, 0.5 * (1.0 - (time.time() - start) / time_budget))
        if new_score < score or rng.random() < np.exp(-(new_score - score) / T):
            a = a_new
            score = new_score
        it += 1
    return a, score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (increase n a little if time allows – 4000 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build initial candidates
    # ------------------------------------------------------------------
    candidates = [
        np.full(n, 1.0 / n),                 # uniform
        spectral_init(n),
        random_initial(n, rng),
        phase_retrieval_initial(n, rng, iters=1500, fft_len=fft_len_conv),
    ]

    # if a previous solution is available, interpolate it
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, prev)
            a_interp = np.maximum(a_interp, 0.0)
            if a_interp.sum() > 0:
                a_interp /= a_interp.sum()
                candidates.append(a_interp)

    # pick the best of the starters
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    phase1_deadline = start_time + 0.20 * budget_s   # coarse L‑BFGS
    phase2_deadline = start_time + 0.45 * budget_s   # refined L‑BFGS
    phase3_deadline = start_time + 0.80 * budget_s   # exponentiated‑gradient
    phase4_deadline = deadline                       # final polishing

    # ------------------------------------------------------------------
    # Phase 1 – many cheap L‑BFGS restarts (coarse smoothing)
    # ------------------------------------------------------------------
    lam_schedule_coarse = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005,
    ]
    maxiter_coarse = 600

    while time.time() < phase1_deadline:
        # choose a diverse start
        r = rng.random()
        if r < 0.5:
            a_start = random_initial(n, rng)
        elif r < 0.85:
            # small perturbation of current best
            a_start = a_start = a_start = a_start = None  # placeholder, will be replaced below
            a_start = best_a.copy()
            # add a tiny amount of Gaussian noise and re‑project
            sigma = rng.uniform(0.0005, 0.005)
            noisy = a_start + rng.normal(0.0, sigma, size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            a_start = noisy / s if s > 0 else best_a.copy()
        else:
            a_start = spectral_init(n)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule_coarse,
            maxiter_coarse,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase‑1] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined L‑BFGS (longer, finer lam schedule)
    # ------------------------------------------------------------------
    lam_schedule_fine = [
        0.4, 0.3, 0.2, 0.15, 0.1, 0.07,
        0.05, 0.03, 0.02, 0.015, 0.01,
        0.008, 0.005, 0.003, 0.0015,
        0.001, 0.0005, 0.0002, 0.0001,
        0.00005, 0.00002,
    ]
    maxiter_fine = 2500

    best_a, best_score = lbfgs_continuation(
        best_a,
        lam_schedule_fine,
        maxiter_fine,
        phase2_deadline,
        fft_len_conv,
        fft_len_g,
        n,
    )
    print(f"[Phase‑2] refined L‑BFGS score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent (long run)
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_deadline,
            n,
            fft_len_conv,
            fft_len_g,
            rng,
        )
        print(f"[Phase‑3] exp‑gradient score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – deterministic hill‑climb + short annealing
    # ------------------------------------------------------------------
    # deterministic hill‑climb
    best_a, best_score = fine_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase4_deadline,
        init_step=0.2,
        min_step=1e-5,
    )
    # a tiny simulated‑annealing burst (use remaining time)
    remaining = deadline - time.time()
    if remaining > 2.0:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            time_budget=remaining,
            n=n,
            fft_len=fft_len_conv,
            rng=rng,
        )
    print(f"[Phase‑4] final polished score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    return final_seq
