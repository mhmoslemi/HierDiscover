"""
Hybrid search for low‑score non‑negative step sequences.

The algorithm proceeds in four stages:
1.  Build a diverse pool of initial probability vectors (uniform,
    spectral, random Dirichlet‑like, triangular, Hann, Gaussian).
2.  Refine the current best vector using a *linear‑programming* step:
    for the current a compute M = max(a*a);
    solve   max Σ b_j  s.t.  a ◦ b ≤ M , b ≥ 0.
    (Here “a ◦ b” denotes the linear convolution a*b.)
    If the LP yields Σb > 1 we either replace a by the normalised b
    (when its own max‑convolution is smaller) or take a small convex
    combination a←(1‑t)a+tb.  Repeated LP updates drive the bound down.
3.  A short L‑BFGS polishing using a log‑sum‑exp smoothing of the ‑max
    (the same machinery as the previous state‑of‑the‑art solution).
4.  Final evaluation with the official `evaluate_sequence` routine.

All phases respect an overall 1000 s budget and keep the best sequence
found.  The core optimisation variable is a probability vector
(i.e. non‑negative and summing to 1) – the objective is scale invariant.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution / objective helpers (sum‑to‑1 vectors)
# ----------------------------------------------------------------------


def conv_max(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of the linear self‑convolution of a (sum(a)=1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * len(a) - 1]
    return float(conv.max())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Exact objective for a probability vector a:
        2·n·max(a*a)
    """
    return 2.0 * len(a) * conv_max(a, fft_len)


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (all return vectors summing to 1)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral “square‑root of uniform” initialisation."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of simple random families."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS polishing (exact same primitives as the reference code)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage
    from the result of the previous stage. Returns the best a found and
    its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> sparse.csc_matrix:
    """
    Construct the rectangular Toeplitz matrix A (shape (2n‑1)×n)
    such that A @ b = conv(f, b) (linear convolution, zero‑padding).
    Each diagonal i (0 ≤ i < n) contains the constant f[i].
    """
    n = f.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal is a vector of length n filled with f[i]
    data = [np.full(n, f_i) for f_i in f]
    A = sparse.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
    return A


def lp_max_sum(f: np.ndarray, M: float, timeout: float = 30.0) -> np.ndarray | None:
    """
    Solve   max Σ b_j   s.t.   f * b ≤ M ,   b ≥ 0 .
    Returns the optimal b (not normalised) or None on failure.
    """
    n = f.shape[0]
    A_ub = build_conv_matrix(f)
    b_ub = M * np.ones(2 * n - 1, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # minimise -Σb  → maximise Σb
    bounds = [(0.0, None)] * n

    # HiGHS (default) works well with sparse matrices.
    options = {"presolve": True, "time_limit": timeout}
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options=options)

    if not res.success:
        return None
    return res.x

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy code safety

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000                      # fixed, good trade‑off
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build an initial pool of candidates
    # ------------------------------------------------------------------
    candidates = []

    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # 1. spectral init
    candidates.append(spectral_init(n))

    # 2. three random draws
    for _ in range(3):
        candidates.append(random_initial(n, rng))

    # 3. classic windows
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(gaussian_init(n))

    # pick the best of the pool (according to the fast objective)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – LP‑based descent (most of the budget)
    # ------------------------------------------------------------------
    lp_deadline = start_time + 0.55 * budget_s
    iteration = 0
    while time.time() < lp_deadline:
        iteration += 1
        M = conv_max(best_a, fft_len)          # current max‑conv
        # Solve the LP: maximise Σ b_j  subject to  best_a * b ≤ M , b ≥ 0
        b_raw = lp_max_sum(best_a, M, timeout=15.0)
        if b_raw is None:
            print("[LP] iteration", iteration, "failed – terminating LP phase")
            break

        sum_b = b_raw.sum()
        # If the LP cannot increase the total mass, we are stuck.
        if sum_b <= 1.0 + 1e-12:
            print("[LP] iteration", iteration,
                  "no mass increase (sum_b={:.6f}) – stop".format(sum_b))
            break

        b = b_raw / sum_b                      # normalise to a probability vector

        # Check whether b alone already improves the max‑convolution.
        M_b = conv_max(b, fft_len)
        if M_b < M:
            best_a = b
            best_score = fast_score(best_a, fft_len)
            print("[LP] iteration", iteration,
                  "direct improvement: fast_score={:.8f}".format(best_score))
            continue

        # Otherwise take a small convex combination.
        t = 0.12                                 # step size (tuned empirically)
        improved = False
        for _ in range(5):                      # backtrack if needed
            a_candidate = (1.0 - t) * best_a + t * b
            M_cand = conv_max(a_candidate, fft_len)
            if M_cand < M:
                best_a = a_candidate
                best_score = fast_score(best_a, fft_len)
                print("[LP] iteration", iteration,
                      "mixed improvement (t={:.4f}) → fast_score={:.8f}".format(t, best_score))
                improved = True
                break
            t *= 0.5
        if not improved:
            # No improvement found – exit the LP loop.
            print("[LP] iteration", iteration,
                  "no improvement despite mixing – stop")

    lbfgs_deadline = start_time + 0.80 * budget_s

                                                lam_schedule,
                                                maxiter_per_lam,

        print(f"[L‑BFGS] post‑LP best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final hill‑climb (very cheap local polishing)
    # ------------------------------------------------------------------
    hill_deadline = deadline
    if time.time() < hill_deadline:
        def hill_climb(a, cur_score, fft_len, n, deadline, max_iter=20000):

            rng_local = np.random.default_rng(seed + 98765)
