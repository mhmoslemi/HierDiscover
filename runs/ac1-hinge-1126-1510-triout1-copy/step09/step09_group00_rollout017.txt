def solve_lp_max_sum(a, M, n):
    # Build sparse matrix A (2n-1 x n) as described.
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.repeat(a, n)
    A = scipy.sparse.coo_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Use linprog with method='highs' (supports sparse)
    bounds = [(0.0, 1000.0)] * n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        b = res.x
        return b
    else:
        return None

"""
search for a step‐function with a low value of

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm uses three ideas:

1. **Cheap parametric sweep** – a large collection of random families
   (exponential, power, cosine, binary‑sparse …) gives a decent starting
   point quickly.

2. **L‑BFGS on a smoothed max** – the true max of the convolution is
   approximated by a log‑sum‑exp; L‑BFGS is run on several values of the
   smoothing parameter to approach a local optimum.

3. **Linear‑programming refinement** – for a fixed vector `f` we solve

       maximise   Σ gₖ
       subject to  f * g ≤ M          (component‑wise)
                  gₖ ≥ 0 , gₖ ≤ 1000

   where `M` is the current maximal convolution of `f`.  The solution `g`
   has a larger ℓ₁‑norm while keeping the convolution bound, and we
   replace `f ← (1‑t) f + t g` (t≈0.5).  Repeating a few times reduces the
   objective dramatically.

The whole search runs for at most 1000 s (the supplied budget).  The best
sequence found is finally evaluated with the official ``evaluate_sequence``
and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def max_conv(a: np.ndarray, fft_len: int) -> float:
    """Maximum entry of a*a (real convolution)."""
    return float(full_conv(a, fft_len).max())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective **under the assumption Σa = 1**.
    Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def normalized_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score of a when we normalise it to sum 1 (identical to evaluate_sequence)."""
    s = a.sum()
    if s == 0.0:
        return np.inf
    a_norm = a / s
    return fast_score(a_norm, fft_len, n)


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# --------------------------------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# --------------------------------------------------------------
# Linear‑programming refinement
# --------------------------------------------------------------

def build_conv_matrix(a: np.ndarray, n: int) -> sparse.csr_matrix:
    """
    Build the (2n‑1) × n matrix A such that (A @ b)[k] = Σ_i a_i·b_{k‑i}
    (linear convolution of a with b).  `a` is assumed non‑negative.
    """
    # Every pair (i,j) contributes a_i to row i+j and column j.
    i_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    j_idx = np.tile(np.arange(n, dtype=np.int32), n)
    rows = i_idx + j_idx                     # 0 … 2n‑2
    data = np.repeat(a, n)                  # a_i repeated n times
    A = sparse.coo_matrix((data, (rows, j_idx)), shape=(2 * n - 1, n))
    return A.tocsr()


def lp_max_sum(a: np.ndarray, M: float, n: int) -> np.ndarray | None:
    """
    Solve   maximise Σ b_i   subject to   a * b ≤ M , 0 ≤ b_i ≤ 1000 .
    Returns the optimal b (as a dense ndarray) or None on failure.
    """
    A = build_conv_matrix(a, n)                     # (2n‑1) × n
    ub = np.full(2 * n - 1, M)                      # upper bound for each convolution entry
    c = -np.ones(n)                                 # maximise Σb ↔ minimise –Σb
    bounds = [(0.0, 1000.0)] * n

    # scipy.linprog understands sparse CSR matrices for the “highs” method
    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if res.success:
        return res.x
    return None


def lp_refinement(a_start: np.ndarray, n: int, fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and mix the solution back
    into the current vector.  The mixing factor t is fixed to 0.5.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = normalized_score(a, fft_len, n)

    while time.time() < deadline:
        M = max_conv(a, fft_len)
        b = lp_max_sum(a, M, n)
        if b is None:
            break

        # convex combination, keep the vector in the feasible box
        t = 0.5
        a = (1.0 - t) * a + t * b
        a = np.clip(a, 0.0, 1000.0)

        cur_score = normalized_score(a, fft_len, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[lp] new best fast_score={best_score:.6f}")
    return best_a, best_score

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lambda in ``lam_schedule``, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # use logits to stay on the simplex

    best_a = a_start.copy()
    best_score = normalized_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = normalized_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = normalized_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true max‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        step = 0.6 / (it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = normalized_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def exp_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = normalized_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.5 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = normalized_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
            print(f"[exp‑grad] new best fast_score={best_score:.6f}")

        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Slower square‑root cooling hill‑climb that transfers a random fraction
    of mass between two coordinates.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.05 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = normalized_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Explore a collection of random parametric families and keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_score = normalized_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10          # 10 % of total budget

    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = normalized_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function

    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = normalized_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # Re‑use a previously stored sequence if present
    # --------------------------------------------------------------

        hs = np.array(globals()["height_sequence_1"], dtype=float)

            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = normalized_score(a1, fft_len, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] parametric sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2     # safety margin

    # allocate time slices
    phase1_end = start_time + 0.40 * budget_s   # more time for L‑BFGS
    phase2_end = start_time + 0.65 * budget_s   # LP refinement
    phase3_end = start_time + 0.85 * budget_s   # sub‑gradient / exp‑gradient
    # remaining time → annealing

    # --------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS with a finer schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # occasional jitter around the current best
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – linear‑programming refinement
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        print("[phase2] starting LP refinement")
        a_refined, sc_refined = lp_refinement(
            best_a,
            n,
            fft_len,
            phase2_end,
        )
        if sc_refined < best_score:
            best_score, best_a = sc_refined, a_refined.copy()
            print(f"[phase2] LP improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient + exponentiated‑gradient
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase3_end, fft_len, fft_len_g, n
        )
        print(f"[phase3] sub‑gradient best fast_score={best_score:.6f}")

        # a short run of exponentiated‑gradient for fine‑tuning
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len, fft_len_g, n
        )
        print(f"[phase3] exp‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – slow‑cooling simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_score = normalized_score(best_a, fft_len, n)
        best_a, best_score = simulated_annealing(
            best_a,
            final_score,
            fft_len,
            n,
            deadline,
            rng,
        )
        print(f"[phase4] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official ``evaluate_sequence``
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
