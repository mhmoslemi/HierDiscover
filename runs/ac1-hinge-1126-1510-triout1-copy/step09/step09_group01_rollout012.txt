def refine_subgradient_line_search(a, deadline):
    best = a ; best_score = fast_score(a)
    while time.time() < deadline:
        # compute conv max and mask
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()

"""
search for low evaluation step sequence using large n, multi‑phase
optimisation (parametric sweep → Adam on smoothed max → projected
sub‑gradient → exponentiated‑gradient → simulated annealing → fine
line‑search refinement).  The algorithm works on the probability simplex
(∑a=1) so that the provided ``evaluate_sequence`` returns
``2·n·max(conv(a,a))`` directly.  The length is set to 8000 for a
better asymptotic bound; all phases respect the global deadline.
"""
import time
import numpy as np

def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 3000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(80.0, 600.0)
        sigma2 = rng.uniform(80.0, 600.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:  # "random"
        a = rng.random(n)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def parametric_sweep(n: int, fft_len: int, budget_s: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap search over several parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.15
    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score

def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float) -> tuple[float, np.ndarray]:
    """Smoothed max of convolution using log‑sum‑exp; return value and gradient w.r.t. y."""
    a = softmax(y)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter proportional to current max
    lam = max_c * lam_factor + 1e-14
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    w = exp_shift / Z          # ∂smooth_max/∂conv_i
    # back‑propagation through convolution: gradient wrt a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = conv_rev[(n - 1) : (n - 1 + n)]   # length n

    # chain rule through softmax
    mean = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean)
    return f_obj, grad_y

def adam_optimize(a_start: np.ndarray, lam_factors, iters_per_lam: int,
                  deadline: float, fft_len_conv: int, fft_len_g: int, n: int
                  ) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smoothed objective."""
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            # Adam updates
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.02 / (np.sqrt(v_hat) + eps)
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score

def subgradient_optimize_momentum(a0: np.ndarray, deadline: float,
                                 fft_len_conv: int, fft_len_g: int,
                                 n: int) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum on true max."""
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score

def exp_gradient_optimize(a0: np.ndarray, deadline: float,
                         fft_len_conv: int, n: int) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent (mirror descent on the simplex)."""
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
        it += 1
    return best_a, best_score

def simulated_annealing(a_start: np.ndarray, start_score: float,
                        fft_len: int, n: int, deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple hill‑climb with linearly decreasing temperature."""
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remaining = max(deadline - time.time(), 0.0)
        T = T0 * (remaining / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score

def refine_local(a_start: np.ndarray, deadline: float,
                 fft_len: int, n: int) -> tuple[np.ndarray, float]:
    """
    Fine‑grained local refinement using a sub‑gradient direction
    and a simple back‑tracking line‑search.
    """
    a = a_start.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len, n)

    while time.time() < deadline:
        # compute sub‑gradient of true max
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # search direction is opposite to the sub‑gradient
        d = -grad

        # line search with projection
        alpha = 0.5
        improved = False
        while alpha > 1e-12:
            a_new = simplex_projection(a + alpha * d)
            sc = fast_score(a_new, fft_len, n)
            if sc < best_score - 1e-12:
                best_score, best_a = sc, a_new.copy()
                a = a_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            # no progress – exit early
            break
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimisation that searches a long non‑negative step
    sequence.  Returns a list of floats (the best sequence found
    before the deadline).  The routine respects the 1000 s limit and
    prints progress.
    """
    rng = np.random.default_rng(seed)

    # problem size – larger n gives a better asymptotic bound
    n = 8000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # baseline uniform (for reference)
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # initialise from a previous global solution if available
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    # Phase 0 – cheap parametric sweep (≈15 % of budget)
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[sweep] parametric best fast_score={best_score:.6f}")

    # Time slots for subsequent phases
    t0 = time.time()
    phase1_end = start_time + 0.45 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.85 * budget_s
    phase4_end = start_time + 0.96 * budget_s

    # Phase 1 – Adam on smoothed max
    lam_factors = np.geomspace(0.7, 2e-6, num=30)
    iters_per_lam = 1200   # generous per‑lam budget
    while time.time() < phase1_end:
        # initialise restart (30 % jitter, 70 % fresh)
        if rng.random() < 0.3:
            a_start = best_a * rng.normal(1.0, 0.015, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = adam_optimize(
            a_start, lam_factors, iters_per_lam,
            phase1_end, fft_len_conv, fft_len_g, n
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score={best_score:.6f}")

    # Phase 2 – projected sub‑gradient with momentum (true max)
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] best fast_score={best_score:.6f}")

    # Phase 3 – exponentiated‑gradient descent (mirror)
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, fft_len_conv, n
        )
        print(f"[exp‑gradient] best fast_score={best_score:.6f}")

    # Phase 4 – simulated annealing
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, phase4_end, rng
        )
        print(f"[annealing] best fast_score={best_score:.6f}")

    # Final fine‑grained refinement (line‑search)
    if time.time() < deadline:
        best_a, best_score = refine_local(
            best_a, deadline, fft_len_conv, n
        )
        print(f"[refine] best fast_score={best_score:.6f}")

    # Evaluate with the provided function (will clip/normalise automatically)
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq

"""
Multi‑phase optimisation for a low‑score step sequence.

The routine works on the probability simplex (∑a = 1) so that the
provided ``evaluate_sequence`` returns ``2·n·max(conv(a,a))`` directly.
A relatively large size (n = 8000) gives a better asymptotic bound.
The search consists of

0. cheap parametric sweep,
1. Adam optimiser on a log‑sum‑exp smooth approximation of the max,
2. projected sub‑gradient descent with momentum (true max),
3. exponentiated‑gradient (mirror descent on the simplex),
4. simulated‑annealing hill‑climb,
5. a final fine‑grained line‑search refinement.

All phases respect a global deadline (default 1000 s).  The best
vector found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility primitives
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returns the first 2·len(a)‑1 entries."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector already normalised to ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simple parametric families (Phase 0)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a simple parametric family."""
    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        if inc:
            a = np.arange(1, n + 1, dtype=float) ** p
        else:
            a = np.arange(n, 0, -1, dtype=float) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 3000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (np.arange(n, dtype=float) - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(80.0, 600.0)
        sigma2 = rng.uniform(80.0, 600.0)
        a1 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((np.arange(n, dtype=float) - mu2) / sigma2) ** 2)
        a = a1 + a2
    else:                                # "random"
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(n: int, fft_len: int, budget_s: float,
                     rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Cheap random sweep over a collection of families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    deadline = time.time() + budget_s * 0.15
    families = ["power", "exp", "gauss", "cos", "bimodal", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed max objective (used by Adam)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the convolution a*a where a = softmax(y).
    Returns the objective value and its gradient w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-14
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_obj = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # back‑propagation through convolution (gradient w.r.t. a)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_grad = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_grad)

    return f_obj, grad_y

# Optimisers

def adam_optimize(a_start: np.ndarray, lam_factors, iters_per_lam: int,
                  deadline: float, fft_len_conv: int, fft_len_g: int, n: int
                  ) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smoothed max objective."""
    y = np.log(a_start + 1e-12)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_factors:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                break
            t += 1
            f_val, grad = objective_and_grad(y, n, fft_len_conv,
                                             fft_len_g, lam)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.02 / (np.sqrt(v_hat) + eps)
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray, deadline: float,
                                 fft_len_conv: int, fft_len_g: int,
                                 n: int) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with momentum on the true max."""
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        sc = fast_score(a_candidate, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score


def exp_gradient_optimize(a0: np.ndarray, deadline: float,
                         fft_len_conv: int, n: int) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent (mirror descent on the simplex)."""
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len_conv, n)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        eta = 0.25 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
        it += 1
    return best_a, best_score


def simulated_annealing(a_start: np.ndarray, start_score: float,
                        fft_len: int, n: int, deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb respecting mass conservation."""
    a = a_start.copy()
    best_a, best_score = a.copy(), start_score
    cur_score = start_score

    T0 = 0.02 * start_score
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        remaining = max(deadline - time.time(), 0.0)
        T = T0 * (remaining / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] == 0.0:
            continue
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
    return best_a, best_score


def refine_local(a_start: np.ndarray, deadline: float,
                 fft_len: int, n: int) -> tuple[np.ndarray, float]:
    """
    Fine‑grained refinement using a back‑tracking line search along the
    sub‑gradient direction of the true max‑convolution.
    """
    a = a_start.copy()
    best_a, best_score = a.copy(), fast_score(a, fft_len, n)

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # move opposite to the sub‑gradient
        d = -grad
        alpha = 0.5
        improved = False
        while alpha > 1e-12:
            a_new = simplex_projection(a + alpha * d)
            sc = fast_score(a_new, fft_len, n)
            if sc < best_score - 1e-12:
                best_score, best_a = sc, a_new.copy()
                a = a_new
                improved = True
                break
            alpha *= 0.5
        if not improved:
            break
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    best_a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------

    if "height_sequence_1" in globals():

            sc1 = fast_score(a1, fft_len_conv, n)
