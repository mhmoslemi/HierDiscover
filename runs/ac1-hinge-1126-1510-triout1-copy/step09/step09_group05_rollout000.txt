"""
Multi‑phase optimizer for the step‑function evaluation problem.

The algorithm consists of four stages within the allotted time budget:

1. **Beta‑shape grid search** – a one‑parameter symmetric family
   a_i ∝ ((i+1)*(n-i))^p is scanned (coarse → fine) to obtain a strong
   starting point.

2. **Heavy L‑BFGS** – a continuation scheme over a log‑sum‑exp
   smooth approximation of the max‑convolution, with many random
   restarts. After each L‑BFGS run the vector is symmetrised and
   projected onto the simplex.

3. **Projected Adam** – sub‑gradient descent on the exact nonsmooth
   objective using the true max‑convolution sub‑gradient; symmetry is
   enforced after each update.

4. **Simulated‑annealing hill‑climb** – repeated transfers of mass from
   high‑gradient to low‑gradient indices, accepting occasional uphill
   moves. The temperature decays exponentially.

All stages respect the total time budget (default 1000 s).  The best
sequence found is returned as a plain Python list and stored in the
global variable ``height_sequence_1`` for warm‑starts in future runs.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Force a symmetric vector (a_i == a_{n-1-i})."""
    return (a + a[::-1]) / 2.0


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) assuming sum(a)=1.
    Uses FFT for the linear convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Exact max‑convolution value and a sub‑gradient for a·a.
    Returns (max_val, g) where g has size n.
    """
    n = a.shape[0]
    # full convolution a * a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed surrogate (log‑sum‑exp) of the max‑convolution.
    The variables `y` are logits; a = softmax(y) satisfies sum(a)=1.
    Returns (objective, gradient w.r.t. y).
    """
    # softmax, numerically stable
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  The vector is kept on the simplex
    (via softmax) and symmetrised after each λ.
    Returns the best vector and its fast_score.
    """
    # log‑odds (logits)
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=lambda y_, *args: objective_and_grad(
                    y_, n, fft_len_conv, fft_len_g, lam
                ),
                x0=y,
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            # anything unexpected – skip this λ
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        a = symmetrize(a)               # enforce symmetry
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} new best={best_score:.6f}")
    return best_a, best_score


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the exact sub‑gradient of max‑convolution.
    Symmetry is enforced after each update.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        _, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = symmetrize(a)
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} best={best_score:.6f}")

        if it % 2000 == 0:
            # jitter to escape flat regions
            sigma = rng.uniform(0.001, 0.005)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = symmetrize(a)
            a = simplex_projection(a)

    return best_a, best_score


def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive hill‑climb with simulated‑annealing acceptance.
    Transfers mass from high‑gradient to low‑gradient entries.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.01 * cur_score, 1e-6)
    decay = 0.9995
    min_temp = 1e-9
    it = 0

    while time.time() < deadline and it < max_iters:
        # exact gradient for the current iterate
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # pick source (high) and target (low) indices
        high_thr = np.percentile(grad, 90)
        low_thr = np.percentile(grad, 10)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i != j:
                frac = rng.uniform(0.08, 0.22)   # proportion of source mass
                delta = cur_a[i] * frac
                cand = cur_a.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            # simple random swap (helps reshaping)
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # enforce feasibility
        cand = np.maximum(cand, 0.0)
        cand = symmetrize(cand)
        cand = simplex_projection(cand)

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score


def beta_shape(p: float, n: int) -> np.ndarray:
    """
    Symmetric beta‑type family: a_i ∝ ((i+1)*(n-i))^p,
    i = 0,…,n‑1.  Returns a normalized probability vector.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    log_factor = np.log(idx) + np.log(n + 1 - idx)     # log((i+1)*(n-i+1))
    # avoid overflow / underflow
    a = np.exp(p * log_factor)
    s = a.sum()
    if s == 0 or not np.isfinite(s):
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``2 * n * max(conv(a,a)) / (sum(a)**2)``.
    Returns the best vector (as a plain list) found within the given
    time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # problem size – a modest length keeps the FFT cheap
    n = 4000

    # initialise from a previously saved sequence if possible
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to current size
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 = symmetrize(a0)
            a0 = simplex_projection(a0)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2   # tiny safety margin

    # fraction of time allocated to each phase
    t_phase_a = start + 0.07 * (budget_s - 0.2)          # beta‑shape search
    t_phase_b = start + 0.27 * (budget_s - 0.2)          # L‑BFGS
    t_phase_c = start + 0.60 * (budget_s - 0.2)          # Adam
    t_phase_d = total_deadline                           # annealing / polish

    # ------------------------------------------------------------------
    # Phase A – quick exploration of the symmetric beta family
    # ------------------------------------------------------------------
    # coarse grid
    p_vals = np.linspace(-2.0, 2.0, 81)
    for p in p_vals:
        a = beta_shape(p, n)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[beta] p={p:.3f} new best={best_score:.6f}")

    # fine refinement around the best p
    p_best = p_vals[np.argmin([fast_score(beta_shape(p, n), fft_len_conv, n) for p in p_vals])]
    step = 0.05
    for _ in range(4):
        p_grid = np.linspace(p_best - step, p_best + step, 41)
        for p in p_grid:
            a = beta_shape(p, n)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                p_best = p
                print(f"[beta‑refine] p={p:.5f} new best={best_score:.6f}")
        step *= 0.3

    # ------------------------------------------------------------------
    # Phase B – heavy L‑BFGS with random restarts
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        0.0005, 0.0002, 0.0001, 5e-05, 1e-05,
    ]
    maxiter_per_lam = 250

    while time.time() < t_phase_b:
        # occasional diversification
        if rng.random() < 0.35:
            # random initial point on the simplex
            rand = rng.random(n)
            rand = np.maximum(rand, 0.0)
            a_start = symmetrize(rand)
            a_start = simplex_projection(a_start)
        else:
            a_start = best_a.copy()

        a_opt, scr_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            t_phase_b,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if scr_opt < best_score:
            best_score = scr_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS restart] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase C – projected Adam on the exact nonsmooth objective
    # ------------------------------------------------------------------
    if time.time() < t_phase_c:
        best_a, best_score = subgradient_adam(
            best_a, t_phase_c, fft_len_conv, fft_len_g, n
        )
        print(f"[Adam] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase D – simulated‑annealing hill‑climb + final polishing
    # ------------------------------------------------------------------
    if time.time() < t_phase_d:
        best_a, best_score = annealing_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            t_phase_d,
        )
        print(f"[Anneal] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation (to report the true objective)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose for potential warm‑starts in a later run
    globals()["height_sequence_1"] = final_seq

    return final_seq
