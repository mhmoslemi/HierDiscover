def optimal_delta_hill_climb(a, current_score, fft_len, n, deadline, min_delta=1e-12):
    best_score = current_score
    while time.time() < deadline:
        # compute conv and partner for current max
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n - 1]
        max_idx = np.argmax(conv)
        # compute partner array
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)
        # pick i_min as index with smallest partner (not same as i_max)
        sorted_by_partner = np.argsort(partner)
        # ensure i_min distinct from i_max
        if sorted_by_partner[0] != i_max:
            i_min = sorted_by_partner[0]
        else:
            i_min = sorted_by_partner[1]

        p_max = partner[i_max]
        p_min = partner[i_min]
        c_max = contrib[i_max]
        c_min = contrib[i_min]

        if p_max + p_min <= 0:
            # cannot improve by moving mass between these indices
            break

        delta = (c_max - c_min) / (p_max + p_min)
        if delta <= 0:
            # nothing to move
            break

        # bound delta by available mass at i_max
        delta = min(delta, a[i_max])

        # prevent extremely tiny steps
        if delta < min_delta:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # renorm to sum 1 (renorm may be needed if numerical errors)
        s = a_new.sum()
        if s <= 0:
            break
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            a = a_new
            best_score = new_score
            # continue near this new state
        else:
            # fallback: Try smaller delta (reduce by factor 0.5) up to few attempts
            improved = False
            for factor in (0.5, 0.25, 0.125):
                d2 = delta * factor
                if d2 < min_delta:
                    break
                a2 = a.copy()
                a2[i_max] -= d2
                a2[i_min] += d2
                a2 /= a2.sum()
                sc2 = fast_score(a2, fft_len, n)
                if sc2 < best_score:
                    a = a2
                    best_score = sc2
                    improved = True
                    break
            if not improved:
                # if still not improved, maybe pick second-best i_min or i_max
                # attempt to pick alternative low partner index
                # Use second smallest partner
                if len(sorted_by_partner) > 2:
                    i_alt = sorted_by_partner[1] if sorted_by_partner[1] != i_max else sorted_by_partner[2]
                    # try moving to i_alt
                    p_alt = partner[i_alt]
                    # compute delta for i_max->i_alt using same formula
                    delta_alt = (c_max - (a[i_alt] * p_alt)) / (p_max + p_alt)
                    delta_alt = min(delta_alt, a[i_max])
                    if delta_alt > min_delta:
                        a_alt = a.copy()
                        a_alt[i_max] -= delta_alt
                        a_alt[i_alt] += delta_alt
                        a_alt /= a_alt.sum()
                        sc_alt = fast_score(a_alt, fft_len, n)
                        if sc_alt < best_score:
                            a = a_alt
                            best_score = sc_alt
                            continue
                # else break out to avoid endless loops
                break
    return a, best_score

"""
Hybrid optimizer for the self‑convolution bound.

The objective is
    2 * n * max(conv(a, a)) / (sum(a)^2)

with a_i ≥ 0.  The algorithm works in three stages:

1.  Populate a pool of diverse initial vectors (including the previous best
    sequence if available) and keep the best by a fast FFT‑based score.
2.  Long‑run exponentiated‑gradient mirror descent on the exact max‑convolution
    (sub‑gradient) – this is cheap, uses only FFTs and quickly pushes the bound
    down.
3.  An aggressive “delta‑hill‑climb”:  at each iteration we move the optimal
    amount of mass from the coordinate that contributes most to the current
    maximal convolution entry to the coordinate with the smallest partner
    value.  The move size is chosen analytically so that the two contributions
    become equal; a back‑off line‑search guarantees monotonic improvement.
4.  A short random‑perturbation hill‑climb to escape tiny local minima.

All stages respect the simplex (∑a = 1, a ≥ 0) and use only numpy / scipy.
The final sequence is returned as a plain Python list; the provided
`evaluate_sequence` function is called once at the end for the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector using several heuristics."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Generate a start vector from a flat spectrum (square‑rooted)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Smoothed max of self‑convolution (log‑sum‑exp). Returns f, grad_y."""
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """Run L‑BFGS through a sequence of smoothing parameters."""
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient descent via exponentiated‑gradient
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Mirror descent with KL‑projection (exponentiated‑gradient) on the true max‑conv.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # classic 1/√t step size, with a small scaling factor
        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Aggressive delta hill‑climb (deterministic)
# ----------------------------------------------------------------------


def optimal_delta_hill_climb(a: np.ndarray,
                             current_score: float,
                             fft_len: int,
                             n: int,
                             deadline: float,
                             min_delta: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Repeatedly move the optimal amount of mass from the coordinate that
    contributes most to the maximal convolution entry toward the coordinate
    with the smallest partner value.  The step size is chosen analytically
    so that the two contributions become equal; a back‑off line‑search
    guarantees monotonic improvement.
    """
    best_score = current_score
    while time.time() < deadline:
        # convolution and critical index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        # contribution of each coordinate to the maximal entry
        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: smallest partner value (avoid picking i_max itself)
        sorted_by_partner = np.argsort(partner)
        i_min = sorted_by_partner[0] if sorted_by_partner[0] != i_max else sorted_by_partner[1]

        p_max = partner[i_max]
        p_min = partner[i_min]
        c_max = contrib[i_max]
        c_min = contrib[i_min]

        if p_max + p_min <= 0:
            break

        # Analytic delta that equalises the two contributions:
        delta = (c_max - c_min) / (p_max + p_min)
        delta = min(delta, a[i_max])          # cannot take more than we have
        if delta <= min_delta:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            a = a_new
            best_score = new_score
            continue

        # back‑off line‑search (geometric)
        improved = False
        for factor in (0.5, 0.25, 0.125):
            d2 = delta * factor
            if d2 <= min_delta:
                break
            a_tmp = a.copy()
            a_tmp[i_max] -= d2
            a_tmp[i_min] += d2
            a_tmp /= a_tmp.sum()
            sc_tmp = fast_score(a_tmp, fft_len, n)
            if sc_tmp < best_score:
                a = a_tmp
                best_score = sc_tmp
                improved = True
                break
        if not improved:
            # try an alternative low‑partner index (second best)
            if len(sorted_by_partner) > 2:
                alt = sorted_by_partner[1] if sorted_by_partner[1] != i_max else sorted_by_partner[2]
                p_alt = partner[alt]
                delta_alt = (c_max - a[alt] * p_alt) / (p_max + p_alt)
                delta_alt = min(delta_alt, a[i_max])
                if delta_alt > min_delta:
                    a_alt = a.copy()
                    a_alt[i_max] -= delta_alt
                    a_alt[alt] += delta_alt
                    a_alt /= a_alt.sum()
                    sc_alt = fast_score(a_alt, fft_len, n)
                    if sc_alt < best_score:
                        a = a_alt
                        best_score = sc_alt
                        continue
            # no improving move found – exit
            break
    return a, best_score

def random_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      step_frac: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Randomly move a small fraction of mass from one coordinate to another.
    Accept only improving moves.  Simple stochastic local search.

    best_score = score
    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)

        delta = a[i] * step_frac

        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:

            best_score = sc
