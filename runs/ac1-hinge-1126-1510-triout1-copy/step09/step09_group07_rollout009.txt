"""
Propose a low‑score non‑negative step sequence.

Algorithm
---------
1. Build a rich pool of initial shapes (random Dirichlet draws,
   classic windows, and many parametrised families such as Kaiser,
   Blackman, Tukey, Gaussian, inverse‑power, etc.).  All are normalised
   to sum = 1 and evaluated with the fast FFT‑based score
   `2·n·max(conv(a,a))`.  The best few seeds are retained.
2. Phase 1 – many short L‑BFGS polishing runs on a smoothed
   max‑objective (log‑sum‑exp).  Each run starts from a random
   mixture of the current best seed and a fresh random vector.
3. Phase 2 – long exponentiated‑gradient descent on the exact
   max‑convolution (the same descent used in the reference solution).
4. Phase 3 – a final high‑precision L‑BFGS continuation with a very
   small smoothing parameter, followed by a greedy hill‑climb that
   moves tiny amounts of mass between two coordinates only if the
   objective improves.
5. Return the best sequence discovered; `evaluate_sequence` will
   re‑scale it and compute the official score.

All steps respect the overall 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import i0
from scipy.signal import tukey

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast evaluation (sum=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Return the exact objective for a probability vector a (sum=1):
        2·n·max(a*a)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    # scale to [-1,1]
    x = (2.0 * i) / (n - 1) - 1.0
    a = i0(beta * np.sqrt(1 - x * x)) / i0(beta)
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a


def tukey_init(n: int, alpha: float) -> np.ndarray:

    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

    left = (i + 1) ** (-p)
    right = (n - i) ** (-p)

    sigma = sigma_factor * n
    offset = separation_frac * n / 2.0
    a = np.exp(-0.5 * ((i - centre - offset) / sigma) ** 2)
    a += np.exp(-0.5 * ((i - centre + offset) / sigma) ** 2)
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          rng: np.random.Generator):
    """
    Exponentiated‑gradient descent on the exact max‑convolution.
    The update a ← a·exp(−η·g) keeps a on the simplex automatically.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        fm = np.fft.rfft(mask, fft_len_g)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 20000,
               delta_frac: float = 0.08):
    """
    Simple greedy hill‑climb: transfer a small random fraction of mass
    from i to j and keep the change only if the score improves.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # for any legacy code that uses np.random

    # --------------------------------------------------------------
    # Problem size (a modest n keeps FFT cheap while giving sufficient freedom)
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool (including many parametrised windows)
    # --------------------------------------------------------------
    candidates = []

    # include previous best if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # classical/hand‑crafted initialisers
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(gaussian_init(n, sigma_factor=0.25))
    candidates.append(blackman_init(n))

    # parameter sweeps over popular windows
    for beta in np.linspace(0.0, 12.0, 7):
        candidates.append(kaiser_init(n, beta))

    for alpha in np.linspace(0.0, 1.0, 6):
        candidates.append(tukey_init(n, alpha))

    for sigma_factor in np.linspace(0.08, 0.35, 6):
        candidates.append(gaussian_init(n, sigma_factor))

    for rate in [0.001, 0.003, 0.008, 0.02, 0.05]:
        candidates.append(exp_decay_init(n, rate))

    for p in np.linspace(0.2, 3.0, 7):
        candidates.append(power_law_init(n, p))

    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        candidates.append(inv_power_init(n, p))

    for p in [1, 2, 3]:
        for sigma_factor in [0.1, 0.2, 0.3]:
            candidates.append(generalized_gaussian_init(n, p, sigma_factor))

    for sep in [0.2, 0.4, 0.6]:
        candidates.append(bimodal_gaussian_init(n,
                                                sigma_factor=0.2,
                                                separation_frac=sep))

    # many random Dirichlet‑like draws
    for _ in range(30):
        candidates.append(random_initial(n, rng))

    # --------------------------------------------------------------
    # Keep the best few seeds (top 5)
    # --------------------------------------------------------------
    scores = [fast_score(c, fft_len, n) for c in candidates]
    top_idx = np.argsort(scores)[:5]
    seed_pool = [candidates[i].copy() for i in top_idx]
    best_a = seed_pool[0].copy()
    best_score = scores[top_idx[0]]
    print(f"[init] n={n}, best initial fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – multiple short L‑BFGS runs from diverse starts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06
    ]
    maxiter_per_lam = 800  # a bit cheaper per run – we will do many restarts

    while time.time() < phase1_end:
        # pick a seed from the pool, perturb it with a random blend
        base = rng.choice(seed_pool)
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.5
        a_start = (1 - alpha) * base + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            # keep pool up‑to‑date (replace worst of pool)
            worst_idx = np.argmax([fast_score(c, fft_len, n) for c in seed_pool])
            seed_pool[worst_idx] = best_a.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent on the exact max
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len, fft_len,
                                                   rng)
        print(f"[exp‑grad] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final high‑precision L‑BFGS polishing (very small λ)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:   # leave a few seconds for hill‑climb
        final_lam_schedule = [5e-07, 2e-07, 1e-07, 5e-08, 2e-08]
        a_opt, sc_opt = lbfgs_continuation(best_a,
                                           final_lam_schedule,
                                           maxiter_per_lam=1200,
                                           deadline=deadline - 5.0,
                                           fft_len_conv=fft_len,
                                           fft_len_g=fft_len,
                                           n=n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[final L‑BFGS] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – greedy hill‑climb for a final polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=40000,
                                        delta_frac=0.06)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (evaluate_sequence will rescale internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # final official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
