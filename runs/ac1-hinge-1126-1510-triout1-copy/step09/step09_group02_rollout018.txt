def beta_initial(n, rng):
    alpha = rng.uniform(0.1, 2.0)   # small for U shape, large for convex shape
    x = np.linspace(0.5/n, 1-0.5/n, n)  # use points
    # compute unnormalized = x**(alpha-1) * (1-x)**(alpha-1)
    unnorm = np.exp((alpha-1)*np.log(x) + (alpha-1)*np.log(1-x))
    # for alpha<1, values near edges become large but still finite.
    a = unnorm
    a = np.maximum(a, 0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum
    return a
```

def simulated_annealing_hill_climb(a, curr_score, fft_len, n, deadline, rng,
                                   start_temp=0.02, end_temp=1e-5,
                                   start_frac=0.3, end_frac=0.001,
                                   recompute_every=1500, phase_start=None):
    if phase_start is None:
        phase_start = time.time()
    best_a = a.copy()
    best_score = curr_score
    # compute conv using FFT
    conv = full_conv(a, fft_len)
    # compute max and score for reliability:
    curr_max = conv.max()
    # Not using fast_score but compute 2*n*curr_max
    curr_score = 2 * n * curr_max
    it = 0
    accepted = 0
    while time.time() < deadline:
        now = time.time()
        progress = (now - phase_start) / (deadline - phase_start)
        # linear schedule:
        temp = start_temp * (1 - progress) + end_temp * progress
        max_frac = start_frac * (1 - progress) + end_frac * progress

        # source weighted by mass
        src = rng.choice(n, p=a)  # a sums to 1
        # ensure a[src]>0
        if a[src] <= 0:
            continue
        # dest not equal src; choose uniform
        dst = rng.integers(0, n)
        if dst == src:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0:
            continue

        # incremental update
        diff = np.zeros_like(conv)
        diff[src:src+n] -= a
        diff[dst:dst+n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < curr_score or rng.random() < np.exp(-(new_score - curr_score) / max(temp, 1e-12)):
            # accept
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # recompute conv from scratch periodically for numeric stability
        if it % recompute_every == 0:
            conv = full_conv(a, fft_len)
            curr_score = 2 * n * conv.max()
        it += 1
    return best_a, best_score

"""
Propose a non‑negative step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser has four stages, all respecting a 1000 s wall‑clock budget:

1. **L‑BFGS on a log‑sum‑exp smoothed max** – many short restarts from
   diverse random initialisations (including U‑shaped beta PDFs) explore
   the basin of low values.

2. **Adam on the same smoothed objective** – a finer‑grained schedule for the
   temperature λ further reduces the smoothed max.

3. **Exponentiated‑gradient descent on the true max** – sub‑gradient steps
   with a decreasing step‑size drive the exact max down.

4. **Simulated‑annealing hill‑climb** – moves a fraction of mass from a heavy
   index to a lighter one; the move is accepted either if it improves the
   objective or with a probability that slowly cools.  Incremental convolution
   updates keep the inner loop cheap.

The sequence length is set to 6000 (larger than the previous 4000) to give
more freedom for flattening the self‑convolution while keeping each iteration
fast.  All phases keep the vector on the probability simplex, so the final
evaluation is simply `2·n·max(conv)`.  The best sequence found before the
deadline is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability distribution."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power',
         'edge', 'bimodal', 'sparse', 'beta']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:                     # beta – a symmetric U‑shaped pdf
        alpha = rng.uniform(0.1, 2.5)    # α<1 yields heavy edges
        x = (np.arange(n) + 0.5) / n
        # pdf ∝ x^{α-1} (1-x)^{α-1}
        with np.errstate(divide='ignore'):
            log_pdf = (alpha - 1) * (np.log(x) + np.log(1 - x))
        a = np.exp(log_pdf)
        a[np.isinf(a)] = 0.0
        # protect against all‑zero
        if a.sum() == 0:
            a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max of
    the convolution.  Returns (objective, gradient) w.r.t. the logits y,
    where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)          # stabilise
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]       # gradient w.r.t. a_i (pre‑softmax)

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if steps % 200 == 0:
                    print(f"[Adam] lam={lam:.5f} step={steps} score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient descent (sub‑gradient of max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill climbing with incremental convolution updates
# ----------------------------------------------------------------------


def simulated_annealing_hill_climb(a: np.ndarray,
                                   curr_score: float,
                                   fft_len: int,
                                   n: int,
                                   deadline: float,
                                   rng: np.random.Generator,
                                   start_temp: float = 0.04,
                                   end_temp: float = 5e-5,
                                   start_frac: float = 0.25,
                                   end_frac: float = 0.001,
                                   recompute_every: int = 1500,
                                   phase_start: float = None):
    """
    Greedy hill‑climb that also accepts worsening moves according to a
    temperature schedule.  Mass is moved from a high‑mass index to a lower‑mass
    one; convolution updates are performed incrementally.
    """
    if phase_start is None:
        phase_start = time.time()

    best_a = a.copy()
    best_score = curr_score

    conv = full_conv(a, fft_len)
    curr_score = 2.0 * n * conv.max()

    diff = np.zeros_like(conv)

    it = 0
    accepted = 0

    while time.time() < deadline:
        now = time.time()
        progress = (now - phase_start) / (deadline - phase_start)
        # linear schedules
        temp = start_temp * (1.0 - progress) + end_temp * progress
        max_frac = start_frac * (1.0 - progress) + end_frac * progress

        # pick source proportional to current mass
        if a.sum() == 0:
            break
        src = rng.choice(n, p=a)
        if a[src] <= 0.0:
            continue
        # destination – uniform among all other indices
        dst = rng.integers(0, n)
        if dst == src:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        accept = False
        if new_score < curr_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - curr_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            # numerical safety
            if a[src] < 0.0:
                a[src] = 0.0
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        if it % recompute_every == 0:
            # occasional full recomputation to avoid drift
            conv = full_conv(a, fft_len)
            curr_score = 2.0 * n * conv.max()
            # re‑normalise just in case
            s = a.sum()
            if s != 0.0:
                a /= s

        it += 1

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 6000 entries gives a richer search space while staying fast
    # --------------------------------------------------------------
    n = 6000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise – try a beta‑shaped distribution (U‑shape) for a good start
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = random_initial(n, rng)
    else:
        a0 = random_initial(n, rng)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.18 * budget_s  # L‑BFGS
    phase2_end = start_time + 0.45 * budget_s  # Adam
    phase3_end = start_time + 0.70 * budget_s  # Exp‑grad
    # remainder → annealing hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS with many short restarts
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.1, 0.07,
                    0.04, 0.025, 0.015, 0.008, 0.004,
                    0.002, 0.001, 5e-4, 2e-4, 1e-4,
                    5e-5, 2e-5, 1e-5, 5e-6]
    maxiter_per_lam = 1000

    while time.time() < phase1_end:
        # 70 % fresh starts, 30 % jittered copy of the current best
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the smooth max
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        lam_schedule_adam = [0.4, 0.3, 0.22, 0.15, 0.09,
                             0.05, 0.03, 0.015, 0.008, 0.004]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam,
            max_steps_per_lam=2500,
            deadline=phase2_end,
            n=n,
            fft_len=fft_len,
            lr0=0.14,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on the true max
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=20.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing hill climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        hill_start = time.time()
        best_a, best_score = simulated_annealing_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            start_temp=0.04,
            end_temp=5e-5,
            start_frac=0.25,
            end_frac=0.001,
            recompute_every=2000,
            phase_start=hill_start,
        )
        print(f"[anneal‑hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
