"""
A mixed integer‑linear‑programming + gradient approach for the
self‑convolution problem.

1.  Randomised Dirichlet and analytic families generate a diverse
    pool of probability vectors.  The best of them seeds the main loop.
2.  In the main loop we solve a linear programme:
        maximise   Σ g_j
        subject to 0 ≤ g ,  (f * g)[k] ≤ M   ∀k
   where f is the current vector and M = max(f * f).  The LP finds a
   new distribution g that can be added without increasing the worst
   autocorrelation entry.  A convex combination f←(1‑t)f+t·g is taken
   (t is line‑searched) which strictly reduces the objective.
3.  After the LP phase a short exponentiated‑gradient descent refines
   the solution, followed by a tiny deterministic hill‑climb.
4.  The best vector found before the deadline is returned as a plain
   Python list; the provided ``evaluate_sequence`` routine is used only
   for the final check.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming Σa = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2*n-1, n)) such that
        A @ g  ==  np.convolve(f, g, mode='full')
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random probability vector from a Dirichlet(alpha) distribution."""
    # pick a concentration parameter on log‑scale
    log_alpha = rng.uniform(-2.0, 1.0)      # α ∈ [~0.01, ~10]
    alpha = 10 ** log_alpha
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s <= 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / s
    return x


def exponential_family(n: int, rng: np.random.Generator) -> np.ndarray:
    lam = rng.uniform(0.0005, 0.02)
    a = np.exp(-lam * np.arange(n))
    a /= a.sum()
    return a


def power_family(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(-2.0, 3.0)
    a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a /= a.sum()
    return a


def gaussian_family(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(n * 0.05, n * 0.25)
    x = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a /= a.sum()
    return a


def hann_family(n: int, rng: np.random.Generator) -> np.ndarray:
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    a /= a.sum()
    return a


def generate_random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a generator at random and produce a candidate."""
    fam = rng.choice(
        [
            random_dirichlet,
            exponential_family,
            power_family,
            gaussian_family,
            hann_family,
        ]
    )
    return fam(n, rng)


# ----------------------------------------------------------------------
# Exponentiated‑gradient refinement (same as the reference code)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray, deadline: float,
                          n: int, fft_len: int, rng: np.random.Generator):
    """Projected exponentiated‑gradient descent on the exact max‑conv objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # Compute full self‑convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fw = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fw * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc


# ----------------------------------------------------------------------
# Fine deterministic hill‑climb (tiny final polishing)
# ----------------------------------------------------------------------


def fine_hill_climb(a: np.ndarray, score: float,
                    fft_len: int, n: int,
                    deadline: float,
                    init_step: float = 0.20,
                    min_step: float = 1e-5) -> tuple[np.ndarray, float]:
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] where defined
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        valid = np.arange(i_start, i_end + 1)
        partner[valid] = a[max_idx - valid]

        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: smallest partner (avoid i_max itself)
        order = np.argsort(partner)
        i_min = int(order[0] if order[0] != i_max else order[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        else:
            step = max(step * 0.8, min_step)
    return a, score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list of heights) with a low
    ``evaluate_sequence`` score.  The routine mixes random sampling,
    a linear‑programming refinement loop, exponentiated‑gradient descent
    and a final tiny hill‑climb, respecting the 1000 s time budget.
    Returns the best list of floats found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – small enough for fast LP while still expressive
    # ------------------------------------------------------------------
    n = 500                     # length of the step function
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Starting pool – incorporate a previous construction if available
    # ------------------------------------------------------------------
    pool = []
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Interpolate to the required length and renormalise
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            pool.append(a)

    pool.append(np.full(n, 1.0 / n))               # uniform
    pool.append(gaussian_family(n, rng))
    pool.append(hann_family(n, rng))

    # Random Dirichlet samples (cheap) to enrich the pool
    for _ in range(30):
        pool.append(random_dirichlet(n, rng))

    best_a = None
    best_score = np.inf
    for cand in pool:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] best score after seeding = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – random Dirichlet search (quick broad exploration)
    # ------------------------------------------------------------------
    phase1_deadline = time.time() + 0.15 * budget_s
    while time.time() < phase1_deadline:
        cand = generate_random_candidate(n, rng)
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            print(f"[rand] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP refinement loop
    # ------------------------------------------------------------------
    phase2_deadline = time.time() + 0.55 * budget_s   # about half the budget
    f = best_a.copy()
    iteration = 0
    while time.time() < phase2_deadline:
        # current max self‑convolution
        fa = np.fft.fft(f, fft_len)
        conv_ff = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv_ff.max()

        # build convolution matrix for (f * g)
        A_ub = build_conv_matrix(f)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # LP: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None) for _ in range(n)]

        # solve
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        except Exception as e:
            print("[LP] exception:", e)
            break

        if res.status != 0:
            # infeasible or unbounded – give up this iteration
            print("[LP] no feasible solution, status:", res.status)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        # normalise g to sum=1 (same scale as f)
        g_norm = g / sum_g

        # line‑search for the best mixing weight t ∈ (0,1]
        # only a few candidates are examined
        best_local_sc = best_score
        best_local_f = None
        best_t = None
        for t in (0.1, 0.2, 0.3, 0.5, 0.7):
            f_new = (1.0 - t) * f + t * g_norm
            f_new /= f_new.sum()
            sc = fast_score(f_new, fft_len, n)
            if sc < best_local_sc:
                best_local_sc = sc
                best_local_f = f_new.copy()
                best_t = t

        # if no improvement, fall back to a tiny step
        if best_local_f is None:
            t = 0.05
            f_new = (1.0 - t) * f + t * g_norm
            f_new /= f_new.sum()
            sc = fast_score(f_new, fft_len, n)
            f = f_new
            cur_sc = sc
        else:
            f = best_local_f
            cur_sc = best_local_sc

        if cur_sc < best_score:
            best_score = cur_sc
            best_a = f.copy()
            print(f"[LP‑iter {iteration}] improved score = {best_score:.6f}")

        iteration += 1
        if time.time() >= phase2_deadline:
            break

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement (long tail)
    # ------------------------------------------------------------------
    phase3_deadline = time.time() + 0.30 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[exp‑grad] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final fine deterministic hill‑climb
    # ------------------------------------------------------------------
    final_deadline = time.time() + 0.05 * budget_s
    best_a, best_score = fine_hill_climb(
        best_a,
        best_score,
        fft_len,
        n,
        final_deadline,
        init_step=0.2,
        min_step=1e-5,
    )
    print(f"[final‑hill] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence returned {final_eval:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)
        final_eval = None

    return final_seq
