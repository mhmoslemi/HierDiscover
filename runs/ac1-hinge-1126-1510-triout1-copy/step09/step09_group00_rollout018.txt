            a_old = np.maximum(a_old, 0.0)
            s = a_old.sum()
            if s > 0:
                a_old /= s
                # quick evaluation – only the score is needed for comparison
                from numpy import convolve
                b = np.convolve(a_old, a_old)
                score_old = float(2.0 * n * b.max() / (s * s))
                # compute our triangular score
                b_tri = np.convolve(a, a)
                score_tri = float(2.0 * n * b_tri.max() / (1.0 * 1.0))
                if score_old < score_tri:
                    best_a = a_old
                    print(f"[init] kept previous construction, score={score_old:.6f}")
                else:
                    best_a = a
                    print(f"[init] triangular construction, score={score_tri:.6f}")
    except Exception:
        # No previous construction – just use the triangular one
        best_a = a
        print("[init] triangular construction (no previous data)")

    # ------------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`.
    # The function is imported in the execution environment.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_score = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_score:.6f}")

    return final_seq
