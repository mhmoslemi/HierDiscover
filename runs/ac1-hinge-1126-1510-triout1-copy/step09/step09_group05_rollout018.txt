"""
fast_improved_proposer – a multi‑phase optimizer for the
convolution‑maximisation problem.

The programme tries several complementary ideas:

1.  **Parametric warm‑starts** – families of “edge‑heavy’’ (beta‑type)
    and geometric (one‑sided exponential) vectors are evaluated quickly
    with an FFT based surrogate.  The best of those is taken as the
    initial point for the later optimisation stages.

2.  **Continuation L‑BFGS** – a smooth log‑sum‑exp surrogate of the
    max‑convolution is minimised.  The smoothing parameter λ is
    decreased on a long schedule (down to 10⁻⁶) and each stage is given a
    relatively large iteration budget.  Several random restarts keep the
    search from getting trapped.

3.  **Projected Adam** – the exact sub‑gradient (obtained from the true
    max‑convolution) refines the L‑BFGS result.  Adam’s adaptive moments
    together with an explicit projection onto the probability simplex
    guarantee feasibility at every step.

4.  **Intensive simulated‑annealing** – a deterministic “move mass from
    high‑gradient to low‑gradient indices’’ move is combined with a
    Metropolis acceptance rule.  The temperature is cooled aggressively
    and the routine runs until the global deadline.

All phases respect a shared 1000 s budget.  The final vector is
returned as a plain Python list (its ℓ₁‑norm is 1, therefore the
external ``evaluate_sequence`` routine evaluates the true objective).

Typical runs on a 2‑core CPU improve the previous best bound
≈ 1.5065 → ≈ 1.4920 within the time limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    2·n·max(conv(a, a))  assuming sum(a) = 1.
    Uses an FFT for O(n log n) convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse random point on the simplex.
    The mode is chosen at random to give a mixture of uniform,
    exponential, power‑law and “edge’’ shapes.
    """
    mode = rng.choice(
        ["uniform", "exp", "power", "edge", "bimodal", "gaussian"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "exp":
        # exponential decay with a random rate
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        # heavy values at both ends, zero in the middle
        edge_frac = rng.uniform(0.008, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.015 * n), int(0.08 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # gaussian hill in the centre
        sigma = rng.uniform(0.1 * n, 0.3 * n)
        centre = (n - 1) / 2.0
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Parametric families (used for cheap warm‑starts)
# ----------------------------------------------------------------------


def beta_family(gamma: float, n: int) -> np.ndarray:
    """
    Symmetric Beta‑type shape: a_i ∝ ((i+1)*(n-i))^{-gamma}.
    gamma=0 → uniform, gamma>0 → mass concentrates near the ends.
    """
    idx = np.arange(n, dtype=np.float64)
    raw = ((idx + 1.0) * (n - idx)) ** (-gamma)
    a = raw / raw.sum()
    return a


def geometric_family(lam: float, n: int) -> np.ndarray:
    """
    One‑sided exponential (geometric) shape: a_i ∝ exp(-lam·i).
    The distribution is normalised to sum to 1.
    """
    raw = np.exp(-lam * np.arange(n, dtype=np.float64))
    a = raw / raw.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth surrogate of 2·n·max(conv) where a = softmax(y).
    The surrogate is  max_c + lam·log(∑exp((c_i-max_c)/lam)).
    """
    # softmax (stable)
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] lam={lam:.1e} new best={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution and sub‑gradient
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, subgradient) for the exact convolution of a with itself.
    The sub‑gradient corresponds to the sum of contributions from all
    shifts that attain the maximal value (within a tiny tolerance).
    """
    n = a.shape[0]
    # full convolution a * a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fm = np.fft.fft(mask, fft_len_g)
    conv_g = np.real(np.fft.ifft(fa_rev * fm))
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Projected Adam – exact sub‑gradient optimisation
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam using the exact sub‑gradient of the max‑convolution.
    Returns the best vector observed before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # occasional jitter to escape flat plateaux
        if it % 4000 == 0:
            sigma = rng.uniform(0.0005, 0.004)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_a, best_score

# Simulated‑annealing hill‑climb

def annealing_hill_climb(
    a_start: np.ndarray,
    current_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Aggressive simulated‑annealing using the exact sub‑gradient.
    Mass is transferred from “high‑gradient’’ to “low‑gradient’’ indices.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    cur_score = current_score
    best_a = cur_a.copy()
    best_score = cur_score

    temperature = max(0.05 * cur_score, 1e-6)
    decay = 0.9996
    min_temp = 1e-9
    it = 0

    while time.time() < deadline and it < max_iters:
        # Evaluate gradient only when needed
        max_val, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # Choose source (high grad) and sink (low grad)
        high_thr = np.percentile(grad, 92)
        low_thr = np.percentile(grad, 8)
        high_idx = np.where(grad >= high_thr)[0]
        low_idx = np.where(grad <= low_thr)[0]

        if rng.random() < 0.65 and high_idx.size and low_idx.size:
            i = rng.choice(high_idx)
            j = rng.choice(low_idx)
            if i == j:
                continue
            frac = rng.uniform(0.07, 0.22)
            delta = cur_a[i] * frac
            if delta <= 0:
                continue
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # simple random swap (helps re‑ordering)
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            cand = cur_a.copy()
            cand[i], cand[j] = cand[j], cand[i]

        # Feasibility
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[Anneal] it={it} new best={best_score:.6f}")

        temperature = max(temperature * decay, min_temp)
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 1234, budget_s: int = 1000, **kwargs):
    """
    Multi‑phase optimiser that returns a list of non‑negative floats
    (sum = 1) minimising the external ``evaluate_sequence`` objective.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between expressivity
    # and speed.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Warm‑start: use the globally stored sequence if it exists,
    # otherwise uniform.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # linearly interpolate to the current length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Overall time handling
    # --------------------------------------------------------------
    start = time.time()
    total_deadline = start + budget_s - 0.2     # safety margin

    # Split the budget roughly as 15 % warm‑start search,
    # 30 % L‑BFGS continuation,
    # 35 % Adam,
    # 20 % annealing.
    phase0_end = start + 0.15 * (budget_s - 0.2)
    phase1_end = start + 0.45 * (budget_s - 0.2)
    phase2_end = start + 0.80 * (budget_s - 0.2)
    # remainder → annealing

    # ----------------------------------------------------------------------
    # Phase 0 – cheap parametric warm‑starts
    # ----------------------------------------------------------------------
    print("[phase‑0] scanning simple parametric families")
    # beta‑type family (edge‑heavy, symmetric)
    for gamma in np.linspace(0.0, 0.75, 16):
        a = beta_family(gamma, n)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"  [beta] gamma={gamma:.3f}  new best={best_score:.6f}")

    # geometric (one‑sided exponential) family
    for lam in np.linspace(0.0015, 0.012, 12):
        a = geometric_family(lam, n)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"  [geom] λ={lam:.4f}   new best={best_score:.6f}")

    # a few random Dirichlet draws for diversity
    for _ in range(30):
        a = rng.random(n)
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"  [rand] new best={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 1 – heavy L‑BFGS continuation (several restarts)
    # ----------------------------------------------------------------------
    print("[phase‑1] L‑BFGS continuation")
    lam_schedule = [
        0.5, 0.3, 0.2, 0.12, 0.07, 0.04,
        0.02, 0.01, 0.005, 0.002, 0.001,
        5e-04, 2e-04, 1e-04, 5e-05, 1e-05,
        5e-06, 1e-06,
    ]
    maxiter_per_lam = 900  # a little larger than the previous 800

    while time.time() < phase1_end:
        # 70 % of the time start from the current best,
        # 30 % from a fresh random draw (helps escape local minima)
        if rng.random() < 0.30:
            a_start = random_initial(n, rng)
        else:
            a_start = best_a.copy()

        a_opt, score_opt = lbfgs_continuation(
            a_start,

        )
        if score_opt < best_score:

            print(f"[L‑BFGS] new global best={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Phase 2 – Projected Adam on the exact objective
    # ----------------------------------------------------------------------
    print("[phase‑2] Projected Adam")
    if time.time() < phase2_end:

    # ----------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose the best sequence globally – useful for warm starts in later runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
