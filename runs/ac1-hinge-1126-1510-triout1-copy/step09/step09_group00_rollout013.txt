def adam_optimize(a_start, lam_schedule, max_iter_per_lam, deadline, fft_len_conv, fft_len_g, n):
    y = np.log(a_start+1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    best_a = a_start.copy()
    best_score = fast_score(a_start, fft_len_conv, n)

    it = 0
    for lam in lam_schedule:
        for _ in range(max_iter_per_lam):
            if time.time()>=deadline: break
            f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            it+=1
            m = beta1*m + (1-beta1)*grad_y
            v = beta2*v + (1-beta2)*(grad_y**2)
            m_hat = m / (1 - beta1**it)
            v_hat = v / (1 - beta2**it)
            step = 0.5 / (np.sqrt(v_hat) + epsilon)  # constant base learning rate adaptively scaled
            y = y - step * m_hat
            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        # optionally re-initialize momentum m,v after schedule shift?

    return best_a, best_score

def generate_family(family, n, rng):
    i = np.arange(n, dtype=float)
    if family == "uniform":
        a = np.ones(n)
    elif family == "linear_inc":
        a = i + 1.0
    elif family == "linear_dec":
        a = n - i
    elif family == "triangular":
        mid = (n-1)/2.0
        a = 1.0 - np.abs(i - mid) / mid
        a = np.maximum(a, 0.0)
    elif family == "inverse_triangular":
        mid = (n-1)/2.0
        a = np.abs(i - mid) / mid
        a = np.maximum(a, 0.0)
    elif family == "hamming":
        a = 0.54 - 0.46 * np.cos(2*np.pi*i/(n-1))
    elif family == "hann":
        a = 0.5 * (1 - np.cos(2*np.pi*i/(n-1)))
    elif family == "blackman":
        a = 0.42 - 0.5*np.cos(2*np.pi*i/(n-1)) + 0.08*np.cos(4*np.pi*i/(n-1))
    elif family == "kaiser":
        beta = rng.uniform(0.5, 10.0)
        from scipy.special import i0
        denom = i0(np.pi*beta)
        a = i0(np.pi*beta * np.sqrt(1 - ((2*i/(n-1))-1)**2)) / denom
    elif family == "beta":
        alpha = rng.uniform(0.1, 10.0)

"""
Propose a step‑function (non‑negative vector) with a low value of

    2 · n · max(a * a) / (sum(a)²)

The algorithm consists of several fast phases that respect a global
deadline (default 1000 s):

1. **Cheap parametric sweep** – sample many simple analytic families
   (uniform, linear, triangular, classic windows, Beta‑distribution,
   mixtures of Gaussians, sparse binary patterns …).  The best vector
   from this cheap sweep seeds the later phases.

2. **Random‑restart L‑BFGS on a smoothed max‑convolution** – the true
   max is replaced by a log‑sum‑exp smoothing; a continuation over
   decreasing smoothing parameters is used, each run starting from a
   jittered copy of the current best.

3. **Adam on the same smoothed objective** – after L‑BFGS we fine‑tune
   the logits of a soft‑max parametrisation using the Adam optimiser,
   again with a decreasing smoothing schedule.

4. **Projected sub‑gradient descent with momentum** – true max‑convolution
   sub‑gradients are used; after each step the iterate is projected onto
   the probability simplex.

5. **Slow‑cooling simulated annealing** – a long stochastic hill‑climb
   that transfers a fraction of mass between two indices and accepts
   uphill moves with a square‑root cooling schedule.

Throughout the run the best vector (according to the inexpensive
`fast_score` based on FFT) is remembered.  At the end we call the
official `evaluate_sequence` on the best vector and return it.

The code is completely self‑contained, uses only numpy / scipy, works
for any length `n` (default 4000) and respects the 1000 s time budget.
"""
import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import i0

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector already normalised (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# --------------------------------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# --------------------------------------------------------------
# Optimisation primitives
# --------------------------------------------------------------

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:           # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    for lam in lam_schedule:
        for _ in range(maxiter_per_lam):
            if time.time() >= deadline:
                break
            f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            it += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            step = 0.25 / (np.sqrt(v_hat) + eps)          # base LR = 0.25
            y = y - step * m_hat
            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[adam] improved fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # true max‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max (via convolution with mask)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size schedule – slower decay than in the original code
        step = 0.5 / (it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a square‑root cooling schedule.
    Mass is moved between two randomly chosen indices (up to 30 % of the source mass).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.05 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

# Richer parametric families for the cheap sweep

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "uniform":
        a = np.ones(n)

    elif family == "linear_inc":
        a = i + 1.0

    elif family == "linear_dec":
        a = n - i

    elif family == "triangular":
        mid = (n - 1) / 2.0
        a = 1.0 - np.abs(i - mid) / mid
        a = np.maximum(a, 0.0)

    elif family == "inverse_triangular":
        mid = (n - 1) / 2.0
        a = np.abs(i - mid) / mid
        a = np.maximum(a, 0.0)

    elif family == "hamming":
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))

    elif family == "hann":
        a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))

    elif family == "blackman":
        a = (
            0.42
            - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
        )

    elif family == "kaiser":
        beta = rng.uniform(0.5, 10.0)
        denom = i0(np.pi * beta)
        a = i0(
            np.pi * beta * np.sqrt(1.0 - ((2.0 * i / (n - 1)) - 1.0) ** 2)
        ) / denom

    elif family == "beta":
        alpha = rng.uniform(0.1, 10.0)
        beta_param = rng.uniform(0.1, 10.0)
        x = i / (n - 1)
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta_param - 1.0) + 1e-12)

    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)

    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)

    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))

    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)

    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))

    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)

    elif family == "random":
        a = rng.random(n)

    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform a cheap sweep over many random parametric families
    and return the best vector together with its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10          # 10 % of total budget

    families = [
        "uniform",
        "linear_inc",
        "linear_dec",
        "triangular",
        "inverse_triangular",
        "hamming",
        "hann",
        "blackman",
        "kaiser",
        "beta",
        "gauss",
        "exp",
        "double_exp",
        "cos",
        "logistic",
        "binary_sparse",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score
