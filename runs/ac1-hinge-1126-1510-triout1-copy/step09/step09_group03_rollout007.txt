def build_conv_matrix(a):
    n = len(a)
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=np.float64)

"""
Step‑function optimizer using a linear‑programming mixing scheme
followed by a short smoothed‑L‑BFGS polish.  The LP step
maximises the total mass that can be added to the current
distribution while keeping the self‑convolution peak unchanged.
Mixing the current vector with the LP‑solution strictly reduces
the peak and therefore the evaluation score.
The whole search respects the 1000 s budget and returns the best
sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize


# -------------------------------------------------------------
# Helper utilities
# -------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a))  for a normalised (sum=1) vector."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# -------------------------------------------------------------
# Smoothed L‑BFGS (identical to the reference implementation)
# -------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                      lam_factor: float = 0.30):
    """
    Smoothed objective based on log‑sum‑exp of the self‑convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax → a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    # chain rule for softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray, lam_sched, maxiter_per_lam,
                       deadline: float, fft_len: int, n: int):
    """Cascade of L‑BFGS optimisations on the smoothed objective."""
    y = np.log(a_start + 1e-12)          # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:               # pragma: no cover
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------
# Linear‑programming mixing step
# -------------------------------------------------------------

def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the matrix A (size (2n‑1)×n) such that A @ b = conv(a,b)
    for any vector b (both non‑negative).
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contains a shifted by j positions
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_mix_step(a: np.ndarray, t: float, rng: np.random.Generator,
                deadline: float) -> tuple[np.ndarray, bool]:
    """
    Perform one LP based mixing iteration.
    Returns the new vector and a flag indicating whether an improvement
    (non‑increase of the max‑convolution) was obtained.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    M = conv.max()
    A = build_conv_matrix(a)                     # (2n‑1) × n
    M_vec = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=np.float64)

    # time left for the solver (tiny safety margin)
    time_left = max(0.0, deadline - time.time() - 0.1)
    # linprog's 'highs' automatically respects reasonable times;
    # we only pass the limit via the options dictionary if supported.
    opts = {'presolve': True}
    if time_left > 0:
        opts['time_limit'] = time_left

    res = linprog(c, A_ub=A, b_ub=M_vec,
                  bounds=(0, None), method='highs', options=opts)

    if res.status != 0 or res.x is None:        # pragma: no cover
        return a, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return a, False

    g = b / sum_b                               # normalise to unit sum

    # try mixing; if the full step (t=1) already respects the bound,
    # we can replace a completely.
    t_cur = t
    improved = False
    while t_cur >= 1e-5:
        a_new = (1.0 - t_cur) * a + t_cur * g
        conv_new = np.convolve(a_new, a_new)
        if conv_new.max() <= M + 1e-14:
            a = a_new
            improved = True
            break
        t_cur *= 0.5
    return a, improved


# -------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with low evaluation score.
    The algorithm consists of:
      1. initialise (interpolate previous best or uniform),
      2. repeatedly apply a linear‑programming mixing step,
      3. finish with a short smoothed L‑BFGS polish,
      4. return the best sequence.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # -----------------------------------------------------------------
    n = 1500                               # moderate size for quick LP solves
    fft_len = next_pow2(2 * n - 1)

    # -----------------------------------------------------------------
    # Initial vector (interpolate previous global if present)
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 1:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:                               # fallback to uniform
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, score={best_score:.6f}")

    # -----------------------------------------------------------------
    # Time budget split
    # -----------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05    # small safety margin
    phase_lp_end = start_time + 0.70 * budget_s
    phase_lbfgs_end = deadline

    # -----------------------------------------------------------------
    # Phase 1 – Linear‑programming mixing
    # -----------------------------------------------------------------
    mix_step = 0.3           # initial mixing factor
    while time.time() < phase_lp_end:
        a, improved = lp_mix_step(a, mix_step, rng, phase_lp_end)
        if not improved:
            # if a step fails to improve, reduce the step size a little
            mix_step = max(0.05, mix_step * 0.8)
        else:
            mix_step = min(0.5, mix_step * 1.02)   # gently grow step size
            cur_score = fast_score(a, fft_len)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[LP iter] new best score {best_score:.6f}  "
                      f"(mix={mix_step:.3f})")
        # safety: break if progress stalls
        if mix_step < 0.06 and not improved:
            break

    # -----------------------------------------------------------------
    # Phase 2 – short L‑BFGS polish on a smoothed max
    # -----------------------------------------------------------------
    if time.time() < phase_lbfgs_end:
        # use a tighter lam‑schedule and a few more iterations per lam
        lam_sched = [
            0.2, 0.15, 0.1, 0.07, 0.04,
            0.025, 0.015, 0.008, 0.004, 0.002,
            0.001, 0.0005, 0.0002, 0.0001, 5e-05,
        ]
        maxiter_per_lam = 800
        a_opt, score_opt = lbfgs_continuation(
            best_a, lam_sched, maxiter_per_lam,
            phase_lbfgs_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS polish] improved score {best_score:.6f}")

    # -----------------------------------------------------------------
    # Final evaluation with the official scorer
    # -----------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()['height_sequence_1'] = final_seq

    return final_seq
