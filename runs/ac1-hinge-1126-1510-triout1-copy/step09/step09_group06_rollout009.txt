def propose_candidate(seed=42, budget_s=1000):

        prev = np.array(globals()['height_sequence_1'], dtype=float)

            a_prev = np.full(n, 1.0/n)
        candidates.append(a_prev)

"""
Propose a low‑score step function.

The algorithm proceeds in four phases within a 1000 s budget:

1️⃣  Build a diverse pool of initial candidates (previous best,
    spectral, random Dirichlet draws, exponential / power‑law / edge‑heavy
    parametric families, triangular shape).

2️⃣  Run a continuation of smooth L‑BFGS on a log‑sum‑exp
    smoothed version of the max‑convolution.  Several seeds are processed
    and the best result is kept.

3️⃣  Perform a long Adam‑style projected sub‑gradient descent on the
    exact max‑convolution.  The gradient of the (hard) max‑convolution
    is obtained by a single FFT‑based back‑propagation.

4️⃣  Polish the result with a massive greedy hill‑climb that repeatedly
    transfers a random fraction of mass from one index to another,
    always keeping the simplex constraints.  The move size decays
    gradually.

All phases continually update the global best solution.  The final
sequence is returned as a plain Python list; its true score is printed
using the supplied `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Fast objective (used during the search)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score for a vector a that already sums to 1:
       2·n·max(a*a)   (exact max‑convolution, no division by sum²)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS (log‑sum‑exp approximation)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed version of 2·n·max(a*a) where a = softmax(y).
    Returns (objective, gradient) w.r.t. y.
    """
    a = softmax(y)

    # full self‑convolution
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. the convolution entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # back‑propagation through convolution: d conv / d a = rev_a ⋆ w
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage. Returns the best a found and its
    fast_score.
    """
    y = np.log(a_start + 1e-12)           # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution gradient (used by Adam & hill‑climb)
# ----------------------------------------------------------------------


def max_conv_grad(a: np.ndarray, fft_len: int):
    """
    Compute gradient of max(self‑convolution) w.r.t. a (ordered pair
    count) and return also the current maximum value.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    grad_full = np.fft.irfft(
        np.fft.rfft(a_rev, fft_len) * np.fft.rfft(mask, fft_len),
        fft_len
    )
    g = 2.0 * grad_full[n - 1:n - 1 + n]          # d max(conv) / d a
    return g, max_val


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient descent on the exact max
# ----------------------------------------------------------------------


def adam_subgrad_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          step0: float = 0.5,
                          beta1: float = 0.9,
                          beta2: float = 0.999,
                          eps: float = 1e-8):
    """
    Adam optimizer on the exact max‑convolution objective.
    The update is projected onto the simplex after each step.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        g, _ = max_conv_grad(a, fft_len)               # gradient of max(conv)
        g_obj = 2.0 * n * g                            # gradient of 2·n·max

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g_obj
        v = beta2 * v + (1.0 - beta2) * (g_obj ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = step0 / np.sqrt(t)
        a_new = a - step * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = best_a.copy()

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------


def hill_climb(a_start: np.ndarray,
               start_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 200000):
    """
    Greedy hill‑climb: repeatedly move a random fraction of the mass
    from one coordinate to another, accepting only improvements.
    The maximal transfer fraction decays slowly.
    """
    best = a_start.copy()
    best_score = start_score
    rng = np.random.default_rng()
    it = 0
    max_frac = 0.12  # initial maximum transferred fraction

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most max_frac of the source mass
        delta = best[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # undo
            best[i] += delta
            best[j] -= delta

        # gradually shrink the transfer size
        max_frac *= 0.99997
        if max_frac < 1e-4:
            max_frac = 1e-4

        it += 1

    return best, best_score

def spectral_init(n: int) -> np.ndarray:
    """Inverse FFT of sqrt of the FFT of a uniform target."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Rayleigh‑like / Dirichlet‑type random draws."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def exponential_shape(n: int, c: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-c·min(i,n‑1‑i))."""
    idx = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-c * idx)
    a /= a.sum()
    return a


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (min(i,n‑1‑i)+1)^(-p)."""
    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = dist.astype(float) ** (-p)
    a /= a.sum()
    return a


def edge_shape(n: int, edge_frac: float, ratio: float,
               interior_zero: bool = False) -> np.ndarray:
    """
    Edge‑heavy shape.
    edge_frac ∈ (0,0.5) fraction of indices taken from each side.
    ratio = w / u where w is weight of edge cells and u weight of interior cells.
    If interior_zero=True the interior receives zero mass.
    """
    m = max(1, int(edge_frac * n))
    if interior_zero:
        a = np.zeros(n)
        a[:m] = 0.5 / m
        a[-m:] = 0.5 / m
        return a
    interior_len = n - 2 * m
    u = 1.0 / (interior_len + 2 * m * ratio)
    w = ratio * u
    a = np.full(n, u)
    a[:m] = w
    a[-m:] = w
    a /= a.sum()
    return a


def triangular_shape(n: int) -> np.ndarray:
    """Tent distribution (peak in the centre, linear to the edges)."""
    mid = (n - 1) / 2.0
    tri = 1.0 - np.abs(np.arange(n) - mid) / mid
    tri[tri < 0] = 0.0
    tri /= tri.sum()
    return tri

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Problem size
    # ------------------------------------------------------------------
    n = 4000                         # length of the step function
    fft_len = next_pow2(2 * n - 1)   # power‑of‑two for fast FFTs

    # ------------------------------------------------------------------
    # Gather a diverse pool of initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # 0️⃣  Previous best (if present)

        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_prev = np.interp(x_new, x_old, prev)
            a_prev = np.clip(a_prev, 0.0, None)
            s = a_prev.sum()
            a_prev = a_prev / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a_prev)

    # 1️⃣  Uniform baseline
    candidates.append(np.full(n, 1.0 / n))

    # 2️⃣  Spectral and random initialisations
    candidates.append(spectral_init(n))
    for _ in range(3):
        candidates.append(random_initial(n, rng))

    # 3️⃣  Parametric families
    # exponential
    for c in np.logspace(-5, -0.5, num=7):
        candidates.append(exponential_shape(n, c))
    # power‑law
    for p in np.linspace(0.1, 4.0, num=9):
        candidates.append(powerlaw_shape(n, p))
    # edge‑heavy
    edge_fracs = [0.003, 0.006, 0.012, 0.025]
    ratios = [5, 10, 20, 40, 80]
    for ef in edge_fracs:
        for r in ratios:
            candidates.append(edge_shape(n, ef, r, interior_zero=False))
            candidates.append(edge_shape(n, ef, r, interior_zero=True))
    # triangular
    candidates.append(triangular_shape(n))

    # ------------------------------------------------------------------
    # Evaluate the initial pool and keep the best few seeds
    # ------------------------------------------------------------------
    scored = [(fast_score(a, fft_len, n), a) for a in candidates]
    scored.sort(key=lambda x: x[0])
    best_score, best_a = scored[0]
    print(f"[init] best initial fast_score = {best_score:.8f}")

    # Keep the top 4 vectors as L‑BFGS seeds
    lbgfs_seeds = [a for _, a in scored[:4]]

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – smooth L‑BFGS (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.6, 0.45, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002, 0.0001
    ]
    maxiter_per_lam = 900

    for seed_a in lbgfs_seeds:
        if time.time() >= phase1_end:
            break
        a_opt, sc_opt = lbfgs_continuation(seed_a,
                                            lam_schedule,
                                            maxiter_per_lam,
                                            phase1_end,
                                            fft_len,
                                            fft_len,
                                            n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam‑style projected sub‑gradient descent (≈50 %)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    if time.time() < phase2_end:
        a_opt, sc_opt = adam_subgrad_optimize(best_a,
                                              phase2_end,
                                              n,
                                              fft_len,
                                              step0=0.4)   # modest initial step
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Adam] new best fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – greedy hill‑climb polishing (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = hill_climb(best_a,
                                   best_score,
                                   fft_len,
                                   n,
                                   deadline,
                                   max_iter=300000)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[Hill‑climb] final fast_score = {best_score:.8f}")

    # ------------------------------------------------------------------
    # Return the result (evaluate with the provided function)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.8f}")

    return final_seq
