"""
Multi‑phase optimizer for the step‑function bound.

The routine works on a vector `a` of length `n` (default 4000) that
satisfies a∈Δ (non‑negative, Σa = 1).  The objective that we minimise is

    2·n·max(conv(a, a))

because `evaluate_sequence` returns this value divided by the squared sum,
which is 1 in our normalisation.

Phases (time‑budget aware)

1. **Heavy L‑BFGS** – many restarts, long inner loops, decreasing
   log‑sum‑exp smoothness.  This finds a good basin for the convex problem.
2. **Projected Adam** – uses the exact sub‑gradient of the true max‑convolution,
   with an adaptive step‑size and simplex projection.
3. **Deterministic hill‑climb** – repeatedly moves a fraction of mass from
   indices with large sub‑gradient to those with small sub‑gradient; the move
   is always kept if it reduces the fast FFT‑based score.
4. **Fine L‑BFGS** – very small smoothing parameters and many iterations to
   polish the solution.
5. **Pruning** – tiny entries are set to zero and the vector is re‑normalised,
   which often reduces the peak of the convolution.

All work is done with NumPy / SciPy, uses only FFTs for fast evaluation,
and respects the 1000 s overall budget.  The best sequence found is returned
as a plain list (sum = 1) and stored in the global variable
`height_sequence_1` for warm‑starts in later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) assuming Σa = 1.
    Uses FFT (real) for the full linear convolution.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random point on the simplex."""
    mode = rng.choice(
        ["uniform", "random", "exp", "power", "edge", "bimodal", "spike"]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.018)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        # heavy mass at both ends
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # spike – a few large spikes, rest tiny
        a = np.full(n, 1e-8)
        k = rng.integers(3, 8)
        positions = rng.choice(n, size=k, replace=False)
        a[positions] = 1.0 / k
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of 2·n·max(conv) where a = softmax(y).
    Returns (value, gradient w.r.t. y).
    """
    # softmax (stable)
    m_y = np.max(y)
    e = np.exp(y - m_y)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through the convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # only the n entries that affect a (central part)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # safe logits (interior of simplex)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact sub‑gradient via max‑convolution
# ----------------------------------------------------------------------


def max_and_subgradient(
    a: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, subgradient) for the exact max‑convolution of a with itself.
    The sub‑gradient corresponds to the sum of contributions from all shifts
    attaining the maximal value (within a tiny tolerance).
    """
    n = a.shape[0]
    # full convolution a * a
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_val = conv_full.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv_full >= max_val - eps).astype(np.float64)

    # sub‑gradient: 2 * (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fm = np.fft.rfft(mask, fft_len_g)
    conv_g = np.fft.irfft(fa_rev * fm, fft_len_g)
    g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
    return max_val, g


# ----------------------------------------------------------------------
# Projected Adam on the exact objective
# ----------------------------------------------------------------------


def subgradient_adam(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (adaptive moment) using the true sub‑gradient of the
    max‑convolution.  Returns the best vector seen before the deadline.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline:
        max_val, grad = max_and_subgradient(a, fft_len_conv, fft_len_g)

        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = 0.5 / np.sqrt(it)
        a = a - step * m_hat / (np.sqrt(v_hat) + eps)

        a = simplex_projection(a)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[Adam] it={it} new best={best_score:.6f}")

        # occasional jitter to escape plateaus
        if it % 5000 == 0:
            sigma = rng.uniform(0.001, 0.006)
            a = a + rng.normal(0.0, sigma, size=n)
            a = np.maximum(a, 0.0)
            a = np.maximum(a, 1e-12)  # avoid all‑zero
            a /= a.sum()

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb (mass‑transfer) phase
# ----------------------------------------------------------------------


def deterministic_hill_climb(
    a_start: np.ndarray,
    cur_score: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of mass from high‑gradient indices to low‑gradient ones.
    Only moves that improve the fast FFT score are kept.
    """
    rng = np.random.default_rng()
    cur_a = a_start.copy()
    best_a = cur_a.copy()
    best_score = cur_score

    it = 0
    while time.time() < deadline and it < max_iters:
        _, grad = max_and_subgradient(cur_a, fft_len_conv, fft_len_g)

        # split around the median gradient
        median = np.median(grad)
        high_idx = np.where(grad > median)[0]
        low_idx = np.where(grad < median)[0]

        if high_idx.size == 0 or low_idx.size == 0:
            break

        i = rng.choice(high_idx)
        j = rng.choice(low_idx)
        if cur_a[i] == 0:
            it += 1
            continue

        # transfer 5‑15 % of source mass
        frac = rng.uniform(0.05, 0.15)
        delta = cur_a[i] * frac
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        # project onto simplex (non‑negative then renormalise)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0:
            it += 1
            continue
        cand /= s

        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < cur_score:
            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
                print(f"[HillClimb] it={it} new best={best_score:.6f}")

        it += 1

    return best_a, best_score

def prune_and_normalize(a: np.ndarray, thresh: float = 1e-8) -> np.ndarray:
    """Zero out entries below `thresh` and renormalise to sum‑1."""
    a = a.copy()
    a[a < thresh] = 0.0
    total = a.sum()
    if total == 0.0:
        # fallback to uniform if everything got killed
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a /= total
    return a

    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = deterministic_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            fft_len_g,
            n,
            phase3_end,
            max_iters=300_000,
        )
        print(f"[HillClimb] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – fine L‑BFGS with very small smoothing
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        fine_lam_schedule = [5e-06, 2e-06, 1e-06, 5e-07, 2e-07, 1e-07]
        fine_maxiter = 2000
        best_a, fine_score = lbfgs_continuation(
            best_a,
            fine_lam_schedule,
            fine_maxiter,
            total_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if fine_score < best_score:
            best_score = fine_score
            print(f"[L‑BFGS fine] new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final pruning of tiny entries
    # ------------------------------------------------------------------
    best_a = prune_and_normalize(best_a, thresh=1e-8)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[Prune] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final external evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluate_sequence returned {final_eval:.6f}")

    # expose globally for warm‑starts
    globals()["height_sequence_1"] = final_seq

    return final_seq
