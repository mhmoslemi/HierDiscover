"""
Optimisation of a non‑negative step sequence for the provided
``evaluate_sequence`` function.

The implementation follows the spirit of the reference code but adds
several extra ideas:

* a richer collection of parametric families (exponential,
  double‑exponential, power‑law, truncated‑Gaussian, sinusoidal)
* many more random restarts (the optimiser is re‑initialised from a
  random point several times)
* a longer L‑BFGS continuation with a finer smoothing schedule
* a high‑precision projected sub‑gradient loop with an adaptive step‑size
* a final coordinate‑wise “mass‑transfer’’ hill‑climb that moves a
  fraction of the weight between two positions; the step size shrinks
  with the square‑root of the iteration count.

All phases respect a global deadline (default 1000 s).  The best
sequence seen at any moment is returned, and the official score is
computed at the end by calling ``evaluate_sequence``.
"""

from __future__ import annotations
import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Utility helpers (no closures)
# ------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # the zero vector is not on the simplex; return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector normalised to Σa = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ------------------------------------------------------------
# Families of parametric sequences (all are normalised to sum=1)
# ------------------------------------------------------------

def gen_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def gen_double_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a /= a.sum()
    return a


def gen_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1.0
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gen_gauss(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def gen_sin(n: int, freq: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.sin(2 * np.pi * freq * i / n) ** 2 + 0.1   # keep strictly positive
    a /= a.sum()
    return a


def random_simplex_point(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point on the simplex (Dirichlet(1,…,1))."""
    x = rng.random(n)
    x = np.maximum(x, 0.0)
    total = x.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n)
    return x / total


# ------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ------------------------------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_back: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # soft‑max to stay on the simplex
    a = np.exp(y - np.max(y))
    a /= a.sum()

    # convolution a * a (linear)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z   # shape (2n-1,)

    # back‑propagation through the convolution to get ∂smooth_max/∂a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_back)
    fw = np.fft.fft(w, fft_len_back)
    tmp = np.real(np.fft.ifft(fa_rev * fw))
    # the gradient w.r.t. a lives in the band [n‑1 … n‑1+n‑1]
    g_a = tmp[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_a)
    grad_y = a * (g_a - mean_g)

    return f, grad_y

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule: list[float],
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_back: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for a list of smoothing parameters.
    Returns the best a (still normalised to sum=1) and its fast_score.
    """
    # initialise logits (log‑softmax) – adding a tiny constant avoids log(0)
    y = np.log(a_start + 1e-20)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_back, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12},
            )
        except Exception as exc:                  # pragma: no cover
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = fast_score(a, fft_len_conv, n)

        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best fast_score={best_sc:.6f}")

    return best_a, best_sc


def subgradient_descent(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_back: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑convolution.
    Uses a simple momentum term and an iteration‑dependent step size.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)    # momentum buffer
    beta = 0.85
    it = 0

    while time.time() < deadline:
        # true convolution via FFT (faster than np.convolve for many steps)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_back)
        fa_rev = np.fft.fft(a_rev, fft_len_back)
        grad_full = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_full[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # decreasing step size (sqrt decay)
        step = 0.5 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_sc = fast_score(a_candidate, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_sc


def hill_climb_transfer(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climb that repeatedly transfers a random
    fraction of mass from one index to another.  The temperature
    follows a square‑root schedule making the search fairly aggressive
    early on and very fine‑grained near the end.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    T0 = max(0.05 * start_score, 1e-6)           # initial temperature
    t0 = time.time()
    total = max(deadline - t0, 1e-12)

    while time.time() < deadline:
        # cooling – slower than linear
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total)

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        if a[i] <= 0.0:
            continue

        # transfer a fraction (up to 30 %) of a[i] to a[j]
        delta = a[i] * rng.random() * 0.30
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_sc = fast_score(a_new, fft_len_conv, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc

def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Explores a modest grid of hand‑crafted families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential
    for rate in np.linspace(0.0001, 0.0015, 45):
        if time.time() >= deadline:
            return best_a, best_sc
        a = gen_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] exp rate={rate:.5f} → {best_sc:.6f}")

    # double‑exponential
    for rate in np.linspace(0.0001, 0.0012, 35):
        if time.time() >= deadline:
            return best_a, best_sc
        a = gen_double_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] dexp rate={rate:.5f} → {best_sc:.6f}")

    # power‑law
    for p in np.linspace(-6.0, -0.2, 21):
        if time.time() >= deadline:
            return best_a, best_sc
        a = gen_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] power p={p:.2f} → {best_sc:.6f}")

    # Gaussian‑like bump
    for sigma in np.linspace(300.0, 1500.0, 13):
        if time.time() >= deadline:
            return best_a, best_sc
        a = gen_gauss(n, sigma)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] gauss sigma={sigma:.0f} → {best_sc:.6f}")

    # sinusoidal (helps to break symmetry)
    for freq in [1, 2, 3, 5]:
        if time.time() >= deadline:
            return best_a, best_sc
        a = gen_sin(n, freq)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] sin freq={freq} → {best_sc:.6f}")

    return best_a, best_sc

# Main entry point required by the platform

def propose_candidate(
    seed: int = 12345,
    budget_s: int = 1000,
    **_kwargs,
) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation
    score.  The function is time‑bounded (default 1000 s) and returns
    the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    # Problem parameters
    # --------------------------------------------------------
    n = 4000  # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)   # FFT length for a*a
    fft_len_back = next_pow2(3 * n - 2)   # for back‑propagation

    # --------------------------------------------------------
    # Baseline – uniform
    # --------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------
    # Try to re‑use a previously discovered construction
    # --------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the required length
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a_old = np.interp(x_new, x_old, old)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            sc_old = fast_score(a_old, fft_len_conv, n)
            if sc_old < best_score:
                best_a, best_score = a_old, sc_old
                print(f"[init] previous best fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.15   # small safety margin

    # --------------------------------------------------------
    # Phase 0 – systematic parametric sweep
    # --------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    # --------------------------------------------------------
    # Time allocation for later phases
    # --------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.95 * budget_s   # hill‑climb

    # --------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed max (multiple restarts)
    # --------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06,
    ]
    maxiter_per_lam = 1500

    while time.time() < phase1_end:
        # a few random starts to avoid getting trapped
        if rng.random() < 0.30:
            a_start = random_simplex_point(n, rng)
        else:
            # exponential bias – often a good basin
            a_start = gen_exp(n, rng.uniform(0.0002, 0.0010))

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_back,
            n,
        )
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------
    # Phase 2 – true sub‑gradient descent with momentum
    # --------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_descent(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_back,
            n,
        )
        print(f"[phase2] sub‑gradient fast_score={best_score:.6f}")

    # --------------------------------------------------------
    # Phase 3 – final hill‑climb (mass transfer)
    # --------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = hill_climb_transfer(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] hill‑climb best fast_score={best_score:.6f}")

    # --------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # type: ignore[name-defined]
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
