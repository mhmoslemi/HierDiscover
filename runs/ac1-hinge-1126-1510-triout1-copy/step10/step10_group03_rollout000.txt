-----------------
1. **Parametric sweep** – explores several low‑dimensional families
   (exponential, power‑law, arcsine, cosine‑shaped and triangular).
   Mixes each family with the uniform distribution and keeps the best
   candidate according to the fast FFT‑based score.

2. **Mirrored SGD (Exponentiated‑gradient)** – performs projected
   sub‑gradient descent on the true max‑convolution objective using the
   exact sub‑gradient (active‑set mask).  The learning rate follows a
   1/√t schedule and the iterate is re‑projected onto the probability
   simplex after each step.

3. **Two‑stage simulated‑annealing** – first a coarse anneal with large
   mass moves, then a fine anneal with tiny moves.  The temperature and
   maximum transfer fraction decay linearly; the convolution is updated
   incrementally and re‑computed periodically for numerical stability.

4. **Final polishing** – a short second EG pass to clean up any remaining
   noise.

The best sequence found is returned and stored globally as
`height_sequence_1` for possible reuse in future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute 2·n·max(a*a) assuming sum(a) = 1.
    Uses FFT for O(n log n) convolution.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of the true max‑convolution objective and
    the corresponding maximal convolution value.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Families for the cheap parametric sweep
# ----------------------------------------------------------------------


def family_exponential(n: int, lam: float) -> np.ndarray:
    a = np.exp(-lam * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def family_powerlaw(n: int, p: float) -> np.ndarray:
    a = np.arange(1, n + 1, dtype=float) ** (-p)
    a /= a.sum()
    return a


def family_arcsine(n: int) -> np.ndarray:
    # Discrete arcsine density (peaks at ends, low in middle)
    i = np.arange(n, dtype=float) + 0.5
    a = 1.0 / np.sqrt(i * (n + 0.5 - i))
    a /= a.sum()
    return a


def family_cosine_endpeak(n: int, gamma: float) -> np.ndarray:
    # Peaks at the ends, zero in the middle when gamma=1
    i = np.arange(n, dtype=float)
    a = 1.0 - gamma * np.cos(np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def family_cosine_midpeak(n: int, gamma: float) -> np.ndarray:
    # Peaks in the centre, zero at the ends when gamma=1
    i = np.arange(n, dtype=float)
    a = 1.0 + gamma * np.cos(np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def family_triangular(n: int) -> np.ndarray:
    # Symmetric triangle, peak at centre
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs(i - (n - 1) / 2) / ((n - 1) / 2)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def mix_with_uniform(base: np.ndarray, alpha: float) -> np.ndarray:
    """Return alpha*base + (1-alpha)*uniform, normalised."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n)
    a = alpha * base + (1.0 - alpha) * uniform
    a /= a.sum()
    return a


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Explore a collection of analytic families; keep the best."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # Helper to possibly update best
    def try_candidate(a_candidate: np.ndarray):
        nonlocal best_a, best_score
        score = fast_score(a_candidate, fft_len)
        if score < best_score:
            best_score, best_a = score, a_candidate.copy()

    # --- exponential ---------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.05), num=200):
        a = family_exponential(n, lam)
        try_candidate(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- power‑law -----------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=150):
        a = family_powerlaw(n, p)
        try_candidate(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- arcsine --------------------------------------------------------
    a = family_arcsine(n)
    try_candidate(a)
    if time.time() >= deadline:
        return best_a, best_score

    # --- cosine end‑peak ------------------------------------------------
    for gamma in np.linspace(0.0, 1.0, num=120):
        a = family_cosine_endpeak(n, gamma)
        try_candidate(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- cosine mid‑peak ------------------------------------------------
    for gamma in np.linspace(0.0, 1.0, num=120):
        a = family_cosine_midpeak(n, gamma)
        try_candidate(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- triangular -----------------------------------------------------
    a = family_triangular(n)
    try_candidate(a)
    if time.time() >= deadline:
        return best_a, best_score

    # --- Mixtures with uniform (exponential) ----------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.05), num=30):
        base = family_exponential(n, lam)
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = mix_with_uniform(base, alpha)
            try_candidate(a)
            if time.time() >= deadline:
                return best_a, best_score

    # --- Mixtures with uniform (power‑law) ------------------------------
    for p in np.linspace(0.3, 4.0, num=30):
        base = family_powerlaw(n, p)
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = mix_with_uniform(base, alpha)
            try_candidate(a)
            if time.time() >= deadline:
                return best_a, best_score

    # --- Random convex combinations of three families -------------------
    families = [
        lambda: family_exponential(n, rng.uniform(1e-5, 0.05)),
        lambda: family_powerlaw(n, rng.uniform(0.2, 5.0)),
        lambda: family_arcsine(n),
        lambda: family_triangular(n),
    ]
    for _ in range(80):
        weights = rng.dirichlet(np.ones(len(families)))
        combo = np.zeros(n)
        for w, f in zip(weights, families):
            combo += w * f()
        combo /= combo.sum()
        try_candidate(combo)
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent (mirrored SGD)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 8.0,
):
    """Projected exponentiated‑gradient descent on the true objective."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 1

    while time.time() < deadline:
        g, _ = subgradient(a, fft_len)
        eta = eta0 / np.sqrt(it)

        # multiplicative update (exponentiated gradient)
        exponents = -eta * g
        # clip to avoid overflow/underflow
        exponents = np.clip(exponents, -50, 50)
        a = a * np.exp(exponents)

        # project onto simplex (L1 = 1)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated annealing (coarse + fine)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
    temp_factor: float = 0.02,
    max_frac_start: float = 0.20,
    max_frac_end: float = 0.001,
    recompute_every: int = 2500,
):
    """
    Random mass transfers between two indices with linear cooling.
    Two stages are realised by varying max_fraction in proportion to temperature.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # Initialise temperature as a small fraction of the current score
    temp0 = max(best_score * temp_factor, 1e-7)
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        # Linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        # max_fraction shrinks with temperature
        max_frac = max_frac_end + (max_frac_start - max_frac_end) * (temp / temp0)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # Transfer from larger to smaller entry (helps keep L1 = 1)
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j

        if a[src] <= 0.0:
            continue

        # Random fraction of src mass
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # Incremental convolution update (ignore O(delta^2) term)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()
        dE = new_score - curr_score

        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        # Periodic full recomputation to avoid drift
        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 4000 offers a good trade‑off between quality and speed.
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm start from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Interpolate to the new length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget partitioning
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase0_end = start_time + 0.05 * budget_s          # 5 %
    phase1_end = start_time + 0.78 * budget_s          # 78 %
    phase2_end = deadline                              # remaining ~17 %

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric families
    # ------------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase1_end, n, fft_len, rng, eta0=8.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated annealing (coarse + fine)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # Coarse anneal (higher temperature, larger moves)
        coarse_deadline = time.time() + 0.10 * (phase2_end - time.time())
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            coarse_deadline,
            n,
            rng,
            temp_factor=0.04,
            max_frac_start=0.25,
            max_frac_end=0.02,
            recompute_every=2000,
        )
        print(f"[anneal‑coarse] fast_score={best_score:.6f}")

        # Fine anneal (low temperature, tiny moves)
        if time.time() < phase2_end:
            fine_deadline = phase2_end
            best_a, best_score = simulated_annealing(
                best_a,
                best_score,
                fine_deadline,
                n,
                rng,
                temp_factor=0.005,
                max_frac_start=0.02,
                max_frac_end=0.0005,
                recompute_every=1000,
            )
            print(f"[anneal‑fine] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional polishing step: short EG pass after annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        polish_deadline = deadline
        best_a, best_score = exp_gradient_optimize(
            best_a, polish_deadline, n, fft_len, rng, eta0=4.0
        )
        print(f"[polish‑EG] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for possible reuse in subsequent runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
