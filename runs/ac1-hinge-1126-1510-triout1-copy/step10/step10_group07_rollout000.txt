"""
Propose a low‑score step‑function sequence.

The optimiser works in several complementary phases inside the 1000 s
budget:

1. **Diverse initialisation** – uniform, spectral, random, exponential,
   edge‑biased and a noisy copy of the best known vector.
2. **Adam on a smoothed max‑convolution** – the hard max is replaced by a
   log‑sum‑exp with a decreasing smoothing factor.  Many short restarts
   (with added noise) explore the landscape.
3. **Exponentiated‑gradient descent** on the true (non‑smoothed) max‑
   convolution, using a decaying step size.
4. **Stochastic local search** – repeatedly moves a small amount of mass
   from a “high‑contribution’’ index to a “low‑partner’’ index; the move
   is accepted only if the objective improves.
5. **Deterministic hill‑climb** (as in the previous solution) for
   final fine‑tuning.
6. **Mixture search** – cheap golden‑section line‑search over convex
   combinations of the top‑K candidates discovered so far.
7. The best vector is finally evaluated with the supplied
   `evaluate_sequence` routine.

All vectors are kept on the probability simplex (non‑negative,
sum‑to‑one) by means of soft‑max parametrisation or explicit normalisation.
Fast FFT‑based convolution is used throughout; the programme continuously
checks the time limit and stops gracefully before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → probability simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective
    2·n·max(a*a) assuming Σa = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate for the max‑convolution
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of a*a using log‑sum‑exp.
    Returns (objective value, gradient w.r.t. logits y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 5.0):
    """
    Exponentiated‑gradient descent on the hard max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb (same as in the reference solution)
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate contributing most
    to the maximal convolution entry toward a coordinate with the
    smallest partner value.  Geometric step reduction.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]  (zero outside)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: index with smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep current step size
        else:
            step = max(step * 0.7, min_step)

    return a, score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among a small pool
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair of candidates perform a cheap golden‑section line
    search over mixing weight t∈[0,1] to minimise the bound.
    Returns best vector and its score found during this phase.
    """
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = _eval_mix(c, a, b, fft_len, n)
            fd, _ = _eval_mix(d, a, b, fft_len, n)
            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = _eval_mix(c, a, b, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = _eval_mix(d, a, b, fft_len, n)

            tm = 0.5 * (a0 + b0)
            fs, vec = _eval_mix(tm, a, b, fft_len, n)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()
    return best_a, best_score


def _eval_mix(t, a, b, fft_len, n):
    """Linear mixture of two probability vectors, evaluation and normalisation."""
    mix = (1.0 - t) * a + t * b
    mix /= mix.sum()
    return fast_score(mix, fft_len, n), mix


# ----------------------------------------------------------------------
# Stochastic local search – move mass from high‑contrib to low‑partner
# ----------------------------------------------------------------------
def local_random_search(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator,
                        init_step: float = 0.2,
                        min_step: float = 1e-6):
    """
    Randomised hill‑climbing that moves a fraction of probability mass
    from an index heavily involved in the maximal convolution entry
    toward an index with a small partner value.
    """
    a = a_start.copy()
    best_score = score_start
    step_factor = init_step

    eps_partner = 1e-12

    while time.time() < deadline:
        # compute convolution and locate one maximal index
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        max_idx = int(rng.choice(max_idxs))

        # partner vector for this max_idx
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1)
        partner[idx_range] = a[max_idx - idx_range]

        # contribution of each coordinate to the maximal entry
        contrib = a * partner

        # source index – probability proportional to contribution
        contrib_sum = contrib.sum()
        if contrib_sum > 0.0:
            probs_src = contrib / contrib_sum
            src = int(rng.choice(n, p=probs_src))
        else:
            src = int(rng.choice(n, p=a))

        # target index – favour tiny partner values (avoid src)
        inv_partner = 1.0 / (partner + eps_partner)
        inv_partner[src] = 0.0
        inv_sum = inv_partner.sum()
        if inv_sum > 0.0:
            probs_tgt = inv_partner / inv_sum
            tgt = int(rng.choice(n, p=probs_tgt))
        else:
            # fallback: any index different from src
            cand = list(range(n))
            cand.remove(src)
            tgt = int(rng.choice(cand))

        # amount to transfer
        delta = a[src] * step_factor * rng.random()
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[tgt] += delta
        s = a_new.sum()
        if s <= 0.0:
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            a = a_new
            best_score = new_score
            # after success we can be a bit bolder
            step_factor = min(step_factor * 1.05, 0.5)
        else:
            # shrink step factor on failure
            step_factor = max(step_factor * 0.90, min_step)

    return a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – keep 4000 (the historic size) for a good trade‑off
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise a pool of diverse candidates
    # ------------------------------------------------------------------
    cand0 = np.full(n, 1.0 / n)                         # uniform
    cand1 = spectral_init(n)                           # spectral
    cand2 = random_initial(n, rng)                     # random
    cand3 = random_initial(n, rng)                     # another random
    cand4 = add_noise(cand0, rng.uniform(0.001, 0.01), rng)  # noisy uniform

    candidates = [cand0, cand1, cand2, cand3, cand4]

    # if a previous best exists, interpolate it to current size
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_interp = np.interp(x_new, x_old, prev)
            a_interp = np.maximum(a_interp, 0.0)
            a_interp /= a_interp.sum()
            candidates.append(a_interp)

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Adam with many restarts and a fine lam schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002, 0.00001, 0.000005, 0.000001
    ]
    steps_per_lam = 180
    max_restarts = 12
    restarts_done = 0

    while time.time() < deadline and restarts_done < max_restarts:
        if restarts_done == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len,
            fft_len,
            n,
        )
        print(f"[Adam‑restart {restarts_done+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")
        restarts_done += 1

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient refinement (longer run)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            scale=10.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Stochastic local search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = local_random_search(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
            init_step=0.2,
            min_step=1e-6,
        )
        print(f"[Random‑local] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after random local search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Deterministic hill‑climb (fine tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            init_step=0.15,
            min_step=1e-6,
        )
        print(f"[Hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after hill‑climb: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Convex‑combination (mixture) search among top vectors
    # ------------------------------------------------------------------
    # Assemble a small pool of promising candidates
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # keep the best K = 5
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    K = 5
    selected = ordered[:K]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len,
            deadline,
        )
        print(f"[Mixture‑search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation via the security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose for possible reuse in subsequent runs
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
