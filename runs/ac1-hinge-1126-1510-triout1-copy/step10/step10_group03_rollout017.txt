def targeted_descent(a_start, start_score, deadline, n, fft_len, rng):
    a = a_start.copy()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_score = start_score
    # Precompute current max for early break
    iteration = 0
    while time.time() < deadline:
        # compute subgradient and max conv value
        g, max_val = subgradient(a, fft_len)
        # identify src index with highest gradient and positive a
        # we may restrict to top few indices
        src = np.argmax(g)
        # if a[src] is negligible, find next
        if a[src] <= 0:
            # find a > threshold with max gradient
            candidates = np.where(a > 1e-12)[0]
            if candidates.size == 0:
                break
            src = candidates[np.argmax(g[candidates])]
        dst = np.argmin(g)  # lowest gradient
        # if dst == src, continue
        if src == dst:
            # perturb random indices
            i, j = rng.integers(0, n, size=2)
            src, dst = i, j
        # fraction decays
        fraction = max(0.05 * (0.99 ** iteration), 1e-4)
        delta = a[src] * fraction
        if delta <= 0:
            break
        # update conv using slice
        # conv_new = conv - 2*delta*shift(src) + 2*delta*shift(dst)
        conv_new = conv.copy()
        conv_new[src:src+n] -= 2.0 * delta * a
        conv_new[dst:dst+n] += 2.0 * delta * a
        new_score = 2.0 * n * conv_new.max()
        if new_score < best_score:
            # accept
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            best_score = new_score
            best_a = a.copy()
        # else maybe reject, not update
        iteration += 1
        # occasional recompute conv from scratch to avoid accumulated errors
        if iteration % 5000 == 0:
            conv = np.convolve(a, a)
    return best_a, best_score

def subgradient(a, fft_len):
    n = len(a)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)

    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val

"""
Propose a step‑function sequence with a low value of
    2·n·max(conv(a,a)) / (sum(a))²

The optimiser works in three stages (all respecting a 1000 s budget):

1. **Cheap parametric scan** – tries a few analytic families (exponential,
   linear, mixtures, power‑law) and keeps the best normalised vector.

2. **Mirror‑gradient optimisation** – multiplicative (exponentiated‑gradient)
   descent on the true max‑convolution objective, using the exact sub‑gradient
   (computed via FFT).  This quickly finds a good “exponential‑like’’ shape.

3. **Targeted descent + fine‑grained annealing** – a deterministic step that
   always moves a fraction of mass from the index with the largest gradient
   (i.e. contributing most to the current maximum) to the index with the
   smallest gradient, updating the convolution incrementally.  After the
   deterministic phase a short stochastic annealing sweep tries to escape any
   remaining local minima.

The final sequence is evaluated with the provided `evaluate_sequence`
function and stored in the global variable `height_sequence_1` for possible
reuse by later runs.
"""

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """2·n·max(conv(a,a)) assuming ‖a‖₁ = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of the true max‑convolution objective and the
    current maximal convolution value.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Very lightweight sweep over a few low‑dimensional families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # pure exponential
    for lam in np.logspace(np.log10(1e-6), np.log10(0.02), num=250):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # linear decreasing (triangular)
    for s in np.linspace(0.0, 1.0 / n, num=200):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # exponential + uniform mixture
    for lam in np.logspace(np.log10(1e-6), np.log10(0.02), num=80):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # power‑law decay
    for p in np.linspace(0.2, 5.0, num=120):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent (mirrored SGD)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 5.0,
):
    """Mirrored SGD with multiplicative updates."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 1

    while time.time() < deadline:
        g, _ = subgradient(a, fft_len)
        eta = eta0 / np.sqrt(it)

        exponents = -eta * g
        exponents = np.clip(exponents, -50, 50)  # avoid overflow
        a = a * np.exp(exponents)

        # re‑project onto the simplex
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – deterministic “gradient‑descent’’ mass transfers
# ----------------------------------------------------------------------


def targeted_descent(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """
    Move a tiny fraction of probability mass from the index with the largest
    sub‑gradient (most responsible for the current max) towards the index with
    the smallest sub‑gradient.  Updates the convolution incrementally, so each
    iteration costs O(n) plus the FFTs needed for the sub‑gradient.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)  # full convolution, length 2n‑1
    best_a = a.copy()
    best_score = start_score
    iteration = 0
    # parameters for step size decay
    start_frac = 0.06
    min_frac = 1e-4

    while time.time() < deadline:
        # sub‑gradient and current maximal value
        g, max_val = subgradient(a, fft_len)

        # source – index with biggest gradient (most contributing to max)
        src = int(np.argmax(g))
        # ensure source actually holds some mass
        if a[src] <= 1e-14:
            # pick the largest‑mass index with a positive gradient
            candidates = np.where(a > 1e-14)[0]
            if candidates.size == 0:
                break
            src = int(candidates[np.argmax(g[candidates])])

        # destination – index with smallest gradient
        dst = int(np.argmin(g))
        if src == dst:
            # fall back to a random pair (unlikely, but safe)
            i, j = rng.integers(0, n, size=2)
            src, dst = i, j

        # decaying fraction of the source mass to move
        frac = max(start_frac * (0.995 ** iteration), min_frac)
        delta = a[src] * frac
        if delta <= 0.0:
            break

        # incremental update of the full convolution (linear term only)
        conv_new = conv.copy()
        conv_new[src : src + n] -= 2.0 * delta * a
        conv_new[dst : dst + n] += 2.0 * delta * a
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score:
            # accept the move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            best_score = new_score
            best_a = a.copy()

        iteration += 1

        # occasional full recompute to wipe accumulated numerical error
        if iteration % 3000 == 0:
            conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – simulated‑annealing fine‑tuning (optional)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Classic Metropolis mass‑transfer annealing."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.18   # how much of the source mass may be moved
    recompute_every = 5000
    accepted = 0
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (same style as in targeted_descent)
        conv_new = conv.copy()
        conv_new[src : src + n] -= 2.0 * delta * a
        conv_new[dst : dst + n] += 2.0 * delta * a
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 8000 strikes a good balance between quality and speed.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previous run if one exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Split the overall budget (roughly 5 % / 30 % / 35 % / 30 % )
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # small safety margin

    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.35 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase1_end, n, fft_len, rng, eta0=5.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic targeted descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = targeted_descent(
            best_a, best_score, phase2_end, n, fft_len, rng
        )
        print(f"[targeted‑descent] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # store for possible reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
