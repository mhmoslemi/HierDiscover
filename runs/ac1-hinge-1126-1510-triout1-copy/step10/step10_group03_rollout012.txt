Proposed hill climbing algorithm:

def max_focused_hill_climb(a_start, start_score, deadline, n, rng):
    a = a_start.copy()
    conv = np.convolve(a, a)
    best_score = start_score
    best_a = a.copy()
    max_fraction = 0.03   # fraction of source mass moved per proposal
    recompute_every = 2000
    iteration = 0

    while time.time() < deadline:
        # compute current max conv and index:
        k_max = int(np.argmax(conv))
        M = conv[k_max]

        # compute contributions vector c
        c = np.zeros(n)
        # partner indices for each i
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        # for i in [i_start, i_end], partner = k_max - i
        i_vals = np.arange(i_start, i_end + 1)
        partners = k_max - i_vals
        c[i_vals] = a[i_vals] * a[partners]

        # source index: choose among high contributions
        if c.sum() == 0:
            # if empty, break
            break
        src = rng.choice(n, p=c / c.sum())
        # destination: choose index with low contribution (or random)
        # to avoid picking src again
        low_mask = np.ones(n, dtype=bool)
        low_mask[src] = False
        # weights inverse to contribution + epsilon
        inv_weights = 1.0 / (c + 1e-12)
        inv_weights[src] = 0
        dst = rng.choice(n, p=inv_weights / inv_weights.sum())
        # ensure source has mass
        if a[src] <= 0:
            # try another random src
            continue
        # determine delta
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0:
            continue
        # incremental conv update:
        diff = np.zeros_like(conv)
        diff[src:src+n] -= a
        diff[dst:dst+n] += a
        conv_new = conv + 2.0 * delta * diff
        new_M = conv_new.max()
        # compute new score (fast_score will be 2*n*new_M, but compute directly)
        new_score = 2.0 * n * new_M
        # accept if improves (lower)
        if new_score < best_score or rng.random() < np.exp(-(new_score - best_score) / (0.001 + best_score)):
            # accept
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # occasionally recompute conv
        if iteration % recompute_every == 0:
            conv = np.convolve(a, a)
        iteration += 1
```

def max_focused_hill_climb(a_start, start_score, deadline, n, rng):
    # start_score is fast_score (2*n*max)
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score
    start_time = time.time()
    temp0 = max(best_score * 0.02, 1e-6)
    max_fraction = 0.03
    recompute_every = 5000
    iteration = 0
    while time.time() < deadline:
        # compute k_max, M
        k_max = int(np.argmax(conv))
        M = conv[k_max]
        # contributions
        c = np.zeros(n)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_vals = np.arange(i_start, i_end + 1)
        partners = k_max - i_vals
        c[i_vals] = a[i_vals] * a[partners]
        # choose src (high product)
        c_sum = c.sum()
        if c_sum == 0:
            break
        src = rng.choice(n, p=c / c_sum)
        # choose dst (low product)
        inv_weights = 1.0 / (c + 1e-15)
        inv_weights[src] = 0  # cannot choose src as dest
        inv_sum = inv_weights.sum()
        if inv_sum == 0:
            # choose random dst not src
            dst_candidates = np.delete(np.arange(n), src)
            dst = rng.choice(dst_candidates)
        else:
            dst = rng.choice(n, p=inv_weights / inv_sum)
        if a[src] <= 0:
            continue
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0:
            continue
        # incremental conv update
        conv_new = conv.copy()
        conv_new[src:src+n] -= 2.0 * delta * a
        conv_new[dst:dst+n] += 2.0 * delta * a
        new_score = 2.0 * n * conv_new.max()
        # temperature
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)
        accept = False
        if new_score < curr_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - curr_score) / temp):
                accept = True
        if accept:
            # apply move
            a[src] -= delta
            a[dst] += delta
            # avoid negative due to rounding
            if a[src] < 0:
                a[src] = 0.0
            conv = conv_new
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()
        # occasional recompute
        if iteration % recompute_every == 0:
            conv = np.convolve(a, a)
            curr_score = 2.0 * n * conv.max()
        iteration += 1
    return best_a, best_score

"""
Candidate generator for the step‑function minimisation task.

Algorithm
---------
1. **Parameter sweep** – a cheap exploration of several analytic families
   (exponential, linear ramp, mixtures, power‑law and a two‑parameter
   exponential mixture).  The best candidate becomes the seed for the next
   phase.

2. **Exponentiated‑gradient descent with momentum** – a mirror‑SGD on the true
   ‑∞‑norm objective.  The gradient is the sub‑gradient of the maximal
   convolution entry (computed by FFT).  A momentum term (β=0.9) stabilises
   the updates and a decaying step‑size η₀/√t is used.

3. **Max‑focused hill‑climbing** – a specialised simulated‑annealing
   routine that concentrates on the shift where the convolution attains its
   maximum.  At each iteration the algorithm moves a small amount of mass
   from an index that contributes strongly to the current peak to an index
   that contributes weakly, accepting moves according to a linear cooling
   schedule.  Incremental O(n) convolution updates keep the loop cheap.

4. **Optional final polishing** – a short run of the exponentiated‑gradient
   optimiser with a small learning‑rate to clean up any remaining noise.

All phases respect a global time budget of ≈1000 s.  The best sequence found
is returned and stored in the global variable ``height_sequence_1`` for
reuse by subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming ‖a‖₁ = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of the true max‑convolution objective and the
    current maximal convolution value.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families (including a double‑exponential)
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Brute‑force over a few low‑dimensional families; return best found."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # --- pure exponential -------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=250):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- linear decreasing ------------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=250):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- mixture exponential + uniform ------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=60):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=13):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # --- power‑law decay --------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=200):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- double exponential mixture ---------------------------------------
    # a = alpha * exp(-lam1*i) + (1-alpha) * exp(-lam2*i)
    lam_vals = np.logspace(np.log10(1e-5), np.log10(0.02), num=30)
    for lam1 in lam_vals[::5]:          # coarse sampling
        for lam2 in lam_vals[::5]:
            if lam2 <= lam1:
                continue
            for alpha in np.linspace(0.0, 1.0, num=7):
                a = alpha * np.exp(-lam1 * np.arange(n)) + (1 - alpha) * np.exp(-lam2 * np.arange(n))
                a /= a.sum()
                score = fast_score(a, fft_len)
                if score < best_score:
                    best_score, best_a = score, a.copy()
                if time.time() >= deadline:
                    return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent with momentum
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
):
    """Mirrored SGD (multiplicative update) on max‑convolution, with momentum."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)   # momentum buffer
    beta = 0.9
    it = 1

    while time.time() < deadline:
        g, _ = subgradient(a, fft_len)
        # momentum update
        m = beta * m + (1.0 - beta) * g
        eta = eta0 / np.sqrt(it)

        exponents = -eta * m
        exponents = np.clip(exponents, -50, 50)   # avoid overflow
        a = a * np.exp(exponents)

        # re‑project onto the simplex
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score

def max_focused_hill_climb(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Hill‑climbing that moves mass away from indices that dominate the current
    maximal convolution entry.  Acceptance follows a linear‑cooling schedule.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score  # this is 2·n·max(conv)
    best_a = a.copy()
    best_score = curr_score

    start_time = time.time()
    temp0 = max(best_score * 0.10, 1e-6)   # a bit hotter than the original SA
    max_fraction = 0.04                  # max portion of source mass moved per step
    recompute_every = 4000
    iteration = 0

    while time.time() < deadline:
        # locate current peak
        k_max = int(np.argmax(conv))

        # contributions to the peak
        c = np.zeros(n)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        i_vals = np.arange(i_start, i_end + 1)
        partners = k_max - i_vals
        c[i_vals] = a[i_vals] * a[partners]

        total_c = c.sum()
        if total_c == 0.0:
            break

        # source index – weighted by contribution
        src = rng.choice(n, p=c / total_c)

        # destination – weighted inversely by contribution
        inv_w = 1.0 / (c + 1e-15)
        inv_w[src] = 0.0
        inv_total = inv_w.sum()
        if inv_total == 0.0:
            # fallback to a completely random destination
            dst_candidates = np.delete(np.arange(n), src)
            dst = rng.choice(dst_candidates)
        else:
            dst = rng.choice(n, p=inv_w / inv_total)

        if a[src] <= 0.0:
            continue

        # mass to move
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (linear term only)
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * delta * a
        conv_new[dst:dst + n] += 2.0 * delta * a
        new_score = 2.0 * n * conv_new.max()

        # temperature (linear cooling)
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        accept = False
        if new_score < curr_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - curr_score) / temp):
                accept = True

        if accept:
            # apply move
            a[src] -= delta
            a[dst] += delta
            # numerical safety
            if a[src] < 0.0:
                a[src] = 0.0
            conv = conv_new
            curr_score = new_score
            if curr_score < best_score:
                best_score = curr_score
                best_a = a.copy()

        if iteration % recompute_every == 0:
            conv = np.convolve(a, a)
            curr_score = 2.0 * n * conv.max()

        iteration += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Optional monotone projection (non‑increasing sequence)
# ----------------------------------------------------------------------


def enforce_monotone_decreasing(a: np.ndarray) -> np.ndarray:
    """
    Project onto the set of non‑increasing vectors by isotonic regression.
    Afterwards renormalise to keep the L¹‑norm equal to 1.
    """
    b = np.maximum.accumulate(a[::-1])[::-1]  # non‑increasing
    b_sum = b.sum()
    if b_sum == 0.0:
        return a
    return b / b_sum


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if one exists.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the required length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase0_end = start_time + 0.05 * budget_s          # 5 %
    phase1_end = start_time + 0.55 * budget_s          # 50 % total (45 % for phase‑1)
    phase2_end = start_time + 0.95 * budget_s          # 40 % total (30 % for phase‑2)

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_a, best_score = cand_a.copy(), cand_score
            print(f"[param‑scan] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient descent (momentum)
    # --------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase1_end, n, fft_len, rng, eta0=6.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – max‑focused hill‑climbing
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = max_focused_hill_climb(
            best_a, best_score, phase2_end, n, rng
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – brief cleanup with a tiny exponentiated‑gradient run
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, rng, eta0=2.0
        )
        print(f"[final‑exp] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Optional monotone projection (sometimes beneficial)
    # --------------------------------------------------------------
    proj_a = enforce_monotone_decreasing(best_a)
    proj_score = fast_score(proj_a, fft_len)
    if proj_score < best_score:
        best_a, best_score = proj_a, proj_score
        print(f"[monotone] fast_score improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
