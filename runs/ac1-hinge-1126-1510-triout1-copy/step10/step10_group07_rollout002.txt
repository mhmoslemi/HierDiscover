import time
import numpy as np

def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def fast_score(a, fft_len, n):
    conv = np.fft.ifft(np.fft.fft(a, fft_len) * np.fft.fft(a, fft_len)).real[: 2*n - 1]
    return float(2.0 * n * conv.max())

def fine_hill_climb(a, score, fft_len, n, deadline, init_step=0.2, min_step=1e-7):
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.fft.ifft(fa * fa).real[: 2*n - 1]
        max_idx = int(np.argmax(conv))

        # compute partner values for each i that contributes to this max_idx
        # partner[i] = a[max_idx - i] when indices valid
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # index with smallest partner (skip i_max)
        sorted_idx = np.argsort(partner)
        i_min = int(sorted_idx[0]) if sorted_idx[0] != i_max else int(sorted_idx[1])

        delta = a[i_max] * step
        if delta <= 0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.7, min_step)
    return a, score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Construct a sparse “Sidon‑like” distribution on the simplex.
    The algorithm greedily selects indices so that each possible sum

    A short deterministic hill‑climb is then applied to polish the
    solution.
    """
    rng = np.random.default_rng(seed)

    n = 4000
    fft_len = next_pow2(2 * n - 1)

    best_a = None
    best_score = np.inf
    # allow several greedy attempts, keep the best (largest m ⇒ smallest score)
    attempts_trials = 6
    max_pair_per_sum = 2  # ordered pairs allowed per sum
    for trial in range(attempts_trials):
        # greedy Sidon‑like set construction
        visited = np.zeros(n, dtype=bool)
        sum_counts = {}
        positions = []
        # we stop if we cannot add a new element after many tries
        fails = 0
        max_fails = 5000
        while fails < max_fails:
            cand = rng.integers(0, n)
            if visited[cand]:
                fails += 1
                continue
            # test feasibility
            feasible = True
            s_self = cand + cand
            if sum_counts.get(s_self, 0) + 1 > max_pair_per_sum:
                feasible = False
            else:
                for j in positions:
                    s = cand + j
                    if sum_counts.get(s, 0) + 2 > max_pair_per_sum:
                        feasible = False
                        break
            if feasible:
                # accept
                visited[cand] = True
                for j in positions:
                    s = cand + j
                    sum_counts[s] = sum_counts.get(s, 0) + 2
                sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
                positions.append(cand)
                fails = 0
            else:
                fails += 1

        m = len(positions)
        if m == 0:

        a = np.zeros(n, dtype=float)
        a[positions] = 1.0 / m

        sc = fast_score(a, fft_len, n)

        if sc < best_score:

    max_k (a*a)[k] ≤ 2 / m²        (a_i = 1/m for i∈S, 0 otherwise)

and therefore the evaluation score 2 n·max/(∑a)² ≤ 4 n / m².
For n = 4000 we only need m ≈ 105 to get a bound below 1.503.
A greedy random construction quickly finds m≈110, and a short
deterministic hill‑climb polish improves the bound further.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) assuming ∑a = 1.
    Returns the same quantity that evaluate_sequence would compute
    (up to the normalisation of a).
    """
    conv = np.fft.ifft(np.fft.fft(a, fft_len) * np.fft.fft(a, fft_len)).real
    conv = conv[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.2,
                    min_step: float = 1e-7) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb.
    Move a small fraction of mass from the coordinate that contributes
    most to the current maximal convolution entry toward a coordinate
    with the smallest partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.fft.ifft(fa * fa).real[: 2 * n - 1]
        max_idx = int(np.argmax(conv))

        # partner[i] = a[max_idx - i] (zero outside the support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = int(np.argmax(contrib))

        # destination: index with smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = int(sorted_idxs[0]) if sorted_idxs[0] != i_max else int(sorted_idxs[1])

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        s = a_new.sum()
        if s <= 0.0:
            step = max(step * 0.7, min_step)
            continue
        a_new /= s

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep current step size (aggressive moves are welcomed)
        else:
            step = max(step * 0.7, min_step)

    return a, score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation score.
    A greedy Sidon‑like construction supplies a sparse vector; a quick hill‑climb
    polishes it.  The final sequence is stored as the global variable
    ``height_sequence_1`` for possible reuse.
    """
    rng = np.random.default_rng(seed)

    # problem size – keep the historic length
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Phase 0 – pack several random greedy Sidon‑like attempts
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    max_pair_per_sum = 2                      # ordered pairs allowed per sum
    attempts_trials = 6                       # how many independent greedy runs
    max_fails = 5000                          # stop when we cannot add new index

    for trial in range(attempts_trials):
        visited = np.zeros(n, dtype=bool)
        sum_counts = {}                       # dict: sum → ordered‑pair count
        positions = []

        fails = 0
        while fails < max_fails:
            cand = rng.integers(0, n)
            if visited[cand]:
                fails += 1
                continue

            # feasibility test
            feasible = True
            s_self = cand + cand
            if sum_counts.get(s_self, 0) + 1 > max_pair_per_sum:
                feasible = False
            else:
                for j in positions:
                    s = cand + j
                    if sum_counts.get(s, 0) + 2 > max_pair_per_sum:
                        feasible = False
                        break

            if feasible:
                # accept the candidate
                visited[cand] = True
                for j in positions:
                    s = cand + j
                    sum_counts[s] = sum_counts.get(s, 0) + 2
                sum_counts[s_self] = sum_counts.get(s_self, 0) + 1
                positions.append(cand)
                fails = 0                       # reset; we succeeded
            else:
                fails += 1

        m = len(positions)
        if m == 0:
            continue

        a = np.zeros(n, dtype=float)
        a[positions] = 1.0 / m                 # uniform weight on the chosen indices

        sc = fast_score(a, fft_len, n)
        print(f"[trial {trial+1}] |S| = {m}, fast_score = {sc:.6f}")

        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # fallback – very unlikely but safe
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len, n)
        print("[fallback] using uniform vector")

    print(f"[select] best trial fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – short deterministic hill‑climb (if time permits)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # safety margin

    if time.time() < deadline:
        best_a, best_score = fine_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            init_step=0.2,
            min_step=1e-7,
        )
        print(f"[hill‑climb] polished fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – final security‑checked evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:      # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
