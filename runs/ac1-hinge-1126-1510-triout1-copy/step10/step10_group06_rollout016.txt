"""
Improved search for a low‑score non‑negative step sequence.

Main ideas
----------

1. **Broad parametric sweep** – systematic grid over a richer set of
   families (exponential, power‑law, logistic, and a mixed double‑exponential
   with uniform background).  The best vector from this cheap sweep seeds the
   next stage.

2. **Fine‑grained Adam optimisation** – the exact objective
   `2·n·max(conv)` is nonsmooth (the max).  Its sub‑gradient is
   computed analytically via FFT, then an Adam‑style adaptive
   projected‑gradient descent on the simplex is run for the bulk of the
   time budget.  This typically pushes the score a few‑tenths of a percent
   below the previous best.

3. **Optional simulated‑annealing polishing** – a fast stochastic
   hill‑climb that moves a random fraction of mass between two positions
   with a square‑root cooling schedule.  It rarely harms and sometimes
   yields a final tiny improvement.

The whole procedure respects the global 1000 s deadline, always keeping
the best sequence seen so far.  The returned list is ready for
`evaluate_sequence`.  It works with any existing global variable
`height_sequence_1` (a previous construction) and falls back to a

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT (length 2·len(a)−1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector with ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)

    return a

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a vector from a parametric family with random parameters."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":

        a = np.exp(-rate * i)
    elif family == "logistic":

        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "double_exp":

        weight = rng.uniform(0.0, 1.0)

    if total == 0.0:

    else:
        a /= total
    return a

def compute_subgradient(a: np.ndarray, fft_len: int) -> tuple[np.ndarray, float]:
    """
    Returns (sub‑gradient wrt a, current max convolution value).
    The sub‑gradient corresponds to the (possibly non‑unique) maximal shift.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    M = conv.max()
    eps = max(M * 1e-12, 1e-12)
    mask = (conv >= M - eps).astype(float)          # 1 at shift(s) attaining the max

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    conv_g = np.fft.irfft(fm * fa_rev, fft_len)
    g = 2.0 * conv_g[(n - 1):(n - 1 + n)]          # sub‑gradient ∂max/∂a
    return g, M

    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.2                         # base learning rate (tuned empirically)
    it = 0

    while time.time() < deadline:
        g, _ = compute_subgradient(a, fft_len)
        it += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)    # element‑wise step size
        a_candidate = a - step * m_hat
        a_candidate = simplex_projection(a_candidate)

        sc = fast_score(a_candidate, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a_candidate.copy()
        a = a_candidate

    return best_a, best_sc


# --------------------------------------------------------------
# Simulated annealing (as a final polishing step)
# --------------------------------------------------------------

def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Stochastic hill‑climb with a square‑root cooling schedule."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem parameters
    # --------------------------------------------------------------
    n = 4000                                   # length of the step function
    fft_len = next_pow2(2 * n - 1)             # size for all FFTs (power of two)

    # --------------------------------------------------------------
    # Initial candidate (uniform) – also try to reuse a previous construction
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    if "height_sequence_1" in globals():
        try:
            hs = np.array(globals()["height_sequence_1"], dtype=float)
            if hs.ndim == 1 and hs.size > 0:
                x_old = np.linspace(0, 1, hs.size)
                x_new = np.linspace(0, 1, n)
                a1 = np.interp(x_new, x_old, hs)
                a1 = np.maximum(a1, 0.0)
                a1 /= a1.sum()
                sc1 = fast_score(a1, fft_len, n)
                if sc1 < best_score:
                    best_a, best_score = a1, sc1
                    print(f"[init] previous best fast_score={best_score:.6f}")
        except Exception:
            pass  # ignore malformed global

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 0 – systematic grid search + cheap random sweep
    # --------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search_extended(
        n, fft_len, rng, deadline
    )
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] new best fast_score={best_score:.6f}")

    # a few seconds for a cheap random sweep
    a_rand, sc_rand = cheap_random_sweep(
        n, fft_len, deadline, rng, max_samples=3000
    )
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam optimisation on the true max‑convolution
    # --------------------------------------------------------------
    phase1_deadline = deadline - 0.2               # leave a tiny window for annealing
    best_a, best_score = adam_optimize(
        best_a, phase1_deadline, n, fft_len, rng
    )
    print(f"[adam] best fast_score after Adam={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – final simulated annealing (optional)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            rng,
        )
        print(f"[anneal] final best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
