"""
Propose a low‑score step‑function sequence.

The search proceeds in several stages, all bounded by the 1000 s budget.
1.  Build a diverse pool of initial vectors (uniform, spectral, random,
    exponential, power‑law, edge‑heavy, and the previous best if available).
2.  Run many Adam restarts on a log‑softmax parametrisation of the
    smooth‑max surrogate of the self‑convolution.  The smoothing factor
    (lam_factor) is decreased gradually; each restart performs several
    thousand gradient steps.
3.  Refine the best vector with a long sub‑gradient (exponentiated‑gradient)
    descent on the true max‑convolution objective.
4.  Perform a stochastic hill‑climbing phase that repeatedly moves a random
    fraction of mass from a random source index to a random target index,
    accepting only improvements.
5.  Combine the top few discovered vectors by a cheap golden‑section line
    search over convex mixtures.
6.  Return the best vector (as a plain list) and store it under the historic
    name ``height_sequence_1`` for possible reuse.

All helper functions are top‑level, use only NumPy/ SciPy, and avoid closures
or lambda‑expressions, satisfying the execution constraints.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility helpers (no closures, no lambdas)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random probability vector."""
    a = rng.random(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def exponential_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.001, 0.02)
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def powerlaw_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.2, 2.5)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def edge_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    edge_frac = rng.uniform(0.006, 0.03)
    m = max(1, int(edge_frac * n))
    a = np.full(n, 1.0 / n)
    heavy = np.sqrt(n / 2.0) / n
    a[:m] = heavy
    a[-m:] = heavy
    a /= a.sum()
    return a


def bimodal_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    centre = n // 2
    width = rng.integers(int(0.02 * n), int(0.07 * n))
    left = max(0, centre - width)
    right = min(n, centre + width)
    a = np.full(n, 1.0 / n)
    peak = rng.uniform(0.5, 1.5) / n
    a[left:right] = peak
    a /= a.sum()
    return a


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    a_sum = a.sum()
    if a_sum <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed surrogate
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 5.0):
    """
    Sub‑gradient descent on the true max‑convolution using an
    exponentiated (mirror‑descent) update.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (ordered‑pair version)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Stochastic hill‑climb (random mass transfers)
# ----------------------------------------------------------------------
def random_hill_climb(a: np.ndarray,
                      score: float,
                      fft_len: int,
                      n: int,
                      deadline: float,
                      rng: np.random.Generator,
                      max_iters: int = 200000):
    """
    Repeatedly move a random fraction of mass from a random source index
    to a random target index, keeping changes only if they improve the bound.
    """
    while time.time() < deadline and max_iters > 0:
        # pick source with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        j = rng.integers(0, n)
        if j == i:
            max_iters -= 1
            continue

        # transfer a random fraction of a[i] (up to 40 %)
        delta = a[i] * rng.random() * 0.4
        if delta <= 0.0:
            max_iters -= 1
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum unchanged → no renormalisation needed

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
        max_iters -= 1
    return a, score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) exploration among top candidates
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """Golden‑section line search over convex mixes of each pair of candidates."""
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)

            fc, _ = _eval_mix(c, a, b, fft_len, n)
            fd, _ = _eval_mix(d, a, b, fft_len, n)

            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = _eval_mix(c, a, b, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = _eval_mix(d, a, b, fft_len, n)

            t_mid = 0.5 * (a0 + b0)
            f_mid, vec_mid = _eval_mix(t_mid, a, b, fft_len, n)
            if f_mid < best_score:
                best_score = f_mid
                best_a = vec_mid.copy()
    return best_a, best_score


def _eval_mix(t, a, b, fft_len, n):
    """Helper: evaluate convex combination."""
    mix = (1.0 - t) * a + t * b
    mix /= mix.sum()
    return fast_score(mix, fft_len, n), mix

# Main entry point required by the task
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep a reasonably large power‑of‑two for FFT efficiency.
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from the historic global variable if present.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Build an initial pool of diverse seeds.
    # ------------------------------------------------------------------
    pool = [
        a_start.copy(),
        spectral_init(n),
        random_initial(n, rng),
        exponential_initial(n, rng),
        powerlaw_initial(n, rng),
        edge_initial(n, rng),
        bimodal_initial(n, rng)
    ]

    # evaluate quick scores and keep the best few
    scores = [fast_score(c, fft_len_conv, n) for c in pool]
    best_idx = int(np.argmin(scores))
    best_a = pool[best_idx].copy()
    best_score = scores[best_idx]
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Adam restarts (longer schedule)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.20, 0.15, 0.10, 0.07, 0.05,
        0.035, 0.025, 0.018, 0.012, 0.008,
        0.005, 0.003, 0.0018, 0.0010,
        0.0006, 0.0003, 0.00015, 0.00007, 0.00003
    ]
    steps_per_lam = 300
    max_restarts = 15
    restarts_done = 0
    while time.time() < deadline and restarts_done < max_restarts:
        # perturb current best a little before each restart
        if restarts_done == 0:
            a0 = best_a.copy()
        else:
            sigma = rng.uniform(0.001, 0.02)
            a0 = add_noise(best_a, sigma, rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam restart {restarts_done + 1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")
        restarts_done += 1

    # ------------------------------------------------------------------
    # Phase 2 – Sub‑gradient (exponentiated‑gradient) refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            scale=5.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Random hill‑climb (mass transfers)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = random_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
            max_iters=200000
        )
        print(f"[Random hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after hill‑climb: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Convex‑combination search among top candidates
    # ------------------------------------------------------------------
    # Gather a handful of promising vectors (including a few fresh random ones)
    extra_vectors = [random_initial(n, rng) for _ in range(4)]
    extra_scores = [fast_score(v, fft_len_conv, n) for v in extra_vectors]

    candidate_vecs = [best_a] + extra_vectors
    candidate_scores = [best_score] + extra_scores

    # keep the best K = 6 vectors for mixture search
    ordered = sorted(zip(candidate_scores, candidate_vecs), key=lambda x: x[0])
    top_k = 6
    selected = ordered[:top_k]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len_conv,
            deadline
        )
        print(f"[Mixture search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the security‑checked routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluation raised an exception:", exc)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
