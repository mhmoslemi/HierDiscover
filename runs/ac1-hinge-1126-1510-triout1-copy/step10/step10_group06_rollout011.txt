def exponentiated_gradient_optimize(a0, deadline, n, fft_len_conv, fft_len_g):
   a = a0.copy()
   best_a = a.copy()
   best_score = fast_score(a, fft_len_conv, n)
   step_counter = 0
   while time.time() < deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa*fa))[: 2*n -1]
        max_val = conv.max()
        # compute mask for indices where conv ~ max (within tolerance)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute subgradient via convolution with mask reversed
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n-1):(n-1+n)]
        # exponentiated update:
        eta = 0.3 / np.sqrt(step_counter + 1)
        a = a * np.exp(-eta * g)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        # evaluate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
        step_counter += 1
   return best_a, best_score

"""
Propose a low‐score step sequence.

The approach consists of five stages (all time‑controlled):

1. **Quick parametric sweep** – systematic grid over several families
   (exponential, power‑law (both signs), linear ramps, logistic, Gaussian,
   double‑exponential) for three different lengths (2000, 3000, 4000).
   The best vector (lowest `fast_score`) is kept.

2. **Exponentiated‑gradient descent** – true max‑convolution sub‑gradient
   is used in an exponentiated‑gradient (mirror‑descent on the simplex)
   update `a ← a·exp(−η·g)`.  A decaying step‑size η≈0.3/√t and occasional
   random jitter provide aggressive local improvement.

3. **L‑BFGS refinement on a smoothed max** – the nonsmooth objective
   `max(conv)` is approximated by a log‑sum‑exp with a decreasing smoothing
   parameter λ.  The continuation runs a sequence of λ values, each
   starting from the result of the previous stage.

4. **Projected sub‑gradient descent with momentum** – as a safety net
   (identical to the original implementation) to escape possible plateaus.

5. **Long simulated annealing** – stochastic hill‑climb that moves a random
   fraction (≤30 %) of mass between two positions, with a square‑root
   cooling schedule.  The temperature is tuned so that many uphill moves
   are still possible early on.

All stages respect the global deadline (`budget_s` seconds).  The best
sequence seen at any moment is returned, and a final call to
`evaluate_sequence` yields the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (used by the L‑BFGS stage)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) assuming ∑a = 1 (used for internal optimisation)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_linear(n: int, inc: bool = True) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=float)
    if not inc:
        i = i[::-1]
    a = i
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_double_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_random(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a /= a.sum()
    return a


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Systematic grid over several families for a fixed length `n`.
    Returns the best normalized vector and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # ----- exponential ----------------------------------------------------
    for rate in np.linspace(0.0002, 0.0020, 31):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # ----- power‑law negative (decreasing) ---------------------------------
    for p in np.linspace(-5.0, -0.2, 25):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power‑neg p={p:.2f} → score={best_sc:.6f}")

    # ----- power‑law positive (increasing) ---------------------------------
    for p in np.linspace(0.2, 4.0, 20):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=False)   # decreasing from left → increasing shape
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power‑pos p={p:.2f} → score={best_sc:.6f}")

    # ----- linear ramps ---------------------------------------------------
    for inc in (True, False):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_linear(n, inc=inc)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] linear inc={inc} → score={best_sc:.6f}")

    # ----- double exponential ---------------------------------------------
    for rate in np.linspace(0.0002, 0.0015, 15):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_double_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] double_exp rate={rate:.5f} → score={best_sc:.6f}")

    # ----- logistic --------------------------------------------------------
    mus = np.linspace(0.2 * n, 0.8 * n, 9)
    sigmas = np.linspace(300.0, 1500.0, 9)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # ----- Gaussian ---------------------------------------------------------
    for sigma in np.linspace(300.0, 2000.0, 12):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_gauss(n, sigma)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] gauss sigma={sigma:.0f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(

    best_a = np.full(n, 1.0 / n)

    start = time.time()

    families = [

    while time.time() < deadline:

# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)  # logits → simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def exponentiated_gradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using exponentiated gradient
    and the exact sub‑gradient of `max(conv)`.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    t = 0
    while time.time() < deadline:
        # compute convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max (same as in the original code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # exponentiated‑gradient update
        eta = 0.30 / np.sqrt(t + 1.0)
        # clip gradient to avoid overflow in exp
        g_clip = np.clip(g, -1e4, 1e4)
        a = a * np.exp(-eta * g_clip)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # evaluate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
        t += 1

    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)  # momentum buffer
    beta = 0.85
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with a square‑root cooling schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature as a noticeable fraction of the current score
    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30  # up to 30 % move
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # safety margin

    # ------------------------------------------------------------------
    # 0️⃣  Use a previously‑found construction if it exists
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    best_n = None
    best_fft_len_conv = None
    best_fft_len_g = None

    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate to each candidate length later
            pass  # will be considered after the length loop

    # ------------------------------------------------------------------
    # 1️⃣  Scan a few different lengths (2000, 3000, 4000)
    # ------------------------------------------------------------------
    for n in (2000, 3000, 4000):
        if time.time() >= deadline:
            break
        fft_len_conv = next_pow2(2 * n - 1)
        # quick parametric exploration
        a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
        if sc_grid < best_score:
            best_score, best_a, best_n = sc_grid, a_grid.copy(), n
            best_fft_len_conv = fft_len_conv
            best_fft_len_g = next_pow2(3 * n - 2)
            print(f"[grid n={n}] best fast_score={best_score:.6f}")

        # cheap random sweep (small time slice)
        a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
        if sc_rand < best_score:
            best_score, best_a, best_n = sc_rand, a_rand.copy(), n
            best_fft_len_conv = fft_len_conv
            best_fft_len_g = next_pow2(3 * n - 2)
            print(f"[rand n={n}] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  If a previous construction exists, try interpolating it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # attempt interpolation to the chosen length `best_n`
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, best_n)
            a_old = np.interp(x_new, x_old, hs)
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            sc_old = fast_score(a_old, best_fft_len_conv, best_n)
            if sc_old < best_score:
                best_a, best_score = a_old, sc_old
                print(f"[init] interpolated previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient descent (dominant phase)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    while time.time() < phase1_end:
        # occasional jitter around the current best to explore neighbourhoods
        if rng.random() < 0.30:
            a_start = best_a * np.exp(rng.normal(0.0, 0.02, size=best_n))
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            # fresh random start from a random family
            fam = rng.choice(["exp", "power_neg", "power_pos", "linear"])
            if fam == "exp":
                rate = rng.uniform(0.0002, 0.0020)
                a_start = generate_exp(best_n, rate)
            elif fam == "power_neg":
                p = rng.uniform(-5.0, -0.2)
                a_start = generate_power(best_n, p, inc=True)
            elif fam == "power_pos":
                p = rng.uniform(0.2, 4.0)
                a_start = generate_power(best_n, p, inc=False)
            else:  # linear
                inc = rng.choice([True, False])
                a_start = generate_linear(best_n, inc=inc)

        a_opt, sc_opt = exponentiated_gradient_optimize(
            a_start, phase1_end, best_fft_len_conv, best_fft_len_g, best_n
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[eg] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  L‑BFGS refinement on a smoothed max
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0005,
        0.0002, 0.0001, 5e-05, 2e-05,
        1e-05, 5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 1800
    while time.time() < phase2_end:
        a_opt, sc_opt = lbfgs_continuation(
            best_a,
            lam_schedule,
            maxiter_per_lam,
            phase2_end,
            best_fft_len_conv,
            best_fft_len_g,
            best_n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[lbfgs] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Projected sub‑gradient descent with momentum (fallback)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = subgradient_optimize_momentum(
            best_a, deadline, best_fft_len_conv, best_fft_len_g, best_n
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[sub‑grad] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Simulated annealing (final stochastic polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            best_fft_len_conv,
            best_n,
            deadline,
            rng,
        )
        print(f"[anneal] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
