def adam_optimize(a_start, max_iters, deadline, fft_len, fft_len_g, n, lam_factor_initial=0.5, lam_factor_final=1e-5):

    y = np.log(a_start + eps)

    m = np.zeros_like(y)

    v = np.zeros_like(y)

    beta1, beta2 = 0.9, 0.999

    t = 0

    best_a = a_start.copy()

    best_score = fast_score(best_a, fft_len, n)

    while time.time() < deadline and t < max_iters:

        t += 1

        a = softmax(y)

        # compute convolution to get max_c
        fa = np.fft.fft(a, fft_len)
        conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_c = conv_full.max()

        lam = max_c * lam_factor_initial  # maybe vary lam per schedule

        # compute smooth max gradient using earlier function objective_and_grad with lam factor maybe
        # Actually we have function objective_and_grad(y,..., lam_factor) expects lam_factor scaling factor of max_c.
        # It uses lam_factor as lam_factor (i.e., lam = max_c * lam_factor + 1e-12). So we can pass lam_factor= current factor.

        f, grad_y = objective_and_grad(y, n, fft_len, fft_len_g, lam_factor=lam_factor_initial)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** t)

"""
Search for a low‑score non‑negative step sequence.

The algorithm consists of four time‑controlled phases:

1. **Systematic parametric sweep** – tests a handful of one‑parameter families
   (exponential, power, logistic, Gaussian, triangular, sinusoidal).  
   The best vector from this cheap grid becomes the starting point.

2. **Adam on a log‑sum‑exp smooth max** – the true objective
   ``2·n·max(conv)`` is approximated by a smooth max.
   A sequence of decreasing smoothing factors (λ) is used and the logits
   are updated with the Adam optimiser.  This phase drives the solution
   close to a local optimum of the true L∞‑objective.

3. **Projected sub‑gradient descent with momentum** – true max‑convolution
   sub‑gradients are computed with FFT, projected onto the probability simplex
   and used with an adaptive step‑size schedule.  It refines the result of
   phase 2.

4. **Guided simulated annealing** – a stochastic hill‑climb that moves a
   random fraction of mass between two indices, with the source chosen from
   the largest entries of the sub‑gradient (most responsible for the current
   peak) and the destination from the smallest entries.  A square‑root
   cooling schedule keeps the temperature high enough for uphill moves.

All phases respect the global deadline (default 1000 s).  The best sequence
encountered is returned; its official score is printed at the end.
"""

import time
import numpy as np
from numpy.fft import fft, ifft

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector with sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------

def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = fft(a_rev, fft_len_g)
    fw = fft(w, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a /= a.sum()
    return a


def generate_gaussian(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def generate_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    a = i * (n + 1 - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_sine(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a = np.maximum(a, 0.0) ** power
    a /= a.sum()
    return a


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Systematic search over a handful of simple families.
    Returns the best vector found.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential
    for rate in np.linspace(2e-4, 1e-3, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → {best_sc:.6f}")

    # power‐law
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → {best_sc:.6f}")

    # logistic
    mus = np.linspace(0.2 * n, 0.8 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} σ={sigma:.0f} → {best_sc:.6f}")

    # Gaussian
    sig_vals = np.linspace(200.0, 2000.0, 9)
    for sigma in sig_vals:
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_gaussian(n, (n - 1) / 2.0, sigma)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] gaussian σ={sigma:.0f} → {best_sc:.6f}")

    # triangular
    a = generate_triangular(n)
    sc = fast_score(a, fft_len, n)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
        print(f"[grid] triangular → {best_sc:.6f}")

    # sine (power 1)
    a = generate_sine(n, power=1.0)
    sc = fast_score(a, fft_len, n)
    if sc < best_sc:
        best_sc, best_a = sc, a.copy()
        print(f"[grid] sine → {best_sc:.6f}")

    return best_a, best_sc

# Optimisation primitives
# ----------------------------------------------------------------------

def adam_optimize(
    a_start: np.ndarray,
    lam_schedule: list[float],
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp smoothed max.
    Works in the log‑probability (logits) space; a = softmax(y).
    """
    eps = 1e-14
    y = np.log(a_start + eps)                # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2 = 0.9, 0.999

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    t = 0
    for lam in lam_schedule:
        for _ in range(iters_per_lam):
            if time.time() >= deadline:
                return best_a, best_sc
            t += 1
            f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=lam)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # step size – classic Adam base = 0.1, scaled per‑coordinate
            step = 0.12 / (np.sqrt(v_hat) + 1e-8)
            y -= step * m_hat

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[adam] lam={lam:.5f} iter={t} → new best {best_sc:.6f}")

    return best_a, best_sc


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum term
    beta = 0.86
    it = 0

    while time.time() < deadline:
        # true max‑convolution (linear)
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        # sub‑gradient of the max (mask may have several active entries)
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum
        v = beta * v + (1.0 - beta) * g

        # step size – 0.7 / sqrt(it+1) gives a gentle decay
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_sc = fast_score(a_candidate, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, a_candidate.copy()
            print(f"[subgrad] it={it} → new best {best_sc:.6f}")

        a = a_candidate
        it += 1

    return best_a, best_sc


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Guided simulated annealing.
    At each step we move a random fraction of mass from a “high‑gradient”
    index (contributing most to the current peak) to a “low‑gradient”
    index.  Square‑root cooling keeps a generous temperature early on.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    # initial temperature as a perceptible fraction of the score
    T0 = max(0.07 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # cooling
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        # compute sub‑gradient once every few steps (cached)
        if rng.random() < 0.02:
            # recompute mask for the current max
            fa = fft(a, fft_len)
            conv = np.real(ifft(fa * fa))[: 2 * n - 1]
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - eps).astype(float)

            a_rev = a[::-1]
            fm = fft(mask, fft_len)
            fa_rev = fft(a_rev, fft_len)
            conv_g = np.real(ifft(fm * fa_rev))
            grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]
        # pick source from the top‑15 % of gradient magnitudes
        src_candidates = np.argpartition(-grad, int(0.15 * n))[: int(0.15 * n)]
        dst_candidates = np.argpartition(grad, int(0.15 * n))[: int(0.15 * n)]

        i = int(rng.choice(src_candidates))
        j = int(rng.choice(dst_candidates))
        if i == j:
            continue

        if a[i] <= 0.0:
            continue

        # transfer up to 30 % of the source mass
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc, best_a = new_sc, a_new.copy()

    return best_a, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem parameters
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation in smoothed max

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Re‑use a previous construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate to current length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_sc:
                best_a, best_sc = a1, sc1
                print(f"[init] imported best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – systematic parametric grid search
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_sc:
        best_a, best_sc = a_grid, sc_grid
        print(f"[grid] best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Timing allocation
    # --------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s   # Adam phase
    phase2_end = start_time + 0.55 * budget_s   # Sub‑gradient phase
    phase3_end = start_time + 0.90 * budget_s   # Simulated annealing

    # --------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (multiple λ)
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.3, 0.15, 0.08, 0.04, 0.02,
        0.01, 0.006, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-05,
    ]
    iters_per_lam = 1200

    while time.time() < phase1_end:
        a_start = best_a.copy()
        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_schedule,
            iters_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_sc = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] sub‑gradient best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – guided simulated annealing
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
