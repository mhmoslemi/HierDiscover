def propose_candidate(seed=42, budget_s=1000, **kwargs):

    rng = np.random.default_rng(seed)

    n = 4000

    fft_len_conv = next_pow2(2 * n - 1)

    fft_len_g = next_pow2(3 * n - 2)   # for grad computing

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety

    # initial candidate: uniform
    best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)

    print(f"[init] uniform fast_score={best_score:.6f}")

    # use prior
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # resample to length n via interpolation
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0)
            if a1.sum() > 0:
                a1 /= a1.sum()
                score1 = fast_score(a1, fft_len_conv, n)
                if score1 < best_score:
                    best_a, best_score = a1, score1
                    print(f"[init] prior fast_score={best_score:.6f}")

    # Phase 0
    grid_a, grid_sc = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if grid_sc < best_score:
        best_a, best_score = grid_a, grid_sc
        print(f"[grid] best fast_score={best_score:.6f}")

    # cheap random sweep maybe 5% of total
    sweep_a, sweep_sc = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sweep_sc < best_score:
        best_a, best_score = sweep_a, sweep_sc
        print(f"[rand] new best fast_score={best_score:.6f}")

    # Determine phase deadlines
    total_budget = budget_s
    t0 = start_time
    # allocate times
    t_phase1_end = t0 + 0.30 * total_budget        # L-BFGS
    t_phase2_end = t0 + 0.65 * total_budget        # Adam optimizer
    t_phase3_end = t0 + 0.95 * total_budget        # Simulated annealing
    # remaining ensures final evaluation

    # Phase 1 L-BFGS
    lam_schedule = [0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
                    0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
                    0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
                    2e-05, 1e-05, 5e-06, 2e-06]

    maxiter_per_lam = 3000

    while time.time() < t_phase1_end:
        # jitter or random start
        if rng.random() < 0.2:
            # jitter current best
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0)
            a_start /= a_start.sum()
        else:
            rate = rng.uniform(0.0002, 0.0012)
            a_start = generate_exp(n, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            t_phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # Phase 2 Adam optimizer on smoothed max
    # Use current best as start
    if time.time() < t_phase2_end:
        # convert best_a to logits y
        y_start = np.log(best_a + 1e-12)
        best_a2, best_score2 = adam_optimize(
            y_start,
            lam_schedule,
            lr=0.15,
            beta1=0.9,
            beta2=0.999,
            eps=1e-12,
            maxiter_per_lam=3000,
            deadline=t_phase2_end,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if best_score2 < best_score:
            best_score, best_a = best_score2, best_a2.copy()
            print(f"[phase2] Adam improved fast_score={best_score:.6f}")

    # Phase 3 Simulated annealing / hill climbing
    if time.time() < t_phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            t_phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # Final evaluation
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq

"""
Propose a low‑score step sequence.

The algorithm consists of four time‑controlled phases:

1. **Broad parametric sweep** – systematic grid search over several
   simple families (exponential, power, logistic, Gaussian) plus a cheap
   random sweep.  The best sequence is kept.

2. **Continuation L‑BFGS on a smoothed L∞‑objective** – the true
   `max(conv)` is approximated by a log‑sum‑exp with a decreasing
   smoothing factor.  For each factor a short L‑BFGS optimisation is run,
   starting from a jittered copy of the current best.

3. **Adam optimisation on the same smoothed objective** – after L‑BFGS a
   longer Adam run (logits → soft‑max) further reduces the smoothed
   maximum.

4. **Long simulated annealing / hill‑climb** – mass is transferred
   between two random positions; uphill moves are accepted with a
   square‑root cooling schedule.  The best vector seen during the whole
   run is finally evaluated with the official `evaluate_sequence`.

All phases respect a global deadline (default 1000 s).  The function
`propose_candidate` returns the best list of floats found before the
deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact score 2·n·max(a*a) for a normalised (sum = 1) vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective and its gradient (log‑sum‑exp)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of `2·n·max(a*a)` with a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from
    the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def adam_optimize(
    y_start: np.ndarray,
    lam_schedule,
    lr: float,
    beta1: float,
    beta2: float,
    eps: float,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smoothed max objective.
    Works directly in the log‑odds space (y) which guarantees the simplex.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t_iter = 0

    best_y = y.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        for _ in range(maxiter_per_lam):
            if time.time() >= deadline:
                break
            t_iter += 1
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_curr = softmax(y)
            score = fast_score(a_curr, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_y = y.copy()
                best_a = a_curr.copy()
                print(f"[adam] new best fast_score={best_score:.6f}")

    best_a = softmax(best_y)
    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Mass‑transfer hill‑climb with a square‑root cooling schedule.
    Moves a random fraction (up to 30 %) from a source index to a target.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # initial temperature
    T0 = max(0.08 * start_score, 1e-8)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-8)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gaussian(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def parametric_grid_search(

    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential
    for rate in np.linspace(0.0002, 0.0015, 61):
        if time.time() >= deadline:

        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power law
    for p in np.linspace(-6.0, -0.2, 61):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic
    mus = np.linspace(0.2 * n, 0.8 * n, 9)
    sigmas = np.linspace(200.0, 1200.0, 9)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # Gaussian (left‑biased)
    mus = np.linspace(0.0, 0.3 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_gaussian(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()

def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a normalised non‑negative vector from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-6.0, -0.2)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    tot = a.sum()
    if tot == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= tot
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Initial candidate (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optionally reused previous construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            total = a1.sum()
            if total > 0.0:
                a1 /= total
                score1 = fast_score(a1, fft_len_conv, n)
                if score1 < best_score:
                    best_a, best_score = a1, score1
                    print(f"[init] prior fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – systematic grid + cheap random sweep
    # ------------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time allocation for the remaining phases
    # ------------------------------------------------------------------
    total_budget = budget_s
    t0 = start_time
    t_phase1_end = t0 + 0.30 * total_budget   # L‑BFGS
    t_phase2_end = t0 + 0.65 * total_budget   # Adam optimisation
    t_phase3_end = t0 + 0.95 * total_budget   # Simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation on a smoothed max
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06,
    ]
    maxiter_per_lam = 2000
    restart_counter = 0
    while time.time() < t_phase1_end and restart_counter < 12:
        if rng.random() < 0.2:
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            rate = rng.uniform(0.0002, 0.0012)
            a_start = generate_exp(n, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            t_phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

        restart_counter += 1

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < t_phase2_end:
        y_start = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(
            y_start,
            lam_schedule,
            lr=0.15,
            beta1=0.9,
            beta2=0.999,
            eps=1e-12,
            maxiter_per_lam=3000,
            deadline=t_phase2_end,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        print(f"[phase2] Adam best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long simulated annealing / hill‑climb
    # ------------------------------------------------------------------
    if time.time() < t_phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            t_phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
