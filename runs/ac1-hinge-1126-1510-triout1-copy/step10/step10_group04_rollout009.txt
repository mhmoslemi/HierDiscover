def lp_alternating(f_start, deadline, rng):
    n = len(f_start)
    fft_len = next_pow2(2*n - 1)
    f = f_start.copy()
    best_f = f.copy()
    best_score = fast_score(f, fft_len)
    iteration = 0
    while time.time() < deadline:
        # compute conv and M
        fa = np.fft.fft(f, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        M = conv.max()
        # build matrix A
        A = build_toeplitz_matrix(f)
        # Solve LP
        c = -np.ones(n)
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {iteration} failed: {res.message}")
            break
        g = res.x
        sum_g = g.sum()
        if sum_g < 1e-12:
            # nothing to improve
            break
        g_norm = g / sum_g
        improved = False
        # Try mixing coefficients
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1 - t) * f + t * g_norm
            f_new /= f_new.sum()
            score_new = fast_score(f_new, fft_len)
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_f = f_new.copy()
                f = f_new
                improved = True
                break
        if not improved:
            # No improvement using mixing; break loop
            break
        iteration += 1
    return best_f, best_score

"""
Hybrid optimisation for the step‑function bound:
  minimise 2·n·max(conv(a,a)) / (sum(a)^2)  (sum is forced to 1).

The search combines three ideas:

1.  A small‑scale alternating optimisation based on a linear programme.
    For a fixed distribution f we solve
        maximise   Σ g_j
        subject to f * g ≤ M   (M = max (f*f))
                g ≥ 0 .
    The LP is cheap for n≈1 k.  Mixing the obtained g with f
    strictly reduces the peak of the auto‑convolution.

2.  Upsample the small solution to a larger size (≈4000) and
    refine it with exponentiated‑gradient descent on the true
    max‑convolution objective.

3.  A final greedy hill‑climb (incremental convolution updates)
    removes any remaining local peaks.

All phases respect the 1000 s time budget and keep the vector on the
probability simplex, so the final `evaluate_sequence` call is simply
`2·n·max(conv)`.  The best sequence found is returned as a plain list
of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) for L‑BFGS / Adam – kept for completeness
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max(conv) w.r.t. a (sub‑gradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # only central n entries

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a fraction of mass from a heavy index
    to a lighter one; accepts only improvements.  Convolution updates
    are performed incrementally.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming (LP) building block (small‑scale)
# ----------------------------------------------------------------------


def build_toeplitz_matrix(f: np.ndarray):
    """
    Construct the (2·n‑1) × n Toeplitz matrix T such that
        conv(f, g) = T @ g    for any g.
    Only non‑negative offsets are needed because the convolution
    is one‑sided (indices outside [0,n‑1] are treated as zero).
    """
    n = f.shape[0]
    # each diagonal offset k (0 ≤ k < n) is filled with the constant f[k]
    data = [np.full(n, f_k) for f_k in f]
    offsets = np.arange(n)                # 0 .. n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def lp_alternating(f_start: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Alternating optimisation:
        * compute M = max(conv(f,f))
        * solve LP maximise Σ g_j  s.t. conv(f,g) ≤ M , g ≥ 0
        * replace f by a convex combination with the normalised LP solution
    Repeats until no improvement or time runs out.
    """
    n = f_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    f = f_start.copy()
    best_f = f.copy()
    best_score = fast_score(f, fft_len)

    iteration = 0
    while time.time() < deadline:
        # current peak
        fa = np.fft.fft(f, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # build Toeplitz constraint matrix
        A = build_toeplitz_matrix(f)

        # linear programme: maximise Σ g_j  ⇔ minimise -Σ g_j
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        lp_res = linprog(c, A_ub=A, b_ub=b_ub,
                         bounds=bounds, method='highs',
                         options={'presolve': True})

        if not lp_res.success:
            # solver failed – abort the alternating loop
            print(f"[LP] iteration {iteration} failed: {lp_res.message}")
            break

        g = lp_res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break

        g_norm = g / sum_g

        # try a few mixing fractions (including t=1 → take g alone)
        improved = False
        for t in (1.0, 0.8, 0.5, 0.3, 0.2, 0.1, 0.05):
            f_new = (1.0 - t) * f + t * g_norm
            f_new /= f_new.sum()
            score_new = fast_score(f_new, fft_len)
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_f = f_new.copy()
                f = f_new          # continue from the improved point
                improved = True
                break

        if not improved:
            # no better mixture found – stop
            break

        iteration += 1

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Phase 0 – small‑scale alternating LP optimisation
    # ------------------------------------------------------------------
    n_small = 1024                         # cheap enough for repeated LP solves
    fft_len_small = next_pow2(2 * n_small - 1)

    # initialise from the global best if it exists, else uniform
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n_small:
                xs_old = np.linspace(0, 1, old.size)
                xs_new = np.linspace(0, 1, n_small)
                f_small = np.interp(xs_new, xs_old, old)
            else:
                f_small = old.copy()
            f_small = np.maximum(f_small, 0.0)
            f_small /= f_small.sum()
        else:
            f_small = np.full(n_small, 1.0 / n_small)
    else:
        f_small = np.full(n_small, 1.0 / n_small)

    # allocate roughly 30 % of the budget for the LP stage
    start_time = time.time()
    deadline = start_time + budget_s - 0.10    # small safety margin
    lp_deadline = start_time + 0.30 * budget_s

    print(f"[init] n_small={n_small}")

    f_small, score_small = lp_alternating(f_small, lp_deadline, rng)
    print(f"[LP] improved small score = {score_small:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – up‑sample to a large problem size
    # ------------------------------------------------------------------
    n_big = 4000
    fft_len_big = next_pow2(2 * n_big - 1)

    xs_small = np.linspace(0, 1, n_small)
    xs_big = np.linspace(0, 1, n_big)
    f_big = np.interp(xs_big, xs_small, f_small)
    f_big = np.maximum(f_big, 0.0)
    f_big /= f_big.sum()

    best_f = f_big.copy()
    best_score = fast_score(best_f, fft_len_big)
    print(f"[up‑sample] n_big={n_big}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the true objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # reserve ~60 % of the remaining time for exp‑grad
        exp_deadline = time.time() + 0.60 * (deadline - time.time())
        best_f, best_score = exp_gradient_optimize(
            best_f, exp_deadline, n_big, fft_len_big, rng, eta0=12.0
        )
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (exact convolution updates)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_f, best_score = hill_climb_incremental(
            best_f, best_score, deadline, rng
        )
        print(f"[hill‑climb] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (clamped by evaluate_sequence)
    # ------------------------------------------------------------------
    final_seq = best_f.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
