def spectral_flatten(a, L, max_iter, deadline):
    best_a = a.copy()
    best_score = fast_score(a, L, n)  # careful L use proper conv fft length
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # pad
        pad = np.zeros(L)
        pad[:n] = a
        A = np.fft.fft(pad)
        mag = np.abs(A)
        # compute target magnitude (constant)
        target_mag = np.sqrt(np.mean(mag**2))
        # update magnitude
        new_mag = target_mag * np.ones_like(mag)  # full flatten
        # keep phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        p_new = np.fft.ifft(A_new).real
        a_new = p_new[:n]
        # enforce nonnegative
        a_new = np.maximum(a_new, 0.0)
        # renorm
        sum_a = a_new.sum()
        if sum_a <= 0.0:
            a_new = a.copy()
        else:
            a_new /= sum_a
        # evaluate
        score = fast_score(a_new, L, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def projected_gradient_descent(a_start, deadline, n, fft_len, init_lr=0.1, decay=0.95):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    lr = init_lr
    it = 0
    while time.time() < deadline:
        max_val, conv, grad = max_conv_and_grad(a, fft_len)
        # gradient descent step
        a_new = a - lr * grad
        # project onto simplex
        a_new = proj_simplex(a_new)

        # update a
        a = a_new
        it += 1
        # decay lr periodically
        if it % 100 == 0:

    return best_a, best_score

def spectral_flatten(a_start, deadline, n, fft_len, max_iter=2000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

        if time.time() >= deadline:
            break
        pad = np.zeros(fft_len)
        pad[:n] = a
        A = np.fft.fft(pad)
        mag = np.abs(A)
        # target magnitude constant (preserving total power)
        target = np.sqrt(np.mean(mag**2))
        # scale magnitude toward target: full replace
        new_A = target * np.exp(1j * np.angle(A))
        p_new = np.fft.ifft(new_A).real
        a_new = p_new[:n]
        # enforce nonneg and renorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0:
            a_new /= s
        else:
            a_new = a
        # evaluate
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def proj_simplex(v):
    # v is 1d array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1 # index zero-based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Propose a low‑score step‑function sequence.

The search proceeds in several stages within the 1000 s budget:

1. **Initial pool** – generate a diverse set of candidates:
   uniform, random, exponential, power‑law, a “edge” shape,
   the historic best (if any) and a spectral‑init.

2. **Adam on a log‑softmax smoothed max‑objective** – several
   restarts with a decreasing smoothing factor (lam_schedule).
   The best vector is kept.

3. **Spectral equalisation** – alternately enforce a flat
   magnitude spectrum (constant power) and the time‑domain
   constraints (non‑negative, unit sum).  This drives the
   autocorrelation towards a constant shape.

4. **Projected gradient descent on the true max‑convolution** –
   using the sub‑gradient of the (hard) max, with projection onto
   the probability simplex.

5. **Deterministic hill‑climb** – move a small fraction of mass
   from the coordinate that contributes most to the current
   maximal convolution entry toward a coordinate with the smallest
   partner value.  Step size shrinks geometrically.

6. **Mixture search** – convex combination of the top‑K vectors
   found so far, using a cheap golden‑section line search.

7. **Final evaluation** – the best sequence is returned and stored
   under the historic name ``height_sequence_1`` for reuse.

All heavy work (FFT, gradient) is O(n log n) with n=4000.
The algorithm is tuned to squeeze the score below the previous
1.506 value and aim for the target 1.503.  No closures or lambdas are

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max → simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT of length ``fft_len``."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1               # zero‑based index
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diverse random probability vectors."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Spectral initialiser – square‑root of a flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def exponential_initial(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n))
    a /= a.sum()
    return a


def powerlaw_initial(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def combined_initial(n: int, p: float, r: float) -> np.ndarray:
    idx = np.arange(n) + 1.0
    a = (idx ** (-p)) * np.exp(-r * (idx - 1.0))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """Log‑sum‑exp surrogate of max‑convolution and its gradient."""
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """Adam optimiser on the log‑sum‑exp surrogate."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Max‑convolution and sub‑gradient (hard max)
# ----------------------------------------------------------------------
def max_conv_and_grad(a: np.ndarray, fft_len: int):
    """Return (max value, full conv, gradient of max w.r.t. a)."""
    n = a.shape[0]
    pad = np.zeros(fft_len)
    pad[:n] = a
    A = np.fft.fft(pad)
    conv = np.real(np.fft.ifft(A * A))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    # mask of positions attaining (or close to) the max
    mask = np.zeros(fft_len)
    mask[: 2 * n - 1] = (conv >= max_val - eps).astype(np.float64)

    a_rev = np.zeros(fft_len)
    a_rev[:n] = a[::-1]
    M = np.fft.fft(mask)
    AR = np.fft.fft(a_rev)
    grad_full = np.real(np.fft.ifft(M * AR))
    grad = 2.0 * grad_full[n - 1: n - 1 + n]   # length n
    return max_val, conv, grad


# ----------------------------------------------------------------------
# Projected gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def projected_gradient_descent(a_start: np.ndarray,
                               deadline: float,
                               n: int,
                               fft_len: int,
                               init_lr: float = 0.1,
                               decay: float = 0.95,
                               decay_every: int = 100):
    """Sub‑gradient descent with projection onto the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    lr = init_lr
    it = 0
    while time.time() < deadline:
        max_val, conv, grad = max_conv_and_grad(a, fft_len)
        a_new = a - lr * grad
        a_new = proj_simplex(a_new)

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1
        if it % decay_every == 0:
            lr *= decay
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climbing (mass transfer)
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """Move a small fraction of mass from the worst contributor to the best."""
    step = init_step
    while time.time() < deadline:
        pad = np.zeros(fft_len)
        pad[:n] = a
        fa = np.fft.fft(pad)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]  (zero outside support)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: index with smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new = proj_simplex(a_new)   # re‑normalise safely

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.7, min_step)

    return a, score


# ----------------------------------------------------------------------
# Convex‑combination search among top candidates
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """Golden‑section line search over convex mixtures of each pair."""
    def eval_mix(t, a, b):
        mix = (1.0 - t) * a + t * b
        mix /= mix.sum()
        return fast_score(mix, fft_len, n), mix

    phi = (1 + np.sqrt(5)) / 2.0
    invphi = (np.sqrt(5) - 1) / 2.0

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b)
            fd, _ = eval_mix(d, a, b)
            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b)

            tm = 0.5 * (a0 + b0)
            fs, vec = eval_mix(tm, a, b)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Spectral equalisation (constant‑magnitude projection)
# ----------------------------------------------------------------------
def spectral_flatten(a_start: np.ndarray,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    max_iter: int = 2000):
    """Iteratively enforce constant magnitude spectrum (Gerchberg‑Saxton)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        pad = np.zeros(fft_len)
        pad[:n] = a
        A = np.fft.fft(pad)
        mag = np.abs(A)
        target_mag = np.sqrt(np.mean(mag ** 2))

        # replace magnitude by the constant target (keep phase)
        A_new = target_mag * np.exp(1j * np.angle(A))
        p_new = np.fft.ifft(A_new).real
        a_new = p_new[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = a  # fallback

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

# Main entry point required by the task
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – keep the historic length (4000)
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)          # length used for all FFTs

    # --------------------------------------------------------------
    # Load historic best if present
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Build diverse initial pool
    # --------------------------------------------------------------
    candidates = [
        a_start.copy(),
        spectral_init(n),
        random_initial(n, rng),
        add_noise(a_start, rng.uniform(0.001, 0.01), rng),
    ]

    # extra parametric families
    for rate in np.logspace(-4, -2, 5):
        candidates.append(exponential_initial(n, rate))
    for p in np.linspace(0.2, 4.0, 5):
        candidates.append(powerlaw_initial(n, p))
    for p, r in [(1.0, 0.001), (1.5, 0.002), (2.0, 0.003)]:
        candidates.append(combined_initial(n, p, r))

    # keep best of the pool
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – Adam restarts on smoothed max‑objective
    # --------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002, 1e-5
    ]
    steps_per_lam = 200
    restart = 0
    max_restarts = 15
    while time.time() < deadline and restart < max_restarts:
        # disturb best before a restart (except first)
        if restart == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule,
            steps_per_lam,
            deadline,
            fft_len,
            fft_len,
            n,
        )
        print(f"[Adam‑restart {restart+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")
        restart += 1

    # --------------------------------------------------------------
    # Phase 2 – Spectral equalisation (constant magnitude)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = spectral_flatten(
            best_a,
            deadline,
            n,
            fft_len,
            max_iter=3000,
        )
        print(f"[Spectral‑flatten] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after spectral flatten: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Projected gradient descent on true max‑conv.
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = projected_gradient_descent(
            best_a,
            deadline,
            n,
            fft_len,
            init_lr=0.1,
            decay=0.95,
            decay_every=200,
        )
        print(f"[PGD] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after PGD: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – fine deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            init_step=0.15,
            min_step=1e-7,
        )
        print(f"[Hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after hill‑climb: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – convex‑combination search among top candidates
    # --------------------------------------------------------------
    # collect a few competitors (including some fresh random ones)
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # keep the K best vectors
    K = 5
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    selected = ordered[:K]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len,
            deadline,
        )
        print(f"[Mixture‑search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new best after mixture: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the provided security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
