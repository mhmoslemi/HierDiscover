def smoothed_gradient_descent(a_start, deadline, n, fft_len, rng, step0=0.5):
    a = np.copy(a_start)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    start = time.time()
    it = 0
    while time.time() < deadline:
        # compute conv
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2*n-1]
        max_conv = conv.max()
        # schedule temperature T (scale factor)
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        # Temperature factor decays from 0.5 to 0.02
        T = max_conv * max(0.02, 0.5 * (1 - frac))
        # compute softmax weights
        # Because conv - max_conv <= 0, we compute exp((conv - max)/T)
        exp_arg = (conv - max_conv) / T
        # Clip range to avoid underflow
        exp_arg = np.clip(exp_arg, -50, 0)
        w = np.exp(exp_arg)
        sum_w = w.sum()
        if sum_w == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / sum_w
        # compute gradient via conv of a_rev with w
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2*n-1]
        grad = 2.0 * grad_full[n-1 : n-1 + n]  # size n
        # step size
        eta = step0 / np.sqrt(it+1)
        # additive update
        a = a - eta * grad
        # projection onto simplex
        a = simplex_projection(a)  # ensures non-negative sum=1
        # evaluate fast_score every so often
        if (it % 10) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        it += 1
    return best_a, best_score

"""
Propose a low‐evaluation step sequence.

The algorithm works in three phases:

1. **Broad parametric scan** – enumerates several low‑dimensional families
   (exponential, power–law, linear, mixtures, plus two stochastic families:
   random Dirichlet draws and random flat‑spectrum signals).  The best
   candidate (sum normalised to 1) is kept.

2. **Smoothed gradient descent** – optimises the true objective
   `2·n·max(conv(a,a))` with a smooth approximation of the max
   (`softmax`).  The gradient of the smoothed objective is obtained by a
   single FFT‑based convolution; an additive update followed by Euclidean
   projection onto the probability simplex is performed.  The temperature
   of the softmax decays linearly, allowing the method to start with a
   smooth landscape and finish with a true sub‑gradient step.

3. **Simulated‑annealing fine‑tuning** – random mass transfers between
   indices are accepted with a Metropolis rule; the convolution is updated
   incrementally (O(n) per proposal) to finish the optimisation.

The best sequence found is returned and stored in the global variable
`height_sequence_1` for reuse by subsequent runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for evaluate_sequence when sum(a) = 1.
    Returns 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random flat‑spectrum generator
# ----------------------------------------------------------------------


def random_flat_spectrum(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a non‑negative vector of length n whose DFT magnitudes are all 1.
    The vector is shifted to be non‑negative and normalised to sum to 1.
    """
    A = np.zeros(n, dtype=np.complex128)
    A[0] = 1.0  # DC component (sets the total sum)
    half = n // 2
    # positive frequencies
    if n % 2 == 0:
        pos = np.arange(1, half)
        # Nyquist term must be real
        A[half] = rng.choice([-1.0, 1.0])
    else:
        pos = np.arange(1, half + 1)

    phases = rng.uniform(0.0, 2.0 * np.pi, size=pos.shape[0])
    for idx, k in enumerate(pos):
        phi = phases[idx]
        val = np.exp(1j * phi)
        A[k] = val
        A[-k] = np.conj(val)

    a = np.real(np.fft.ifft(A))
    a -= a.min()                     # shift to non‑negative
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Parametric families – cheap exhaustive scan
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """
    Explore a handful of low‑dimensional families and keep the best
    (according to fast_score).  Stops early if time runs out.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    def consider(a_candidate: np.ndarray):
        nonlocal best_a, best_score
        if a_candidate.ndim != 1 or a_candidate.size != n:
            return
        a_candidate = np.maximum(a_candidate, 0.0)
        s = a_candidate.sum()
        if s <= 0.0:
            return
        a_candidate = a_candidate / s
        score = fast_score(a_candidate, fft_len)
        if score < best_score:
            best_score, best_a = score, a_candidate.copy()

    # --- exponential decay -------------------------------------------------
    for lam in np.logspace(-6, -1.5, num=250):
        a = np.exp(-lam * np.arange(n))
        consider(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- linear decreasing ------------------------------------------------
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=250):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        consider(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- mixture exponential + uniform ------------------------------------
    for lam in np.logspace(-6, -1.5, num=60):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=13):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            consider(a)
            if time.time() >= deadline:
                return best_a, best_score

    # --- power‑law decay --------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=180):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        consider(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- exponential * power‑law -----------------------------------------
    for lam in np.logspace(-5, -1.5, num=80):
        exp_part = np.exp(-lam * np.arange(n))
        for p in np.linspace(0.2, 3.0, num=40):
            a = exp_part * (np.arange(1, n + 1, dtype=float) ** (-p))
            consider(a)
            if time.time() >= deadline:
                return best_a, best_score

    # --- random flat‑spectrum candidates ----------------------------------
    for _ in range(400):
        a = random_flat_spectrum(n, rng)
        consider(a)
        if time.time() >= deadline:
            return best_a, best_score

    # --- random Dirichlet / exponential draws ----------------------------
    for _ in range(400):
        a = rng.exponential(scale=1.0, size=n)
        consider(a)
        if time.time() >= deadline:
            return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Smoothed gradient descent on the true objective
# ----------------------------------------------------------------------


def smoothed_gradient_descent(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              rng: np.random.Generator,
                              step0: float = 0.8):
    """
    Mirror descent with a soft‑max approximation of the max‑convolution.
    A simple additive update is followed by Euclidean projection onto the
    simplex.  Temperature decreases linearly, turning the smooth objective
    into the true max‑convolution near the end.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    start = time.time()
    while time.time() < deadline:
        # convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_conv = conv.max()

        # temperature schedule (linear decay)
        elapsed = time.time() - start
        frac = elapsed / max(1e-12, deadline - start)
        # start with 0.5·max_conv, finish at about 0.02·max_conv
        T = max_conv * max(0.02, 0.5 * (1.0 - frac))

        # soft‑max weights (numerically stable)
        exp_arg = (conv - max_conv) / T
        exp_arg = np.clip(exp_arg, -60.0, 0.0)          # avoid underflow
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2·(a_rev * w)  (convolution)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1: n - 1 + n]

        # additive mirror update
        eta = step0 / np.sqrt(it + 1.0)
        a = a - eta * grad

        # Euclidean projection onto simplex
        a = simplex_projection(a)

        # occasional evaluation of the surrogate score
        if (it & 15) == 0:            # every 16th iteration
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (same as previous implementation)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator):
    """Random mass transfers with Metropolis acceptance and linear cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.20
    recompute_every = 4000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / max(1e-12, deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move mass from larger to smaller index
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of convolution (exact for linear term)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = dE < 0.0 or (rng.random() < np.exp(-dE / temp))
        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()
            if accepted % recompute_every == 0:
                conv = np.convolve(a, a)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 gives a good trade‑off between quality and speed
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # initialise from a previous run if possible
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    total_deadline = start_time + budget_s - 0.05   # safety margin

    # allocate time slices
    phase0_end = start_time + 0.07 * budget_s      # parametric sweep
    phase1_end = start_time + 0.78 * budget_s      # smoothed gradient descent
    # remaining time → simulated annealing

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = smoothed_gradient_descent(
            best_a, phase1_end, n, fft_len, rng, step0=0.8
        )
        print(f"[smooth‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < total_deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, total_deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
