"""
Propose a low‑score non‑negative step sequence.

Algorithm (time‑controlled ≈1000 s):
  0. Initialise: use the previous best if present, otherwise uniform.
  1. Fast parametric sweep over several families (exponential,
     power‑law, logistic, asymmetric double‑exponential) and a cheap
     random sweep.  Keep the best normalised vector.
  2. A short L‑BFGS phase on the smoothed max (log‑sum‑exp) to get out
     of poor basins.  The schedule of smoothing parameters λ is dense
     near 0 for fine optimisation.
  3. Long projected sub‑gradient descent with momentum on the true
     max‑convolution.  The step size follows 0.6/√t; the loop runs for
     most of the remaining budget.
  4. Simulated annealing (square‑root cooling) as a final stochastic
     hill‑climb.
  5. Return the best candidate (normalised to sum = 1).  The official
     score equals the fast score `2·n·max(conv)`.

All helper functions are top‑level, use only numpy / scipy and respect
the global time limit.  The routine works for any length; we use
n = 4000 (the length that gave the previous best result).
"""

import time
import numpy as np
from scipy.optimize import minimize

# Utility functions

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Return 2·n·max(a*a) assuming Σa = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1.0
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_asymm_double_exp(n: int, rate1: float, rate2: float,
                              coeff: float) -> np.ndarray:
    """a_i = e^{-rate1*i} + coeff·e^{-rate2*(n-1-i)} (asymmetric)."""
    i = np.arange(n, dtype=float)
    a = np.exp(-rate1 * i) + coeff * np.exp(-rate2 * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_random_family(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random parametric family and random parameters."""
    family = rng.choice(
        ["exp", "power", "logistic", "asymm_double_exp", "random"]
    )
    if family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        return generate_exp(n, rate)
    if family == "power":
        p = rng.uniform(-7.0, -0.5)
        inc = rng.choice([True, False])
        return generate_power(n, p, inc)
    if family == "logistic":
        mu = rng.uniform(0.2 * n, 0.8 * n)
        sigma = rng.uniform(200.0, 2000.0)
        return generate_logistic(n, mu, sigma)
    if family == "asymm_double_exp":
        rate1 = rng.uniform(0.0002, 0.0015)
        rate2 = rng.uniform(0.0002, 0.0015)
        coeff = rng.uniform(0.1, 5.0)
        return generate_asymm_double_exp(n, rate1, rate2, coeff)
    # pure random
    a = rng.random(n)
    a /= a.sum()
    return a

def cheap_random_sweep(n: int, fft_len: int, budget_s: float,

# Smoothed max (log‑sum‑exp) – L‑BFGS
# --------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient wrt y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule via softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray, lam_schedule,
                       maxiter_per_lam: int, deadline: float,
                       fft_len_conv: int, fft_len_g: int, n: int):
    """Run L‑BFGS over the λ schedule, starting each stage from the
    previous result."""
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


# --------------------------------------------------------------
# Projected sub‑gradient descent with momentum (true max)
# --------------------------------------------------------------

def subgradient_optimize_momentum(a0: np.ndarray, deadline: float,
                                 fft_len_conv: int, fft_len_g: int, n: int):
    """Projected sub‑gradient descent with momentum on the true max."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.85
    it = 0

    while time.time() < deadline:
        # true convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max (2·a⋆mask)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        grad = 2.0 * np.real(np.fft.ifft(fm * fa_rev))[(n - 1):(n - 1 + n)]

        # momentum
        v = beta * v + (1.0 - beta) * grad

        # step size decays like 0.6 / sqrt(it+1)
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)

        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


# --------------------------------------------------------------
# Simulated annealing (final stochastic hill‑climb)
# --------------------------------------------------------------

def simulated_annealing(a_start: np.ndarray, start_score: float,
                        fft_len: int, n: int, deadline: float,
                        rng: np.random.Generator):
    """Square‑root‑cooled annealing moving a random mass fraction."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimensions
    # ------------------------------------------------------------------
    n = 4000  # length of the step function (kept as in the reference)
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # ------------------------------------------------------------------
    # Baseline candidate (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Use a previously saved construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing setup
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.15  # small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – systematic parametric grid + cheap random sweep
    # ------------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv,
                                         budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase allocation (seconds)
    # ------------------------------------------------------------------
    total = budget_s
    phase1_end = start_time + 0.30 * total      # L‑BFGS
    phase2_end = start_time + 0.85 * total      # long sub‑gradient descent
    # rest -> annealing

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed max (few restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 800  # fewer iterations – we only need a quick boost

    while time.time() < phase1_end:
        # Two kinds of starts: (i) jittered current best, (ii) fresh exponential
        if rng.random() < 0.35:
            jitter = rng.normal(1.0, 0.03, size=n)
            a_start = best_a * jitter
            a_start = np.maximum(a_start, 0.0)
            if a_start.sum() == 0.0:
                a_start = np.full(n, 1.0 / n)
            else:
                a_start /= a_start.sum()
        else:
            rate = rng.uniform(0.0003, 0.0012)
            a_start = generate_exp(n, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] sub‑gradient best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing (final refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation (official)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
