    best_set = None

    for trial in range(max_trials):

        S = []

        candidates = list(range(n))
        rng.shuffle(candidates)
        for idx in candidates:
            # also bound total so far
            # Check if adding idx would exceed g for any sum
            ok = True
            if sum_counts[idx+idx] >= g:
                ok = False
            else:
                for j in S:
                    if sum_counts[idx+j] >= g:
                        ok = False
                        break
            if ok:
                # Add

                sum_counts[idx+idx] += 1
                for j in S:
                    sum_counts[idx+j] += 1
                S.append(idx)
                if len(S) >= m_target:

        # Evaluate max count
        current_max = sum_counts.max()

"""
Propose a low‑score non‑negative step sequence.

The algorithm follows a different philosophy from the previous
implementation:

1. **Sparse Sidon‑like construction** – we greedily build a set of
   indices `S ⊂ {0,…,n‑1}` such that each unordered sum `i+j` occurs at
   most `g` times (here `g=3`).  With `|S|≈180` the uniform weight
   `a_i = 1/|S|` on this support already yields a bound
   `2·n·max(a*a) < 1.5`.

2. **Local support annealing** – we improve the position set by a
   simulated‑annealing move (transfer a fraction of mass from one
   index to another).  The acceptance probability follows a
   square‑root cooling schedule.

3. **Projected sub‑gradient descent with momentum** – the final
   refinement runs true sub‑gradient steps on the exact max‑convolution
   objective, projecting onto the probability simplex after each step.

The whole search respects a global time budget (default 1000 s).  The
best sequence found at any moment is returned; it is finally evaluated
with the official `evaluate_sequence` function.
"""

import time
import numpy as np
from typing import List, Tuple

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Compute 2·n·max(a*a) for a vector whose entries sum to 1.
    The convolution is performed via FFT for speed.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # Only the first 2·n‑1 entries contain the linear convolution.
    return float(2.0 * n * np.max(conv[: 2 * n - 1]))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Greedy Sidon‑like set construction
# --------------------------------------------------------------


def greedy_sidonic_set(
    target_m: int,
    max_multiplicity: int,
    n: int,
    rng: np.random.Generator,
    max_trials: int = 30,
) -> Tuple[np.ndarray, int, int]:
    """
    Greedily build a set S of distinct indices in [0, n) such that each
    unordered sum i+j appears at most `max_multiplicity` times.
    Returns the best set found (its size and the maximum observed sum
    multiplicity).
    """
    best_S = np.empty(0, dtype=int)
    best_len = -1
    best_max = max_multiplicity + 1

    for _ in range(max_trials):
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)   # unordered pair counts
        S: List[int] = []
        # Random permutation of the index space
        cand = np.arange(n, dtype=int)
        rng.shuffle(cand)

        for idx in cand:
            # check self‑sum
            if sum_counts[idx + idx] >= max_multiplicity:
                continue
            # check sums with already chosen positions
            ok = True
            for j in S:
                if sum_counts[idx + j] >= max_multiplicity:
                    ok = False
                    break
            if not ok:
                continue

            # add idx to the set and update counts
            for j in S:
                sum_counts[idx + j] += 1
                # (j+idx) is the same index, counted only once per unordered pair
            sum_counts[idx + idx] += 1
            S.append(idx)

            if len(S) >= target_m:
                break

        cur_len = len(S)
        cur_max = int(sum_counts.max())
        if cur_len > best_len or (cur_len == best_len and cur_max < best_max):
            best_S = np.array(S, dtype=int)
            best_len = cur_len
            best_max = cur_max
            if best_len >= target_m and best_max <= max_multiplicity:
                break

    return best_S, best_len, best_max


# --------------------------------------------------------------
# Simulated annealing on the support (mass transfer moves)
# --------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> Tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that moves a random fraction of mass between
    two indices.  A square‑root cooling schedule is used.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t0 = time.time()
    total_time = max(deadline - t0, 1e-6)

    while time.time() < deadline:
        # temperature: sqrt decay
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        # pick source and destination indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        # transfer a random fraction (up to 30 %)
        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# --------------------------------------------------------------
# Projected sub‑gradient descent with momentum (true max)
# --------------------------------------------------------------


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Momentum (`beta≈0.85`) and a diminishing step size are used.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.85
    it = 0

    while time.time() < deadline:
        # exact auto‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # sub‑gradient of max

        # gradient of convolution w.r.t. `a`
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ~ 0.6 / sqrt(it+1)
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()

        a = a_candidate
        it += 1

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size
    n = 4000
    fft_len = next_pow2(2 * n - 1)        # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for back‑propagation in sub‑gradient

    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # safety margin

    # ------------------------------------------------------------------
    # 0. Uniform baseline (keeps a valid initial answer)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Sparse Sidon‑like construction
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.35 * budget_s
    target_m = 180           # desired support size
    max_multiplicity = 3     # at most 3 unordered pairs per sum

    best_sparse_a = None
    best_sparse_score = float("inf")
    attempts = 0

    while time.time() < phase1_deadline:
        attempts += 1
        S, sz, max_cnt = greedy_sidonic_set(
            target_m, max_multiplicity, n, rng, max_trials=5
        )
        if sz == 0:
            continue
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / sz          # uniform mass on the chosen support
        sc = fast_score(a, fft_len, n)

        if sc < best_sparse_score:
            best_sparse_score = sc
            best_sparse_a = a.copy()
            print(
                f"[sparse] attempt {attempts:04d} size={sz} max_unordered={max_cnt}"
                f" → score={sc:.6f}"
            )
        if sc < 1.45:            # early stop if already comfortably below target
            break

    if best_sparse_a is not None and best_sparse_score < best_score:
        best_a, best_score = best_sparse_a, best_sparse_score
        print(f"[phase1] best sparse score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated annealing on the support (mass transfers)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.65 * budget_s
    if time.time() < phase2_deadline and best_a is not None:
        a_anneal, sc_anneal = simulated_annealing(
            best_a,
            best_score,
            fft_len,
            n,
            phase2_deadline,
            rng,
        )
        if sc_anneal < best_score:
            best_a, best_score = a_anneal, sc_anneal
            print(f"[phase2] annealed score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected sub‑gradient descent (final polishing)
    # ------------------------------------------------------------------
    phase3_deadline = deadline
    if time.time() < phase3_deadline and best_a is not None:
        a_final, sc_final = subgradient_optimize_momentum(
            best_a, phase3_deadline, fft_len, fft_len_g, n
        )
        if sc_final < best_score:
            best_a, best_score = a_final, sc_final
            print(f"[phase3] sub‑gradient final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # provided by the platform
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
