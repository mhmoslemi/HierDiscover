"""
Improved step‑function optimizer.

Strategy
--------
0️⃣  Interpolate a previously found solution (if any) to the target length.
1️⃣  Fast parametric scan over a diverse family of analytic shapes
    (exponential, power‑law, beta‑type, two‑exponential mixtures and the
    arcsine law).  The best candidate becomes the start point for the next
    phase.
2️⃣  Smooth‑max gradient descent: the true max‑convolution is approximated by a
    log‑sum‑exp with a temperature that is increased linearly (continuation).
    The gradient is obtained by a single pair of FFT‑based convolutions,
    followed by a Euclidean projection onto the probability simplex.
3️⃣  Short exponentiated‑gradient (mirrored SGD) run to explore a different
    descent direction.
4️⃣  Simulated‑annealing fine‑tuning with larger mass‑transfer steps.

All phases respect an overall 1000 s budget.  The best sequence is stored in
the global variable `height_sequence_1` for reuse by later runs.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Return the value 2·n·max(conv(a,a)) assuming ‖a‖₁ = 1.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def param_scan(

    """

    """
    # start with uniform
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    idx = np.arange(n, dtype=float)

    # --------------------------------------------------------------
    # 1) pure exponential
    # --------------------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=200):
        a = np.exp(-lam * idx)
        a_sum = a.sum()
        if a_sum == 0.0:
            continue
        a /= a_sum
        score = fast_score(a, fft_len)
        if score < best_score:

        if time.time() >= deadline:
            return best_a, best_score

    # 2) power‑law
    # --------------------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=150):

        a_sum = a.sum()
        if a_sum == 0.0:

        a /= a_sum
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --------------------------------------------------------------
    # 3) beta‑type (i+1)^{p-1} (n-i)^{q-1}
    # --------------------------------------------------------------
    for _ in range(300):
        p = rng.uniform(0.2, 5.0)

        a_sum = a.sum()
        if a_sum == 0.0:

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --------------------------------------------------------------

    for _ in range(400):

        a = alpha * np.exp(-lam1 * idx) + (1.0 - alpha) * np.exp(-lam2 * idx)

def smooth_gradient_descent(
    a_start: np.ndarray,
    start_time: float,
    phase_end: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv(a,a)) using a smoothed max (log‑sum‑exp).
    The smoothing parameter α grows linearly from 5 to 200.
    """
    a = a_start.copy()
    # initial convolution & score
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 1
    eta0 = 0.05  # base step size
    alpha_start = 5.0
    alpha_end = 200.0

    while time.time() < phase_end:
        # current max of the convolution
        max_conv = conv.max()

        # continuation schedule for the smoothing parameter
        elapsed = time.time() - start_time
        total = phase_end - start_time
        fraction = min(1.0, elapsed / total)
        alpha = alpha_start + fraction * (alpha_end - alpha_start)

        # soft‑max weights (numerically stable)
        shift = max_conv
        exps = np.exp(alpha * (conv - shift))
        w = exps / exps.sum()

        # gradient of the smoothed objective
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fw = np.fft.rfft(w, fft_len)
        grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # step size (decaying sqrt schedule)
        eta = eta0 / np.sqrt(it)

        # additive update + projection onto simplex
        a = a - eta * grad
        a = simplex_projection(a)

        # recompute convolution for the next iteration
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()

        it += 1

    return best_a, best_score

def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
) -> tuple[np.ndarray, float]:
    """Mirrored SGD (multiplicative update) on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 1

    while time.time() < deadline:
        # sub‑gradient of the true max (identical to the previous code)
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len)
        fa_rev = np.fft.rfft(a_rev, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it)

        # multiplicative (exponentiated) update
        exponents = -eta * grad
        exponents = np.clip(exponents, -50, 50)   # prevent overflow
        a = a * np.exp(exponents)

        # simplex projection (by simple normalisation, non‑negativity already holds)
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score

# Phase 3 – simulated annealing (targeted hill‑climb)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Random mass transfers with a Metropolis rule and a linear cooling schedule."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # temperature starts as a modest fraction of the current score
    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.30   # allow larger moves than the original code
    recompute_every = 2000
    accepted = 0

    while time.time() < deadline:
        # linear cooling factor
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (O(n) work)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 4000 provides a good balance between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # initialise from a previous run if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # linearly interpolate to the new size
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # split the time budget (≈5 % / 66 % / 15 % / 14 %)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.71 * budget_s
    phase2_end = start_time + 0.86 * budget_s
    # remaining time goes to simulated annealing

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (more families than before)
    # ------------------------------------------------------------------
    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (continuation)
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        best_a, best_score = smooth_gradient_descent(
            best_a, start_time, phase1_end, n, fft_len, rng
        )
        print(f"[smooth‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=6.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
