def adam_subgradient_optimize(a0, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    while time.time() < deadline:
        t += 1
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2*n-1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(float)
        # gradient of max w.r.t a:
        # compute conv_g = 2 * (mask convolve reversed a)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2 * conv_g[(n-1): (n-1 + n)]
        # projection step:
        # We need descent direction = grad - np.mean(grad) *? Actually moving opposite to grad reduces max.
        # Use descent = grad - np.mean(grad) to keep sum zero.
        descent = grad - grad.mean()
        # update moments:

"""
Improved search for a low‑score non‑negative step sequence.

The algorithm consists of five stages (all time‑controlled):

1. **Parametric sweep** – cheap grid search over exponential, power and
   logistic families (plus a small random sweep).  The best vector is kept.

2. **L‑BFGS on a smoothed max** – minimise a log‑sum‑exp approximation of
   ``max(conv)`` using the soft‑max (log‑its) parametrisation.  The smoothing
   parameter λ is decreased gradually; each stage starts from the result of
   the previous one.

3. **Adam projected sub‑gradient descent** – the true nonsmooth objective
   ``max(a*a)`` is optimised with an Adam‑style update on the simplex.
   The gradient of the active (max‑valued) convolution entries is obtained
   by FFT‑based back‑propagation.  The iterate is projected onto the simplex
   after every step.

4. **Deterministic pair‑wise mass transfer** – at each iteration the
   element with the largest sub‑gradient (i.e. contributing most to the peak)
   donates a small amount of mass to the element with the smallest gradient.
   The move is kept only if it improves the score.

5. **Final simulated annealing** – a short stochastic hill‑climb with a
   square‑root cooling schedule to escape possible tiny local minima.

All stages respect a global deadline (default 1000 s).  The best sequence
seen at any moment is returned; a final call to ``evaluate_sequence`` yields
the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT; length 2·len(a)-1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector with Σa = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Random member of a simple parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_grid_search(n: int, fft_len: int, rng: np.random.Generator,
                           deadline: float) -> tuple[np.ndarray, float]:
    """Systematic grid search over a handful of simple families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power‑law
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic
    mus = np.linspace(0.2 * n, 0.8 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(n: int, fft_len: int, budget_s: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Quick random sweep over many families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + 0.10 * budget_s

    families = [
        "power", "exp", "gauss", "cos",
        "bimodal", "logistic", "binary_sparse",
        "double_exp", "random"
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")
    return best_a, best_score


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (logits → probabilities)
# --------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray, lam_schedule,
                       maxiter_per_lam: int, deadline: float,
                       fft_len_conv: int, fft_len_g: int, n: int) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits – always on the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:               # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score

# Adam projected sub‑gradient descent (true max)
# --------------------------------------------------------------

def adam_subgradient_optimize(a0: np.ndarray,
                              deadline: float,
                              fft_len_conv: int,
                              fft_len_g: int,
                              n: int) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected sub‑gradient descent on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    t = 0

    while time.time() < deadline:
        t += 1
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(float)

        # gradient of the active term(s)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # make the step a descent direction (projected to the tangent plane)
        descent = grad - grad.mean()

        # Adam update on the descent direction
        m = beta1 * m + (1.0 - beta1) * descent
        v = beta2 * v + (1.0 - beta2) * (descent ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = 0.12 / (np.sqrt(v_hat) + eps)          # learning‑rate schedule
        a_new = a - step * m_hat
        a_new = simplex_projection(a_new)

        cur_score = fast_score(a_new, fft_len_conv, n)

        if cur_score < best_score:
            best_score, best_a = cur_score, a_new.copy()
            # occasional progress report
            if t % 500 == 0:
                print(f"[adam] iter={t} best_score={best_score:.6f}")

        a = a_new

    return best_a, best_score

def pairwise_refine(a_start: np.ndarray,
                    deadline: float,
                    fft_len_conv: int,
                    fft_len_g: int,
                    n: int,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy refinement that moves a small amount of mass from the index with
    the largest gradient to the index with the smallest gradient, accepting
    only improving moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(float)

        # gradient of the active peak(s)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # indices with extreme gradients
        i = np.argmax(grad)          # source (most responsible for the peak)
        j = np.argmin(grad)          # destination (least responsible)

        if a[i] <= 1e-15:
            # pick a random non‑zero source if the chosen one is empty
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            i = rng.choice(nz)
        # transfer a modest fraction (up to 25 %) of the source mass
        delta = a[i] * rng.random() * 0.25
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < best_score:
            best_score, best_a = new_score, a_new.copy()
            # keep the improvement
            a = a_new
        # occasional status
        if it % 2000 == 0:
            print(f"[pairwise] iter={it} best_score={best_score:.6f}")

    return best_a, best_score

def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts uphill moves with a square‑root
    cooling schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.07 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.35
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score

                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Timing allocation
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    phase0_end = start_time + 0.15 * budget_s
    phase1_end = start_time + 0.30 * budget_s
    phase2_end = start_time + 0.60 * budget_s
    phase3_end = start_time + 0.85 * budget_s
    phase4_end = start_time + budget_s

    # --------------------------------------------------------------
    # Phase 0 – systematic grid + cheap random sweep
    # --------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, phase0_end)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[phase0] grid best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[phase0] random sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed max (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.4, 0.25, 0.18, 0.12, 0.07, 0.04,
        0.025, 0.015, 0.009, 0.005, 0.003, 0.0015,
        0.0008, 0.0004, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06, 5e-07,
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # occasional jitter of current best
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.025, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            rate = rng.uniform(0.0003, 0.0009)
            a_start = generate_exp(n, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_subgradient_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] Adam sub‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deterministic pair‑wise refinement
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = pairwise_refine(
            best_a,
            phase3_end,
            fft_len_conv,
            fft_len_g,
            n,
            rng,
        )
        print(f"[phase3] pairwise best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final simulated annealing (short polishing)
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_end,
            rng,
        )
        print(f"[phase4] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] official evaluated score = {final_eval:.6f}")

    return final_seq
