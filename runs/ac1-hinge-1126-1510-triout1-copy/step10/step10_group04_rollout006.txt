"""
Gradient‑guided simplex hill‑climb for the max‑convolution objective.

We keep the vector `a` on the probability simplex (sum = 1, a_i ≥ 0).
The true objective is

    f(a) = 2·n·max(conv(a,a)) / (∑ a)^2

which, after fixing ∑ a = 1, reduces to `2·n·max(conv)`.  The algorithm
iteratively:

* computes the convolution `conv = a * a` by FFT,
* builds the set of maximising indices (`mask`),
* obtains the sub‑gradient of `max(conv)` w.r.t. `a`:
      g_i = 2·∑_{k∈mask} a_{k-i}
  via another FFT,
* moves a fraction of mass from the index with the largest gradient
  (source) to the index with the smallest gradient (destination),
  accepting the move only when the peak of the convolution strictly
  decreases.  The step size (`delta`) is a decreasing fraction of the
  source weight, allowing large moves early and fine‐grained updates
  later.

When progress stalls we inject a small random perturbation to escape
local minima.  The whole search runs until the time budget (≈1000 s)
expires, after which the best sequence found is returned and finally
evaluated with the provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) for a normalised vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def gradient_max_conv(a: np.ndarray, conv: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current distribution (sum = 1).
    conv : ndarray, shape (2*n-1,)
        Full convolution a*a.
    fft_len : int
        Length of the FFT (a power of two).

    Returns
    -------
    g : ndarray, shape (n,)
        Gradient (2 * sum_{k∈argmax} a_{k-i}).
    """
    n = a.shape[0]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    # convolution mask * a_rev gives the required sums
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1 : n - 1 + n]
    return g


def perturb(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Small random mass‑shuffle to help escape plateaus."""
    n = a.shape[0]
    a2 = a.copy()
    # perform a few tiny moves
    for _ in range(8):
        src = rng.choice(n, p=a2 / a2.sum())
        dst = rng.integers(0, n)
        if src == dst or a2[src] <= 0.0:
            continue
        delta = a2[src] * scale * rng.random()
        a2[src] -= delta
        a2[dst] += delta
    a2 = np.maximum(a2, 0.0)
    s = a2.sum()
    if s > 0.0:
        a2 /= s
    return a2


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def improve_sequence(a: np.ndarray,
                     fft_len: int,
                     deadline: float,
                     rng: np.random.Generator,
                     best_a: np.ndarray,
                     best_score: float) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill–climb with occasional random perturbations.
    Returns the best sequence and its score seen during the run.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len)

    # counters for stagnation detection
    iter_without_improve = 0
    max_no_improve = 800  # after this many steps we perturb

    while time.time() < deadline:
        max_val = conv.max()
        # gradient of the *current* max
        g = gradient_max_conv(a, conv, fft_len)

        # source : largest gradient, destination : smallest
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # make sure we have something to move
        if src == dst or a[src] <= 0.0:
            # fall back to a random biased move
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(0, n)
            if src == dst or a[src] <= 0.0:
                # nothing sensible, skip this iteration
                iter_without_improve += 1
                if iter_without_improve >= max_no_improve:
                    a = perturb(a, rng)
                    conv = full_conv(a, fft_len)
                    iter_without_improve = 0
                continue

        # step size proportional to the source weight, decays over time
        elapsed = time.time() - (deadline - (deadline - time.time()))
        total = deadline - (deadline - (deadline - time.time()))  # approx total budget
        # a smooth decay from 0.20 → 0.02
        max_frac = 0.02 + 0.18 * (1.0 - (time.time() - (deadline - (deadline - time.time()))) / total)
        delta = a[src] * max_frac
        if delta <= 0.0:
            continue

        # build the linear convolution update term (diff)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a

        improved = False
        # try a few decreasing fractions of delta
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
            d = delta * factor
            if d <= 0:
                continue
            conv_new = conv + 2.0 * d * diff
            new_max = conv_new.max()
            if new_max < max_val - 1e-15:          # strict improvement
                # accept the move
                a[src] -= d
                a[dst] += d
                conv = conv_new
                cur_score = 2.0 * n * new_max
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(
                        f"[improve] t={time.time():.1f}s score={best_score:.6f} "
                        f"src={src} dst={dst} Δ={d:.2e}"
                    )
                improved = True
                iter_without_improve = 0
                break

        if not improved:
            # no improvement – count it towards stagnation
            iter_without_improve += 1
            if iter_without_improve >= max_no_improve:
                # inject a small random perturbation and recompute conv
                a = perturb(a, rng, scale=0.01)
                conv = full_conv(a, fft_len)
                iter_without_improve = 0

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
        2·n·max(conv(a,a)) / (sum(a)**2)

    The method starts from the previously discovered best sequence (if
    available), then performs a gradient‑guided simplex hill‑climb with
    occasional random perturbations for the remaining time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Load or initialise a starting distribution
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.array(globals()["height_sequence_1"], dtype=float)
        n = len(seq0)
        a0 = np.maximum(seq0, 0.0)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 /= s
    else:
        n = 4000
        a0 = np.full(n, 1.0 / n)

    # ensure normalization
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial score
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, initial fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    # We run the gradient‑guided hill‑climb until the budget expires.
    # Periodically we restart from a slightly perturbed version of the
    # current best to avoid being trapped in a local optimum.
    while time.time() < deadline:
        # allocate a modest chunk for a focussed descent
        sub_deadline = min(deadline, time.time() + 12.0)  # ~12 s per chunk
        # start from a perturbed copy of the current best
        a_start = perturb(best_a, rng, scale=0.015)
        best_a, best_score = improve_sequence(

            fft_len,

    # Final evaluation with the official routine
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
