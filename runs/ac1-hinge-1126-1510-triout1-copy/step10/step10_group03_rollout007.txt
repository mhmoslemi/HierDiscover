def soft_gradient(a, fft_len, beta=50.0):
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2*n -1]
    max_val = conv.max()
    # compute softmax weights.
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()
    # compute gradient:
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fw, fft_len)[:2*n -1]
    grad = 2.0 * grad_full[n-1 : n-1 + n]
    return grad, conv.max()
```

"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. **Parametric sweep** – Explore several low‑dimensional families:
   * pure exponential,
   * power‑law,
   * polynomial decay ( (1‑x)^α ),
   * mixtures of the above.
   Every candidate is normalised to a probability vector and scored with
   a fast FFT routine.

2. **Adam‑based mirror descent** – Optimise the (smoothed) max‑convolution
   objective directly on the simplex.
   The true convolution is obtained by FFT, a soft‑max with temperature β
   turns the non‑smooth max into a smooth weight vector and a gradient is
   computed analytically (∂(a∗a)_k/∂a_i = 2·a_{k‑i}).
   Adam’s momentum and adaptive step size accelerate convergence.
   After each update the iterates are projected onto the probability simplex.

3. **Simulated‑annealing polishing** – Random mass transfers between two
   indices are accepted with a Metropolis rule using a linear cooling
   schedule.  The convolution is updated incrementally (O(n)) to keep the
   routine fast.

The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming ‖a‖₁ = 1."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def soft_gradient(a: np.ndarray, fft_len: int, beta: float):
    """
    Compute a smooth gradient of the max‑convolution objective.
    Returns (grad, max_conv).
    """
    n = a.shape[0]

    # convolution a * a
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()

    # soft‑max weights on the convolution entries
    # shift by max to avoid overflow
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()          # shape (2n‑1,)

    # gradient = 2 * (a_rev * w)  (convolution)
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fw, fft_len)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice length n
    return grad, max_val


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """
    Scan several low‑dimensional families of sequences.
    Returns the best (a, score) encountered before deadline.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # ---- exponential families ----
    for lam in np.logspace(np.log10(1e-5), np.log10(0.05), num=300):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ---- power‑law families ----
    for p in np.linspace(0.05, 3.0, num=250):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ---- polynomial decay ( (1‑x)^α ) ----
    xs = np.arange(n, dtype=float) / (n - 1)
    for alpha in np.linspace(0.1, 5.0, num=200):
        a = (1.0 - xs) ** alpha
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # ---- mixtures: exponential + power‑law ----
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=40):
        exp_base = np.exp(-lam * np.arange(n))
        exp_base /= exp_base.sum()
        for p in np.linspace(0.3, 2.0, num=30):
            pow_base = (np.arange(1, n + 1, dtype=float) ** (-p))
            pow_base /= pow_base.sum()
            for alpha in np.linspace(0.0, 1.0, num=6):
                a = alpha * exp_base + (1.0 - alpha) * pow_base
                a /= a.sum()
                score = fast_score(a, fft_len)
                if score < best_score:
                    best_score, best_a = score, a.copy()
                if time.time() >= deadline:
                    return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – Adam optimiser on smoothed max‑convolution
# ----------------------------------------------------------------------


def adam_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    alpha: float = 0.15,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    beta_soft: float = 30.0,
    beta_soft_final: float = 200.0,
):
    """
    Adam (mirrored) SGD on the smooth max‑convolution objective.
    Returns (best_a, best_score) found before deadline.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    while time.time() < deadline:
        t += 1
        # slowly increase the softness (beta) so we start smooth
        beta_cur = beta_soft + (beta_soft_final - beta_soft) * min(1.0, t / 2000.0)

        grad, _ = soft_gradient(a, fft_len, beta_cur)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a = a - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # project back onto the simplex (guarantees non‑negativity & sum=1)
        a = simplex_projection(a)

        # occasionally compute the exact fast_score
        if t % 30 == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing refinement (incremental updates)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers with Metropolis acceptance and linear cooling.
    Returns (best_a, best_score) found before deadline.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.12   # smaller steps for fine tuning
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incrementally update the convolution (O(n))
        # convolution update: conv_new = conv + 2*delta*(outer product of unit vectors)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    # --------------------------------------------------------------
    # Problem size – 4000 gives a good speed / quality trade‑off.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate the old sequence to the new length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Time budget split (5 % / 70 % / 25 %)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.03   # safety margin

    phase0_end = start_time + 0.05 * budget_s
    phase1_end = start_time + 0.75 * budget_s
    # rest → simulated annealing

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    if time.time() < phase0_end:
        cand_a, cand_score = param_scan(n, fft_len, phase0_end, rng)
        if cand_score < best_score:
            best_a, best_score = cand_a.copy(), cand_score

    # --------------------------------------------------------------

    if time.time() < phase1_end:

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,

            deadline,
            n,
            rng,

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
