- compute b_sequence = np.convolve(sequence, sequence)
- max_b = max(b_sequence)
- sum_a = sum(sequence)

if sum_a < 0.01 -> return inf. else return 2 * n * max_b / (sum_a**2)

Sequence may be any length. For each sequence length n, the score is 2*n*max_b/(sum_a**2). The search wants to minimize this. So we want to find sequences with small max convolution, while sum of elements is not too small (but we can scale). Since the function is homogeneous under scaling: For scaling factor c>0, max_b scales as c^2, sum_a scales as c, so 2*n*c^2*max_b0/(c^2 * sum_a0^2) = 2*n*max_b0/(sum_a0^2) independent of scaling. So the objective is scale-invariant: scaling the vector doesn't change the score (provided not clipped). The code also clamps each value to [0, 1000] and ensures nonnegative; sequence sum is not normalized, but you can handle scaling.

Thus we want to minimize max(convolution)/sum^2.

    def project_simplex(v):
        # v is real vector
        # ensure sum(v_i) = 1 and v_i >=0
        # W. Duchi et al 2008 algorithm
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, len(v)+1)
        cond = u - cssv / ind > 0
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

def subgradient_descent(a_start, deadline, n, fft_len, rng):

    a = a_start.copy()

    best_a = a.copy(); best_score = fast_score(a,...)

    it = 0

    while time.time() < deadline:

        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # tolerance relative: maybe 1e-12 * max_val
        mask = (conv >= max_val - max_val*1e-12).astype(np.float64)  # mask entries at max
        # compute gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad = 2.0 * np.real(np.fft.ifft(fa_rev * fm))
        grad_vec = grad[(n-1):(n-1+n)]
        # step size schedule: eta = step0 / sqrt(it+1)
        eta = 0.5 / np.sqrt(it + 1)  # step0 maybe ~ 0.5 (depending)
        # we must ensure step doesn't exceed a[i] values, else projection will enforce.

        a_next = a - eta * grad_vec
        # project onto simplex
        a = project_simplex(a_next)
        # compute new score
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score

A diverse set of seeds is generated, after which several complementary
optimisation phases are carried out within the allotted time budget:

1.  Adam on a log‑softmax parametrisation of a *smoothed* max‑convolution
    (log‑sum‑exp) with a decreasing “smoothness factor’’ schedule.
2.  Projected sub‑gradient descent on the *true* max‑convolution
    (additive updates with Euclidean projection onto the simplex).
3.  Mirror‑descent (exponentiated‑gradient) on the same non‑smooth objective.
4.  Deterministic hill‑climbing that moves a fraction of mass from the
    coordinate that contributes most to the current maximal convolution entry
    toward a coordinate with the smallest partner value; the step size shrinks
    geometrically.
5.  Fast stochastic local search – random tiny mass transfers that are kept
    whenever they improve the score.
6.  Pairwise convex‑combination (golden‑section line‑search) among the best
    vectors discovered so far.

All phases respect the global deadline; the best vector found is stored in
the historic name ``height_sequence_1`` and returned as a plain list of
floats.  The implementation uses only NumPy and SciPy, no closures or
lambda‑expressions, and works for any length `n` (default 4000).
"""

import time
import numpy as np
from scipy.optimize import minimize


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation for a normalised vector (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implements the algorithm of Duchi et al. (ICML 2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Start vector obtained from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def arcsine_init(n: int) -> np.ndarray:
    """Discrete arcsine‑type start (more mass near the ends)."""
    i = np.arange(n, dtype=float)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using a log‑sum‑exp surrogate.
    Returns (objective value, gradient w.r.t. logits y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns the best vector and its score discovered during this run.
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


def subgradient_descent_max(a_start: np.ndarray,
                            deadline: float,
                            n: int,
                            fft_len: int,
                            rng: np.random.Generator,
                            step0: float = 0.5):
    """
    Additive projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector and its score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        # mask of entries that are (almost) maximal
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max w.r.t. a (sub‑gradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # step size decays like 1/√t
        eta = step0 / np.sqrt(it + 1.0)

        a_new = a - eta * g
        a = project_simplex(a_new)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


def random_local_search(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator):
    """
    Very cheap stochastic local optimisation:
    move a tiny random amount of mass between two coordinates.
    Accept only improving moves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # Choose source (must have some mass) and destination
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # Relative step shrinks slowly with the number of trials
        step_factor = 0.02 * (0.9995 ** it)
        epsilon = min(a[i] * step_factor, 0.005)

        a_new = a.copy()
        a_new[i] -= epsilon
        a_new[j] += epsilon
        # sum unchanged, non‑negativity guaranteed by epsilon choice
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        it += 1
    return best_a, best_score


def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Deterministic hill‑climb that moves a fraction of mass from the index
    contributing the most to the current maximal convolution entry toward the
    index with the smallest partner value.  Step size decays geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i]   (zero outside the range)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: index with the smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new /= a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size
        else:
            step = max(step * 0.7, min_step)

    return a, score


def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    Pairwise convex‑combination search among the top candidates.
    For each pair a cheap golden‑section line search over t ∈ [0,1] is
    performed to minimise the bound.
    """
    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
            fd = fast_score((1.0 - d) * a + d * b, fft_len, n)
            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd = fast_score((1.0 - d) * a + d * b, fft_len, n)
            tm = 0.5 * (a0 + b0)
            fs = fast_score((1.0 - tm) * a + tm * b, fft_len, n)
            if fs < best_score:
                best_score = fs
                best_a = ((1.0 - tm) * a + tm * b) / ((1.0 - tm) * a + tm * b).sum()
    return best_a, best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a power‑of‑two length works well with the FFT.
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from a historic result if present.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Build a small pool of diverse seeds.
    # ------------------------------------------------------------------
    cand0 = a_start.copy()
    cand1 = spectral_init(n)
    cand2 = random_initial(n, rng)
    cand3 = arcsine_init(n)
    cand4 = add_noise(cand0, rng.uniform(0.001, 0.01), rng)
    candidates = [cand0, cand1, cand2, cand3, cand4]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global time budget.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – Adam with a fine smoothness‑factor schedule.
    # ------------------------------------------------------------------
    lam_schedule_fine = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 150
    restarts = 6
    for r in range(restarts):
        if time.time() >= deadline:
            break
        # Perturb the current best before each restart (except the first)
        if r == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule_fine,
            steps_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam‑restart {r+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the true max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = subgradient_descent_max(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            step0=0.6,
        )
        print(f"[Sub‑grad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after sub‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient refinement (mirrored descent).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            scale=5.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after exp‑gradient: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Deterministic hill‑climb.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_step=0.15,
            min_step=1e-6,
        )
        print(f"[Hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after hill‑climb: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Fast random local search.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = random_local_search(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
        )
        print(f"[Random‑local] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after random local search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – Convex‑combination exploration among the top vectors.
    # ------------------------------------------------------------------
    # Gather a handful of promising vectors (including the current best)
    top_vectors = [best_a]
    top_scores = [best_score]
    for _ in range(3):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len_conv, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # Keep the K best (K=5)
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    K = 5
    selected = ordered[:K]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len_conv,
            deadline,
        )
        print(f"[Mixture‑search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official security‑checked routine.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector globally for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
