"""
Search for a non‑negative step sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a three‑phase approach:

1. **Random restarts of exponentiated‑gradient descent** on the true
   max‑convolution objective.  The update a←a·exp(−η·g) (followed by a
   projection onto the simplex) follows the sub‑gradient g of the
   current peak of the convolution.

2. **Refined exponentiated‑gradient run** from the best vector found
   in phase 1, using a slower decay of the step size.

3. **Directed hill‑climb** that repeatedly moves a small amount of
   mass from the index with the largest sub‑gradient (most harmful)
   to the index with the smallest sub‑gradient (most beneficial).
   The convolution is updated **exactly** (including the quadratic
   δ²‑terms) so that the true max‑convolution is never approximated.

All phases respect the 1000 s time budget; the last remaining time is
spent on the hill‑climb.  The current best vector is returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution and peak mask
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1:n - 1 + n]

        # mirror‑descent update (exponentiated‑gradient)
        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # keep the best iterate
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Directed hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def directed_hill_climb(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the index with the largest
    sub‑gradient to the index with the smallest sub‑gradient.
    Updates the convolution exactly (including quadratic δ² terms).
    """
    a = a_start.copy()
    # exact linear convolution
    conv = np.convolve(a, a)          # length 2*n‑1
    best_a = a.copy()
    best_score = 2.0 * n * conv.max()

    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        # recompute sub‑gradient from current conv
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient g = 2·(a_rev * mask) with FFT
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1:n - 1 + n]

        # pick source (largest g) and destination (smallest g)
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] <= 0.0:
            # fallback to a random sensible move
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(0, n)
            if src == dst or a[src] <= 0.0:
                continue

        # move size – decays from 0.20 → 0.02
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)
        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # ----- exact update of the convolution -----
        # linear part: 2·δ·(shift_dst - shift_src)
        conv_new = conv.copy()
        conv_new[src:src + n] -= 2.0 * delta * a
        conv_new[dst:dst + n] += 2.0 * delta * a

        # quadratic part: δ²·(e_src*e_src + e_dst*e_dst - 2·e_src*e_dst)
        idx_ss = 2 * src
        idx_dd = 2 * dst
        idx_sd = src + dst
        conv_new[idx_ss] += delta * delta
        conv_new[idx_dd] += delta * delta
        conv_new[idx_sd] -= 2.0 * delta * delta

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            # accept move
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            best_score = new_score
            best_a = a.copy()
            accepted += 1

            # occasional full recomputation to avoid drift
            if accepted % 3000 == 0:
                conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 entries works well for the time budget.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previous run if available; otherwise uniform.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget handling
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline_total = start_time + budget_s - 0.05  # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – random restarts of exponentiated‑gradient
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = exp_gradient_optimize(
            a_start, phase1_end, n, fft_len, rng, eta0=10.0
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[EG‑restart] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – refined exponentiated‑gradient from the current best
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    if time.time() < phase2_end:
        a_opt, score_opt = exp_gradient_optimize(
            best_a, phase2_end, n, fft_len, rng, eta0=12.0
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[EG‑refine] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – directed hill‑climb with exact updates
    # ------------------------------------------------------------------
    if time.time() < deadline_total:
        a_opt, score_opt = directed_hill_climb(
            best_a, deadline_total, n, fft_len, rng
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
