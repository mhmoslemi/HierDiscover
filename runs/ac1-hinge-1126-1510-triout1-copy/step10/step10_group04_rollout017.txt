def subgradient_optimize(a_start, deadline, n, fft_len, rng, step0=0.1):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    t=0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute gradient of max conv w.r.t a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1:n-1+n]   # shape (n,)
        # step size schedule
        step = step0 / np.sqrt(t+1)
        # update using projected gradient descent onto simplex
        a_new = a - step * g
        a_new = simplex_projection(a_new)  # ensures nonnegative, sum 1
        a = a_new
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        t += 1
    return best_a, best_score
```

"""
A fast mini‑optimizer for the max‑convolution problem.

The optimisation proceeds in four logical steps:

1. **Sidon‑like support construction** – a greedy algorithm builds a set of
   indices whose pairwise sums are bounded by a small integer B.
   With only a few collisions the resulting auto‑convolution is already close

2. **Smooth L‑BFGS / Adam refinement** – starting from the support vector,
   we minimise a smooth log‑sum‑exp approximation of the max‑convolution.
   Several λ‑levels and many random restarts are performed, but the overall
   budget allocated to this phase is modest.

3. **True‑max sub‑gradient / exponentiated‑gradient descent** – a large number
   of cheap projected (or multiplicative) sub‑gradient steps are taken on the
   exact objective 2·n·max(conv(a,a)).  The step‑size decays as 1/√t.

4. **Final local polishing** – a short simulated‑annealing walk followed by
   an incremental hill‑climb (the same code used in the reference solution)
   cleans up any remaining tiny irregularities.

All phases keep the vector on the probability simplex, therefore the final
evaluation is simply `2·n·max(conv)`.  The routine returns the best sequence
found before the 1000 s deadline.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # d smooth_max / d conv_i

    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution (additive updates)
# ----------------------------------------------------------------------


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         step0: float = 0.2):
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    t = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        step = step0 / np.sqrt(t + 1.0)
        a_new = a - step * g
        a = simplex_projection(a_new)

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        t += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (global random walk)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled random walk on the simplex.
    Moves a random fraction of mass from a source index (chosen
    proportionally to its current weight) to a destination index.
    Accepts uphill steps with probability exp(‑Δ/ T).
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12                # initial temperature ~5 % of score
    max_frac = 0.20                               # max move fraction at start

    while time.time() < deadline:
        # linear cooling
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        # adapt move size
        max_frac = 0.02 + 0.18 * fraction

        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a fraction of mass from a heavy index
    to a lighter one, accepting only improvements.  Convolution updates
    are performed incrementally.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Greedy Sidon‑like support construction
# ----------------------------------------------------------------------


def greedy_support(n: int, B: int, rng: np.random.Generator,
                  time_limit: float) -> np.ndarray:
    """
    Build a set S⊂{0,…,n‑1} such that each unordered pair (i,j) contributes
    to a sum i+j at most B times.
    The algorithm repeatedly processes a random permutation of the indices
    and keeps every element that does not violate the bound.
    The best set found before the deadline is returned.
    """
    best_S = np.array([], dtype=int)
    best_len = 0
    theoretical = int(np.sqrt(2 * B * (2 * n - 1)))  # upper bound on |S|

    while time.time() < time_limit:
        order = rng.permutation(n)
        cnt = np.zeros(2 * n - 1, dtype=np.int16)   # counts of unordered pairs per sum
        S = []
        for idx in order:
            # diagonal contribution
            if cnt[2 * idx] + 1 > B:
                continue
            ok = True
            for s in S:
                sm = idx + s
                if cnt[sm] + 1 > B:
                    ok = False
                    break
            if not ok:
                continue
            # accept idx
            S.append(idx)
            cnt[2 * idx] += 1
            for s in S[:-1]:
                cnt[idx + s] += 1
            if len(S) >= theoretical:
                break
        if len(S) > best_len:
            best_len = len(S)
            best_S = np.array(S, dtype=int)
            print(f"[support] B={B}, size={best_len}")
            # early stop if we are close to the theoretical maximum
            if best_len >= theoretical - 2:
                break
    return best_S


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – keep it modest for fast FFTs
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previous run if possible; otherwise uniform.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – greedy Sidon‑like support construction (≈15 % time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * budget_s
    for B in (2, 3, 4):
        if time.time() >= phase1_end:
            break
        # allocate a slice of the remaining time for this B
        remaining = phase1_end - time.time()
        S = greedy_support(n, B, rng, remaining)
        if S.size == 0:
            continue
        a_candidate = np.zeros(n, dtype=float)
        a_candidate[S] = 1.0 / S.size
        score = fast_score(a_candidate, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()
            print(f"[support‑init] B={B}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short L‑BFGS runs from random starts (≈30 % time)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.45 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-7, num=15)
    maxiter_per_lam = 800
    while time.time() < phase2_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase2_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam on the smoothed objective (≈20 % time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase3_end = start_time + 0.65 * budget_s
        y0 = np.log(best_a + 1e-12)
        lam_sched = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01,
                     0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
        best_a, best_score = adam_optimize(
            y0, lam_sched, max_steps_per_lam=4000,
            deadline=phase3_end, n=n, fft_len=fft_len, lr0=0.15
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – sub‑gradient & exponentiated‑gradient on the true max (≈20 % time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        phase4_end = start_time + 0.85 * budget_s
        best_a, best_score = subgradient_optimize(
            best_a, phase4_end, n, fft_len, rng, step0=0.3
        )
        print(f"[sub‑grad] fast_score={best_score:.6f}")

        # a short final exponentiated‑gradient pass
        best_a, best_score = exp_gradient_optimize(
            best_a, phase4_end, n, fft_len, rng, eta0=15.0
        )
        print(f"[exp‑grad‑final] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – global simulated annealing + fine hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, rng
        )
        print(f"[SA] fast_score={best_score:.6f}")

        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
