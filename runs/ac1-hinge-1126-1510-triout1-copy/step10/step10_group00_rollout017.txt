"""
Improved step‑function search.
------------------------------

The evaluation metric is 2·n·max(conv(a,a)) / (sum(a)^2).  The score is
scale‑invariant, therefore we may restrict ourselves to probability
vectors (sum = 1).  The aim is to minimise the peak of the (linear)
self‑convolution.

The optimiser works in five stages

1. **Diverse initial pool** – a mixture of classic windows, spectral
   guesses, and many random Dirichlet draws.  All candidates are normalised
   to sum = 1.

2. **L‑BFGS on a smooth approximation** – the max‑convolution is
   replaced by a log‑sum‑exp (parameterised by a schedule of smoothing
   factors).  Optimisation is performed on the log‑softmax logits, which
   guarantees positivity and the simplex constraint.  Several random
   restarts are performed within the time budget.

3. **Adam optimisation on the same smooth objective** – an Adam optimiser
   (first‑order moment estimates) is used to exploit the curvature
   information that L‑BFGS can miss when the smoothing factor becomes very
   small.

4. **Projected sub‑gradient descent (exact max)** – after the smooth
   phases we switch to the exact non‑smooth objective using a simple
   diminishing‑step sub‑gradient method on the simplex.

5. **Targeted simulated‑annealing & greedy hill‑climb** – final
   exploration moves mass from the positions that contribute most to the
   current maximal convolution entry to positions that contribute least,
   followed by a strict greedy polishing.

All phases respect a global 1000 s wall‑clock limit.  The best sequence
found so far is reported with the official ``evaluate_sequence`` call.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Fast evaluation (probability vectors)
# --------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector (sum = 1):
        2·n·max(a * a)
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# --------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # Full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # Gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # Back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # Extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_dirichlet_initial(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Dirichlet draw with configurable concentration parameter."""
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalized to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    """Classic Hann window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    """Blackman window (good sidelobe suppression)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    """Kaiser window with a given beta."""
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    """Discrete Gaussian centred in the middle, normalised."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each smoothing factor in lam_schedule, starting each
    stage from the result of the previous one.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator,
                  lam_schedule,
                  max_iter_per_lam: int = 1200):
    """
    Adam optimiser on the log‑sum‑exp smoothed max‑convolution.
    Operates on logits (y) to keep the simplex constraint.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        it = 0
        while time.time() < deadline and it < max_iter_per_lam:
            t += 1
            _, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam)
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = 0.02 / (np.sqrt(t) + 1.0)   # simple diminishing schedule
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            it += 1
    return best_a, best_score


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         step0: float = 0.3):
    """
    Projected sub‑gradient descent on the *exact* max convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # top‑few mask (helps sub‑gradient stability)
        kth = max(1, int(0.02 * conv.size))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        # descent direction (zero‑sum)
        mean_g = np.mean(g)
        d = -(g - mean_g)

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = simplex_projection(a_new)

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score


def targeted_annealing(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       rng: np.random.Generator,
                       init_temp: float = 5e-4,
                       final_temp: float = 1e-7,
                       max_moves: int = 150000):
    """
    Simulated annealing that preferentially moves mass from the positions
    that contribute most to the current maximal convolution entry to those
    that contribute the least.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    move = 0
    total_moves = max_moves

    while time.time() < deadline and move < total_moves:
        # Current convolution and the index of the peak
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        k_max = int(conv.argmax())
        max_val = conv[k_max]

        # Contributions to the peak
        # c_i = a_i * a_{k_max-i} (if the partner index exists)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        # Choose source from the top‑10 % contributors
        src_candidates = np.where(contrib > np.percentile(contrib, 90))[0]
        if src_candidates.size == 0:
            src = rng.integers(0, n)
        else:
            src = rng.choice(src_candidates)

        # Choose destination from the bottom‑10 % contributors
        dst_candidates = np.where(contrib < np.percentile(contrib, 10))[0]
        if dst_candidates.size == 0:
            dst = rng.integers(0, n)
        else:
            dst = rng.choice(dst_candidates)

        if src == dst:
            move += 1
            continue

        # Transfer a fraction of the source mass
        delta = a[src] * rng.random() * 0.12
        if delta <= 0.0:
            move += 1
            continue

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta

        sc_candidate = fast_score(a_candidate, fft_len, n)

        accept = False
        if sc_candidate < best_score:
            accept = True
        else:
            t_frac = move / total_moves
            T = init_temp * (final_temp / init_temp) ** t_frac
            prob = np.exp(-(sc_candidate - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            if sc_candidate < best_score:
                best_score = sc_candidate
                best_a = a_candidate.copy()
        move += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 60000,
               delta_frac: float = 0.07):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # legacy safety

    # --------------------------------------------------------------
    # Problem size – a larger n gives a finer degree of freedom.
    # --------------------------------------------------------------
    n = 8000                       # doubled size compared to the reference
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build an enriched pool of initial candidates
    # --------------------------------------------------------------
    candidates = []

    # If a previous best construction is available, interpolate it.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Randomised families (including Dirichlet draws with various α)
    for _ in range(3):
        candidates.append(random_initial(n, rng))
    for alpha in [0.3, 0.7, 1.2]:
        candidates.append(random_dirichlet_initial(n, rng, alpha))

    # A few Kaiser windows with random β
    for _ in range(2):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Pick the best of the pool (fast_score is exact for sum=1)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – intensive L‑BFGS polishing (≈30 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004, 0.0002,
        0.0001, 5e-05, 2e-05, 1e-05, 5e-06,
        2e-06, 1e-06, 5e-07, 2e-07, 1e-07,
        5e-08, 2e-08, 1e-08
    ]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector to keep diversity.
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3          # mix ratio ≤ 0.3
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation on the smoothed max (≈30 % of budget)
    # --------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = adam_optimize(best_a,
                                           phase2_end,
                                           n,
                                           fft_len,
                                           rng,
                                           lam_schedule,
                                           max_iter_per_lam=1500)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (≈15 % of budget)
    # --------------------------------------------------------------
    phase3_end = start_time + 0.75 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase3_end,
                                                  n,
                                                  fft_len,
                                                  step0=0.35)
        print(f"[sub‑grad] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 4 – targeted simulated annealing (≈8 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_annealing(best_a,
                                                deadline,
                                                n,
                                                fft_len,
                                                rng,
                                                init_temp=5e-4,
                                                final_temp=1e-7,
                                                max_moves=180000)
        print(f"[targeted SA] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        deadline,
                                        rng,
                                        max_iter=90000,
                                        delta_frac=0.06)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (the official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
