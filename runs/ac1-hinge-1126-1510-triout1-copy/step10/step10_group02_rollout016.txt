    best_a_full = a.copy()

"""
prime‑step optimizer – 1000 s budget

Design a non‑negative step function (vector a) that minimises

    2·n·max(conv(a,a)) / (sum a)²

where conv is the full linear convolution.  The optimal shape makes the
self‑convolution as flat as possible (the “sum of two independent draws”
distribution with a uniform density).  The search proceeds in four
stages:

1.   Build a diverse pool of low‑dimensional “block” candidates
     (uniform, power‑law, exponential, Beta‑family).  The best block
     is expanded to length n and stored.

2.   Refine the block representation with a cascade of L‑BFGS
     optimisations on a smoothed max‑convolution (log‑sum‑exp) – the
     optimisation works in the log‑softmax (mirror) space and respects
     the simplex constraint automatically.

3.   Starting from the thus‑obtained full‑length vector, run a long
     exponentiated‑gradient (mirror‑SGD) descent on the *exact* L∞‑objective.
     The update is multiplicative and keeps the vector on the simplex.

4.   Finish with a simple simulated‑annealing hill‑climb that moves
     mass between arbitrary coordinates.  The last few seconds apply a
     brief extra mirror‑SGD polish.

All heavy work uses FFT‑based convolution (O(n log n)).  The routine
keeps track of the best sequence and finally returns it (and stores it
as ``height_sequence_1`` for later runs).  The implementation obeys the
no‑closure rule – every helper is a top‑level function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the official objective for a vector a (already on the simplex).
    Returns 2·n·max(conv(a,a)) / (sum a)².
    """
    n = a.shape[0]
    s = a.sum()
    if s == 0.0:
        return np.inf
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(v: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = v + rng.normal(0.0, sigma, size=v.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return v.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed max‑convolution (log‑sum‑exp) – objective for L‑BFGS
# ----------------------------------------------------------------------


def block_smooth_obj_and_grad(log_w: np.ndarray,
                              lam: float,
                              n: int,
                              fft_len: int,
                              block_len: int) -> tuple[float, np.ndarray]:
    """
    Smoothed max‑convolution on a block‑parameterised vector.
    log_w : log‑softmax parameters of the block weights (length B).
    lam   : smoothing factor (larger → smoother).
    Returns (objective, gradient wrt log_w).
    """
    # ----- block → full vector (still on the simplex) -----
    w = np.exp(log_w - np.max(log_w))          # prevent overflow
    w = w / w.sum()
    a = np.repeat(w, block_len)[:n]            # expand and truncate

    # ----- convolution & smoothed max -----
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    eps = 1e-12
    lam_eff = max_c * lam + eps                  # avoid division by zero
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)

    # objective (sum(a)=1 → denominator = 1)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max wrt a (full vector) -----
    w_conv = exp_shift / Z                       # d smooth_max / d conv[k]
    a_rev = a[::-1]
    fw = np.fft.fft(w_conv, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_a = 2.0 * np.real(np.fft.ifft(fa_rev * fw))[:n]

    # ----- aggregate gradient to block variables -----
    B = w.shape[0]
    grad_w = np.empty_like(w)
    for j in range(B):
        lo = j * block_len
        hi = min(lo + block_len, n)
        grad_w[j] = grad_a[lo:hi].sum()

    # ----- chain rule through the softmax (log_w) -----
    mean_g = np.dot(w, grad_w)
    grad_log = w * (grad_w - mean_g)   # ∂f/∂log_w

    return float(f), grad_log


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the *exact* max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 15.0,
                          max_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Mirror‑SGD (exponentiated‑gradient) that directly minimises the true
    L∞‑objective.  The update stays on the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline:
        # convolution and mask of (almost) maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        # gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = 2.0 * np.real(np.fft.ifft(fa_rev * fm))[:n]

        # step‑size (standard 1/√t schedule)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * grad_full, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        cur_score = fast_score(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator,
                        max_frac: float = 0.20) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves accepted with a Metropolis rule.
    The temperature linearly cools from a fraction of the starting
    score down to near zero.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)   # initial temperature

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = dE < 0.0 or rng.random() < np.exp(-dE / temp)

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – moderate resolution (blocks give a coarse shape)
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Block representation (each block is constant)
    # --------------------------------------------------------------
    block_len = 20                     # 4000 / 20 = 200 blocks

    # --------------------------------------------------------------
    # Generate a diverse pool of block‑vectors
    # --------------------------------------------------------------
    pool_blocks = []

    # uniform
    pool_blocks.append(np.full(B, 1.0 / B))

    # power‑law decreasing
    for _ in range(4):
        p = rng.uniform(0.2, 2.5)
        w = np.arange(1, B + 1, dtype=float) ** (-p)
        pool_blocks.append(w / w.sum())

    # exponential decay
    for _ in range(4):
        rate = rng.uniform(0.001, 0.02)
        w = np.exp(-rate * np.arange(B))
        pool_blocks.append(w / w.sum())

    # Beta‑family (symmetric and skewed)
    for _ in range(6):
        alpha = rng.uniform(0.1, 3.0)
        beta = rng.uniform(0.1, 3.0)
        x = (np.arange(B, dtype=float) + 0.5) / B
        w = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
        w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
        if w.sum() > 0:
            pool_blocks.append(w / w.sum())

    # completely random block vectors
    for _ in range(6):
        w = rng.random(B)
        pool_blocks.append(w / w.sum())

    # optionally seed from previous global result
    if 'height_sequence_1' in globals():

    # --------------------------------------------------------------
