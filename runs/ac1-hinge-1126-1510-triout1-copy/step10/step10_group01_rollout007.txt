"""
Propose a step‑function (non‑negative vector) that minimises

        2·n·max(a*a) / (sum(a)**2)

under the constraints of ``evaluate_sequence``.

Algorithm
---------

1. **Problem size** – work with n=4000 (the same size as the reference
   solution).  All vectors are kept on the probability simplex
   (sum=1) so that the objective simplifies to 2·n·max(conv).

2. **Utility routines** – fast convolution using FFT, projection onto the
   simplex, softmax (log‑odds parametrisation) and a numerically stable
   log‑sum‑exp smooth approximation of the max.

3. **Cheap parametric sweep** – generate many candidates from a
   collection of families (including a new “edge_peaks” family that puts
   most mass near the ends) and keep the best.

4. **Smooth L‑BFGS optimisation** – minimise the smooth surrogate
   (log‑sum‑exp of the convolution) using many random restarts.
   The smoothing parameter λ is driven down through a long schedule,
   ending at 1e‑07 so the surrogate is virtually the true max.

5. **Fine‑grained projected sub‑gradient descent** – starting from the
   best L‑BFGS result, run a classic sub‑gradient method with an adaptive
   back‑tracking line‑search.  The iterate is always projected onto the
   simplex.

6. **Polishing via local random moves** – a short simulated‑annealing
   phase that transfers a tiny amount of mass between two coordinates.
   Accepted moves are those that improve the true (unsmoothed) objective,
   otherwise they are accepted with probability exp(−Δ/T).

7. **Final evaluation** – the best vector found is returned as a plain
   Python list.  ``evaluate_sequence`` will automatically apply the
   required clipping and normalisation.

All phases respect the global deadline (default 1000 s).  The best
candidate seen at any time is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize
from numpy.fft import fft, ifft

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) under the assumption ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if len(rho) == 0:
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed objective 2·n·smooth_max(conv) where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    Fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(Fa * Fa))[: 2 * n - 1]

    # ----- smooth max via log‑sum‑exp -----
    max_c = np.max(conv_full)
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient wrt conv entries -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # ----- back‑prop through convolution (a'*a) -----
    a_rev = a[::-1]
    Fa_rev = fft(a_rev, fft_len_g)
    Fw = fft(w, fft_len_g)
    conv_rev = np.real(ifft(Fa_rev * Fw))
    # extract the part that corresponds to gradient w.r.t a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def run_lbfgs_multi_restart(
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    rng: np.random.Generator,
    n_restarts: int = 12,
) -> tuple[np.ndarray, float]:
    """
    Run many random‑restart L‑BFGS optimisations on the smoothed objective.
    Returns the best vector found (sum=1) and its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)

    # a schedule of smoothing parameters that ends essentially at the true max
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06,
        5e-07, 2e-07, 1e-07,
    ]

    maxiter_per_lam = 2500

    for restart in range(n_restarts):
        if time.time() >= deadline:
            break

        # random start: Dirichlet with concentration 0.5 (often sparse)
        a0 = rng.gamma(0.5, 1.0, size=n)
        a0 /= a0.sum()
        y = np.log(a0 + 1e-12)          # logits

        for lam in lam_schedule:
            if time.time() >= deadline:
                break

            res = minimize(
                fun=smooth_obj_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
            y = res.x
            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[lbfgs] restart {restart} new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_descent(
    a0: np.ndarray,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with a simple back‑tracking line search.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # compute exact sub‑gradient of max(conv)
        Fa = fft(a, fft_len_conv)
        conv = np.real(ifft(Fa * Fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        Fmask = fft(mask, fft_len_g)
        Fa_rev = fft(a_rev, fft_len_g)
        tmp = np.real(ifft(Fa_rev * Fmask))
        g = 2.0 * tmp[(n - 1) : (n - 1 + n)]

        # try a decreasing step size until improvement
        step = 0.6 / (it + 1.0)
        improved = False
        for _ in range(10):
            a_trial = simplex_projection(a - step * g)
            sc = fast_score(a_trial, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a_trial.copy()
                improved = True
                a = a_trial
                break
            # accept a non‑improving move only if it reduces the true max
            cur_sc = fast_score(a, fft_len_conv, n)
            if sc < cur_sc:
                a = a_trial
                improved = True
                break
            step *= 0.5
        if not improved:
            # if no improvement after line‑search, just project with current step
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


def annealing_refinement(
    a0: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climb that moves a tiny amount of mass
    between two positions.  The temperature decays with the square‑root
    of the remaining time.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    cur_score = best_score

    start = time.time()
    total = deadline - start
    T0 = max(0.05 * best_score, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue
        delta = a[i] * rng.random() * 0.25     # move at most 25 % of source mass
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep (including a new one)
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative vector (sum=1) from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        base = i + 1 if inc else n - i
        a = base ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        a = mask.astype(float)
        if a.sum() == 0.0:
            a[0] = 1.0
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "sinusoid":
        freq = rng.integers(1, 10)
        a = np.maximum(np.sin(2 * np.pi * freq * i / n), 0.0) + 0.01
    elif family == "edge_peaks":
        # concentrate mass near both ends; p is negative for a heavy tail
        p = rng.uniform(-3.0, -0.2)
        a = (i + 1) ** p + (n - i) ** p
        a = np.maximum(a, 0.0)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_frac: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Lightweight random sweep over several families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    end = time.time() + budget_frac * 0.10   # 10 % of total budget
    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "sinusoid",
        "edge_peaks",
        "random",
    ]

    while time.time() < end:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")
    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence of length 4000 that minimises
    the evaluation function.  Returns a list of floats (the height sequence).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # ------------------------------------------------------------------
    # initialise with uniform distribution (sum=1)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # if a previous construction exists, interpolate it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] imported previous fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # ------------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, budget_s, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] sweep best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # global timing
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # allocate time slices (tuned for the 1000 s budget)
    phase1_end = start_time + 0.60 * budget_s   # heavy L‑BFGS
    phase2_end = start_time + 0.85 * budget_s   # sub‑gradient fine tune
    phase3_end = deadline                       # final annealing

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart L‑BFGS on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < phase1_end:
        a_lbfgs, sc_lbfgs = run_lbfgs_multi_restart(
            n,
            fft_len_conv,
            fft_len_g,
            phase1_end,
            rng,
            n_restarts=20,
        )
        if sc_lbfgs < best_score:
            best_score, best_a = sc_lbfgs, a_lbfgs.copy()
            print(f"[phase1] L‑BFGS best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with back‑tracking
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        a_sub, sc_sub = subgradient_descent(
            best_a,
            fft_len_conv,
            fft_len_g,
            phase2_end,
        )
        if sc_sub < best_score:
            best_score, best_a = sc_sub, a_sub.copy()
            print(f"[phase2] sub‑gradient best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simple annealing refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        a_ann, sc_ann = annealing_refinement(
            best_a,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        if sc_ann < best_score:
            best_score, best_a = sc_ann, a_ann.copy()
            print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final evaluation with the official function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
