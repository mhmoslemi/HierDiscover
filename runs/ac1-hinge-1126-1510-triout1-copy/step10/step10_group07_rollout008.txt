def project_simplex(v): 
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def proj_subgradient_optimize(a_start, deadline, n, fft_len, rng, scale=1.0):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline:
        # compute convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)
        # compute gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]
        # step size
        eta = scale / np.sqrt(it + 1.0)  # diminishing
        # gradient descent step (additive)
        a_new = a - eta * g
        # projection to simplex
        a_new = project_simplex(a_new)
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score

def simulated_annealing_optimize(a_start, deadline, n, fft_len, rng, init_temp=0.01, decay=0.9999):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    temp = init_temp
    while time.time() < deadline:
        # generate neighbor: choose i, j distinct; delta random up to fraction of a[i]
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            j += 1
        # ensure non-zero i
        if a[i] == 0:
            # maybe choose new i
            continue
        delta = rng.random() * a[i] * 0.5  # move up to half mass
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # ensure nonnegative
        a_new[i] = max(a_new[i], 0.0)
        # renormalize to sum 1 (small drift)
        a_new /= a_new.sum()
        score = fast_score(a_new, fft_len, n)
        # acceptance
        if score < best_score or rng.random() < np.exp((best_score - score) / (temp + 1e-12)):
            a = a_new
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
        temp = temp * decay
    return best_a, best_score

"""
Optimiser for the step‑function bound.

The algorithm consists of several orthogonal phases that together
explore the feasible simplex aggressively while staying within a
single‑CPU 1000 s budget.

1.  Initialise a diverse pool of candidates (uniform, random, exponential,
    power‑law, spectral, and the previous best if available).
2.  Run a few Adam optimisations on a log‑softmax parametrisation with a
    decreasing “smooth‑max’’ parameter (log‑sum‑exp).  Each restart is
    stopped as soon as the global deadline is reached.
3.  Refine the current best vector with a *projected sub‑gradient* method
    on the true max‑convolution (additive updates + projection onto the
    probability simplex).
4.  Apply an exponentiated‑gradient descent (multiplicative updates) for
    a short burst – it is cheap and often pushes the solution further.
5.  Perform a deterministic hill‑climb that moves a small fraction of
    probability mass from the “worst’’ coordinate (largest contribution
    to the maximal convolution entry) toward the “best’’ one.
6.  Run a lightweight simulated‑annealing search that perturbs two
    coordinates at a time and accepts uphill moves with a geometrically
    decaying temperature.
7.  Finally carry out a convex‑combination (mixture) search among the
    top‑K vectors discovered so far using a cheap golden‑section line
    search.

All vectors are kept on the simplex (∑a = 1, a ≥ 0).  The fast score
`fast_score` evaluates the bound without the expensive denominator,
because the denominator is constant (=1) on the simplex.
The best vector is returned as a plain Python list; it is also stored
in the global name `height_sequence_1` for possible reuse in later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming ∑a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero, return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initialisers
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random probability vector."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and project back onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start‑vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed surrogate
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.06

    y = np.log(a_start + 1e-12)          # initialise logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def proj_subgrad_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 1.0):
    """
    Additive sub‑gradient descent on the true max‑convolution,
    projected onto the simplex after each step.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        a_new = a - eta * g
        a_new = project_simplex(a_new)

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 5.0):
    """
    Multiplicative (exponentiated) gradient descent on the hard max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a = project_simplex(a)

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb with geometric step reduction
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Move a fraction of mass from the coordinate that contributes most
    to the current maximal convolution entry toward a coordinate with the
    smallest partner value.  Step size shrinks geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner values for the active index
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: smallest partner (skip the same index)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new = project_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size for possible aggressive moves
        else:
            step = max(step * 0.7, min_step)

    return a, score


# ----------------------------------------------------------------------
# Simple simulated‑annealing search
# ----------------------------------------------------------------------
def simulated_annealing_optimize(a_start: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 fft_len: int,
                                 rng: np.random.Generator,
                                 init_temp: float = 0.02,
                                 decay: float = 0.9999):
    """
    Randomly move a fraction of mass between two coordinates;
    accept uphill moves with a geometrically decaying temperature.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    temp = init_temp

    while time.time() < deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 0.0:
            continue
        delta = rng.random() * a[i] * 0.5   # move at most half of a[i]
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_simplex(a_new)

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            diff = score - best_score
            accept_prob = np.exp(-diff / (temp + 1e-16))
            if rng.random() < accept_prob:
                a = a_new
        temp = max(temp * decay, 1e-6)

    return best_a, best_score


# ----------------------------------------------------------------------
# Convex‑combination exploration among top candidates
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair of candidates, cheap golden‑section line search over t∈[0,1]
    to minimise the bound.
    Returns (best_a, best_score).
    """
    phi = (1 + np.sqrt(5.0)) / 2.0
    invphi = (np.sqrt(5.0) - 1) / 2.0

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
            fd = fast_score((1.0 - d) * a + d * b, fft_len, n)

            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc = fast_score((1.0 - c) * a + c * b, fft_len, n)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd = fast_score((1.0 - d) * a + d * b, fft_len, n)

            tm = 0.5 * (a0 + b0)
            fs = fast_score((1.0 - tm) * a + tm * b, fft_len, n)
            if fs < best_score:
                best_score = fs
                best_a = ((1.0 - tm) * a + tm * b) / ((1.0 - tm) * a + tm * b).sum()

    return best_a, best_score

# Main entry point required by the task
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we keep the historic length (4000) which balances
    # quality and speed.  The FFT length is the next power of two.
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from previous global result if present
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Build a modest pool of diverse start vectors
    # ------------------------------------------------------------------
    cand0 = a_start.copy()
    cand1 = spectral_init(n)
    cand2 = random_initial(n, rng)
    cand3 = add_noise(cand0, rng.uniform(0.001, 0.01), rng)
    candidates = [cand0, cand1, cand2, cand3]

    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budgeting
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – multiple Adam restarts with a finer lam schedule
    # --------------------------------------------------------------
    lam_schedule_fine = [
        0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005, 0.0002,
        0.0001, 0.00005, 0.00002, 0.00001, 5e-06, 2e-06
    ]
    steps_per_lam = 300
    restarts = 8

    for r in range(restarts):
        if time.time() >= deadline:
            break
        if r == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule_fine,
            steps_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam‑restart {r+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best from Adam: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (long run)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = proj_subgrad_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            scale=1.2,
        )
        print(f"[Projected‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after sub‑grad: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            scale=4.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after exp‑gradient: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – fine deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_step=0.12,
            min_step=1e-8,
        )
        print(f"[Hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after hill‑climb: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – simulated‑annealing exploration
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = simulated_annealing_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            init_temp=0.02,
            decay=0.9995,
        )
        print(f"[Sim‑anneal] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after annealing: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – convex‑combination search among the top few vectors
    # --------------------------------------------------------------
    # Gather the top‑K vectors discovered so far (including the current best)
    top_vectors = [best_a]
    top_scores = [best_score]

    # Add a few random enrichments
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len_conv, n)
        top_vectors.append(a_tmp)
        top_scores.append(sc_tmp)

    # Keep the best K of them
    K = 5
    ordered = sorted(zip(top_scores, top_vectors), key=lambda x: x[0])
    selected = ordered[:K]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len_conv,
            deadline,
        )
        print(f"[Mixture‑search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose the best vector under the historic name for possible reuse
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
