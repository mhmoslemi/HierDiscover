Now define valley_shape:

```
def valley_shape(n, lo_ratio, exponent):
    # lo_ratio: value at centre relative to ends (0..1)
    # exponent: shape exponent controlling steepness; >=0
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre) / centre   # from 0 at centre to 1 at ends
    shape = lo_ratio + (1 - lo_ratio) * (dist ** exponent)
    return shape
```

def adam_optimize(a_start, deadline, n, fft_len, rng,
                  init_lr=0.2,
                  beta1=0.9, beta2=0.999,
                  eps=1e-8,
                  max_iters=5000):
    # returns best a and its fast_score within time.
    y = np.log(a_start + 1e-12)  # logit
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)
    t = 0
    total_iters = max_iters
    # We'll schedule lam_factor decreasing over iterations
    while time.time() < deadline and t < total_iters:
        lam_factor = 0.3 * (0.5 ** (t / total_iters))  # decays from 0.3 to small
        f, grad_y = objective_and_grad(y, n, fft_len, fft_len, lam_factor)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step
        # compute current a
        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        t += 1
    return best_a, best_score

"""
Propose a low‑score non‑negative step sequence.

Approach
--------
* Work on the simplex (soft‑max of logits) → sum = 1, so the official
  objective reduces to 2·n·max(conv(a,a)).
* Initialise a diverse pool (uniform, spectral, window functions, random
  Dirichlet / exponential / “valley” shapes) and keep the best.
* Repeatedly optimise the best candidate with Adam on the *smoothed* max
  of the self‑convolution (log‑sum‑exp).  The smoothing factor (lam_factor)
  decays exponentially so the optimisation gradually focuses on the true
  max.
* After the Adam phase run a few rounds of exact exponentiated‑gradient
  descent (top‑fraction mask) and a projected sub‑gradient descent to
  polish the solution.
* Finish with an aggressive greedy hill‑climb that moves small mass
  between random indices only when the score improves.
* All phases respect a global 1000 s budget; the currently best sequence
  is returned (converted to a plain list) and evaluated with
  ``evaluate_sequence`` at the end.
"""

import time
import numpy as np
from scipy.optimize import minimize  # only for fallback, not essential


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact objective for a probability vector (sum(a)=1):
        2·n·max(conv(a,a))
    """
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisers (all generate non‑negative vectors that are later normalised)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_pad = np.zeros(L)
    u_pad[:target_len] = uniform
    S = np.fft.fft(u_pad)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a = np.real(np.fft.ifft(A))[:n]
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑based Dirichlet draw (concentration <1 gives sparse vectors)."""
    concentration = rng.uniform(0.05, 0.5)
    a = rng.gamma(concentration, 1.0, size=n)
    a = a / a.sum()
    return a


def exponential_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Discrete exponential a_i ∝ exp(-c·i).  c drawn at random."""
    c = rng.uniform(0.001, 0.02)          # slower decay → smoother shape
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    a = a / a.sum()
    return a


def valley_shape(n: int, lo_ratio: float, exponent: float) -> np.ndarray:
    """
    Symmetric valley: low in the centre, high at the ends.
    lo_ratio ∈ [0,1] is the relative height at the centre.
    exponent ≥ 0 controls steepness.
    """
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    dist = np.abs(idx - centre) / centre                # 0 at centre, 1 at ends
    shape = lo_ratio + (1.0 - lo_ratio) * (dist ** exponent)
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    return shape


def random_valley(n: int, rng: np.random.Generator) -> np.ndarray:
    lo = rng.uniform(0.0, 0.8)          # centre can be almost zero
    exp = rng.uniform(0.3, 5.0)         # steepness
    return valley_shape(n, lo, exp)


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

# Optimisation primitives
# ----------------------------------------------------------------------


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  init_lr: float = 0.4,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-8,
                  max_iters: int = 4000) -> tuple[np.ndarray, float]:
    """
    Adam on the smoothed max‑convolution.
    Returns the best vector found during the run and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    # decay schedule for the smoothing factor
    lam_start = 0.30
    lam_end = 5e-07
    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # exponential decay of lam_factor
        lam_factor = lam_start * (lam_end / lam_start) ** (it / max_iters)

        f, grad = objective_and_grad(y, n, fft_len, fft_len, lam_factor)

        # Adam update (minimisation → move opposite to grad)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        # limit the step size to keep y well‑behaved
        step = np.clip(step, -5.0, 5.0)

        y = y - step

        a = softmax(y)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          top_fraction: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the *exact* max convolution.
    A mask of the top‑fraction entries smooths the sub‑gradient.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline:
        # exact self‑convolution
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        # construct mask of top entries
        kth = max(1, int(np.ceil(top_fraction * conv.size)))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        # gradient wrt a (cf. derivation in original script)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_conv = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = 1.0 / np.sqrt(it + 1.0)         # diminishing step size
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
        else:
            a = a / a_sum

        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        it += 1
    return best_a, best_sc


def subgradient_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         step0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline:
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]

        kth = max(1, int(0.02 * conv.size))
        thresh = np.partition(conv, -kth)[-kth]
        mask = (conv >= thresh).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        grad_full = np.fft.irfft(fa_rev * fm, fft_len)
        g = 2.0 * grad_full[(n - 1):(n - 1 + n)]

        mean_g = np.mean(g)
        d = -(g - mean_g)                      # feasible descent direction

        alpha = step0 / np.sqrt(it + 1.0)
        a_new = a + alpha * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new = a_new / s
        else:
            a_new = a.copy()

        sc_new = fast_score(a_new, fft_len, n)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_sc


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 120000,
               delta_frac: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Strictly improving greedy hill‑climb (pairwise mass transfer).
    """
    best = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # rollback
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a longer vector gives a finer discretisation.
    # --------------------------------------------------------------
    n = 8000                     # 8 k points is still fast for FFTs
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a diverse initial pool
    # --------------------------------------------------------------
    pool = []

    # previous construction (if any) interpolated to the new size
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0:
                a0 = a0 / s
            else:
                a0 = np.full(n, 1.0 / n)
            pool.append(a0)

    # deterministic families
    pool.append(spectral_init(n))
    pool.append(hann_init(n))
    pool.append(blackman_init(n))
    pool.append(gaussian_init(n))
    pool.append(np.full(n, 1.0 / n))            # uniform

    # random families
    for _ in range(4):
        pool.append(random_dirichlet(n, rng))
    for _ in range(4):
        pool.append(exponential_init(n, rng))
    for _ in range(4):
        pool.append(random_valley(n, rng))

    # pick the best start according to the *exact* fast_score
    best_a = pool[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in pool[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Phase A – Adam on smoothed max (many short runs)
    # --------------------------------------------------------------
    runs = 0
    while time.time() < deadline - 30.0:           # leave ~30 s for polishing
        # blend the current best with a fresh random candidate to keep diversity
        rand_vec = random_dirichlet(n, rng)
        alpha = rng.random() * 0.3                  # ≤30 % new mass
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()

        a_opt, sc_opt = adam_optimize(a_start,
                                      deadline,
                                      n,
                                      fft_len,
                                      init_lr=0.4,
                                      max_iters=2500)
        runs += 1
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[Adam run {runs}] new best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Phase B – exponentiated‑gradient on the exact max
    # --------------------------------------------------------------
    if time.time() < deadline - 15.0:
        best_a, best_sc = exp_gradient_optimize(best_a,
                                                deadline,
                                                n,
                                                fft_len,
                                                rng,
                                                top_fraction=0.02)
        print(f"[exp‑grad] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Phase C – projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        best_a, best_sc = subgradient_optimize(best_a,
                                               deadline,
                                               n,
                                               fft_len,
                                               step0=0.35)
        print(f"[sub‑grad] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Phase D – final greedy hill‑climb (deep polish)
    # --------------------------------------------------------------
    if time.time() < deadline:
        cur_sc = fast_score(best_a, fft_len, n)
        best_a, best_sc = hill_climb(best_a,
                                     cur_sc,
                                     fft_len,
                                     n,
                                     deadline,
                                     rng,
                                     max_iter=180000,
                                     delta_frac=0.05)
        print(f"[hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the integer‑friendly list and the official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)     # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
