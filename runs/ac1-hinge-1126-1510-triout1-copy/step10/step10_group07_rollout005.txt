def simplex_projection(v):
    v = np.maximum(v, 0)
    if v.sum() == 0:
        return np.full_like(v, 1.0/len(v))
    # else project onto simplex with sum=1
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # Find rho
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

Implementation:

def project_to_simplex(v):
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    # shift by subtracting max for numerical stability not needed.
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1 - cssv) / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        # all non-positive => return uniform
        return np.full_like(v, 1.0/n)
    rho = rho[-1]  # index
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Proposes a low‑score non‑negative step function sequence.

The search combines several complementary phases under the 1000 s budget:

1. **Diverse initialisation** – uniform, spectral, random, exponential,
   power‑law, edge‑heavy families and the previous global best (if any).
   All are normalised to the simplex (∑a = 1).

2. **Adam optimisation on a log‑softmax parametrisation** – a
   log‑sum‑exp surrogate of the max‑convolution is minimised.
   Several restarts with a decreasing “smooth‑max’’ factor (lam) are
   performed; the best vector is retained.

3. **Exponentiated‑gradient descent** on the true (hard) max‑convolution
   objective – a mirror‑descent step with multiplicative updates.

4. **Projected sub‑gradient descent** – a classic sub‑gradient step on
   the max‑convolution followed by a Euclidean projection onto the
   simplex (ensuring feasibility).  A simple back‑tracking line‑search
   is employed.

5. **Deterministic hill‑climbing** – moves a fraction of mass from the
   coefficient that contributes most to the current maximal convolution
   entry toward a coefficient with the smallest partner value; step size
   shrinks geometrically.

6. **Random local‑search (simulated‑annealing style)** – repeatedly
   transfers a small random amount of mass between two coordinates,
   accepting only improving moves; step size decays with time.

7. **Mixture search** – golden‑section line‑search over convex
   combinations of the top‑K candidates discovered so far.

The best vector found is finally evaluated with the provided
`evaluate_sequence` routine, printed and returned.  The sequence is also
exposed under the historic name `height_sequence_1` for reuse in later
runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective under the assumption ∑a = 1:
        score = 2·n·max(a*a)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project vector v onto the probability simplex {x ≥ 0, ∑x = 1}.
    """
    n = v.shape[0]
    # shift to avoid negative values before sorting (they become zero later)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive → return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Initialisation families
# ----------------------------------------------------------------------
def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector from a diverse set of modes."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.005, 0.04)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.015 * n), int(0.08 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Perturb a with Gaussian noise and re‑normalise onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def spectral_init(n: int) -> np.ndarray:
    """Construct a start vector from a square‑rooted flat spectrum."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def exponential_shape(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential decay a_i ∝ exp(-rate·i) + exp(-rate·(n‑1‑i))."""
    idx = np.arange(n, dtype=float)
    a = np.exp(-rate * idx) + np.exp(-rate * (n - 1 - idx))
    a /= a.sum()
    return a


def powerlaw_shape(n: int, p: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)⁻ᵖ + (n‑i)⁻ᵖ."""
    idx = np.arange(1, n + 1, dtype=float)
    a = (idx ** (-p)) + ((n - idx + 1) ** (-p))
    a /= a.sum()
    return a


def edge_heavy_shape(n: int, edge_frac: float, weight: float) -> np.ndarray:
    """Heavy mass on both edges, uniform elsewhere."""
    m_edge = max(1, int(n * edge_frac))
    a = np.full(n, 1.0 / n)
    a[:m_edge] = weight
    a[-m_edge:] = weight
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed surrogate (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed approximation of the max‑convolution using log‑sum‑exp.
    Returns (objective value, gradient w.r.t. y).
    """
    a = softmax(y)

    # full self‑convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))

    # extract gradient w.r.t. a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimisation on the smoothed surrogate
# ----------------------------------------------------------------------
def adam_optimize(a_start: np.ndarray,
                  lam_schedule,
                  steps_per_lam: int,
                  deadline: float,
                  fft_len_conv: int,
                  fft_len_g: int,
                  n: int):
    """
    Adam optimiser on the log‑sum‑exp surrogate.
    Returns (best_a, best_score).
    """
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05

    y = np.log(a_start + 1e-12)          # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            loss, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len_conv: int,
                          rng: np.random.Generator,
                          scale: float = 5.0):
    """
    Multiplicative mirror‑descent on the true max‑convolution.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        eta = scale / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with back‑tracking line‑search
# ----------------------------------------------------------------------
def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len_conv: int,
                                   rng: np.random.Generator,
                                   init_step: float = 0.5,
                                   min_step: float = 1e-6):
    """
    Classic sub‑gradient descent on the max‑convolution, followed by
    Euclidean projection onto the simplex.
    Returns (best_a, best_score).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    step = init_step
    while time.time() < deadline:
        # current max‑convolution and mask of active entries
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_conv)
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[(n - 1):(n - 1 + n)]

        # tentative move
        a_new = a - step * g
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len_conv, n)

        if new_score < best_score:
            # accept, increase step a little
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.1, 1.0)   # prevent overflow
        else:
            # reject – shrink step
            step = max(step * 0.7, min_step)

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Deterministic hill‑climb
# ----------------------------------------------------------------------
def fine_hill_climb(a: np.ndarray,
                    score: float,
                    fft_len: int,
                    n: int,
                    deadline: float,
                    init_step: float = 0.15,
                    min_step: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Transfer a fraction of mass from the coefficient that contributes most
    to the max‑convolution entry toward the coefficient with the smallest
    partner value.  Step size is reduced geometrically.
    """
    step = init_step
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_idx = np.argmax(conv)

        # partner[i] = a[max_idx - i] (zero for out‑of‑range)
        partner = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1)
        partner[idx] = a[max_idx - idx]

        contrib = a * partner
        i_max = np.argmax(contrib)

        # destination: smallest partner (avoid i_max itself)
        sorted_idxs = np.argsort(partner)
        i_min = sorted_idxs[0] if sorted_idxs[0] != i_max else sorted_idxs[1]

        delta = a[i_max] * step
        if delta <= 0.0:
            break

        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < score:
            a = a_new
            score = new_score
            # keep step size (allow aggressive moves)
        else:
            step = max(step * 0.7, min_step)
    return a, score


# ----------------------------------------------------------------------
# Convex‑combination (mixture) search among top candidates
# ----------------------------------------------------------------------
def mixture_search(candidates, scores, n, fft_len, deadline):
    """
    For each pair (a,b) perform a cheap golden‑section line‑search over
    t∈[0,1] to minimise the bound on the convex combination.
    Returns best vector and its score found during this phase.
    """
    def eval_mix(t, a, b):
        mix = (1.0 - t) * a + t * b
        mix /= mix.sum()
        return fast_score(mix, fft_len, n), mix

    phi = (1 + np.sqrt(5)) / 2
    invphi = (np.sqrt(5) - 1) / 2

    best_a = candidates[0]
    best_score = scores[0]

    K = len(candidates)
    for i in range(K):
        for j in range(i + 1, K):
            a = candidates[i]
            b = candidates[j]
            a0, b0 = 0.0, 1.0
            c = b0 - invphi * (b0 - a0)
            d = a0 + invphi * (b0 - a0)
            fc, _ = eval_mix(c, a, b)
            fd, _ = eval_mix(d, a, b)
            while time.time() < deadline and (b0 - a0) > 1e-4:
                if fc < fd:
                    b0, fd = d, fc
                    d = c
                    c = b0 - invphi * (b0 - a0)
                    fc, _ = eval_mix(c, a, b)
                else:
                    a0, fc = c, fd
                    c = d
                    d = a0 + invphi * (b0 - a0)
                    fd, _ = eval_mix(d, a, b)
            tm = 0.5 * (a0 + b0)
            fs, vec = eval_mix(tm, a, b)
            if fs < best_score:
                best_score = fs
                best_a = vec.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Random local search (greedy “annealing” style)
# ----------------------------------------------------------------------
def random_local_search(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        init_frac: float = 0.05,
                        min_frac: float = 1e-5):
    """
    Greedy stochastic mass‑transfer search.
    At each iteration a random source i (a[i]>0) and a random destination j
    are chosen; a fraction of the source mass is moved.  Only improving
    moves are kept.  The fraction decays over time.
    """
    a = a_start.copy()
    best_score = fast_score(a, fft_len, n)

    total_time = deadline - time.time()
    while time.time() < deadline:
        elapsed = time.time() - (deadline - total_time)
        frac = init_frac * (1.0 - elapsed / total_time) + min_frac * (elapsed / total_time)

        # pick a source with non‑trivial mass
        src_candidates = np.where(a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        i = rng.choice(src_candidates)
        # pick a distinct destination
        j = rng.integers(n)
        while j == i:
            j = rng.integers(n)

        delta = a[i] * frac
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = project_to_simplex(a_new)

        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            a = a_new
    return a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a minimal evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 provides a good trade‑off between quality and speed.
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Gather previous best if it exists.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # interpolate to current length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, prev)
            a_start = np.maximum(a_start, 0.0)
            a_start = a_start / a_start.sum()
        else:
            a_start = np.full(n, 1.0 / n)
    else:
        a_start = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # --------------------------------------------------------------
    candidates = [
        a_start,
        spectral_init(n),
        random_initial(n, rng),
        add_noise(a_start, rng.uniform(0.001, 0.01), rng)
    ]

    # add a few parametrised families
    for _ in range(5):
        rate = rng.uniform(0.0005, 0.02)
        candidates.append(exponential_shape(n, rate))
        p = rng.uniform(0.2, 3.0)
        candidates.append(powerlaw_shape(n, p))
        edge_frac = rng.uniform(0.005, 0.04)
        weight = rng.uniform(0.5, 3.0) * (np.sqrt(n / 2.0) / n)
        candidates.append(edge_heavy_shape(n, edge_frac, weight))

    # evaluate and keep the top few
    scored = [(fast_score(c, fft_len_conv, n), c) for c in candidates]
    scored.sort(key=lambda x: x[0])
    top_k = 8
    top_vectors = [vec for _, vec in scored[:top_k]]
    top_scores = [sc for sc, _ in scored[:top_k]]
    best_a = top_vectors[0].copy()
    best_score = top_scores[0]

    print(f"[init] n={n}, best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budgeting
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – Adam optimiser (multiple restarts, finer schedule)
    # --------------------------------------------------------------
    lam_schedule_fine = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005,
        0.0002, 0.0001, 0.00005, 0.00002
    ]
    steps_per_lam = 180
    restarts = 6
    for r in range(restarts):
        if time.time() >= deadline:
            break
        if r == 0:
            a0 = best_a.copy()
        else:
            a0 = add_noise(best_a, rng.uniform(0.001, 0.02), rng)

        a_opt, sc_opt = adam_optimize(
            a0,
            lam_schedule_fine,
            steps_per_lam,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[Adam‑restart {r+1}] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after Adam: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = exp_gradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            scale=5.0,
        )
        print(f"[Exp‑gradient] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after exp‑gradient: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = projected_subgradient_optimize(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            init_step=0.6,
            min_step=1e-5,
        )
        print(f"[Proj‑subgrad] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after proj‑subgrad: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – deterministic hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = fine_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_step=0.15,
            min_step=1e-6,
        )
        print(f"[Hill‑climb] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after hill‑climb: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – random local search (greedy annealing)
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_opt, sc_opt = random_local_search(
            best_a,
            deadline,
            n,
            fft_len_conv,
            rng,
            init_frac=0.04,
            min_frac=1e-5,
        )
        print(f"[Random‑local] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after random‑local: {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – convex‑combination mixture search among top candidates
    # --------------------------------------------------------------
    # collect a fresh pool of top‑K vectors
    pool_vectors = [best_a]
    pool_scores = [best_score]
    for _ in range(4):
        a_tmp = random_initial(n, rng)
        sc_tmp = fast_score(a_tmp, fft_len_conv, n)
        pool_vectors.append(a_tmp)
        pool_scores.append(sc_tmp)

    # keep the best K = 5
    combined = list(zip(pool_scores, pool_vectors))
    combined.sort(key=lambda x: x[0])
    K = 5
    selected = combined[:K]
    top_scores, top_vectors = zip(*selected)

    if time.time() < deadline:
        a_opt, sc_opt = mixture_search(
            list(top_vectors),
            list(top_scores),
            n,
            fft_len_conv,
            deadline,
        )
        print(f"[Mixture‑search] score={sc_opt:.6f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"   ==> new global best after mixture: {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with security‑checked routine
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:          # pragma: no cover
        print("[final] evaluation raised an exception:", e)
        final_eval = None

    # expose for possible reuse by downstream runs
    global height_sequence_1
    height_sequence_1 = final_seq

    return final_seq
