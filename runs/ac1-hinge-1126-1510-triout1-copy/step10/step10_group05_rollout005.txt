def annealing_exact(a_start, start_score, deadline, n, fft_len, rng):
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score
    start_time = time.time()
    total_time = deadline - start_time
    # parameters
    T0 = max((best_score - 1.5) * 0.02, 1e-5) # maybe
    # fractions: start large (0.2) -> end small (0.005)
    while time.time() < deadline:
        elapsed = time.time() - start_time
        fraction = (deadline - time.time()) / total_time
        T = max(T0 * fraction, 1e-12)
        max_frac = 0.02 + 0.25 * fraction  # start 0.27 maybe
        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # ensure moving from larger to smaller
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0:
            continue
        # pick delta: random fraction up to max_frac
        delta = a[src] * rng.random() * max_frac
        if delta <= 0:
            continue
        # Save old a for updates
        a_old = a.copy()
        # Update conv exactly
        conv_new = conv.copy()
        # shift contributions
        conv_new[src:src+n] -= 2*delta*a_old
        conv_new[dst:dst+n] += 2*delta*a_old
        # delta^2 contributions
        idx2src = 2*src
        if idx2src < len(conv_new):
            conv_new[idx2src] += delta*delta
        idx2dst = 2*dst
        if idx2dst < len(conv_new):
            conv_new[idx2dst] += delta*delta
        idxsd = src+dst
        if idxsd < len(conv_new):
            conv_new[idxsd] -= 2*delta*delta
        # Compute new max
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        # Accept condition
        if new_score <= curr_score or rng.random() < np.exp((curr_score - new_score)/T):
            # accept
            a = a_old.copy()
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        # optionally recompute conv exactly every certain steps to prevent drift
        # We can recompute after some steps number or time.
        # But using exact update, drift minimal.

    return best_a, best_score

Improved optimiser for the step‑function evaluation problem.

The algorithm works in four stages (all respecting the 1000 s budget):

1. **L‑BFGS on a log‑sum‑exp smoothed max‑convolution** – many random
   restarts, a long cascade of smoothing factors that go down to 5·10⁻⁷.
   Variables are the soft‑max logits, guaranteeing the probability simplex.

2. **Adam on the same smoothed objective** – continues the refinement with
   a tighter smoothing schedule and a larger initial learning‑rate.

3. **Exponentiated‑gradient descent on the hard max‑convolution** – many
   iterations (≈3·10⁵) with an aggressive η‑schedule.  This moves mass
   away from indices that contribute most to the current maximal
   autocorrelation.

4. **Exact simulated‑annealing hill‑climb** – moves a random fraction of
   mass from a “source’’ index to a “destination’’ index while updating
   the convolution *exactly* (including the δ² terms).  Temperature and
   move‑size both decay linearly, and the best sequence seen is returned.

All stages keep the vector on the probability simplex, so the final
score equals `2·n·max(conv(a,a))`.  The routine works for any length;
a length of 8000 gives a finer search space while still being fast.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient w.r.t. logits
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # soft‑max
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-15
    shifted = conv_full - max_c               # ≤ 0
    exp_shift = np.exp(shifted / lam)         # safe because shifted≤0
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. conv entries
    w = exp_shift / Z                         # weights, sum to 1
    # back‑prop through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]      # gradient w.r.t. a (before soft‑max)

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} → new best fast_score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[Adam] λ={lam:.1e} step={steps} → score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (hard max)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]    # sub‑gradient of max‑conv

        eta = eta0 / np.sqrt(it + 1.0)

        # mirror‑descent / exponentiated‑gradient step
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[exp‑grad] it={it} → score={best_score:.6f}")

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact simulated‑annealing hill‑climb (Δ²‑exact convolution update)
# ----------------------------------------------------------------------


def annealing_exact(a_start: np.ndarray,
                    start_score: float,
                    deadline: float,
                    n: int,
                    fft_len: int,
                    rng: np.random.Generator):
    """
    Simulated‑annealing hill‑climb with *exact* convolution updates.
    Moves a random fraction of mass from a source index to a destination
    index, accepts according to a linear cooling schedule and keeps the
    best sequence seen.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact linear convolution, length 2n-1
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time

    # temperature schedule parameters
    T0 = max((best_score - 1.5) * 0.02, 1e-5)          # start temperature
    max_frac_start = 0.25
    max_frac_end = 0.01
    recompute_every = 5000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        # linear cooling
        T = max(T0 * remaining_frac, 1e-12)
        max_frac = max_frac_end + (max_frac_start - max_frac_end) * remaining_frac

        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue

        # always move mass from larger entry to smaller one
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # --- exact convolution update ------------------------------------
        # conv_new = conv - 2*delta*shift(a, src) + 2*delta*shift(a, dst)
        #          + delta² * (e_dst*e_dst + e_src*e_src - 2*e_src*e_dst)
        conv_new = conv.copy()
        # shift contributions (vectorised slice updates)
        conv_new[src : src + n] -= 2.0 * delta * a
        conv_new[dst : dst + n] += 2.0 * delta * a

        # delta² terms
        idx2src = 2 * src
        if idx2src < conv_new.size:
            conv_new[idx2src] += delta * delta
        idx2dst = 2 * dst
        if idx2dst < conv_new.size:
            conv_new[idx2dst] += delta * delta
        idxsd = src + dst
        if idxsd < conv_new.size:
            conv_new[idxsd] -= 2.0 * delta * delta
        # -----------------------------------------------------------------

        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            # occasional full recompute to flush numerical drift
            if accepted and (accepted % recompute_every) == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()

    print(f"[anneal‑exact] accepted moves={accepted}, best_score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 8000 entries (a good trade‑off between quality and speed)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        old_n = old.size
        n = max(8000, old_n * 2)
        # Interpolate old solution to the new length
        x_old = np.linspace(0.0, 1.0, old_n)
        x_new = np.linspace(0.0, 1.0, n)
        a0 = np.interp(x_new, x_old, old)
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
    else:
        n = 8000
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split (roughly 20 %/25 %/20 %/35 %)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05      # tiny safety margin

    phase1_end = start_time + 0.20 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.45 * budget_s   # Adam
    phase3_end = start_time + 0.70 * budget_s   # Exponentiated‑gradient
    # remaining → exact simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with many restarts and a very fine λ‑schedule
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
        0.05, 0.03, 0.018, 0.01, 0.006,
        0.003, 0.0015, 0.0008, 0.0004,
        0.0002, 0.0001, 0.00005, 0.00002,
        0.00001, 5e-6, 2e-6, 1e-6, 5e-7
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # 20 % of restarts are small Gaussian perturbations of the current best,
        # the rest are fresh random initialisations.
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)

        lam_schedule_adam = [
            0.4, 0.25, 0.15, 0.08, 0.04,
            0.02, 0.01, 0.005, 0.002, 0.001,
            0.0005, 0.0002, 0.0001, 5e-5, 2e-5,
            1e-5, 5e-6, 2e-6, 1e-6, 5e-7
        ]
        best_a, best_score = adam_optimize(
            y0,
            lam_schedule_adam,
            max_steps_per_lam=3000,
            deadline=phase2_end,
            n=n,
            fft_len=fft_len,
            lr0=0.14,
        )
        print(f"[Adam] post‑phase best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase3_end,
            n,
            fft_len,
            rng,
            eta0=24.0,               # more aggressive initial step‑size
        )
        print(f"[exp‑grad] post‑phase best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact simulated annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = annealing_exact(
            best_a,
            best_score,
            deadline,
            n,
            fft_len,
            rng,
        )
        print(f"[anneal‑exact] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the framework
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
