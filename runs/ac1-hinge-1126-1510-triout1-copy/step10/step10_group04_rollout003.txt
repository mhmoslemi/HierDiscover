"""
search for a non‑negative step sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser is a four‑stage pipeline:

1. **Parametric seeding** – we draw many candidates from a few
   analytically‑motivated families (power‑law, geometric and Beta‑shaped)
   on the probability simplex and keep the best one.
2. **L‑BFGS on a log‑sum‑exp smoothed max‑conv** – this quickly drives the
   vector into a good basin on the simplex (the variables are the logits of
   a soft‑max).
3. **Exponentiated‑gradient descent on the true max‑convolution** – a
   parameter‑free sub‑gradient method that stays on the simplex.
4. **Greedy hill‑climb + simulated annealing** – fine‑grained local moves
   and a temperature‑controlled random walk to escape any remaining local
   minima.

All stages watch the global deadline (1000 s) and stop early if time runs
out.  The final sequence is returned as a plain list of floats.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Score = 2·n·max(conv(a,a)) when sum(a)=1.
    a is assumed to be a probability vector.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Variables are logits y (a = softmax(y)).
    Returns (objective, gradient) w.r.t. y.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)          # initialise logits
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 12.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # sub‑gradient mask

        # sub‑gradient w.r.t. a (see derivation in the original code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        # multiplicative (EG) update
        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (global search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple temperature‑controlled random walk on the simplex.
    Moves a random fraction of mass from a source index (chosen
    proportionally to its current weight) to a destination index.
    Accepts uphill steps with probability exp(‑Δ/ T).
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12                # initial temperature (~5 % of score)

    while time.time() < deadline:
        # linear cooling
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction

        # adapt move size (large at the start, tiny at the end)
        max_frac = 0.02 + 0.18 * fraction

        # source biased by current mass, destination uniform
        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score

def hill_climb_incremental(a_start: np.ndarray,

    Greedy hill‑climb that moves a fraction of mass from a heavy index
    to a lighter one, accepting only improvements.  Convolution updates
    are performed incrementally for speed.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        # Fraction shrinks from ~20 % → ~2 % as time runs out
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        # source weighted by its mass, destination uniform
        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # occasional full recompute to avoid numeric drift
        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Parametric seeding (random families)
# ----------------------------------------------------------------------


def generate_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Draw a random distribution from one of three simple families
    that are known to produce low max‑convolution values.
    """
    family = rng.choice(['power', 'geom', 'beta'])
    if family == 'power':
        # a_i ∝ (i+off)^(-p) + (n-1-i+off)^(-p)
        p = rng.uniform(0.1, 4.0)
        off = rng.uniform(0.01, 10.0)
        i = np.arange(n, dtype=float)
        a = (i + off) ** (-p) + (n - 1 - i + off) ** (-p)
    elif family == 'geom':
        # a_i ∝ r**i + r**(n-1-i)
        r = rng.uniform(0.85, 0.9999)
        i = np.arange(n, dtype=float)
        a = r ** i + r ** (n - 1 - i)
    else:  # beta
        # a_i ∝ x^{α‑1} (1‑x)^{β‑1},  x = (i+0.5)/n
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        x = (np.arange(n, dtype=float) + 0.5) / n
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – a larger n gives more freedom to flatten the convolution.
    # --------------------------------------------------------------
    n = 8000                      # feel free to adjust; 8 k works comfortably in <1 s per FFT
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Optional warm‑start from a previously saved variable.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # linear interpolation to the new resolution
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 0 – random parametric seeding (≈10 % of the budget)
    # --------------------------------------------------------------
    phase0_end = start_time + 0.10 * budget_s
    while time.time() < phase0_end:
        a_cand = generate_candidate(n, rng)
        score = fast_score(a_cand, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_cand.copy()
            print(f"[seed] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on the smoothed max‑conv objective (≈30 % of budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s
    if time.time() < phase1_end:
        lam_sched = np.geomspace(0.5, 1e-6, num=12)
        best_a, best_score = lbfgs_optimize(best_a,
                                            lam_sched,
                                            maxiter_per_lam=400,
                                            deadline=phase1_end,
                                            fft_len=fft_len,
                                            n=n)
        print(f"[L‑BFGS] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient descent on the true max (≈35 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   deadline,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   eta0=15.0)
        print(f"[exp‑grad] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – greedy hill‑climb + simulated annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        # hill‑climb
        best_a, best_score = hill_climb_incremental(best_a,
                                                    best_score,
                                                    deadline,
                                                    rng)
        print(f"[hill‑climb] best fast_score={best_score:.6f}")

    if time.time() < deadline:
        # simulated annealing as a final polish
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 deadline,
                                                 rng)
        print(f"[SA] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
