"""
Proposes a non‑negative step‑function (probability) vector that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser is a multi‑phase pipeline (≤1000 s total) that
 * works on a larger problem size (n≈8000) – the bound
   2·n/(2·n‑1)≈1.00006 shows there is room for improvement over the
   previous 1.506 result.
 * uses many random restarts of a log‑sum‑exp smoothed‐max L‑BFGS,
   followed by Adam, exponentiated‑gradient, projected sub‑gradient
   and a simulated‑annealing hill‑climb.
 * finishes with a very fine L‑BFGS run (λ≈5·10⁻⁶, 5000 iter) to
   polish the best candidate.
All stages keep the vector on the probability simplex, so the
evaluation is simply `2·n·max(conv)`.  The returned list is ready for
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp max) – used by LBFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.  Variables y are logits; the actual vector is
    a = softmax(y). Returns (objective, grad w.r.t. y).
    """
    # soft‑max
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries (weights of the log‑sum‑exp)
    w = exp_shift / Z

    # back‑prop through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]          # gradient wrt a before softmax

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations using decreasing smoothing
    parameters λ.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Adam optimiser on smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns best a and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if steps % 200 == 0:
                    print(f"[Adam] lam={lam:.5f} step={steps} score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

# Exact exponentiated‑gradient descent on the true max
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent on the hard max‑convolution.
    Returns the best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def compute_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        alpha0: float = 0.5):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        conv = compute_conv_fft(a, fft_len)
        k = int(np.argmax(conv))

        # sub‑gradient of conv[k] w.r.t. a
        g = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        if idx.size:
            g[idx] = a[k - idx]

        alpha = alpha0 / np.sqrt(it + 1.0)

        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = best_a.copy()
            s = a_new.sum()

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score

def simulated_hill_climb(a_start: np.ndarray,
                         start_score: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Greedy hill‑climb with a temperature schedule.  Uses O(n) incremental
    updates of the convolution.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)            # exact linear convolution (2n‑1)
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    # annealing schedule
    T0 = max((best_score - 1.5) * 0.02, 1e-5)
    max_fraction_start = 0.25
    max_fraction_end = 0.02
    recompute_every = 1200
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        T = max(T0 * remaining_frac, 1e-12)
        max_fraction = max_fraction_end + (max_fraction_start - max_fraction_end) * remaining_frac

        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update (ignoring O(delta²) terms)
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            accepted += 1
            if accepted and (accepted % recompute_every) == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program improvement step (optional, invoked if time permits)
# ----------------------------------------------------------------------


def lp_improve(a: np.ndarray,
               n: int,
               fft_len: int,
               deadline: float,
               rng: np.random.Generator,
               max_iters: int = 20,
               t: float = 0.25):
    """
    Implements the classic LP‑based improvement from the paper:
    given a, solve  max Σ b_j  s.t. b≥0 and conv(a,b) ≤ M,
    then mix a←(1‑t)a + t·(b/Σb) and repeat.
    Uses SciPy's HiGHS linear programming solver (sparse).
    """
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = full_conv(a, fft_len)
        M = conv.max()

        # Build convolution matrix A (2n‑1 rows, n cols) as sparse diagonal
        # matrix where each diagonal offset d = -j (j=0..n-1) has
        # constant value a[j] and length n.
        offsets = -np.arange(n)               # [0, -1, -2, ..., -(n-1)]
        data = [np.full(n, a[j]) for j in range(n)]
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        c = -np.ones(n)                       # maximise Σ b_j  ⇒ minimise -Σ b_j
        bounds = [(0, None) for _ in range(n)]
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status != 0:
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break
        g = b / sum_b                        # unit‑sum candidate
        a = (1 - t) * a + t * g
        a = simplex_projection(a)            # numerical safety
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[LP] iter={it} score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 8000 entries gives a better asymptotic bound
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Starting point – interpolate from any previous solution, otherwise uniform
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Split the 1000 s budget among stages
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    phase1_end = start_time + 0.18 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.38 * budget_s   # Adam
    phase3_end = start_time + 0.56 * budget_s   # Exp‑grad
    phase4_end = start_time + 0.74 * budget_s   # Sub‑grad
    phase5_end = start_time + 0.92 * budget_s   # Sim‑anneal
    # remaining → optional LP refinement + final fine L‑BFGS

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with many random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.6, 0.4, 0.3, 0.22, 0.15, 0.1,
                    0.07, 0.045, 0.028, 0.016, 0.009,
                    0.005, 0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # 20 % of the restarts begin from a slightly perturbed best point,
        # the rest are fresh random draws.
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        lam_schedule_adam = [0.35, 0.25, 0.16, 0.09, 0.05,
                             0.03, 0.017, 0.009, 0.004, 0.002]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam,
            max_steps_per_lam=2500,
            deadline=phase2_end,
            n=n, fft_len=fft_len, lr0=0.13,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=12.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent on the hard max
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = subgradient_descent(
            best_a, phase4_end, n, fft_len, rng,
            alpha0=0.5,
        )
        print(f"[sub‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase5_end:
        best_a, best_score = simulated_hill_climb(
            best_a, best_score,
            phase5_end,
            n, fft_len,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional LP‑based refinement (if time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:   # keep ~30 s for final polishing
        best_a, best_score = lp_improve(
            best_a, n, fft_len, deadline, rng,
            max_iters=12, t=0.18,
        )
        print(f"[LP‑refine] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑grained L‑BFGS on a very small λ
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        lam_fine = [5e-6, 2e-6, 1e-6, 5e-7]
        a_fine, score_fine = lbfgs_continuation(
            best_a, lam_fine, maxiter_per_lam=3000,
            deadline=deadline, fft_len=fft_len, n=n,
        )
        if score_fine < best_score:
            best_score = score_fine
            best_a = a_fine.copy()
            print(f"[final‑L‑BFGS] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final official evaluation (may apply clipping/capping internally)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
