"""
Propose a low‑score step‑function.

The objective is 2·n·max(conv(a,a)) / (sum(a)^2).  Because the
objective is homogeneous, we restrict the search to probability
vectors (sum = 1, a_i ≥ 0).  The algorithm proceeds in four phases:

1. **Cheap parametric sweep** – explores a few analytic families
   (exponential, linear, mixtures, power‑law) and keeps the best
   candidate.

2. **Multiple exponentiated‑gradient runs** – mirror‑SGD on the true
   max‑convolution objective.  Several random restarts (Dirichlet
   draws) are performed; each run projects on the simplex after a
   multiplicative update.  The best result of all runs is kept.

3. **Deterministic gradient‑descent mass transfer** – using the exact
   sub‑gradient of the maximal convolution value we repeatedly move a
   small amount of probability mass from the index with the largest
   gradient to the index with the smallest gradient.  The convolution
   is updated incrementally, and the exact convolution is recomputed
   periodically to avoid numerical drift.

4. **Fine‑tuning simulated annealing** – random mass transfers are
   accepted with a Metropolis rule and a linear cooling schedule.

All phases respect the global time budget (`budget_s`).  The best
sequence found is returned and stored in the global variable
`height_sequence_1` for possible warm‑starts in future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming ‖a‖₁ = 1 (probability vector)."""
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def subgradient(a: np.ndarray, fft_len: int):
    """
    Compute a sub‑gradient of the max‑convolution objective and its value.

    Returns
    -------
    grad : np.ndarray, shape (n,)
        d/d a_i max_k (a * a)[k]  (the sub‑gradient).
    max_val : float
        The current maximal convolution entry.
    """
    n = a.shape[0]
    # Full convolution via FFT (fast)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    max_val = conv.max()

    # Tolerance for active set
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.rfft(mask, fft_len)
    fa_rev = np.fft.rfft(a_rev, fft_len)
    grad_full = np.fft.irfft(fa_rev * fm, fft_len)[: 2 * n - 1]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def param_scan(n: int, fft_len: int, deadline: float, rng: np.random.Generator):
    """Brute‑force over a few low‑dimensional families; return best found."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # Pure exponential
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=200):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # Linear decreasing
    max_slope = 1.0 / n
    for s in np.linspace(0.0, max_slope, num=200):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # Mixture exponential + uniform
    for lam in np.logspace(np.log10(1e-5), np.log10(0.02), num=50):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=11):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # Power‑law decay
    for p in np.linspace(0.2, 5.0, num=150):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – exponentiated‑gradient descent (multiple restarts)
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 6.0,
):
    """
    Mirrored SGD (multiplicative update) on the max‑convolution
    objective.  Returns the best vector found before `deadline`.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 1

    while time.time() < deadline:
        g, _ = subgradient(a, fft_len)
        eta = eta0 / np.sqrt(it)

        # Multiplicative update (exponentiated gradient)
        exponents = -eta * g
        exponents = np.clip(exponents, -50, 50)   # avoid overflow
        a = a * np.exp(exponents)

        # Project back onto simplex
        total = a.sum()
        if total <= 0.0:
            a = best_a.copy()
        else:
            a = a / total

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()

        it += 1

    return best_a, best_score

def deterministic_mass_transfer(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
):
    """
    Repeatedly move a small amount of probability mass from the index
    with the largest sub‑gradient to the index with the smallest.
    Updates the convolution incrementally; recomputes it from scratch
    periodically to avoid drift.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact convolution (size 2n‑1)
    best_a = a.copy()
    best_score = start_score

    max_fraction = 0.3      # max fraction of source mass moved per step
    recompute_interval = 5000
    min_delta = 1e-7
    it = 0

    while time.time() < deadline:
        # Gradient of the current maximal convolution entry
        g, max_val = subgradient(a, fft_len)

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if a[src] <= min_delta:
            # nothing to move – break to avoid endless looping
            break

        # Choose a transfer amount
        delta = a[src] * max_fraction * rng.random()
        delta = max(delta, min_delta)
        delta = min(delta, a[src])   # cannot move more than we have

        # Save copy of a for incremental update
        a_before = a.copy()

        # Perform mass transfer
        a[src] -= delta
        a[dst] += delta

        # Incremental update of the convolution:
        # conv_new = conv + 2·delta·diff   where diff = (e_dst - e_src) * a_before
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a_before
        diff[dst : dst + n] += a_before
        conv = conv + 2.0 * delta * diff

        # Periodic exact recomputation to keep numerical errors bounded
        if it % recompute_interval == 0:
            conv = np.convolve(a, a)

        curr_score = 2.0 * n * conv.max()
        if curr_score < best_score:
            best_score, best_a = curr_score, a.copy()
            print(f"[gd] iter {it:06d} best fast_score={best_score:.6f}")

        it += 1

        # Leave a small safety margin before the deadline
        if time.time() > deadline - 2.0:
            break

    return best_a, best_score

def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """Random mass transfers accepted via Metropolis rule with linear cooling."""
    a = a_start.copy()
    conv = np.convolve(a, a)
    curr_score = start_score
    best_a = a.copy()
    best_score = curr_score

    # Initialise temperature as a small fraction of the current score
    temp0 = max(best_score * 0.02, 1e-6)
    start_time = time.time()
    max_fraction = 0.20   # maximum fraction of source mass per proposal
    recompute_every = 4000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-9)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # always move from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental convolution update (linear in n)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - curr_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            accepted += 1
            if curr_score < best_score:
                best_score, best_a = curr_score, a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem size – 4000 is a good trade‑off between quality and speed.
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # Initialise from a previous run if one exists.
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            if a0.sum() == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (≈5 % of the budget)
    # ------------------------------------------------------------------
    phase0_deadline = start_time + 0.05 * budget_s
    if time.time() < phase0_deadline:
        cand_a, cand_score = param_scan(n, fft_len, phase0_deadline, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – multiple exponentiated‑gradient runs (≈50 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.55 * budget_s
    run_idx = 0
    while time.time() < phase1_deadline:
        # Choose starting point: mostly the current best, occasionally random
        if rng.random() < 0.7:
            a_start = best_a.copy()
        else:
            # Dirichlet with concentration 0.5 yields moderately sparse patterns
            a_start = rng.dirichlet(np.full(n, 0.5))

        # Allocate a modest slice of time for this run (max 15 s)
        remaining = deadline - time.time()
        slice_deadline = time.time() + min(15.0, remaining * 0.2)
        a_opt, score_opt = exp_gradient_optimize(
            a_start, slice_deadline, n, fft_len, rng, eta0=4.0
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[exp‑grad‑run{run_idx}] new best fast_score={best_score:.6f}")

        run_idx += 1
        if remaining < 30.0:
            break

    # ------------------------------------------------------------------
    # Phase 2 – deterministic gradient‑descent mass transfer (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.85 * budget_s
    best_a, best_score = deterministic_mass_transfer(
        best_a, best_score, phase2_deadline, n, fft_len, rng
    )
    print(f"[gd] phase completed, best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing fine‑tuning (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, n, rng
        )
        print(f"[anneal] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
