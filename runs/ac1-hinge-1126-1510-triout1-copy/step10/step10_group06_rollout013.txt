def generate_beta(n, p, q):
    i = np.arange(n)
    a = (i + 1) ** p * (n - i) ** q
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

"""
Improved step‑function search for the autocorrelation minimisation challenge.

The algorithm refines the previous method by:
* Adding a two‑parameter Beta family (a_i ∝ (i+1)^p·(n−i)^q) to the
  systematic grid search.
* Using a small Adam optimiser on the smooth log‑sum‑exp surrogate of the
  max‑convolution, providing a complementary first‑order method to the
  L‑BFGS routine.
* Keeping the proven sub‑gradient‑with‑momentum and simulated‑annealing
  phases, but with slightly tighter step‑size schedules.
* After a good solution for n=4000 is found, the vector is interpolated
  to n=8000 and a brief final sub‑gradient run is performed, exploiting
  the extra degrees of freedom to lower the objective further.

All phases respect a shared deadline (default ≈ 1000 s).  The best
candidate seen at any moment is returned, and the final official score
is printed via the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) under the invariant Σa = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. logits y
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_beta(n: int, p: float, q: float) -> np.ndarray:
    """Beta‑like shape: a_i ∝ (i+1)^p·(n-i)^q."""
    i = np.arange(n, dtype=float)
    a = (i + 1) ** p * (n - i) ** q
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a normalized non‑negative vector from a named family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "beta":
        p = rng.uniform(-5.0, -0.5)
        q = rng.uniform(0.0, 5.0)
        a = (i + 1) ** p * (n - i) ** q
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    else:
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Search over a handful of families (exp, power, logistic, beta)
    using modest grids.  Returns the best vector found.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # ----- exponential grid -------------------------------------------------
    for rate in np.linspace(0.0002, 0.0010, 41):          # 41 points
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # ----- power‑law grid ----------------------------------------------------
    for p in np.linspace(-5.0, -0.5, 19):                # 19 points
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # ----- logistic grid ------------------------------------------------------
    mus = np.linspace(0.2 * n, 0.8 * n, 7)               # 7 values
    sigmas = np.linspace(200.0, 1200.0, 7)               # 7 values
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # ----- Beta‑like grid -----------------------------------------------------
    # p ∈ [-5,-0.5] (decreasing) , q ∈ [0,5] (increasing tail)
    p_vals = np.linspace(-5.0, -0.5, 11)
    q_vals = np.linspace(0.0, 5.0, 11)
    for p in p_vals:
        for q in q_vals:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_beta(n, p, q)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] beta p={p:.2f} q={q:.2f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Quick random sweep over many families – sanity‑check that random
    sampling cannot beat the systematic grid.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10          # 10 % of total budget

    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "beta",
        "binary_sparse",
        "double_exp",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    L‑BFGS on the smoothed max objective for each λ in lam_schedule.
    """
    y = np.log(a_start + 1e-12)          # logits – stay on the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def adam_optimize(
    a_start: np.ndarray,
    lam_schedule,
    steps_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed surrogate.
    Operates on logits y to keep the simplex constraint automatically.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                return best_a, best_score
            f_val, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_y
            v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            # learning rate tuned for our scale
            lr = 0.07
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[adam] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum buffer
    beta = 0.85               # slightly larger momentum
    it = 0

    while time.time() < deadline:
        # true max‑convolution (FFT)
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decays like 0.6 / sqrt(it+1)
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts uphill moves with a square‑root
    cooling schedule.  Mass is transferred between two random indices.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem parameters (primary length)
    # ------------------------------------------------------------------
    n0 = 4000                                 # base length
    fft_len_conv0 = next_pow2(2 * n0 - 1)      # for a*a
    fft_len_g0 = next_pow2(3 * n0 - 2)         # for back‑propagation

    # ------------------------------------------------------------------
    # Initial candidate (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n0, 1.0 / n0)
    best_score = fast_score(best_a, fft_len_conv0, n0)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # If a previous construction is present, interpolate it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n0)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv0, n0)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Overall timing
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 0 – systematic grid search + cheap random sweep
    # ------------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n0, fft_len_conv0, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n0, fft_len_conv0, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed max (multiple restarts)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.33 * budget_s
    lam_schedule_lbfgs = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06,
    ]
    maxiter_per_lam = 800  # a bit lighter – many restarts are cheap

    while time.time() < phase1_end:
        # occasional jitter to escape plateaus
        if rng.random() < 0.3:
            a_start = best_a * rng.normal(1.0, 0.02, size=n0)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            rate = rng.uniform(0.0003, 0.0009)
            a_start = generate_exp(n0, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule_lbfgs,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv0,
            fft_len_g0,
            n0,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smoothed max (complementary to L‑BFGS)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    lam_schedule_adam = [
        0.5, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002,
    ]
    steps_per_lam = 120

    while time.time() < phase2_end:
        # initialise Adam from current best or a fresh exponential start
        if rng.random() < 0.5:
            a_start = best_a.copy()
        else:
            rate = rng.uniform(0.0002, 0.0010)
            a_start = generate_exp(n0, rate)

        a_opt, sc_opt = adam_optimize(
            a_start,
            lam_schedule_adam,
            steps_per_lam,
            phase2_end,
            fft_len_conv0,
            fft_len_g0,
            n0,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase2‑adam] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.80 * budget_s
    best_a, best_score = subgradient_optimize_momentum(
        best_a, phase3_end, fft_len_conv0, fft_len_g0, n0
    )
    print(f"[phase3] sub‑gradient best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing (final stochastic refinement)
    # ------------------------------------------------------------------
    phase4_end = start_time + 0.92 * budget_s
    best_a, best_score = simulated_annealing(
        best_a,
        best_score,
        fft_len_conv0,
        n0,
        phase4_end,
        rng,
    )
    print(f"[phase4] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optional upscale to n=8000 for a final polishing pass
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:  # spend at least ~30 s on the up‑scaled run
        n_big = n0 * 2
        fft_len_conv_big = next_pow2(2 * n_big - 1)
        fft_len_g_big = next_pow2(3 * n_big - 2)

        # linear interpolation of the best vector
        x_old = np.linspace(0, 1, n0)
        x_new = np.linspace(0, 1, n_big)
        a_big = np.interp(x_new, x_old, best_a)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # brief sub‑gradient refinement on the larger problem
        big_deadline = time.time() + min(remaining - 5.0, 200.0)
        a_big, score_big = subgradient_optimize_momentum(
            a_big, big_deadline, fft_len_conv_big, fft_len_g_big, n_big
        )
        # accept the up‑scaled solution if it improves the score
        if score_big < best_score:
            best_score, best_a = score_big, a_big
            n0 = n_big
            fft_len_conv0 = fft_len_conv_big
            fft_len_g0 = fft_len_g_big
            print(f"[upscale] improved fast_score={best_score:.6f} at n={n0}")

    # ------------------------------------------------------------------
    # Final exact evaluation with the official `evaluate_sequence`
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
