"""
Propose a low‑score step‑function sequence.

Algorithm
---------
1. **Parameter sweep** – a cheap exhaustive scan of a few low‑dimensional
   analytic families (exponential, linear, mixed exponential‑uniform, power‑law)
   normalised to a probability simplex.  The best candidate becomes the seed.

2. **Smoothed exponentiated‑gradient descent** – we minimise a smooth
   approximation of the max‑convolution using the log‑sum‑exp
   ``L(a)=log∑_k exp(α·(a*a)[k])``.
   The gradient of ``L`` is obtained via FFT‑based convolution and the
   update is performed in the exponentiated‑gradient (mirror‑descent) style,
   keeping the vector on the simplex.  The smoothing parameter ``α`` is
   increased linearly from a small value to a large one, so the optimisation
   first sees a smooth landscape and later focuses on the true max.

3. **Fine‑grained simulated annealing** – after the gradient phase we
   perform a Metropolis‑style random mass‑transfer search with incremental
   convolution updates.  The temperature is lowered linearly.

All phases respect a strict 1000 s wall‑clock budget.  The best sequence
found is stored in ``height_sequence_1`` for reuse and returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the official score under the normalisation
    ``sum(a)=1``.  Returns ``2·n·max(conv(a,a))``.
    """
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 0 – cheap parametric families
# ----------------------------------------------------------------------


def param_scan(
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
):
    """Scan a few low‑dimensional families and return the best candidate."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)

    # --- pure exponential -------------------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(2e-2), num=250):
        a = np.exp(-lam * np.arange(n))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- linear decreasing ------------------------------------------------
    for s in np.linspace(0.0, 1.0 / n, num=250):
        a = np.maximum(0.0, 1.0 - s * np.arange(n))
        if a.sum() == 0.0:
            continue
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    # --- mixture exponential + uniform ------------------------------------
    for lam in np.logspace(np.log10(1e-5), np.log10(2e-2), num=60):
        base = np.exp(-lam * np.arange(n))
        base /= base.sum()
        for alpha in np.linspace(0.0, 1.0, num=12):
            a = alpha * base + (1.0 - alpha) * np.full(n, 1.0 / n)
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score, best_a = score, a.copy()
            if time.time() >= deadline:
                return best_a, best_score

    # --- power‑law decay --------------------------------------------------
    for p in np.linspace(0.2, 5.0, num=180):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        if time.time() >= deadline:
            return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 1 – smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------


def smooth_exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 2.0,
    alpha_start: float = 5.0,
    alpha_end: float = 1500.0,
):
    """
    Mirror‑descent (exponentiated‑gradient) on a log‑sum‑exp
    approximation of the max‑convolution.
    """
    a = a_start.copy()
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0:
        a = a / a_sum
    else:
        a = np.full(n, 1.0 / n)

    # initial best
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()

    start_time = time.time()
    it = 1

    while time.time() < deadline:
        # ---------- compute convolution and smooth weights ----------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = conv.max()

        # linear schedule for the smoothing parameter α
        elapsed = time.time() - start_time
        phase_frac = elapsed / (deadline - start_time)
        alpha = alpha_start + phase_frac * (alpha_end - alpha_start)

        # softmax weights on the convolution values
        w_raw = np.exp(alpha * (conv - max_val))          # max exponent = 0
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(w_raw, 1.0 / w_raw.size)
        else:
            w = w_raw / w_sum

        # ---------- gradient of the smoothed objective ----------
        a_rev = a[::-1]
        grad_full = np.fft.irfft(
            np.fft.rfft(a_rev, fft_len) * np.fft.rfft(w, fft_len),
            fft_len,
        )[: 2 * n - 1]
        # derivative of log‑sum‑exp brings an extra factor α
        g = 2.0 * alpha * grad_full[n - 1 : n - 1 + n]

        # ---------- exponentiated‑gradient update ----------
        eta = eta0 / np.sqrt(it)
        exponents = -eta * g
        # clip to avoid overflow/underflow in the exponent
        exponents = np.clip(exponents, -50.0, 50.0)
        a = a * np.exp(exponents)

        # ---------- projection onto the simplex ----------
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = best_a.copy()          # safeguard

        # ---------- evaluate current objective ----------
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers with Metropolis acceptance.
    Incremental convolution updates keep the loop cheap.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    # temperature as a small fraction of the current score
    temp0 = max(best_score * 0.02, 1e-8)
    start_time = time.time()
    max_fraction = 0.25  # max fraction of source mass moved in one proposal
    recompute_every = 3000
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move from larger entry to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        # random proportion of src mass (up to max_fraction)
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the full auto‑convolution
        diff = np.zeros_like(conv)
        diff[src : src + n] -= a
        diff[dst : dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_score = 2.0 * n * conv_new.max()

        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            cur_score = new_score
            accepted += 1
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

        if accepted and (accepted % recompute_every) == 0:
            conv = np.convolve(a, a)   # refresh to avoid drift

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence (step heights) that minimises the
    provided evaluation function.  Returns the best list found before the
    deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 balances quality and speed.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise from a previous run if possible.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to new length
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split: 5 % param scan, 80 % smoothed gradient,
    # 15 % simulated annealing, 0.05 s safety margin.
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05
    phase0_dead = start_time + 0.05 * budget_s
    phase1_dead = start_time + 0.85 * budget_s

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    if time.time() < phase0_dead:
        cand_a, cand_score = param_scan(n, fft_len, phase0_dead, rng)
        if cand_score < best_score:
            best_score, best_a = cand_score, cand_a.copy()
            print(f"[param‑scan] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed exponentiated‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase1_dead:
        best_a, best_score = smooth_exp_gradient_optimize(
            best_a,
            phase1_dead,
            n,
            fft_len,
            rng,
            eta0=2.0,
            alpha_start=5.0,
            alpha_end=1500.0,
        )
        print(f"[smooth‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    # expose for the next run (optional)
    globals()["height_sequence_1"] = final_seq
    return final_seq
