    while time.time() < phase_lp_end:

def de_optimize(n, fft_len, rng, deadline):
    m = 60  # number of knots
    x_knots = np.linspace(0, 1, m)
    # objective function using closure (needs n, fft_len, x_knots)
    def eval_params(p):
        v = np.exp(p)  # positive
        v = np.sort(v)[::-1]  # enforce decreasing
        a = np.interp(np.linspace(0, 1, n), x_knots, v)
        s = a.sum()
        if s <= 0:
            return np.inf
        a = a / s
        # compute max conv using fft
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_conv = conv.max()
        # compute objective 2 * n * max_conv (since sum normalized)
        return float(2.0 * n * max_conv)
    # bounds for each p
    bounds = [(-8.0, 8.0)] * m
    result = scipy.optimize.differential_evolution(eval_params, bounds, maxiter=200, popsize=15, workers=1, polish=False, seed=rng.integers(2**32))
    # result.x is best param
    # construct best a
    v = np.exp(result.x)
    v = np.sort(v)[::-1]
    a = np.interp(np.linspace(0, 1, n), x_knots, v)
    a = a / a.sum()
    return a, result.fun

"""
A mixed‑strategy optimizer for the step‑function problem.

The algorithm consists of three blocks:

1. **Alternating linear‑programming** – starting from a decreasing
   exponential distribution we repeatedly
     * compute M = max(conv(f,f)),
     * solve the LP  maximise Σg   subject to   f∗g ≤ M   and g≥0,
     * replace f by the convex combination (1‑α)f + α·(g/Σg) (α=0.5).
   The LP is a very cheap way to drive down the worst
   convolution value.

2. **Smooth L‑BFGS / Adam refinement** –  the best vector from step 1 is
   used as a warm start for the log‑sum‑exp smoothed objective
   (the code from the previous solution).  A cascade of λ values is
   applied followed by Adam.

3. **Final polishing** – exponentiated‑gradient, projected sub‑gradient
   and a short simulated‑annealing hill‑climb are run while time permits.

All phases keep the vector on the probability simplex,
so the final score is simply `2·n·max(conv)`.  The routine returns the
best list of floats found before the 1000 s deadline.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse as sp

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'sparse'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = scipy.optimize.minimize(fun=objective_and_grad,
                                          x0=y,
                                          args=(n, fft_len, lam),
                                          method='L-BFGS-B',
                                          jac=True,
                                          options={'maxiter': maxiter_per_lam,
                                                   'gtol': 1e-12,
                                                   'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if steps % 200 == 0:
                    print(f"[Adam] lam={lam:.5f} step={steps} score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def compute_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def subgradient_descent(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        alpha0: float = 0.5):
    """
    Projected sub‑gradient descent on the true max‑convolution.
    Returns the best a and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        conv = compute_conv_fft(a, fft_len)
        k = int(np.argmax(conv))

        # sub‑gradient of conv[k] w.r.t. a
        g = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        if idx.size:
            g[idx] = a[k - idx]

        # diminishing step size
        alpha = alpha0 / np.sqrt(it + 1.0)

        a_new = a - alpha * g
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 0.0:
            a_new /= s
        else:
            a_new = best_a.copy()
            s = a_new.sum()

        score = fast_score(a_new, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1
    return best_a, best_score

def simulated_hill_climb(a_start: np.ndarray,
                         start_score: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Greedy hill‑climb that also accepts uphill moves according to a
    temperature schedule.  Uses O(n) incremental convolution updates.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)            # exact convolution (length 2n‑1)
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    # annealing schedule parameters
    T0 = max((best_score - 1.5) * 0.02, 1e-5)   # start temperature
    max_fraction_start = 0.25
    max_fraction_end = 0.02
    recompute_every = 1200
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        # linear cooling
        T = max(T0 * remaining_frac, 1e-12)
        max_fraction = max_fraction_end + (max_fraction_start - max_fraction_end) * remaining_frac

        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution (ignoring O(delta²) terms)
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            accepted += 1
            if accepted and (accepted % recompute_every) == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()

    return best_a, best_score

def build_toeplitz_matrix(a: np.ndarray, n: int) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix T such that (T @ g) = a ∗ g.
    Uses a sparse diagonal representation.
    """
    m = 2 * n - 1
    data = []
    offsets = []
    # each diagonal offset i contains the constant a[i]
    for i in range(n):
        length = m - i
        data.append(np.full(length, a[i]))
        offsets.append(i)
    return sp.diags(data, offsets, shape=(m, n), format='csc')


def alternating_lp(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   max_iters: int,
                   deadline: float) -> np.ndarray:
    """
    Alternating LP scheme described in the statement.
    Returns the best vector (on the simplex) discovered.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial bound
    conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) ** 2))[: 2 * n - 1]
    best_M = conv.max()
    best_score = 2.0 * n * best_M
    print(f"[LP] start M={best_M:.6e} score={best_score:.6f}")

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current bound
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # build constraint matrix T_a
        T = build_toeplitz_matrix(a, n)

        # linear program: maximise Σg  ⇔ minimise –Σg
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = scipy.optimize.linprog(c,
                                     A_ub=T,
                                     b_ub=b_ub,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True})
        if not res.success:
            print(f"[LP] iteration {it} – LP failed: {res.message}")
            break
        g = res.x
        s = g.sum()
        if s <= 0.0:
            break
        g_norm = g / s

        # convex combination with α=0.5
        alpha = 0.5
        a = (1.0 - alpha) * a + alpha * g_norm
        a = a / a.sum()   # safeguard

        # new bound
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        new_M = conv.max()
        new_score = 2.0 * n * new_M

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_M = new_M
            best_a = a.copy()
            print(f"[LP] iter {it} → score={best_score:.6f}")

        # stop early if improvement stalls
        if new_M >= M * (1 - 1e-9):
            break

    return best_a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between resolution and LP cost.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – use a decreasing exponential or fall back to a
    # stored construction if available.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # a gently decreasing exponential gives a low starting bound
        rate = rng.uniform(0.001, 0.015)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # Phase allocations (seconds)
    phase_lp_end   = start_time + 0.20 * budget_s   # LP stage
    phase_lbfgs_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase_adam_end = start_time + 0.65 * budget_s   # Adam
    phase_exp_end  = start_time + 0.80 * budget_s   # exp‑grad
    phase_sub_end  = start_time + 0.92 * budget_s   # sub‑grad
    # remainder → simulated annealing

    # ------------------------------------------------------------------
    # Phase 0 – alternating linear program (the new ingredient)
    # ------------------------------------------------------------------
    if time.time() < phase_lp_end:
        lp_iters = 7  # small number – each LP already costs a few seconds
        a_lp = alternating_lp(best_a, n, fft_len, lp_iters, phase_lp_end)
        score_lp = fast_score(a_lp, fft_len)
        if score_lp < best_score:
            best_score = score_lp
            best_a = a_lp
            print(f"[LP] best after LP phase: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS on the smoothed max (as before)
    # ------------------------------------------------------------------
    if time.time() < phase_lbfgs_end:
        lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                        0.05, 0.03, 0.018, 0.01, 0.006,
                        0.003, 0.0015, 0.0008, 0.0004,
                        0.0002, 0.0001, 0.00005]
        maxiter_per_lam = 800
        best_a, best_score = lbfgs_continuation(
            best_a, lam_schedule, maxiter_per_lam,
            phase_lbfgs_end, fft_len, n
        )
        print(f"[L‑BFGS] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase_adam_end:
        y0 = np.log(best_a + 1e-12)
        lam_schedule_adam = [0.4, 0.25, 0.15, 0.08, 0.04,
                             0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam,
            max_steps_per_lam=2000,
            deadline=phase_adam_end,
            n=n, fft_len=fft_len, lr0=0.13
        )
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < phase_exp_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase_exp_end, n, fft_len, rng, eta0=12.0
        )
        print(f"[exp‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent on the hard max
    # ------------------------------------------------------------------
    if time.time() < phase_sub_end:
        best_a, best_score = subgradient_descent(
            best_a, phase_sub_end, n, fft_len, rng,
            alpha0=0.5
        )
        print(f"[sub‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated‑annealing hill‑climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_score = 2.0 * n * np.max(np.convolve(best_a, best_a))
        best_a, best_score = simulated_hill_climb(
            best_a, final_score,
            deadline,
            n, fft_len,
            rng
        )
        print(f"[anneal] final score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
