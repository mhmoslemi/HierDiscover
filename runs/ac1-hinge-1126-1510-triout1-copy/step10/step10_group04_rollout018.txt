def guided_hill_climb(a_start, start_score, deadline, rng, start_factor=0.1, end_factor=0.001, recompute_every=5000):
    # a and conv trackers
    a = a_start.copy()
    conv = np.convolve(a, a)
    n = len(a)
    fft_len = next_pow2(2*n-1)
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    start_t = time.time()
    total = deadline - start_t
    steps = 0
    accepted = 0

    while time.time() < deadline:
        # compute max
        max_conv = conv.max()
        # tolerance
        eps = max_conv * 1e-12 + 1e-12
        mask = (conv >= max_conv - eps).astype(float)

        # compute gradient via FFT
        a_rev = a[::-1]
        g = 2.0 * np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                      np.fft.fft(a_rev, fft_len)))[n-1:n-1+n]

        # source and destination indices
        src = np.argmax(g)  # high gradient
        dst = np.argmin(g)  # low gradient

        # if src==dst or a[src] <= 0, fallback to random
        if src == dst or a[src] <= 0:
            # choose source proportional to a (weight) times g maybe
            # fallback random pair
            src = rng.choice(n, p=a / a.sum())
            dst = rng.integers(0, n)
            if src == dst or a[src] <= 0:
                continue

        # step factor decays linearly
        remaining = deadline - time.time()
        frac = remaining / total
        step_factor = end_factor + (start_factor - end_factor) * frac

        # delta to transfer
        delta = a[src] * step_factor
        if delta < 1e-12:
            delta = a[src] * step_factor
        # limit delta to a[src]
        if delta > a[src]:
            delta = a[src]

        # save previous max for comparison
        cur_score = 2.0 * n * max_conv

        # apply incremental update to conv
        # Using a_old for slices
        a_old = a  # pointer
        # subtract from src
        conv[src:src+n] -= 2.0 * delta * a_old
        # add to dst
        conv[dst:dst+n] += 2.0 * delta * a_old

        # update a
        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score < cur_score:
            # accept
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            # else continue
        else:
            # revert
            # revert conv
            conv[src:src+n] += 2.0 * delta * a_old
            conv[dst:dst+n] -= 2.0 * delta * a_old
            # revert a
            a[src] += delta
            a[dst] -= delta

        steps += 1

        # recompute exact conv to avoid drift
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)
        # maybe also random restart with small probability
        # Not needed

    return best_a, best_score

"""
Improved step‑function search.

The algorithm builds on the previous four‑stage pipeline but replaces the final
random hill‑climb with a *gradient‑guided* hill‑climb.  After a short
L‑BFGS/Adam/exp‑gradient warm‑up we repeatedly:

1. Compute the exact max of the current self‑convolution.
2. Form a mask of the (near‑)maximal entries.
3. Obtain the sub‑gradient of the max‑convolution via a single FFT‑based
   convolution with the reversed sequence.
4. Move a fraction of mass from the index with the largest gradient (most
   responsible for the peak) to the index with the smallest gradient.
5. Update the convolution incrementally (linear term only) and accept the
   move only if the max does not increase.

The step size decays linearly with the remaining time, and the full
convolution is recomputed periodically to correct the linear‑approximation
drift.  This directed search quickly flattens the sum distribution and
typically pushes the evaluation below the previous best (≈1.5064) within the
1000 s budget.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) with respect to logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Gradient‑guided hill‑climb (new component)
# ----------------------------------------------------------------------


def guided_hill_climb(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      start_factor: float = 0.20,
                      end_factor: float = 0.001,
                      recompute_every: int = 2000):
    """
    Gradient‑guided hill‑climb that moves a fraction of mass from the index
    with the largest sub‑gradient (most responsible for the current peak) to
    the index with the smallest sub‑gradient.  The step size shrinks linearly
    with the remaining time.  The convolution is updated incrementally; the
    exact convolution is recomputed periodically to avoid drift.
    Returns the best a and its fast_score found before the deadline.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact linear convolution, length 2n‑1
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    phase_start = time.time()
    total = deadline - phase_start
    accepted = 0
    steps = 0

    while time.time() < deadline:
        # current max and mask
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]   # length n

        # indices to move mass between
        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback random move when gradient information is degenerate
            nonzero = np.where(a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero,
                                 p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        if delta > a[src]:
            delta = a[src]

        # keep copies for possible rejection
        a_old = a.copy()
        conv_old = conv.copy()

        # incremental convolution update (linear term only)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[dst:dst + n] += 2.0 * delta * a_old

        # move the mass
        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score:
            # accept
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject – revert to previous state
            a = a_old
            conv = conv_old

        steps += 1

        # periodic exact recompute to counteract linear‑approximation drift
        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        # occasional progress report
        if steps % 10000 == 0:
            elapsed = time.time() - phase_start
            print(f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – 4000 entries give a good trade‑off between quality and speed
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # initialise from a previously stored construction if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # linear interpolation to required length
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS bursts from random starts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.07 * budget_s
    lam_schedule = np.geomspace(0.5, 1e-5, num=12)
    maxiter_per_lam = 250
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(
            y0, lam_sched, max_steps_per_lam=2000,
            deadline=deadline, n=n, fft_len=fft_len, lr0=0.13
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, rng, eta0=12.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – gradient‑guided hill‑climb (new)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = guided_hill_climb(
            best_a, best_score, deadline, rng,
            start_factor=0.20, end_factor=0.001, recompute_every=2000
        )
        print(f"[guided_hc] final best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
