def targeted_descent(a, max_iter, deadline):
    n=len(a)
    # compute conv
    conv = np.convolve(a, a)
    best_score = 2*n*conv.max()
    best_a = a.copy()
    t=0
    while time.time()<deadline and t<max_iter:
        # find index of max conv
        k_max = conv.argmax()
        # overlapping indices:
        low = max(0, k_max-(n-1))
        high = min(k_max, n-1)
        # contributions
        idx = np.arange(low, high+1)
        counterpart = k_max - idx
        p = a[idx] * a[counterpart]
        # find largest contribution
        i = idx[p.argmax()]
        # choose destination j different from i
        # choose j random, but prefer far from i and counterpart
        if rng.random()<0.7:
            # choose far
            # sample uniform from [0, n-1] but ensure |j - i| > n/4 and |j - counterpart| > n/4
            while True:
                j = rng.integers(0, n)
                if j != i and abs(j-i)>n//4 and abs(j-counterpart)>n//4:
                    break
        else:
            # random
            j = rng.integers(0, n)
            while j==i:
                j=rng.integers(0, n)
        # delta mass to move, up to e.g., 20% of a[i] or small fraction
        max_delta = a[i] * 0.2
        if max_delta<1e-12:
            t+=1
            continue
        delta = rng.random()*max_delta
        # compute incremental conv update
        # For each offset k, change in conv due to mass change at index i and j.
        # Define diff: subtract delta from a[i], add delta to a[j].
        # conv_new = conv + 2*delta*(conv_contrib(j) - conv_contrib(i)),
        # where conv_contrib(p)[k] = a[k-p] if that index within bounds.
        # We need to update using original a (before move).
        # We can compute diff array as:
        # diff_i = zeros length 2n-1; diff_i[i: i+n] -= a
        # diff_j = zeros length 2n-1; diff_j[j: j+n] += a
        # conv_new = conv + 2*delta*(diff_j - diff_i)
        # This is same approach as previous anneal.
        # Instead of recompute whole conv for each move, we do incremental.
        diff = np.zeros_like(conv)
        diff[i:i+n] -= a
        diff[j:j+n] += a
        conv_new = conv + 2*delta*diff
        new_max = conv_new.max()
        if new_max < conv.max():
            # accept
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            # optional: do normalization? sum should stay same.
            # No need.
            # update best
            best_score = 2*n*new_max
            best_a = a.copy()
        else:
            # reject
            pass
        t +=1
    return best_a, best_score

def local_search(a_init, deadline, n, fft_len, rng):
    a = a_init.copy()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_score = 2*n*conv.max()
    best_conv_max = conv.max()

    start_time = time.time()
    total_time = deadline - start_time

    step_fraction_initial = 0.1  # max fraction of src mass to move
    recompute_every = 2000
    it = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        # adapt step size
        step_fraction = step_fraction_initial * remaining_frac

        # choose src (high a)
        # choose src from top percentile of a values
        # We might sample src by probability proportional to a_i^2 (since big contributions)
        src = np.argmax(a)  # naive biggest
        # choose dst far
        dst = rng.integers(0, n)
        while dst == src or abs(dst - src) < n//4:
            dst = rng.integers(0, n)

        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * step_fraction
        if delta <= 0:
            continue

        diff = np.zeros_like(conv)
        diff[src:src+n] -= a
        diff[dst:dst+n] += a
        conv_new = conv + 2*delta*diff
        new_max = conv_new.max()
        # accept if better
        if new_max < conv.max() or rng.random() < np.exp(-(new_max - conv.max()) / (best_score*remaining_frac + 1e-12)):
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            if new_max < best_conv_max:
                best_conv_max = new_max
                best_a = a.copy()
                best_score = 2*n*best_conv_max
                # optional print
        # periodic recompute
        it += 1
        if it % recompute_every == 0:
            conv = np.convolve(a, a)
    return best_a, best_score

def targeted_local_search(a_start, deadline, n, fft_len, rng, max_moves=10**9):
    a = a_start.copy()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_max = conv.max()
    best_score = 2*n*best_max

    move_counter = 0
    recompute_interval = 5000
    # Determine initial temperature based on best_score
    T0 = max((best_score - 1.0) * 0.01, 1e-4)

    start_time = time.time()
    total_time = deadline - start_time
    while time.time() < deadline:
        # compute current max and offset
        k_max = conv.argmax()
        max_val = conv.max()

        # contributions to max
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx = np.arange(low, high + 1)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        src = idx[np.argmax(contrib)]

        # choose destination far from src and its partner
        prob = rng.random()
        if prob < 0.7:
            # pick far region
            while True:
                dst = rng.integers(0, n)
                if dst != src and abs(dst - src) > n // 4 and abs(dst - partner[np.argmax(contrib)]) > n // 4:
                    break
        else:
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)

        if a[src] <= 0:
            continue
        # choose delta relative to src mass
        max_delta = a[src] * 0.2
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        # incremental conv update
        diff = np.zeros_like(conv)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()

        # temperature schedule linear decay
        elapsed = time.time() - start_time
        remaining = max(0.0, deadline - time.time())
        T = T0 * remaining / total_time

        accept = False
        if new_max <= max_val:
            accept = True
        else:
            if T > 0:
                prob = np.exp(-(new_max - max_val) / (T + 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            # Apply move using original a for update
            conv = conv_new
            a[src] -= delta
            a[dst] += delta
            # Normalize? sum unchanged.
            # Update best if improved
            if new_max < best_max:
                best_max = new_max
                best_score = 2 * n * best_max
                best_a = a.copy()
                # print progress
                print(f"[local] new best score={best_score:.6f} iter={move_counter}")

        move_counter += 1

        if move_counter % recompute_interval == 0:
            conv = np.convolve(a, a)  # exact recompute

    return best_a, best_score

"""
Proposes a non‑negative step‑function sequence that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser uses several complementary phases within a 1000 s budget:

1. **L‑BFGS on a log‑sum‑exp smoothed max** – a few random restarts give a
   good initial shape on the probability simplex.

2. **Targeted local search** – repeatedly identifies the convolution entry
   that attains the current maximum, picks the pair of indices that
   contributes most to it and moves a small amount of mass from the “worst”
   index to a distant one.  The move is accepted if it improves the max
   (or with a temperature‑controlled probability).  An incremental
   update of the convolution avoids O(n²) recomputation; a full recompute
   is performed periodically to correct drift.

3. **Final simulated‑annealing hill‑climb** – a generic annealer refines the
   solution further and guards against any remaining local minima.

All steps keep the vector on the probability simplex, therefore the final
evaluation is simply `2·n·max(conv)`.  The algorithm returns the best list
of floats found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # soft‑max
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z
    # back‑prop through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]          # gradient wrt a before softmax

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len: int,
                       n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

def targeted_local_search(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          recompute_every: int = 4000):
    """
    Greedy descent that repeatedly attacks the convolution entry that attains
    the current maximum.  Mass is moved from the index that contributes most
    to this entry to a distant index.  The move is accepted if it lowers the
    maximum or with a temperature‑controlled probability.
    Returns the best a and its fast_score.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)                     # exact convolution (len = 2n‑1)
    best_a = a.copy()
    best_max = conv.max()
    best_score = 2.0 * n * best_max

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    # initialise a temperature proportional to the current gap
    T0 = max((best_score - 1.0) * 0.02, 1e-5)

    move_cnt = 0
    while time.time() < deadline:
        # locate the current bottleneck
        k_max = int(conv.argmax())
        max_val = conv.max()

        # indices that actually contribute to this entry
        low = max(0, k_max - (n - 1))
        high = min(k_max, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        # source index – the one that participates with the largest product
        src = idx[np.argmax(contrib)]
        src_partner = partner[np.argmax(contrib)]

        # destination: pick far away to avoid creating a new large overlap
        if rng.random() < 0.7:
            # enforce a minimal distance from both src and its partner
            while True:
                dst = rng.integers(0, n)
                if dst != src and abs(dst - src) > n // 4 and abs(dst - src_partner) > n // 4:
                    break
        else:
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)

        # ensure we move only a modest fraction of the source mass
        max_delta = a[src] * 0.15          # at most 15 % of the source weight
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        # ---- incremental convolution update (linearised) ----
        diff.fill(0.0)
        diff[src:src + n] -= a          # contribution loss of src
        diff[dst:dst + n] += a          # contribution gain of dst
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()

        # temperature linearly decreasing
        remaining = max(0.0, deadline - time.time())
        T = T0 * remaining / total_time

        accept = False
        if new_max <= max_val:
            accept = True
        else:
            # uphill acceptance probability
            prob = np.exp(-(new_max - max_val) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the change permanently
            conv = conv_new
            a[src] -= delta
            a[dst] += delta
            # (sum stays 1, non‑negativity guaranteed by construction)

            if new_max < best_max:
                best_max = new_max
                best_score = 2.0 * n * best_max
                best_a = a.copy()
                print(f"[local] new best score={best_score:.6f}  moves={move_cnt}")

        move_cnt += 1

        # occasional exact recomputation to avoid drift
        if move_cnt % recompute_every == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

# Simulated‑annealing hill‑climb (as in the reference implementation)

def simulated_hill_climb(a_start: np.ndarray,
                         start_score: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator):
    """
    Greedy hill‑climb that also accepts uphill moves according to a
    temperature schedule.  Uses O(n) incremental convolution updates.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)            # exact convolution (length 2n‑1)
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    # annealing schedule parameters
    T0 = max((best_score - 1.5) * 0.02, 1e-5)   # start temperature
    max_fraction_start = 0.25
    max_fraction_end = 0.02
    recompute_every = 1500
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        T = max(T0 * remaining_frac, 1e-12)
        max_fraction = max_fraction_end + (max_fraction_start - max_fraction_end) * remaining_frac

        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution (ignoring O(delta²) terms)
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            accepted += 1
            if accepted % recompute_every == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low evaluation
    score.  Returns a plain list of floats (the best found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size offers a nice trade‑off.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previous run if possible, otherwise uniform.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    phase1_end = start_time + 0.15 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # targeted local search
    phase3_end = start_time + 0.90 * budget_s   # final annealing

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with a handful of random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # 80 % of restarts are from a random pattern, 20 % are noisy variants of the current best
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.008)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – targeted local descent (the core improvement step)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = targeted_local_search(
            best_a, phase2_end, n, fft_len, rng,
            recompute_every=2500,
        )
        print(f"[local] best fast_score after local search={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_hill_climb(
            best_a, best_score, phase3_end,
            n, fft_len, rng,
        )
        print(f"[anneal] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
