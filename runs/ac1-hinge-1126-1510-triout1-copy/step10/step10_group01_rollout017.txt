def subgradient_optimize(a0):
    a = a0.copy()
    best_local = a.copy(); best_sc = fast_score(a)
    v = np.zeros_like(a)
    beta=0.9
    it=0

        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
        max_val = conv.max()
        # mask for entries close to max
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        # gradient: a_rev and mask conv
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1 : n-1+n]

        step = 0.5/(it+1)

        sc = fast_score(a_new)
        if sc < best_sc:
            best_sc = sc; best_local = a_new.copy()
        a = a_new
        it+=1
    return best_local, best_sc

"""
Search for a non‑negative step sequence that minimises

        2·n·max(a*a) / (∑ a)²

The algorithm consists of five stages

1️⃣  cheap parametric sweep over several families (double‑exp,
    power, edge‑blocks, binary‑sparse);
2️⃣  L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution;
3️⃣  projected sub‑gradient descent with momentum;
4️⃣  greedy pair‑wise mass transfer (always accept improvement);
5️⃣  final evaluation with the supplied ``evaluate_sequence``.

All stages are bounded by the global deadline (default 1000 s) and the
best sequence seen so far is returned.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len: int,
    lam: float,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # forward convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    # smoothing parameter
    lam_eff = max_c * lam + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # back‑propagation through convolution (a_rev * w)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean)

    return f, grad_y


# ----------------------------------------------------------------------
# Scoring utilities (fast, assumes ∑a = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) under the assumption ∑a = 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_optimize(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smoothed objective for a decreasing λ‑schedule."""
    y = np.log(a_start + 1e-12)  # logits, stays in interior of the simplex
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")
    return best_a, best_score


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent with simple momentum."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        # back‑propagation of the (sub‑)gradient of max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g

        step = 0.6 / (it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def greedy_pairwise_improve(
    a_start: np.ndarray,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Simple deterministic hill‑climb:
    repeatedly move a random fraction of mass from i→j and keep the move
    iff the fast_score improves.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    rng = np.random.default_rng(0)

    while time.time() < deadline:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or a[i] <= 0.0:
            continue
        # transfer up to 10 % of the source mass
        delta = a[i] * rng.random() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # projection is unnecessary because we keep the total sum unchanged
        # and never create negative entries (delta limited by a[i]).
        sc = fast_score(a_new, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            # print occasionally to monitor progress
            if int(time.time()) % 30 == 0:
                print(f"[greedy] improved fast_score={best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for cheap sweep
# ----------------------------------------------------------------------


def generate_double_exp(n, r, p, d):
    """a_i ∝ exp(-r * d_i^p) where d_i = min(i, n‑1‑i)."""
    w = np.exp(-r * (d ** p))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full(n, 1.0 / n)
    return w / w_sum


def generate_power(n, p, inc):
    i = np.arange(n, dtype=float)
    base = i + 1 if inc else n - i
    a = base ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_edge_block(n, e):
    """Edges of length e have equal weight, interior is zero."""
    if e == 0:
        return np.full(n, 1.0 / n)
    a = np.zeros(n, dtype=float)
    a[:e] = 1.0 / (2 * e)
    a[-e:] = 1.0 / (2 * e)
    return a


def generate_binary_sparse(n, prob, rng):
    mask = rng.random(n) < prob
    if not np.any(mask):
        mask[0] = True
    a = mask.astype(float)
    a /= a.sum()
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_frac: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Randomised sweep over several families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_frac

    # distance vector for double‑exp family
    idx = np.arange(n)
    d = np.minimum(idx, n - 1 - idx).astype(float)

    # 1️⃣ double‑exp (with an optional exponent p)
    p_vals = [1.0, 1.5, 2.0, 2.5]
    r_vals = np.logspace(-4, -1.7, 30)  # ≈ [1e‑4 … 0.02]
    betas = np.linspace(0.0, 1.0, 21)   # mixture with uniform

    for p in p_vals:
        for r in r_vals:
            w = generate_double_exp(n, r, p, d)          # w already normalised
            for beta in betas:
                a = (1.0 - beta) * (1.0 / n) + beta * w
                sc = fast_score(a, fft_len, n)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    print(f"[sweep‑exp] new best fast_score={best_score:.6f}")

                if time.time() >= deadline:
                    return best_a, best_score

    # 2️⃣ power‑law families (both increasing and decreasing)
    for inc in (True, False):
        for p in np.linspace(-3, 3, 25):
            a = generate_power(n, p, inc)
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[sweep‑pow] new best fast_score={best_score:.6f}")
            if time.time() >= deadline:
                return best_a, best_score

    # 3️⃣ edge‑block (vary length of the high‑edge region)
    for e in range(5, n // 2 + 1, 50):
        w = generate_edge_block(n, e)
        for beta in betas:
            a = (1.0 - beta) * (1.0 / n) + beta * w
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                print(f"[sweep‑edge] new best fast_score={best_score:.6f}")
            if time.time() >= deadline:
                return best_a, best_score

    # 4️⃣ binary sparse random masks
    for _ in range(200):
        prob = rng.uniform(0.02, 0.20)
        a = generate_binary_sparse(n, prob, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep‑sparse] new best fast_score={best_score:.6f}")
        if time.time() >= deadline:
            return best_a, best_score

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative floats that (approximately) minimise the
    evaluation function.  The method combines a parametric sweep,
    L‑BFGS on a smooth approximation, sub‑gradient descent,
    and a simple greedy pair‑wise improvement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem dimensions
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)  # for a*a

    # --------------------------------------------------------------
    # Initialise with uniform distribution
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # If a previous construction exists, try to interpolate it
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, old)
            interp = np.maximum(interp, 0.0)
            interp /= interp.sum()
            sc = fast_score(interp, fft_len, n)
            if sc < best_score:
                best_a, best_score = interp, sc
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time allocation
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin
    sweep_deadline = start_time + 0.10 * budget_s   # 10 % for cheap sweep

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep
    # --------------------------------------------------------------
    a_sweep, sc_sweep = parametric_sweep(
        n, fft_len, sweep_deadline - time.time(), rng
    )
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Remaining timeline split among the refinement phases
    # --------------------------------------------------------------
    t1_end = start_time + 0.45 * budget_s   # L‑BFGS
    t2_end = start_time + 0.70 * budget_s   # sub‑gradient
    t3_end = start_time + 0.90 * budget_s   # greedy pairwise
    # everything after t3_end is just final evaluation

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS with a decreasing λ‑schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.7,
        0.5,
        0.35,
        0.25,
        0.18,
        0.12,
        0.08,
        0.05,
        0.03,
        0.018,
        0.01,
        0.006,
        0.003,
        0.0015,
        0.0008,
        0.0005,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
        5e-06,
        2e-06,
        1e-06,
        5e-07,
        2e-07,
        1e-07,
    ]
    maxiter_per_lam = 1500

    while time.time() < t1_end:
        # occasional jitter of the current best
        if rng.random() < 0.3:
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_binary_sparse(n, rng.uniform(0.05, 0.20), rng)

        a_opt, sc_opt = lbfgs_optimize(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            t1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < t2_end:
        best_a, best_score = subgradient_optimize(
            best_a, t2_end, fft_len, n
        )
        print(f"[phase2] sub‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – greedy pairwise improvement (deterministic hill‑climb)
    # --------------------------------------------------------------
    if time.time() < t3_end:
        best_a, best_score = greedy_pairwise_improve(
            best_a, t3_end, fft_len, n
        )
        print(f"[phase3] greedy best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation with the official ``evaluate_sequence``
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
