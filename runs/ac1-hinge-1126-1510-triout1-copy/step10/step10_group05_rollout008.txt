def directed_local_search(a, conv, fft_len, deadline, rng):
    n = len(a)
    best_a = a.copy()
    best_max = conv.max()
    best_score = 2.0 * n * best_max

    # set temperature schedule parameters
    start_score = best_score
    T0 = max(start_score * 0.02, 1e-5)  # start temperature for max conv
    start_time = time.time()
    total_time = deadline - start_time

    iteration = 0
    while time.time() < deadline:
        # Compute gradient of active constraints
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_full[n-1:n-1+n]

        # choose src (largest gradient) where a > 0
        src_candidates = np.argsort(-g)  # descending
        src = None
        for idx in src_candidates:
            if a[idx] > 1e-15:
                src = idx
                break
        if src is None:
            break

        # choose dst (smallest gradient)
        dst = np.argmin(g)
        if dst == src:
            # pick second smallest
            dst = np.argsort(g)[1]

        # compute epsilon: fraction of source mass but not too small
        epsilon = min(a[src] * 0.05, 0.001)  # can tune
        if epsilon <= 1e-12:
            break

        # incremental conv update
        delta = np.zeros_like(conv)
        delta[src:src+n] -= a
        delta[dst:dst+n] += a
        conv_new = conv + 2.0*epsilon*delta
        new_max = conv_new.max()
        delta_score = (new_max - max_val) * 2.0 * n  # since score = 2*n*max

        # temperature schedule
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)
        T = max(T0 * remaining_frac, 1e-12)

        accept = False
        if delta_score <= 0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # commit move
            a[src] -= epsilon
            a[dst] += epsilon
            conv = conv_new
            max_val = new_max
            # update best
            if new_max < best_max:
                best_max = new_max
                best_a = a.copy()
                best_score = 2.0 * n * best_max
                # optional print
                if iteration % 200 == 0:
                    print(f"[Directed] iter={iteration} best_score={best_score:.6f}")

"""
Proposes a non‑negative step‑function sequence minimizing

    2·n·max(conv(a, a)) / (sum(a)**2)

The optimiser runs five phases inside the 1000 s budget:

1. **L‑BFGS on a log‑sum‑exp smoothed max‑convolution** – many random
   restarts, decreasing smoothing λ and generous iteration limits.
   Variables are logits of a soft‑max, guaranteeing the simplex.

2. **Adam** – continues on the same smoothed objective with a schedule
   of λ‑values and adaptive learning rates.

3. **Exponentiated‑gradient descent** – operates on the true (hard)
   max‑convolution, exploiting the exact sub‑gradient.

4. **Gradient‑directed local search** – at each step the active set
   (indices where the convolution attains its maximum) is identified,
   the sub‑gradient g_i = Σ_{k∈active} a_{k‑i} is computed, and a small
   amount of probability mass is moved from the index with the largest
   g_i to the index with the smallest g_i.  Moves that increase the
   maximum are occasionally accepted according to a linear cooling
   temperature schedule.  The convolution is updated incrementally,
   giving an O(n) per‑iteration cost.

5. **Simulated‑annealing hill‑climb** (fallback) – the same routine as
   in the reference implementation, starting from the best point
   produced by phase 4.

All phases keep the vector on the probability simplex, so the final
evaluation is simply `2·n·max(conv(a, a))`.  The best sequence found
before the deadline is returned as a plain Python list.
"""

import time
import numpy as np

# Basic utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    conv = full_conv(a, fft_len)
    return float(2.0 * a.shape[0] * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed objective and L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int,
                       lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    of the convolution.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                       # weight for each conv entry
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]      # gradient w.r.t. a before softmax

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)           # chain rule through soft‑max

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray, lam_schedule,
                       maxiter_per_lam: int, deadline: float,
                       fft_len: int, n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule,
    starting each stage from the previous result.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(y_start: np.ndarray, lam_schedule,
                  max_steps_per_lam: int, deadline: float,
                  n: int, fft_len: int, lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                if steps % 200 == 0:
                    print(f"[Adam] lam={lam:.5f} step={steps} score={best_score:.6f}")

            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray, deadline: float,
                          n: int, fft_len: int,
                          rng: np.random.Generator, eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)

        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def directed_local_search(a: np.ndarray, conv: np.ndarray, fft_len: int,
                          deadline: float, rng: np.random.Generator):
    """
    Uses the exact sub‑gradient of the active max‑convolution to move
    probability mass from the most harmful index to the least harmful
    one.  A simple linear‑cooling temperature schedule allows occasional
    uphill moves.  The convolution is updated incrementally, giving an
    O(n) per‑iteration cost.
    Returns the best a and the associated fast_score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_max = conv.max()
    best_score = 2.0 * n * best_max

    start_time = time.time()
    total_time = deadline - start_time
    T0 = max(best_score * 0.02, 1e-5)  # starting temperature

    it = 0
    while time.time() < deadline:
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-14
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the active constraints
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_full[n - 1: n - 1 + n]

        # source: index with largest gradient that still has mass
        src_order = np.argsort(-g)  # descending
        src = None
        for idx in src_order:
            if a[idx] > 1e-15:
                src = idx
                break
        if src is None:
            break

        # destination: index with smallest gradient
        dst = np.argmin(g)
        if dst == src:
            # pick the second smallest if they coincide
            dst = np.argsort(g)[1]

        # try a few step sizes (large to small)
        success = False
        for factor in (0.1, 0.05, 0.02, 0.01, 0.005):
            epsilon = min(a[src] * factor, 0.002)  # cap to avoid too big steps
            if epsilon <= 1e-15:
                continue

            # incremental update of the convolution
            delta = np.zeros_like(conv)
            delta[src:src + n] -= a
            delta[dst:dst + n] += a
            conv_new = conv + 2.0 * epsilon * delta
            new_max = conv_new.max()
            delta_score = (new_max - max_val) * 2.0 * n  # change in objective

            # temperature (linear cooling)
            remaining_frac = max(0.0, (deadline - time.time()) / total_time)
            T = max(T0 * remaining_frac, 1e-12)

            accept = False
            if delta_score <= 0.0:
                accept = True
            else:
                prob = np.exp(-delta_score / T)
                if rng.random() < prob:
                    accept = True

            if accept:
                # commit move
                a[src] -= epsilon
                a[dst] += epsilon
                conv = conv_new
                max_val = new_max
                success = True
                # track best solution
                if new_max < best_max:
                    best_max = new_max
                    best_a = a.copy()
                    best_score = 2.0 * n * best_max
                    if it % 200 == 0:
                        print(f"[Directed] iter={it} best_score={best_score:.6f}")
                break  # move accepted – go to next iteration

        if not success:
            # if no step size succeeded, try a random perturbation to escape
            if rng.random() < 0.01:
                rnd_src = rng.integers(0, n)
                rnd_dst = rng.integers(0, n)
                if a[rnd_src] > 1e-12 and rnd_src != rnd_dst:
                    eps_rand = min(a[rnd_src] * 0.02, 0.001)
                    delta = np.zeros_like(conv)
                    delta[rnd_src:rnd_src + n] -= a
                    delta[rnd_dst:rnd_dst + n] += a
                    conv = conv + 2.0 * eps_rand * delta
                    a[rnd_src] -= eps_rand
                    a[rnd_dst] += eps_rand
            # otherwise just continue – the loop will recompute a new gradient
            pass

        it += 1

    return best_a, best_score

def simulated_hill_climb(a_start: np.ndarray, start_score: float,
                         deadline: float, n: int, fft_len: int,
                         rng: np.random.Generator):
    """
    Greedy hill‑climb that also accepts uphill moves according to a
    temperature schedule.  Uses O(n) incremental convolution updates.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)          # exact convolution (length 2n‑1)
    curr_score = start_score
    best_a = a.copy()
    best_score = start_score

    start_time = time.time()
    total_time = deadline - start_time
    diff = np.empty_like(conv)

    # annealing schedule parameters
    T0 = max((best_score - 1.5) * 0.02, 1e-5)   # start temperature
    max_fraction_start = 0.25
    max_fraction_end = 0.02
    recompute_every = 1200
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)

        # linear cooling
        T = max(T0 * remaining_frac, 1e-12)
        max_fraction = max_fraction_end + (max_fraction_start - max_fraction_end) * remaining_frac

        src = rng.integers(0, n)
        dst = rng.integers(0, n)
        if src == dst:
            continue
        if a[src] < a[dst]:
            src, dst = dst, src
        if a[src] <= 0.0:
            continue
        delta = a[src] * rng.random() * max_fraction
        if delta <= 0.0:
            continue

        # incremental update of the convolution (ignoring O(delta²) terms)
        diff.fill(0.0)
        diff[src:src + n] -= a
        diff[dst:dst + n] += a
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max
        delta_score = new_score - curr_score

        accept = False
        if delta_score <= 0.0:
            accept = True
        else:
            prob = np.exp(-delta_score / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[src] -= delta
            a[dst] += delta
            conv = conv_new
            curr_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
            accepted += 1
            if accepted and (accepted % recompute_every) == 0:
                conv = np.convolve(a, a)
                curr_score = 2.0 * n * conv.max()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 4000 offers a very good trade‑off between quality
    # and runtime of the heavy phases.
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – reuse a previously discovered sequence if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Interpolate to the current length

            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Partition the budget
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # a small safety margin

    phase1_end = start_time + 0.18 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.44 * budget_s   # Adam
    phase3_end = start_time + 0.66 * budget_s   # Exp‑grad
    phase4_end = start_time + 0.85 * budget_s   # Directed local search
    # Remaining time → simulated annealing hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS with random restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.18, 0.12, 0.08,
                    0.05, 0.03, 0.018, 0.01, 0.006,
                    0.003, 0.0015, 0.0008, 0.0004,
                    0.0002, 0.0001, 0.00005]
    maxiter_per_lam = 1000

    while time.time() < phase1_end:
        # occasionally perturb the current best, otherwise draw a fresh start
        if rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.010)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)

        lam_schedule_adam = [0.4, 0.25, 0.15, 0.08, 0.04,
                             0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(
            y0, lam_schedule_adam,
            max_steps_per_lam=2500,
            deadline=phase2_end,
            n=n, fft_len=fft_len, lr0=0.13,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = exp_gradient_optimize(
            best_a, phase3_end, n, fft_len, rng, eta0=12.0,
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Gradient‑directed local search (new)
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        conv = full_conv(best_a, fft_len)          # current convolution
        best_a, best_score = directed_local_search(
            best_a, conv, fft_len, phase4_end, rng,
        )
        print(f"[Directed] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – simulated annealing hill‑climb (fallback)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_hill_climb(
            best_a, best_score,
            deadline, n, fft_len, rng,
        )
        print(f"[Anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
