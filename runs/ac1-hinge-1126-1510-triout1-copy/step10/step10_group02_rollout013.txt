def adam_exp_gradient_optimize(a_start, deadline, n, fft_len, rng, alpha0=0.2, beta1=0.9, beta2=0.999, epsilon=1e-8, max_iters=10_000_000):

    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        # compute gradient of max conv similarly
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n-1:n-1+n]

        # momentum updates
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

        # compute step size: learning rate decays with sqrt(t) maybe
        alpha = alpha0 / np.sqrt(t) # or keep constant

        # exponentiated update using corrected gradient: -alpha * m_hat / (sqrt(v_hat)+epsilon)
        update = -alpha * m_hat / (np.sqrt(v_hat) + epsilon)
        # exponentiated gradient
        exp_term = np.exp(np.clip(update, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score

   edge‑heavy, and a few classic windows.  The previous best solution
   (if present) is also interpolated to the current size.

2. **Multi‑restart L‑BFGS** – optimisation of a smooth log‑sum‑exp
   approximation of the max‑convolution.  A decreasing λ‑schedule
   (from 0.5 down to 1e‑6) is used, each run starting from a random pool
   element.  The best vector is kept.

3. **Adam‑style exponentiated‑gradient descent** – true sub‑gradient of the
   hard max‑convolution is used; updates are performed in the
   exponentiated‑gradient (mirror) space with Adam momentum.  This phase
   runs until the allocated time expires.

4. **Simulated‑annealing hill‑climb** – simple mass‑transfer moves with a
   linear cooling schedule.  A short final L‑BFGS polish is performed if
   time permits.

All vectors are kept on the probability simplex (∑a = 1, a_i ≥ 0) so the
returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for the official score.
    For a summing to 1 it returns 2·n·max(conv(a, a)).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "beta",
            "edge",
            "bimodal",
            "sparse",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "beta":
        alpha = rng.uniform(0.2, 2.5)
        beta = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) - 0.5) / n
        a = (x ** (alpha - 1)) * ((1.0 - x) ** (beta - 1))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise then re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (L‑BFGS)
# ----------------------------------------------------------------------


def smoothed_obj_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed obj = 2·n·smooth_max(conv(a, a))  with a = softmax(y).
    Returns value and gradient w.r.t. y.
    """
    # softmax ➜ a on simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i
    # back‑propagation through convolution
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]  # same shape as a

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns best vector and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smoothed_obj_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def adam_exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    alpha0: float = 0.25,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 10_000_000,
):
    """
    Mirror‑descent (exponentiated) with Adam momentum on the exact
    L∞‑objective (max convolution).  Returns best vector and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        # convolution and active mask for the maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_mask = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_mask).astype(np.float64)

        # gradient of (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]  # size n

        # Adam momentum update (in the log‑space of the simplex)
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        step = -alpha0 / np.sqrt(t) * m_hat / (np.sqrt(v_hat) + eps)

        # exponentiated‑gradient update
        exp_term = np.exp(np.clip(step, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (enhanced)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Simple mass‑transfer moves with a linear cooling schedule.
    Returns best vector and its fast_score.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-9)  # initial temperature

    max_frac = 0.30  # up to 30 % of source mass may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 points gives a good trade‑off
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # Previous construction (if any)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # Random families
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # Classic windows (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser

        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Timing splits
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    phase1_end = start_time + 0.20 * budget_s  # L‑BFGS
    phase2_end = start_time + 0.80 * budget_s  # Adam‑exp‑grad
    # remaining → annealing + final polish

    # --------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS on the smoothed surrogate
    # --------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.18,
        0.13,
        0.09,
        0.06,
        0.04,
        0.025,
        0.015,
        0.009,
        0.005,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        5e-05,
        2e-05,
        1e-05,
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float("inf")

    while time.time() < phase1_end:
        a_start = rng.choice(pool)
        if rng.random() < 0.35:
            a_start = add_noise(
                a_start,
                sigma=rng.uniform(0.0005, 0.008),
                rng=rng,
            )
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:  # fallback – uniform vector
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # --------------------------------------------------------------
    # Phase 2 – Adam‑style exponentiated‑gradient descent (sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = adam_exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            alpha0=0.3,
            max_iters=5_000_000,
        )
        print(f"[adam‑grad] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Optional final polish – short L‑BFGS with very small λ
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_a, final_score = lbfgs_cascade(
            best_a,
            lam_sched=[5e-06, 2e-06, 1e-06],
            maxiter_per_lam=800,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if final_score < best_score:
            best_a, best_score = final_a, final_score
            print(f"[final‑L‑BFGS] fast_score={best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
