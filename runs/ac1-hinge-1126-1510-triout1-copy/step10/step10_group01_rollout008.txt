def local_search(a_init, deadline, rng):
    a = a_init.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    # compute initial conv and max
    while time.time() < deadline:
        # compute conv
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        cur_max = conv.max()
        cur_score = 2 * n * cur_max
        # find shift indices achieving near max (within tolerance)
        eps = cur_max * 1e-12 + 1e-12
        # pick one
        k = np.argmax(conv)  # shift index where max occurs
        # find pairs i s.t. i and j = k - i within bound.
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = k - idx_i
        # contributions
        contrib = a[idx_i] * a[idx_j]
        # choose pair weighted by contribution
        total = contrib.sum()
        if total <= 0:
            # fallback random move
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            delta = a[i] * rng.random() * 0.5
            a[i] -= delta
            a[j] += delta
            a = np.clip(a, 0, None)  # ensure nonneg
            a /= a.sum()
            continue
        # sample according to contributions
        cum = np.cumsum(contrib)
        rnd = rng.random() * total
        idx = np.searchsorted(cum, rnd)
        i = idx_i[idx]
        j = idx_j[idx]
        # we have i and j pair; choose two distinct indices perhaps i != j
        # Choose a third index r not equal to i or j

"""
Propose a step‑function (non‑negative sequence) that (approximately) minimizes

        2·n·max(a*a) / (sum(a)**2)

using a combination of:
 * a cheap parametric sweep,
 * a long L‑BFGS continuation on a log‑sum‑exp smooth approximation of the max‑conv,
 * an Adam–style refinement on the same smooth objective,
 * a targeted stochastic hill‑climb that preferentially moves mass away from the
   convolution index that attains the current maximum.
The whole search respects a 1000 s deadline; the best sequence seen at any
moment is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast version of the evaluation (assumes sum(a)=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient with respect to logits
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    The smoothing parameter is `lam_factor * max(conv)`.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth_max w.r.t. conv entries ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through the convolution (a * a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # The relevant slice corresponds to the derivative w.r.t. a (length n)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in ``lam_schedule``.
    ``a_start`` must already be normalised (sum = 1).
    """
    y = np.log(a_start + 1e-12)          # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")
    return best_a, best_score


def adam_refine(
    y0: np.ndarray,
    lam_schedule,
    iters_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation on the smooth max objective.
    Returns the best vector (as probabilities) seen.
    """
    y = y0.copy()
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.05                     # base learning rate

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        m = np.zeros_like(y)
        v = np.zeros_like(y)
        for t in range(1, iters_per_lam + 1):
            if time.time() >= deadline:
                break
            f, g = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # simple learning‑rate schedule (decays with t)
            lr = lr0 / (np.sqrt(t) + 1.0)
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)

            if t % 200 == 0:
                a = softmax(y)
                sc = fast_score(a, fft_len_conv, n)
                if sc < best_score:
                    best_score, best_a = sc, a.copy()
                    print(f"[adam] new best fast_score={best_score:.6f}")

    # final check
    a = softmax(y)
    sc = fast_score(a, fft_len_conv, n)
    if sc < best_score:
        best_score, best_a = sc, a.copy()
        print(f"[adam] final best fast_score={best_score:.6f}")
    return best_a, best_score


def targeted_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that preferentially moves mass away from the
    convolution index that currently attains the maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # temperature schedule (starts relatively warm, cools to zero)
    T0 = max(0.02 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # recompute the full convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        k = int(np.argmax(conv))                     # shift where the max occurs

        # Build the list of pairs (i, j) with i + j = k
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = k - idx_i
        contrib = a[idx_i] * a[idx_j]
        total_contrib = contrib.sum()

        if total_contrib > 0.0:
            # sample a pair proportionally to its contribution
            r = rng.random() * total_contrib
            cum = np.cumsum(contrib)
            pair_idx = np.searchsorted(cum, r)
            i = int(idx_i[pair_idx])
            j = int(idx_j[pair_idx])
        else:
            # fallback: pick any two indices uniformly
            i = rng.integers(n)
            j = rng.integers(n)

        # choose a source index (the larger of the two) and a random destination
        src = i if a[i] > a[j] else j
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # transfer a modest fraction of the source mass
        delta = a[src] * rng.uniform(0.0, 0.15)
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # numerical safety
        a_new = np.clip(a_new, 0.0, None)

        # evaluate the new candidate (fast_score assumes sum = 1, which we preserve)
        new_score = fast_score(a_new, fft_len, n)

        # acceptance rule (greedy + occasional SA)
        if new_score < cur_score:
            accept = True
        else:
            # simple simulated‑annealing acceptance
            elapsed = time.time() - t_start
            remain = deadline - time.time()
            T = T0 * np.sqrt(remain / total_time)
            accept = rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12))

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Parametric families for the cheap sweep
# ----------------------------------------------------------------------


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a normalised non‑negative vector from a simple parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-3.0, 3.0)
        inc = rng.choice([True, False])
        a = i + 1 if inc else n - i
        a = a ** p
    elif family == "exp":
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        mu = (n - 1) / 2.0
        a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        mu1 = rng.uniform(0.0, n / 2.0)
        mu2 = rng.uniform(n / 2.0, n - 1)
        sigma1 = rng.uniform(100.0, 500.0)
        sigma2 = rng.uniform(100.0, 500.0)
        a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
        a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
        a = a1 + a2
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0005, 0.015)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    elif family == "sinusoid":
        freq = rng.integers(1, 10)
        a = np.maximum(np.sin(2 * np.pi * freq * i / n), 0.0) + 0.01
    else:                                   # fallback: pure random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Cheap random sweep over a collection of families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10          # 10 % of allocated budget

    families = [
        "power",
        "exp",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "sinusoid",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")
    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises `evaluate_sequence`.
    The function respects a global budget (seconds) and returns the best
    sequence discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)       # for a*a
    fft_len_g = next_pow2(3 * n - 2)          # for back‑propagation in the smooth max

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Re‑use a previously saved construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (slightly larger budget)
    # --------------------------------------------------------------
    sweep_budget = budget_s * 0.12
    a_sweep, sc_sweep = parametric_sweep(n, fft_len_conv, sweep_budget, rng)
    if sc_sweep < best_score:
        best_a, best_score = a_sweep, sc_sweep
        print(f"[phase0] sweep best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # small safety margin

    # Allocate time slices for each phase
    phase1_end = start_time + 0.50 * budget_s      # L‑BFGS
    phase2_end = start_time + 0.75 * budget_s      # Adam refinement
    phase3_end = deadline                          # targeted annealing

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS continuation with a finer λ‑schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 1e-06,
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # occasional jitter of the current best to diversify restarts
        if rng.random() < 0.30:
            a_start = best_a * rng.normal(1.0, 0.02, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_family("random", n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam‑style refinement on the smooth max
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        y0 = np.log(best_a + 1e-12)
        best_a, best_score = adam_refine(
            y0,
            lam_schedule,
            iters_per_lam=2000,
            deadline=phase2_end,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
            rng=rng,
        )
        print(f"[phase2] adam best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – targeted simulated annealing / hill‑climb
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = targeted_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the official `evaluate_sequence`
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
