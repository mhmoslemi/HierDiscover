import time, numpy as np

def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def fast_score(a, fft_len):
    conv = np.real(np.fft.ifft(np.fft.fft(a, fft_len) * np.fft.fft(a, fft_len)))[:2*a.shape[0]-1]
    return float(2.0 * a.shape[0] * conv.max())

def arcsine_distribution(n):
    i = np.arange(n, dtype=np.float64)
    x0 = i / n
    x1 = (i+1) / n
    # clip for safety
    s0 = np.sqrt(np.clip(x0, 0.0, 1.0))
    s1 = np.sqrt(np.clip(x1, 0.0, 1.0))

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0/n)
    else:
        a /= a_sum
    return a

=======================================

The evaluation function is scale‑invariant:
    2·n·max(conv(a,a)) / (sum(a)**2)

    a_i = F((i+1)/n) – F(i/n) ,   F(x)= (2/π)·arcsin(√x) ,

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

    """
    Fast evaluation of 2·n·max(conv(a,a)) for a vector a that already
    satisfies sum(a)=1.  Uses FFT for O(n log n) time.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * a.shape[0] - 1]
    return float(2.0 * a.shape[0] * conv.max())

    """
    Discretise the arcsine (Beta(½,½)) distribution on n bins.
    a_i = F((i+1)/n) – F(i/n) where F(x) = (2/π)·arcsin(√x).
    The result is a probability vector (sum == 1, all entries ≥0).

    x0 = i / n
    x1 = (i + 1) / n

    s1 = np.sqrt(np.clip(x1, 0.0, 1.0))

    total = a.sum()
    if total == 0.0:                       # fallback – should never happen
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a
