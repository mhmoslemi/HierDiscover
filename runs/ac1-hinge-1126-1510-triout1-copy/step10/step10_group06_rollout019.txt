    Temperature follows a square‑root cooling schedule; uphill moves are
    accepted with probability exp(−Δ/T).  The annealer also tries occasional
    “swap” moves (exchanging masses of two pairs) to escape plateaus.

All stages respect a global deadline; the best vector seen so far is
returned and finally evaluated with the official ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """2·n·max(a*a) for a vector that already satisfies ∑a = 1."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Parametric families
# --------------------------------------------------------------

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_triangular(n: int, width: float, centre: float) -> np.ndarray:
    """Symmetric triangular bump (height 1 at centre, linear to 0 at centre±width)."""
    i = np.arange(n, dtype=float)
    a = np.maximum(0.0, 1.0 - np.abs(i - centre) / width)
    a /= a.sum()
    return a


def generate_tent(n: int, left: int, right: int) -> np.ndarray:
    """Linear rise from 0 to the peak at (left+right)//2 and back to 0."""
    i = np.arange(n, dtype=float)
    centre = (left + right) // 2
    slope = 2.0 / (right - left) if right != left else 0.0
    a = np.where(i < left, 0.0,
                 np.where(i > right, 0.0,
                          np.where(i <= centre,
                                   slope * (i - left),
                                   slope * (right - i))))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def parametric_grid_search(n: int, fft_len: int, rng: np.random.Generator,
                           deadline: float) -> tuple[np.ndarray, float]:
    """Explore a handful of analytic families; keep the best (fast_score)."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential
    for rate in np.linspace(0.00015, 0.0012, 55):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → {best_sc:.6f}")

    # power‑law (negative exponents)
    for p in np.linspace(-6.0, -0.3, 23):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → {best_sc:.6f}")

    # logistic
    mus = np.linspace(0.2 * n, 0.8 * n, 9)
    sigmas = np.linspace(150.0, 1300.0, 9)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → {best_sc:.6f}")

    # triangular
    widths = np.linspace(200.0, 2000.0, 11)
    centres = np.linspace(0.2 * n, 0.8 * n, 9)
    for w in widths:
        for c in centres:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_triangular(n, w, c)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] triangular w={w:.0f} c={c:.0f} → {best_sc:.6f}")

    # tent (piecewise linear)
    for left in range(0, n, 400):
        for right in range(left + 400, n, 400):
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_tent(n, left, right)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] tent [{left},{right}] → {best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(n: int, fft_len: int, budget_s: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very cheap random exploration of a few handcrafted families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + 0.10 * budget_s  # 10 % of total budget

    families = ["exp", "power", "logistic", "triangular", "tent", "gauss", "bimodal"]

    while time.time() < deadline:
        fam = rng.choice(families)
        if fam == "exp":
            a = generate_exp(n, rng.uniform(0.0001, 0.0015))
        elif fam == "power":
            p = rng.uniform(-6.0, -0.2)
            inc = rng.choice([True, False])
            a = generate_power(n, p, inc=inc)
        elif fam == "logistic":
            mu = rng.uniform(0.0, n - 1)
            sigma = rng.uniform(150.0, 1500.0)
            a = generate_logistic(n, mu, sigma)
        elif fam == "triangular":
            w = rng.uniform(100.0, 2500.0)
            c = rng.uniform(0.2 * n, 0.8 * n)
            a = generate_triangular(n, w, c)
        elif fam == "tent":
            left = rng.integers(0, n - 400)
            right = rng.integers(left + 400, n)
            a = generate_tent(n, left, right)
        elif fam == "gauss":
            sigma = rng.uniform(150.0, 2500.0)
            mu = (n - 1) / 2.0
            i = np.arange(n, dtype=float)
            a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
            a = np.maximum(a, 0.0)
            a /= a.sum()
        elif fam == "bimodal":
            mu1 = rng.uniform(0.0, n / 2.0)
            mu2 = rng.uniform(n / 2.0, n - 1)
            sigma1 = rng.uniform(100.0, 500.0)
            sigma2 = rng.uniform(100.0, 500.0)
            i = np.arange(n, dtype=float)
            a = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2) + \
                np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            a = rng.random(n)

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} ({fam})")

    return best_a, best_score


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# --------------------------------------------------------------

def smoothed_objective_and_grad(y: np.ndarray, n: int,
                                fft_len_conv: int, fft_len_g: int,
                                lam: float) -> tuple[float, np.ndarray]:
    """
    Compute a smoothed version of ``max(conv)`` and its gradient w.r.t. the
    logits y (soft‑max parametrisation).  The smoothing parameter λ>0.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # log‑sum‑exp regularisation
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. each convolution entry
    w = exp_shift / Z                     # d smooth_max / d conv_k

    # back‑propagation through convolution (linear, so we can reuse FFT)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # gradient w.r.t. a (size n) – the relevant slice:
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# --------------------------------------------------------------
# Optimisation primitives
# --------------------------------------------------------------

def adam_continuation(a_start: np.ndarray,
                     lam_schedule,
                     maxiter_per_lam: int,
                     deadline: float,
                     fft_len_conv: int,
                     fft_len_g: int,
                     n: int,
                     lr: float = 6e-2) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective.  For each λ we perform a
    fixed number of iterations (or stop earlier if the deadline is hit).
    The result of the previous λ is used as the start point for the next.
    """
    # initialise logits
    y = np.log(a_start + 1e-16)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    t_iter = 1
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        for it in range(maxiter_per_lam):
            if time.time() >= deadline:
                break
            f, grad = smoothed_objective_and_grad(y, n,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  lam)
            # Adam updates on the logits
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            y -= step
            t_iter += 1

            a = softmax(y)
            sc = fast_score(a, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, a.copy()
                # verbose output only for strict improvements
                print(f"[adam] λ={lam:.2e} iter={it} → best={best_score:.6f}")

    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the *true* max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.85
    it = 0

    while time.time() < deadline:
        # true convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)   # true sub‑gradient

        # back‑propagation of the sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum step
        v = beta * v + (1.0 - beta) * g
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Square‑root cooling SA that occasionally performs a four‑point swap
    (exchange mass between two pairs) to escape local plateaus.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # cooling
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        # decide move type (mass transfer vs. swap)
        if rng.random() < 0.15:
            # 4‑point swap: pick i<j<k<l and exchange a[i]↔a[k], a[j]↔a[l]
            idx = rng.choice(n, size=4, replace=False)
            idx.sort()
            i, j, k, l = idx
            a_new = a.copy()
            a_new[i], a_new[k] = a_new[k], a_new[i]
            a_new[j], a_new[l] = a_new[l], a_new[j]
        else:
            # simple mass transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.30
            if delta <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Fixed problem size (used throughout the search)
    # --------------------------------------------------------------
    n = 4000                                     # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)          # for a*a
    fft_len_g = next_pow2(3 * n - 2)             # for back‑propagation

    # --------------------------------------------------------------
    # Initial uniform candidate
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Warm‑start from a previous solution if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Global timing
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    # --------------------------------------------------------------
    # Phase 0 – systematic grid + cheap random sweep
    # --------------------------------------------------------------
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] new best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Allocate time for the remaining phases
    # --------------------------------------------------------------
    phase1_end = start_time + 0.30 * budget_s   # Adam / smoothed L‑BFGS
    phase2_end = start_time + 0.70 * budget_s   # Sub‑gradient
    phase3_end = deadline                       # Simulated annealing

    # --------------------------------------------------------------
    # Phase 1 – Adam on smoothed max (progressively smaller λ)
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e‑05,
        2e‑05, 1e‑05, 5e‑06, 2e‑06,
    ]
    maxiter_per_lam = 300               # a few hundred inner steps per λ

    while time.time() < phase1_end:
        # occasional jitter to escape plateaus
        if rng.random() < 0.2:
            a_start = best_a * rng.normal(1.0, 0.03, size=n)
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            # fresh exponential start (good baseline for Adam)
            rate = rng.uniform(0.0002, 0.0009)
            a_start = generate_exp(n, rate)

        a_opt, sc_opt = adam_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent with momentum
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] sub‑gradient best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing (long final refinement)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
