"""
propose_candidate
-----------------

A fast mixed‑strategy optimiser for the step‑function inequality.

The algorithm proceeds in three phases inside the 1000 s budget:

1. **Parametric exploration** – we generate a large number of simple
   families (exponential, power‑law, two‑segment piecewise‑constant,
   linear) with random parameters, normalise them to a probability vector
   and keep the best according to the fast FFT‑based score.

2. **Exponentiated‑gradient refinement** – starting from the current best
   (and from a few random restarts) we perform mirror‑descent on the
   *true* non‑smoothed objective.  The sub‑gradient of the max‑convolution
   is obtained analytically via the FFT, the step size decays as
   η₀/√t and the iterate stays on the simplex by an element‑wise
   exponential update.

3. **Fine‑grained hill‑climb** – a greedy move that transfers a tiny
   fraction of mass from a heavy index to a lighter one, accepting only
   improvements.  Convolution updates are performed incrementally to keep
   the cost low.

All stages keep the vector non‑negative, automatically satisfy the
security checks in ``evaluate_sequence`` and finally return the best
list of floats found before the deadline.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize  # only for optional L‑BFGS (not used here)


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation: 2·n·max(conv(a,a)) (sum‑norm is 1)."""
    n = a.shape[0]
    conv = full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a random probability vector.
    Several modes (uniform, exponential, power‑law, bimodal, sparse)
    are mixed to obtain a diverse set of starts.
    """
    mode = rng.choice(
        ['uniform', 'exp', 'power', 'bimodal', 'sparse', 'linear', 'piece2']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif mode == 'power':
        p = rng.uniform(0.3, 3.0)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'bimodal':
        # two exponential peaks at random locations
        centre1 = rng.integers(0, n // 2)
        centre2 = rng.integers(n // 2, n)
        width = rng.integers(5, 30)
        a = np.zeros(n, dtype=float)
        a[max(0, centre1 - width):min(n, centre1 + width)] = np.exp(
            -0.2 * np.abs(np.arange(-width, width))
        )
        a[max(0, centre2 - width):min(n, centre2 + width)] = np.exp(
            -0.2 * np.abs(np.arange(-width, width))
        )
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.08 * n))
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=float)
        a[idx] = rng.random(m) + 1e-6
    elif mode == 'linear':
        # decreasing linear from a_max to a_min
        a_max = rng.uniform(0.1, 0.5)
        a_min = rng.uniform(0.0, a_max / 10)
        a = np.linspace(a_max, a_min, n)
    else:  # piecewise constant with two levels
        frac = rng.uniform(0.05, 0.95)          # break position
        mass_first = rng.uniform(0.1, 0.9)       # proportion of total mass in first part
        n1 = int(frac * n)
        n2 = n - n1
        if n1 == 0 or n2 == 0:
            a = np.full(n, 1.0 / n)
        else:
            v1 = mass_first / n1
            v2 = (1.0 - mass_first) / n2
            a = np.concatenate([np.full(n1, v1), np.full(n2, v2)])
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 15.0,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution.
    Returns the best vector encountered and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of (max conv) w.r.t. a:
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]   # shape (n,)

        # step‑size schedule
        eta = eta0 / math.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        # keep best
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with incremental convolution update
# ----------------------------------------------------------------------


def hill_climb_incremental(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improvements.
    The convolution is updated incrementally for speed.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        # fraction shrinks from 0.20 → 0.02 as the deadline approaches
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= best
        diff[dst : dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        # occasionally recompute exact convolution to avoid drift
        if accepted and accepted % 3000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises

        2·n·max(conv(a,a)) / (sum(a)**2)

    The search combines cheap parametric sampling, a long
    exponentiated‑gradient descent on the exact objective and a final
    greedy hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – 4000 entries offers a good trade‑off.
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise with a uniform vector (score = 2) and keep the best.
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget handling
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    # --------------------------------------------------------------
    # Stage 1 – parametric random search (≈10 % of the budget)
    # --------------------------------------------------------------
    param_end = start_time + 0.10 * budget_s
    families = ['exp', 'power', 'piece2', 'linear']
    while time.time() < param_end:
        # pick a family uniformly
        fam = rng.choice(families)
        if fam == 'exp':
            rate = rng.uniform(0.0002, 0.015)
            a = np.exp(-rate * np.arange(n, dtype=float))
        elif fam == 'power':
            p = rng.uniform(0.4, 2.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
        elif fam == 'piece2':
            # two‑segment piecewise‑constant
            frac = rng.uniform(0.05, 0.95)          # break position
            mass_first = rng.uniform(0.05, 0.95)    # total mass in first part
            n1 = max(1, int(frac * n))
            n2 = n - n1
            v1 = mass_first / n1
            v2 = (1.0 - mass_first) / n2
            a = np.concatenate([np.full(n1, v1), np.full(n2, v2)])
        else:  # linear decreasing
            a_max = rng.uniform(0.1, 0.5)
            a_min = rng.uniform(0.0, a_max / 8)
            a = np.linspace(a_max, a_min, n)

        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            continue
        a /= a.sum()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[param] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient descent on the true max‑conv.
    # --------------------------------------------------------------
    # We keep the current best and also launch a few short random
    # restarts to escape possible local minima.
    exp_end = start_time + 0.70 * budget_s
    while time.time() < exp_end:
        # decide whether to start from the global best or a fresh random start
        if rng.random() < 0.6:
            a_start = best_a.copy()
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = exp_gradient_optimize(
            a_start, exp_end, n, fft_len, rng, eta0=18.0
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[exp‑grad] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – final fine‑grained hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[hill‑climb] start")
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation (outside the optimisation loop)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
