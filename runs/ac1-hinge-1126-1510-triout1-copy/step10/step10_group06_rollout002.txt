"""
Search for a low‑score non‑negative step function.

The score is   2·n·max(conv(a,a)) / (sum(a)^2) .
Since the score is invariant under scaling we keep the vector on the simplex
(sum = 1).  The goal is to flatten the self‑convolution (ordered sum of two
independent draws) as much as possible.

Algorithm
---------

1. **Parametric grid** – systematic sweep over several families
   (exponential, power, logistic, edge‑heavy power, discrete beta).
   The best vector is kept.

2. **Random sweep** – a cheap random sampling of the same families.

3. **Smoothed L‑BFGS** – minimise a log‑sum‑exp surrogate of the max
   using many values of the smoothing factor λ.  Each run starts from a
   jittered copy of the current best vector.

4. **Exponentiated‑gradient (mirror descent)** – true sub‑gradient of the
   max‑convolution is computed (via FFT) and a KL‑proximal step
   `a = softmax(log(a) – η·g)` is performed.  Momentum is not used
   because the KL‑projection already provides stability.

5. **Simulated annealing** – a stochastic hill‑climb that moves a
   random fraction of mass between two positions, accepting uphill moves
   with a square‑root cooling schedule.

All phases respect a global deadline (default 1000 s).  The best
sequence seen at any time is returned and finally evaluated with the
provided `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """2·n·max(a*a) assuming sum(a)=1."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a /= a.sum()
    return a


def generate_edge_power(n: int, alpha: float) -> np.ndarray:
    """Edges are heavier: a_i ∝ ((i+1)*(n-i))^{-α}."""
    i = np.arange(n, dtype=float)
    denom = (i + 1) * (n - i)
    a = np.where(denom > 0, denom ** (-alpha), 0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def generate_beta_shape(n: int, p: float, q: float) -> np.ndarray:
    """Discrete Beta density: a_i ∝ (i+1)^{p‑1} (n‑i)^{q‑1}."""
    i = np.arange(n, dtype=float) + 1.0
    a = (i ** (p - 1.0)) * ((n - i + 1.0) ** (q - 1.0))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Sample a random vector from the given family."""
    if family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        return generate_exp(n, rate)
    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        return generate_power(n, p, inc)
    if family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        return generate_logistic(n, mu, sigma)
    if family == "edge_power":
        alpha = rng.uniform(0.05, 0.6)
        return generate_edge_power(n, alpha)
    if family == "beta":
        p = rng.uniform(0.2, 1.0)
        q = rng.uniform(0.2, 1.0)
        return generate_beta_shape(n, p, q)
    # fallback: pure random
    a = rng.random(n)
    a /= a.sum()
    return a

def parametric_grid_search(n: int, fft_len: int, rng: np.random.Generator,
                           deadline: float) -> tuple[np.ndarray, float]:
    """Systematic search over a modest grid of families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → {best_sc:.6f}")

    # power law (decreasing)
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → {best_sc:.6f}")

    # logistic
    mus = np.linspace(0.2 * n, 0.8 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → {best_sc:.6f}")

    # edge‑heavy power
    for alpha in np.linspace(0.05, 0.55, 11):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_edge_power(n, alpha)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] edge_power α={alpha:.2f} → {best_sc:.6f}")

    # beta shape
    for p in np.linspace(0.2, 0.9, 8):
        for q in np.linspace(0.2, 0.9, 8):
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_beta_shape(n, p, q)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] beta p={p:.2f} q={q:.2f} → {best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(n: int, fft_len: int, deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very fast random exploration of all families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    families = ["exp", "power", "logistic", "edge_power", "beta", "random"]
    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[rnd] new best {best_sc:.6f} from {fam}")

    return best_a, best_sc


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) objective for L‑BFGS
# --------------------------------------------------------------

def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30) -> tuple[float, np.ndarray]:
    """Smoothed max of a*a where a = softmax(y)."""
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n * smooth_max

    # gradient w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution (reverse a)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """Run L‑BFGS for each λ in lam_schedule, restarting each stage."""
    y = np.log(a_start + 1e-12)          # logits – stays on the simplex
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method="L-BFGS-B",
                           jac=True,
                           options={"maxiter": maxiter_per_lam,
                                    "gtol": 1e-12,
                                    "disp": False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


# --------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) on the true max
# --------------------------------------------------------------

def exp_grad_optimize(a0: np.ndarray,
                      deadline: float,
                      fft_len_conv: int,
                      fft_len_g: int,
                      n: int) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using KL‑divergence.
    Sub‑gradient is taken w.r.t. the true max‑convolution.
    """
    a = a0.copy()
    log_a = np.log(a + 1e-12)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # compute convolution and the mask of maximal entries
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the masked sum
        fm = np.fft.fft(mask, fft_len_g)
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # learning rate schedule (decreasing)
        eta = 0.5 / np.sqrt(it + 1.0)

        # KL‑proximal update
        log_a = log_a - eta * g
        a = softmax(log_a)

        # keep best
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score, best_a = cur_score, a.copy()
        it += 1

    return best_a, best_score


# --------------------------------------------------------------
# Simulated annealing
# --------------------------------------------------------------

def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Square‑root cooling stochastic hill‑climb."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()

    return best_a, best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats discovered before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem parameters
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Warm‑start from a previously saved construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_score:
                best_a, best_score = a1, sc1
                print(f"[init] imported prev fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – systematic grid + cheap random sweep
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2          # safety margin

    # allocate 10 % of the budget to grid + random
    phase0_end = start_time + 0.10 * budget_s

    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, phase0_end)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    # cheap random sweep (another 5 % of total time)
    rnd_end = start_time + 0.15 * budget_s
    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, rnd_end, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time allocation for the remaining phases
    # --------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.68 * budget_s   # Exponentiated‑gradient
    phase3_end = start_time + 0.95 * budget_s   # Simulated annealing
    # remaining time → final evaluation

    # --------------------------------------------------------------
    # Phase 1 – L‑BFGS on smoothed max (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06,
    ]
    maxiter_per_lam = 2000

    while time.time() < phase1_end:
        # occasional jitter to diversify basins
        if rng.random() < 0.25:
            jitter = rng.normal(1.0, 0.02, size=n)
            a_start = best_a * jitter
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            rate = rng.uniform(0.0003, 0.0009)
            a_start = generate_exp(n, rate)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient (KL‑proximal) descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_grad_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] exp‑grad best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing (final stochastic refinement)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
