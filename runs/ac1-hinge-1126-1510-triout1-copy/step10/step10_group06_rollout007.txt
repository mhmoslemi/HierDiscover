def greedy_local_search(a, deadline, rng):
    best_a = a.copy()
    best_score = fast_score(best_a)
    while time.time() < deadline:
        # compute conv and active set
        conv = full_conv(a)
        max_val = conv.max()
        # identify active shift indices near max (within tolerance)
        eps = max_val * 1e-12 + 1e-15
        active_shifts = np.where(conv >= max_val - eps)[0]
        # pick a random shift from active_shifts
        k = rng.choice(active_shifts)
        # compute contributions for this shift:
        i_start = max(0, k - (len(a) - 1))
        i_end = min(len(a) - 1, k)
        i_range = np.arange(i_start, i_end + 1)
        j_range = k - i_range
        prod = a[i_range] * a[j_range]
        # pick index with largest product
        idx = np.argmax(prod)
        i = i_range[idx]
        j = j_range[idx]
        # Move fraction from whichever index has larger a
        source = i if a[i] >= a[j] else j
        target = rng.integers(len(a))
        if target == source:
            continue
        # Transfer a fraction of source mass
        delta = a[source] * rng.random() * 0.2
        a_new = a.copy()
        a_new[source] -= delta
        a_new[target] += delta
        new_score = fast_score(a_new)
        if new_score < best_score or rng.random() < np.exp((best_score - new_score) / (0.01 * best_score)):
            a = a_new
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
    return best_a, best_score
```

"""
Improved step‑function search.

The algorithm (time‑controlled) consists of:

0. Load a previous construction if available.
1. Systematic grid search over several parametric families
   (exponential, power, logistic, double‑exponential) – extremely cheap.
2. Projected sub‑gradient descent with momentum on the true max‑convolution.
   Runs for the bulk of the time budget.
3. Targeted hill‑climbing: repeatedly pick the shift where the
   convolution attains its maximum and move a small fraction of mass
   from the “worst” contributing index to a location that does not
   participate in that shift.  This fine‑tunes the peak value.
4. A short simulated‑annealing phase to escape any remaining local
   minima.
5. Return the best sequence found (as a plain list of floats) and its
   official evaluation score.

All phases respect a global deadline (default 1000 s).  The best
sequence seen at any moment is returned; the final call to
`evaluate_sequence` gives the official score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# --------------------------------------------------------------
# Basic utilities
# --------------------------------------------------------------

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a*a) for a vector with ∑a = 1.  Fast surrogate for the real score."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Smoothed max (log‑sum‑exp) used by L‑BFGS
# --------------------------------------------------------------

def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)          # logits – stays on the simplex
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


# --------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# --------------------------------------------------------------

def subgradient_optimize_momentum(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with momentum on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)      # momentum buffer
    beta = 0.85               # momentum coefficient
    it = 0

    while time.time() < deadline:
        # true max‑convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max (using the mask)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decays like c / sqrt(it+1)
        step = 0.9 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


# --------------------------------------------------------------
# Targeted hill‑climbing (reducing the current peak)
# --------------------------------------------------------------

def targeted_hill_climb(
    a_start: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    n: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly attack the shift that attains the current maximum.
    Move a small fraction of mass from the worst contributing index
    to a location that does *not* participate in that shift.
    Accept only if the max‑convolution value decreases (occasionally
    accept a tiny uphill move to escape flat plateaux).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    cur_score = best_score

    while time.time() < deadline:
        # current full convolution and the worst shift
        conv = full_conv(a, fft_len_conv)
        max_val = conv.max()
        k = int(np.argmax(conv))                 # shift index where the max occurs

        # indices that actually contribute to this shift
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        j_range = k - i_range

        # contribution of each i in the active interval
        contrib = a[i_range] * a[j_range]
        src_local = int(i_range[np.argmax(contrib)])   # source that contributes most

        # pick a target outside the active interval (so the move cannot increase this max)
        # try a few random attempts before falling back to a completely random index
        for _ in range(5):
            tgt = rng.integers(n)
            if not (i_min <= tgt <= i_max):
                break
        else:
            # deterministic fallback: first index outside the interval
            if i_min > 0:
                tgt = rng.integers(i_min)
            else:
                tgt = rng.integers(i_max + 1, n)

        if a[src_local] <= 0.0:
            continue

        # transfer a modest fraction of the source mass
        frac = rng.uniform(0.02, 0.15)           # a bit less aggressive than annealing
        delta = a[src_local] * frac
        a_new = a.copy()
        a_new[src_local] -= delta
        a_new[tgt] += delta

        new_score = fast_score(a_new, fft_len_conv, n)

        # Accept if it improves; otherwise accept with a tiny probability
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score, best_a = new_score, a_new.copy()
        else:
            # occasional uphill acceptance to avoid getting stuck
            if rng.random() < 0.001:
                a = a_new
                cur_score = new_score

    return best_a, best_score


# --------------------------------------------------------------
# Simulated annealing (fallback global search)
# --------------------------------------------------------------

def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len_conv: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb that accepts uphill moves with a square‑root
    cooling schedule.  Mass is transferred between two random positions.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if a[i] <= 0.0:
            continue

        delta = a[i] * rng.random() * 0.30
        if delta <= 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = fast_score(a_new, fft_len_conv, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_double_exp(n: int, rate1: float, rate2: float, w: float) -> np.ndarray:
    """
    Asymmetric double‑exponential: w·exp(-rate1·i) + (1−w)·exp(-rate2·(n−1−i))
    """
    i = np.arange(n, dtype=float)
    left = np.exp(-rate1 * i)
    right = np.exp(-rate2 * (n - 1 - i))
    a = w * left + (1.0 - w) * right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Systematic search over a handful of well‑known families (exp, power,
    logistic, double‑exp).  Returns the best vector seen.
    """
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # ----- exponential -------------------------------------------------
    for rate in np.linspace(0.00015, 0.0012, 51):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # ----- power‑law ---------------------------------------------------
    for p in np.linspace(-6.0, -0.3, 25):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # ----- logistic ----------------------------------------------------
    mus = np.linspace(0.25 * n, 0.75 * n, 9)
    sigmas = np.linspace(150.0, 1300.0, 9)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # ----- double exponential -------------------------------------------
    rates1 = np.linspace(0.0002, 0.0010, 9)
    rates2 = np.linspace(0.0002, 0.0010, 9)
    ws = [0.3, 0.5, 0.7]
    for r1 in rates1:
        for r2 in rates2:
            for w in ws:
                if time.time() >= deadline:
                    return best_a, best_sc
                a = generate_double_exp(n, r1, r2, w)
                sc = fast_score(a, fft_len, n)
                if sc < best_sc:
                    best_sc, best_a = sc, a.copy()
                    print(f"[grid] double_exp r1={r1:.5f} r2={r2:.5f} w={w:.2f} → score={best_sc:.6f}")

    return best_a, best_sc

            sc1 = fast_score(a1, fft_len_conv, n)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,

            maxiter_per_lam,
            phase1_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1‑lbfgs] improved fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ----------------------------------------------------------
    phase2_deadline = start_time + 0.65 * budget_s
    if time.time() < phase2_deadline:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2‑subgrad] best fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 3 – targeted hill‑climbing on the peak shift
    # ----------------------------------------------------------
    phase3_deadline = start_time + 0.85 * budget_s
    if time.time() < phase3_deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            phase3_deadline,
            fft_len_conv,
            n,
            rng,
        )
        print(f"[phase3‑hill] best fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Phase 4 – final short simulated annealing
    # ----------------------------------------------------------
    phase4_deadline = end_time
    if time.time() < phase4_deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase4_deadline,
            rng,
        )
        print(f"[phase4‑anneal] best fast_score={best_score:.6f}")

    # ----------------------------------------------------------
    # Official evaluation
    # ----------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] official evaluated score = {final_eval:.6f}")

    return final_seq
