"""
search_step_function.py
-----------------------

Improved search for a non‑negative step sequence that minimizes the
evaluation function defined in ``evaluate_sequence``.

Strategy
~~~~~~~~
1. Load the previous best sequence if it exists (``height_sequence_1``);
   otherwise start with a uniform distribution of length 2000.
2. Build a large pool of cheap parametric seeds (exponential, power‑law,
   Gaussian, sinusoid, Dirichlet draws, convex combinations of exponentials).
   Keep the best few seeds.
3. For each seed perform a short greedy hill‑climb (strictly improving
   mass transfers) – this quickly discards bad seeds.
4. Starting from the current best distribution, run a *permutation*‑based
   simulated annealing: the multiset of values is kept fixed while the
   ordering is permuted.  By spreading large values apart the peak of the
   convolution can be reduced.
5. Refine the best permutation with a projected sub‑gradient descent that
   works directly on the true max‑convolution (exact sub‑gradient).
6. Final polishing with a longer greedy hill‑climb.

All phases respect the probability simplex (non‑negative, sum = 1) and
use FFT‑based convolutions (O(n log n)).  The routine adapts automatically
to the available time budget (default 1000 s) and returns the best
sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` when ``a`` is already
    normalised to sum = 1.  Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { x ≥ 0, sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer moves)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Simple greedy hill‑climb that transfers a small random fraction of mass
    from one coordinate to another.  The sum is always preserved.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the donor mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) using the true max‑convolution as the
    sub‑gradient.  A simple back‑tracking line‑search is employed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # compute full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12

        # mask of indices attaining (approximately) the maximum
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        # gradient w.r.t. a (only the n entries that belong to the original vector)
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(4):
            cand = simplex_projection(a - step * g)
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1
    return best_a, best_score

        if best_perm_score < best_score:
            best_score, best_a = best_perm_score, best_perm_a.copy()
            print(f"[SA‑perm] improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (exact max‑convolution)
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        best_a, best_score = subgradient_optimize(best_a,
                                                  phase3_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑grad] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (more iterations)
    # ------------------------------------------------------------------
    if time.time() < phase4_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline=phase4_deadline,
                                        max_iter=15000)
        print(f"[final hill] score={best_score:.6f}")

    # Return as plain Python list and report the true score
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # externally provided
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
