"""
propose_candidate – hybrid large‑scale optimizer.

The algorithm focuses on a single large problem size (n≈8000) –
the asymptotic constant 1.5 is approached only for long sequences.
It mixes several optimisation ingredients:

*  Random / parametric initialisations (uniform, Dirichlet,
   power‑law, exponential, edge‑biased, triangular).
*  Continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
   max‑convolution (the same idea as the reference code but with
   a finer λ‑schedule and more iterations per stage).
*  An enhanced projected sub‑gradient method that uses the *full*
   active set of maximal convolution indices and also tries an
   exponentiated‑gradient step.
*  Targeted and random hill‑climbs for polishing.
*  A final deep refinement that repeatedly applies a low‑λ L‑BFGS
   pass followed by the active‑set sub‑gradient descent.

All steps respect the global time budget (≈1000 s).  The best
sequence found is returned; it is stored in the global variable
`height_sequence_1` for possible warm‑starting in later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT, trimmed to length 2n‑1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the simplex (Euclidean)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1} (Wang‑Carreira‑Perpiñán)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """Smooth approximation of max(a*a) and its gradient w.r.t. logits y."""
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                     # make the maximum zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of the smooth max ----
    w = exp_shift / Z                    # ∂smooth_max/∂conv (length 2n‑1)

    # convolution of a (reversed) with w gives the gradient w.r.t a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # back‑propagation through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """Continuation L‑BFGS with decreasing smoothing parameter."""
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Enhanced sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_active_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.40,
    rng_seed: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient (and exponentiated‑gradient) descent on
    f(a)=2·n·max(a*a).  Uses the *full* active set of maximal
    convolution indices.
    """
    rng = np.random.default_rng(rng_seed)
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    idx = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]

        # gradient of the max (average over active indices)
        grad = np.zeros_like(a)
        for k in active:
            j = k - idx
            mask = (j >= 0) & (j < n)
            grad[mask] += 2.0 * a[j[mask]]
        if len(active) > 1:
            grad /= len(active)

        eta = eta0 / np.sqrt(t + 1.0)

        # Euclidean projection step
        a_euc = project_simplex(a - eta * grad)
        score_euc = fast_score(a_euc, fft_len, n)

        # Exponentiated‑gradient step
        exp_factor = np.exp(-eta * grad)
        a_eg = a * exp_factor
        sum_eg = a_eg.sum()
        if sum_eg > 0:
            a_eg /= sum_eg
        else:
            a_eg = np.full_like(a, 1.0 / n)
        score_eg = fast_score(a_eg, fft_len, n)

        # pick the better candidate
        if score_euc <= score_eg:
            a_new, new_score = a_euc, score_euc
        else:
            a_new, new_score = a_eg, score_eg

        if new_score < best_score:
            best_score, best_a = new_score, a_new.copy()

        # simple acceptance rule (always move to the better of the two)
        a = a_new

        # occasional random jitter to escape plateaus
        if (t + 1) % 300 == 0 and rng.random() < 0.05:
            noise = rng.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Random / targeted hill‑climbs (as in the reference implementation)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves mass from the largest to smallest contributor."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Initial sequence generators (various parameterised families)
# ----------------------------------------------------------------------


def generate_initial(n: int, mode: str, rng: np.random.Generator) -> np.ndarray:
    """Return a non‑negative probability vector of length n."""
    if mode == "uniform":
        a = np.full(n, 1.0 / n)

    elif mode == "random":
        a = rng.dirichlet(np.ones(n))

    elif mode == "power":
        p = rng.uniform(0.2, 3.0)
        a = np.arange(1, n + 1, dtype=float) ** (-p)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.05)
        a = np.exp(-rate * np.arange(n, dtype=float))

    elif mode == "tri":
        # non‑symmetric linear‑ish decay
        p = rng.uniform(0.5, 2.0)
        a = (1.0 - np.arange(n, dtype=float) / (n - 1)) ** p

    elif mode == "edge":
        # large weight on a few edge entries
        edge_frac = rng.uniform(0.01, 0.05)
        m_edge = max(1, int(edge_frac * n))
        high = 0.5 / (2 * m_edge)          # allocate half of the mass to edges
        low = (1.0 - 0.5) / (n - 2 * m_edge)
        a = np.full(n, low)
        a[:m_edge] = high
        a[-m_edge:] = high

    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.full(n, 1.0 / n)
    return a

        # --------------------------------------------------------------
        # 3) Targeted hill‑climb
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=2000,
            frac=0.07,
        )

        # --------------------------------------------------------------
        # 4) Random hill‑climb for polishing
        # --------------------------------------------------------------
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=2500,
        )
        iteration += 1

    # ------------------------------------------------------------------
    # Deep final refinement (low‑λ L‑BFGS + sub‑gradient)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.5:
        # final low‑λ L‑BFGS passes
        y_start = np.log(best_a + 1e-12)
        lam_fine = np.geomspace(0.1, 1e-8, num=20)
        y_ref = schedule_optimize(
            y_start=y_start,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_fine,
            maxiter_per_stage=800,
        )
        best_a = softmax(y_ref)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final‑L‑BFGS] score = {best_score:.6f}")

        # final active‑set sub‑gradient descent
        sub_deadline = deadline - 0.4
        best_a, best_score = subgrad_active_descent(
            best_a,
            n,
            fft_len,
            max_iters=5000,
            deadline=sub_deadline,
            eta0=0.30,
            rng_seed=seed + 9999,
        )
        print(f"[final‑sub] score = {best_score:.6f}")

        # brief random hill‑climb
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline - 0.1,
            max_iter=3000,
        )
        print(f"[final‑hill] score = {best_score:.6f}")

    return best_a, best_score

    Search for a step‑function sequence with a low evaluation score.

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 8000  # 8‑k length gives a good trade‑off between quality & speed
    per_size_budget = (total_deadline - time.time()) * 0.95

    # Warm‑start from any previously discovered construction
    prev = globals().get("height_sequence_1")

        prev_arr = np.asarray(prev, dtype=float)

    # expose the best sequence for possible reuse
    globals()["height_sequence_1"] = final_seq
    return final_seq
