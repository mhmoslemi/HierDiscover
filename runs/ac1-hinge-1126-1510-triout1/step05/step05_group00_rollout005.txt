def expgrad_optimize(a0, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it =0
    while time.time() < deadline:

        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_val = conv.max()
        # mask for entries close to max
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[n-1:n-1+n]  # get length n slice

        eta = 0.5 / np.sqrt(it + 1)

        it += 1

"""
propose_candidate
-----------------
Search for a non‑negative step‑function (a probability vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

where `conv` is the linear convolution.  The algorithm combines:
* a small family‑search over simple analytic shapes,
* many random‑restart L‑BFGS optimisations on a smooth log‑sum‑exp
  approximation of the max‑convolution,
* an exponentiated‑gradient sub‑gradient stage (multiplicative mirror
  descent on the simplex),
* a short Adam fine‑tune on the same smooth objective,
* a simulated‑annealing pass and a final greedy hill‑climb.

The whole search respects a 1000 s wall‑clock budget (default) and
returns the best sequence found.  All intermediate scores are printed
for progress tracking.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT helpers & simple utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using the log‑sum‑exp trick.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # --- Convolution (a*a) ---
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --- Gradient of the smooth max w.r.t. convolution entries ---
    w = exp_shift / Z  # d smooth_max / d conv_i

    # Back‑propagation through the convolution: d conv / d a = (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] new best score={best_score:.6f} (lam={lam:.4f})")
    return best_a, best_score


def expgrad_optimize(a0: np.ndarray,
                     deadline: float,
                     fft_len_conv: int,
                     fft_len_g: int,
                     n: int):
    """
    Projected exponentiated‑gradient descent on the (non‑smooth) max‑convolution.
    Uses multiplicative updates a ← a·exp(−η·g) and projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size (1/√t) schedule
        eta = 0.5 / np.sqrt(it + 1.0)

        # multiplicative update + simplex projection
        a_new = a * np.exp(-eta * g)
        a_new = simplex_projection(a_new)

        score = fast_score(a_new, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    print(f"[expgrad] finished after {it} iters, best={best_score:.6f}")
    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max‑convolution objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    print(f"[Adam] finished after {it} iters, best={best_score:.6f}")
    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simple simulated‑annealing that moves a random fraction of mass between
    two coordinates.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10  # initial temperature
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T = max(T * 0.85, 5e-5)
    print(f"[anneal] finished after {it} iters, best={best_score:.6f}")
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    print(f"[hill‑climb] finished after {it} iters, best={best_score:.6f}")
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a diverse random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Analytic families (used for a fast deterministic warm‑start)
# ----------------------------------------------------------------------


def cosine_power_shape(n: int, p: float) -> np.ndarray:
    """Symmetric edge‑heavy shape: (0.5·(1+cos(π·i/(n‑1))))^p."""
    i = np.arange(n)
    base = 0.5 * (1.0 + np.cos(np.pi * i / (n - 1)))
    a = base ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def edge_abs_power_shape(n: int, p: float) -> np.ndarray:
    """Shape that grows towards the ends: |2·i/(n‑1)‑1|^p."""
    i = np.arange(n, dtype=float) / (n - 1)
    a = np.abs(2.0 * i - 1.0) ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum


def triangular_shifted_shape(n: int, shift: int) -> np.ndarray:
    """A simple triangular window shifted by `shift` positions."""
    i = (np.arange(n) + shift) % n
    x = i / (n - 1)
    a = 1.0 - np.abs(2.0 * x - 1.0)  # peak in the middle, zero at edges
    a_sum = a.sum()
    if a_sum == 0.0:
        return np.full(n, 1.0 / n)
    return a / a_sum

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000                         # keep the size that proved tractable
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise with a deterministic family search
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)     # uniform baseline
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform baseline score={best_score:.6f}")

    # if a previous construction is present, bring it in as another candidate
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # stretch/truncate to length n by linear interpolation
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, old)
            a = np.maximum(a, 0.0)
            a_sum = a.sum()
            if a_sum > 0:
                a = a / a_sum
                sc = fast_score(a, fft_len_conv, n)
                if sc < best_score:
                    best_a, best_score = a, sc
                    print(f"[init] imported height_sequence_1 score={sc:.6f}")

    # Sweep a few cheap analytic families
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        a = cosine_power_shape(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[family] cosine^p={p} score={sc:.6f}")

    for p in [0.5, 1.0, 2.0, 3.0, 4.0]:
        a = edge_abs_power_shape(n, p)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[family] edge‑abs^p={p} score={sc:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    phase1_end = start_time + 0.55 * budget_s      # intensive L‑BFGS
    phase2_end = start_time + 0.80 * budget_s      # exponentiated‑gradient
    phase3_end = start_time + 0.93 * budget_s      # Adam fine‑tune
    phase4_end = start_time + 0.98 * budget_s      # annealing
    final_deadline = deadline                      # hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – many random‑restart L‑BFGS with a continuation schedule
    # --------------------------------------------------------------
    lam_schedule = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3,
                    0.2, 0.15, 0.1, 0.07, 0.04,
                    0.02, 0.01, 0.005, 0.002, 0.001]
    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Occasionally perturb the current best; otherwise random start
        if rng.random() < 0.20:
            sigma = rng.uniform(0.0008, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                          lam_schedule,
                                          maxiter_per_lam,
                                          phase1_end,
                                          fft_len_conv,
                                          fft_len_g,
                                          n)
        if sc_opt < best_score:
            best_a, best_score = a_opt, sc_opt
            print(f"[phase1] improved score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = expgrad_optimize(best_a,
                                              phase2_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        print(f"[phase2] exp‑grad score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning on the smoothed objective
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           phase3_end,
                                           lam_factor=0.002,
                                           lr=0.15)
        print(f"[phase3] Adam score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – short simulated annealing
    # --------------------------------------------------------------
    if time.time() < phase4_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase4_end)
        print(f"[phase4] anneal score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # --------------------------------------------------------------
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=10000)
    print(f"[final] hill‑climb score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluate_sequence gives {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence raised:", e)
        final_eval = None

    return final_seq
