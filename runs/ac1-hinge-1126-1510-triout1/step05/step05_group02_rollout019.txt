def greedy_support(n, target_frac, rng, max_trials=30):

    target_size = int(target_frac*n)

    support = np.zeros(n, dtype=bool)

    pair_counts = np.zeros(2*n-1, dtype=int)

    size = 0

    while size < target_size:

        # sample candidate indices not in support
        candidates = rng.choice(np.where(~support)[0], size=min(max_trials, np.sum(~support)), replace=False)

        best_i = None
        best_max = np.iinfo(np.int32).max
        # store contributions for best candidate to reuse later maybe compute quickly
        best_contrib = None
        # to accelerate we compute indices of existing support:
        supp_idx = np.where(support)[0]  # array of size size

        for i in candidates:
            # compute contributions
            # idx = i + supp_idx -> add 2
            # we can compute using np.bincount
            idxs = i + supp_idx
            # Since sum range up to 2n-2. Use np.bincount with minlength=2*n-1.
            contrib = np.bincount(idxs, minlength=2*n-1) * 2
            # diag
            contrib[2*i] += 1
            # compute new max
            cur_max = pair_counts.max()
            new_max = np.maximum(pair_counts, pair_counts + contrib).max()
            # Actually pair_counts + contrib may be >cur_max; evaluate new_max = max(pair_counts+contrib)
            new_max = (pair_counts + contrib).max()
            if new_max < best_max:
                best_max = new_max
                best_i = i
                best_contrib = contrib
        if best_i is None:
            # no candidate improved, break (rare)
            break
        # update
        support[best_i] = True
        pair_counts += best_contrib
        size += 1

    return support, pair_counts

def build_sparse_sequence(n, target_size, rng, sample_per_iter=30):
    support = np.zeros(n, dtype=bool)
    pair_counts = np.zeros(2*n-1, dtype=int)
    while support.sum() < target_size:
        remaining = np.where(~support)[0]
        if remaining.size == 0:
            break
        # sample candidates
        sample = rng.choice(remaining, size=min(sample_per_iter, remaining.size), replace=False)
        sup_idx = np.where(support)[0]  # can cache or update each loop
        best_max = np.iinfo(np.int64).max
        best_i = -1
        best_contrib = None
        # iterate candidates
        for i in sample:
            if sup_idx.size == 0:
                # no existing support
                contrib = np.zeros(2*n-1, dtype=int)
                contrib[2*i] = 1
            else:
                idxs = i + sup_idx
                contrib = np.bincount(idxs, minlength=2*n-1, dtype=int) * 2

            new_max = (pair_counts + contrib).max()

                best_max = new_max
                best_i = i
                best_contrib = contrib
        # add best_i
        support[best_i] = True
        pair_counts += best_contrib
    return support

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len_conv: int,
                       fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in freq. domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    # initialise logits
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_mirror_descent(a0: np.ndarray,
                       deadline: float,
                       fft_len_conv: int,
                       n: int):
    """
    Projected exponentiated‑gradient descent on the exact max‑convolution.
    Returns best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())

        # sub‑gradient of the max (exact, not smoothed)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]          # d max / d a_i

        # larger step‑size early on
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective (fine‑tuning)
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective starting from logits y0.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing phase
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T *= 0.92
            T = max(T, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialiser (dense families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    A mixture of families is used to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sine', 'halfsine'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    else:                       # halfsine
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Greedy construction of a low‑peak support
# ----------------------------------------------------------------------


def greedy_support(n: int,
                   target_frac: float,
                   rng: np.random.Generator,
                   max_trials_per_step: int = 30) -> np.ndarray:
    """
    Build a binary support S (size ≈ target_frac·n) such that the
    maximal number of ordered pairs (i,j)∈S×S with i+j=k is small.
    Returns a bool array ``support`` where support[i] is True iff i∈S.
    """
    target_size = max(2, int(target_frac * n))
    support = np.zeros(n, dtype=bool)
    pair_counts = np.zeros(2 * n - 1, dtype=np.int64)

    # maintain list of currently selected indices for fast index access
    selected = np.empty(0, dtype=np.int64)

    while support.sum() < target_size:
        # choose a modest batch of candidates among the still‑free positions
        free = np.where(~support)[0]
        if free.size == 0:
            break
        cand_batch = rng.choice(free,
                                size=min(max_trials_per_step, free.size),
                                replace=False)

        best_max = np.iinfo(np.int64).max
        best_i = -1
        best_contrib = None

        for i in cand_batch:
            # contributions of adding i to the current set
            if selected.size == 0:
                contrib = np.zeros(2 * n - 1, dtype=np.int64)
                contrib[2 * i] = 1                     # diagonal (i,i)
            else:
                idxs = i + selected                    # i + j for all j∈S
                contrib = np.bincount(idxs,
                                      minlength=2 * n - 1,
                                      dtype=np.int64) * 2   # ordered pairs (i,j) & (j,i)
                contrib[2 * i] += 1                    # diagonal term
            new_max = (pair_counts + contrib).max()
            if new_max < best_max:
                best_max = new_max
                best_i = i
                best_contrib = contrib

        # add the chosen index to the support
        support[best_i] = True
        pair_counts += best_contrib
        selected = np.append(selected, best_i)

    return support


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000                     # we can afford up to 4000 steps
    n = max_n // 2                  # keep FFT sizes reasonable (2000)
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient‑related convolutions

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – greedy low‑peak support construction (quick)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)         # uniform fallback
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform score = {best_score:.6f}")

    # try a few different target densities
    for target_frac in (0.5, 0.6, 0.7, 0.8):
        if time.time() >= deadline:
            break
        support = greedy_support(n, target_frac, rng, max_trials_per_step=25)
        if support.sum() < 2:
            continue
        a = support.astype(float)
        a /= a.sum()                     # normalise to sum = 1
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[greedy] target_frac={target_frac:.2f} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase division (time allocation)
    # ------------------------------------------------------------------
    now = time.time()
    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS / random restarts
    phase2_end = start_time + 0.80 * budget_s   # mirror‑descent + Adam
    # remaining time will be used for anneal & hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts (including the greedy seed)
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001]
    maxiter_per_lam = 800
    while time.time() < phase1_end:
        a_start = random_initial(n, rng) if rng.random() < 0.6 else best_a.copy()
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – mirror‑descent + Adam fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        md_deadline = phase2_end - 0.5
        best_a, best_score = exp_mirror_descent(best_a,
                                                md_deadline,
                                                fft_len_conv,
                                                n)
        print(f"[exp‑mirror] best fast_score = {best_score:.6f}")

        if time.time() < phase2_end - 0.2:
            adam_deadline = phase2_end - 0.2
            logits = np.log(best_a + 1e-12)
            best_a, best_score = adam_optimize(logits,
                                               n,
                                               fft_len_conv,
                                               fft_len_g,
                                               adam_deadline,
                                               lam_factor=0.002,
                                               lr=0.15)
            print(f"[Adam] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.3
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    deadline,
                                    max_iter=8000)
    print(f"[final hill‑climb] best fast_score = {best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
