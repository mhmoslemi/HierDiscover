"""
Optimisation of a non‑negative step‑function sequence to minimise
evaluate_sequence(...).  The algorithm combines:
• Randomised restarts of a continuation L‑BFGS on a smooth‑max surrogate.
• Targeted/random hill‑climbs.
• An LP‑based refinement that locally linearises the max‑convolution
  constraints and solves the resulting linear programme to decrease the
  maximal convolution value.
• A final projected‑gradient descent on the smooth‐max objective.
"""

from scipy.optimize import minimize, linprog

def lp_refine(a, n, deadline, max_iter=200):
    """Linear‑programming refinement of a to lower its maximum auto‑convolution."""
    fft_len = next_pow2(2*n-1)
    # initial convolution
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    best_a = a
    best_conv = conv
    best_val = conv.max()
    for _ in range(max_iter):
        if time.time() >= deadline:
            break
        # choose the most critical shifts (up to 12 of them)
        idx_order = np.argsort(-conv)[:12]
        # build LP matrices
        m = len(idx_order)
        A_ub = np.zeros((m, n+1))
        b_ub = np.empty(m)
        for row, k in enumerate(idx_order):
            low = max(0, k - (n-1))
            high = min(k, n-1)
            i_idx = np.arange(low, high+1)
            j_idx = k - i_idx
            A_ub[row, i_idx] = 2.0 * a[j_idx]
            A_ub[row, n] = -1.0
            b_ub[row] = -conv[k]
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = [(-a[i], a[i]) for i in range(n)] + [(0.0, None)]
        # Solve
        res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'time_limit': max(0.0, deadline - time.time())})
        if not res.success:
            break
        d = res.x[:n]
        # apply full step
        a_new = a + d
        a_new = np.maximum(a_new, 0.0)
        # renormalise – the equality already forces sum unchanged,
        # but rounding errors may accumulate
        s = a_new.sum()
        if s <= 0:
            break
        a_new /= s
        # evaluate new convolution
        fa = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        t_new = conv_new.max()
        if t_new < best_val - 1e-12:
            best_a = a_new
            best_conv = conv_new
            best_val = t_new
            a = a_new
            conv = conv_new
            # continue refining
        else:
            # no improvement – stop early
            break
    return best_a, best_val

"""
Optimisation of a non‑negative step‑function sequence to minimise
``evaluate_sequence``.  The algorithm combines several complementary
techniques:

1. Randomised restarts of a continuation L‑BFGS optimisation on a
   smooth‑max surrogate of the max‑auto‑convolution.
2. Two hill‑climbs (targeted and random) that move mass between
   indices only when the fast surrogate score improves.
3. An LP‑based refinement: at the current point we linearise the
   active maximal convolution constraints and solve a small linear
   programme that guarantees a decrease of the maximal value.
4. A final projected‑gradient descent on the smooth‑max objective
   (with a tiny smoothing factor) and one last random hill‑climb.

All steps keep the vector on the probability simplex; after the search
the sequence is evaluated with the official ``evaluate_sequence`` and
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a (full) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max ----
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max w.r.t. conv ----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ---- chain rule back to a (gradient vector of length n) ----
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    In each step the shift giving the maximal value is identified, the most
    contributing index pair is found, and a fraction of mass is moved away
    from the worst contributor to a location whose partner under the same
    shift is (almost) empty.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        # choose the worst contributor (largest product)
        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift (used to reject bad targets)
        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        # sample a few candidate target indices and pick the one whose partner
        # under the critical shift is smallest
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        # transfer a fraction of mass from i → j
        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns a vector of length n (without the outer 2·n factor).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # derivative of smooth max w.r.t. a (outer factor 2 will be added later)
    return 2.0 * g


def lp_refine(a: np.ndarray,
              n: int,
              deadline: float,
              max_iter: int = 200) -> tuple[np.ndarray, float]:
    """
    Linear‑programming based refinement.
    At the current point we linearise the (few) active maximal
    convolution constraints and solve the resulting LP to obtain a
    feasible descent direction on the simplex.
    """
    fft_len = next_pow2(2 * n - 1)
    # initial convolution
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    best_val = conv.max()
    best_a = a.copy()

    for _ in range(max_iter):
        if time.time() >= deadline:
            break

        # indices of the most critical shifts (up to 12 of them)
        idx = np.argsort(-conv)[:12]

        m = len(idx)
        A_ub = np.zeros((m, n + 1))
        b_ub = np.empty(m)

        for row, k in enumerate(idx):
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            i_idx = np.arange(low, high + 1)
            j_idx = k - i_idx
            A_ub[row, i_idx] = 2.0 * a[j_idx]     # derivative of conv_k w.r.t. a_i
            A_ub[row, n] = -1.0                    # -τ
            b_ub[row] = -conv[k]

        # equality: sum of changes = 0 so that total mass stays 1
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        # bounds:  -a_i ≤ Δa_i ≤ 1‑a_i   (allow increase up to full mass)
        bounds = [(-a[i], 1.0) for i in range(n)] + [(0.0, None)]

        # minimise the new maximal value τ
        res = linprog(c=np.concatenate([np.zeros(n), [1.0]]),
                      A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs')

        if not res.success:
            break

        d = res.x[:n]

        a_new = a + d
        a_new = np.maximum(a_new, 0.0)
        # sum is preserved by the equality but we renormalise against round‑off
        s = a_new.sum()
        if s <= 0.0:
            break
        a_new /= s

        # evaluate new convolution
        fa = np.fft.fft(a_new, fft_len)
        conv_new = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        new_val = conv_new.max()

        if new_val < best_val - 1e-12:
            best_val = new_val
            best_a = a_new
            a = a_new
            conv = conv_new
        else:
            # no improvement – stop early
            break

    return best_a, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction when possible
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # L‑BFGS continuation schedule (global search)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=45)   # finer λ‑schedule
    maxiter_per_stage = 800                         # work per λ‑stage

    iteration = 0
    # Outer loop: many random restarts until time runs out
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            center_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, center_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturb the current best solution
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb -----
        tc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- fallback random hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=4000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # LP based refinement (often yields the last few percent)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[lp] starting LP refinement")
        best_a, _ = lp_refine(best_a, n, deadline, max_iter=200)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[lp] after refinement fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected‑gradient descent on the smooth‑max (λ → 0)
    # ------------------------------------------------------------------
    lam_factor_pg = 1e-5
    step0 = 0.28
    max_pg_iter = 3000
    pg_iter = 0
    while time.time() < deadline - 2.0 and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg) * (2.0 * n)
        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            lam_factor_pg = max(lam_factor_pg * 0.9, 1e-7)

    print(f"[pg] projected‑gradient iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
