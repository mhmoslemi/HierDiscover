def lp_improve(a, n, deadline, t_init=0.1, max_iters=12):
    import scipy.optimize as opt
    from scipy import sparse

    a = a / a.sum()  # ensure normalized

    t = t_init
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Determine active constraints (top values near M)
        eps = 1e-9 * M
        active_idx = np.where(conv >= M - eps)[0]
        # If too many active constraints, randomly sample some
        if active_idx.size > n:
            # sample n indices
            active_idx = np.random.choice(active_idx, size=n, replace=False)
        # Build A_ub matrix
        rows = []
        data = []
        col_idx = []
        for row_num, k in enumerate(active_idx):
            # for each offset k, build row
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            # for i in [start_i, end_i], j=k-i
            for i in range(start_i, end_i + 1):
                j = k - i
                rows.append(row_num)
                col_idx.append(j)
                data.append(a[i])
        A_ub = sparse.csr_matrix((data, (rows, col_idx)), shape=(len(active_idx), n))

        # Upper bounds: M
        b_ub = np.full(len(active_idx), M)

        # Objective: maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n)

        # Bounds for b: (0, None)
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status != 0:
            break
        b_val = np.maximum(res.x, 0)
        s = b_val.sum()
        if s == 0:
            break
        b_scaled = b_val / s

        a_new = (1 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2*n-1))
        old_score = fast_score(a, n, next_pow2(2*n-1))

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init
        else:
            t *= 0.5
            if t < 1e-5:
                break

        it += 1

    return a
```

def lp_improve(a, n, deadline, t_init=0.12, max_iters=8):
    import scipy.optimize as opt
    import scipy.sparse as sp

    a = a.copy()
    a = a / a.sum()
    t = t_init
    best_score = fast_score(a, n, next_pow2(2*n-1))
    best_a = a

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()
        # find active constraints
        eps = max(1e-12, M * 0.001)  # 0.1% tolerance
        idx_active = np.where(conv >= M - eps)[0]
        if idx_active.size == 0:
            break
        # cap size
        max_constraints = min(idx_active.size, n*2)
        if idx_active.size > max_constraints:
            idx_active = np.random.choice(idx_active, size=max_constraints, replace=False)

        rows = []
        cols = []
        data = []
        for row_id, k in enumerate(idx_active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # loop over i
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(row_id)
                cols.append(j)
                data.append(a[i])
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(idx_active), n))

        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(len(idx_active), M)

        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            break
        b_val = np.maximum(res.x, 0)
        if b_val.sum() <= 0:
            break
        b_scaled = b_val / b_val.sum()
        a_new = (1 - t) * a + t * b_scaled
        a_new /= a_new.sum()
        new_score = fast_score(a_new, n, next_pow2(2*n-1))
        if new_score < best_score - 1e-12:
            a = a_new
            best_a = a
            best_score = new_score
            t = t_init
        else:
            t *= 0.5
            if t < 1e-6:
                break
    return best_a, best_score
```

    candidates = []

        tot = arr.sum()
        if tot <= 0:
            return

        sc = fast_score(arr, n, fft_len_conv)

    add_candidate(np.arange(n, 0, -1, dtype=float))

    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:

"""
A practical search for a low‐evaluation step function.

The algorithm proceeds in four stages:

1.  Cheap parametric sweep – generate a diverse set of candidate sequences
    (uniform, linear, exponential, power‑law, Hann/Hamming windows, random
    Dirichlet draws) and keep the best few.
2.  Continuation L‑BFGS – minimise a smooth (log‑sum‑exp) surrogate of the
    peak of the auto‑convolution, gradually decreasing the smoothing
    factor.  Each L‑BFGS run uses a soft‑max parametrisation to enforce the
    simplex, and after every λ‑stage a short hill‑climb is performed.
3.  Linear‑programming “balancing’’ – solve a small LP (via SciPy’s
    high‑performance `linprog`) that maximises the total mass subject to the
    current peak bound.  Mixing a fraction of the LP solution provably reduces
    the peak.
4.  Projected sub‑gradient descent followed by a longer greedy hill‑climb
    (mass transfer) to polish the result.

All steps respect the simplex (∑a = 1, a_i ≥ 0) and use FFT‑based
convolutions (O(n log n)).  The time budget (default 1000 s) is split
automatically; the best sequence found before the deadline is returned.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Score 2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    The smoothing parameter is lam = max_c * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                         # ∂smooth_max / ∂conv_i
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # short hill‑climb after each λ (0.4 s)
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent
    with a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # move a random fraction (≤5 %) of mass from i1 to i2
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming balancing step (SciPy implementation)
# ----------------------------------------------------------------------
def lp_improve(a: np.ndarray,
               n: int,
               deadline: float,
               t_init: float = 0.12,
               max_iters: int = 8):
    """
    Perform a few iterations of the LP “balancing’’ step:

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M    for all k

    Only the constraints that are close to the current peak M are kept,
    yielding a small sparse matrix.  After solving the LP the solution
    b is mixed back into a with a step size t (halved on failure).
    """
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    t = t_init
    best_score = fast_score(a, n, next_pow2(2 * n - 1))

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = np.convolve(a, a)
        M = conv.max()

        # active constraints: indices where conv is near its maximum
        eps = max(1e-12, M * 0.001)  # 0.1 % tolerance
        idx_active = np.where(conv >= M - eps)[0]

        # limit the number of constraints to keep the LP small
        max_cons = min(len(idx_active), 2 * n)
        if len(idx_active) > max_cons:
            idx_active = np.random.choice(idx_active, size=max_cons, replace=False)

        rows = []
        cols = []
        data = []
        for row_id, k in enumerate(idx_active):
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                j = k - i
                rows.append(row_id)
                cols.append(j)
                data.append(a[i])
        A_ub = sp.csr_matrix((data, (rows, cols)), shape=(len(idx_active), n))
        b_ub = np.full(len(idx_active), M)

        c = -np.ones(n)                     # maximise sum(b) → minimise –sum(b)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success or res.x is None:
            break

        b_val = np.maximum(res.x, 0.0)
        if b_val.sum() == 0.0:
            break
        b_scaled = b_val / b_val.sum()

        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()
        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))

        if new_score < best_score - 1e-12:
            a = a_new
            best_score = new_score
            t = t_init
        else:
            t *= 0.5
            if t < 1e-6:
                break

    return a, best_score
