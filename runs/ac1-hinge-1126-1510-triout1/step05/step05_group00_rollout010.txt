def evaluate_sequence(seq):
    # security checks...
    seq = [max(0, x) for x in seq] # clamp to non-negative
    seq = [min(1000.0, x) for x in seq] # clamp to max 1000
    n = len(seq)
    b_seq = np.convolve(seq, seq) # linear convolution length 2n-1.
    max_b = max(b_seq)
    sum_a = sum(seq)
    if sum_a < 0.01: return inf
    return 2 * n * max_b / (sum_a**2)

def generate_exponential(n, r):
    if r==1: uniform else...
    a = (1-r) * (r**np.arange(n)) / (1 - r**n)

def generate_symmetric_exponential(n, r):
    left = (1-r) * (r**np.arange(n))  # decays
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a

def generate_triangular(n):
    i = np.arange(n)
    a = np.where(i <= n//2, i+1, n-i)
    a = a.astype(np.float64)
    a /= a.sum()
    return a

def generate_cosine_window(n):

def generate_dpss(n, NW):
    from scipy.signal.windows import dpss

    w = np.maximum(w, 0.0)

"""
Propose a step sequence with a low evaluation score.

The algorithm consists of several stages:
1. Generate a diverse pool of initial candidates (uniform, exponential,
   triangular, cosine window, DPSS windows, random Dirichlet draws).
2. Perform many random‑restart L‑BFGS optimisations on a smoothed
   log‑sum‑exp approximation of the max‑convolution.  The smoothing
   factor (lam) is decreased gradually (continuation) so the optimiser
   first explores globally and then refines locally.
3. Apply a single linear‑programming “active‑set” update (the classic
   f‑g optimisation of the statement) to push the max‑convolution down.
   The LP solves maximise ∑g_j subject to non‑negativity and
   (f*g)(k) ≤ M (M = max (f*f)).  The optimum g is mixed with f.
4. Refine with a projected sub‑gradient descent with momentum,
   then Adam on the smoothed objective.
5. Conduct a simulated‑annealing pass followed by a greedy hill‑climb.
All heavy convolutions use FFT (real‑FFT for speed).  The routine respects
the 1000 s time budget and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_convolve(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) when sum(a)=1."""
    conv = full_convolve(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate generators (all return a vector that sums to 1)
# ----------------------------------------------------------------------


def generate_uniform(n: int) -> np.ndarray:
    a = np.full(n, 1.0 / n)
    return a


def generate_exponential(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (0<r<1)."""
    if r <= 0.0 or r >= 1.0:
        return generate_uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1 - r) * (r ** i) / (1 - r ** n)
    return a


def generate_symmetric_exponential(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two exponentials, decaying from both ends."""
    i = np.arange(n, dtype=np.float64)
    left = (1 - r) * (r ** i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def generate_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    half = (n - 1) / 2.0
    a = 1.0 - np.abs(i - half) / (half + 1.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** 2
    a /= a.sum()
    return a


def generate_dpss_window(n: int, nw: float) -> np.ndarray:
    from scipy.signal.windows import dpss

    w = dpss(n, nw, Kmax=1, sym=False)[0]
    w = np.maximum(w, 0.0)
    if w.sum() == 0:
        return generate_uniform(n)
    w /= w.sum()
    return w


def random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 0.3) -> np.ndarray:
    """Sample a Dirichlet vector (sparse when alpha<1)."""
    a = rng.gamma(alpha, 1.0, size=n)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mix several families to create a diverse starting point."""
    modes = ["uniform", "exp", "symexp", "tri", "cos", "dpss", "dir"]
    m = rng.choice(modes)
    if m == "uniform":
        return generate_uniform(n)
    if m == "exp":
        r = rng.uniform(0.01, 0.99)
        return generate_exponential(n, r)
    if m == "symexp":
        r = rng.uniform(0.01, 0.99)
        return generate_symmetric_exponential(n, r)
    if m == "tri":
        return generate_triangular(n)
    if m == "cos":
        return generate_cosine_window(n)
    if m == "dpss":
        nw = rng.uniform(2.0, 4.5)
        return generate_dpss_window(n, nw)
    # dirichlet
    alpha = rng.uniform(0.1, 0.7)
    return random_dirichlet(n, rng, alpha)


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) via log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y (where a = softmax(y)).
    """
    a = softmax(y)

    # ----- convolution a*a (full length 2n‑1) -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. convolution entries -----
    w = exp_shift / Z  # d smooth_max / d conv_i, shape (2n‑1,)

    # convolution of w with reversed a gives d f / d a
    a_rev = a[::-1]
    fw = np.fft.rfft(w, fft_len_g)
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # extremely rare, just skip this lam
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # ----- full convolution a*a -----
        fa = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)  # active set

        # ----- sub‑gradient of the max -----
        a_rev = a[::-1]
        fm = np.fft.rfft(mask, fft_len_g)
        fa_rev = np.fft.rfft(a_rev, fft_len_g)
        conv_g = np.fft.irfft(fa_rev * fm, fft_len_g)
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # ----- momentum update -----
        v = beta * v + (1.0 - beta) * g
        step = 0.7 / np.sqrt(it + 1.0)          # diminishing step
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing that moves a random fraction of mass between two
    coordinates, accepting worsening moves with a temperature that decays.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.15                     # start temperature (a bit higher than before)
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Move up to 20 % of the mass at i
        delta = cur_a[i] * rng.random() * 0.20
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.90
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10  # up to 10 % of mass at i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add zero‑mean Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Linear‑programming step (active‑set update)
# ----------------------------------------------------------------------


def lp_update(f: np.ndarray, M: float, n: int, rng: np.random.Generator):
    """
    Solve   max 1ᵀg   s.t.   C_f g ≤ M,   g ≥ 0,
    where (C_f g)_k = (f * g)_k.  Return the normalised g.
    """
    # Build sparse constraint matrix C_f (2n‑1) × n
    rows = 2 * n - 1
    data = []
    row_ind = []
    col_ind = []
    # For each column i, entries appear on rows i … i+n‑1 with value f[i]
    for i in range(n):
        length = n - i  # number of rows for this offset
        rows_i = np.arange(i, i + length)
        cols_i = np.full(length, i, dtype=int)
        data.append(np.full(length, f[i]))
        row_ind.append(rows_i)
        col_ind.append(cols_i)
    # Concatenate
    data = np.concatenate(data)
    row_ind = np.concatenate(row_ind)
    col_ind = np.concatenate(col_ind)
    A_ub = sp.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

    b_ub = np.full(rows, M)

    c = -np.ones(n)                     # maximise sum(g) == minimise -sum(g)
    bounds = [(0, None) for _ in range(n)]

    # Solve with HiGHS (fast for large sparse problems)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success or res.x is None:
        return None
    g0 = res.x
    if g0.sum() == 0:
        return None
    g = g0 / g0.sum()
    return g

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial population
    # ------------------------------------------------------------------
    init_pool = []

    # deterministic families
    init_pool.append(generate_uniform(n))
    init_pool.append(generate_triangular(n))
    init_pool.append(generate_cosine_window(n))

    for _ in range(3):
        r = rng.uniform(0.01, 0.99)
        init_pool.append(generate_exponential(n, r))
        init_pool.append(generate_symmetric_exponential(n, r))

    for nw in [2.0, 2.5, 3.0, 3.5, 4.0]:
        try:
            init_pool.append(generate_dpss_window(n, nw))
        except Exception:
            pass

    # random Dirichlet draws
    for _ in range(5):
        init_pool.append(random_dirichlet(n, rng, alpha=rng.uniform(0.1, 0.7)))

    # normalise (should already sum to 1 but safeguard)
    init_pool = [a / a.sum() if a.sum() > 0 else generate_uniform(n) for a in init_pool]

    # start with the best of these (fast_score)
    best_a = init_pool[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    for a in init_pool[1:]:
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.65 * budget_s   # LP update + sub‑gradient
    phase3_end = start_time + 0.85 * budget_s   # Adam + annealing
    # remaining time for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many random‑restart continuation L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        if best_score < np.inf and rng.random() < 0.20:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP active‑set update followed by sub‑gradient descent
    # ------------------------------------------------------------------
    while time.time() < phase2_end:
        # ---- LP update -------------------------------------------------
        conv = full_convolve(best_a, fft_len_conv)
        M = conv.max()
        g = lp_update(best_a, M, n, rng)
        if g is None:
            break
        # mix a little (t controls how aggressive the step is)
        t = 0.25
        a_new = (1.0 - t) * best_a + t * g
        a_new = simplex_projection(a_new)   # numerical safety
        new_score = fast_score(a_new, fft_len_conv, n)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[LP] new best score={best_score:.6f}")
        # ---- sub‑gradient descent (short burst) -----------------------
        sub_deadline = min(phase2_end, time.time() + 5.0)   # a couple of seconds
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                          sub_deadline,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] interim score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning, simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        # Adam with tighter smoothing
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           phase3_end,
                                           lam_factor=0.002,
                                           lr=0.15)
        print(f"[Adam] score={best_score:.6f}")

        # Short simulated annealing
        if time.time() < phase3_end:
            anneal_deadline = phase3_end - 0.2
            best_a, best_score = random_anneal(best_a,
                                               best_score,
                                               fft_len_conv,
                                               n,
                                               anneal_deadline)
            print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final evaluation using the external function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
