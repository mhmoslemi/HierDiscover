def solve_lp_opt(f):
    n = f.shape[0]
    # Build T using toeplitz, may be dense -> convert to CSR
    T_dense = scipy.linalg.toeplitz(np.concatenate((f, np.zeros(n-1))), np.concatenate((np.array([f[0]]), np.zeros(n-1))))
    # Convert to CSR
    T = scipy.sparse.csr_matrix(T_dense)
    # Build A_ub
    neg_one_col = -np.ones((2*n-1, 1))
    A_ub = scipy.sparse.hstack([T, scipy.sparse.csr_matrix(neg_one_col)])
    b_ub = np.zeros(2*n-1)
    # Equality
    A_eq = scipy.sparse.csr_matrix(np.hstack((np.ones((1, n)), np.zeros((1, 1)))))
    b_eq = np.array([1.0])
    #Objective
    c = np.concatenate((np.zeros(n), np.array([1.0])))
    # Bounds
    bounds = [(0, None)] * n + [(0, None)]
    res = scipy.optimize.linprog(c=c, A_ub=A_ub, b_ub=b_ub,
                                A_eq=A_eq, b_eq=b_eq,
                                bounds=bounds, method='highs')
    if not res.success:
        return None, None
    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, M_opt

def improve_by_lp_mix(f, best_score, fft_len_conv, deadline):
    # f normalized sum=1
    n = len(f)
    improved = False
    while time.time() < deadline:
        b_opt, M_opt = solve_lp_opt(f)
        if b_opt is None:
            break
        # Evaluate mixture candidates
        candidate_best = f
        candidate_score = best_score
        best_t = 0.0
        # try mix fractions
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            new_f = (1 - t) * f + t * b_opt
            new_f_sum = new_f.sum()
            if new_f_sum == 0:
                continue
            new_f = new_f / new_f_sum
            score = fast_score(new_f, fft_len_conv, n)
            if score < candidate_score:
                candidate_score = score
                candidate_best = new_f
                best_t = t
        if candidate_score < best_score - 1e-9:
            f = candidate_best
            best_score = candidate_score
            improved = True
            print(f"[LP] improved to score={best_score:.6f} (t={best_t})")
        else:
            # no improvement, break to avoid endless loop
            break
    return f, best_score, improved

"""
Improved step‑function search using LP‑based mixing.

The algorithm:
1. Generate a few simple parametric seeds and keep the best.
2. Start from the best seed (normalized to sum = 1).
3. Repeatedly:
   a) Solve a linear program that finds a distribution `b` with the
      smallest possible peak of the convolution `f*b` under
      `b≥0, Σb=1`.  The LP is
          minimize   M
          subject to T_f·b ≤ M,  Σb = 1,  b ≥ 0
      where `T_f` is the Toeplitz matrix representing convolution with `f`.
   b) Mix the current `f` with the obtained `b` using several step sizes
      (convex combinations).  Keep the mixture that gives the lowest
      fast evaluation `2·n·max(f*f)` (the true score is scale‑invariant,
      so we normalise to Σf = 1 and ignore the denominator).
   c) Stop when no mixing step improves the score.
4. Refine the result with a projected sub‑gradient descent on the exact
   max‑convolution and a final greedy hill‑climb.
All convolutions are performed by FFT for speed; the final sequence is
returned as a plain Python list.
"""

import time
import numpy as np
import scipy.linalg
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a distribution already
    normalised to sum = 1.
    Returns 2·n·max(conv) (the denominator Σa² is 1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """Greedy mass‑transfer hill‑climb (keeps sum = 1)."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.04  # ≤4 % of a[i]
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # exact convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # gradient of max (sub‑gradient)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(4):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming (LP) helper
# ----------------------------------------------------------------------
def solve_lp_min_max(f: np.ndarray) -> tuple[np.ndarray | None, float | None]:
    """
    Solve
        minimise   M
        s.t.       T_f · b ≤ M
                   Σ b = 1
                   b ≥ 0
    where T_f is the Toeplitz matrix representing convolution with `f`.
    Returns the optimal b (as a dense ndarray) and the optimal M.
    """
    n = f.shape[0]

    # Build Toeplitz matrix T_f (shape (2n‑1) × n)
    first_col = np.concatenate((f, np.zeros(n - 1, dtype=float)))
    first_row = np.concatenate((np.array([f[0]]), np.zeros(n - 1, dtype=float)))
    T_dense = scipy.linalg.toeplitz(first_col, first_row)
    T = scipy.sparse.csr_matrix(T_dense)

    # Inequality A_ub · [b; M] ≤ 0   (T·b - M·1 ≤ 0)
    neg_one = -np.ones((2 * n - 1, 1))
    A_ub = scipy.sparse.hstack([T, scipy.sparse.csr_matrix(neg_one)])
    b_ub = np.zeros(2 * n - 1)

    # Equality Σ b = 1
    A_eq = scipy.sparse.csr_matrix(np.hstack((np.ones((1, n)), np.zeros((1, 1)))))
    b_eq = np.array([1.0])

    # Objective: minimise M
    c = np.concatenate((np.zeros(n), np.array([1.0])))

    bounds = [(0, None)] * n + [(0, None)]  # b_i ≥ 0, M ≥ 0

    # Solve with HiGHS (fast for moderate sizes)
    res = scipy.optimize.linprog(c=c,
                                 A_ub=A_ub,
                                 b_ub=b_ub,
                                 A_eq=A_eq,
                                 b_eq=b_eq,
                                 bounds=bounds,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        return None, None
    b_opt = res.x[:n]
    M_opt = res.x[-1]
    return b_opt, M_opt


def lp_mixing_phase(f_start: np.ndarray,
                    best_score_start: float,
                    fft_len_conv: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve the LP described above and mix the current distribution
    with the LP solution.  The best-found distribution is returned.
    """
    f = f_start.copy()
    best_score = best_score_start
    n = f.shape[0]

    # mix steps to try (convex combination weights)
    mix_weights = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

    while time.time() < deadline:
        b_opt, M_opt = solve_lp_min_max(f)
        if b_opt is None:
            break

        # try a few convex mixes
        improved = False
        for t in mix_weights:
            candidate = (1.0 - t) * f + t * b_opt
            # re‑normalise (numerical safety)
            s = candidate.sum()
            if s <= 0:
                continue
            candidate /= s
            cand_score = fast_score(candidate, fft_len_conv, n)
            if cand_score + 1e-12 < best_score:   # strict improvement
                best_score = cand_score
                f = candidate
                improved = True
                print(f"[LP] mix t={t:.2f} → score={best_score:.6f}")
                break   # accept the first improving mix

        if not improved:
            # No improvement with any mix weight – we are at a fixed point.
            break

    return f, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial distribution
    # ------------------------------------------------------------------
    max_n = 2000                     # keep the size modest for speed
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            total = a0.sum()
            a0 = a0 / total if total > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for sub‑gradient helper (unused here)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep to collect a few extra seeds
    # ------------------------------------------------------------------
    seeds = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr /= arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        seeds.append((sc, arr))

    # uniform (baseline)
    add_candidate(np.full(n, 1.0 / n))

    # linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_candidate(lin)

    # exponential families
    for rate in [0.0005, 0.001, 0.002, 0.004, 0.008]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # power‑law families
    for p in [0.3, 0.6, 0.9, 1.2, 1.8]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # edge‑heavy
    for edge_frac in [0.005, 0.01, 0.02]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0]:
        for _ in range(2):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # include the previously discovered sequence if present
    if 'height_sequence_1' in globals():
        add_candidate(np.array(globals()['height_sequence_1'], dtype=float))

    # keep the top few seeds
    seeds.sort(key=lambda t: t[0])
    top_seeds = [arr for _sc, arr in seeds[:6]]
    print(f"[sweep] best sweep score = {seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Choose a starting point (the best from the sweep)
    # ------------------------------------------------------------------
    best_a = top_seeds[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – LP mixing / fixed‑point iterations
    # ------------------------------------------------------------------
    lp_deadline = min(deadline, start_time + 0.45 * budget_s)
    best_a, best_score = lp_mixing_phase(best_a, best_score,
                                         fft_len_conv, lp_deadline)
    print(f"[LP phase] best score after mixing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (fine tuning)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        sub_deadline = min(deadline, start_time + 0.75 * budget_s)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sub_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (as much time as left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=12000)
        print(f"[hill‑climb] final fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence (plain Python list) and report the true score
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function, already imported
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
