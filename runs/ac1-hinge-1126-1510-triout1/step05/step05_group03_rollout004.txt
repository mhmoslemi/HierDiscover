def adam_optimize(a_init, n, fft_len_conv, fft_len_g, deadline, max_iter=2000, lam_factor_start=0.3, lam_factor_end=1e-8):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    while time.time() < deadline and t < max_iter:
        t += 1
        # optionally schedule lam_factor decreasing
        lam_factor = lam_factor_start * (lam_factor_end/lam_factor_start) ** (t / max_iter)
        grad = grad_full(a, lam_factor)  # shape n
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.5 / (np.sqrt(t) + 1)  # maybe constant
        # uses adam update rule:
        a_new = a - 0.1 * m_hat / (np.sqrt(v_hat) + eps)
        # project onto simplex
        a = proj_simplex(a_new)
        # evaluate fast_score
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def grad_full(a, lam_factor):
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z
    # compute gradient of smooth max w.r.t a (not outer scaling)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (2 * n - 1)]
    # full gradient = 2 * n * g_vec

"""
Hybrid optimization for the “step‑function” inequality problem.

The algorithm combines three phases:

1. **Global multi‑restart L‑BFGS** on a smooth approximation of the
   objective (log‑soft‑max parametrisation + a geometric λ‑schedule).

2. **Aggressive local improvement** – a focused “targeted” hill‑climb that
   moves mass away from the worst contributors to the current maximal
   convolution entry, followed by a classic random mass‑transfer hill‑climb.

3. **Fine‑tuning with Adam** – a projected Adam optimiser that works directly
   on the original (unsmoothed) max‑convolution objective using the exact
   sub‑gradient (derived from the smooth‑max limit).  The update is projected
   onto the probability simplex after each step.

The routine respects the 1000 s budget, progressively allocating time to the
above phases and prints intermediate scores.  The best sequence found is
returned (and stored as ``height_sequence_1`` for reuse by later runs).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of ``evaluate_sequence`` for a probability vector."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- chain rule back to a (gradient vector of length n) -----
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient of f = 2·n·smooth_max w.r.t. a is 2·n·g_vec
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    In each step the shift giving the maximal value is identified, the most
    contributing index pair is found, and a fraction of mass is moved away
    from the worst contributor to a location whose partner under the same
    shift is (almost) empty.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))  # shift with maximal overlap

        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        partner_weight = np.full(n, np.inf)
        in_range = mask
        partner_weight[in_range] = best[partner_idx[in_range]]

        # sample candidate targets
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_full(a: np.ndarray, lam_factor: float, n: int) -> np.ndarray:
    """
    Exact gradient of the (unsmoothed) objective ``2·n·max(conv)`` w.r.t.
    the probability vector ``a``.
    The gradient is approximated via the smooth‑max with a tiny λ.
    """
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (2 * n - 1)]

    # Full gradient of f = 2·n·smooth_max
    return 2.0 * n * g_vec


def adam_optimize(
    a_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    max_iter: int = 3000,
    lam_start: float = 0.3,
    lam_end: float = 1e-9,
    alpha: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the (smoothed) max‑convolution objective.
    The step size decays automatically via Adam’s moment estimates.
    """
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        # linear schedule for the smoothing factor
        lam_factor = lam_start * (lam_end / lam_start) ** (t / max_iter)

        grad = grad_full(a, lam_factor, n)          # shape (n,)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = proj_simplex(a - step)

        cur_score = fast_score(a_candidate, fft_len_conv, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a_candidate.copy()
            a = a_candidate
        else:
            # reject – keep current a
            pass

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0          # margin for final evaluation

    # ------------------------------------------------------------------
    # Problem size – reuse previous construction if possible, otherwise use a
    # larger dimension (≈8000) to give the optimiser more flexibility.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    target_n = 8000
    if prev is not None:
        prev_arr = np.asarray(prev, dtype=float)
        if prev_arr.ndim == 1 and prev_arr.size > 0:
            # embed the previous vector (or truncate) into the new size
            if prev_arr.size >= target_n:
                best_a = prev_arr[:target_n].copy()
            else:
                best_a = np.zeros(target_n, dtype=float)
                best_a[: prev_arr.size] = prev_arr
        else:
            best_a = np.full(target_n, 1.0 / target_n)
    else:
        best_a = np.full(target_n, 1.0 / target_n)

    # Normalise to the probability simplex (sum = 1, non‑negative)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(target_n, 1.0 / target_n)
    else:
        best_a = best_a / best_a.sum()

    n = best_a.size
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Global multi‑restart L‑BFGS phase
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-8, num=45)   # very fine λ‑schedule
    maxiter_per_stage = 700                         # more work per stage

    iteration = 0
    while time.time() < deadline - 8.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.015, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            center_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, center_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best solution
            y0 = best_y + rng.normal(scale=0.09, size=n)

        # ----- L‑BFGS continuation -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- Aggressive local refinement -----

            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.35,
        )
        best_y = np.log(best_a + 1e-12)

        # random hill‑climb (helps escape tiny plateaus)
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=6000,
            max_move_frac=0.22,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Fine‑tuning with projected Adam (uses the true max‑convolution gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        adam_deadline = deadline - 2.0
        best_a, best_score = adam_optimize(
            a_start=best_a,
            n=n,
            fft_len_conv=fft_len_conv,
            deadline=adam_deadline,
            max_iter=2500,

            alpha=0.12,
        )

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------

        best_a, best_score = hill_climb(

    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
