"""
Propose a low‑score step function.
The algorithm combines a fast parametric grid search (product‑power,
sum‑power, mixtures) with a continuation L‑BFGS on a smooth log‑sum‑exp
approximation of the max‑convolution, followed by a projected Adam‑style
sub‑gradient descent on the true nonsmooth objective and a final greedy
hill‑climb.  The whole search respects the 1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, sum(x)=1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                 # now max = 0
    exp_shift = np.exp(shifted / lam)      # all ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ---- propagate gradient through the convolution ----------------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simple families for a quick grid search
# ----------------------------------------------------------------------


def shape_prod_power(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-alpha}."""
    i = np.arange(n)
    a = ((i + 1) * (n - i)) ** (-alpha)
    return a / a.sum()


def shape_sum_power(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^{-alpha} + (n-i)^{-alpha}."""
    i = np.arange(n)
    a = (i + 1) ** (-alpha) + (n - i) ** (-alpha)
    return a / a.sum()


# ----------------------------------------------------------------------
# Local refinement utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    epsilon: float = 1e-8,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam‑style descent on the true objective
        f(a) = 2·n·max_i (a*a)_i .
    The projected sub‑gradient for the active index m is
        g_i = 2·a_{m‑i}   (if 0 ≤ m‑i < n, else 0).
    """
    a = a_start.copy()
    m_exp = np.zeros_like(a)
    v_exp = np.zeros_like(a)

    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam moments
        m_exp = beta1 * m_exp + (1 - beta1) * g
        v_exp = beta2 * v_exp + (1 - beta2) * (g * g)

        # bias‑correction
        m_hat = m_exp / (1 - beta1 ** it)
        v_hat = v_exp / (1 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.size == n and np.sum(tmp) > 0:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Fast parametric grid search (product / sum power families)
    # ------------------------------------------------------------------
    alphas = np.linspace(0.20, 0.80, 31)  # 31 candidates

    # product‑power family
    for alpha in alphas:
        a = shape_prod_power(alpha, n)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a, best_y = sc, a.copy(), np.log(a + 1e-12)
            print(f"[grid] prod α={alpha:.3f} → fast score = {best_score:.6f}")

    # sum‑power family
    for alpha in alphas:
        a = shape_sum_power(alpha, n)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a, best_y = sc, a.copy(), np.log(a + 1e-12)
            print(f"[grid] sum α={alpha:.3f} → fast score = {best_score:.6f}")

    # mixtures of product‑power with uniform
    for alpha in alphas:
        a_prod = shape_prod_power(alpha, n)
        for lam in np.linspace(0.0, 1.0, 6):          # 0, .2, … ,1
            a_mix = lam * a_prod + (1.0 - lam) * (np.full(n, 1.0 / n))
            a_mix /= a_mix.sum()
            sc = fast_score(a_mix, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a, best_y = sc, a_mix.copy(), np.log(a_mix + 1e-12)
                print(f"[grid] mix α={alpha:.3f} λ={lam:.2f} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Continuation L‑BFGS on the best candidate found so far
    # ------------------------------------------------------------------
    # a relatively long smoothing schedule
    lam_schedule = list(np.geomspace(0.5, 1e-9, num=80))
    y_opt = schedule_optimize(
        best_y,
        n,
        fft_len_conv,
        fft_len_g,
        lam_schedule,
        maxiter_per_stage=800,
    )
    a_opt = softmax(y_opt)
    sc_opt = fast_score(a_opt, fft_len_conv, n)
    if sc_opt < best_score:
        best_score, best_a, best_y = sc_opt, a_opt.copy(), y_opt.copy()
        print(f"[L‑BFGS] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2B) Small random perturbations + short L‑BFGS refinements
    # ------------------------------------------------------------------
    # we allow a few quick restarts to escape shallow local minima
    restarts = 5
    for r in range(restarts):
        if time.time() >= deadline - 150.0:
            break
        y0 = best_y + rng.normal(scale=0.02, size=n)
        y_ref = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule[-30:],          # only the finest part
            maxiter_per_stage=300,
        )
        a_ref = softmax(y_ref)
        sc_ref = fast_score(a_ref, fft_len_conv, n)
        if sc_ref < best_score:
            best_score, best_a, best_y = sc_ref, a_ref.copy(), y_ref.copy()
            print(f"[restart {r}] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected Adam descent on the true nonsmooth objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 100.0:
        adam_deadline = deadline - 80.0
        print("[Adam] start")
        best_a, best_score = adam_descent(
            best_a,
            fft_len_conv,
            n,
            adam_deadline,
            lr=0.05,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[Adam] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final aggressive hill‑climb (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final HC] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final HC] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
