"""
Hybrid step‑function optimizer.

The search combines several complementary phases:

1.  A continuation L‑BFGS on a smooth approximation of the
    max‑convolution (soft‑max).  This quickly moves the vector into a
    promising region of the search space.

2.  Exact projected sub‑gradient descent on the true
    objective  f(a)=2·n·max(a∗a).  The sub‑gradient is the active
    convolution peak, and the iterates are kept on the probability
    simplex.

3.  Targeted greedy hill‑climb that moves mass from the largest
    contributor to the smallest contributor of the current dominant
    lag.

4.  A short random hill‑climb for polishing.

5.  A linear‑programming refinement step.  With the current vector
    *f* fixed, the constraints  f∗g ≤ M (where *M* is the current
    maximum convolution) are linear in *g*.  We maximise ∑gᵢ under
    those constraints, normalise the result and take a convex
    combination (1‑t)·f + t·g.  Repeating this step shrinks *M*
    further.

All phases are run repeatedly until the allocated budget (≈1000 s)
is exhausted.  The best sequence found is finally evaluated with the
provided ``evaluate_sequence`` routine.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i ≥ 0)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (l‑BFGS stage)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                      # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max
    w = exp_shift / Z  # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).

    Returns the best vector seen and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of the (non‑smooth) max at the active index k
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)

        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional perturbation to escape flat plateaus
        if (t + 1) % 250 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.008, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the largest contributor to the
    smallest contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def lp_refine_step(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    mix_t: float = 0.18,
    act_rel: float = 0.95,
) -> tuple[np.ndarray, float]:
    """
    One LP refinement step.

    Solve: maximise Σ g_i  subject to g≥0 and (a ∗ g)[k] ≤ M for a subset of
    active lags k where the current convolution peak M is almost attained.
    The resulting g is normalised to sum 1 and mixed with the current a.
    """
    if time.time() > deadline:
        return a, fast_score(a, fft_len, n)

    conv = conv_fft(a, fft_len, n)
    M = conv.max()

    # pick active constraints (those close to the peak)
    active = np.where(conv >= M * act_rel)[0]
    if active.size == 0:
        return a, fast_score(a, fft_len, n)

    # Build sparse matrix A (rows = active lags, cols = g_i)
    rows = []
    cols = []
    data = []
    for row_idx, k in enumerate(active):
        i_low = max(0, k - (n - 1))
        i_high = min(k, n - 1)
        # coefficient for g_{k-i} is a_i
        for i in range(i_low, i_high + 1):
            j = k - i
            rows.append(row_idx)
            cols.append(j)
            data.append(a[i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(len(active), n))
    b_ub = np.full(len(active), M, dtype=float)

    # linear program: minimise -sum(g)  (i.e. maximise sum(g))
    c = -np.ones(n, dtype=float)
    bounds = [(0, None)] * n

    # use HiGHS – fast for medium‑size LPs
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return a, fast_score(a, fft_len, n)

    g = res.x
    if g.sum() <= 0:
        return a, fast_score(a, fft_len, n)

    g_norm = g / g.sum()
    new_a = (1.0 - mix_t) * a + mix_t * g_norm
    new_a = project_simplex(new_a)  # keep exact sum‑1 safety
    new_score = fast_score(new_a, fft_len, n)

    # only accept if we made progress
    if new_score < fast_score(a, fft_len, n):
        return new_a, new_score
    else:
        return a, fast_score(a, fft_len, n)


# ----------------------------------------------------------------------
# Main optimisation driver for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length n within the allotted time slice.
    Returns the best vector (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ---------- initialise ----------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # continuation schedule for L‑BFGS
    lam_factors = np.geomspace(0.5, 5e-6, num=25)

    iteration = 0
    # ------------------------------------------------------------------
    # Main optimization loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb current best
            y0 = best_y + rng.normal(scale=0.06, size=n)

        # 1) L‑BFGS on smoothed objective
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_score:.6f}")

        # 2) Exact sub‑gradient descent
        sub_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = subgrad_descent(
            best_a,
            n,
            fft_len,
            max_iters=2500,
            deadline=sub_deadline,
            eta0=0.36,
        )
        best_y = np.log(best_a + 1e-12)

        # 3) Targeted hill‑climb
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=3000,
            frac=0.08,
        )
        best_y = np.log(best_a + 1e-12)

        # 4) Linear‑programming refinement (quick)
        lp_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = lp_refine_step(
            best_a,
            n,
            fft_len,
            lp_deadline,
            mix_t=0.15,
            act_rel=0.94,
        )
        best_y = np.log(best_a + 1e-12)

        # 5) Random hill‑climb (polish)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Deep final refinement – more L‑BFGS & LP passes
    # ------------------------------------------------------------------
    while time.time() < deadline - 2.0:
        # final low‑temperature L‑BFGS passes
        for lam in lam_factors[-8:]:  # the tightest smoothing levels
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 600, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)

        # an extra LP pass after the L‑BFGS sweep
        if time.time() + 0.6 < deadline:
            best_a, best_score = lp_refine_step(
                best_a, n, fft_len, deadline, mix_t=0.12, act_rel=0.93
            )
            best_y = np.log(best_a + 1e-12)

    return best_a, best_score


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its sum will be 1 after
    internal normalisation).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Focus the whole budget on a single, reasonably large size.
    # Increasing n beyond 4000 gives diminishing returns for the same
    # time budget, so we stay at n=4000.
    # ------------------------------------------------------------------
    n = 4000
    remaining = total_deadline - time.time()
    if remaining < 5.0:
        raise RuntimeError("Insufficient time budget.")

    print(f"[main] Optimising length n={n} with full budget ≈{remaining:.1f}s")
    best_a, best_score = optimise_one_size(
        n=n,
        budget_s=remaining,
        seed=seed,
        prev_seq=None,
    )
    print(f"[main] Finished optimisation, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore  # evaluate_sequence is pre‑imported
    print(f"[final] evaluate_sequence score = {final_val:.6f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
