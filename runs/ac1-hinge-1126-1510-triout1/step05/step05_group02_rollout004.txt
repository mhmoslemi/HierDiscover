"""
Propose a low‑score non‑negative step sequence.

The search proceeds in several stages:

  * Phase 0 – broad random sampling using many families of
    simplex‑distributed candidates (Dirichlet, power‑law,
    exponential, sine, etc.). The best sample seeds later phases.

  * Phase 1 – a continuation L‑BFGS on a smooth log‑sum‑exp
    approximation of the max‑convolution.  Several restarts are
    performed; each restart runs a short schedule of decreasing
    smoothing parameters.

  * Phase 2 – exponentiated‑gradient (mirror) descent on the exact
    max‑convolution (sub‑gradient of the L∞‑norm).

  * Phase 3 – a directed “max‑descent’’ that repeatedly moves a
    small fraction of mass from the index most responsible for the
    current maximal convolution entry to an index that does not
    affect that entry.  This aggressively reduces the L∞‑peak.

  * Phase 4 – a final greedy hill‑climb that swaps small amounts of
    mass between two coordinates only when the move strictly
    improves the fast objective.

All phases respect the simplex (non‑negative, sum = 1) and the
overall wall‑clock budget (≈1000 s).  The best sequence found is
returned and finally evaluated with the provided ``evaluate_sequence``
function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a (full length 2n-1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in freq. domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Phase 1 – continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage
    from the result of the previous stage. Returns the best sequence
    found during this run.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            # keep current y and continue
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – mirror descent (exponentiated gradient)
# ----------------------------------------------------------------------


def exp_mirror_descent(a0: np.ndarray,
                       deadline: float,
                       fft_len_conv: int,
                       n: int):
    """
    Projected exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())

        # sub‑gradient of the max (exact, not smoothed)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]  # derivative of max w.r.t. a_i

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:  # safeguard
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – directed max‑descent (aggressive peak reduction)
# ----------------------------------------------------------------------


def max_descent(a_start: np.ndarray,
                deadline: float,
                fft_len: int,
                n: int):
    """
    Repeatedly moves a small fraction of mass from the index that
    contributes most to the current maximal convolution entry to an
    index that does not contribute.  Accepts only improving moves;
    otherwise the move is reverted.  Returns best sequence seen.
    """
    cur_a = a_start.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    best_a = cur_a.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    it = 0
    # linear decay of step size: start at 0.2, decay to 0.02 over the phase
    start_step = 0.2
    end_step = 0.02

    while time.time() < deadline:
        # update step size
        elapsed = time.time() - (deadline - (deadline - start_step))  # dummy
        # Simple linear decay based on iteration count
        frac = it / max(1, (deadline - time.time()))  # dummy, later replaced
        # Use a simple schedule:
        tau = max(0.0, min(1.0, (deadline - time.time()) / (deadline - (deadline - (deadline - start_step)))))
        # Actually we just compute a linearly decreasing step:
        step = start_step - (start_step - end_step) * (it / max(1, (deadline - time.time()) + 1e-9))

        # compute current convolution and its maximal index
        conv = full_conv(cur_a, fft_len)
        k_max = int(conv.argmax())

        # derivative of the maximal entry w.r.t. each a_i:
        # g_i = 2 * a_{k_max - i} for indices where k_max-i in [0,n-1], else 0
        g = np.zeros_like(cur_a)
        lo = max(0, k_max - (n - 1))
        hi = min(k_max, n - 1)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = k_max - i_idx
        g[i_idx] = 2.0 * cur_a[j_idx]

        # Choose donor (largest g) and receiver (smallest g)
        # Prefer indices inside the valid range; for others g is zero.
        donor = i_idx[np.argmax(g[i_idx])]
        receiver = i_idx[np.argmin(g[i_idx])]

        # If donor has no mass, pick a random donor with positive mass
        if cur_a[donor] <= 1e-15:
            pos = np.where(cur_a > 1e-12)[0]
            if pos.size == 0:
                break
            donor = rng.choice(pos)

        if donor == receiver:
            # pick a random different receiver
            cand = rng.integers(0, n)
            while cand == donor:
                cand = rng.integers(0, n)
            receiver = cand

        # transfer a fraction of the donor's mass
        delta = cur_a[donor] * step
        if delta <= 0.0:
            break

        cur_a[donor] -= delta
        cur_a[receiver] += delta

        # Evaluate new score
        new_score = fast_score(cur_a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
        else:
            # revert move
            cur_a[donor] += delta
            cur_a[receiver] -= delta

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 4 – final greedy hill‑climb (strict improvement)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Random initialiser (many families)
# ----------------------------------------------------------------------


def random_initial_extended(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex using a mixture of
    families to increase diversity.
    """
    mode = rng.choice(['uniform', 'dirichlet', 'power_decr', 'power_incr',
                       'exp_decr', 'exp_incr', 'quadratic', 'logistic',
                       'sine', 'halfsine', 'bimodal', 'edge'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'dirichlet':
        alpha = rng.uniform(0.05, 5.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a /= a.sum()
    elif mode == 'power_decr':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a /= a.sum()
    elif mode == 'power_incr':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (p))
        a /= a.sum()
    elif mode == 'exp_decr':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n, dtype=float))
        a /= a.sum()
    elif mode == 'exp_incr':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(rate * np.arange(n, dtype=float))
        a /= a.sum()
    elif mode == 'quadratic':
        x = np.linspace(-1, 1, n)
        a = 1.0 - x**2  # hump shape (largest in the centre)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 'logistic':
        x = np.linspace(-6, 6, n)
        a = 1.0 / (1.0 + np.exp(-x))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 'halfsine':
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
        a /= a.sum()
    else:  # edge
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000                         # we try the full budget size
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient‑related convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    overall_deadline = start_time + budget_s - 0.5   # safety margin

    # Phase allocations (fractions of total time)
    phase0_end = start_time + 0.10 * budget_s   # random sampling
    phase1_end = start_time + 0.60 * budget_s   # L‑BFGS restarts
    phase2_end = start_time + 0.80 * budget_s   # mirror descent
    phase3_end = start_time + 0.95 * budget_s   # directed max‑descent
    # remaining time for final hill‑climb

    # ------------------------------------------------------------------
    # Phase 0 – broad random sampling
    # ------------------------------------------------------------------
    samples = 0
    while time.time() < phase0_end:
        a_cand = random_initial_extended(n, rng)
        score = fast_score(a_cand, fft_len_conv, n)
        samples += 1
        if score < best_score:
            best_score = score
            best_a = a_cand.copy()
            print(f"[sample #{samples}] new best fast_score={best_score:.6f}")
    print(f"[phase0] completed {samples} samples, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (many restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.25, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]
    maxiter_per_lam = 300                 # relatively cheap per restart
    while time.time() < phase1_end:
        a_start = random_initial_extended(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        md_deadline = phase2_end - 0.5
        best_a, best_score = exp_mirror_descent(best_a,
                                                md_deadline,
                                                fft_len_conv,
                                                n)
        print(f"[exp‑mirror] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – aggressive max‑descent
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        md_deadline = phase3_end - 0.5
        best_a, best_score = max_descent(best_a,
                                         md_deadline,
                                         fft_len_conv,
                                         n)
        print(f"[max‑descent] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = overall_deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    print(f"[final hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided secure wrapper
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
