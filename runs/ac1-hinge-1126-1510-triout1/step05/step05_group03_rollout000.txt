"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm Overview
------------------
1. **Initialisation** – Load a previous construction if available,
   otherwise start from a uniform distribution of length `n = 5000`.

2. **Global optimisation** – Repeated random restarts of a smooth‑max
   surrogate minimisation using L‑BFGS.  A geometric λ‑schedule
   (λ is the smooth‑max temperature) is applied, and the parameters are
   kept in the log‑soft‑max space so that the simplex constraint is
   satisfied automatically.

3. **Local refinement** – Two hill‑climbs:
   * *Targeted* – moves mass away from the pair of indices that
     contributes most to the current maximal convolution entry.
   * *Random* – classic mass‑transfer hill‑climb.

4. **Projected sub‑gradient descent** – Directly minimises the true
   objective `max(conv(a,a))` on the simplex by repeatedly moving
   opposite to the sub‑gradient of the maximal convolution entry and
   projecting back onto the probability simplex.

5. **Final polishing** – a last random hill‑climb with any remaining
   time.

The whole search stops before the given time budget (default 1000 s)
and returns the best sequence found.  The final sequence is stored in
the global variable ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog, OptimizeResult
from scipy import sparse

# ----------------------------------------------------------------------
# Utility functions (pure, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast surrogate of ``evaluate_sequence``:
    2·n·max(conv(a,a))  where the convolution is computed via FFT.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained log‑soft‑max parameters `y`.
    """
    a = softmax(y)

    # --- convolution a*a (full) ---
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # --- smooth max ---
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f_val = 2.0 * n * smooth_max

    # --- gradient of smooth max w.r.t. conv ---
    w = exp_shift / Z

    # --- propagate gradient back to a ---
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient of f wrt a
    grad_a = 2.0 * n * g_vec

    # chain rule through soft‑max: ∂a/∂y = diag(a) - a aᵀ
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f_val), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 500,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that reduces the current maximal convolution entry.
    In each step the shift giving the maximal overlap is identified,
    the most contributing index pair is found, and a fraction of mass
    is moved away from the worst contributor to a location whose partner
    under the same shift is (almost) empty.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (two FFTs)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                # shift with maximal overlap

        # contribution of each index to that shift
        idx = np.arange(n)
        partner_idx = k - idx
        mask = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner_idx[mask]]

        # choose the worst contributor (largest product)
        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # partner weight under the same shift (used to reject bad targets)
        partner_weight = np.full(n, np.inf)
        in_range = (partner_idx >= 0) & (partner_idx < n)
        partner_weight[in_range] = best[partner_idx[in_range]]

        # sample a few candidate target indices and pick the one whose partner
        # under the critical shift is smallest
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        # transfer a fraction of mass from i → j
        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def max_convolution_subgrad(
    a: np.ndarray,
    n: int,
    fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Compute the maximal auto‑convolution value and a sub‑gradient of the
    (non‑smoothed) max‑operator.
    Returns (max_value, gradient) where ``gradient`` has the same shape as ``a``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    k_max = int(np.argmax(conv))

    # map to the symmetric half so that 0 <= k <= n-1
    if k_max > n - 1:
        k = 2 * n - 2 - k_max
    else:
        k = k_max

    grad = np.zeros_like(a)
    if k == 0:
        # derivative of sum_i a_i^2 is 2*a_i
        grad = 2.0 * a
    else:
        # forward contribution
        grad[: n - k] += a[k:]
        # backward contribution
        grad[k:] += a[: n - k]
    return max_c, grad


def init_y(rng: np.random.Generator, n: int, best_a: np.ndarray, iteration: int) -> np.ndarray:
    """
    Produce a starting point `y` (log‑soft‑max parameters) for L‑BFGS.
    Various heuristics are mixed to keep diversity across restarts.
    """
    mode = rng.choice(["random", "exp", "power", "edge", "best"])
    if mode == "random":
        return rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        edge_val = np.sqrt(n / 2.0) / n
        center_val = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, center_val)
        a0[:m_edge] = edge_val
        a0[-m_edge:] = edge_val
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)
    else:  # perturb the current best solution
        return np.log(best_a + 1e-12) + rng.normal(scale=0.07, size=n)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops short of this limit.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # keep a safety margin

    # ------------------------------------------------------------------
    # Problem size and initial point
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 2000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to probability simplex (the objective is scale‑invariant)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_y = np.log(best_a + 1e-12)
    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # L‑BFGS global search (multiple random restarts)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-7, num=45)   # fine λ‑schedule
    maxiter_per_stage = 800                         # more work per stage
    iteration = 0

    while time.time() < deadline - 5.0:
        y0 = init_y(rng, n, best_a, iteration)

        y_opt = schedule_optimize(
            y_start=y0,

        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS improved fast score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Targeted hill‑climb
        # ------------------------------------------------------------------
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1500,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ------------------------------------------------------------------
        # Random hill‑climb (helps to escape tiny plateaus)
        # ------------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=5000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    step0 = 0.45
    subgrad_iter = 0

        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - eps:

            # adapt step size slowly if we are making progress

            n=n,
            deadline=deadline,

    # Official evaluation and return

    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
