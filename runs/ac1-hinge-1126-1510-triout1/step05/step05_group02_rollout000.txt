def hill_climb_exact(a, best_score, n, fft_len, deadline):
    best = a.copy()
    best_score = best_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = full_conv(best, fft_len)
        k = int(conv.argmax())
        # compute set of indices i that contribute to max
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i_idx = np.arange(lo, hi+1)
        j_idx = k - i_idx
        # gradient for each i: 2 * best[j_idx]
        grad = np.zeros_like(best)
        grad[i_idx] = 2*best[j_idx]
        # pick i with largest grad among those with positive a[i]
        # pick j random with low grad
        candidates = np.where(best > 1e-12)[0]
        if candidates.size == 0: break
        i = candidates[np.argmax(grad[candidates])]
        # choose j as index with minimal grad (but not same)
        j = np.argmin(grad)
        if i == j:
            # pick random j
            j = rng.integers(0, n)
            if j == i: continue

        delta = best[i] * rng.random() * 0.02  # move at most 2%
        if delta <= 0:
            continue
        # Apply move
        best[i] -= delta
        best[j] += delta
        # reproject to simplex? sum remains same.
        # Evaluate
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-9: # require improvement
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score
```

"""
propose_candidate
=================

Search for a non‑negative step sequence that minimizes
    2·n·max(conv(a,a))  (with sum(a)=1)

Algorithm
---------

1. **Random initialisation** – a diverse family of shapes (uniform,
   exponential, power‑law, edge‑heavy, bimodal, sine, half‑sine,
   triangular, Gaussian).

2. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing factor
   (lam_factor) is decreased in a continuation schedule; each stage
   runs a limited L‑BFGS optimisation on the simplex (via a softmax
   re‑parameterisation).  Many restarts are performed until ≈55 % of
   the total budget is spent.

3. **Linear‑program refinement** – given the current distribution `a`,
   solve the LP
        maximise   1ᵀg   subject to   T(a)·g ≤ M ,  g ≥ 0
   where `M = max(conv(a,a))` and `T(a)` is the convolution matrix
   (linear map `g ↦ conv(a,g)`).  The LP finds a distribution `g`
   that spreads the mass as much as possible without increasing the
   peak.  We replace `a` by the average `(a + ĝ)/2` where `ĝ` is the
   normalised LP solution.  The LP is run a few times (as time permits).

4. **Greedy exact‑subgradient hill‑climb** – repeatedly move a small
   fraction of mass from a coordinate that contributes most to the
   current maximal convolution entry towards a coordinate that contributes
   least.  Only strictly improving moves are kept.  This fine‑tunes the
   solution with many inexpensive iterations.

5. **Final evaluation** – the best sequence is returned; its exact score
   is obtained with the provided `evaluate_sequence` function.

The code respects the 1000 s time budget, uses only NumPy, SciPy and the
standard library, and prints intermediate progress.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed maximum based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in freq. domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    # initialise logits
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a random distribution on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sine', 'halfsine', 'triangular', 'gaussian'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    elif mode == 'halfsine':
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
    elif mode == 'triangular':
        a = np.concatenate([np.linspace(0, 1, n // 2, endpoint=False),
                            np.linspace(1, 0, n - n // 2)])
    else:  # gaussian
        sigma = rng.uniform(0.05, 0.2) * n
        centre = n / 2
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

def build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix T(a) such that T(a)·g = conv(a,g).
    The matrix is stored as a CSR sparse matrix.
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    # each output index i (0 … 2n‑2) receives contributions from a[j]·g[i‑j] with valid j
    for i in range(2 * n - 1):
        lo = max(0, i - (n - 1))
        hi = min(i, n - 1)
        # j runs from lo to hi, and the coefficient for g[j] is a[i‑j]
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            data.append(a[i - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_refinement(a: np.ndarray, deadline: float, n: int,
                  fft_len_conv: int) -> (np.ndarray, float):
    """
    Perform a few LP‑based refinements.
    Returns the best sequence and its fast_score.
    """
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    while time.time() < deadline:
        # current peak
        conv = full_conv(best_a, fft_len_conv)
        M = conv.max()

        # build convolution matrix for the current a
        A = build_conv_matrix(best_a)

        # Solve LP: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        # Use the high‑performance HiGHS solver (default in SciPy >=1.6)
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        g_norm = g0 / sum_g0  # now sum(g_norm)=1

        # average the current a and the LP solution
        a_new = 0.5 * (best_a + g_norm)

        # evaluate
        new_score = fast_score(a_new, fft_len_conv, n)

        # accept only if improvement
        if new_score < best_score - 1e-12:
            best_a = a_new
            best_score = new_score
            # continue refining with the improved a
            continue
        else:
            # no improvement – stop LP refinement
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Exact‑subgradient greedy hill‑climb
# ----------------------------------------------------------------------


def exact_hill_climb(a: np.ndarray, current_score: float,
                     fft_len: int, n: int,
                     deadline: float, max_iter: int = 20000) -> (np.ndarray, float):
    """
    Greedy mass‑transfer hill‑climb that uses the exact sub‑gradient
    of the max‑convolution.  Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = full_conv(best, fft_len)
        k = int(conv.argmax())                     # index of current maximal entry

        # indices contributing to entry k
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_idx = np.arange(lo, hi + 1)
        j_idx = k - i_idx

        # sub‑gradient (d max / d a_i) = 2 * a_j for i in i_idx
        grad = np.zeros_like(best)
        grad[i_idx] = 2.0 * best[j_idx]

        # select source i (large gradient, positive mass) and target j (small gradient)
        pos_indices = np.where(best > 1e-12)[0]
        if pos_indices.size == 0:
            break
        # source: among positions with positive mass, pick the one with largest gradient
        i = pos_indices[np.argmax(grad[pos_indices])]
        # target: pick index with smallest gradient (could be the same)
        j = np.argmin(grad)
        if i == j:
            # fallback: random different target
            j = rng.integers(0, n)
            if j == i:
                it += 1
                continue

        # amount to move – a small fraction of source mass
        delta = best[i] * rng.random() * 0.02   # up to 2 %
        if delta <= 0.0:
            it += 1
            continue

        # propose move
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score - 1e-12:
            best = cand
            best_score = new_score
        # else reject

        it += 1

    return best, best_score

    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
