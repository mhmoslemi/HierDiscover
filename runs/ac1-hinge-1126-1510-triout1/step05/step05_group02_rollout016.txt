def subgradient_descent(a0, deadline):

    best = a.copy()
    best_score = fast_score(a)
    it=0
    while time.time() < deadline:
        conv = np.convolve(a, a)
        k = np.argmax(conv)
        # compute gradient
        g = np.zeros_like(a)
        lo = max(0, k - (len(a)-1))
        hi = min(k, len(a)-1)
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] = 2.0 * a[j]
        # step size schedule
        eta = 0.05 / np.sqrt(it+1)  # maybe adjust

        # project onto simplex
        a_new = simplex_projection(a_new)
        # evaluate
        score = fast_score(a_new)
        if score < best_score:
            best_score = score
            best = a_new.copy()
        a = a_new
        it+=1
    return best, best_score

"""
Propose a low‑score non‑negative step sequence.

Algorithm outline (≈1000 s budget):
  1. Initialise a with the best known construction (global ``height_sequence_1``)
     or a random simplex sample.
  2. Phase 1 – many cheap L‑BFGS restarts on a smooth log‑sum‑exp
     approximation of the max‑convolution to obtain a good baseline.
  3. Phase 2 – exponentiated‑gradient (mirror descent) on the exact max‑convolution.
  4. Phase 3 – massive mass‑transfer simulated‑annealing.
     Moves consist of transferring a random fraction of the mass from one
     coordinate to another; the convolution is updated *incrementally* in O(n)
     using slice operations, so millions of moves can be tried.
  5. Phase 4 – short greedy hill‑climb (strictly improving swaps).
  6. Return the best sequence found (as a list of floats).  The final score
     is printed via ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) for a distribution already
    normalised to sum = 1.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sine",
            "halfsine",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sine":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    else:  # halfsine
        a = np.sin(np.pi * np.arange(n) / (n - 1))

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,

    Runs cheap L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    # initialise logits
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
    return best_a, best_score


def exp_mirror_descent(a0: np.ndarray,
                       deadline: float,
                       n: int):
    """
    Projected exponentiated‑gradient descent on the exact max‑convolution.
    Returns best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a)
    it = 0
    eta0 = 1.0

    while time.time() < deadline:
        conv = np.convolve(a, a)
        k = int(conv.argmax())

        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        sc = fast_score(a)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Massive mass‑transfer simulated annealing (incremental updates)
# ----------------------------------------------------------------------


def mass_transfer_anneal(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        rng: np.random.Generator):
    """
    Simulated‑annealing where a random fraction of mass is moved from one
    coordinate to another.  Convolution is updated incrementally in O(n).
    Returns the best sequence and its max‑convolution value.
    """
    a = a_start.copy()
    conv = np.convolve(a, a)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    # temperature proportional to current max
    T = cur_max * 0.2
    cooling = 0.9995
    min_T = cur_max * 1e-6
    it = 0

    while time.time() < deadline:
        # pick source with non‑zero mass
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        # random fraction of source (up to 30 %)
        max_frac = 0.30
        delta = a[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # Incremental update of the convolution
        conv_new = conv.copy()
        # linear part (2 δ · a shifted)
        conv_new[j:j + n] += 2.0 * delta * a
        conv_new[i:i + n] -= 2.0 * delta * a
        # quadratic part
        idx_2j = 2 * j
        idx_2i = 2 * i
        idx_ij = i + j
        conv_new[idx_2j] += delta * delta
        conv_new[idx_2i] += delta * delta
        conv_new[idx_ij] -= 2.0 * delta * delta

        new_max = conv_new.max()

        # acceptance test
        if new_max < cur_max:
            accept = True
        else:
            prob = np.exp(-(new_max - cur_max) / T)
            accept = rng.random() < prob

        if accept:
            # apply move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_a = a.copy()

        # cooling schedule
        if it % 1000 == 0:
            T = max(min_T, T * cooling)
        it += 1

    return best_a, best_max


def hill_climb(a: np.ndarray,
               cur_max: float,
               deadline: float,
               n: int,
               rng: np.random.Generator,
               max_iter: int = 5000):
    """
    Greedy hill‑climb: move a small random fraction of mass from one index
    to another, accepting only strictly improving moves.
    """
    best_a = a.copy()
    best_max = cur_max
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j or best_a[i] <= 0.0:
            it += 1
            continue
        delta = best_a[i] * rng.random() * 0.03  # up to 3 % of source
        if delta <= 0.0:
            it += 1
            continue

        # incremental update (same as in anneal)
        conv = np.convolve(best_a, best_a)
        conv[i:i + n] -= 2.0 * delta * best_a
        conv[j:j + n] += 2.0 * delta * best_a
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta
        new_max = conv.max()
        if new_max < best_max:
            best_a[i] -= delta
            best_a[j] += delta
            best_max = new_max
        it += 1
    return best_a, best_max

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and warm‑start
    # ------------------------------------------------------------------
    max_n = 4000
    if "height_sequence_1" in globals():
        init_seq = np.array(globals()["height_sequence_1"], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n // 2)  # keep it modest
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n // 2
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n // 2
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Timing split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_end = start_time + 0.20 * budget_s  # L‑BFGS restarts
    phase2_end = start_time + 0.45 * budget_s  # mirror‑descent
    phase3_end = deadline - 0.25               # massive annealing
    # ------------------------------------------------------------------
    # Phase 1 – cheap L‑BFGS restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001]
    maxiter_per_lam = 300  # few iterations → many restarts
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len_conv,
                                           fft_len_g,
                                           n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        md_deadline = phase2_end - 0.2
        best_a, best_score = exp_mirror_descent(best_a,
                                                md_deadline,
                                                n)
        print(f"[exp‑mirror] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – massive mass‑transfer simulated annealing
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        anneal_deadline = phase3_end - 0.2
        best_a, best_max = mass_transfer_anneal(best_a,
                                                anneal_deadline,
                                                n,
                                                rng)
        best_score = 2.0 * n * best_max
        print(f"[mass‑anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        # compute current max for hill‑climb
        cur_conv = np.convolve(best_a, best_a)
        cur_max = cur_conv.max()
        best_a, best_max = hill_climb(best_a,
                                      cur_max,
                                      final_deadline,
                                      n,
                                      rng,
                                      max_iter=8000)
        best_score = 2.0 * n * best_max
        print(f"[final hill‑climb] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result (post‑process to list of floats)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # external function
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
