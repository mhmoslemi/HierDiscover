"""
Propose a low‑score non‑negative step‑function sequence.

Algorithm Overview
------------------
1.  Choose a problem size `n`.  If a previous construction (`height_sequence_1`)
    exists we reuse its length, otherwise we start from `n = 5000`.

2.  Initialise a probability vector `a` (∑a = 1).  Several randomised
    strategies (uniform, exponential, power‑law, edge‑biased and a
    perturbed copy of the current best) are used as restarts.

3.  Global optimisation with a continuation of L‑BFGS on a *smooth‑max*
    surrogate of the true objective `2·n·max(conv)`.  The smoothing
    parameter λ follows a fine geometric schedule (≈ 70 stages) and each
    stage receives many L‑BFGS iterations (≈ 1200) → a very thorough
    search of the smooth landscape.

4.  Two complementary local searches are applied after each global run:
       * **Targeted hill‑climb** – attacks the shift where the convolution
         attains its maximum and moves mass away from the worst contributing
         index.
       * **Random hill‑climb** – classic mass‑transfer moves on the simplex.

5.  A simulated‑annealing phase explores uphill moves with a slowly
    decreasing temperature to escape shallow local minima.

6.  A final projected‑gradient descent on the true (non‑smoothed) objective
    refines the solution to high precision.

All moves keep the vector on the probability simplex (non‑negative,
∑a = 1).  The expensive official ``evaluate_sequence`` is called only once,
at the very end.

The code respects the 1000 s time budget and prints intermediate progress.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Surrogate of ``evaluate_sequence`` for a probability vector.
    Returns 2·n·max(conv) (the exact score when ∑a = 1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth‑max objective and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Returns the smoothed max of the auto‑convolution and its gradient
    w.r.t. the unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max -----
    shifted = conv - max_c                # peak at 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # ----- back‑propagation to a (length n) -----
    # Convolution of a_rev with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # Gradient w.r.t. y (soft‑max Jacobian)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 1200,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the final parameter vector `y`.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y

# Local search utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer hill‑climb on the simplex.
    Only strictly improving moves are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Focused hill‑climb that attacks the shift with the current maximal
    convolution value.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution via FFT
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))

        # contributions of each index to this shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))
        if best[i] <= 0.0:
            continue

        # pick a destination whose partner under this shift is tiny
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 0.02,
    move_frac: float = 0.18,
    decay: float = 0.999,
    min_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Classic simulated‑annealing on the simplex.
    Accepts worsening moves with a temperature that decays exponentially.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    T = init_temp

    while time.time() < deadline and T > min_temp:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score

        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

        # geometric cooling
        T *= decay
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth‑max of the auto‑convolution w.r.t. a.
    Returns the gradient *before* the outer 2·n factor.
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    return 2.0 * g  # without the final 2·n factor

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds). The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 500:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise onto the simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}  fast score={best_score:.6f}")

    # --------------------------------------------------------------
    # Pre‑computed FFT lengths
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # L‑BFGS continuation schedule (global search)
    # --------------------------------------------------------------
    # start with a relatively large λ‑factor and finish with a very small one
    lam_factors = np.geomspace(0.5, 1e-8, num=70)
    maxiter_per_stage = 1200

    iteration = 0
    while time.time() < deadline - 5.0:
        # ------ random initialisation strategy ------
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            # concentrate a little on the two edges
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            middle_val = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, middle_val)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturb the current best solution
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ------ continuation L‑BFGS ------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode}  improve -> {best_score:.6f}")

        # ------ aggressive targeted hill‑climb ------
        tc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=tc_deadline,
            max_iter=1200,
            move_frac=0.30,
        )
        best_y = np.log(best_a + 1e-12)

        # ------ fallback random hill‑climb ------
        rc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=rc_deadline,
            max_iter=8000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Simulated annealing phase (exploration of uphill moves)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        sa_deadline = deadline - 1.0
        best_a, best_score = simulated_annealing(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=sa_deadline,
            init_temp=0.03,
            move_frac=0.12,
            decay=0.998,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[sa] after annealing  score={best_score:.6f}")

    # --------------------------------------------------------------
    # Projected‑gradient descent on the *true* non‑smoothed max
    # --------------------------------------------------------------
    lam_factor_pg = 5e-6
    step0 = 0.25
    max_pg_iter = 3500
    pg_iter = 0
    while time.time() < deadline - 1.0 and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)      # without outer 2·n
        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            lam_factor_pg = max(lam_factor_pg * 0.85, 1e-8)

    print(f"[pg] iterations={pg_iter}  best fast score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing hill‑climb (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=25000,
            max_move_frac=0.22,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    globals()["height_sequence_1"] = final_seq          # store for future runs
    final_val = evaluate_sequence(final_seq)           # official score
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
