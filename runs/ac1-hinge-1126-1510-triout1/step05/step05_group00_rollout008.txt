def frank_wolfe_refine(a, deadline, n, fft_len_conv):
    best_a = a.copy()
    conv_a = full_conv(a, fft_len_conv)
    best_score = fast_score(a, fft_len_conv, n)
    t = 0
    while time.time() < deadline:
        M = conv_a.max()
        eps = M * 1e-12 + 1e-14
        mask = (conv_a >= M - eps).astype(np.float64)
        # compute subgradient
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len_conv)
        fm = np.fft.fft(mask, fft_len_conv)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))

        conv_ii = np.zeros_like(conv_a)

The evaluation function  
\[
E(a)=\frac{2n\;\max_k (a*a)_k}{(\sum_i a_i)^2}
\]
is *scale‑invariant*: we may restrict ourselves to the probability simplex  
\(\sum_i a_i=1,\;a_i\ge 0\).  
The term \(\max_k (a*a)_k\) is the maximum entry of the *self‑convolution*  
\(c_k=\sum_{i+j=k}a_i a_j\).  

The algorithm below combines three complementary optimisation ideas

1. **Smoothed L‑BFGS** – a continuation method that minimises a smooth
   log‑sum‑exp surrogate of the max‑convolution.
2. **Projected sub‑gradient descent** – directly tackles the hard max‑convolution
   with a momentum term.
3. **Frank‑Wolfe refinement** – the convex sub‑gradient of the max‑convolution
   is computed, the best vertex of the simplex (the index with the smallest
   sub‑gradient component) is selected, and a line‑search is performed on the
   exact objective.  This step is extremely cheap (only a few FFTs per iteration)
   and is very effective for the final few thousandths of improvement.

All phases respect the 1000 s deadline, continually keep the best sequence
found and finally return it.

```python
"""
search for a step function with a minimal evaluation score.

Algorithm
---------
* Work on the probability simplex (sum of heights = 1).  The evaluation
  function is invariant to scaling, so this does not lose generality.

* Phase‑1 (≈55 % of the time) – continuation L‑BFGS on a smooth log‑sum‑exp
  approximation of the max‑convolution.  Random restarts and occasional
  Gaussian perturbations of the current best are used.

* Phase‑2 (≈30 % of the time) – projected sub‑gradient descent with momentum
  on the exact hard max‑convolution.

* Phase‑3 (≈20 % of the time) – Frank‑Wolfe style refinement.
  The sub‑gradient of the max‑convolution is computed, the index with the
  smallest component is selected (the optimal vertex of the simplex),
  and a line‑search (binary halving) is performed on the true objective.
  This converges very quickly to the final optimum.

* A short greedy hill‑climb is performed at the very end.

The best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when Σa = 1:
    score = 2·n·max(conv).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * w) in F domain
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Random initialisers
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Continuation L‑BFGS over the supplied smoothing schedule.
    Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with a simple momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/sqrt(it)
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def frank_wolfe_refine(a_start: np.ndarray,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Frank‑Wolfe style refinement for the max‑convolution objective.
    At each iteration we:
        - compute the sub‑gradient,
        - pick the simplex vertex (index) with smallest component,
        - perform a back‑tracking line‑search on the *exact* objective,
          moving a fraction of mass to that vertex.
    """
    a = a_start.copy()
    conv_a = full_conv(a, fft_len_conv)
    best_a = a.copy()
    best_score = 2.0 * n * conv_a.max()

    it = 0
    while time.time() < deadline:
        M = conv_a.max()
        eps = M * 1e-12 + 1e-14
        mask = (conv_a >= M - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (same as in sub‑gradient step)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        subgrad = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # vertex giving steepest descent = smallest component
        i = int(np.argmin(subgrad))

        # Pre‑compute the three convolution terms needed for the line search
        conv_ai = np.zeros_like(conv_a)
        conv_ai[i:i + n] = a                     # a * δ_i
        conv_ii = np.zeros_like(conv_a)
        idx_ii = 2 * i
        if idx_ii < conv_ii.size:
            conv_ii[idx_ii] = 1.0               # δ_i * δ_i

        # line‑search (binary halving) on γ ∈ (0,1]
        gamma = 0.5
        improved = False
        while gamma > 1e-6:
            coeff_self = (1.0 - gamma) ** 2
            coeff_cross = 2.0 * gamma * (1.0 - gamma)
            coeff_ii = gamma ** 2
            conv_candidate = (coeff_self * conv_a +
                              coeff_cross * conv_ai +
                              coeff_ii * conv_ii)
            score_cand = 2.0 * n * conv_candidate.max()
            if score_cand < best_score - 1e-12:
                improved = True
                break
            gamma *= 0.5

        if not improved:
            # No improvement with this vertex – try a random one to escape stagnation.
            i = np.random.randint(0, n)
            it += 1
            if it > 5000:  # safety break
                break
            continue

        # accept the move
        a = (1.0 - gamma) * a
        a[i] += gamma
        conv_a = conv_candidate
        best_score = score_cand
        best_a = a.copy()
        it += 1

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another. Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    # ------------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
```
