"""
Propose a low‑evaluation step function.

Algorithm
---------
1.  Build a pool of cheap parametric seeds (uniform, linear, exponential,
    power‑law, sinusoid, logistic, random Dirichlet, etc.) and keep the
    best few after normalising each to the simplex.
2.  For each seed run a continuation L‑BFGS optimiser on a smooth
    log‑sum‑exp surrogate of   max(conv(a,a)).  The variables are the
    un‑normalised log‑weights; the soft‑max guarantees a simplex point.
    After each λ‑stage a short hill‑climb is performed.
3.  Apply a classic “M‑balancing’’ LP step, now solved with SciPy’s
    high‑performance HiGHS LP solver (instead of cvxpy/ECOS).  The LP
    maximises Σb   subject to   a∗b ≤ M, b≥0, where M is the current
    peak of a∗a.  The solution b is mixed back into a with a tunable
    factor t and the process repeats while time permits.
4.  Refine the result with projected sub‑gradient descent on the exact
    max‑convolution (the sub‑gradient is obtained by convolving the mask
    of active peak positions with the reversed vector).
5.  Finish with an extensive greedy hill‑climb that repeatedly transfers
    a tiny random mass between two coordinates, accepting only strictly
    improving moves.

All phases respect the simplex constraint (∑a = 1, a_i ≥ 0).  The
budget (default 1000 s) is split automatically between the phases.
The best sequence found before the deadline is returned as a plain
list of non‑negative floats.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i

    # gradient wrt a : convolution of w with reversed a
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple λ values)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.
    Returns the best simplex vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # tiny hill‑climb after each λ to escape small plateaus
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Direct minimisation of 2·n·max(a*a) via sub‑gradient descent.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple diminishing‑step rule
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Random mass‑transfer hill‑climb: pick two indices, move a tiny
    random fraction from one to the other and keep the move only if it
    strictly improves the score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # move at most 5 % of the source mass
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# LP “M‑balancing’’ step (SciPy HiGHS)
# ----------------------------------------------------------------------
def lp_balancing(a: np.ndarray,
                 n: int,
                 fft_len_conv: int,
                 deadline: float,
                 t_init: float = 0.15,
                 max_iters: int = 15):
    """
    Perform a few iterations of the linear‑programming balancing step:

        maximise   Σ b_j
        s.t.       b_j ≥ 0,
                   (a * b)_k ≤ M   ∀k,

    where M = max(a*a).  The solution b is mixed back into a with a
    shrinking mixing factor t.
    """
    a = a.copy()
    t = t_init

    while time.time() < deadline and max_iters > 0:
        # current peak
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        M = conv.max()

        # Build convolution matrix T (size (2n‑1) × n)
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear programme: maximise Σ b  <=> minimise –Σ b
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M, dtype=float)
        bounds = [(0, None)] * n

        # remaining time for the solver (HiGHS accepts a time limit via options)
        remaining = max(0.0, deadline - time.time())
        opts = {"presolve": True, "time_limit": remaining}
        try:
            res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                          method='highs', options=opts)
        except Exception as e:
            print("[lp] linprog exception:", e)
            break

        if not res.success:
            # Infeasibility or timeout – abort the LP phase
            print("[lp] linprog not successful:", res.message)
            break

        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s <= 0.0:
            break
        b_scaled = b / s

        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, fft_len_conv)
        old_score = fast_score(a, n, fft_len_conv)

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init                # reset mixing factor on success
            max_iters -= 1
        else:
            t *= 0.5
            if t < 1e-5:
                break
            # keep the same iteration count – we will try again with a smaller t

    return a

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats (sum > 0, all ≥ 0).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size and base vector
    # ------------------------------------------------------------------
    max_n = 2500                     # safe size for the LP phase
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0[a0 < 0] = 0.0
        total = a0.sum()
        a0 = a0 / total if total > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (collect seeds)
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len_conv)
        candidates.append((sc, arr))

    # a few analytic families
    add_candidate(np.full(n, 1.0 / n))                              # uniform
    add_candidate(np.arange(n, 0, -1, dtype=float))                # linear decreasing
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))                # exponential families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))   # power‑law
    # sinusoid²
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)
    # logistic shape (symmetric S‑curve)
    scale = n / 6.0
    x = np.arange(n) - n / 2.0
    logistic = 1.0 / (1.0 + np.exp(x / scale))
    add_candidate(logistic)
    # random Dirichlet–like draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:8]]
    print(f"[sweep] best seed score={candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    phase1_end = start + 0.45 * budget_s      # L‑BFGS / restarts
    phase2_end = start + 0.80 * budget_s      # LP + sub‑gradient
    # remaining time → final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.9, 0.6, 0.4, 0.2, 0.1,
        0.07, 0.04, 0.02, 0.01, 0.006,
        0.003, 0.001, 0.0005, 0.0002, 0.0001
    ]
    maxiter_per_lam = 400
    restarts = 0
    while time.time() < phase1_end and restarts < 25:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-4
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – LP balancing + sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # ----- LP balancing (SciPy HiGHS) -----
        lp_deadline = min(phase2_end, deadline - 0.3)
        print("[LP] starting balancing iterations")
        best_a = lp_balancing(best_a, n, fft_len_conv, lp_deadline,
                              t_init=0.15, max_iters=12)
        best_score = fast_score(best_a, n, fft_len_conv)
        print(f"[LP] post‑LP score={best_score:.6f}")

        # ----- Sub‑gradient descent (more iterations) -----
        if time.time() < phase2_end:
            sg_deadline = min(phase2_end, deadline - 0.2)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[sub‑gradient] after phase2 score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (as much time as left)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=20000)

    print(f"[final] best internal score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence (plain list) and report the official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] official evaluated score = {final_eval:.6f}")
    return final_seq
