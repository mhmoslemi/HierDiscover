"""
Propose a step sequence with low evaluation score.

The algorithm combines several optimisation phases:

1.  Random‑restart L‑BFGS with a log‑sum‑exp smooth approximation of the
    max‑convolution.  The smoothing factor is gradually reduced (continuation)
    to first explore globally and then refine locally.

2.  Projected sub‑gradient descent with momentum followed by an Adam run,
    both operating on the simplex (soft‑max parametrisation).

3.  A linear‑programming refinement: for the current distribution `a`,
    we maximise the total mass `∑ b_i` under the constraint
    `conv(a, b) ≤ max(conv(a,a))`.  The resulting `b` (renormalised) is a
    promising new candidate.  A few convex combinations of `a` and `b`
    are also tested.

4.  A final greedy hill‑climb that moves a small fraction of mass between
    two coordinates only when the move strictly improves the objective.

All heavy convolution work uses FFT (O(n log n)).  The time budget (default
1000 s) is split automatically; the best sequence found before the deadline
is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


# ----------------------------------------------------------------------
# Basic utilities (FFT helpers, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z    # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in F domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Very rare; just skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with a simple momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/sqrt(it)
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another.  Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

# Linear‑programming refinement (optional)
# ----------------------------------------------------------------------


def lp_refine_candidate(a: np.ndarray,
                        fft_len: int,
                        n: int,
                        lp_deadline: float,
                        top_k: int = 800):
    """
    Solve a linear programme that maximises the L1‑mass of a new vector `b`
    under the constraint ``conv(a, b) ≤ max(conv(a,a))``.
    Only the `top_k` most critical convolution rows are kept as constraints.
    Returns a (possibly) improved normalised vector and its fast_score.
    """
    # current convolution and its maximum
    conv_cur = full_conv(a, fft_len)
    M_cur = conv_cur.max()

    # indices of the most binding constraints
    k_sel = min(top_k, len(conv_cur))
    idx = np.argpartition(conv_cur, -k_sel)[-k_sel:]

    # Build a sparse matrix for the selected rows
    rows = []
    cols = []
    data = []
    for row_idx, kk in enumerate(idx):
        i_start = max(0, kk - (n - 1))
        i_end = min(n - 1, kk)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        d = kk - i_range
        vals = a[d]                     # a[d] = f_{kk-i}
        rows.extend([row_idx] * len(i_range))
        cols.extend(i_range.tolist())
        data.extend(vals.tolist())
    A_sub = csr_matrix((data, (rows, cols)), shape=(len(idx), n))

    # Linear programme: maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)
    b_ub = np.full(A_sub.shape[0], M_cur)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception as e:
        print("[LP] exception:", e)
        return a, fast_score(a, fft_len, n)

    if not res.success:
        # LP failed – return original
        return a, fast_score(a, fft_len, n)

    b = res.x
    if b.sum() == 0:
        return a, fast_score(a, fft_len, n)

    # Renormalise to sum = 1
    b_norm = b / b.sum()
    cand_score = fast_score(b_norm, fft_len, n)

    # Also test convex combinations of a and b_norm
    best_vec = a
    best_score = fast_score(a, fft_len, n)

    if cand_score < best_score:
        best_vec = b_norm
        best_score = cand_score

    for t in (0.3, 0.5, 0.7):
        mix = (1 - t) * a + t * b_norm
        mix = mix / mix.sum()
        sc = fast_score(mix, fft_len, n)
        if sc < best_score:
            best_vec = mix
            best_score = sc
            break

    return best_vec, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (fixed to 4000 – a good trade‑off)
    # ------------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        n = max_n
        old_n = init_seq.size
        if old_n != n:
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, init_seq)
        else:
            a0 = init_seq.copy()
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.80 * budget_s   # sub‑gradient + Adam
    phase3_end = start_time + 0.90 * budget_s   # LP refinement
    # remaining time for hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many random‑restart continuation L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = [0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005,
                    0.0003, 0.0002, 0.0001]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
        if best_score < np.inf and rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.02)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent + Adam fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    if time.time() < phase2_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           phase2_end,
                                           lam_factor=0.0015,
                                           lr=0.12)
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        # perform a handful of LP refinements (each cheap)
        lp_iters = 5
        for it_lp in range(lp_iters):
            if time.time() >= phase3_end:
                break
            new_a, new_score = lp_refine_candidate(best_a,
                                                   fft_len_conv,
                                                   n,
                                                   phase3_end,
                                                   top_k=800)
            if new_score < best_score:
                best_score = new_score
                best_a = new_a.copy()
                print(f"[LP] improved score={best_score:.6f} (iteration {it_lp+1})")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final evaluation using the provided external function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
