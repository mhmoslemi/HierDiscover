def targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_iter=5000, move_frac=0.2):

    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter>0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))
        # indices that contribute to conv[m]
        idx_low = max(0, m - (n - 1))
        idx_high = min(n - 1, m)
        # list of i where i contributes
        contrib_i = np.arange(idx_low, idx_high+1)
        # choose i where best[i] > 0
        # pick i with probability proportional to best[i] to move mass out.
        weights = best[contrib_i]
        if weights.sum() == 0:
            # fallback random move
            i = rng.integers(0, n)
        else:
            i = rng.choice(contrib_i, p=weights/weights.sum())
        # choose j from complement not overlapping m (where i not in contrib)
        # we can pick random j
        j = rng.integers(0, n)
        if i == j:
            continue
        # amount to move: delta = move_frac * best[i]
        delta = best[i] * rng.random() * move_frac
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert changes
            best[i] += delta
            best[j] -= delta
    return best, best_score

def propose_candidate(seed=42, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small margin

    # Determine n
    init_seq = globals().get('height_sequence_1', None)
    default_n = 8000
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(init_seq, dtype=float).ravel()
        if arr.ndim == 1 and arr.size >= 3000:
            n = min(arr.size, default_n)
            # Use as initial best candidate if same size
            if arr.size == n:
                base_seq = arr
            else:
                # take first n entries, renormalize
                base_seq = arr[:n]
        else:
            n = default_n
            base_seq = None
    else:
        n = default_n
        base_seq = None

    fft_len = next_pow2(2*n - 1)
    # for gradient convolution we need length for g: 3*n-2 maybe.
    fft_len_g = next_pow2(3*n - 2)

    # Initialize best candidate
    if base_seq is not None:
        best_a = base_seq.astype(float)
        best_a = np.maximum(best_a, 0.0)
        total = best_a.sum()
        if total < 1e-12:
            best_a = np.full(n, 1.0/n)
        else:
            best_a = best_a / total
    else:
        best_a = np.full(n, 1.0/n)

    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # Phase 1: L-BFGS restarts
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=40))
    stages_maxiter = 500
    modes = ["random", "exp", "power", "edge", "sidon", "binary", "bump", "best"]
    # Determine phase deadline (e.g., 0.5 * budget)
    phase1_deadline = start_time + 0.55 * budget_s

    restart_count = 0
    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(y0, n, fft_len, fft_len_g, lam_schedule, maxiter_per_stage=stages_maxiter)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart {restart_count}] mode={mode} improved fast_score={best_score:.6f}")

        # Short hill climb (targeted)
        hc_deadline = min(deadline, time.time() + 0.5)  # 0.5 sec for quick refine
        best_a, best_score = targeted_hill_climb(best_a, best_score, fft_len, n,
                                                 hc_deadline,
                                                 max_iter=2000,
                                                 move_frac=0.2)
        best_y = np.log(best_a + 1e-12)

        restart_count += 1

    # Phase 2: Projected subgradient descent (long)
    sg_deadline = min(deadline - 10.0, start_time + 0.85 * budget_s)
    print("[subgrad] start")
    best_a, best_score = subgradient_descent(best_a, fft_len, n,
                                             sg_deadline,
                                             eta0=0.4,
                                             max_iters=800_000)
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] final fast_score {best_score:.6f}")

    # Final targeted hill climbing till deadline
    if time.time() < deadline:
        print("[final] targeted hill climb")
        best_a, best_score = targeted_hill_climb(best_a, best_score,

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

"""
Hybrid optimizer for the step‑function minimisation problem.

The score to be minimised is
    score(a) = 2·n·max(conv(a,a)) / (sum(a)**2) .
Because the objective is scale‑invariant we restrict ourselves to the
probability simplex (sum = 1).  The algorithm consists of three stages

1.  Multi‑restart continuation L‑BFGS on a smooth log‑sum‑exp
    approximation of the max–convolution.  Many random initialisations
    (Gaussian, exponential, power‑law, edge‑heavy, Sidon‑set, binary
    mask, Gaussian bump and perturbations of the current best) are tried.
    The smoothing factor λ is decreased geometrically (≈ 40 stages).

2.  A long projected sub‑gradient descent on the true non‑smooth
    objective.  The sub‑gradient of the maximal convolution entry is
    computed analytically and the iterate is projected onto the simplex
    after each step.  The step size η decays like η₀/√t.

3.  Targeted greedy hill‑climbing.  In every iteration the index of the
    current maximal convolution entry m is identified; mass is moved
    from indices that contribute to that entry to indices that do not.
    Moves are kept only when they improve the fast score; a small
    amount of random exploration is also performed.

The whole search respects the 1000 s budget; the best sequence found is
returned as a plain list of non‑negative floats (the harness will call
``evaluate_sequence`` on it).  The code works for any initial sequence
available as the global variable ``height_sequence_1``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed to have length n.
    """
    # forward FFT of a (zero‑padded)
    fa = np.fft.fft(a, fft_len)
    # product in Fourier domain and inverse FFT
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)   (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor   (lam_factor is the smoothing parameter).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                # max becomes zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # d(smooth_max)/d(conv)
    w = exp_shift / Z

    # derivative of conv w.r.t. a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # middle n entries correspond to ∂conv/∂a
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a sequence of optimisation stages with a
    decreasing smoothing factor lam_factor.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that preferentially moves mass away from the
    indices contributing to the currently maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # identify the current max convolution index
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # range of i such that i + j = m with 0≤i,j<n
        i_low = max(0, m - (n - 1))
        i_high = min(n - 1, m)
        contrib = np.arange(i_low, i_high + 1)   # all i that appear in the sum

        # pick a source index i from the contributing set,
        # weighted by its current mass (more mass → more impact)
        probs = best[contrib]
        if probs.sum() > 0:
            i = rng.choice(contrib, p=probs / probs.sum())
        else:
            i = rng.integers(0, n)

        # pick a destination index j uniformly from the whole set (could be
        # anywhere, even inside the contributing set – the acceptance check
        # will filter non‑improving moves)
        j = rng.integers(0, n)
        if i == j:
            continue

        delta = best[i] * rng.random() * move_frac
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = 2·n·max(conv(a,a)) (equivalently minimise max(conv) with sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1

        # compute the full convolution and the index of the maximum entry
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of conv[m] with respect to a
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        # For each i that participates in the max entry
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------

    # Use a size large enough to give a rich search space but still fast.
    default_n = 8000
    init_seq = globals().get("height_sequence_1", None)
    base_seq = None
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            # Use at most default_n elements; if the provided sequence is
            # longer we truncate, otherwise we pad later.
            n = min(tmp.size, default_n)
            base_seq = tmp[:n]
        else:
            n = default_n
    else:
        n = default_n

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if base_seq is not None:
        best_a = base_seq.copy()
        best_a = np.maximum(best_a, 0.0)
        s = best_a.sum()
        if s < 1e-12:
            best_a = np.full(n, 1.0 / n)
        else:
            best_a = best_a / s
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart continuation L‑BFGS
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-8, num=40))
    maxiter_per_stage = 500
    modes = ["random", "exp", "power", "edge", "sidon", "binary", "bump", "best"]
    phase1_deadline = start_time + 0.55 * budget_s
    restart_id = 0

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":

        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[restart {restart_id:04d}] mode={mode} improved fast_score={best_score:.6f}")

        # short targeted hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.6)

            fft_len,
