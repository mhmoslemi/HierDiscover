"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The algorithm is a multi‑size, multi‑restart local optimiser that combines:
* smooth max‑approximation (softmax parametrisation + L‑BFGS continuation)
* projected sub‑gradient descent on the exact max‑convolution
* greedy targeted hill‑climb
* random hill‑climb
* a short simulated‑annealing polishing phase

Various random initialisers (uniform, exponential, power‑law, edge‑biased,
sinusoidal and Gaussian) are used to avoid premature convergence.
The total runtime is limited to the supplied budget (default 1000 s) and
the best sequence found (its sum is 1) is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0:
        return np.full_like(y, 1.0 / len(y))
    return e / s


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact evaluation of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the simplex {x ≥ 0, sum x = 1}
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth approximation of max‑convolution (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    Returns (function value, gradient w.r.t. y).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                         # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                               # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS on a decreasing smoothing schedule.
    Returns the final log‑probability vector y.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).

    Returns the best vector encountered and its fast_score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)  # constant index array for vectorised gradient

    for t in range(max_iters):
        if time.time() > deadline:
            break

        # linear convolution and index of the current maximum
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # Sub‑gradient of the (non‑smooth) max at active index k
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(t + 1.0)

        # gradient step + projection onto simplex
        a = project_simplex(a - eta * grad)

        # keep the best solution seen so far
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional perturbation to leave plateaus
        if (t + 1) % 300 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray, cur_score: float, fft_len: int, n: int,
               deadline: float, max_iter: int = 2500) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.09

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(a: np.ndarray, cur_score: float, fft_len: int, n: int,
                        deadline: float, max_moves: int = 4000,
                        frac: float = 0.07) -> tuple[np.ndarray, float]:
    """
    Greedy climb that moves mass from the biggest contributor to the
    smallest contributor for the currently maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


def simulated_annealing(a: np.ndarray, cur_score: float, fft_len: int, n: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Lightweight Metropolis‑based annealing.  Small random mass swaps are
    accepted with probability exp(‑Δ/ T) where temperature T decays linearly
    with remaining time.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    total = deadline - start
    T0 = 0.5 * (best_score)            # initial temperature proportional to score

    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start
        T = max(1e-6, T0 * (1.0 - elapsed / total))

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.06
        if cur[i] < delta:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score

def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑n step function within the allotted time slice.
    Returns the best probability vector (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ------------------------------------------------------------------
    # Initialise a0
    # ------------------------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # smoothing schedule – a finer geometric progression than the baseline
    lam_factors = np.geomspace(0.5, 1e-8, num=36)

    iteration = 0

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.time() < deadline - 6.0:
        mode = rng.choice(
            ["uniform", "random", "exp", "power", "edge", "sin", "gauss", "best"]
        )

        if mode == "uniform":
            a = np.full(n, 1.0 / n)
            y0 = np.log(a + 1e-12)
        elif mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.0008, 0.018)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 2.2)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            edge_frac = rng.uniform(0.01, 0.045)
            m_edge = max(1, int(n * edge_frac))
            edge_val = np.sqrt(n / 2.0) / n      # heuristic edge height
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "sin":
            x = np.arange(n, dtype=float) + 1
            a = np.sin(np.pi * x / (n + 1))
            a = np.clip(a, 0, None)
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "gauss":
            sigma = rng.uniform(n / 30.0, n / 8.0)
            centre = (n - 1) / 2.0
            a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
            a += 1e-8
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:          # "best"
            y0 = best_y + rng.normal(scale=0.045, size=n)

        # --------------------------------------------------------------
        # 1) Continuation L‑BFGS on the smoothed objective
        # --------------------------------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] init={mode} L‑BFGS fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Exact sub‑gradient descent
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 1.1)
        best_a, best_score = subgrad_descent(
            best_a,
            n,
            fft_len,
            max_iters=3000,
            deadline=sub_deadline,
            eta0=0.48,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 3) Targeted hill‑climb
        # --------------------------------------------------------------
        targ_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            targ_deadline,
            max_moves=5000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 4) Random hill‑climb
        # --------------------------------------------------------------
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=3500,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 5) Simulated annealing polishing (short)
        # --------------------------------------------------------------
        anneal_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len,
            n,
            anneal_deadline,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final low‑λ L‑BFGS sweep (consume any remaining time)
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.2:
        for lam in reversed(lam_factors[-12:]):   # a few very low λ steps
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 500, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    return best_a, best_score

# Global entry point – invoked by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its elements sum to 1).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin for final evaluation

    # ------------------------------------------------------------------
    # Choose a set of problem sizes – larger n yields more freedom.
    # ------------------------------------------------------------------
    sizes = [3000, 5000, 8000]                # three reasonably large sizes
    per_size_budget = (total_deadline - time.time()) / len(sizes)

    # warm‑start from any previous construction
    prev = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev, dtype=float) if prev is not None else None

    best_overall_a = None
    best_overall_score = np.inf

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 3.0:
            break
        # allocate a slice, leaving a little for the next sizes
        slice_budget = min(per_size_budget, remaining - (len(sizes) - idx - 1) * 2.5)

        print(
            f"[size] Optimising length n={n} with slice budget ≈{slice_budget:.1f}s"
        )
        a_n, score_n = optimise_one_size(
            n=n,
            budget_s=slice_budget,
            seed=seed + idx * 199,
            prev_seq=prev_arr,
        )
        print(f"[size] n={n} best fast score = {score_n:.6f}")

        if score_n < best_overall_score:
            best_overall_score = score_n
            best_overall_a = a_n.copy()

        # keep the best sequence as a warm‑start for the next size
        prev_arr = best_overall_a.copy()

    # ------------------------------------------------------------------
    # Final evaluation via the provided secure wrapper
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose for potential reuse in subsequent calls
    globals()["height_sequence_1"] = final_seq
    return final_seq
