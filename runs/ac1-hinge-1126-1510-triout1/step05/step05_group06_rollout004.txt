"""
Optimisation of a non‑negative step function for the evaluation metric

    2·n·max(conv(a,a)) / (sum(a)²)

The algorithm combines:

1.   A diversified parametric sweep (uniform, beta‑shapes, double‑Gaussian,
     power‑law, exponential, sinusoid, Dirichlet draws) to collect several
     promising seeds.
2.   Continuation L‑BFGS on a smooth log‑sum‑exp approximation of the
     max‑convolution (decreasing λ schedule, many restarts).
3.   An LP “balancing’’ step implemented with SciPy’s high‑performance
     `linprog`.  After solving the LP we perform a golden‑section line‑search
     for the optimal mixing factor `t` between the current vector and the LP
     solution, guaranteeing a non‑increasing peak.
4.   Projected sub‑gradient descent (exact max‑convolution) for a short
     polishing phase.
5.   A greedy hill‑climb that repeatedly transfers a tiny random mass
     between two coordinates while preserving the simplex constraint.

All phases respect the simplex ∑a = 1, use FFT‑based (O(n log n))
convolutions and stop automatically when the 1000 s budget expires.
The best sequence found is returned as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, maps ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the objective for a simplex‑normalised vector:
    2·n·max(conv(a,a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max via log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                       # numeric stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv_i
    w = exp_shift / Z
    # convolution of w with reversed a gives derivative wrt a
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, passing the result of one stage to
    the next. Returns the best vector and its score.
    """
    y = np.log(a_start + 1e-12)          # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # short hill‑climb after each λ (helps escape tiny local traps)
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent with a simple
    back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# LP balancing with optimal mixing factor (SciPy linprog + golden section)
# ----------------------------------------------------------------------
def optimal_mixing_t(a: np.ndarray, b: np.ndarray,
                     n: int, fft_len: int, max_iter: int = 20) -> float:
    """
    Golden‑section search on t∈[0,1] for a_t = (1‑t)a + t b minimizing
    2·n·max(conv(a_t,a_t)).
    """
    phi = (np.sqrt(5) - 1) / 2.0
    lo, hi = 0.0, 1.0
    # helper to evaluate the score for a given t
    def score(t):
        at = (1.0 - t) * a + t * b
        return fast_score(at, n, fft_len)

    c = hi - phi * (hi - lo)
    d = lo + phi * (hi - lo)
    fc = score(c)
    fd = score(d)
    for _ in range(max_iter - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - phi * (hi - lo)
            fc = score(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + phi * (hi - lo)
            fd = score(d)
    t_opt = (lo + hi) / 2.0
    return t_opt


def lp_improve_mix(a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 12):
    """
    Perform a few LP “balancing’’ iterations.  After each LP we mix the LP
    solution with the current vector using the optimal mixing factor found
    by golden‑section search.
    """
    a = a.copy()
    a = a / a.sum()                # ensure on simplex
    fft_len_conv = next_pow2(2 * n - 1)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current peak
        conv = full_conv(a, fft_len_conv)
        M = conv.max()

        # Build convolution matrix T (size (2n‑1)×n)
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear programme: maximise ∑b  ⇔ minimise -∑b
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None) for _ in range(n)]

        # Solve with HiGHS (fast, works with sparse matrices)
        lp_res = linprog(c, A_ub=T, b_ub=b_ub,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})

        if not lp_res.success or lp_res.x is None:
            # LP failed – abort further LP steps
            break

        b_vec = np.maximum(lp_res.x, 0.0)
        if b_vec.sum() == 0.0:
            break
        b_vec /= b_vec.sum()                # normalise to simplex

        # Find best mixing factor
        t_opt = optimal_mixing_t(a, b_vec, n, fft_len_conv, max_iter=15)
        a_new = (1.0 - t_opt) * a + t_opt * b_vec
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, fft_len_conv)
        old_score = fast_score(a, n, fft_len_conv)

        if new_score < old_score - 1e-12:
            a = a_new
            it += 1
        else:
            # No improvement – stop early
            break

    return a
