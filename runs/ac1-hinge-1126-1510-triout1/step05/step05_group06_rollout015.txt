```
    best_a = softmax(y)
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam_factor in lam_factors:
        for s in range(steps_per_factor):
            if time.time() >= deadline:
                break
            f, g = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
            # Adam update
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)

            # evaluate actual objective
            score = fast_score(a, n, fft_len_conv)
            if score < best_score:
                best_score = score
                best_a = a.copy()
```

```
def lp_improve(a, n, deadline, t_init=0.1, max_iters=6):
    a = a.copy()
    a = a / a.sum()
    t = t_init
    it =0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()
        # Create T matrix:
        data = [np.full(n, a[i]) for i in range(n)]  # each column diag
        offsets = np.arange(n)
        T = diags(data, offsets, shape=(2*n-1, n), format='csc')
        # Solve LP: maximize sum(b) => minimize -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # A_ub = T
        b_ub = np.full(2*n-1, M)
        try:
            res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        except Exception as e:
            print("[lp] linprog exception:", e)
            break
        if not res.success:
            print("[lp] linprog not successful:", res.message)
            break
        b_val = np.maximum(res.x, 0.0)
        sum_b = b_val.sum()
        if sum_b == 0:
            break
        b_scaled = b_val / sum_b
        a_new = (1-t) * a + t * b_scaled
        a_new = a_new / a_new.sum()
        new_score = fast_score(a_new, n, fft_len_conv)  # need fft_len_conv variable
        old_score = fast_score(a, n, fft_len_conv)
        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init
        else:
            t *= 0.5
            if t < 1e-4:
                break
        it+=1
    return a
```

"""
Improved optimizer for the step‑function evaluation metric

The code follows a multi‑phase pipeline:

1. **Generate a diverse set of seeds** – uniform, exponential, power‑law,
   sinusoidal, arcsine, beta‑shaped and random Dirichlet draws (plus the
   previously best known construction if available).

2. **Adam on a smooth log‑sum‑exp approximation** – the objective
   `2·n·max(conv(a,a))` is approximated by a smoothed max with a tunable
   λ‑factor.  For each seed we run Adam with a decreasing λ schedule.
   The parameters are kept on the probability simplex via a softmax
   representation.

3. **Projected sub‑gradient descent** – after the smooth optimisation we
   directly minimise the exact max‑convolution using a simple back‑tracking
   line‑search and Euclidean projection onto the simplex.

4. **Linear‑programming balancing (SciPy)** – a short LP phase maximises
   the ℓ₁‑norm of a vector `b` under the linear constraints
   `(a * b) ≤ M` (where `M` is the current peak).  Mixing a small
   fraction of the scaled solution back into `a` provably does not increase
   the peak.

5. **Greedy hill‑climbing (mass transfer)** – a final intensive hill‑climb
   moves tiny random packets of mass between coordinates while always
   preserving the simplex.  Only strictly improving moves are kept.

All heavy work uses FFT‑based convolutions (`O(n log n)`).  The total
budget (default 1000 s) is split automatically; the best sequence found
before the deadline is returned as a plain list of non‑negative floats.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Small utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    # keep only the needed part
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    2·n·max(conv(a,a))   (a is already normalised to sum = 1)
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by Adam
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_k
    # convolution of w with reversed a (chain rule)
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    # indices corresponding to a component in the gradient
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------
def adam_optimize(y0: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int,
                  lam_schedule, steps_per_factor, deadline):
    """
    Adam optimisation on the smooth log‑sum‑exp surrogate.
    Returns the best probability vector found and its score.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    it = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        for _ in range(steps_per_factor):
            if time.time() >= deadline:
                return best_a, best_score
            f, g = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            it += 1
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = softmax(y)
            score = fast_score(a, n, fft_len_conv)
            if score < best_score:
                best_score = score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent
    with a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active mask
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer) with optional simulated‑annealing
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # transfer up to 5 % of the donor mass
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1
    return best, best_score


# ----------------------------------------------------------------------
# LP balancing step using SciPy’s high‑level linear programming
# ----------------------------------------------------------------------
def lp_improve(a: np.ndarray,
               n: int,
               deadline: float,
               fft_len_conv: int,
               t_init: float = 0.12,
               max_iters: int = 8):
    """
    Perform a few iterations of the LP balancing step:
    maximise Σ b_j  s.t.  b ≥ 0, (a * b)_k ≤ M ∀k  (M = max(a*a)).
    After each LP we mix a small fraction of the (scaled) solution back.
    """
    a = a.copy()
    a = a / a.sum()                       # ensure simplex
    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)         # exact linear conv
        M = conv.max()

        # Build convolution matrix T : (2n‑1)×n, each column i has a[i] on a
        # diagonal of offset i.
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear program: maximise sum(b)  -> minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M)

        res = linprog(c, A_ub=T, b_ub=b_ub,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1.0, deadline - time.time())})
        if not res.success:
            # if the LP fails we abort the balancing loop
            break
        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        b_scaled = b_val / s

        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, fft_len_conv)
        old_score = fast_score(a, n, fft_len_conv)

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init               # reset mixing factor after success
            it += 1
        else:
            t *= 0.5
            if t < 1e-5:
                break
            it += 1
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step sequence.
    Returns a plain Python list of floats (non‑negative, sum > 0).
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size / initial vector
    # ------------------------------------------------------------------
    max_n = 2500                     # safe size for the LP stage
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0[a0 < 0] = 0.0
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    # ------------------------------------------------------------------
    # Seed generation (a diverse soup of candidates)
    # ------------------------------------------------------------------
    seeds = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        seeds.append(arr)

    # Uniform
    add_candidate(np.full(n, 1.0 / n))

    # Linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # Exponential families
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # Double‑exponential (heavy at both ends)
    for rate in [5e-4, 1e-3, 2e-3]:
        arr = np.exp(-rate * np.arange(n)) + np.exp(-rate * np.arange(n - 1, -1, -1))
        add_candidate(arr)

    # Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2

    # Arcsine‑type distribution (≈ 1/√(x(1‑x)))
    i = np.arange(n) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    add_candidate(w)

    # Random Dirichlet draws (Gamma)
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        for _ in range(2):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Beta‑shaped families
    for a_param in [0.5, 1.0, 2.0]:
        for b_param in [0.5, 1.0, 2.0]:
            x = np.arange(1, n + 1, dtype=float)
            arr = (x ** (a_param - 1)) * ((n - x + 1) ** (b_param - 1))
            add_candidate(arr)

    # Edge‑heavy (mass on the two ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # Previous best (if present)
    if "height_sequence_1" in globals():
        add_candidate(np.asarray(globals()["height_sequence_1"], dtype=float))

    # ------------------------------------------------------------------
    # Keep the most promising few seeds
    # ------------------------------------------------------------------
    seed_scores = [(fast_score(s, n, fft_len_conv), s) for s in seeds]
    seed_scores.sort(key=lambda t: t[0])
    top_seeds = [arr for _, arr in seed_scores[:8]]
    print(f"[seed] best initial score = {seed_scores[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing split

    deadline = start + budget_s - 0.01          # safety margin
    phase1_end = start + 0.65 * budget_s       # Adam optimisation
    phase2_end = start + 0.90 * budget_s       # sub‑gradient + LP

    lam_schedule = [0.8, 0.6, 0.4, 0.2, 0.1, 0.08, 0.05,
                    0.03, 0.02, 0.01, 0.008, 0.005,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    steps_per_factor = 180   # moderate depth per λ
    best_a = a0.copy()

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (use whatever time is left)
    # ------------------------------------------------------------------
    final_deadline = deadline

                                    best_score,

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)    # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
