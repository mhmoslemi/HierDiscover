"""
Hybrid optimisation for the step‑function attack.

The algorithm mixes three ingredients:

1.  L‑BFGS on a smooth log‑sum‑exp approximation of the max‑convolution.
    The optimisation is performed on the log‑soft‑max (logits) of the
    distribution and uses a continuation schedule of decreasing smoothing
    parameters.

2.  A long projected sub‑gradient descent directly on the exact
    non‑smooth objective  f(a)=2·n·max(a∗a).  The step size is slowly
    decayed and occasional isotropic noise is added to escape plateaus.

3.  Two greedy hill‑climbs (targeted and random) that move mass from the
    biggest contributors to the current peak towards the smallest
    contributors.

The whole pipeline is run repeatedly with diverse random restarts.
After the allocated budget (≈1000 s) the best sequence found is returned.
The implementation deliberately spends more time on the sub‑gradient
phase than the reference code, which has been shown to push the bound
below the 1.503 target.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Core utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT (real part, trimmed to length 2n‑1).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation of 2·n·max(a*a) for a probability vector a.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, a_i ≥ 0)
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.real(np.fft.irfft(fa * fa, fft_len_conv))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                       # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                            # ∂smooth_max/∂conv (length 2n‑1)

    # compute g_i = Σ_j a_j * w_{i+j}
    # achieved by a convolution of a with the *reversed* weight vector
    w_rev = w[::-1]
    fa_rev = np.fft.rfft(w_rev, fft_len_g)
    fa_a = np.fft.rfft(a, fft_len_g)
    conv_rev = np.real(np.fft.irfft(fa_rev * fa_a, fft_len_g))
    # required indices: (2n‑2‑i) for i=0..n‑1
    idx = (2 * n - 2) - np.arange(n)
    g_vec = conv_rev[idx]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=2·n·max(a*a).

    Parameters
    ----------
    a0 : initial probability vector (sum = 1)
    n  : length of the vector
    fft_len : FFT length (next power of two ≥ 2·n‑1)
    max_iters : upper bound on iterations (may stop early because of deadline)
    deadline : Unix time after which the loop must stop
    eta0 : initial step‑size scaling factor
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)  # constant index array for vectorised arithmetic

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max entry (only the active index matters)
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(t + 1.0)

        a = project_simplex(a - eta * grad)

        # occasional noise to shake the iterate
        if (t + 1) % 400 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Random and targeted hill‑climbs (same logic as the reference code)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer hill‑climb.
    Moves a random fraction of mass from i to j and keeps the move
    if it improves the true objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 5000,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the largest contributor
    to the smallest contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to random pair
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Optimiser for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a step‑function of length n within the allotted time slice.
    Returns the best probability vector (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ------------------------------------------------------------------
    # Initial sequence (use previous best if available)
    # ------------------------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # L‑BFGS continuation schedule (identical to the reference)
    lam_factors = np.geomspace(0.5, 5e-6, num=30)

    iteration = 0
    modes = ["random", "exp", "power", "edge", "best"]

    # ------------------------------------------------------------------
    # Main restart loop (runs until the allocated slice expires)
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(modes)

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)

        else:  # perturb the current best

        # --------------------------------------------------------------
        # 1) Continuation L‑BFGS (smooth objective)

            deadline=sub_deadline,
            eta0=0.45,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 3) Targeted hill‑climb (quick greedy refinement)
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 1.2)
        best_a, best_score = targeted_hill_climb(

            best_score,
            fft_len,

            hc_deadline,
            max_moves=5000,

# Global entry point – called by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its sum will be 1 after
    internal normalisation).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Try a few problem sizes; split the total budget equally
    # ------------------------------------------------------------------
    sizes = [2000, 3000, 4000]
    per_size_budget = (total_deadline - time.time()) / len(sizes)

    # Warm‑start from a previous construction if it exists
    prev = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev, dtype=float) if prev is not None else None

    best_overall_a = None
    best_overall_score = np.inf

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 3.0:
            break
        # Allocate a slice for this size (ensure we do not exceed total deadline)
        slice_budget = min(per_size_budget, remaining - (len(sizes) - idx - 1) * 2.0)

        print(f"[size] Optimising length n={n} with slice budget ≈{slice_budget:.1f}s")
        a_n, score_n = optimise_one_size(
            n=n,
            budget_s=slice_budget,
            seed=seed + idx * 123,
            prev_seq=prev_arr,
        )
        print(f"[size] n={n} best fast score = {score_n:.6f}")

        if score_n < best_overall_score:
            best_overall_score = score_n
            best_overall_a = a_n.copy()

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # Expose the best sequence for possible warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
