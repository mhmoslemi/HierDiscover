def anneal(a, cur_score, fft_len, n, deadline):
    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    T0 = 0.1 * cur_score  # initial temperature magnitude
    T_end = 1e-5
    # schedule based on elapsed fraction
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * (1 - frac) + T_end * frac
        # propose move
        i, j = rng.integers(0,n, size=2)
        if i==j: continue
        # delta fraction
        max_move = min(0.3, cur_a[i])  # cannot exceed cur_a[i]
        delta = max_move * rng.random()
        if delta <= 0: continue
        cur_a[i] -= delta
        cur_a[j] += delta
        new_score = fast_score(cur_a, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < math.exp(-diff / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        else:
            # revert
            cur_a[i] += delta
            cur_a[j] -= delta
    return best_a, best_score
```

        v_hat = v / (1-beta2**t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return y

"""
Propose a step‑function (list of non‑negative heights) with a low evaluation
score.

Algorithm
---------
1. Choose a problem size n (≈ 5 000‑8 000).  Work on the probability simplex
   (sum = 1) because the scaling cancels out in the objective.
2. Multi‑restart stochastic optimisation:
   – Initialise y (pre‑softmax) with many diverse strategies
     (random Gaussian, exponential / power‑law decay, edge‑heavy,
     sparse Sidon‑set, binary mask, single Gaussian bump,
     two‑bump, sinusoidal wiggle and “best” – a perturbed copy of the
     current best point).
   – Run a continuation L‑BFGS on a smoothed log‑sum‑exp version of the
     objective.  The smoothing factor λ (relative to the current max‑conv)
     is decreased geometrically (≈ 30 stages).
   – After each L‑BFGS run apply a short greedy hill‑climb.
   – Keep the best candidate found so far.
3. Simulated‑annealing phase:
   – Randomly move a fraction of mass between two positions.
   – Accept worsening moves with probability exp(‑Δ/T) where the temperature
     T decays linearly.
   – Keep the best solution discovered.
4. Final aggressive hill‑climb to polish the result.
5. Scale the probability vector to total sum = √(2 n) (well below the
   1000‑value cap) and return it as a plain list.

All heavy work is done with FFT‑based convolutions (O(n log n)).  The
time budget (default 1000 s) is split roughly 60 % L‑BFGS, 30 % annealing,
10 % final hill‑climb.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(conv)   (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, n=fft_len_conv)
    conv = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z  # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_g)
    fw = np.fft.rfft(w, n=fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, n=fft_len_g)

    # only the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing.
    Each stage starts from the result of the previous stage.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # L‑BFGS on the smoothed objective
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        # Even if the optimizer fails to converge use its last point
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initialisation helpers (pre‑softmax vectors)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes:
        random – normal distribution
        exp    – exponential decay
        power  – power‑law decay
        edge   – mass at the two edges
        sidon  – sparse Sidon‑set
        binary – random 0/1 mask
        bump   – a single Gaussian bump
        twobump– two separated Gaussian bumps
        saw    – sinusoidal wiggle
        best   – perturbed copy of the current best (handled outside)
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy construction of a Sidon set (size ≈ √n)
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * cand not in sums):
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "twobump":
        centre1 = rng.integers(0, n // 2)
        centre2 = rng.integers(n // 2, n)
        sigma = rng.uniform(0.04, 0.10) * n
        idx = np.arange(n)
        a0 = (np.exp(-0.5 * ((idx - centre1) / sigma) ** 2) +
              np.exp(-0.5 * ((idx - centre2) / sigma) ** 2))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "saw":
        # Sinusoidal pattern with many periods
        periods = rng.integers(5, 15)
        phase = rng.random() * 2 * np.pi
        a0 = np.abs(np.sin(2 * np.pi * np.arange(n) * periods / n + phase)) + 0.01
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_delta = best[i] * max_move
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def anneal(
    a_start: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the true objective.
    Randomly transfer mass; sometimes accept worsening moves.
    """
    best_a = a_start.copy()
    best_score = cur_score
    cur_a = a_start.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    # Temperature schedule: linear decay
    T0 = max(1e-3, 0.05 * cur_score)
    T_end = 1e-6

    while time.time() < deadline:
        elapsed = time.time() - start
        frac = elapsed / (deadline - start)
        T = T0 * (1 - frac) + T_end * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # ensure we do not take more mass than available at i
        max_delta = cur_a[i] * 0.25
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta
        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < math.exp(-diff / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = cur_a.copy()
        else:
            # revert move
            cur_a[i] += delta
            cur_a[j] -= delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Determine problem size n
    # --------------------------------------------------------------
    # If a previous construction is available, use its length (capped at 8000).
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)
        else:
            n = 5000
    else:
        n = 5000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1: multi‑restart L‑BFGS (≈ 60 % of budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=30))
    modes = [
        "random", "exp", "power", "edge", "sidon",
        "binary", "bump", "twobump", "saw", "best"
    ]
    phase1_deadline = start_time + 0.60 * budget_s
    iteration = 0

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # short greedy polishing
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Phase 2: simulated annealing (≈ 30 % of budget)
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.90 * budget_s
    print("[anneal] starting simulated annealing")
    best_a, best_score = anneal(
        best_a,
        best_score,
        fft_len_conv,
        n,
        phase2_deadline,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[anneal] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3: final aggressive hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Rescale to a reasonable total sum (√(2 n) keeps numbers modest)
    # --------------------------------------------------------------
    scale = math.sqrt(2.0 * n) / np.sum(best_a)
    final_seq = (best_a * scale).tolist()

    # --------------------------------------------------------------
    # Official evaluation (the harness will also call evaluate_sequence)
    # --------------------------------------------------------------
    try:
        final_val = evaluate_sequence(final_seq)  # type: ignore
        print(f"[final] evaluated score = {final_val:.6f}")
    except Exception as e:  # pragma: no cover
        print("[final] evaluation failed:", e)

    return final_seq
