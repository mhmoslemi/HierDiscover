"""
Propose a low‑score non‑negative step sequence.

Approach
--------
1. Initialise with a moderate length (n≈2000) and a uniform distribution.
2. Perform a fast parametric sweep over several simple families
   (edge‑heavy, monotonic exponential, power‑law, edge‑decay,
   sin‑power).  The sweep uses the cheap FFT‑based fast_score
   (2·n·max(conv)) and keeps the best candidate found.
3. Starting from the current best, run a continuation L‑BFGS optimiser
   on the smooth log‑sum‑exp surrogate of the max‑convolution.
   The lam schedule goes down to 5e‑4 and the optimiser is allowed many
   iterations.
4. Refine further with a projected sub‑gradient descent on the exact
   max‑convolution (additive update + Euclidean simplex projection).
5. Finish with a greedy hill‑climb that moves a small random fraction
   of mass between two indices and accepts only strictly improving
   moves.
6. Return the best sequence (as a list) before the 1000 s deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# -------------------------------------------------------------
# Basic utilities
# -------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# -------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# -------------------------------------------------------------
# Continuation L‑BFGS
# -------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n,
                       lam_factor_schedule=None):
    """
    Runs L‑BFGS for each lam in lam_schedule.
    Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)          # initialise logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for idx, lam in enumerate(lam_schedule):
        if time.time() >= deadline:
            break
        lam_factor = (lam_factor_schedule[idx]
                      if lam_factor_schedule is not None else 0.30)
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam_factor),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# -------------------------------------------------------------
# Projected sub‑gradient descent on the exact max
# -------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                        deadline: float,
                        fft_len: int,
                        n: int):
    """
    Additive sub‑gradient descent on max (a*a) with Euclidean
    projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    eta0 = 0.5

    while time.time() < deadline:
        conv = full_conv(a, fft_len)
        k = int(conv.argmax())

        # sub‑gradient of the max (exact)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        eta = eta0 / np.sqrt(it + 1.0)
        a = a - eta * g
        a = simplex_projection(a)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# -------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# -------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move up to 5 % of the mass currently at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# -------------------------------------------------------------
# Parameterised families (used for a quick global search)
# -------------------------------------------------------------


def edge_weight_sequence(n: int, m: int, w: float) -> np.ndarray:
    """m edge entries on each side carry total weight w (0≤w≤0.5)."""
    if m == 0:
        return np.full(n, 1.0 / n)
    interior = (1.0 - w) / (n - 2 * m)
    edge_val = w / (2.0 * m)
    a = np.full(n, interior, dtype=np.float64)
    a[:m] = edge_val
    a[-m:] = edge_val
    return a


def monotonic_exponential(n: int, rate: float) -> np.ndarray:
    """Decreasing exponential a_i ∝ exp(-rate·i)."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64))
    return a / a.sum()


def edge_decay_exponential(n: int, rate: float) -> np.ndarray:
    """Symmetric decay from the edges: a_i ∝ exp(-rate·dist_to_edge)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = np.exp(-rate * d.astype(np.float64))
    return a / a.sum()


def power_decay(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} (monotone)."""
    a = 1.0 / (np.arange(1, n + 1, dtype=np.float64) ** p)
    return a / a.sum()


def edge_decay_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (dist_to_edge+1)^{-p}."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1)) + 1
    a = 1.0 / (d.astype(np.float64) ** p)
    return a / a.sum()


def sin_power(n: int, beta: float) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n)^{beta}."""
    a = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    a = np.maximum(a, 0.0) ** beta
    if a.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return a / a.sum()

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # Problem size
    # ---------------------------------------------------------
    max_n = 4000
    # Using a moderate size keeps FFT cheap while still offering good quality.
    n = max_n // 2                     # 2000
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient‑related convolutions

    # ---------------------------------------------------------
    # Warm‑start from any globally‑available sequence
    # ---------------------------------------------------------
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size >= n:
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # ---------------------------------------------------------
    # Phase 0 – quick parametrised scan
    # ---------------------------------------------------------
    scan_deadline = start_time + 0.20 * budget_s   # 20 % of total budget
    print("[scan] starting parametric search")
    # --- edge‑heavy ------------------------------------------------
    edge_ms = [2, 4, 8, 16, 32, 64, 128, 256, 512]
    for m in edge_ms:
        for w in np.linspace(0.05, 0.5, 10):
            if time.time() >= scan_deadline:
                break
            cand = edge_weight_sequence(n, m, w)
            sc = fast_score(cand, fft_len_conv, n)
            if sc < best_score:
                best_score, best_a = sc, cand.copy()
                print(f"[scan] edge m={m}, w={w:.3f} → {best_score:.6f}")

    # --- monotonic exponential ------------------------------------
    for rate in np.logspace(-4, -0.3, 20):
        if time.time() >= scan_deadline:
            break
        cand = monotonic_exponential(n, rate)
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[scan] mono exp rate={rate:.5f} → {best_score:.6f}")

    # --- edge‑decay exponential ------------------------------------
    for rate in np.logspace(-4, -0.3, 20):
        if time.time() >= scan_deadline:
            break
        cand = edge_decay_exponential(n, rate)
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[scan] edge decay rate={rate:.5f} → {best_score:.6f}")

    # --- power‑law (monotone) --------------------------------------
    for p in np.linspace(0.1, 3.0, 15):
        if time.time() >= scan_deadline:
            break
        cand = power_decay(n, p)
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[scan] power p={p:.2f} → {best_score:.6f}")

    # --- edge‑decay power ------------------------------------------
    for p in np.linspace(0.5, 4.0, 15):
        if time.time() >= scan_deadline:
            break
        cand = edge_decay_power(n, p)
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[scan] edge‑power p={p:.2f} → {best_score:.6f}")

    # --- sin‑power -------------------------------------------------
    for beta in np.linspace(0.2, 3.0, 15):
        if time.time() >= scan_deadline:
            break
        cand = sin_power(n, beta)
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score, best_a = sc, cand.copy()
            print(f"[scan] sin‑power beta={beta:.2f} → {best_score:.6f}")

    print(f"[scan] best after parametric search fast_score={best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (more aggressive)
    # ---------------------------------------------------------
    if time.time() < deadline:
        lbfgs_end = time.time() + 0.45 * (deadline - time.time())
        lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                        0.07, 0.045, 0.03, 0.018, 0.01,
                        0.006, 0.003, 0.001, 0.0005]
        maxiter_per_lam = 1500
        # a small lam‑factor schedule that tightens the smooth approximation
        lam_factor_sched = [0.30] * 5 + [0.20] * 5 + [0.10] * 4
        # at least as long as lam_schedule
        while len(lam_factor_sched) < len(lam_schedule):
            lam_factor_sched.append(0.05)

        # a few restarts (including current best)
        restarts = 4
        for r in range(restarts):
            if time.time() >= deadline:
                break
            a_start = best_a if r == 0 else random_initial(n, rng)
            a_opt, sc_opt = lbfgs_continuation(
                a_start,
                lam_schedule,
                maxiter_per_lam,
                lbfgs_end,
                fft_len_conv,
                fft_len_g,
                n,
                lam_factor_schedule=lam_factor_sched
            )
            if sc_opt < best_score:
                best_score, best_a = sc_opt, a_opt.copy()
                print(f"[L‑BFGS r{r}] new best fast_score={best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on exact max
    # ---------------------------------------------------------
    if time.time() < deadline:
        subgrad_deadline = deadline - 0.2
        best_a, best_score = subgradient_descent(
            best_a,
            subgrad_deadline,
            fft_len_conv,
            n
        )
        print(f"[subgrad] fast_score={best_score:.6f}")

    # ---------------------------------------------------------
    # Phase 3 – final greedy hill‑climb
    # ---------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000
        )
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    # ---------------------------------------------------------
    # Return the final sequence and report the true evaluation
    # ---------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
