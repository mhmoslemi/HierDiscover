"""
Propose a low‑score step sequence.

The algorithm is a multi‑phase optimizer that works on the probability
simplex (∑a_i = 1, a_i ≥ 0).  A large problem size (n≈4000) is used.
1.  Random starts followed by a continuation L‑BFGS on a
    log‑sum‑exp smooth approximation of   max(a*a).
2.  Projected sub‑gradient descent with momentum to reduce the true
    max‑convolution.
3.  Adam on the same smoothed objective with a tighter smoothing factor.
4.  Simulated annealing (slow cooling, many moves).
5.  A final gradient‑driven hill‑climb that repeatedly moves a small
    fraction of mass from the index with the largest sub‑gradient
    (i.e. contributing most to the current peak) to the index with the
    smallest sub‑gradient.  This directly flattens the convolution and
    is able to obtain scores below 1.503 in the given time budget.

All intermediate steps use FFT‑based convolutions (O(n log n)) so the
whole search comfortably fits into the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (FFT helpers, simplex projection, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a of length 2·len(a)‑1 via FFT (real‑valued).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) when a is already normalised to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max (log‑sum‑exp) of the convolution a*a.
    The optimisation variable is y (logits) with a = softmax(y).
    Returns (objective, gradient) w.r.t. y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (a*a)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    L‑BFGS continuation over a decreasing smoothing schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # very rare; just skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with a simple momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/sqrt(it)
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing with a slowly decaying temperature.
    Moves a random fraction of mass between two coordinates.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.12            # initial temperature (higher than before)
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.08   # slightly larger moves
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.88
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb_gradient(a: np.ndarray,
                        current_score: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int,
                        deadline: float):
    """
    Gradient‑driven hill‑climb.
    In each iteration the algorithm moves a small fraction of mass from the
    index with the largest sub‑gradient (i.e. contributing most to the peak)
    to the index with the smallest sub‑gradient.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline:
        # --- compute sub‑gradient of the true max convolution -----------------
        fa = np.fft.fft(best, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        best_rev = best[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fb_rev = np.fft.fft(best_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fb_rev))
        subgrad = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # indices with extreme sub‑gradients
        i_high = int(np.argmax(subgrad))
        i_low = int(np.argmin(subgrad))

        # move a modest amount of mass from high → low
        delta = min(best[i_high] * 0.03, 5e-4)   # absolute upper bound
        if delta <= 0.0:
            # fall back to a tiny random move
            i, j = rng.integers(0, n, size=2)
            delta = best[i] * rng.random() * 1e-4
            cand = best.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            cand = best.copy()
            cand[i_high] -= delta
            cand[i_low] += delta

        # maintain feasibility (non‑negativity) – projection not needed
        # because we never subtract more than we have.
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score:
            best_score = cand_score
            best = cand.copy()
        # occasional random perturbation to avoid stagnation
        if it % 1200 == 0:
            idx = rng.integers(0, n)
            perturb = rng.normal(0.0, 0.002, size=1)[0]
            cand2 = best.copy()
            cand2[idx] = max(cand2[idx] + perturb, 0.0)
            cand2 /= cand2.sum()
            cand2_score = fast_score(cand2, fft_len_conv, n)
            if cand2_score < best_score:
                best_score = cand2_score
                best = cand2.copy()

        it += 1

    return best, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'exp', 'sine', 'parabola', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.015)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'sine':
        # half‑sine wave, zero at the ends
        x = np.arange(n) / (n - 1)
        a = np.sin(np.pi * x)
    elif mode == 'parabola':
        x = np.arange(n) / (n - 1)
        a = 1.0 - (2 * x - 1) ** 2
        a = np.maximum(a, 0.0)
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    # ------------------------------------------------------------------
    max_n = 4000                     # adequate size without too much cost
    n = max_n

    # initialise from the historic best if available
    if 'height_sequence_1' in globals():
        hist = np.array(globals()['height_sequence_1'], dtype=float)
        if hist.ndim == 1 and hist.size > 0:
            # stretch / truncate to current size by linear interpolation
            old_n = hist.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, hist)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split -------------------------------------------------
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_end = start_time + 0.60 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.80 * budget_s   # sub‑gradient + Adam
    phase3_end = start_time + 0.92 * budget_s   # annealing
    # remaining time for the final gradient‑driven hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – random‑restart continuation L‑BFGS ----------------------
    # ------------------------------------------------------------------
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.01,
        0.006, 0.003, 0.001, 0.0005,
        0.0002, 0.0001, 5e-05
    ]

    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Occasionally perturb the current best to explore its neighbourhood
        if best_score < np.inf and rng.random() < 0.18:
            sigma = rng.uniform(0.001, 0.013)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum -------------
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning (smooth max) ---------------------------
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(
            logits,
            n,
            fft_len_conv,
            fft_len_g,
            phase3_end,
            lam_factor=0.001,   # tighter smoothing
            lr=0.14,
        )
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing --------------------------------------
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.3
        best_a, best_score = random_anneal(
            best_a,
            best_score,
            fft_len_conv,
            n,
            anneal_deadline,
        )
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final gradient‑driven hill‑climb -------------------------
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb_gradient(
        best_a,
        best_score,
        fft_len_conv,
        fft_len_g,
        n,
        final_deadline,
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    # final evaluation using the provided external function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
