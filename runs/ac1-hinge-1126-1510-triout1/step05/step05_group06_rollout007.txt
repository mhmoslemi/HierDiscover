"""
Propose a low‑evaluation step‑function sequence.

The algorithm combines a cheap parametric sweep, multiple restarts of a
smooth‑max L‑BFGS optimiser, a linear‑programming “balancing’’ step
(using SciPy’s HiGHS solver), a short sub‑gradient descent, and a final
greedy hill‑climb.  All iterates stay on the probability simplex
(sum = 1, non‑negative).  The routine respects the 1000 s budget and
prints progress information.

The main entry point is `propose_candidate(seed=42, budget_s=1000)`.
"""

import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags, csc_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (full length 2·len(a)‑1)."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    # map unconstrained y to simplex via soft‑max
    a = softmax(y)

    # --- convolution a*a ---
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # --- smooth max ---
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt conv entries
    w = exp_shift / Z                          # ∂smooth_max/∂conv_i   (size 2n-1)

    # back‑propagate through convolution: we need ∂conv/∂a
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    # indices that correspond to the derivative w.r.t. each a_i
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple λ‑stages, multiple restarts)
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best a and its score.
    """
    y = np.log(a_start + 1e-12)        # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # Tiny hill‑climb after each λ stage (helps escape tiny local traps)
        h_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        h_deadline, max_iter=1200)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming balancing step (uses SciPy’s HiGHS)
# ----------------------------------------------------------------------
def lp_balancing(a: np.ndarray,
                 n: int,
                 deadline: float,
                 t_init: float = 0.12,
                 max_iters: int = 8) -> np.ndarray:
    """
    Perform a few LP balancing iterations:
        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k   (M = max(a*a))

    The solution b is mixed back into a with factor t.
    """
    a = a.copy()
    a = a / a.sum()                     # ensure simplex

    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build convolution matrix T (size (2n−1)×n) as a sparse CSC matrix.
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)

        lp_res = linprog(c, A_ub=T, b_ub=ub,
                         bounds=(0, None),
                         method='highs',
                         options={'presolve': True, 'time_limit': max(0, deadline - time.time())})

        if lp_res.status != 0 or lp_res.x is None:
            print("[lp] LP failed or timed out (status=%d)" % lp_res.status)
            break

        b_val = np.maximum(lp_res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        b_scaled = b_val / s

        # Mix and renormalise
        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a, n, next_pow2(2 * n - 1))

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init                # reset mixing factor after successful step
            it += 1
            print(f"[lp] iter {it}: improved score={new_score:.6f}")
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1

    return a


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set mask
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # transfer up to 5 % of the source mass
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score

def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑type density on {0,…,n‑1} with parameters α,β > 0."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    # avoid pow(0,negative) → replace 0 by tiny value
    x = np.clip(x, 1e-12, 1 - 1e-12)
    arr = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    if arr.sum() == 0:
        arr = np.full_like(arr, 1.0 / n)
    else:
        arr = arr / arr.sum()
    return arr

            print(f"[L‑BFGS] new global best score={best_score:.6f}")

        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – LP balancing + projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # LP balancing (calls SciPy HiGHS, fast for n ≤ 2500)
        lp_deadline = min(phase2_end, deadline - 0.3)
        print("[LP] starting balancing iterations")
        best_a = lp_balancing(best_a, n, lp_deadline,
                              t_init=0.12, max_iters=8)
        best_score = fast_score(best_a, n, fft_len_conv)
        print(f"[LP] post‑LP score={best_score:.6f}")

        # Sub‑gradient descent (short, uses exact max)
        if time.time() < phase2_end:
            sg_deadline = min(phase2_end, deadline - 0.2)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (as much time as remains)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=15000)

    # ------------------------------------------------------------------
    # Return the sequence (plain list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # evaluate with the externally provided function (for the final log line)
    try:
        final_eval = evaluate_sequence(final_seq)    # noqa: F821
    except Exception:
        final_eval = None
    if final_eval is not None:
        print(f"[final] evaluated score = {final_eval:.6f}")
    else:
        print("[final] evaluation not performed (function unavailable)")

    return final_seq
