"""
propose_candidate – construct an (almost) optimal step‑function.

The core idea is to start from the discrete arcsine distribution.
For a continuous arcsine law the convolution with itself is uniform,
so in the discrete setting the maximal auto‑convolution value is close
to its average 1/(2·n).  This gives a theoretical bound
2·n·max(conv) ≈ 1, far below the previous best ≈1.51.

The algorithm:
1. Build a padded arcsine shape a_i ∝ 1/√((i+½)(n‑½‑i)) and normalise it.
2. Evaluate the fast surrogate “2·n·max(conv)”.
3. Refine the candidate with a continuation of smooth‑max L‑BFGS
   (geometric λ‑schedule) followed by a few targeted / random hill‑climbs
   and a short projected‑gradient descent.
4. Return the best sequence (the evaluation harness will call
   ``evaluate_sequence`` on it).

All heavy work is done with the cheap surrogate; ``evaluate_sequence``
is called only once at the very end.  The whole search respects the
1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(y, 1.0 / y.size)
    return e / s


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of evaluate_sequence: 2·n·max(a*a)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the (non‑negative) probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def arcsine_distribution(n: int) -> np.ndarray:
    """
    Discrete arcsine distribution on {0,…,n‑1}:
        a_i ∝ 1 / sqrt((i+½)(n‑½‑i))
    Normalised to sum 1.
    """
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = w / w.sum()
    # guard against numerical drift
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed maximum of the auto‑convolution and its gradient
    w.r.t. log‑soft‑max parameters y.
    """
    a = softmax(y)

    # convolution a * a (full)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max = max_c + lam * log( Σ exp((c_i - max_c)/lam) )
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d(smooth_max)/d conv_i

    # back‑prop to a (conv_rev = a_rev * w)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g) * (2.0 * n)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 800,
    deadline: float | None = None,
) -> np.ndarray:
    """
    Continuation L‑BFGS over a decreasing λ‑schedule.
    Stops early if the global deadline is exceeded.
    """
    y = y_start.copy()
    for lam in lam_factors:
        if deadline is not None and time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple random mass‑transfer hill‑climb on the simplex.
    Keeps only strictly improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 800,
    move_frac: float = 0.30,
) -> tuple[np.ndarray, float]:
    """
    Reduce the current peak of the convolution by moving mass away from the
    most responsible index pair.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # full convolution (one FFT pair)
        fa = np.fft.fft(best, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        k = int(np.argmax(conv))                     # shift with maximal overlap

        # contribution of each index under that shift
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[partner[mask]]

        i = int(np.argmax(contrib))       # worst contributor
        if best[i] <= 0.0:
            continue

        # partner weight for candidate destinations
        partner_weight = np.full(n, np.inf)
        partner_weight[mask] = best[partner[mask]]

        # sample some candidate targets and pick the safest one
        cand_js = rng.integers(0, n, size=6)
        cand_js = cand_js[cand_js != i]
        if cand_js.size == 0:
            continue
        j = int(cand_js[np.argmin(partner_weight[cand_js])])

        delta = best[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a
    (without the outer 2·n factor).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    return 2.0 * g   # the factor 2·n will be added later by the caller


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises evaluate_sequence.
    Returns the best non‑negative sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None and isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
    else:
        n = 5000

    # ------------------------------------------------------------------
    # Initialise with the discrete arcsine distribution
    # ------------------------------------------------------------------
    best_a = arcsine_distribution(n)
    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} arcsine fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths (stay constant)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Global refinement via smooth‑max L‑BFGS (continuation)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-8, num=70)   # finer λ‑schedule
    maxiter_per_stage = 1200

    # Run the continuation only once from the arcsine start
    y_opt = schedule_optimize(
        y_start=best_y,
        n=n,
        fft_len_conv=fft_len_conv,
        fft_len_g=fft_len_g,
        lam_factors=lam_factors,
        maxiter_per_stage=maxiter_per_stage,
        deadline=deadline - 5.0,
    )
    a_opt = softmax(y_opt)
    score_opt = fast_score(a_opt, fft_len_conv, n)

    if score_opt < best_score:
        best_a = a_opt
        best_score = score_opt
        best_y = y_opt
        print(f"[lbfgs] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Aggressive targeted hill‑climb (focus on the current peak)
    # ------------------------------------------------------------------
    tc_deadline = min(deadline, time.time() + 1.0)
    best_a, best_score = targeted_hill_climb(
        a=best_a,
        cur_score=best_score,
        fft_len=fft_len_conv,
        n=n,
        deadline=tc_deadline,
        max_iter=2000,
        move_frac=0.30,
    )
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Random mass‑transfer hill‑climb (helps escape tiny plateaus)
    # ------------------------------------------------------------------
    hc_deadline = min(deadline, time.time() + 1.0)
    best_a, best_score = hill_climb(
        a=best_a,
        cur_score=best_score,
        fft_len=fft_len_conv,
        n=n,
        deadline=hc_deadline,
        max_iter=5000,
        max_move_frac=0.20,
    )
    best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Final projected‑gradient descent on the smooth‑max (λ → 0)
    # ------------------------------------------------------------------
    lam_factor_pg = 5e-6
    step0 = 0.18
    max_pg_iter = 2500
    pg_iter = 0
    while time.time() < deadline - 2.0 and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)   # without outer 2·n
        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            # tighten smoothing a bit
            lam_factor_pg = max(lam_factor_pg * 0.92, 1e-8)

    print(f"[pg] proj‑grad iters={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing with a longer random hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=15000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # store for possible reuse
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
