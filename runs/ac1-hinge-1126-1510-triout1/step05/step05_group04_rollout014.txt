def adam_optimize(y_start, n, fft_len, deadline):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # dynamic lambda schedule: lam = max_c * lam_factor * (decay_factor**iteration) maybe?
        # Actually we want lambda to become small over time: lam = max_c * max(lam_factor * (0.5**(iteration//2000)), 1e-12)
        # But to avoid too small early on, we can start with lam=0.1*max_c and shrink.
        a = softmax(y)
        # compute conv and max
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_c = conv.max()
        # choose lambda = max_c * max(lam_factor * (0.5**(iteration/2000)), 1e-12)
        lam = max_c * max(lam_factor * (0.5**(iteration/5000)), 1e-12)
        # compute smooth max gradient similar to objective_and_grad but with current lam
        # compute shifted = conv - max_c
        shifted = conv - max_c
        exp_shift = np.exp(shifted / lam)
        Z = exp_shift.sum()
        w = exp_shift / Z

        # compute reverse convolution gradient using FFT (g_vec)
        # compute fa_rev = fft of reversed a
        fa_rev = np.fft.fft(a[::-1], fft_len)
        fw = np.fft.fft(w, fft_len)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        g_vec = conv_rev[n-1:n-1+n]
        # chain rule
        mean_g = np.dot(a, g_vec)
        grad_y = a * (g_vec - mean_g)

        # Adam update
        lr = initial_lr / (1 + iteration * 1e-5)  # optional decay
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate fast_score after certain intervals (like every 100 iterations)
        if iteration % 100 == 0:
            a_curr = softmax(y)
            score = fast_score(a_curr, fft_len, n)
            if score < best_score:
                best_score = score
                best_a = a_curr.copy()
        # else continue
    return best_a, best_score

"""
Propose a low‑score step‑function.

Algorithm
---------
* Initialise a probability vector `a` (sum = 1) – either the provided
  `height_sequence_1` or a uniform distribution.
* **Phase 1 – multi‑restart L‑BFGS with continuation smoothing**.
  Several random initialisations are generated (Gaussian, exponential,
  power‑law, edge‑heavy, Sidon‑set, binary mask, single bump, a
  perturbed copy of the current best).  For each restart a
  continuation L‑BFGS is run with a decreasing smoothing factor
  (log‑sum‑exp approximation of the max‑convolution).  After the L‑BFGS
  run a short greedy hill‑climb is performed.
* **Phase 2 – Adam‑style projected optimisation**.
  Starting from the best point found, we minimise the smoothed objective
  `2·n·smooth_max(conv(a,a))` using an Adam optimiser on the
  unconstrained soft‑max parameters.  The smoothing factor is
  gradually reduced.  The best (exact‑score) vector is kept.
* **Phase 3 – aggressive hill‑climb** until the time budget expires.
* The final sequence (converted back from the probability vector) is
  returned; the harness will evaluate it with `evaluate_sequence`.

All numbers are non‑negative; the sum is forced to be positive by the
soft‑max parametrisation.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (outputs a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS / Adam)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point `y` (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = any((candidate + c) in sums or (2 * candidate) in sums for c in chosen)
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

# Local search helpers

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    init_lr: float = 0.30,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
    lam_initial: float = 0.30,
    lam_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed objective.
    Returns the best probability vector (sum = 1) and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    iteration = 0
    lam_factor = lam_initial

    while time.time() < deadline:
        iteration += 1

        # occasionally decay the smoothing parameter
        if iteration % 500 == 0:
            lam_factor = max(lam_factor * lam_decay, 1e-9)

        _, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam moment updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        y = y - init_lr * m_hat / (np.sqrt(v_hat) + eps)

        # periodic exact evaluation
        if iteration % 200 == 0:
            a_curr = softmax(y)
            cur_score = fast_score(a_curr, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_curr.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)       # allow a bit larger vectors
        else:
            n = 5000
    else:
        n = 5000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS (≈ 55 % of the budget)
    # --------------------------------------------------------------
    main_deadline = start_time + 0.55 * budget_s
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=30))
    modes = ["random", "exp", "power", "edge", "sidon", "binary", "bump", "best"]
    it = 0

    while time.time() < main_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # Continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=250,      # a bit lighter to allow many restarts
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{it:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # short greedy polishing
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)
        it += 1

    # --------------------------------------------------------------
    # Phase 2 – Adam refinement (≈ 30 % of the budget)
    # --------------------------------------------------------------
    adam_deadline = min(deadline - 2.0, start_time + 0.85 * budget_s)
    print("[adam] starting Adam refinement")
    best_a, best_score = adam_optimize(
        best_y,
        n,
        fft_len_conv,
        fft_len_g,
        adam_deadline,
        init_lr=0.30,
        lam_initial=0.30,
        lam_decay=0.9995,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[adam] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final aggressive hill‑climb (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30000,
            max_move=0.35,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Return result
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
