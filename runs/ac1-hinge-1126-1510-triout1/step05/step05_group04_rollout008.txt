def accelerated_subgradient_descent(a_start, fft_len, n, deadline, eta0=0.5):
    a = a_start.copy()
    a_prev = a.copy()
    t_prev = 1.0
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        # compute subgradient at current a
        conv = conv_fft(a, fft_len, n)
        m = np.argmax(conv)
        # compute subgradient g = 2 * shifted a
        g = np.zeros_like(a)
        # Determine overlapping indices
        start = max(0, m - (n - 1))
        end = min(m, n - 1)
        idx = np.arange(start, end+1)
        rev = m - idx
        g[idx] = 2.0 * a[rev]
        # step size
        eta = eta0 / np.sqrt(it)
        # Nesterov extrapolation
        y = a + ((t_prev - 1) / (t_prev + 1)) * (a - a_prev)  # alternative
        # Actually algorithm:
        # y_k = a_k + beta_k * (a_k - a_{k-1})
        beta = (t_prev - 1) / (t_prev + 2)  # common schedule? may use typical Nesterov: beta_k = (t_{k-1} - 1)/t_k, 
        y = a + beta * (a - a_prev)
        # compute subgradient at y (need conv_y)
        conv_y = conv_fft(y, fft_len, n)
        m_y = np.argmax(conv_y)
        g_y = np.zeros_like(y)
        start_y = max(0, m_y - (n - 1))
        end_y = min(m_y, n - 1)
        idx_y = np.arange(start_y, end_y+1)
        rev_y = m_y - idx_y
        g_y[idx_y] = 2.0 * y[rev_y]
        # gradient step
        a_new = y - eta * g_y
        # projection onto probability simplex (sum=1)
        a_new = project_onto_simplex(a_new)
        # evaluate
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        # update variables for Nesterov
        a_prev = a
        a = a_new
        # update t
        t_new = (1 + np.sqrt(1 + 4 * t_prev**2)) / 2
        t_prev = t_new
    return best_a, best_score

def simulated_annealing_climb(a, init_score, fft_len, n, deadline, start_temp=0.01, end_temp=1e-6, max_iter=20000):
    best = a.copy()
    best_score = init_score
    cur = a.copy()
    cur_score = init_score
    rng = np.random.default_rng()
    total_time = deadline - time.time()
    max_iter = max_iter
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # temperature schedule linearly decay
        t = start_temp + (end_temp - start_temp) * (it / max_iter)
        # propose move
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        # propose delta proportional to cur[i]
        delta = cur[i] * rng.random() * 0.2
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        delta_score = new_score - cur_score
        # accept if improve or with prob exp(-delta_score/t)
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(t, 1e-12)):
            cur_score = new_score
            # maybe update best
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score
```

"""
Advanced search for a low‑value step‑function.

The evaluation score is
    2·n·max(a*a) / (sum(a)^2)          (scale‑invariant)
so we work on the probability simplex (sum = 1).  The algorithm consists of
four phases, each using a different optimisation principle.

1.  Multi‑restart L‑BFGS on a smooth log‑sum‑exp surrogate of the max‑convolution.
2.  Short greedy hill‑climbs that move a random fraction of mass between two indices.
3.  Accelerated projected sub‑gradient descent on the true (non‑smoothed) objective.
4.  Simulated‑annealing hill‑climb for a final polish.

A variety of random initialisations is tried (Gaussian, exponential decay,
power‑law, edge‑heavy, arctan‑shape, Beta‑shaped and the current best).  The
search is time‑bounded (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from numpy.fft import fft, ifft

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = fft(a, fft_len)
    conv = ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
         2·n·max(conv)   (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                       # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                            # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (reverse convolution)
    fa_rev = fft(a[::-1], fft_len_g)
    fw = fft(w, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    # the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

# Initialisation helpers (many diverse modes)
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes:
        random – standard normal
        exp    – exponential decay
        power  – power‑law decay
        edge   – heavy mass at the two edges
        arcsine– 1/sqrt(i*(n‑i)) profile (U‑shaped)
        beta   – Beta(α,α) with random α∈(0.1,0.8) (U‑shaped)
        binary – random 0/1 mask
        bump   – a single Gaussian bump
        best   – perturbed copy of the current best (handled outside)
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n) + 0.5
        a0 = 1.0 / np.sqrt(i * (n + 1 - i))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "beta":
        # draw a shape parameter α < 1 to obtain a U‑shaped beta density
        alpha = rng.uniform(0.1, 0.8)
        i = np.arange(n) + 0.5
        t = i / n
        a0 = (t ** (alpha - 1)) * ((1 - t) ** (alpha - 1))
        a0 += 1e-12
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0

# Local search helpers

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep the move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def accelerated_subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.50,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected accelerated sub‑gradient descent on the true objective.
    Uses Nesterov‑style extrapolation and a 1/√t step‑size schedule.
    """
    a = a_start.copy()
    a_prev = a.copy()
    t_prev = 1.0
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute sub‑gradient at the extrapolated point
        beta = (t_prev - 1) / (t_prev + 2)
        y = a + beta * (a - a_prev)

        conv_y = conv_fft(y, fft_len, n)
        m = int(np.argmax(conv_y))                     # index of maximal entry

        # sub‑gradient of that entry (vector of length n)
        g = np.zeros_like(y)
        start = max(0, m - (n - 1))
        end = min(m, n - 1)
        idx = np.arange(start, end + 1)
        rev = m - idx
        g[idx] = 2.0 * y[rev]

        eta = eta0 / np.sqrt(it)                       # classic diminishing step
        a_new = y - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        # update Nesterov parameters
        a_prev = a
        a = a_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        t_prev = t_new

    return best_a, best_score


def simulated_annealing_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 30_000,
    start_temp: float = 0.02,
    end_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climb; accepts occasional worsening moves.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score

    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        # linear temperature schedule
        T = start_temp + (end_temp - start_temp) * (it / max_iter)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.25
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta

        new_score = fast_score(cur, fft_len, n)
        dE = new_score - cur_score

        if dE < 0 or rng.random() < np.exp(-dE / max(T, 1e-12)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject – revert
            cur[i] += delta
            cur[j] -= delta

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) with a low
    evaluation score.  The search runs for ``budget_s`` seconds and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # a small safety margin

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    # Base size 5000 – adapt if a longer external seed is supplied.
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # parameters for soft‑max
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS + short hill‑climb (≈ 55 % time)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=25))
    iteration = 0
    modes = ["random", "exp", "power", "edge", "arcsine",
             "beta", "binary", "bump", "best"]
    phase1_deadline = start_time + 0.55 * budget_s

    while time.time() < phase1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.025, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # Continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # Short greedy hill‑climb to polish the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – accelerated projected sub‑gradient descent (≈ 30 %)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting accelerated sub‑gradient descent")
    best_a, best_score = accelerated_subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.55,
        max_iters=600_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated‑annealing polishing (use remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:

        best_a, best_score = simulated_annealing_climb(
            best_a,

            fft_len_conv,
            n,
            deadline,

        )

    final_seq = best_a.tolist()

    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
