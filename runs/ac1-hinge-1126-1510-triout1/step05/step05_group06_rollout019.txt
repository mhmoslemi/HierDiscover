    """

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists

    # Check each element in the list for validity

"""
A fast multi‑phase optimizer for the step‑function evaluation metric

    score = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works on the probability simplex (sum = 1) so the denominator
is constant and the objective reduces to 2·n·max(conv).  It consists of:

1.  A cheap parametric sweep producing a few diverse seeds.
2.  Projected sub‑gradient descent on the exact max‑convolution.
3.  A SciPy‑based linear‑programming “M‑balancing’’ step:
    maximise Σ b_j  subject to  b≥0 and (a * b) ≤ M (current peak).
    The solution b is mixed back into a to strictly reduce the peak.
4.  A final greedy hill‑climb that transfers a tiny random mass between
    two coordinates while preserving the simplex.

All phases respect the simplex constraint and use FFT‑based convolutions
for speed.  The total 1000 s budget is split automatically; the best
sequence found before the deadline is returned as a plain list.
"""

import time
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import linprog
import scipy.linalg


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) – a is assumed to sum to 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Returns the best vector found before ``deadline``.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # exact convolution and active set
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# LP “M‑balancing’’ step (SciPy implementation)
# ----------------------------------------------------------------------
def lp_improve_scipy(a: np.ndarray,
                     n: int,
                     deadline: float,
                     t_init: float = 0.2,
                     max_iters: int = 12):
    """
    Perform a few iterations of the classic LP balancing step using SciPy.
    At each iteration we solve

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k

    where M = max(a*a).  The resulting b is mixed back into a with factor
    ``t``; ``t`` is halved whenever a step fails to improve the objective.
    """
    a_cur = a.copy()
    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        # current peak
        conv = full_conv(a_cur, next_pow2(2 * n - 1))
        M = conv.max()

        # build convolution matrix T (size (2n‑1) × n)
        # T @ b = a_cur * b (linear convolution)
        col = np.concatenate([a_cur, np.zeros(n - 1)], dtype=np.float64)
        row = np.concatenate([a_cur[:1], np.zeros(n - 1)], dtype=np.float64)
        T = scipy.linalg.toeplitz(col, row)

        # linear programme: maximise Σ b -> minimise -Σ b
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # time‑limit for the solver is the remaining budget
        solver_time = max(0.0, deadline - time.time())
        try:
            res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                          method='highs',
                          options={'presolve': True,
                                   'time_limit': solver_time})
        except Exception as e:
            print("[lp] linprog exception:", e)
            break

        if not res.success or res.x is None:
            # LP failed – exit early
            break

        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break

        # scale b back to the simplex
        b_scaled = b_val / s

        # mixture and re‑normalisation
        a_new = (1.0 - t) * a_cur + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a_cur, n, next_pow2(2 * n - 1))

        if new_score < old_score - 1e-12:
            a_cur = a_new
            t = t_init               # reset mixing factor
        else:
            t *= 0.5
            if t < 1e-4:
                break

        it += 1

    return a_cur


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step sequence.
    Returns a plain list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Choose problem size and an initial distribution
    # ------------------------------------------------------------------
    max_n = 2500
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0[a0 < 0] = 0.0
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, n, fft_len)
    print(f"[init] n={n}, init_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (collect seeds)
    # ------------------------------------------------------------------
    seeds = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        seeds.append(arr)

    # uniform
    add_candidate(np.full(n, 1.0 / n))

    # linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # exponential families
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # sinusoidal‑squared
    add_candidate(np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2 + 1e-3)

    # random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0]:
        for _ in range(3):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # keep the best few seeds according to the fast score
    seed_scores = [(fast_score(s, n, fft_len), s) for s in seeds]
    seed_scores.sort(key=lambda t: t[0])
    top_seeds = [s for _, s in seed_scores[:6]]
    print(f"[sweep] best seed score={seed_scores[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    phase1_deadline = start + 0.65 * budget_s      # sub‑gradient phase
    phase2_deadline = start + 0.90 * budget_s      # LP phase (remaining time)

    # ------------------------------------------------------------------
    # Phase 1 – projected sub‑gradient descent from each seed
    # ------------------------------------------------------------------
    for seed_vec in top_seeds:
        if time.time() >= phase1_deadline:
            break
        a_start = seed_vec + rng.random(n) * 1e-4
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, sc_opt = subgradient_optimize(a_start,
                                              phase1_deadline,
                                              fft_len,
                                              fft_len,
                                              n)
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[sg] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP “M‑balancing’’ steps (SciPy linprog)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("[lp] starting SciPy LP balancing")
        best_a = lp_improve_scipy(best_a,
                                  n,
                                  phase2_deadline,
                                  t_init=0.25,
                                  max_iters=12)
        best_score = fast_score(best_a, n, fft_len)
        print(f"[lp] post‑LP score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final hill‑climb polishing (use whatever time remains)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len,
                                    n,
                                    final_deadline,
                                    max_iter=20000)

    # ------------------------------------------------------------------
    # Return the sequence (plain list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)    # external function supplied by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
