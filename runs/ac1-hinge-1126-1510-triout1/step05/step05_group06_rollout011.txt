def lp_optimize(a, n, deadline, max_iters=20):
    best_a = a.copy()
    best_score = fast_score(best_a, n, next_pow2(2*n-1))
    # Pre-build convolution matrix building may be heavy each iteration,
    # but we can construct using sparse matrix each iteration using diags.
    # We'll compute T each time because a changes.
    while time.time() < deadline and max_iters > 0:
        # Build constraints matrix
        # For each k from 0 to 2*n-2, row k has a_i where i in [max(0,k-n+1), min(k,n-1)].
        # Using scipy.sparse.diags with offsets.
        data = [np.copy(a) for _ in range(n)] # each diagonal
        offsets = np.arange(n)
        T = diags(data, offsets, shape=(2*n-1, n), format="csc")
        # construct inequality: T @ b - t * 1 <= 0.
        # linprog expects A_ub @ x <= b_ub.
        # Variables: [b (n), t].
        # So we need A_ub shape (2n-1, n+1).
        # We'll create by stacking T and -1 column.
        A_ub = np.hstack([T.toarray(), -np.ones((2*n-1, 1))])
        # For equality: sum(b) = 1
        A_eq = np.hstack([np.ones((1, n)), np.zeros((1, 1))])
        bounds = [(0, None)] * n + [(0, None)] # t >=0
        # solve
        res = linprog(c=np.array([0]*n + [1]), A_ub=A_ub, b_ub=np.zeros(2*n-1),
                      A_eq=A_eq, b_eq=np.array([1]), bounds=bounds,
                      method='highs', options={'time_limit': max(0, deadline - time.time())})
```

def exp_gradient_optimize(a0, deadline):
    a = a0.copy()
    best_a, best_score = a.copy(), fast_score(a, n, fft_len)
    it = 0
    while time.time() < deadline:
        # compute mask where conv is near max
        conv = full_conv(a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[n-1:n-1+n]  # n-long gradient
        # step size
        eta = 0.5 / (it + 1)   # decreasing
        # exponentiated update
        a = a * np.exp(-eta * g)
        a = simplex_projection(a)   # ensure normalization
        score = fast_score(a, n, fft_len)
        if score < best_score:
            best_score, best_a = score, a.copy()
        it += 1
    return best_a, best_score

"""
Propose a low‑evaluation non‑negative step sequence.

Algorithm
---------

1. **Parameter sweep** – generate a handful of handcrafted families
   (uniform, linear, exponential, power‑law, edge‑heavy,
   sinusoid, symmetric geometric) and keep the best few seeds.

2. **Continuation L‑BFGS** – optimise a smoothed version of the
   objective (log‑sum‑exp approximation of the max‑convolution) on the
   probability simplex.  The schedule of smoothing factors (λ) is
   progressively refined; many short restarts are performed.

3. **Exponentiated‑gradient descent** – minimise the exact max‑convolution
   using a mirror‑descent step with the (sub‑)gradient of the active
   constraints and an entropic projection (simplex projection).

4. **Linear‑programming balancing** – solve the LP

        minimise   t
        s.t.       conv(a,b) ≤ t   (elementwise)
                   Σ b = 1 ,  b ≥ 0

   (using `scipy.optimize.linprog`).  The solution `b` is mixed back into
   the current vector to strictly lower the peak.

5. **Final greedy hill‑climb** – repeatedly transfer a tiny random
   amount of mass between two coordinates while preserving the simplex.

All steps respect the simplex constraint, use FFT‑based convolutions
(`O(n log n)`) and stop automatically when the allotted time (default
1000 s) runs out.  The best sequence found is returned as a plain list
of floats.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and its gradient – for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple restarts)
# ----------------------------------------------------------------------
def run_lbfgs_restart(a_start: np.ndarray,
                      lam_schedule,
                      maxiter_per_lam,
                      deadline,
                      fft_len_conv,
                      fft_len_g,
                      n):
    """
    Run continuation L‑BFGS from a_start.
    Returns the best sequence found during this run.
    """
    # initialise soft‑max parameters
    y = np.log(a_start + 1e-12)
    best_a = softmax(y)
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # occasional numerical problems – skip this λ
            print("[lbfgs] exception:", e)
            continue

        if not res.success:
            # still accept the point (might be better)
            pass
        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
    return best_a, best_score

    rng = np.random.default_rng()

    restarts = 0
    while time.time() < budget_deadline and restarts < max_restarts:
        # Choose a random seed and jitter it slightly
        seed_vec = rng.choice(seeds)
        jitter = rng.random(n) * 1e-3
        a0 = seed_vec + jitter
        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

        a_opt, sc_opt = run_lbfgs_restart(a0, lam_schedule,
                                          maxiter_per_lam,
                                          budget_deadline,
                                          fft_len_conv,
                                          fft_len_g,
                                          n)
        if sc_opt < best_score:
            best_a, best_score = a_opt.copy(), sc_opt
            print(f"[L‑BFGS] new best score={best_score:.6f}")

        restarts += 1
    # fallback: if not improved keep first seed
    if best_a is None:
        best_a = seeds[0].copy()
        best_score = fast_score(best_a, n, fft_len_conv)
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def exp_gradient_optimize(a_start: np.ndarray,
                          n: int,
                          fft_len_conv: int,
                          fft_len_g: int,
                          deadline: float,
                          max_iters: int = 8000):
    """Mirror‑descent (Kullback‑Leibler) on the exact max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0
    while time.time() < deadline and it < max_iters:
        # current convolution and active mask
        fa = fft(a, fft_len_conv)
        conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv_full.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv_full >= max_val - eps).astype(float)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_rev = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_rev[n - 1:n - 1 + n]

        # step size (diminishing)
        eta = 0.5 / (it + 1.0)

        # exponentiated (multiplicative) update + projection onto simplex
        a = a * np.exp(-eta * g)
        a = simplex_projection(a)

        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming (LP) balancing step – uses SciPy linprog (highs)
# ----------------------------------------------------------------------
def lp_balance(a: np.ndarray,
               n: int,
               deadline: float,
               max_iters: int = 6,
               mix_alpha: float = 0.2):
    """
    Repeatedly solve
          minimise   t
          s.t.       conv(a,b) ≤ t,  Σb = 1,  b ≥ 0
    and mix the result back into a.  Stops when no improvement.
    """
    best_a = a.copy()
    best_score = fast_score(best_a, n, next_pow2(2 * n - 1))
    fft_len = next_pow2(2 * n - 1)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # Build Toeplitz convolution matrix T (2n‑1)×n
        data = [np.copy(a) for _ in range(n)]
        offsets = np.arange(n, dtype=int)
        T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear programme:
        # minimise   t
        # subject to  T @ b - t ≤ 0
        #             Σ b = 1
        #             b ≥ 0, t ≥ 0
        A_ub = sp.hstack([T, -sp.csc_matrix(np.ones((2 * n - 1, 1)))], format="csc")
        b_ub = np.zeros(2 * n - 1)
        A_eq = sp.hstack([sp.csc_matrix(np.ones((1, n))),
                          sp.csc_matrix(np.zeros((1, 1)))], format="csc")
        b_eq = np.array([1.0])
        c = np.concatenate([np.zeros(n), np.array([1.0])])   # minimise t
        bounds = [(0, None)] * n + [(0, None)]

        # Solve with HiGHS (interior‑point)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds,
                          method='highs')
        except Exception as e:
            print("[lp] linprog exception:", e)
            break

        if not res.success or res.x is None:
            print("[lp] linprog failed or no solution")
            break

        b_opt = res.x[:n]
        # Mix a bit of the new vector into the current one
        a_new = (1.0 - mix_alpha) * a + mix_alpha * b_opt
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, fft_len)
        if new_score + 1e-12 < best_score:
            best_score, best_a = new_score, a_new.copy()
            a = a_new                     # continue from improved point
            # keep mixing factor steady
        else:
            # if no improvement, reduce the mixing factor
            mix_alpha *= 0.5
            if mix_alpha < 1e-4:
                break
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, simplex‑preserving)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """Greedy improvement by moving a tiny random mass between two coordinates."""
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # move at most 2 % of the source coordinate
        delta = best[i1] * rng.random() * 0.02
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta
        i += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point – the function that will be called by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step sequence that minimises
        2·n·max(conv(a,a)) / (sum(a)²)

    Returns a plain list of non‑negative floats (the sequence).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # -------------------- problem size & initial vector --------------------
    max_n = 2500                     # safe upper bound for LP matrix size
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = np.maximum(prev[:n], 0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a convolution
    fft_len_g = next_pow2(3 * n - 2)      # length for gradient‑related convolutions

    # -------------------- cheap parametric sweep --------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len_conv)
        candidates.append((sc, arr))

    # Uniform
    add_candidate(np.full(n, 1.0 / n))

    # Linear decreasing (edges heavy)
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # Exponential families
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # Edge‑heavy
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # Sinusoid‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # Symmetric geometric families (r in log‑space)
    for r in np.logspace(-3, -0.05, num=30):
        idx = np.arange(n, dtype=float)
        geom = r ** idx + r ** (n - 1 - idx)
        add_candidate(geom)

    # Random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(3):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Keep a handful of the best seeds
    candidates.sort(key=lambda tup: tup[0])
    top_seeds = [arr for _score, arr in candidates[:8]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # -------------------- time budget split --------------------
    start = time.time()
    deadline = start + budget_s - 0.5        # final safety margin
    phase1_deadline = start + 0.45 * budget_s   # L‑BFGS
    phase2_deadline = start + 0.70 * budget_s   # exp‑gradient
    phase3_deadline = start + 0.90 * budget_s   # LP balancing
    # remaining time → final hill‑climb

    # -------------------- Phase 1 – continuation L‑BFGS --------------------
    best_a, best_score = lbfgs_phase(top_seeds, n,
                                     fft_len_conv, fft_len_g,
                                     phase1_deadline,
                                     max_restarts=12)
    print(f"[phase1] best score after L‑BFGS = {best_score:.6f}")

    # -------------------- Phase 2 – exponentiated‑gradient descent --------------------
    if time.time() < phase2_deadline:
        best_a, best_score = exp_gradient_optimize(best_a, n,
                                                   fft_len_conv,
                                                   fft_len_g,
                                                   phase2_deadline,
                                                   max_iters=12000)
        print(f"[phase2] score after exp‑gradient = {best_score:.6f}")

    # -------------------- Phase 3 – LP balancing --------------------
    if time.time() < phase3_deadline:
        best_a, best_score = lp_balance(best_a, n,
                                        phase3_deadline,
                                        max_iters=8,
                                        mix_alpha=0.15)
        print(f"[phase3] score after LP balancing = {best_score:.6f}")

    # -------------------- Final hill‑climb polishing --------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=15000)

    # -------------------- Return the best sequence --------------------
    final_seq = best_a.tolist()
    # Force the external evaluator to compute the official score (shown for logging)
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
