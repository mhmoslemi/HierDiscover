def propose_candidate(seed=123, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    max_n = 4000
    n = max_n

    # Use length n for all candidates
    fft_len = next_pow2(2*n - 1)
    fft_len_g = next_pow2(3*n - 2)

    # Helper to compute score quickly given a (sum to 1)
    def fast_score_arr(arr):
        conv = full_conv(arr, fft_len)  # returns length 2n-1
        return float(2.0 * n * conv.max())

    # Parametric families
    candidates = []
    x = np.arange(n, dtype=float)

    # Uniform
    a = np.full(n, 1.0/n)
    candidates.append(('uniform', a))

    # Triangular (peak at center)
    tri = np.minimum(x+1, n-x)
    a = tri / tri.sum()
    candidates.append(('triangular', a))

    # Quadratic (parabola)
    quad = (x+1)*(n-x)
    a = quad / quad.sum()
    candidates.append(('quadratic', a))

    # Sine
    sine = np.sin(np.pi * (x+0.5) / n)
    a = sine / sine.sum()
    candidates.append(('sine', a))

    # Sine^2
    sine2 = np.sin(np.pi * (x+0.5) / n) ** 2
    a = sine2 / sine2.sum()
    candidates.append(('sine2', a))

    # Cosine window (Hann)
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * x / (n-1))
    a = hann / hann.sum()
    candidates.append(('hann', a))

    # Hamming
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n-1))
    a = hamming / hamming.sum()
    candidates.append(('hamming', a))

    # Power law with random exponent
    for _ in range(5):
        p = rng.uniform(0.2, 2.5)
        powv = (x+1) ** (-p)
        a = powv / powv.sum()
        candidates.append((f'power_{p:.2f}', a))

    # Exponential decay
    for _ in range(5):
        lam = rng.uniform(0.0005, 0.02)
        expv = np.exp(-lam * x)
        a = expv / expv.sum()
        candidates.append((f'exp_{lam:.4f}', a))

    # Arcsine-like (beta 0.5,0.5) approximated
    eps = 1e-6
    beta = np.sqrt(((x+0.5)/n) * (1 - (x+0.5)/n) + eps)
    arcs = 1.0 / beta
    a = arcs / arcs.sum()
    candidates.append(('arcsine', a))

    # Edge heavy (non-uniform)
    edge_frac = 0.02
    m_edge = max(1, int(n * edge_frac))
    edge = np.full(n, 1.0/n)
    heavy = np.sqrt(n/2.0) / n
    edge[:m_edge] = heavy
    edge[-m_edge:] = heavy
    a = edge / edge.sum()
    candidates.append(('edge', a))

    # Bimodal
    centre = n // 2
    width = rng.integers(int(0.02*n), int(0.07*n))
    bimodal = np.full(n, 1.0/n)
    left = max(0, centre - width)
    right = min(n, centre + width)
    peak = rng.uniform(0.5, 1.5) / n
    bimodal[left:right] = peak
    a = bimodal / bimodal.sum()
    candidates.append(('bimodal', a))

    best_a = None
    best_score = np.inf

    for name, arr in candidates:
        score = fast_score_arr(arr)
        if score < best_score:
            best_score = score
            best_a = arr.copy()
            print(f"[init] candidate {name} score={score:.6f}")

    # Ensure we have current best
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_score = fast_score_arr(best_a)

    # Now random restart L-BFGS
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    phase1_end = start_time + 0.45 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.90 * budget_s
    # L-BFGS parameters
    lam_schedule = [0.6,0.4,0.3,0.2,0.1,0.05,0.03,0.02,0.01,0.005,0.002,0.001,0.0005,0.0002,0.0001]
    maxiter_per_lam = 1200

    # Helper: get y (logits) from a
    def logits_from_a(a):
        return np.log(a + 1e-15)

    while time.time() < phase1_end:
        if best_score < np.inf and rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            # Random initial from families
            a_start = random_initial(n, rng)
        # L-BFGS continuation
        y = np.log(a_start + 1e-15)
        best_local_a = a_start.copy()
        best_local_score = fast_score_arr(best_local_a)

        for lam in lam_schedule:
            if time.time() >= phase1_end:
                break
            try:
                res = minimize(fun=objective_and_grad,
                               x0=y,
                               args=(n, fft_len, fft_len_g, lam),
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': maxiter_per_lam,
                                        'gtol': 1e-12,
                                        'disp': False})
            except Exception as e:
                continue
            y = res.x
            a_new = softmax(y)
            score = fast_score_arr(a_new)
            if score < best_local_score:
                best_local_score = score
                best_local_a = a_new.copy()
        # update global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local_a.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

    # Subgradient with momentum
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(best_a, phase2_end, fft_len, fft_len_g, n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # Adam fine‑tuning
    if time.time() < phase3_end:
        logits = logits_from_a(best_a)
        best_a, best_score = adam_optimize(logits, n, fft_len, fft_len_g, phase3_end,
                                           lam_factor=0.0015, lr=0.12)
        print(f"[Adam] score={best_score:.6f}")

    # Simulated annealing
    cur_deadline = deadline
    if time.time() < cur_deadline:
        best_a, best_score = random_anneal(best_a, best_score, fft_len, n, cur_deadline - 5.0)
        print(f"[anneal] score={best_score:.6f}")

    # Final greedy hill‑climb with more aggressive moves
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a, best_score, fft_len, n, deadline, max_iter=12000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq

"""
propose_candidate
-----------------
Search for a non‑negative step sequence that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the linear (full) convolution.  The search proceeds in
several stages within the 1000 s budget:

1. **Parametric warm‑starts** – a collection of analytically‑defined
   windows (uniform, triangular, quadratic, sine, Hann, Hamming, power
   laws, exponentials, arcsine‑like, edge‑heavy, bimodal) are generated,
   normalised to sum = 1 and the best is kept.

2. **Random‑restart continuation L‑BFGS** – the smooth “log‑sum‑exp”
   approximation of the max‑convolution is minimised.  A schedule of
   decreasing smoothing parameters (``lam_factor``) is used; each stage
   runs a limited number of L‑BFGS iterations.  Restarts are taken from
   the current best solution (with added Gaussian noise) as well as from
   the random families.

3. **Projected sub‑gradient descent with momentum** – after the smooth
   stage the true non‑smooth objective is attacked directly by a
   sub‑gradient method that projects onto the probability simplex.

4. **Adam fine‑tuning** on the smoothed objective with a tighter
   smoothing factor.

5. **Simulated annealing** – stochastic mass‑moves between two
   coordinates accept worsening moves with a decaying temperature.

6. **Greedy hill‑climb** – a final deterministic pass that moves a
   small fraction of the mass between random pairs only when the move
   strictly improves the objective.

All heavy operations use FFT‑based convolutions (O(n log n)).  The
function returns the best list of floats found before the deadline.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT helpers and simple utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z    # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random probability vector using several heuristics."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    x = np.arange(n, dtype=float)

    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * x)
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (x + 1.0) ** (-p)
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:        # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * full_conv(a, fft_len_conv).max()

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        step = 0.8 / np.sqrt(it + 1.0)   # decreasing step
        a_candidate = simplex_projection(a - step * v)

        cand_score = 2.0 * n * full_conv(a_candidate, fft_len_conv).max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation on the smoothed log‑sum‑exp max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = 2.0 * n * full_conv(best_a, fft_len_conv).max()

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = 2.0 * n * full_conv(a, fft_len_conv).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simulated annealing: move a random fraction of mass between two
    coordinates, accepting worsening moves with a decaying temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.12
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = 2.0 * n * full_conv(cand, fft_len).max()
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 4000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy hill‑climb: move a small random fraction of the mass between
    two coordinates and keep the move only if it strictly improves the
    objective.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue

        best[i] -= delta
        best[j] += delta

        new_score = 2.0 * n * full_conv(best, fft_len).max()
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (fixed to 4000 – a good trade‑off between quality
    # and speed)
    # ------------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Parametric warm‑starts
    # ------------------------------------------------------------------
    x = np.arange(n, dtype=float)

    candidates = []

    # Uniform
    a = np.full(n, 1.0 / n)
    candidates.append(('uniform', a))

    # Triangular (tent)
    tri = np.minimum(x + 1, n - x)
    a = tri / tri.sum()
    candidates.append(('triangular', a))

    # Quadratic (parabolic)
    quad = (x + 1) * (n - x)
    a = quad / quad.sum()
    candidates.append(('quadratic', a))

    # Sine (single half‑period)
    sine = np.sin(np.pi * (x + 0.5) / n)
    a = sine / sine.sum()
    candidates.append(('sine', a))

    # Sine squared
    sine2 = sine ** 2
    a = sine2 / sine2.sum()
    candidates.append(('sine2', a))

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2 * np.pi * x / (n - 1))
    a = hann / hann.sum()
    candidates.append(('hann', a))

    # Hamming window
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    a = hamming / hamming.sum()
    candidates.append(('hamming', a))

    # Random power‑law families
    for _ in range(5):
        p = rng.uniform(0.2, 2.5)
        powv = (x + 1) ** (-p)
        a = powv / powv.sum()
        candidates.append((f'power_{p:.2f}', a))

    # Random exponentials
    for _ in range(5):
        lam = rng.uniform(0.0005, 0.02)
        expv = np.exp(-lam * x)
        a = expv / expv.sum()
        candidates.append((f'exp_{lam:.4f}', a))

    # Arcsine‑like (Beta(0.5,0.5)) – clipped to avoid infinities
    eps = 1e-8
    beta = np.sqrt(((x + 0.5) / n) * (1 - (x + 0.5) / n) + eps)
    arcs = 1.0 / beta
    a = arcs / arcs.sum()
    candidates.append(('arcsine', a))

    # Edge‑heavy (small fraction of large values at both ends)
    edge_frac = 0.02
    m_edge = max(1, int(n * edge_frac))
    edge = np.full(n, 1.0 / n)
    heavy = np.sqrt(n / 2.0) / n
    edge[:m_edge] = heavy
    edge[-m_edge:] = heavy
    a = edge / edge.sum()
    candidates.append(('edge', a))

    # Bimodal bump
    centre = n // 2
    width = rng.integers(int(0.02 * n), int(0.07 * n))
    bimodal = np.full(n, 1.0 / n)
    left = max(0, centre - width)
    right = min(n, centre + width)
    peak = rng.uniform(0.5, 1.5) / n
    bimodal[left:right] = peak
    a = bimodal / bimodal.sum()
    candidates.append(('bimodal', a))

    # ------------------------------------------------------------------
    # Choose the best parametric starter
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    def fast_score_arr(arr):
        """Fast objective for a vector already normalised to sum = 1."""
        conv = full_conv(arr, fft_len_conv)
        return float(2.0 * n * conv.max())

    for name, arr in candidates:
        s = fast_score_arr(arr)
        if s < best_score:
            best_score = s
            best_a = arr.copy()
        print(f"[parametric] {name:12s} score={s:.6f}")

    # ------------------------------------------------------------------
    # 2) Random‑restart continuation L‑BFGS (smooth approximation)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # keep a few seconds for final eval
    phase1_end = start_time + 0.45 * budget_s
    phase2_end = start_time + 0.70 * budget_s
    phase3_end = start_time + 0.90 * budget_s

    lam_schedule = [0.6, 0.4, 0.3, 0.2, 0.15, 0.1,
                    0.07, 0.05, 0.03, 0.02, 0.01,
                    0.006, 0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 900

    def logits_from_a(a):
        return np.log(a + 1e-15)

    # Main L‑BFGS loop (random restarts)
    while time.time() < phase1_end:
        # Occasionally perturb the current best to explore its neighbourhood
        if best_score < np.inf and rng.random() < 0.2:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        y = logits_from_a(a_start)
        best_local_a = a_start.copy()
        best_local_score = fast_score_arr(best_local_a)

        for lam in lam_schedule:
            if time.time() >= phase1_end:
                break
            try:
                res = minimize(fun=objective_and_grad,
                               x0=y,
                               args=(n, fft_len_conv, fft_len_g, lam),
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': maxiter_per_lam,
                                        'gtol': 1e-12,
                                        'disp': False})
            except Exception:
                continue
            y = res.x
            a_new = softmax(y)
            score_new = fast_score_arr(a_new)
            if score_new < best_local_score:
                best_local_score = score_new
                best_local_a = a_new.copy()

        if best_local_score < best_score:
            best_score = best_local_score
            best_a = best_local_a.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Adam fine‑tuning on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        logits = logits_from_a(best_a)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           phase3_end,
                                           lam_factor=0.0015,
                                           lr=0.12)
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           deadline - 2.0)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=12000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (list of floats) and print the
    # evaluation by the provided external function.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
