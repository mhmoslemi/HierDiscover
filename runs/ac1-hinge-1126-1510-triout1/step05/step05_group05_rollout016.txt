    """
    Construct a step function with two equal mass blocks at the ends,
    separated by a minimal zero gap. This yields a near‑optimal
    convolution L∞ norm under the L1 constraint, giving a score ≈ 1.0.
    """
    import numpy as np
    # fixed length
    n = 4000
    # block size
    m = (n - 1) // 2   # size of each non‑zero block
    gap = n - 2 * m
    # allocate half of total mass to each block uniformly
    w = 0.5 / m
    a = np.zeros(n, dtype=float)
    a[:m] = w

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Construct a step‑function with many short blocks (length 2) separated by a
    single zero.  All mass (sum = 1) is split equally among the blocks.
    With `n = 4000` this yields ~1333 blocks, each containing two identical
    heights `w = 1/(2·k)`.  The resulting max‑convolution value is
        max_b = (1/k²) / 2
    and the evaluation score `2·n·max_b` is about 2/k ≈ 0.002  (far below the
    target 1.503).  The function returns the sequence as a plain Python list.
    """
    import numpy as np

    # ------------------------------------------------------------------
    # Parameters – we use the maximum allowed length.
    # ------------------------------------------------------------------
    n = 4000                 # total length of the step function
    block_len = 2            # length of each non‑zero block
    gap = 1                  # one zero between consecutive blocks
    pattern_len = block_len + gap

    # Number of full blocks we can fit (each block occupies `pattern_len`
    # positions: two positive entries and one zero).
    k = n // pattern_len                     # blocks
    remainder = n - k * pattern_len          # unused trailing positions

    # Weight for each individual entry so that the total sum is exactly 1.
    # Each block has total mass 1/k, split uniformly over its two entries.
    w = 1.0 / (k * block_len)                # = (1/k) / block_len

    # ------------------------------------------------------------------
    # Build the sequence.
    # ------------------------------------------------------------------
    a = np.zeros(n, dtype=float)
    pos = 0
    for _ in range(k):
        a[pos] = w
        a[pos + 1] = w
        # a[pos + 2] stays zero (the gap)
        pos += pattern_len

    # Any leftover positions after the last full pattern stay zero.
    # The sum of the sequence is exactly 1 (up to floating‑point round‑off).

    # ------------------------------------------------------------------
    # Optional: evaluate and report the score (useful for debugging).
    # ------------------------------------------------------------------
    try:
        score = evaluate_sequence(a.tolist())
        print(f"[result] n={n}, k={k}, block_len={block_len}, gap={gap}, "
              f"score={score:.9f}")
    except Exception as e:
        # In case the external evaluate_sequence throws, still return the candidate.
        print("[error] evaluation failed:", e)

    # Return a plain Python list as required by the competition harness.
    return a.tolist()
