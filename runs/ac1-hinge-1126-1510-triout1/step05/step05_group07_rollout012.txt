"""
Propose a low‑score step‑function using a multi‑stage stochastic optimiser.

The optimiser works on several problem sizes (default: 2000, 4000, 6000, 8000)
and spends (≈ equal) time on each size.  For a single size it repeatedly:

1.   Generates a diverse initial point (uniform, random Gaussian logits,
     exponential, power‑law, edge‑biased or a perturbation of the current
     best).
2.   Refines it with a continuation L‑BFGS on a smooth log‑sum‑exp
     approximation of the max‑convolution.  The smoothing factor λ (controlled
     by ``lam_factor``) is decreased geometrically.
3.   Performs an accelerated projected sub‑gradient descent on the exact
     max‑convolution (objective = 2·n·max(a∗a) with sum(a)=1).  The step size
     decays as 1/√t and the iterate is projected onto the probability simplex.
4.   Applies a greedy “targeted” hill‑climb that moves mass from the largest
     contributor to the smallest contributor of the current maximal convolution
     entry.
5.   Finishes with a short random hill‑climb for polishing.

All intermediate evaluations use a fast FFT‑based convolution (``fast_score``),
which for a normalised vector (∑a=1) returns the exact value of the
evaluation function up to the constant factor 2·n.  After the time budget is
exhausted the best sequence found across all sizes is evaluated with the
provided ``evaluate_sequence`` and returned.

The algorithm builds on the existing hybrid optimiser but adds:
    * a larger set of problem sizes (up to 8000 entries),
    * an accelerated projected sub‑gradient method (Nesterov momentum),
    * a longer and finer L‑BFGS continuation schedule,
    * slightly larger step‑size in the sub‑gradient phase,
    * more restarts and tighter time slicing.
These improvements give the optimiser more chance to push the score
below the previous 1.509 → 1.503 target.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (real part, trimmed to length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact value of 2·n·max(a*a) for a probability vector a."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1} (Wang‑Carreira‑Perpiñán)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Smooth approximation of max(a*a) and its gradient w.r.t. logits y.

    The smoothing parameter is λ = lam_factor * max(c) + 1e-12,
    where c = a*a.  The log‑sum‑exp provides a smooth approximation of max.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                     # make max zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                           # ∂smooth_max/∂conv

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1  (correlation)
    # we compute via convolution with reversed a
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS with a decreasing smoothing schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Accelerated projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def accelerated_subgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Nesterov‑accelerated projected sub‑gradient descent on
    f(a)=2·n·max(a*a) (sum(a)=1).

    Returns the best vector seen and its fast_score.
    """
    a = a0.copy()
    y = a.copy()          # momentum point
    t = 1.0                # Nesterov parameter
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    idx = np.arange(n)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current max‑convolution entry
        conv = conv_fft(y, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max at active index k
        j = k - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(y)
        grad[mask] = 2.0 * y[j[mask]]

        eta = eta0 / np.sqrt(it + 1.0)

        a_next = project_simplex(y - eta * grad)

        # Nesterov update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a_next + ((t - 1.0) / t_next) * (a_next - a)
        a = a_next
        t = t_next

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if (it + 1) % 500 == 0 and np.random.rand() < 0.08:
            noise = np.random.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)
            y = a.copy()   # reset momentum after jitter

    return best_a, best_score


# ----------------------------------------------------------------------
# Random and targeted hill‑climbs (as in the reference program)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2500,
) -> tuple[np.ndarray, float]:
    """Simple random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3500,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy move of mass from the largest contributor to the smallest
    contributor of the current maximal convolution entry.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Optimisation driver for a single problem size
# ----------------------------------------------------------------------


def optimise_one_size(
    n: int,
    budget_s: float,
    seed: int,
    prev_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimises a step‑function of length ``n`` inside the allocated time slice.
    Returns the best probability vector (sum = 1) and its fast_score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s

    # ------------------------------------------------------------------
    # Initialise a feasible vector
    # ------------------------------------------------------------------
    if prev_seq is not None and prev_seq.size >= n:
        a0 = prev_seq[:n].copy()
        a0 = np.clip(a0, 0.0, np.inf)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    # smoothing schedule (geometric decay)
    lam_factors = np.geomspace(1.0, 5e-9, num=45)

    iteration = 0

    while time.time() < deadline - 3.0:
        # ---- pick a random initialisation pattern --------------------
        mode = rng.choice(["random", "exp", "power", "edge", "best"])
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # --------------------------------------------------------------
        # 1) Continuation L‑BFGS on a smooth approximation
        # --------------------------------------------------------------
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS fast score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) Accelerated sub‑gradient descent on the exact max‑convolution
        # --------------------------------------------------------------
        sub_deadline = min(deadline, time.time() + 0.9)  # 0.9 s budget per run
        best_a, best_score = accelerated_subgrad(
            best_a,
            n,
            fft_len,
            max_iters=4000,
            deadline=sub_deadline,
            eta0=0.85,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 3) Targeted hill‑climb (quick greedy refinement)
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=3500,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # --------------------------------------------------------------
        # 4) Random hill‑climb for final polishing
        # --------------------------------------------------------------
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Final L‑BFGS refinement with the smallest smoothing factor,
    # using any remaining time.
    # ------------------------------------------------------------------
    while time.time() < deadline - 1.5:
        for lam in reversed(lam_factors):  # start from the smallest λ
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=best_y,
                args=(n, fft_len, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 800, "gtol": 1e-14, "disp": False},
            )
            best_y = res.x
            best_a = softmax(best_y)
            best_score = fast_score(best_a, fft_len, n)
            print(f"[final‑L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Global entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns the best non‑negative sequence (its sum will be 1 after internal
    normalisation).  The function respects the 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose problem sizes. Larger n gives more degrees of freedom.
    # ------------------------------------------------------------------
    sizes = [2000, 4000, 6000, 8000]
    # Allocate a roughly equal slice for each size.
    per_slice = (total_deadline - time.time()) / len(sizes)

    prev_seq = globals().get("height_sequence_1")
    prev_arr = np.asarray(prev_seq, dtype=float) if prev_seq is not None else None

    best_overall_a = None
    best_overall_score = np.inf

    for idx, n in enumerate(sizes):
        remaining = total_deadline - time.time()
        if remaining < 2.0:
            break
        # Ensure we keep at least a couple of seconds for later sizes
        slice_budget = min(per_slice, remaining - (len(sizes) - idx - 1) * 2.0)

        print(f"[size] Optimising n={n} with slice budget ≈{slice_budget:.1f}s")
        a_n, score_n = optimise_one_size(
            n=n,
            budget_s=slice_budget,
            seed=seed + idx * 12345,
            prev_seq=prev_arr,
        )
        print(f"[size] n={n} best fast score = {score_n:.6f}")

        if score_n < best_overall_score:
            best_overall_score = score_n
            best_overall_a = a_n.copy()

        # keep the best result as warm‑start for larger sizes
        prev_arr = best_overall_a.copy()

    # ------------------------------------------------------------------
    # Final evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_overall_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose the best sequence for possible future warm‑starts
    globals()["height_sequence_1"] = final_seq
    return final_seq
