def adam_subgrad(a_start, n, fft_len, deadline, eta=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best = a.copy()
    best_score = fast_score(a, fft_len, n)
    rng = np.random.default_rng()
    while time.time() < deadline:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        # compute subgradient
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # gradient step
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a_new.copy()
        a = a_new
    return best, best_score
```

def simulated_annealing(a_start, n, fft_len, deadline, initial_temp=0.1, final_temp=0.001):
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    # compute temperature schedule based on iteration count
    total_iters_est = 500000
    it = 0
    while time.time() < deadline and it < total_iters_est:
        it += 1
        # temperature linearly decreasing
        T = initial_temp * (1 - it / total_iters_est) + final_temp * (it / total_iters_est)
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # maximum mass we can move from i is a[i]
        max_move = min(a[i], rng.random() * 0.2)  # up to 20% of source
        if max_move <= 0:
            continue
        delta = max_move
        a[i] -= delta

"""
Propose a non‑negative step‑function (list of floats) with a low evaluation
score.

Algorithm (≈ 1000 s budget)
--------------------------------
1. Choose a problem size `n` (default 8000, or the size of the previous
   construction if available).  All vectors are treated as probability
   vectors (∑a=1); the evaluation function is then simply
   `2·n·max(conv(a))`.

2. Initialise several candidate distributions on the simplex:
   * random Gaussian,
   * decreasing exponential (left‑biased or symmetric),
   * power‑law,
   * edge‑heavy,

   * a greedy Sidon set,
   * a perturbation of the current best.

3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp surrogate of
   `max(conv)`.  The smoothing factor `λ = max(conv)·lam_factor` is reduced
   geometrically over 50 stages (λ‑factor from 0.5 down to 1e‑7).  Each stage

   objective.  The sub‑gradient of `max_i (a*a)_i` w.r.t. `a` is
   `g_j = 2·a_{m−j}` where `m` is the index of the maximal convolution
   entry.  Adam moments are maintained and after each step `a` is projected
   onto the probability simplex.

   from one coordinate to another.  Moves that increase the objective are
   accepted with probability `exp((old‑new)/T)`, with temperature `T`
   decreasing linearly.

6. **Aggressive hill‑climb** – a final greedy refinement that only accepts
   strictly improving moves.

7. Return the best vector found.  The harness evaluates the same objective
   (scale‑invariant), so the probability vector can be returned directly.

The whole search respects the 1000 s time limit; each phase reserves a
portion of the remaining budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (∑a = 1):
        2·n·max(a*a)
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, ∑x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx. of max)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where
        f = 2·n·smooth_max(a*a)   (a = softmax(y))
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = max(conv)·lam_factor.
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ----- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c                 # max = 0 now
    exp_shift = np.exp(shifted / lam)      # ≤ 1
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient w.r.t. conv -------------------------------------------
    w = exp_shift / Z                      # ∂smooth_max/∂conv_i

    # ----- propagate gradient through the convolution ----------------------
    # we need grad_a = ∂f/∂a ; then ∂a/∂y = diag(a) - a a^T
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    # indices (n‑1) … (n‑1+n‑1) correspond to i = 0 … n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a a^T)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12,
                     "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Initial distribution generators
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Produce a starting point y (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_sym":
        rate = rng.uniform(0.001, 0.02)
        d = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-rate * d)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "arcsine":
        i = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if (cand + c) in sums or (2 * cand) in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: repeatedly move a random fraction of mass from i to j.
    Accept the move only if the fast objective improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def adam_subgrad(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    eta: float = 0.20,
    beta1: float = 0.90,
    beta2: float = 0.999,
    eps: float = 1e-8,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimisation on the true max‑convolution objective.
    Sub‑gradient for the active convolution entry m is
        g_i = 2·a_{m−i}  (if the index is valid).
    """
    a = a_start.copy()
    m1 = np.zeros_like(a)  # first moment
    m2 = np.zeros_like(a)  # second moment
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t = 0
    rng = np.random.default_rng()

    while time.time() < deadline and t < max_iters:
        t += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # Adam moments
        m1 = beta1 * m1 + (1.0 - beta1) * g
        m2 = beta2 * m2 + (1.0 - beta2) * (g * g)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        step = eta * m1_hat / (np.sqrt(m2_hat) + eps)
        a_new = a - step
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_temp: float = 0.20,
    final_temp: float = 0.001,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing refinement.  Mass is moved from a random source
    to a random target; worsening moves are occasionally accepted with a
    temperature‑dependent probability.
    """
    a = a_start.copy()
    cur_score = fast_score(a, fft_len, n)
    best_a = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_moves:
        it += 1
        # linear temperature schedule
        T = init_temp + (final_temp - init_temp) * (it / max_moves)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j or a[i] == 0.0:
            continue

        # move up to 15 % of source mass
        delta = a[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        a[i] -= delta
        a[j] += delta

        new_score = fast_score(a, fft_len, n)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        else:
            # revert
            a[i] += delta
            a[j] -= delta

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluate_sequence score.
    Returns a plain list of non‑negative floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0        # safety margin

    # ------------------------------------------------------------------
    # Problem size selection
    # ------------------------------------------------------------------
    prev_seq = globals().get("height_sequence_1", None)
    if isinstance(prev_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(prev_seq, dtype=float).ravel()
        # keep at least 4000 entries, cap at 12000 for speed
        n = max(4000, min(tmp.size, 12000))
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability vector)
    # ------------------------------------------------------------------
    if isinstance(prev_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(prev_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / tmp.sum()
        else:
            # truncate / pad to length n and normalise
            best_a = np.zeros(n, dtype=float)
            length = min(tmp.size, n)
            best_a[:length] = tmp[:length]
            best_a /= best_a.sum()
    else:
        best_a = np.full(n, 1.0 / n)

    best_y = np.log(best_a + 1e-12)          # soft‑max parameters
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS on smoothed objective
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-7, num=50))
    # reserve ~350 s for this phase; later phases will take the remainder
    lbfgs_deadline = deadline - 350.0
    iteration = 0
    modes = ["random", "exp", "exp_sym", "power", "edge", "arcsine", "sidon", "best"]

    while time.time() < lbfgs_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=400,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[L‑BFGS {iteration:04d}] mode={mode} → improved fast score = {best_score:.6f}")

        # a quick hill‑climb to polish this L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam sub‑gradient descent on the true max‑conv.
    # ------------------------------------------------------------------
    if time.time() < deadline - 170.0:
        sub_deadline = deadline - 120.0
        print("[Adam sub‑grad] start")
        best_a, best_score = adam_subgrad(
            best_a,
            n,
            fft_len_conv,
            sub_deadline,
            eta=0.22,
            beta1=0.90,
            beta2=0.999,
            max_iters=800_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[Adam sub‑grad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        sa_deadline = deadline - 30.0
        print("[Simulated annealing] start")
        best_a, best_score = simulated_annealing(
            best_a,
            n,
            fft_len_conv,
            sa_deadline,
            init_temp=0.20,
            final_temp=0.001,
            max_moves=600_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[Simulated annealing] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final hill‑climb] start")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=20000,
        )
        print(f"[final hill‑climb] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return result (the harness will evaluate it)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # supplied by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
