def lp_balancing(a, n, deadline, max_iters=15, t_init=0.1):
    import numpy as np
    from scipy.optimize import linprog
    # ensure a simplex normalized
    a = a.copy()
    total = a.sum()
    if total == 0:
        a = np.full_like(a, 1.0/n)
    else:
        a = a / total
    t = t_init
    best_a = a.copy()
    best_score = fast_score(best_a, n, next_pow2(2*n-1))
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = np.convolve(best_a, best_a)
        M = conv.max()
        # Build convolution matrix
        diags_data = [np.full(n, best_a[d]) for d in range(n)]
        offsets = np.arange(n)
        T = sp.diags(diags_data, offsets, shape=(2*n-1, n), format='csc')
        # solve LP
        c = -np.ones(n)  # minimize -sum(b)
        bounds = [(0, None)] * n
        # A_ub x <= b_ub
        A_ub = T
        b_ub = np.full(2*n-1, M)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            break
        b = res.x
        if b.sum() == 0:
            break
        b = b / b.sum()
        # mix
        new_a = (1 - t) * best_a + t * b
        new_a = new_a / new_a.sum()
        new_score = fast_score(new_a, n, next_pow2(2*n-1))
        if new_score < best_score:
            best_a = new_a
            best_score = new_score
            t = t_init
        else:
            t *= 0.5
            if t < 1e-5:
                break
    return best_a, best_score
```

"""
Propose a low‑evaluation step sequence.

Algorithm overview
------------------
1.  Parameter sweep – generate a variety of simple candidate
    sequences (uniform, linear, exponential, power‑law, edge‑heavy,
    sinusoidal‑squared, random Dirichlet draws).  Keep the top few
    seeds (by the exact objective) for further optimisation.

2.  Continuation L‑BFGS – minimise a smooth log‑sum‑exp approximation
    of the max‑convolution.  A schedule of smoothing parameters
    (λ) is used, each stage starting from the previous optimum.
    Multiple restarts are performed from the best seeds.

3.  LP balancing – repeatedly solve the classic linear‑programming
    “M‑balancing’’ step:
        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k,
    where M is the current peak of a*a.
    The LP is solved with SciPy's HiGHS interior‑point solver
    (no external dependencies).  The solution b is mixed back
    into a with a small factor t; t is halved when a step does not
    improve the exact objective.  Several iterations are performed.

4.  Projected sub‑gradient descent – optimise the exact (non‑smoothed)
    objective using the sub‑gradient of the max‑convolution and a
    simple back‑tracking line‑search, always projecting onto the simplex.

5.  Greedy hill‑climb – transfer a tiny random amount of mass between
    two coordinates while preserving the sum; only strictly improving
    moves are kept.  This is used as a final polish.

All phases respect the simplex constraint (non‑negative entries,
sum = 1).  The total running time is limited by the supplied
budget (default 1000 s).  The best sequence found before the deadline
is returned as a plain Python list.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags, csc_matrix

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max landing on the probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective (2·n·max(a*a)) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)                                 # a ≥ 0, Σa = 1

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max: λ is a fraction of the current max (to keep scale)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                    # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t convolution entries
    w = exp_shift / Z                               # ∂smooth_max/∂conv_i
    a_rev = a[::-1]

    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]           # gradient w.r.t a

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple λ values, optional restarts)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result.  Returns the best sequence found and its exact score.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score = {best_score:.6f}")

        # a quick hill‑climb after each λ (tiny budget)
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline, max_iter=800)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact objective
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent with a simple
    back‑tracking line‑search.  Returns best sequence found and its score.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current full convolution
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max (using mask as weight)
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # step size schedule
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer while preserving the simplex)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        # move at most 5 % of the source coordinate’s mass
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# LP “M‑balancing’’ step (SciPy HiGHS LP solver)
# ----------------------------------------------------------------------


def lp_balancing(a: np.ndarray,
                 n: int,
                 deadline: float,
                 t_init: float = 0.12,
                 max_iters: int = 12):
    """
    Perform several iterations of the LP balancing step:

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k,   M = max(a*a).

    After each LP solution b, mix a fraction t of b back into a.
    The mixing factor t is halved whenever a step does not improve the
    exact objective.
    """
    a = a.copy()
    total = a.sum()
    a = a / total if total > 0 else np.full_like(a, 1.0 / n)

    t = t_init
    best_a = a.copy()
    best_score = fast_score(best_a, n, next_pow2(2 * n - 1))

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current peak M
        conv = np.convolve(best_a, best_a)
        M = conv.max()
        if M <= 0:
            break

        # Build convolution matrix T (size (2n‑1)×n) as a sparse CSR matrix.
        # Each column j contributes a[j] on a diagonal offset j.
        diags_data = [np.full(n, best_a[d]) for d in range(n)]
        offsets = np.arange(n)
        T = diags(diags_data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise sum(b)   ↔   minimise -sum(b)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success or res.x is None:
            # if the LP fails, abort the balancing stage
            print("[lp] LP failed or infeasible at iteration", it)
            break

        b = res.x
        s = b.sum()
        if s <= 0:
            break
        # scale to the simplex
        b_scaled = b / s

        # mix
        a_new = (1.0 - t) * best_a + t * b_scaled
        a_new /= a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        if new_score + 1e-12 < best_score:      # strict improvement
            best_a = a_new
            best_score = new_score
            t = t_init    # reset mixing factor after a success
        else:
            t *= 0.5
            if t < 1e-5:
                break

    return best_a, best_score
