"""
Propose a step‑function sequence with a low ‑∞ convolution bound.

Algorithm Overview
------------------
1. **Problem size** – work with n≈4000 (large enough for a low bound,
   still cheap to FFT).  All vectors are kept on the probability simplex
   (sum = 1) because the evaluation function is scale‑invariant.
2. **Initial families** – generate a handful of parametric families
   (exponential, power‑law, edge‑heavy, bimodal, etc.), evaluate them
   quickly and keep the best one as a warm start.
3. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  The smoothing factor is
   gradually reduced; each stage starts from the previous optimum.
4. **Projected sub‑gradient descent** – using the exact sub‑gradient
   of the hard max, push the current solution further down.
5. **Adam on the smooth objective** – fine‑tune with adaptive steps.
6. **Permutation local search** – the ordering of the coefficients
   strongly influences the convolution.  Random swaps (and occasional
   full shuffles) are accepted only if they lower the max‑convolution.
7. **Greedy mass‑transfer hill‑climb** – move a small fraction of mass
   between coordinates to capture any remaining improvement.
8. **Final evaluation** – the external ``evaluate_sequence`` function
   is called once at the end.

The whole routine respects the 1000 s budget, printing progress
messages and always returning the best sequence found before termination.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities (FFT helpers, simplex projection, softmax, etc.)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective when a is already normalised
    (sum(a)=1). Returns 2·n·max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max of (a*a) using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z    # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the gradient components that correspond to a_i
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best a found and its score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Very rare – just skip this stage
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with momentum for the hard max‑objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    # distance from the maximum at which we treat an entry as active
    eps_factor = 1e-12

    while time.time() < deadline:
        # Full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * eps_factor + eps_factor
        mask = (conv >= max_val - eps).astype(np.float64)

        # Sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # Momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/√it
        step = 0.7 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation on the smoothed max‑objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simple simulated annealing that swaps mass between two coordinates.
    Accepts worsening moves with a temperature that decays.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.10            # initial temperature
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # move up to 12 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 3000 == 0:
            T *= 0.85
            T = max(T, 5e-5)

    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy hill‑climb: move a small random fraction of the mass from one
    coordinate to another.  Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0

    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def permutation_optimize(a_start: np.ndarray,
                         start_score: float,
                         fft_len: int,
                         n: int,
                         deadline: float):
    """
    Local search on the ordering of the coefficients.
    Randomly swap two entries; keep the swap only if it reduces the
    max‑convolution.  Occasionally performs a full random shuffle.
    """
    cur_a = a_start.copy()
    cur_score = start_score
    best_a = cur_a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    # how often we perform a full shuffle instead of a simple swap
    full_shuffle_every = 5000
    it = 0

    while time.time() < deadline:
        if it % full_shuffle_every == 0 and it > 0:
            cand = np.copy(cur_a)
            rng.shuffle(cand)                # full random permutation
        else:
            cand = cur_a.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cand[i], cand[j] = cand[j], cand[i]

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score

        it += 1
    return best_a, best_score


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """
    Add zero‑mean Gaussian noise and re‑project onto the simplex.
    """
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s

def exponential_family(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam * i)   (lam ≥ 0)."""
    a = np.exp(-lam * np.arange(n, dtype=float))
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    return a / a.sum()


def powerlaw_family(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)   (p ≥ 0)."""
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    return a / a.sum()


def edge_heavy_family(n: int, lam: float) -> np.ndarray:
    """
    a_i ∝ exp(-lam * min(i, n-1-i))   – large mass near both ends.
    """
    idx = np.arange(n, dtype=float)
    a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    return a / a.sum()


def bimodal_family(n: int, centre_frac: float, width_frac: float, peak_factor: float) -> np.ndarray:
    """
    Two symmetric bumps around the centre.
    centre_frac – relative position of centre (0.5 is true centre).
    width_frac  – width of each bump as a fraction of n.
    peak_factor – multiplicative factor for bump height (relative to uniform).
    """
    centre = int(centre_frac * n)
    width = max(1, int(width_frac * n))
    a = np.full(n, 1.0 / n)
    left = max(0, centre - width)
    right = min(n, centre + width)
    bump = peak_factor / n
    a[left:right] = bump
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Fallback random initialisation (as in the previous code)."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        lam = rng.uniform(0.001, 0.02)
        a = edge_heavy_family(n, lam)
    else:  # bimodal
        centre = 0.5
        width = rng.uniform(0.02, 0.07)
        peak = rng.uniform(0.5, 1.5)
        a = bimodal_family(n, centre, width, peak)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – we keep n=4000 (good trade‑off between quality and speed)
    # ------------------------------------------------------------------
    max_n = 4000
    n = max_n

    # FFT lengths
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise from families / previous best if available
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)          # fallback uniform
    best_score = fast_score(best_a, fft_len_conv, n)

    # If a previous solution exists, interpolate it to the current size
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            old_n = prev.size
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, n)
            a_interp = np.interp(x_new, x_old, prev)
            a_interp = np.maximum(a_interp, 0.0)
            if a_interp.sum() > 0:
                best_a = a_interp / a_interp.sum()
                best_score = fast_score(best_a, fft_len_conv, n)

    # ------------------------------------------------------------------
    # Quick sweep over parametric families (very cheap)
    # ------------------------------------------------------------------
    family_trials = 60   # number of random draws per family
    # exponential
    for _ in range(family_trials):
        lam = rng.uniform(0.0, 0.02)
        a = exponential_family(n, lam)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a.copy()
    # power‑law
    for _ in range(family_trials):
        p = rng.uniform(0.1, 3.0)
        a = powerlaw_family(n, p)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a.copy()
    # edge‑heavy
    for _ in range(family_trials):
        lam = rng.uniform(0.0, 0.02)
        a = edge_heavy_family(n, lam)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a.copy()
    # bimodal
    for _ in range(family_trials):
        centre_frac = 0.5
        width_frac = rng.uniform(0.02, 0.07)
        peak = rng.uniform(0.5, 1.5)
        a = bimodal_family(n, centre_frac, width_frac, peak)
        s = fast_score(a, fft_len_conv, n)
        if s < best_score:
            best_score, best_a = s, a.copy()
    print(f"[init families] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Phase split (percent of total budget)
    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.78 * budget_s   # sub‑gradient
    phase3_end = start_time + 0.92 * budget_s   # Adam
    phase4_end = start_time + 0.985 * budget_s  # permutation search
    # remaining time → final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (many restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.7, 0.5, 0.35, 0.25, 0.18, 0.12,
                    0.08, 0.05, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005]

    maxiter_per_lam = 1200

    while time.time() < phase1_end:
        # Occasionally jitter the current best to explore a neighbourhood
        if rng.random() < 0.18:
            sigma = rng.uniform(0.001, 0.015)
            a_start = add_noise(best_a, sigma, rng)
        else:
            a_start = random_initial(n, rng)

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning (smooth max)
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           phase3_end,
                                           lam_factor=0.002,
                                           lr=0.15)
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – permutation local search (swap / full shuffle)
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        perm_deadline = phase4_end
        best_a, best_score = permutation_optimize(best_a,
                                                  best_score,
                                                  fft_len_conv,
                                                  n,
                                                  perm_deadline)
        print(f"[permutation] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb (mass transfer)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)
    print(f"[final hill‑climb] score={best_score:.6f}")

    # Convert to Python list for the external evaluator
    final_seq = best_a.tolist()
    # Final evaluation using the provided external function
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
