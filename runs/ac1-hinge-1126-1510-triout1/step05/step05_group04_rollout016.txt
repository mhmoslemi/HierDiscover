def exp_gradient_descent(a_start, fft_len_conv, fft_len_grad, n, deadline, eta0=0.5,
                        decay=0.98, min_eta=1e-4):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    eta = eta0
    L = 2*n - 1
    while time.time() < deadline and eta > min_eta:
        conv = conv_fft(a, fft_len_conv, n)
        M = conv.max()
        # mask of maxima
        eps = 1e-12
        mask = (conv >= M - eps).astype(float)
        # compute gradient via convolution with mask reversed
        mask_rev = mask[::-1]
        # Use fft length for gradient
        fa = np.fft.fft(a, fft_len_grad)
        fm = np.fft.fft(mask_rev, fft_len_grad)
        conv_rev = np.real(np.fft.ifft(fa * fm))
        # extract gradient entries
        g = 2.0 * conv_rev[L - 1 - np.arange(n)]
        # update via exponentiated gradient
        # to avoid overflow, we clip g
        # Use exponentiated mirror descent: a_i *= exp(-eta * g_i)
        a = a * np.exp(-eta * g)
        # keep non-negative, renorm to sum 1
        sum_a = a.sum()
        if sum_a == 0 or np.isnan(sum_a) or np.isinf(sum_a):
            # fallback to uniform
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a
        # evaluate
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # decay eta
        eta *= decay
    return best_a, best_score
```

"""
Propose a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm
---------
* The objective 2·n·max(a*a) is scale‑invariant, so we keep the coefficients
  on the probability simplex (sum = 1).
* A multi‑restart L‑BFGS optimiser works on the unconstrained
  soft‑max parameters y.  The objective is a smooth log‑sum‑exp
  approximation of the max‑convolution; the smoothing factor λ is reduced
  in a continuation schedule.
* After the L‑BFGS phase we run a *mirror‑descent* (exponentiated‑gradient)
  optimisation directly on the true objective  max(a*a).  The sub‑gradient
  of the max‑convolution is obtained by a convolution with the indicator of
  the maximal positions – this step is cheap (FFT) and very effective at
  shaving off the last few hundredths of the bound.
* Finally a greedy hill‑climb (with optional large moves) polish the result.
* The whole search respects the 1000 s budget and prints progress.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    a is assumed length n, fft_len ≥ 2*n‑1.
    Returns the first 2*n‑1 entries (the linear convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv) .
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a (linear)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # weights for the smooth max
    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv w.r.t a (reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # middle n entries correspond to ∂conv/∂a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "tent":
        # symmetric triangular shape (peak in the middle)
        idx = np.arange(n)
        a0 = 1.0 - np.abs((idx - (n - 1) / 2) / ((n - 1) / 2))
        a0 = np.clip(a0, 0, None)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

# Local search helpers
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def exp_gradient_descent(
    a_start: np.ndarray,
    fft_len_conv: int,
    fft_len_grad: int,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    decay: float = 0.97,
    min_eta: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true objective (max convolution).
    Updates a via a_i ← a_i·exp(−η·g_i) and renormalises.
    g is the sub‑gradient of max(a*a) obtained by convolving a with
    the indicator of the maximal positions.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    L = 2 * n - 1
    eta = eta0
    while time.time() < deadline and eta > min_eta:
        conv = conv_fft(a, fft_len_conv, n)
        M = conv.max()
        # points that are (almost) maximal
        tol = max(1e-12, 1e-6 * M)
        mask = (conv >= M - tol).astype(float)          # shape (2n‑1,)
        # gradient = 2 * (a * mask)  (cross‑correlation)
        mask_rev = mask[::-1]
        fa = np.fft.fft(a, fft_len_grad)
        fm = np.fft.fft(mask_rev, fft_len_grad)
        conv_rev = np.real(np.fft.ifft(fa * fm))
        # extract the needed slice: g_i = 2 * conv_rev[L‑1‑i]
        g = 2.0 * conv_rev[L - 1 - np.arange(n)]

        # exponentiated‑gradient update
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0 or np.isnan(s) or np.isinf(s):
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        eta *= decay
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    n = 5000
    # If a previous construction is available, adapt n to its size.
    init_seq = globals().get("height_sequence_1", None)

        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and 3000 <= tmp.size <= 8000:
            n = int(tmp.size)

    fft_len_conv = next_pow2(2 * n - 1)        # for a*a
    fft_len_grad = next_pow2(3 * n - 2)        # for gradient (a * mask)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(tmp, 0.0)
            if best_a.sum() < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-7, num=35))
    Lbfgs_deadline = start_time + 0.50 * budget_s
    iteration = 0
