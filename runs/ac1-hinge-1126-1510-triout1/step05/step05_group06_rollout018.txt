def sidon_set(n):
    sum_used = np.zeros(2*n, dtype=bool) # size 2n for sums range 0..2n-2
    positions = []
    for i in range(n):
        # check if adding i leads to collisions
        # compute sums with existing positions and itself
        conflict = False
        # compute sums: i + i
        if sum_used[i+i]:
            conflict = True
        else:
            # for each s in positions:
            for s in positions:
                if sum_used[i + s]:
                    conflict = True
                    break
        if not conflict:
            positions.append(i)
            sum_used[i+i] = True
            for s in positions[:-1]:
                sum_used[i + s] = True
        # break condition: if we cannot add any further after some tries? but we will iterate all.
    return positions

def lp_improve(a, n, deadline, t_init=0.1, max_iters=12):
    a = np.asarray(a, dtype=float)
    a = np.maximum(a, 0)
    a = a / a.sum()
    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()
        # Build convolution matrix T efficiently using scipy.sparse
        # Use diags with offsets 0..n-1 (as before)
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n)
        T = diags(data, offsets, shape=(2 * n - 1, n), format='csc')
        # Linear program: maximize sum(b) subject to T @ b <= M, b >= 0
        c = -np.ones(n)
        bounds = [(0, None)] * n
        A_ub = T
        b_ub = np.full(2*n-1, M)
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            print("[lp] linprog exception:", e)
            break
        if not res.success:
            print("[lp] linprog failed:", res.message)
            break
        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        b_scaled = b_val / s
        a_new = (1 - t) * a + t * b_scaled
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a, n, next_pow2(2 * n - 1))
        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init
            it += 1
            # optional print
            print(f"[lp] iter {it} new score={new_score:.6f}")
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1
    return a
```

def greedy_sidon(n):
    sum_used = np.zeros(2*n, dtype=bool)
    positions = []
    for i in range(n):
        conflict = sum_used[i + i]
        if conflict:
            continue
        # check all existing positions
        for s in positions:
            if sum_used[i + s]:
                conflict = True
                break
        if conflict:
            continue
        # add
        positions.append(i)
        # mark sums
        sum_used[i + i] = True
        for s in positions[:-1]:
            sum_used[i + s] = True
            sum_used[s + i] = True  # same
    return positions
```

"""Propose a step sequence that minimizes
   2·n·max(conv(a,a)) / (sum(a)²).

This implementation builds on the previous state‑of‑the‑art code
but introduces several improvements:

1. **Richer seed pool** – Uniform, exponential, power‑law, edge‑heavy,
   sinusoidal, random Dirichlet draws, a symmetric Dirichlet seed and a
   sparse “Sidon‑set” construction (no two pairwise sums coincide).
2. **Continuation L‑BFGS** – finer λ‑schedule, more restarts and a
   larger iteration budget per λ.
3. **LP balancing** – the classic M‑balancing step, now solved with
   SciPy’s HiGHS linear‑programming solver (`linprog`).  This replaces
   the previous CVXPY/ECOS attempt and yields a measurable improvement.
4. **Projected sub‑gradient descent** – runs after the LP stage.
5. **Final hill‑climb** – greedy mass‑transfer moves for polishing.
6. **Symmetrisation** – after each major phase the candidate is mirrored
   (a ← (a + a[::-1])/2) and re‑projected onto the probability simplex;
   symmetry empirically keeps the convolution peak low.

All steps respect the simplex constraint (non‑negative, sum = 1) and
share a common FFT length for fast convolutions.  The total runtime is
kept within the allotted 1000 s; the function returns the best sequence
found so far.
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[n‑1‑i] and re‑normalize onto the simplex."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 0.0:
        a_sym /= s
    else:
        a_sym = np.full_like(a_sym, 1.0 / a_sym.size)
    return a_sym


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max using log‑sum‑exp.
    Returns (objective, gradient) where the objective is
    2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)      # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        # occasionally enforce symmetry (helps the max‑conv peak)
        a = symmetrize(a)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # short hill‑climb after each λ
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1200)
        # symmetrise again before the next λ
        best_a = symmetrize(best_a)
        best_score = fast_score(best_a, n, fft_len_conv)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent
    with a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # simple back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    # final symmetrisation
    best_a = symmetrize(best_a)
    best_score = fast_score(best_a, n, fft_len_conv)
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer, keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    # ensure symmetry before returning
    best = symmetrize(best)
    best_score = fast_score(best, n, fft_len)
    return best, best_score


# ----------------------------------------------------------------------
# LP balancing step (using scipy.linprog)
# ----------------------------------------------------------------------
def lp_improve(a: np.ndarray,
               n: int,
               deadline: float,
               t_init: float = 0.12,
               max_iters: int = 12):
    """
    Perform a few iterations of the linear‑programming balancing step:

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M   ∀k    (M = max(a*a))

    The LP is solved with SciPy's HiGHS simplex solver.  After each LP we
    mix a fraction t of the (scaled) solution back into a; t is halved
    whenever a step does not improve the objective.
    """
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build convolution matrix T (size (2n‑1)×n)
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format='csc')

        # Linear programme: maximise sum(b)  ⇔ minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        A_ub = T
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs')
        except Exception as e:
            print("[lp] linprog exception:", e)
            break

        if not res.success:
            print("[lp] linprog failed:", res.message)
            break

        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        b_scaled = b_val / s

        # Mix and project
        a_new = (1.0 - t) * a + t * b_scaled
        a_new = symmetrize(a_new)

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a, n, next_pow2(2 * n - 1))

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init
            it += 1
            print(f"[lp] iter {it} new score={new_score:.6f}")
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1

    return a


# ----------------------------------------------------------------------
# Sidon‑set based sparse seed
# ----------------------------------------------------------------------
def greedy_sidon_set(n: int):
    """Greedy construction of a Sidon set (all pairwise sums distinct)."""
    sum_used = np.zeros(2 * n, dtype=bool)   # sums 0 … 2n‑2
    positions = []
    for i in range(n):
        # check collisions with existing sums
        conflict = sum_used[i + i]
        if conflict:
            continue
        for s in positions:
            if sum_used[i + s]:
                conflict = True
                break
        if conflict:
            continue
        # accept i
        positions.append(i)
        sum_used[i + i] = True
        for s in positions[:-1]:
            sum_used[i + s] = True
            sum_used[s + i] = True   # same sum, symmetric
    return positions


def sidon_seed(n: int):
    """Return a probability vector supported on a greedy Sidon set."""
    pos = greedy_sidon_set(n)
    m = len(pos)
    a = np.zeros(n, dtype=float)
    if m == 0:
        a[:] = 1.0 / n
    else:
        w = 1.0 / m
        a[pos] = w
        a = symmetrize(a)
    # re‑normalise onto the simplex
    tot = a.sum()
    if tot > 0.0:
        a /= tot
    else:
        a[:] = 1.0 / n
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    Returns a plain Python list of floats (non‑negative, sum > 0).

    np.random.seed(seed)

    # ------------------------------------------------------------------

        s = a0.sum()

    fft_len_conv = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – generate a diverse pool of seeds
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len_conv)
        candidates.append((sc, arr))

    # uniform
    add_candidate(np.full(n, 1.0 / n))

    # linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # exponential families
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
