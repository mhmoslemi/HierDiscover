"""
Propose a low‑evaluation step function.

Algorithm
---------
* Work on the probability simplex (a_i ≥ 0, Σa_i = 1) – the scale of the
  vector does not affect the evaluation.
* Several cheap random initialisations are tried (uniform, edge‑heavy,
  Dirichlet‑sparse, random Gaussian).  For each start we run a
  continuation Adam optimiser that minimises a smooth approximation of
  the max‑convolution:
        smooth_max = max(conv) + λ·log Σ exp((conv−max)/λ) .
  The smoothing factor λ is decreased geometrically (≈30 stages).
  The gradient is computed analytically using FFTs; the outer factor
  2·n from the evaluation function is taken into account.
* The best candidate of the restart loop is refined by a projected
  sub‑gradient descent on the true objective
        f(a) = 2·n·max(conv) .
  Step‑size η = η0 / √t with η0 tuned for the current size.
* Finally an aggressive greedy hill‑climb (random mass transfers)
  runs until the time budget is exhausted.
* The resulting list of non‑negative heights is returned; the harness
  will call ``evaluate_sequence`` on it.
"""

import time
import numpy as np
from numpy.fft import fft, ifft


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·n‑1, result truncated)."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))[: 2 * n - 1]
    return conv


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(conv) for a probability vector (Σa=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto {x | x ≥ 0, Σx = 1}. """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp of the convolution)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
) -> tuple[float, np.ndarray]:
    """
    Return (f, grad_y) where
        f = 2·n·smooth_max(conv)
    and grad_y is the gradient w.r.t. the unconstrained vector y.
    The smoothing parameter is λ = lam_factor * max(conv).
    """
    a = softmax(y)                     # probability vector, Σa = 1
    # ---- convolution a*a -------------------------------------------------
    fa = fft(a, fft_len_conv)
    conv = np.real(ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12    # smoothing scale
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth_max w.r.t. a ---------------------------------
    # w_k = ∂ smooth_max / ∂ conv_k
    w = exp_shift / Z                    # shape (2n‑1,)
    # conv_rev = a_rev * w   (linear convolution)
    fa_rev = fft(a[::-1], fft_len_g)
    fw = fft(w, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    # derivative of conv_k w.r.t. a_i is 2·a_{k−i}
    g_a = 2.0 * conv_rev[(n - 1) : (n - 1 + n)]    # length n
    # include outer factor 2·n from the definition of f
    g_a *= 2.0 * n

    # ---- chain rule through softmax --------------------------------------
    mean_g = np.dot(a, g_a)
    grad_y = a * (g_a - mean_g)
    return f, grad_y


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    lr: float = 0.03,
    max_iter_per_lam: int = 1200,
) -> np.ndarray:
    """
    Continuation Adam optimiser.
    Starts from y0 and runs a decreasing smoothing schedule.
    Returns the final y (pre‑softmax parameters).
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_schedule:
        for _ in range(max_iter_per_lam):
            t += 1
            _, grad = smooth_obj_and_grad(
                y, n, fft_len_conv, fft_len_g, lam_factor=lam
            )
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            y -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return y


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.45,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a) = 2·n·max(conv).
    Returns the best vector seen and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))                     # index of maximal entry
        # sub‑gradient of conv[k] w.r.t. a:
        #   ∂/∂a_i conv[k] = 2·a_{k−i}  (0 if out of range)
        idx = np.arange(n)
        partner = k - idx
        mask = (partner >= 0) & (partner < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[partner[mask]]

        eta = eta0 / np.sqrt(it)                     # diminishing step size
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (random mass transfers)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 100_000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Accept only improving moves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move – up to max_move of the source mass
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Available modes:
        uniform – equal mass
        random  – Gaussian
        edge    – heavy on both edges (exponential tail)
        dirich  – Dirichlet with small α (sparse)
        bump    – single Gaussian bump (anywhere)
    """
    if mode == "uniform":
        a0 = np.full(n, 1.0 / n)
        return np.log(a0 + 1e-12)

    if mode == "random":
        y0 = rng.standard_normal(n)
        return y0

    if mode == "edge":
        # exponential decay from both borders
        rate = rng.uniform(0.003, 0.015)
        left = np.exp(-rate * np.arange(n))
        right = np.exp(-rate * np.arange(n)[::-1])
        a0 = left + right
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    if mode == "dirich":
        # Dirichlet with concentration α << 1 → very sparse
        alpha = rng.uniform(0.05, 0.2)
        a0 = rng.dirichlet(np.full(n, alpha))
        return np.log(a0 + 1e-12)

    if mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.03, 0.08) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        return np.log(a0 + 1e-12)

    raise ValueError(f"unknown mode {mode}")

        # short hill‑climb to polish this run (few hundred moves)
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=1500,
            max_move=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the exact objective
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 10.0, start_time + 0.85 * budget_s)
    print("[subgrad] start projected sub‑gradient descent")
    best_a, best_score = subgradient_descent(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.45,
        max_iters=1_500_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished with fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final aggressive hill‑climb until time expires
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=200_000,
            max_move=0.30,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Return the sequence (list of non‑negative floats)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
