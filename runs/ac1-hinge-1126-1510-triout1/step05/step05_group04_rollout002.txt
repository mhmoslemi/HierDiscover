    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    for t in range(1, steps+1):
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)
        # Optionally break if gradient norm small
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        y = y - lr * m_hat/(np.sqrt(v_hat)+eps)
        # optionally random perturb to escape flat
        if t % 1000 == 0:
            # compute current a and score for debug
            pass
    return y

"""
Propose a non‑negative step‑function (list of floats) with a low evaluation score.

Algorithm
---------
1. Initialise a probability vector (sum = 1) – either from the provided
   `height_sequence_1` or from a few hand‑crafted analytic shapes.
2. Run several multi‑restart L‑BFGS optimisations on a smooth
   log‑sum‑exp approximation of the true objective.
   * Each restart uses a different random initialisation (Gaussian,
     exponential decay, power‑law, edge‑heavy, Sidon‑set, binary mask,
     single bump, or a perturbed copy of the current best).
   * A continuation schedule with decreasing smoothing (`lam`) is used.
   * After each L‑BFGS run a short greedy hill‑climb polishes the result.
3. Refine the best candidate with a projected sub‑gradient method that
   includes momentum (similar to Nesterov/Heavy‑ball) to push the
   maximum convolution entry down.
4. Use a long aggressive hill‑climb, gradually reducing the maximum
   move size, and finish with a tiny simulated‑annealing phase to
   escape possible shallow local minima.
5. Return the final sequence; the harness will call `evaluate_sequence`
   on it.

The whole search respects the 1000 s budget (default 1 000 s) and
always returns a valid non‑negative list.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
    2·n·max(conv)  (the denominator sum_a**2 = 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z    # ∂smooth_max/∂conv_i

    # derivative of conv wrt a (via reverse convolution)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the middle n entries correspond to gradient components
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation L‑BFGS optimisation with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    Modes:
        random – normal distribution
        exp    – exponential decay
        power  – power‑law decay
        edge   – mass at the two edges
        sidon  – Sidon‑set with equal mass
        binary – random 0/1 mask with uniform mass on the 1's
        bump   – a single Gaussian bump
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums or 2 * candidate in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
                if len(chosen) >= int(np.sqrt(n) * 1.5):
                    break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "binary":
        mask = rng.random(n) < 0.20
        if not mask.any():
            mask[0] = True
        a0 = np.zeros(n)
        a0[mask] = 1.0 / mask.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "bump":
        centre = rng.integers(0, n)
        sigma = rng.uniform(0.05, 0.15) * n
        idx = np.arange(n)
        a0 = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a random fraction (≤ max_move) of the source mass
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_momentum(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    gamma: float = 0.90,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with heavy‑ball momentum.
    Optimises the true objective f(a)=2·n·max(conv) (a lies on the simplex).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    v = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m_idx = int(np.argmax(conv))
        # sub‑gradient of that entry
        idx = np.arange(n)
        counterpart = m_idx - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]
        # diminishing step size
        eta = eta0 / math.sqrt(it)
        # momentum update
        v = gamma * v + eta * g
        a_new = a - v
        # project back to simplex
        a_new = project_onto_simplex(a_new)
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    start_temp: float = 1e-3,
    end_temp: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Tiny simulated‑annealing phase to escape shallow local minima.
    Proposes random mass transfers and accepts them with Metropolis
    probability based on a linearly decreasing temperature.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    total_time = deadline - time.time()
    while time.time() < deadline:
        t = (time.time() - (deadline - total_time)) / total_time
        T = start_temp * (1 - t) + end_temp * t
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10  # limited move
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        delta_E = new_score - cur_score
        accept = False
        if delta_E < 0:
            accept = True
        else:
            prob = math.exp(-delta_E / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def pattern_powerlaw(n: int, p: float) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    if a.sum() == 0:

def pattern_exponential(n: int, rate: float) -> np.ndarray:
    a = np.exp(-rate * np.arange(n))
    if a.sum() == 0:
        a = np.ones(n)
    return a / a.sum()


def pattern_cosine(n: int) -> np.ndarray:
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    return a / a.sum()


def pattern_bump(n: int, centre: int, sigma: float) -> np.ndarray:
    idx = np.arange(n)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    return a / a.sum()

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    # Use a size that balances richness of the search space with FFT speed.
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 3000:
            n = min(int(tmp.size), 8000)
        else:
            n = 5000
    else:
        n = 5000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / tmp.sum()
        elif tmp.ndim == 1 and tmp.size > n:
            best_a = tmp[:n]
            best_a = best_a / best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Quick analytic pattern sweep (a few seconds)
    # --------------------------------------------------------------
    pattern_trials = 12
    for _ in range(pattern_trials):
        mode = rng.choice(["power", "exp", "cosine", "bump"])
        if mode == "power":
            p = rng.uniform(0.1, 2.5)
            a_pat = pattern_powerlaw(n, p)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a_pat = pattern_exponential(n, rate)
        elif mode == "cosine":
            a_pat = pattern_cosine(n)
        elif mode == "bump":
            centre = rng.integers(0, n)
            sigma = rng.uniform(0.04, 0.20) * n
            a_pat = pattern_bump(n, centre, sigma)
        else:
            continue
        s_pat = fast_score(a_pat, fft_len_conv, n)
        if s_pat < best_score:
            best_score = s_pat
            best_a = a_pat
            best_y = np.log(best_a + 1e-12)
            print(f"[pattern] new best fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – Multi‑restart L‑BFGS (≈ 55 % of budget)
    # --------------------------------------------------------------
    stage1_deadline = start_time + 0.55 * budget_s
    lam_schedule = list(np.geomspace(0.5, 1e-6, num=30))
    modes = ["random", "exp", "power", "edge", "sidon", "binary", "bump", "best"]
    restart_counter = 0

    while time.time() < stage1_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # Continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=600,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[L‑BFGS {restart_counter:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # A short greedy hill‑climb to polish this restart
        hc_deadline = min(stage1_deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2500,
            max_move=0.22,
        )
        best_y = np.log(best_a + 1e-12)
        restart_counter += 1

    # --------------------------------------------------------------
    # Stage 2 – Projected sub‑gradient with momentum (≈ 30 % of budget)
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline - 20.0, start_time + 0.90 * budget_s)
    print("[subgrad] starting projected sub‑gradient with momentum")
    best_a, best_score = subgradient_momentum(
        best_a,
        fft_len_conv,
        n,
        subgrad_deadline,
        eta0=0.30,
        gamma=0.90,
        max_iters=500_000,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] finished fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – Final aggressive hill‑climb + tiny annealing (remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30_000,
            max_move=0.30,
        )
        # small simulated‑annealing polish if any time left
        remaining = deadline - time.time()
        if remaining > 5.0:
            print("[final] simulated‑annealing polish")
            best_a, best_score = simulated_annealing(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                start_temp=5e-4,
                end_temp=1e-7,
            )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # pylint: disable=undefined-variable
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
