def directed_hill_climb(a, score, fft_len, n, deadline, max_iter=2000):
    while time.time() < deadline and it<max_iter:
        conv = full_conv(a, fft_len)
        k = np.argmax(conv)
        # compute contributions
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i_idx = np.arange(lo, hi+1)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx] * 2.0 # each product counted twice? Actually each product appears once except when i=j but fine.
        # Weighted selection: choose i proportionally to contrib
        if contrib.sum() == 0:
            break
        prob_i = contrib / contrib.sum()
        i = np.random.choice(i_idx, p=prob_i)
        # choose j from indices with low contribution
        # For simplicity choose random j distinct from i
        j = np.random.randint(0, n)
        while j == i:
            j = np.random.randint(0, n)
        # move small delta from i to j (or move from i to j if i >? ensure mass reduces peak)
        delta = a[i] * np.random.random() * 0.05
        a[i] -= delta
        a[j] += delta
        # re-evaluate score
        new_score = fast_score(a, fft_len, n)
        if new_score < score:
            score = new_score
        else:
            # revert
            a[i] += delta
            a[j] -= delta
        it += 1

"""
Improved search for a low‑score non‑negative step sequence.

Approach
--------
* Use a larger problem size (n = 4000) – the theoretical lower bound
  (uniform convolution) drops from 2.0 to ≈1.0, giving more head‑room.
* Phase 1 – many random restarts of a continuation L‑BFGS on a
  log‑sum‑exp smooth‑max of the convolution (the same idea as the
  reference code but with a finer λ‑schedule and a larger budget).
* Phase 2 – Adam optimisation on the **exact** max‑convolution
  (sub‑gradient on the active shift only).  Operates on logits so the
  simplex constraint is satisfied automatically.
* Phase 3 – directed hill‑climb that always attacks the current
  maximal convolution entry (instead of random moves).
* Phase 4 – final mixing with the uniform distribution and a short
  greedy hill‑climb for polishing.
* All phases respect the 1000 s deadline and keep the best sequence
  seen at any time.

The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) – assumes sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) – used by continuation L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
):
    """
    Return (smooth‑max objective, gradient) w.r.t. logits y.
    The smooth‑max is built from the convolution a*a.
    """
    a = softmax(y)

    # --------------------------------------------------------------
    # Convolution a*a (linear)
    # --------------------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # --------------------------------------------------------------
    # Smooth max using log‑sum‑exp
    # --------------------------------------------------------------
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # --------------------------------------------------------------
    # Gradient of smooth max w.r.t. convolution entries
    # --------------------------------------------------------------
    w = exp_shift / Z  # d smooth_max / d conv_i

    # Back‑propagation through convolution: d conv / d a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # --------------------------------------------------------------
    # Chain rule through soft‑max
    # --------------------------------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (phase 1)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam,
    deadline,
    fft_len_conv,
    fft_len_g,
    n,
):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            # ignore numerical problems and continue with current y
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (phase 2)
# ----------------------------------------------------------------------


def max_conv_and_subgrad(a: np.ndarray, fft_len: int):
    """
    Compute max(convolution) and a sub‑gradient with respect to a.
    Sub‑gradient g satisfies d max / d a_i = 2 * a_{k-i} for the active shift k.
    """
    conv = full_conv(a, fft_len)
    k = int(conv.argmax())  # index of the maximal entry

    # Build gradient of the max entry
    g = np.zeros_like(a)
    lo = max(0, k - (len(a) - 1))
    hi = min(k, len(a) - 1)
    i = np.arange(lo, hi + 1)
    j = k - i
    g[i] = 2.0 * a[j]  # derivative of a_i * a_{k-i}
    max_val = conv[k]
    return max_val, g


def adam_optimize_exact(
    y0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    lr: float = 0.12,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
):
    """
    Adam optimisation of the *exact* max‑convolution (non‑smooth).
    Operates on logits y (softmax(y) lives on the simplex).
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    a = softmax(y)
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        a = softmax(y)
        _, g_a = max_conv_and_subgrad(a, fft_len)

        # gradient w.r.t. logits: a * (g_a - <a,g_a>)
        grad_y = a * (g_a - np.dot(a, g_a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Directed hill‑climb (phase 3) – always attacks the current worst shift
# ----------------------------------------------------------------------


def directed_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    move_frac: float = 0.05,
):
    """
    Greedy hill‑climb that moves a small fraction of mass from a coordinate
    participating heavily in the maximal convolution entry to a random
    other coordinate.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        conv = full_conv(best, fft_len)
        k = int(conv.argmax())

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_idx = np.arange(lo, hi + 1)
        j_idx = k - i_idx
        # contribution of each i to the max entry (symmetrised)
        contrib = best[i_idx] * best[j_idx] * 2.0
        if contrib.sum() == 0.0:
            break
        prob_i = contrib / contrib.sum()
        i = rng.choice(i_idx, p=prob_i)

        # pick a random destination different from i
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        delta = best[i] * rng.random() * move_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simple mixing with the uniform distribution (phase 4)
# ----------------------------------------------------------------------


def mix_with_uniform(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
    max_mix: float = 0.20,
):
    """
    Small convex combinations with the uniform distribution.
    Occasionally a little mixing with the uniform vector reduces spikes.
    """
    best = a.copy()
    best_score = cur_score
    uniform = np.full(n, 1.0 / n)
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        t = rng.random() * max_mix
        cand = (1.0 - t) * best + t * uniform
        new_score = fast_score(cand, fft_len, n)
        if new_score < best_score:
            best = cand
            best_score = new_score
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Random initialiser – richer family than the reference code
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector on the simplex.
    A richer mixture of families (including Dirichlet) is used to increase
    diversity of starting points.
    """
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sine",
            "halfsine",
            "dirichlet",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sine":
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    elif mode == "halfsine":
        a = np.sin(np.pi * np.arange(n) / (n - 1))
    else:  # dirichlet
        # Dirichlet with random concentration parameters
        alpha = rng.uniform(0.1, 5.0, size=n)
        a = rng.dirichlet(alpha)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – we use the full budget size (4000)
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n
    fft_len_conv = next_pow2(2 * n - 1)          # for a*a
    fft_len_g = next_pow2(3 * n - 2)             # for smoothed‑max gradient

    # --------------------------------------------------------------
    # Initial sequence (uniform) – will be beaten by the optimiser
    # --------------------------------------------------------------
    a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5                # small safety margin
    phase1_end = start_time + 0.45 * budget_s             # L‑BFGS phase
    phase2_end = start_time + 0.75 * budget_s             # exact‑max Adam
    phase3_end = start_time + 0.90 * budget_s             # directed hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with a geometric λ‑schedule
    # --------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.15,
        0.10,
        0.07,
        0.045,
        0.03,
        0.018,
        0.010,
        0.006,
        0.003,
        0.001,
        5e-4,
        2e-4,
        1e-4,
    ]
    maxiter_per_lam = 300  # generous but not too heavy

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on the exact max‑convolution (sub‑gradient)
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize_exact(
            logits, n, fft_len_conv, phase2_end, lr=0.12
        )
        print(f"[Adam‑exact] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – directed hill‑climb (mass transfer targeting the worst shift)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = directed_hill_climb(
            best_a, best_score, fft_len_conv, n, phase3_end, max_iter=8000
        )
        print(f"[directed hill‑climb] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final polishing: mixing with uniform + short random hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = mix_with_uniform(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=4000
        )
        # a final tiny random hill‑climb (same routine as the reference)
        best_a, best_score = directed_hill_climb(
            best_a, best_score, fft_len_conv, n, deadline, max_iter=2000, move_frac=0.02
        )
        print(f"[final polishing] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the candidate sequence
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
