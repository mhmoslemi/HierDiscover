def adam_optimize(y0, n, fft_len_conv, fft_len_g, lam_schedule, deadline, lr0=0.1):
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter=0
    best_y = y.copy()
    best_score = fast_score(np.exp(y)/np.exp(y).sum(), fft_len_conv, n)
    for lam in lam_schedule:
        while time.time() < deadline and t_iter < 100000:
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t_iter += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad**2)
            m_hat = m / (1 - beta1**t_iter)
            v_hat = v / (1 - beta2**t_iter)
            y = y - lr0 * m_hat / (np.sqrt(v_hat) + eps)
            # optional symmetrize:
            y = (y + y[::-1]) * 0.5
            a = softmax(y)
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score = score
                best_y = y.copy()
                # optionally break early to move to next lam
                # but continue to search
    return best_y, best_score
```

```python
"""
Efficient search for a non‑negative step function minimizing evaluate_sequence.

The algorithm consists of three phases:
  1. A fast parametric sweep over several families of symmetric
     shapes (Beta‑type, exponential tails, power‑law and mixtures).
  2. Refinement of the best shape by a continuation L‑BFGS optimisation
     on a smoothed log‑sum‑exp surrogate of the max‑convolution,
     followed by an Adam‑style mirror descent on the true softmax
     parameters.
  3. A final projected sub‑gradient descent and an aggressive greedy
     hill‑climb to polish the solution.

All steps respect the probability simplex (sum = 1) and enforce symmetry
by averaging with the reversed vector, which is known to be optimal for
the convolution objective.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ---- basic utilities ----------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using a real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ---- smoothed objective -------------------------------------------------
def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where f = 2·n·smooth_max(a*a) and a = softmax(y).
    The smoothing parameter is λ = max(conv)·lam_factor (+1e-12).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                     # ∂smooth_max/∂conv

    # gradient back‑propagation through the convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # gradient through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """Continuation L‑BFGS on the smoothed surrogate."""
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y

# ---- families of symmetric shapes ---------------------------------------
def shape_beta(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta‑type density with shape parameter alpha (α>0)."""
    x = (np.arange(n) + 0.5) / n               # mid‑point coordinates
    # use log‑space to avoid overflow when α<1
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    a = np.exp(log_a)
    a = a / a.sum()
    return a

def shape_exponential(rate: float, n: int) -> np.ndarray:
    """Symmetric decreasing exponential from the borders."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-rate * d)
    a = a / a.sum()
    return a

def shape_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^{-p} + (n-i)^{-p}."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p) + (n + 1 - i) ** (-p)
    a = a / a.sum()
    return a

def shape_edge(m_frac: float, n: int) -> np.ndarray:
    """Few heavy edge buckets (fraction of the length) and flat interior."""
    m_edge = max(1, int(np.round(n * m_frac)))
    a = np.full(n, 1.0 / n)
    # give a factor sqrt(n) to the edge entries (same trick as the original code)
    edge_val = np.sqrt(n) / n
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a = a / a.sum()
    return a

def shape_mix(alpha: float, w: float, n: int) -> np.ndarray:
    """Convex combination of a Beta‑type shape and a uniform density."""
    beta = shape_beta(alpha, n)
    uniform = np.full(n, 1.0 / n)
    a = w * beta + (1.0 - w) * uniform
    a = a / a.sum()
    return a

# ---- local optimisation utilities ----------------------------------------
def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that transfers a random fraction of mass."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the true max‑convolution."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    deadline: float,
    lr0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the smoothed objective.
    Returns the logits that gave the best (lowest) fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_y = y.copy()
    best_score = fast_score(softmax(y), fft_len_conv, n)
    t_iter = 0

    for lam in lam_schedule:
        while time.time() < deadline:
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            t_iter += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            y = y - lr0 * m_hat / (np.sqrt(v_hat) + eps)

            # enforce symmetry (helps both convergence and objective)
            y = (y + y[::-1]) * 0.5

            # evaluate every few iterations to save time
            if t_iter % 10 == 0:
                a = softmax(y)
                cur_score = fast_score(a, fft_len_conv, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_y = y.copy()
            # safety break if we got very close to the deadline
            if time.time() > deadline - 0.01:
                break
    return best_y, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluate_sequence score. The routine combines a parametric sweep,
    continuation L‑BFGS, Adam refinement, sub‑gradient descent and a final
    greedy hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------------
    # Length of the sequence – reuse a previous solution when possible.
    # --------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
            best_a = tmp / tmp.sum()
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, fast score = {best_score:.6f}")

    # --------------------------------------------------------------------
    # Phase 1 – fast parametric sweep
    # --------------------------------------------------------------------
    sweep_deadline = deadline - 350.0   # reserve ~350 s for later refinement
    param_candidates = []

    # beta shapes
    for alpha in np.linspace(0.12, 0.90, 15):
        param_candidates.append(('beta', alpha, None))

    # exponential tails
    for rate in np.linspace(0.001, 0.03, 12):
        param_candidates.append(('exp', rate, None))

    # power‑law
    for p in np.linspace(0.1, 2.2, 14):
        param_candidates.append(('power', p, None))

    # edge‑heavy
    for frac in [0.005, 0.01, 0.02, 0.04]:
        param_candidates.append(('edge', frac, None))

    # mixes of beta and uniform
    for alpha in [0.2, 0.3, 0.4]:
        for w in [0.2, 0.35, 0.5]:
            param_candidates.append(('mix', alpha, w))

    # random perturbations of the current best (to escape possible local minima)
    for _ in range(8):
        param_candidates.append(('perturb', None, None))

    for mode, p1, p2 in param_candidates:
        if time.time() > sweep_deadline:
            break
        if mode == 'beta':
            a = shape_beta(p1, n)
        elif mode == 'exp':
            a = shape_exponential(p1, n)
        elif mode == 'power':
            a = shape_power(p1, n)
        elif mode == 'edge':
            a = shape_edge(p1, n)
        elif mode == 'mix':
            a = shape_mix(p1, p2, n)
        elif mode == 'perturb':
            a = best_a + rng.normal(scale=0.02, size=n)

            a = a / a.sum()
        else:

        # enforce symmetry explicitly (helps the objective)
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_y = np.log(a + 1e-12)

    # --------------------------------------------------------------------

    a_opt = a_opt / a_opt.sum()
    sc_opt = fast_score(a_opt, fft_len_conv, n)
    if sc_opt < best_score:
        best_score = sc_opt
        best_a = a_opt.copy()
        best_y = np.log(best_a + 1e-12)
        print(f"[L‑BFGS] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------------
    # Phase 3 – Adam‑style mirror descent on the smoothed objective
    # --------------------------------------------------------------------
    if time.time() < deadline - 180.0:

        lam_sched_adam = list(np.geomspace(0.5, 1e-8, num=40))

            best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------------

            best_a,

    # --------------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq

"""
Efficient search for a non‑negative step function minimizing evaluate_sequence.

The algorithm runs several phases:
  1. Fast parametric sweep over a few families of symmetric shapes
     (Beta‑type, exponential tails, power‑law, edge‑heavy and mixtures).
  2. Continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
     max‑convolution, followed by an Adam‑style mirror descent.
  3. Projected sub‑gradient descent on the true max‑convolution.
  4. Aggressive greedy hill‑climb to polish the final candidate.

All vectors are kept on the probability simplex (sum = 1) and are forced
to be symmetric – a property that is optimal for the convolution objective.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max‑convolution surrogate
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Returns (f, grad_y) where f = 2·n·smooth_max(a*a) and a = softmax(y).
    The smoothing parameter is λ = max(conv)·lam_factor (+1e-12).
    """
    a = softmax(y)

    # ---- convolution a*a -------------------------------------------------
    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # ---- smooth max (log‑sum‑exp) ----------------------------------------
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient w.r.t. conv --------------------------------------------
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # ---- propagate through the convolution -------------------------------
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---- chain rule through soft‑max -------------------------------------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)          # (diag(a) - a aᵀ)·g_vec

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 500,
    deadline: float | None = None,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a series of optimisations with decreasing
    smoothing (lam_factor).  Each stage starts from the previous result.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        if deadline is not None and time.time() > deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Families of symmetric shapes (used in the parametric sweep)
# ----------------------------------------------------------------------


def shape_beta(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta‑type density – shape parameter α>0."""
    x = (np.arange(n) + 0.5) / n
    # use log‑space to keep stability when α<1
    log_a = (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    a = np.exp(log_a)
    a = a / a.sum()
    return a


def shape_exponential(rate: float, n: int) -> np.ndarray:
    """Symmetric exponential decay from the borders."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-rate * d)
    a = a / a.sum()
    return a


def shape_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ i⁻ᵖ + (n‑i)⁻ᵖ."""
    i = np.arange(1, n + 1, dtype=float)
    a = i ** (-p) + (n + 1 - i) ** (-p)
    a = a / a.sum()
    return a


def shape_edge(m_frac: float, n: int) -> np.ndarray:
    """Few heavy edge buckets (fraction m_frac of the length)."""
    m_edge = max(1, int(np.round(n * m_frac)))
    a = np.full(n, 1.0 / n)
    edge_val = np.sqrt(n) / n
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a = a / a.sum()
    return a


def shape_mix(alpha: float, w: float, n: int) -> np.ndarray:
    """Convex combination of a Beta shape and the uniform density."""
    beta = shape_beta(alpha, n)
    uniform = np.full(n, 1.0 / n)
    a = w * beta + (1.0 - w) * uniform
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Local optimisation utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a random fraction of mass between indices."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on   f(a) = max_i (a*a)_i."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score


def adam_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    deadline: float,
    lr0: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the smoothed surrogate.
    Returns the logits that gave the best fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_y = y.copy()
    best_score = fast_score(softmax(y), fft_len_conv, n)
    it = 0

    for lam in lam_schedule:
        while time.time() < deadline:
            f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam)
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            y = y - lr0 * m_hat / (np.sqrt(v_hat) + eps)

            # enforce symmetry (optimal for the objective)
            y = (y + y[::-1]) * 0.5

            # evaluate occasionally
            if it % 10 == 0:
                a = softmax(y)
                cur_score = fast_score(a, fft_len_conv, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_y = y.copy()

            if time.time() > deadline - 0.01:
                break

    return best_y, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluate_sequence score.  The routine mixes a parametric sweep,
    continuation L‑BFGS, Adam refinement, sub‑gradient descent and a final
    greedy hill‑climb.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose a problem size.  Re‑use a previous construction if available.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)
            best_a = tmp / tmp.sum()
        else:
            n = 8000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 8000
        best_a = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – fast parametric sweep
    # ------------------------------------------------------------------
    sweep_deadline = deadline - 350.0   # reserve time for refined phases
    candidates = []

    # beta‑type shapes
    for alpha in np.linspace(0.12, 0.90, 15):
        candidates.append(("beta", alpha, None))

    # exponential tails
    for rate in np.linspace(0.001, 0.03, 12):
        candidates.append(("exp", rate, None))

    # power‑law shapes
    for p in np.linspace(0.1, 2.2, 14):
        candidates.append(("power", p, None))

    # edge‑heavy shapes
    for frac in [0.005, 0.01, 0.02, 0.04]:

    # mixtures of beta and uniform
    for alpha in [0.2, 0.3, 0.4]:

    # random perturbations of the current best
    for _ in range(8):

    for mode, p1, p2 in candidates:
        if time.time() > sweep_deadline:

        if mode == "beta":

        elif mode == "exp":
            a = shape_exponential(p1, n)
        elif mode == "power":
            a = shape_power(p1, n)
        elif mode == "edge":
            a = shape_edge(p1, n)
        elif mode == "mix":
            a = shape_mix(p1, p2, n)
        elif mode == "perturb":
            a = best_a + rng.normal(scale=0.02, size=n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            continue

        # enforce exact symmetry
        a = (a + a[::-1]) * 0.5
        a = a / a.sum()

        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_y = np.log(a + 1e-12)
            print(f"[sweep] mode={mode} param={p1},{p2} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – continuation L‑BFGS on a smoothed objective
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(1.0, 1e-7, num=45))
    y_opt = schedule_optimize(
        best_y,
        n,
        fft_len_conv,
        fft_len_g,
        lam_schedule,
        maxiter_per_stage=800,
        deadline=deadline - 200.0,
    )
    a_opt = softmax(y_opt)
    a_opt = (a_opt + a_opt[::-1]) * 0.5
    a_opt = a_opt / a_opt.sum()
    sc_opt = fast_score(a_opt, fft_len_conv, n)
    if sc_opt < best_score:
        best_score = sc_opt
        best_a = a_opt.copy()
        best_y = np.log(best_a + 1e-12)
        print(f"[L‑BFGS] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam‑style mirror descent on the smoothed surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline - 150.0:
        adam_deadline = deadline - 150.0
        lam_sched_adam = list(np.geomspace(0.5, 1e-8, num=40))
        y_adam, sc_adam = adam_optimize(

        )

        a_adam = (a_adam + a_adam[::-1]) * 0.5
        a_adam = a_adam / a_adam.sum()
        if sc_adam < best_score:
            best_score = sc_adam
            best_a = a_adam.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[Adam] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected sub‑gradient descent on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 70.0:
        sub_deadline = deadline - 70.0
        print("[subgrad] start")
        best_a, best_score = subgradient_descent(
            best_a,

        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final aggressive hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:

        best_a, best_score = hill_climb(

        )

    # ------------------------------------------------------------------

    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
