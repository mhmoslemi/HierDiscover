"""
Propose a low‑evaluation step function.

Algorithm outline
-----------------
1. **Parameter sweep** – generate a few dozen deterministic and random
   candidates (uniform, linear, exponential, power‑law, edge‑heavy,
   sinusoidal, double‑edge, etc.).  Keep the best few.
2. **Continuation L‑BFGS** – minimise a smooth log‑sum‑exp
   approximation of the max‑convolution.  A schedule of decreasing
   smoothing parameters (λ) is used and each stage is limited to a few
   hundred L‑BFGS iterations.  Several restarts from the sweep seeds are
   performed.
3. **LP balancing** – solve the classic “M‑balancing’’ linear programme
   (maximise Σ b subject to b≥0 and (a∗b)≤M) using SciPy’s high‑speed
   `linprog`.  Mix the scaled LP solution back into the current vector.
4. **Exact sub‑gradient descent** – minimise the original
   2·n·max(a∗a) directly with a simple back‑tracking line‑search.
5. **Fine‑grained L‑BFGS** – another continuation run with very small
   λ values to polish the solution.
6. **Final polishing** – a long random hill‑climb followed by a
   *targeted* hill‑climb that repeatedly reduces the contribution of the
   coefficient pair responsible for the current maximal convolution value.

All steps respect the simplex constraint (non‑negative, sum = 1) and use
FFT‑based convolutions for speed.  The total runtime is split across the
phases so that the function always returns a feasible sequence before the
deadline (default 1000 s).
"""

import time
import math
import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = fft(a, fft_len)
    conv = np.real(ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) assuming a is already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) and gradient for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max via log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = fft(a, fft_len_conv)
    conv_full = np.real(ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = fft(w, fft_len_g)
    fa_rev = fft(a_rev, fft_len_g)
    conv_rev = np.real(ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS driver
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous λ.  Returns best (a, score) seen.
    """
    y = np.log(a_start + 1e-12)               # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, n, fft_len_conv)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, n, fft_len_conv)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # a short hill‑climb after each λ to escape possible stalls
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        hc_deadline,
                                        max_iter=1200)

    return best_a, best_score


# ----------------------------------------------------------------------
# LP balancing – solved with SciPy's linprog (HiGHS)
# ----------------------------------------------------------------------
def lp_improve(a: np.ndarray,
               n: int,
               deadline: float,
               t_init: float = 0.12,
               max_iters: int = 12):
    """
    Perform a few iterations of the linear‑programming balancing step:

        maximise   Σ b_j
        s.t.       b ≥ 0,
                   (a * b)_k ≤ M  ∀k    (M = max(a*a))

    After each LP we mix a small fraction of the (scaled) solution back
    into the current vector a.
    """
    a = a.copy()
    a = a / a.sum()          # ensure simplex
    t = t_init
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = np.convolve(a, a)
        M = conv.max()

        # Build the convolution matrix T (size (2n‑1)×n) – diagonal i has a[i]
        data = [np.full(n, a[i]) for i in range(n)]
        offsets = np.arange(n, dtype=int)
        T = diags(data, offsets, shape=(2 * n - 1, n), format="csc")

        # Linear program: minimise -sum(b)  <=> maximise sum(b)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=T, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if res.status != 0 or res.x is None:
            # LP failed – abort further LP steps
            print("[lp] linprog failed, status:", res.status)
            break

        b_val = np.maximum(res.x, 0.0)
        s = b_val.sum()
        if s <= 0.0:
            break
        b_scaled = b_val / s

        # Mix and re‑normalise
        a_new = (1.0 - t) * a + t * b_scaled
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, n, next_pow2(2 * n - 1))
        old_score = fast_score(a, n, next_pow2(2 * n - 1))

        if new_score < old_score - 1e-12:
            a = a_new
            t = t_init
            it += 1
        else:
            t *= 0.5
            if t < 1e-4:
                break
            it += 1

    return a


# ----------------------------------------------------------------------
# Exact sub‑gradient descent (max‑convolution)
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent with a simple
    back‑tracking line search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = fft(a, fft_len_conv)
        conv = np.real(ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = fft(mask, fft_len_g)
        fa_rev = fft(a_rev, fft_len_g)
        conv_g = np.real(ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, n, fft_len_conv)
            if cand_score < best_score:
                best_score, best_a = cand_score, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random hill‑climb (mass transfer)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy hill‑climb:
    pick two indices, move a tiny random fraction from one to the other,
    accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            i += 1
            continue
        delta = best[i1] * rng.random() * 0.05
        if delta <= 0.0:
            i += 1
            continue
        best[i1] -= delta
        best[i2] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i1] += delta
            best[i2] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (reducing the dominant convolution contribution)
# ----------------------------------------------------------------------
def hill_climb_targeted(a: np.ndarray,
                        current_score: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        max_iter: int = 5000,
                        frac: float = 0.05):
    """
    Focused hill‑climb:
    repeatedly identify the coefficient that most contributes to the
    current maximal convolution entry and move a small fraction of its
    mass to a random other position.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    i = 0
    while time.time() < deadline and i < max_iter:
        # compute full convolution once per iteration
        conv = np.real(ifft(fft(best, fft_len) * fft(best, fft_len)))[: 2 * n - 1]
        k_max = int(conv.argmax())
        # determine the index with largest a[i] * a[k_max - i]
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        best_contrib = -1.0
        i_max = None
        for ii in range(i_start, i_end + 1):
            jj = k_max - ii
            contrib = best[ii] * best[jj]
            if contrib > best_contrib:
                best_contrib = contrib
                i_max = ii
        if i_max is None:
            break
        delta = best[i_max] * frac
        if delta <= 0.0:
            i += 1
            continue
        # choose a random target index different from i_max
        j_rand = rng.integers(0, n)
        while j_rand == i_max:
            j_rand = rng.integers(0, n)

        best[i_max] -= delta
        best[j_rand] += delta

        new_score = fast_score(best, n, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i_max] += delta
            best[j_rand] -= delta

        i += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step sequence.
    Returns a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 2500                              # keep LP tractable
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        n = min(prev.size, max_n)
        a0 = prev[:n].copy()
        a0[a0 < 0] = 0.0
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a convolution
    fft_len_g = next_pow2(3 * n - 2)      # length for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, n, fft_len_conv)
    print(f"[init] n={n}, initial fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (collect seeds)
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.ndim != 1 or arr.size == 0:
            return
        arr = np.maximum(arr, 0.0)
        tot = arr.sum()
        if tot <= 0.0:
            return
        arr = arr / tot
        sc = fast_score(arr, n, fft_len_conv)
        candidates.append((sc, arr))

    # Uniform
    add_candidate(np.full(n, 1.0 / n))

    # Linear decreasing
    add_candidate(np.arange(n, 0, -1, dtype=float))

    # Exponential families
    for rate in [5e-4, 1e-3, 2.5e-3, 5e-3, 1e-2]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # Edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # Double‑edge power‑law (both ends heavy)
    for p in [0.5, 1.0, 1.5]:
        left = (np.arange(1, n // 2 + 1, dtype=float) ** (-p))
        right = left[::-1]
        arr = np.concatenate([left, right]) if n % 2 == 0 else np.concatenate([left, [0.0], right])
        add_candidate(arr)

    # Random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Keep the best few seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:6]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin
    phase1_end = start + 0.55 * budget_s      # L‑BFGS (majority)
    phase2_end = start + 0.80 * budget_s      # LP + sub‑gradient
    # remaining time → final polishing

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.8, 0.6, 0.4, 0.2, 0.1, 0.08, 0.05,
                    0.03, 0.02, 0.01, 0.008, 0.005,
                    0.003, 0.001, 0.0005, 0.0002, 0.0001]
    maxiter_per_lam = 350
    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – LP balancing + sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # LP balancing (fast)
        lp_deadline = min(phase2_end, deadline - 0.4)
        print("[LP] starting balancing iterations")
        best_a = lp_improve(best_a, n, lp_deadline, t_init=0.12, max_iters=12)
        best_score = fast_score(best_a, n, fft_len_conv)
        print(f"[LP] post‑LP fast_score={best_score:.6f}")

        # Sub‑gradient descent (short but more iterations)
        if time.time() < phase2_end:
            sg_deadline = min(phase2_end, deadline - 0.2)
            best_a, best_score = subgradient_optimize(best_a,
                                                      sg_deadline,
                                                      fft_len_conv,
                                                      fft_len_g,
                                                      n)
            print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine‑grained L‑BFGS (very small λ)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        fine_lam_sched = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
        fine_maxiter = 200
        a_opt, score_opt = lbfgs_continuation(best_a,
                                              fine_lam_sched,
                                              fine_maxiter,
                                              deadline - 5.0,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[fine‑L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final polishing (random + targeted hill‑climb)
    # ------------------------------------------------------------------
    final_deadline = deadline
    # Random hill‑climb
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    # Targeted hill‑climb – allocate whatever time remains
    if time.time() < final_deadline:
        best_a, best_score = hill_climb_targeted(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 final_deadline,
                                                 max_iter=8000,
                                                 frac=0.04)
    # One more random sweep in case we still have time
    if time.time() < final_deadline:
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=6000)

    # ------------------------------------------------------------------
    # Return the sequence (plain list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # Evaluate finally (the competition runner will also evaluate)
    try:
        final_eval = evaluate_sequence(final_seq)  # type: ignore  # noqa: F821
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence not available or raised:", e)
    return final_seq
