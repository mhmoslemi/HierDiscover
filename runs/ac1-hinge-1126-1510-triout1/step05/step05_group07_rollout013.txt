def lp_step(a, M, t=0.5):
    # Build sparse matrix A: shape (2*n - 1, n)
    # For each shift k, row k: for i in range(n): if 0 <= k - i < n => row k has coefficient a[i].
    # Equivalent to Toeplitz: first column = a concatenated with zeros? Actually matrix largest row length.
    # We'll construct using scipy.sparse.lil

    m = 2*n-1
    rows = []
    cols = []
    data = []
    for i in range(n):
        a_i = a[i]
        start = i  # row index where a_i first appears
        # row indices start..start+n-1 (length n)
        rows.extend(range(start, start+n))
        cols.extend([i]*n)
        data.extend([a_i]*n)

    # Build matrix as csr
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n), dtype=np.float64)

    # Solve LP:
    # maximize sum(b)  => minimize -sum(b)
    c = -np.ones(n)
    # constraints: A @ b <= M
    # b >= 0.

    res = linprog(c, A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)]*n, method='highs')
    b = res.x

    # Normalize and mix
    sum_b = b.sum()
    if sum_b == 0:
        # fallback
        return a
    g = b / sum_b
    new_a = (1-t) * a + t * g
    return new_a

def lp_improve(a, M, t_init=0.5, min_t=1e-4):
    n = len(a)
    # compute A
    rows = (np.arange(n)[:, None] + np.arange(n)).ravel()
    cols = np.tile(np.arange(n), n)
    data = np.tile(a, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    # Solve LP
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        print("LP failed", res.message)
        return a, False
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a, False
    g = b / sum_b
    t = t_init
    while t >= min_t:

        M_new = conv_fft(a_new, fft_len, n).max()
        if M_new < M - 1e-12:
            return a_new, True
        t *= 0.5
    # fallback: no improvement
    return a, False

def init_edge_sequence(n, rng):
    # Use edge-biased pattern used earlier:
    m_edge = max(1, int(n * 0.02)) # 2% of n for edges maybe.
    edge_val = np.sqrt(n / 2.0) / n
    center_val = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, center_val)
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a /= a.sum()
    return a

def lp_improve(a, M, fft_len, n, iteration, deadline):
    # Build convolution matrix A (n*n nonzeros)
    # Use sparse lil building for clarity
    # We'll break early if time near deadline
    if time.time() > deadline - 2.0:
        return a, False
    m = 2*n - 1
    # Build via vectorized approach for speed
    # Row indices
    i = np.arange(n, dtype=np.int32)
    rows = (i[:, None] + i[None, :]).ravel()
    cols = np.tile(i, n)
    data = np.tile(a, n)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))
    # Solve LP
    c = -np.ones(n)  # maximize sum(b)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                 method='highs', options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
    # free A quickly
    del A
    if not res.success:
        return a, False
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return a, False
    g = b / sum_b

    t = 0.5
    while t >= 1e-4:
        a_new = (1 - t) * a + t * g
        # compute new max conv
        M_new = conv_fft(a_new, fft_len, n).max()
        if M_new < M - 1e-12:
            return a_new, True
        t *= 0.5
    return a, False

"""
Propose a candidate step‑function sequence.

The algorithm is a hybrid of three ideas:
1.  Start from an “edge‑biased’’ distribution which already lowers the
    centre of the self‑convolution.
2.  Repeatedly solve a linear program:
        max   Σ b_i
        s.t.  (a * b)_k ≤ M   for all k
              b_i ≥ 0
    where * denotes linear convolution, a is the current sequence and
    M = max(a*a).  The LP finds a direction b that can carry more mass
    without raising any convolution entry above the current worst value.
    The new sequence is a convex combination a←(1‑t)a + t·(b/Σb) with a
    back‑tracking line search on t, guaranteeing monotone decrease of M.
3.  Between LP steps a cheap projected sub‑gradient descent on the exact
    objective 2·n·max(a*a) is performed, followed by a short random
    hill‑climb for polishing.

All vectors are kept on the probability simplex (sum = 1, a_i ≥ 0),
hence the evaluation function `evaluate_sequence` reduces to
`2·n·max(convolution)`.  The implementation works for n≈4000
and stays well within the 1000 s budget.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# --------------------------------------------------------------
# Core utilities
# --------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT (real part,
    trimmed to length 2·n‑1).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact score 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# --------------------------------------------------------------


def subgrad_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    deadline: float,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on f(a)=2·n·max(a*a)."""
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    i = np.arange(n)  # index array for vectorised gradient

    for t in range(max_iters):
        if time.time() > deadline:
            break

        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of max at the active index k
        j = k - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(t + 1.0)
        a = project_simplex(a - eta * grad)

        score = fast_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # occasional jitter to escape plateaus
        if (t + 1) % 300 == 0 and np.random.rand() < 0.07:
            noise = np.random.normal(scale=0.006, size=n)
            a = project_simplex(a + noise)

    return best_a, best_score


# --------------------------------------------------------------
# Simple hill‑climbs (polishing)
# --------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Plain random mass‑transfer hill‑climb."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    max_move_frac = 0.10
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# --------------------------------------------------------------
# Linear‑programming improvement step
# --------------------------------------------------------------


def lp_improve(
    a: np.ndarray,
    M: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, bool]:
    """
    Solve
        max Σ b_i
        s.t.   (a * b)_k ≤ M   ∀k
               b_i ≥ 0
    Returns the mixed vector (1‑t)·a + t·(b/Σb) with a back‑track on t.
    """
    if time.time() > deadline - 2.0:
        return a, False

    m = 2 * n - 1

    # Build the convolution matrix A where A_{k,i} = a_{k‑i} (0 if out of range).
    idx = np.arange(n, dtype=np.int32)
    rows = (idx[:, None] + idx[None, :]).ravel()           # shape n², values 0..2n‑2
    cols = np.tile(idx, n)                                 # each column index repeated n times
    data = np.tile(a, n)                                   # a repeated for every column
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(m, n))

    # Linear program (highs is fast and reliable)
    c = -np.ones(n)                                        # maximise Σb_i  → minimise -Σb_i
    rhs = np.full(m, M, dtype=np.float64)                  # (a * b)_k ≤ M
    bounds = [(0, None)] * n

    lp_res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=rhs,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    # free the big matrix early
    del A

    if not lp_res.success:
        return a, False

    b = lp_res.x
    total_b = b.sum()
    if total_b <= 0.0:
        return a, False
    g = b / total_b          # normalise to a probability vector

    # back‑track line search on the mixing weight t
    t = 0.5
    while t >= 1e-4:
        a_new = (1.0 - t) * a + t * g
        M_new = conv_fft(a_new, fft_len, n).max()
        if M_new < M - 1e-12:          # strict improvement
            return a_new, True
        t *= 0.5
    return a, False


# --------------------------------------------------------------
# Initial edge‑biased sequence
# --------------------------------------------------------------


def initial_edge_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Edge‑biased distribution used in earlier successful runs.
    A few cells at both ends receive a higher weight.
    """
    m_edge = max(1, int(n * rng.uniform(0.015, 0.04)))  # 1–4 % of the indices
    edge_val = np.sqrt(n / 2.0) / n
    centre_val = 1.0 / np.sqrt(2.0) / n
    a = np.full(n, centre_val, dtype=np.float64)
    a[:m_edge] = edge_val
    a[-m_edge:] = edge_val
    a /= a.sum()
    return a

# Main entry point required by the harness
# --------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found within the allotted time.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_deadline = start_time + budget_s - 5.0   # small safety margin

    # problem size (the larger the better, 4000 works comfortably)
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # start from the edge‑biased shape
    a = initial_edge_sequence(n, rng)

    # keep track of the overall best
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    iteration = 0
    lp_steps = 0
    # primary loop – mix LP steps with cheap sub‑gradient refinements
    while time.time() < total_deadline - 200.0:   # reserve ~200 s for final polishing
        M = conv_fft(a, fft_len, n).max()

        # 1) LP improvement (performed at most 12 times to stay within budget)
        if lp_steps < 12:
            a_new, improved = lp_improve(a, M, fft_len, n, total_deadline)
            if improved:
                a = a_new
                cur_score = fast_score(a, fft_len, n)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[Iter {iteration}] LP step improved score = {best_score:.6f}")
                lp_steps += 1

        # 2) short sub‑gradient descent (500 iterations or until time runs out)
        sub_deadline = min(total_deadline, time.time() + 0.9)
        a, sg_score = subgrad_descent(
            a, n, fft_len, max_iters=500, deadline=sub_deadline, eta0=0.34
        )
        if sg_score < best_score:
            best_score = sg_score
            best_a = a.copy()
            print(f"[Iter {iteration}] Sub‑grad descent improved score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final polishing – a few aggressive hill‑climbs
    # ------------------------------------------------------------------
    polishing_deadline = total_deadline
    a = best_a.copy()
    cur_score = best_score
    while time.time() < polishing_deadline:
        # random hill‑climb
        a, cur_score = hill_climb(
            a, cur_score, fft_len, n, polishing_deadline, max_iter=2500
        )
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[polish] Hill‑climb improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluate_sequence()
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # expose for possible warm‑starting of later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
