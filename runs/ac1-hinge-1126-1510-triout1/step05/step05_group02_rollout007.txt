def subgrad_descent(a0, deadline, fft_len, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    step0 = 0.05
    it = 0
    while time.time() < deadline:
        conv = full_conv(a, fft_len)
        k = int(conv.argmax())
        # compute gradient
        g = np.zeros(n)
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i = np.arange(lo, hi+1)
        j = k - i
        g[i] = 2.0 * a[j]
        # step size
        step = step0 / np.sqrt(it + 1)
        a_new = a - step * g
        # projection onto simplex
        a_new = simplex_projection(a_new)
        # evaluate
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        it += 1
    return best_a, best_score
```

def targeted_hill_climb(a0, start_score, deadline, fft_len, n, max_iter=5000):
    a = a0.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = full_conv(a, fft_len)
        k = int(conv.argmax())
        lo = max(0, k-(n-1))
        hi = min(k, n-1)
        i_indices = np.arange(lo, hi+1)
        j_indices = k - i_indices
        contributions = a[i_indices] * a[j_indices]
        # Select the index with largest contribution
        idx = np.argmax(contributions)
        i = i_indices[idx]
        j = j_indices[idx]
        # Amount to move
        eps = min(a[i], a[j]) * 0.2  # move up to 20% of the smaller
        if eps <= 0:
            it += 1
            continue
        # subtract from i and j
        a[i] -= eps
        a[j] -= eps
        # choose two random destinations (could be same)
        p, q = rng.integers(0, n, size=2)
        # add equally
        a[p] += eps/2
        a[q] += eps/2
        # ensure nonnegative minor negative due to floating rounding
        a = np.maximum(a, 0.0)
        # reproject onto simplex to preserve sum
        a = simplex_projection(a)
        new_score = fast_score(a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # revert changes
            a = best_a.copy()
            cur_score = best_score
        it += 1
    return best_a, best_score

The search proceeds in several stages, all respecting the simplex
constraint Σ a_i = 1 and a_i ≥ 0.

1. **Many random restarts of a continuation L‑BFGS** on a smooth
   log‑sum‑exp approximation of the max‑convolution.  The smoothing
   parameter λ is decreased according to a schedule, allowing the
   optimizer to approach the true non‑smooth objective.

2. **Exponential‑mirror descent** (a multiplicative sub‑gradient method)
   on the exact max‑convolution.  This quickly reduces the peak of the
   autocorrelation.

3. **Projected sub‑gradient descent** on the exact max‑convolution
   with a diminishing step size and Euclidean simplex projection.
   This phase gives a monotone improvement and can escape the
   plateaus left by the mirror step.

4. **Adam optimisation** on the smoothed objective for fine‑tuning.

5. **Short simulated annealing** to diversify the search.

6. **Targeted hill‑climb** that moves mass away from the pair of
   coordinates that contribute most to the current maximal convolution
   entry.

All phases are time‑boxed inside the total 1000 s budget.  The best
sequence found (as a list of floats) is returned and its exact score
is printed via ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a sequence with Σ a = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑prop through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            # ignore failures and continue with current y
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Mirror‑descent with exponentiated‑gradient updates
# ----------------------------------------------------------------------


def exp_mirror_descent(a0: np.ndarray,
                       deadline: float,
                       fft_len_conv: int,
                       n: int):
    """
    Projected exponentiated‑gradient descent on the exact max‑convolution.
    Returns best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    it = 0
    eta0 = 1.0
    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())

        # sub‑gradient of the max (exact)
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgrad_descent(a0: np.ndarray,
                    deadline: float,
                    fft_len_conv: int,
                    n: int):
    """
    Simple projected sub‑gradient descent on f(a)=max(a*a).
    Monotonically reduces the objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    cur_score = best_score
    it = 0
    step0 = 0.05

    while time.time() < deadline:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())

        # sub‑gradient of the active linear piece
        g = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g[i] = 2.0 * a[j]

        step = step0 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * g)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
        # else keep the old a (monotone descent)
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective (fine‑tuning)
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective starting from logits y0.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (short)
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len_conv: int,
                  n: int,
                  deadline: float):
    """
    Classic simulated annealing that moves a random fraction of mass
    between two coordinates.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len_conv, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T = max(T * 0.92, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb that attacks the active convolution entry
# ----------------------------------------------------------------------


def targeted_hill_climb(a0: np.ndarray,
                        start_score: float,
                        deadline: float,
                        fft_len_conv: int,
                        n: int,
                        max_iter: int = 4000):
    """
    Greedy mass‑transfer that always removes a fraction from the pair
    (i,j) contributing most to the current maximal convolution entry.
    Accepts only improving moves.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        conv = full_conv(a, fft_len_conv)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_idx = np.arange(lo, hi + 1)
        j_idx = k - i_idx
        contrib = a[i_idx] * a[j_idx]
        # most harmful pair
        mx = np.argmax(contrib)
        i = i_idx[mx]
        j = j_idx[mx]
        # move up to 20 % of the smaller of the two masses
        eps = min(a[i], a[j]) * 0.20
        if eps <= 0.0:
            it += 1
            continue

        prev_a = a.copy()
        a[i] -= eps
        a[j] -= eps

        # two random recipients (may coincide)
        p, q = rng.integers(0, n, size=2)
        a[p] += eps / 2.0
        a[q] += eps / 2.0

        # re‑project onto simplex to keep Σa = 1
        a = simplex_projection(a)

        new_score = fast_score(a, fft_len_conv, n)
        if new_score < cur_score:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a = prev_a
            cur_score = cur_score  # unchanged
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialiser (diverse families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random distribution on the simplex using a mixture of families."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sine', 'halfsine', 'triangular'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    elif mode == 'halfsine':
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)
    else:  # triangular (peak in the centre, zero at ends)
        i = np.arange(n)
        a = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
        a = np.maximum(a, 0.0)

    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (use the warm‑start if available, otherwise maximum)
    # ------------------------------------------------------------------
    MAX_N = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, MAX_N)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = MAX_N // 2
            a0 = np.full(n, 1.0 / n)
    else:
        n = MAX_N
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient‑related convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init]   n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    phase1_end = start_time + 0.45 * budget_s      # L‑BFGS
    phase2_end = start_time + 0.65 * budget_s      # mirror‑descent
    phase3_end = start_time + 0.80 * budget_s      # sub‑gradient descent
    phase4_end = start_time + 0.92 * budget_s      # Adam + anneal
    # remaining time → targeted hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS with many restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001, 0.0005]
    maxiter_per_lam = 1000
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – exponentiated‑gradient mirror descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_mirror_descent(
            best_a,
            phase2_end,
            fft_len_conv,
            n)
        print(f"[mirror] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the exact max
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = subgrad_descent(
            best_a,
            phase3_end,
            fft_len_conv,
            n)
        print(f"[subgrad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Adam fine‑tuning + short annealing
    # ------------------------------------------------------------------
    if time.time() < phase4_end:
        # Adam on the smooth surrogate
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(
            logits,
            n,
            fft_len_conv,
            fft_len_g,
            phase4_end,
            lam_factor=0.002,
            lr=0.15)
        print(f"[Adam] fast_score={best_score:.6f}")

        # Very short simulated annealing to escape possible local traps
        if time.time() < deadline - 0.2:
            anneal_deadline = deadline - 0.2
            best_a, best_score = random_anneal(
                best_a,
                best_score,
                fft_len_conv,
                n,
                anneal_deadline)
            print(f"[anneal] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – targeted hill‑climb (final refinement)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            deadline,
            fft_len_conv,
            n,
            max_iter=12000)
        print(f"[targeted hill‑climb] fast_score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
