"""
Propose a low‑score step sequence.
-------------------------------------------------
Algorithm (≈ 1000 s budget)

1. Initialise a probability vector `f` (sum = 1).  Use the
   best known construction (`height_sequence_1`) if available,
   otherwise start from the uniform distribution.

2. Repeatedly improve `f` by solving a *linear programme*:
   maximise   Σ g_j   subject to   g≥0   and   (f * g)_i ≤ M
   where   M = max(f * f) and `*` denotes the ordinary
   (zero‑padded) convolution.  This is the exact optimisation
   step used in the reference iterative scheme, but here we
   solve it with `scipy.optimize.linprog` (high‑speed HiGHS
   solver).  The LP only needs the *active* constraints,
   i.e. indices where the current bound `M` is tight, which keeps
   the problem small (≈ n constraints instead of 2 n).

   The optimal `g` is normalised to sum = 1 and mixed with the
   current vector:
        f ← (1‑t)·f + t·g ,   t = 0.4 (fixed step size)
   The mixture stays on the simplex, so the evaluation
   function (scale‑invariant) is unchanged.

3. Iterate the LP/mixture step as long as time permits (≈ 10‑15
   iterations).  Keep the best vector seen.

4. Finish with a cheap greedy hill‑climb that moves a small
   random fraction of the mass between two coordinates only
   when it strictly reduces the objective.  This refines the
   solution to the final precision.

All heavy work (convolution, FFT length, simplex projection,
hill‑climbing) is performed with NumPy.  The code respects the
time budget, prints progress and finally returns the best list
of floats found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) assuming Σa = 1.
    This equals the value returned by ``evaluate_sequence``.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, Σ x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of the mass from one
    index to another, accepting only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 10 % of the mass at i
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000  # keep the same size as the previous best solution
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Initialise f (probability vector)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate / truncate to the required length
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            f = np.interp(x_new, x_old, old)
            f = np.maximum(f, 0.0)
            f_sum = f.sum()
            if f_sum <= 0.0:
                f = np.full(n, 1.0 / n)
            else:
                f /= f_sum
        else:
            f = np.full(n, 1.0 / n)
    else:
        f = np.full(n, 1.0 / n)

    best_f = f.copy()
    best_score = fast_score(best_f, fft_len, n)
    print(f"[init] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # keep a tiny safety margin

    # --------------------------------------------------------------
    # Phase 1 – LP‑based improvement (core loop)
    # --------------------------------------------------------------
    max_lp_iters = 15           # safety cap – usually far fewer are needed
    lp_iter = 0
    step_size = 0.4             # mixing coefficient (1‑t)·f + t·g

    while time.time() < deadline - 5.0 and lp_iter < max_lp_iters:
        # 1) current max convolution value M
        conv_f = full_conv(f, fft_len)
        M = conv_f.max()

        # 2) build the Toeplitz convolution matrix A such that
        #    (A @ g)[i] = (f * g)[i]   (zero‑padded linear conv)
        #    only keep rows where the bound is (almost) active
        tol = M * 1e-12 + 1e-12
        active_rows = np.where(conv_f >= M - tol)[0]

        #   Each diagonal offset d contains the constant f[d]
        diagonals = [np.full(n, val) for val in f]
        offsets = np.arange(n)
        A_full = diags(diagonals, offsets, shape=(2 * n - 1, n), format="csc")
        A = A_full[active_rows, :]   # keep just the active constraints

        # 3) linear programme:
        #    maximise   Σ g_j    (i.e. minimise –Σ g_j )
        #    s.t.      A @ g ≤ M  ,   g ≥ 0
        c = -np.ones(n)                     # -Σ g_j
        b_ub = np.full(active_rows.shape[0], M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method="highs", options={"presolve": True})

        if not res.success:
            print(f"[LP] iteration {lp_iter} failed: {res.message}")
            break

        g = res.x
        g_sum = g.sum()
        if g_sum <= 0.0:
            print(f"[LP] iteration {lp_iter} produced zero‑sum g")
            break
        g = g / g_sum                     # enforce Σg = 1

        # 4) convex combination (mixing)
        f_new = (1.0 - step_size) * f + step_size * g
        # Numerical drift may break the simplex – re‑project occasionally
        if lp_iter % 3 == 0:
            f_new = simplex_projection(f_new)

        new_score = fast_score(f_new, fft_len, n)

        # record improvement
        if new_score < best_score - 1e-7:
            best_score = new_score
            best_f = f_new.copy()
            print(f"[LP] iter {lp_iter:02d} new best: {best_score:.6f}")

        f = f_new
        lp_iter += 1

    # --------------------------------------------------------------
    # Phase 2 – final greedy hill‑climb (fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_f, best_score = hill_climb(best_f, best_score,
                                        fft_len, n, deadline,
                                        max_iter=8000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return result
    # --------------------------------------------------------------
    final_seq = best_f.tolist()
    final_eval = evaluate_sequence(final_seq)     # external function
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
