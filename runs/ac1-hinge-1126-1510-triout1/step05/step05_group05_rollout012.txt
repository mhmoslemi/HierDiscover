def propose_candidate(seed=42, budget_s=1000, **kwargs):

    import time, numpy as np, etc.

    rng = np.random.default_rng(seed)

    max_n = 4000

    n = max_n

    fft_len = next_pow2(2*n - 1) # for conv and other.

    def norm(a): return a / a.sum() if a.sum()>0 else np.full_like(a, 1/a.shape[0])

    # seeds
    seeds = []

    def add(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() == 0: return
        arr = norm(arr)
        sc = fast_score(arr, fft_len, n)
        seeds.append((sc, arr))

    add(np.full(n, 1.0/n))
    # linear decreasing
    add(np.linspace(1, 0.01, n)) # ensure positivity
    # exponential
    for rate in np.logspace(-4, -1, 10):
        add(np.exp(-rate*np.arange(n)))
    # power law
    for p in np.linspace(0.1, 3.0, 12):
        add((np.arange(1,n+1)**(-p)))
    # sin^2
    add(np.sin(np.pi * (np.arange(n)+0.5)/n)**2 + 1e-3)
    # random Dirichlet
    for alpha in [0.3, 0.7, 1.5, 3.0]:
        for _ in range(3):
            add(rng.gamma(alpha, 1.0, size=n))
    # include prior sequence if any: try to resample to length n using interpolation.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.size > 0:
            # upsample via linear interpolation to length n
            x_old = np.linspace(0,1,prev.size)
            x_new = np.linspace(0,1,n)
            import scipy.interpolate as si
            f = si.interp1d(x_old, prev, kind='linear', fill_value='extrapolate')
            add(f(x_new))

    seeds.sort(key=lambda t: t[0])
    top_seeds = [arr for _,arr in seeds[:10]]
    best_a = top_seeds[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, best_sweep_score={best_score}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Phase L-BFGS
    phase1_deadline = start_time + 0.45 * budget_s
    lam_schedule = [0.4, 0.3, 0.2, 0.1, 0.07, 0.04, 0.02, 0.01, 0.006, 0.003, 0.001]
    maxiter_per_lam = 400
    restarts = 0
    while time.time() < phase1_deadline and restarts < 30:
        # pick seed:
        seed_vec = top_seeds[rng.integers(len(top_seeds))]
        jitter = rng.random(n) * 5e-4
        a_start = np.maximum(seed_vec + jitter, 0)
        a_start = norm(a_start)
        a_opt, sc_opt = lbfgs_continuation(a_start, lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
        restarts += 1

    # Phase flatten using L2 loss
    phase2_deadline = start_time + 0.70 * budget_s

""" 
Search for a low‑max‑convolution step sequence.

The algorithm builds a set of parametric seeds, refines them with a
continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
max‑convolution, then flattens the convolution by minimising an
L2‑distance to the uniform target, follows a multiplicative
sub‑gradient descent for the true max‑convolution and finishes with a
greedy hill‑climb.  All iterates stay on the probability simplex
(sum = 1, non‑negative).  The code respects the 1000 s budget and
returns a plain Python list.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # optimisation may fail; just skip this λ
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # a short hill‑climb after each λ helps to escape tiny stalls
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=800)
    return best_a, best_score


# ----------------------------------------------------------------------
# L2‑flattening (minimise squared deviation from uniform convolution)
# ----------------------------------------------------------------------
def l2_loss_and_grad(a: np.ndarray,
                     fft_len_conv: int,
                     fft_len_g: int):
    """
    Returns (loss, gradient) of Σ (conv - avg)² where avg = 1/(2n‑1).
    Gradient w.r.t a is 2·conv_rev(diff) restricted to the original n entries.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len_conv)[: 2 * n - 1]
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = np.sum(diff ** 2)

    a_rev = a[::-1]
    diff_rev = diff[::-1]
    fd = np.fft.fft(diff_rev, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fd * fa_rev))
    g = 2.0 * conv_g[(n - 1):(n - 1 + n)]
    return loss, g


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07   # up to 7 % of mass at i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    # --------------------------------------------------------------

        s = v.sum()
        if s > 0:
            return v / s
        else:
            return np.full_like(v, 1.0 / v.shape[0])

    # --------------------------------------------------------------
    seeds = []

        if arr.size == 0:

        arr = normalise(arr)
        sc = fast_score(arr, fft_len_conv, n)
        seeds.append((sc, arr))

    # 1) Uniform (baseline)
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    add_candidate(np.linspace(1.0, 0.001, n))

    # 3) Exponential families (rates covering many decades)
    for rate in np.logspace(-4, -0.5, 12):
        add_candidate(np.exp(-rate * np.arange(n)))

    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # 6) Random Dirichlet / Gamma draws with several concentrations
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(2):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

                add_candidate(rep)

    phase2_end = start_time + 0.68 * budget_s   # L2‑flattening (Adam)
    phase3_end = start_time + 0.88 * budget_s   # multiplicative sub‑gradient
    # final hill‑climb runs until the global deadline

    # Phase 1 – continuation L‑BFGS (many restarts)
    # --------------------------------------------------------------

    restarts = 0
    while time.time() < phase1_end and restarts < 40:

        jitter = rng.random(n) * 5e-4
        a_start = normalise(seed_vec + jitter)

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len_conv,
                                           fft_len_g,
                                           n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.25
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0

            loss, grad = l2_loss_and_grad(a, fft_len_conv, fft_len_g)
            t += 1

            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            a_candidate = a - step
            a_candidate = simplex_projection(a_candidate)
            cur_score = fast_score(a_candidate, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_candidate.copy()
                print(f"[L2‑Adam] new best score={best_score:.6f}")
            a = a_candidate

    # --------------------------------------------------------------
    # Phase 3 – multiplicative sub‑gradient descent on the true max‑conv
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        a = best_a.copy()
        it = 0
        while time.time() < phase3_end:
            conv = full_conv(a, fft_len_conv)[: 2 * n - 1]
            max_val = conv.max()
            # mask of entries that are essentially maximal
            delta = max_val * 1e-6 + 1e-12
            mask = (conv >= max_val - delta).astype(float)

            a_rev = a[::-1]
            fm = np.fft.fft(mask, fft_len_g)
            fa_rev = np.fft.fft(a_rev, fft_len_g)
            conv_g = np.real(np.fft.ifft(fm * fa_rev))

            step = 0.45 / np.sqrt(it + 1.0)
            a_new = a * np.exp(-step * g)
            a_new = normalise(a_new)

            score_new = fast_score(a_new, fft_len_conv, n)
            if score_new < best_score:
                best_score = score_new
                best_a = a_new.copy()
                print(f"[sub‑mult] iter {it:5d}, score={best_score:.6f}")

    # Phase 4 – final greedy hill‑climb (as much as time permits)
    # --------------------------------------------------------------
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    deadline,
                                    max_iter=15000)

    # Return the sequence (as a plain Python list)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")

""" 
Search for a low‑max‑convolution step sequence.

The algorithm builds a set of parametric seeds, refines them with a
continuation L‑BFGS on a smooth log‑sum‑exp surrogate of the
max‑convolution, then flattens the convolution by minimising an
L2‑distance to the uniform target, follows a multiplicative
sub‑gradient descent for the true max‑convolution and finishes with a
greedy hill‑climb.  All iterates stay on the probability simplex
(sum = 1, non‑negative).  The code respects the 1000 s budget and
returns a plain Python list.
"""
import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (real part, truncated)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(a*a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        # small hill‑climb after each λ (helps to escape tiny stalls)
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=800)
    return best_a, best_score


# ----------------------------------------------------------------------
# L2‑flattening (minimise squared deviation from uniform convolution)
# ----------------------------------------------------------------------
def l2_loss_and_grad(a: np.ndarray,
                     fft_len_conv: int,
                     fft_len_g: int):
    """
    Returns (loss, gradient) of Σ (conv - avg)² where avg = 1/(2n‑1).
    Gradient w.r.t a is 2·conv_rev(diff) restricted to the original n entries.
    """
    n = a.shape[0]
    conv = full_conv(a, fft_len_conv)[: 2 * n - 1]
    avg = 1.0 / (2 * n - 1)
    diff = conv - avg
    loss = np.sum(diff ** 2)

    a_rev = a[::-1]
    diff_rev = diff[::-1]
    fd = np.fft.fft(diff_rev, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_g = np.real(np.fft.ifft(fd * fa_rev))
    g = 2.0 * conv_g[(n - 1):(n - 1 + n)]
    return loss, g


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07   # up to 7 % of mass at i
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – we use the maximal allowed length.
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n

    # FFT lengths
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for gradient / mask convolutions

    # --------------------------------------------------------------
    # Helper to normalise a vector to the simplex.
    # --------------------------------------------------------------
    def normalise(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s > 0:
            return v / s
        else:
            return np.full_like(v, 1.0 / v.shape[0])

    # --------------------------------------------------------------
    # Generate a collection of seeds from simple parametric families.
    # --------------------------------------------------------------
    seeds = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.size == 0:
            return
        arr = normalise(arr)
        sc = fast_score(arr, fft_len_conv, n)
        seeds.append((sc, arr))

    # 1) Uniform (baseline)
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    add_candidate(np.linspace(1.0, 0.001, n))

    # 3) Exponential families (rates covering many decades)
    for rate in np.logspace(-4, -0.5, 12):
        add_candidate(np.exp(-rate * np.arange(n)))

    # 4) Power‑law families
    for p in np.linspace(0.1, 3.0, 15):
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Sinusoidal‑squared (always non‑negative)
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # 6) Random Dirichlet / Gamma draws with several concentrations
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(2):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # 7) Use the previously found construction if available (rescaled)
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.size > 0:
            try:
                import scipy.interpolate as si
                x_old = np.linspace(0.0, 1.0, prev.size)
                x_new = np.linspace(0.0, 1.0, n)
                f = si.interp1d(x_old, prev, kind='linear', fill_value='extrapolate')
                add_candidate(f(x_new))
            except Exception:
                add_candidate(np.resize(prev, n))

    # Keep the best few seeds for warm‑starts.
    seeds.sort(key=lambda t: t[0])
    top_seeds = [arr for _, arr in seeds[:12]]
    best_a = top_seeds[0].copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, best sweep score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Time budget management
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS
    phase2_end = start_time + 0.68 * budget_s   # L2‑flattening (Adam)
    phase3_end = start_time + 0.88 * budget_s   # multiplicative sub‑gradient

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (many restarts)
    # --------------------------------------------------------------
    lam_schedule = [0.4, 0.3, 0.2, 0.1, 0.07, 0.04,
                    0.02, 0.01, 0.006, 0.003, 0.001]
    maxiter_per_lam = 400
    restarts = 0
    while time.time() < phase1_end and restarts < 40:
        seed_vec = top_seeds[rng.integers(len(top_seeds))]
        jitter = rng.random(n) * 5e-4
        a_start = normalise(seed_vec + jitter)

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len_conv,
                                           fft_len_g,
                                           n)
        if sc_opt < best_score:

            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
        restarts += 1

    # --------------------------------------------------------------
    # Phase 2 – Adam on the L2 flattening objective
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        a = best_a.copy()
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.25
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
        while time.time() < phase2_end:
            loss, grad = l2_loss_and_grad(a, fft_len_conv, fft_len_g)

            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_candidate = a - step
            a_candidate = simplex_projection(a_candidate)
            cur_score = fast_score(a_candidate, fft_len_conv, n)
            if cur_score < best_score:

                best_a = a_candidate.copy()
                print(f"[L2‑Adam] new best score={best_score:.6f}")
            a = a_candidate

    # --------------------------------------------------------------
    if time.time() < phase3_end:
        a = best_a.copy()
        it = 0
        while time.time() < phase3_end:

            a_rev = a[::-1]
            fm = np.fft.fft(mask, fft_len_g)
            fa_rev = np.fft.fft(a_rev, fft_len_g)
            conv_g = np.real(np.fft.ifft(fm * fa_rev))

            score_new = fast_score(a_new, fft_len_conv, n)

                best_a = a_new.copy()
                print(f"[sub‑mult] iter {it:5d}, score={best_score:.6f}")
            a = a_new
            it += 1

    # --------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (as much as time permits)
    # --------------------------------------------------------------
    best_a, best_score = hill_climb(best_a, best_score,
                                    fft_len_conv, n,
                                    deadline,
                                    max_iter=15000)

    # --------------------------------------------------------------
