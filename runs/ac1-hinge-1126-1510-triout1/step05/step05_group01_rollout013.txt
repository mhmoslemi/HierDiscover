    deadline = time.time() + budget_s - 1

    best_a = None
    best_score = float('inf')

    # We will run M iterations of random sparse initialization, each with subgradient descent for some time.

"""
Non‑negative step function search (target < 1.503).

Idea
----
The objective is 2·n·max(conv)/sum² and is invariant under scaling,
so we can treat the sequence as a probability distribution (sum = 1).
For any n the average value of the auto‑convolution is 1/(2n).
If we place the mass on a *sparse random* set of indices (size k),
the number of ordered pairs that fall onto any given sum is
approximately Poisson with mean μ = k²/(2n).  The maximal count is
≈ μ+√(μ·log n), giving a fast‑score ≈ 1+√(2n·log n)/k.  Choosing
k≈√(2n·log n)/0.5 yields a score well below 1.5.

Algorithm
---------
1. Fix a moderate size n (8000).  All vectors are kept normalised to
   sum = 1, so the fast evaluation `fast_score` equals the true score.
2. Generate many random candidates:
   * pick a random sparsity fraction p∈[0.07, 0.25];
   * choose k = int(p·n) distinct indices;
   * either put equal weight 1/k on the chosen indices, or draw
     random exponential weights and renormalise.
   * keep the best candidate (lowest fast_score).
3. Refine the best candidate with a projected sub‑gradient descent on
   the true (non‑smoothed) max‑convolution objective.
4. Finish with a short greedy hill‑climb that repeatedly moves a
   random fraction of mass from one coordinate to another, accepting
   only improvements.
5. Return the final sequence; the harness will call `evaluate_sequence`
   on it.

The whole search respects the 1000 s time limit and prints intermediate
scores for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using a real FFT.
    a is assumed length n.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return np.real(conv)


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector (sum = 1):
        2·n·max(a*a)
    This equals the value returned by evaluate_sequence for the same
    normalised vector.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent_time(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    time_limit: float,
    eta0: float = 0.25,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for a fixed amount of wall‑clock time.
    Returns the best vector found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    t0 = time.time()
    it = 0

    while it < max_iters and (time.time() - t0) < time_limit:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        j = m - idx
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (fast_score is used as the objective)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Move a random fraction of mass from i to j, accepting only
    improvements of the fast objective.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10  # up to 10 % of source mass per move

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] == 0.0:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score
