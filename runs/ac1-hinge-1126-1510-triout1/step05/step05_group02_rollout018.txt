def subgradient_descent(a, deadline, fft_len, n):
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    step0 = 1.0
    while time.time() < deadline:
        conv = full_conv(a, fft_len)
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # compute g_i = 2*a_{k-i}
        g = np.zeros_like(a)
        i_range = np.arange(lo, hi+1)
        j_range = k - i_range
        g[i_range] = 2.0 * a[j_range]

        # determine step size (decreasing)
        step = step0 / np.sqrt(it + 1.0)

        # try backtracking line search
        new_a = a - step * g
        # projection to simplex
        new_a = simplex_projection(new_a)
        new_score = fast_score(new_a, fft_len, n)
        # if not improved, reduce step
        if new_score < best_score:
            a = new_a
            best_a = new_a.copy()
            best_score = new_score
        else:
            # adjust step down factor (e.g., halve step)
            step *= 0.5
            new_a = a - step * g
            new_a = simplex_projection(new_a)
            new_score = fast_score(new_a, fft_len, n)
            if new_score < best_score:
                a = new_a
                best_a = new_a.copy()
                best_score = new_score
        it += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

"""
Propose a low‑score non‑negative step sequence.

The search proceeds in four stages (≈1000 s total):
1.  Random‑parameter families (exponential, power, linear‑decrease, etc.)
    are sampled and the best sequence (by the fast surrogate score) is kept.
2.  A continuation L‑BFGS optimisation refines that best point on a
    smooth log‑sum‑exp approximation of the max‑convolution.
3.  Projected sub‑gradient descent on the exact max‑convolution
    (with a simple back‑tracking line search) further reduces the score.
4.  A short simulated‑annealing + greedy hill‑climb polish the final
    candidate before it is handed to ``evaluate_sequence``.

All sequences are kept on the probability simplex (sum = 1) and are
kept non‑negative.  The code works for any length up to ``max_n`` (here
4000) and automatically respects the global time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate: 2·n·max(a*a) for a sequence already normalised
    to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Random families (parameterised)

    family = rng.choice([

        'exp',
        'exp_rev',
        'sym_exp',
        'power',
        'sym_power',
        'linear_dec',
        'quad_dec',
        'inverse_sine',
    ])

    if family == 'uniform':
        a = np.full(n, 1.0)

    elif family == 'exp':
        rate = rng.uniform(0.0005, 0.02)          # small positive
        a = np.exp(-rate * np.arange(n))

    elif family == 'exp_rev':
        rate = rng.uniform(0.0005, 0.02)
        a = np.exp(-rate * np.arange(n - 1, -1, -1))

    elif family == 'sym_exp':
        c = rng.uniform(0.0005, 0.02)
        a = np.exp(-c * np.abs(np.arange(n) - (n - 1) / 2.0))

    elif family == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))

    elif family == 'sym_power':
        p = rng.uniform(0.2, 2.5)
        a = (1.0 + np.abs(np.arange(n) - (n - 1) / 2.0)) ** (-p)

    elif family == 'linear_dec':
        a = np.arange(n, 0, -1, dtype=float)

    elif family == 'quad_dec':
        q = rng.uniform(1.0, 4.0)
        a = (np.arange(n, 0, -1, dtype=float) ** q)

    elif family == 'inverse_sine':
        a = 1.0 - np.sin(np.pi * np.arange(n) / (n - 1))
    else:
        a = np.full(n, 1.0)

    # normalise to the simplex
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.02):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in freq. domain
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each lambda in lam_schedule, starting each stage from the
    result of the previous stage.
    """
    y = np.log(np.maximum(a_start, 1e-12))

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(a0: np.ndarray,
                       deadline: float,
                       fft_len: int,
                       n: int,
                       init_step: float = 1.0):
    """
    Projected sub‑gradient method on the exact max‑convolution.
    Uses a simple back‑tracking line search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline:
        conv = full_conv(a, fft_len)
        k = int(conv.argmax())

        # sub‑gradient of the max (exact)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i = np.arange(lo, hi + 1)
        j = k - i
        g = np.zeros_like(a)
        g[i] = 2.0 * a[j]

        # diminishing step size with back‑tracking
        step = init_step / np.sqrt(it + 1.0)
        for _ in range(8):
            cand = simplex_projection(a - step * g)
            cand_score = fast_score(cand, fft_len, n)
            if cand_score < best_score:
                a = cand
                best_a = cand.copy()
                best_score = cand_score
                break
            step *= 0.5
        it += 1
        if it >= 5000:                 # safety cap
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing + greedy hill‑climb
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.015
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta
        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 2000 == 0:
            T *= 0.93
            T = max(T, 1e-5)
    return best_a, best_score


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move a random fraction (up to 8 %) of mass from i to j
        delta = best[i] * rng.random() * 0.08
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use the maximum allowed size for a tighter bound)
    # --------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin

    phase0_end = start_time + 0.10 * budget_s       # random families
    phase1_end = start_time + 0.65 * budget_s       # L‑BFGS restarts
    phase2_end = start_time + 0.80 * budget_s       # sub‑gradient descent
    # remaining time for annealing & hill‑climb

    # --------------------------------------------------------------
    # Phase 0 – random families
    # --------------------------------------------------------------
    while time.time() < phase0_end:
        a_cand = random_family_sequence(n, rng)
        score_cand = fast_score(a_cand, fft_len_conv, n)
        if score_cand < best_score:
            best_score, best_a = score_cand, a_cand.copy()
            print(f"[family] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts (continuation)
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001]
    maxiter_per_lam = 600                           # more iterations per λ

    while time.time() < phase1_end:
        a_start = random_family_sequence(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the exact max
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_descent(best_a,
                                                 phase2_end,
                                                 fft_len_conv,
                                                 n,
                                                 init_step=1.0)
        print(f"[sub‑grad] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – annealing + hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        # simulated annealing
        anneal_deadline = deadline - 0.30
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] fast_score={best_score:.6f}")

        # final greedy hill‑climb
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=12000)
        print(f"[hill‑climb] fast_score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
