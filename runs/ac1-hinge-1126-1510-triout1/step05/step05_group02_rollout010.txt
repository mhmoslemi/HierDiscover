"""
Propose a low‑score non‑negative step‑function.

The search consists of four stages:

  1.  Random‑restart continuation L‑BFGS on a smooth log‑sum‑exp
      surrogate of the max‑convolution (≈40 % of the budget).
  2.  A sequence of linear‑programming updates:
      given the current a, solve
          max   Σ g_j
          s.t.  (a * g)(k) ≤ M   for all shifts k,
                g ≥ 0,
      where M = max(a*a).  The solution g has a larger ℓ₁‑norm;
      after normalising g back to ℓ₁=1 we obtain a new a with a
      strictly smaller 2·n·max(a*a) (≈30 % of the budget).
  3.  Adam optimisation of the smooth objective for fine‑tuning
      (≈20 % of the budget).
  4.  A cheap greedy hill‑climb to polish the result (remainder).

All phases respect the ℓ₁‑normalisation Σ a_j = 1, therefore the
evaluation score returned by ``evaluate_sequence`` equals the fast
score 2·n·max(conv) used throughout the optimisation.

The algorithm works for up to 4000 coefficients; here we use
n = 3000 (a good trade‑off between expressiveness and speed).
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    # keep only the useful part
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already
    normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * w) in freq. domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part that corresponds to a‑indices
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (many restarts)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence and its fast_score.
    """
    # initialise logits from the start distribution
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            # if L‑BFGS fails we simply keep the current y
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Random initialiser (various families)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    A mixture of families is used to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge',
                       'bimodal', 'sine', 'halfsine'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sine':
        a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
        a = np.maximum(a, 0.0)
    else:  # halfsine
        a = np.sin(np.pi * np.arange(n) / (n - 1))
        a = np.maximum(a, 0.0)

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a_sum
    return a


# ----------------------------------------------------------------------
# Linear‑programming update (LP‑step)
# ----------------------------------------------------------------------


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2·n‑1) × n) such that A @ g = a * g,
    i.e. each column j contains a shifted copy of a.
    """
    n = a.shape[0]
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = a
    return A


def lp_iterate(a: np.ndarray,
               fft_len: int,
               deadline: float,
               max_iter: int = 15) -> tuple[np.ndarray, float]:
    """
    Perform several LP based updates.
    Starting from a (ℓ₁‑normalised), each iteration solves:
        max Σ g_j  s.t.  (a * g) ≤ M, g ≥ 0,
    where M = max(a*a).  The solution g is normalised back to ℓ₁=1.
    If the new sequence improves the fast_score it is kept.
    Returns the best sequence and its fast_score.
    """
    n = a.shape[0]
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iter:
        # current convolution bound
        conv = full_conv(a, fft_len)
        M = conv.max()

        # build linear constraints a * g ≤ M
        A = build_conv_matrix(a)                     # dense shape (2n‑1, n)
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # linear programme: maximise Σ g_j   => minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)

        # run the LP (highs is the default high‑performance solver)
        try:
            res = linprog(c,
                          A_ub=A,
                          b_ub=b,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': max(0.0, deadline - time.time())})
        except Exception:
            break

        if not res.success:
            # LP failed – exit the LP loop
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-12:
            # no increase in ℓ₁‑norm → cannot improve further
            break

        # normalise back to ℓ₁ = 1
        g_norm = g / sum_g

        new_score = fast_score(g_norm, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            best_a = g_norm.copy()
            a = g_norm
            # continue with the new a
        else:
            # no improvement – stop LP iterations
            break

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser (fine‑tuning)
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective starting from logits y0.
    Returns the best sequence (ℓ₁‑normalised) and its fast_score.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps ℓ₁ = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 5 % of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # 40 % L‑BFGS
    phase1_end = start_time + 0.40 * budget_s

    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.001]
    maxiter_per_lam = 600

    # many short restarts – keep trying as long as we have time
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 30 % Linear‑programming refinement
    # --------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = lp_iterate(best_a,
                                        fft_len_conv,
                                        phase2_end,
                                        max_iter=12)
        print(f"[LP] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 20 % Adam fine‑tuning (smooth surrogate)
    # --------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = deadline - 0.3
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           adam_deadline,
                                           lam_factor=0.002,
                                           lr=0.15)
        print(f"[Adam] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Remaining time – greedy hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=8000)
        print(f"[final hill‑climb] fast_score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function supplied by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
