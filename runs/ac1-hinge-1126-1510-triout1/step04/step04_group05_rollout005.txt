def proj_simplex(v):
    # v is a numpy array

Implementation details for `proj_simplex`:

```
def proj_simplex(v):
    # projects onto {x: x >= 0, sum x = 1}
    # from the paper "Efficient projections onto the l1-ball", but for simplex:
    n = v.shape[0]
    if v.sum() == 1 and np.all(v >= 0):
        return v.copy()
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(1, n+1)) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid exponentiated‑gradient + hill‑climbing optimizer.

The search works on a simplex (non‑negative entries summing to 1) of a
chosen length *n*.  Each restart begins from a random or heuristic
distribution.  A fast exponentiated‑gradient (EG) descent directly
minimises the true objective `max(a*a)` by moving mass away from the
indices that contribute most to the current peak of the autocorrelation.
The EG update keeps the iterate inside the simplex without an explicit
projection.  After a short EG phase we polish the current best solution
with the targeted hill‑climb from the reference implementation and a
fallback random hill‑climb.  The whole loop runs until the time budget
expires and finally returns the best sequence found.

The method differs from the previous L‑BFGS based approach: it uses a
true sub‑gradient of the max‑convolution (rather than a smoothed
approximation) and a KL‑mirror descent (exponentiated‑gradient) update,
which is cheap, memory‑friendly and well‑suited for many restarts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two >= x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT.
    Returns the first 2*n‑1 entries (the full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) for a probability vector a.
    This is exactly the value returned by ``evaluate_sequence`` when sum(a)=1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x >= 0, sum(x) = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Hill‑climbing refinements (re‑used from the reference implementation)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 4000,
) -> tuple[np.ndarray, float]:
    """Random hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 5000,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Greedy sub‑gradient hill‑climb that reduces the current peak."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to the maximal convolution entry
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random move
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def eg_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Run exponentiated‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and its score.
    """
    a = a_start.copy()
    # ensure strict positivity to avoid stagnation
    eps = 1e-12
    a = np.maximum(a, eps)
    a = a / a.sum()

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline and t < max_iter:
        t += 1
        eta = eta0 / np.sqrt(t)

        conv = conv_fft(a, fft_len, n)
        k_max = int(np.argmax(conv))

        # sub‑gradient of max‑convolution (positive direction)
        d = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        d[mask] = 2.0 * a[idx[mask]]

        # EG update: a ← a ⊙ exp(−η d)  (⊙ = element‑wise product)
        a = a * np.exp(-eta * d)
        s = a.sum()
        if s == 0.0:
            # should not happen, but recover from numerical collapse
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        # occasional early exit if improvement stalls
        if t % 500 == 0 and best_score < 1.51:  # heuristic early stop
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative vector) that minimises

        2·n·max(a*a) / (sum(a)^2)

    using exponentiated‑gradient descent, followed by targeted and random
    hill‑climbs.  The function returns the best sequence found within the
    given time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin for final eval

    # --------------------------------------------------------------
    # Problem size – a compromise between flexibility and speed
    # --------------------------------------------------------------
    default_n = 3000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            n = default_n
            a0 = arr[:n].copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            a0 = a0 / s if s > 1e-12 else np.full(n, 1.0 / n)
        else:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    fft_len = next_pow2(2 * n - 1)

    # keep the globally best solution
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)

    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    restart = 0

    # --------------------------------------------------------------
    # Main optimisation loop – many short restarts with EG descent
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        # ----------- choose a diverse starting distribution -----------
        mode = rng.choice(["uniform", "random", "exp", "power", "edge", "perturb"])
        if mode == "uniform":
            a_start = np.full(n, 1.0 / n)
        elif mode == "random":
            a_start = rng.random(n)
            a_start = a_start / a_start.sum()
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a_start = np.exp(-rate * np.arange(n))
            a_start = a_start / a_start.sum()
        elif mode == "power":
            p = rng.uniform(0.2, 2.5)
            a_start = (np.arange(1, n + 1, dtype=float) ** (-p))
            a_start = a_start / a_start.sum()
        elif mode == "edge":
            # concentrate a modest amount at both ends
            m_edge = max(1, int(n * rng.uniform(0.005, 0.02)))
            edge_val = (np.sqrt(n / 2.0) / n)  # heuristic larger edge weight
            a_start = np.full(n, 1.0 / n)
            a_start[:m_edge] = edge_val
            a_start[-m_edge:] = edge_val
            a_start = a_start / a_start.sum()
        else:  # perturb the current best solution
            a_start = best_a + rng.normal(scale=0.05, size=n)
            a_start = np.maximum(a_start, 0.0)
            s = a_start.sum()
            a_start = a_start / s if s > 1e-12 else np.full(n, 1.0 / n)

        # -------------- EG descent on the exact max convolution -------
        eg_deadline = min(deadline, time.time() + 0.9)  # limit per restart
        a_opt, score_opt = eg_descent(
            a_start,
            fft_len,
            n,
            eg_deadline,
            max_iter=4000,
            eta0=0.9,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{restart}] mode={mode} EG improved fast score = {best_score:.6f}")

        # -------------- polishing with targeted hill‑climb ------------
        polish_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            polish_deadline,
            max_moves=4000,
            frac=0.07,
        )
        restart += 1

    # --------------------------------------------------------------
    # Final deep polishing (random hill‑climb) if time remains
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            final_deadline,
            max_iter=8000,
        )
        # one last quick targeted pass
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            final_deadline,
            max_moves=2000,
            frac=0.06,
        )

    # --------------------------------------------------------------
    # Evaluate and return the best sequence
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
