def active_set_lp_refine(a, n, deadline, n_iter=8, alpha=0.5):
    # a: array sum=1.
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    # Determine fft lengths for conv
    fft_len_conv = next_pow2(2*n-1)
    for it in range(n_iter):
        if time.time() >= deadline: break
        # conv
        conv = full_conv(best_a, fft_len_conv)
        M = conv.max()
        # active shifts
        eps = M * 1e-6 + 1e-12
        active_idxs = np.where(conv >= M - eps)[0]
        # limit size to at most Kmax=10, choose random subset if >10
        Kmax = 12
        if active_idxs.size > Kmax:
            # choose top Kmax with highest conv values
            sorted_idxs = active_idxs[np.argsort(-conv[active_idxs])][:Kmax]
            active_idxs = sorted_idxs
        # Build A matrix
        rows = []
        cols = []
        data = []
        # For each k in active_idxs:
        for r, k in enumerate(active_idxs):
            # i loop
            i_low = max(0, k-(n-1))
            i_high = min(n-1, k)
            # For i in [i_low,i_high], j = k-i
            for i in range(i_low, i_high+1):
                j = k - i
                rows.append(r)
                cols.append(j)
                data.append(best_a[i])
        # Add last column for M_var with coefficient -1
        M_var_index = n  # last variable
        # We will have n variables b_0...b_{n-1} and M_var at index n
        total_vars = n + 1
        # build A_ub matrix
        from scipy.sparse import coo_matrix
        # Build constraint for each active row: Σ a_i b_{k-i} - M_var <= 0
        # A_ub exists as coo matrix with shape (len(active_idxs), total_vars)
        # Fill with data for b coefficients and -1 for M_var column
        rows_arr = np.array(rows, dtype=int)
        cols_arr = np.array(cols, dtype=int)
        data_arr = np.array(data, dtype=float)
        # Add column for M_var
        rows_m = np.arange(len(active_idxs))
        cols_m = np.full(len(active_idxs), M_var_index, dtype=int)
        data_m = np.full(len(active_idxs), -1.0, dtype=float)

        rows_comb = np.concatenate([rows_arr, rows_m])
        cols_comb = np.concatenate([cols_arr, cols_m])
        data_comb = np.concatenate([data_arr, data_m])
        A_ub = coo_matrix((data_comb, (rows_comb, cols_comb)), shape=(len(active_idxs), total_vars)).tocsc()
        b_ub = np.zeros(len(active_idxs), dtype=float)

        # Equality constraint: sum b = 1
        A_eq = coo_matrix((np.ones(n), (np.zeros(n), np.arange(n))), shape=(1, total_vars)).tocsc()
        # Actually need shape (1,total_vars). But easier: use SciPy's linprog which accepts dense arrays for equality; we can give A_eq as dense row.
        A_eq_dense = np.zeros((1, total_vars))
        A_eq_dense[0,:n] = 1.0
        b_eq = np.array([1.0])

        # bounds: b_i >=0, M_var >=0
        bounds = [(0, None)] * n + [(0, None)]

        # Use linprog
        from scipy.optimize import linprog
        res = linprog(c=np.concatenate([np.zeros(n), np.array([1.0])]), A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq_dense, b_eq=b_eq, bounds=bounds,
                      method='highs')
        if not res.success:
            # fallback to simple gradient step
            break
        b_opt = res.x[:n]
        # updating a as convex combination
        # use decreasing alpha: e.g., alpha = max(0.1, alpha * 0.6)
        a_new = (1 - alpha) * best_a + alpha * b_opt
        a_new = a_new / a_new.sum()
        # compute new score
        score_new = fast_score(a_new, fft_len_conv, n)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new
        # reduce alpha
        alpha *= 0.6
        # maybe early stop if no improvement
    return best_a, best_score

"""
Advanced two‑stage optimizer for the step‑function score.

Stage 1 – many continuation L‑BFGS runs (smooth log‑sum‑exp max) on
randomly generated simplex points.  Stage 2 – a cheap linear‑programming
refinement that directly minimises the peak of the *aperiodic*
autocorrelation (the quantity appearing in the score).  After these
phases a short simulated‑annealing + greedy hill‑climb finish the job.

The LP works on the *active* constraints only (the shifts where the
convolution is currently maximal) which makes each solve tiny
(~10 × n variables) and allows many refinement steps within the time
budget.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1 (the denominator of evaluate_sequence becomes 1).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used by L‑BFGS/Adam)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * w) in freq domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)            # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # optimisation failure – skip this lam
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimisation (fine‑tuning)
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimiser on the smoothed‑max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n,
                                     fft_len_conv, fft_len_g,
                                     lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (active‑set)
# ----------------------------------------------------------------------


def lp_refine_active(a: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 8,
                     alpha_start: float = 0.5,
                     eps_rel: float = 1e-3):
    """
    Small LP that directly minimises the maximal asterisk‑convolution.
    Only the shifts that are (almost) maximal are kept as constraints.
    Returns a refined sequence and its fast_score.
    """
    fft_len = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    alpha = alpha_start

    while time.time() < deadline and max_iters > 0:
        max_iters -= 1

        conv = full_conv(best_a, fft_len)
        M = conv.max()
        # active shifts (within a relative tolerance)
        active = np.where(conv >= M - eps_rel * M)[0]
        # keep at most 12 constraints for speed
        if active.size > 12:
            # keep the indices with largest values
            order = np.argsort(-conv[active])[:12]
            active = active[order]

        rows = len(active)
        # build dense A_ub (rows × (n+1))
        A_ub = np.zeros((rows, n + 1), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            # fill the row with a_i coefficients
            for i in range(i_low, i_high + 1):
                j = k - i
                A_ub[r, j] = best_a[i]
            A_ub[r, -1] = -1.0                     # -M_var ≤ 0

        b_ub = np.zeros(rows, dtype=np.float64)

        # equality: sum(b) = 1
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        # bounds: b_i ≥ 0, M_var ≥ 0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        # objective: minimise M_var (last variable)
        c = np.concatenate([np.zeros(n), np.array([1.0])])

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            # LP failed – abort refinement
            break

        b_opt = res.x[:n]

        # convex combination with decreasing alpha
        a_new = (1.0 - alpha) * best_a + alpha * b_opt
        a_new /= a_new.sum()
        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score:
            best_score = new_score
            best_a = a_new
        # reduce step size for next iteration
        alpha = max(0.05, alpha * 0.6)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Simple annealing – move a random fraction of mass between two cells,
    occasionally accepting worsening moves.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()

    T = 0.02
    it = 0
    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.08   # up to 8 % of source mass
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 2500 == 0:
            T *= 0.90
            T = max(T, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 4000):
    """
    Deterministic hill‑climb – move a small random fraction from one index
    to another only if the move strictly improves the score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # up to 4 % of the mass at i is moved
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Random initialiser (produces points on the simplex)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability distribution.  Several families are mixed
    to increase diversity.
    """
    mode = rng.choice(['uniform', 'dirichlet', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'dirichlet':
        # small concentration => sparsity
        conc = rng.uniform(0.05, 0.5)
        a = rng.gamma(conc, 1.0, size=n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:                     # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000
    # try to use the previously found construction if it exists
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n // 2
            a0 = np.full(n, 1.0 / n)
    else:
        # start with a medium‑sized uniform distribution
        n = max_n // 2            # 2000 – a good trade‑off between speed and flexibility
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a
    fft_len_g = next_pow2(3 * n - 2)      # length for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    phase1_end = start_time + 0.45 * budget_s   # L‑BFGS phase
    phase2_end = start_time + 0.70 * budget_s   # Adam fine‑tuning
    phase3_end = start_time + 0.85 * budget_s   # LP refinement
    phase4_end = start_time + 0.95 * budget_s   # simulated annealing

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.01,
                    0.006, 0.003, 0.0015, 0.0008, 0.0004]
    maxiter_per_lam = 600

    while time.time() < phase1_end:
