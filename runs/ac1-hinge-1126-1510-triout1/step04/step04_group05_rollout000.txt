def projected_subgradient_descent(a0, max_iter, time_limit):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    start = time.time()
    for it in range(max_iter):
        # compute conv and max value with current a
        conv = conv_fft(a, fft_len, n)
        max_val = conv.max()
        # early exit if time
        if time.time() - start > time_limit:
            break
        # indexes achieving max (within tolerance)
        k_idxs = np.where(np.abs(conv - max_val) <= 1e-12)[0]
        # compute subgradient g
        g = np.zeros_like(a)
        for k in k_idxs:
            # g_i includes a_{i+k} and a_{i-k}
            if k < n:
                g[:n-k] += a[k:]
            if k > 0:
                g[k:] += a[:n-k]
        # gradient of objective f = 2*n*max_val: df/da = 2*n * g (since derivative of max contribution)
        # Actually need to multiply by 2*n
        grad = 2 * n * g
        # step size schedule: α = C / sqrt(it+1)
        step = 1.0 / np.sqrt(it + 1)
        a_new = a - step * grad
        # projection onto simplex (nonnegative, sum=1)
        # Use standard Euclidean projection onto simplex
        # ensure nonnegative (though a_new may be negative)
        a_new = project_simplex(a_new)
        # evaluate new objective
        score = fast_score(a_new, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
```

def project_simplex(v):
    v = np.maximum(v, 0) # ensure nonneg?
    # but need to project onto sum=1, all entries >=0.
    # algorithm:
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid sub‑gradient + hill‑climb optimizer for the max‑auto‑correlation
objective.

The algorithm works on the probability simplex (the coefficients are
non‑negative and sum to one).  It repeatedly

1. draws a random start (uniform, exponential, power‑law or Dirichlet);
2. performs a projected sub‑gradient descent on the *exact* objective
   `2·n·max(a*a)` (the max of the linear convolution);
3. refines the result with a fast greedy hill‑climb that moves mass from
   the indices that contribute most to the current worst convolution peak
   to the indices that contribute the least;
4. periodically applies a tiny random hill‑climb for polishing.

The sub‑gradient of `max_k c_k(a)` with respect to a_i is a sum of the
shifted vectors belonging to the *active* lag(s) k*.  After each gradient
step the vector is projected back onto the simplex (Euclidean projection
à la Duchi et al.).  The method is lightweight, uses only FFTs for the
convolution, and can be run many times within the 1000 s budget.  By
using a larger problem size (n≈8000) the theoretical lower bound
`2·n·1/(2n‑1)≈1` becomes attainable, giving a good chance to beat the
target 1.503.

The final sequence is returned as a plain Python list; `evaluate_sequence`
will be invoked once to obtain the official score.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize  # only for the final safety polish (optional)

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns the first 2*n‑1 entries.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector a.
    Returns 2·n·max(a*a) (the denominator sum(a)^2 equals 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho_arr.size == 0:
        # All entries are non‑positive – return the uniform distribution.
        return np.full(n, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# Optimisation primitives
# ----------------------------------------------------------------------


def subgradient_descent(
    a0: np.ndarray,
    fft_len: int,
    n: int,
    time_limit: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact objective.
    Returns the best vector seen and its fast_score.
    """
    a = a0.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    start = time.time()
    for it in range(1, max_iter + 1):
        if time.time() > time_limit:
            break

        # current convolution and the maximal value
        conv = conv_fft(a, fft_len, n)
        max_val = conv.max()

        # indices attaining the maximal value (within a tight tolerance)
        eps = 1e-12
        ks = np.where(np.abs(conv - max_val) <= eps)[0]

        # gradient of max_val w.r.t. a
        g = np.zeros_like(a)
        # each active lag contributes a_i shifted left and right
        for k in ks:
            if k < n:
                g[: n - k] += a[k:]
            if k > 0:
                g[k:] += a[: n - k]

        # full gradient of 2·n·max_val
        grad = 2.0 * n * g

        # diminishing stepsize (tuned empirically)
        step = 0.5 / math.sqrt(it)

        a = a - step * grad
        a = project_simplex(a)

        # evaluate the new point
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best = a.copy()

    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that repeatedly transfers a small fraction of mass
    from the index that contributes most to the current worst convolution
    peak to the index that contributes the least.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each coordinate to the peak k_max
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        # fallback if somehow all contributions equal
        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-14:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1
    return best, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb on the simplex.
    Randomly moves a fraction of mass between two positions and keeps the
    move only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10
    while time.time() < deadline and max_iter > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        max_iter -= 1
    return best, best_score

        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(
                f"[{iteration}] mode={mode} sub‑grad improved fast score = {best_score:.6f}"
            )

        # --------------------------------------------------------------
        # Aggressive targeted hill‑climb for polishing
        # --------------------------------------------------------------
        hc_deadline = min(deadline, time.time() + 0.5)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            hc_deadline,
            max_moves=4000,
            frac=0.07,
        )

        iteration += 1

    # ------------------------------------------------------------------
    # Final deep polishing (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        # a longer sub‑gradient pass
        best_a, best_score = subgradient_descent(
            best_a, fft_len, n, final_deadline, max_iter=8000
        )
        print(f"[final‑subgrad] fast score = {best_score:.6f}")

        # a random hill‑climb for escaping tiny flat‑spots
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=5000,
        )

    # ------------------------------------------------------------------
    # Return the sequence after the official evaluation (clipping, etc.)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
