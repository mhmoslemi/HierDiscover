"""
Optimisation of a non‑negative step sequence for the evaluation function

    E(a) = 2·n·max_k (a*a)[k] / (sum a)^2 .

The algorithm consists of the following stages (all respecting the simplex
constraint sum a = 1, a_i ≥ 0):

  0. Quick parametric sweep (uniform, linear, exponential, power‑law,
     sinusoid, edge‑heavy) plus a large *random* sweep over several
     families (exponential, double‑exponential, sigmoid, Gaussian,
     mixtures …).  The best few seeds are kept.

  1. Continuation L‑BFGS on a smoothed max (log‑sum‑exp) with a decreasing
     λ‑schedule.  Each seed is run a few times with a tiny jitter and the
     best result of all restarts is retained.

  2. Projected sub‑gradient descent with a simple back‑tracking line‑search.
     The optimisation is run several times from the current best point
     (different jitter each run) to escape shallow minima.

  3. Simulated‑annealing that accepts worsening moves with a slowly cooling
     temperature, followed by a short greedy hill‑climb.

  4. Final extensive hill‑climb for polishing the result.

All convolutions use FFTs (O(n log n)).  The total time budget is split
automatically; the best sequence found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1.  The true score is 2·n·max/ sum(a)^2, but the ratio is
    scale‑invariant, so we only need max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with simple back‑tracking line‑search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution and active mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (non‑smooth) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Random moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 5000 == 0:
            T = max(T * 0.92, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS (smooth max) with jittered restarts
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  The best vector seen is returned.
    """
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Occasionally L‑BFGS may fail – ignore and continue.
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()

        # a short hill‑climb after each λ can sharpen the result
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score

def generate_random_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Sample a random shape from a variety of families and normalise it.
    """
    typ = rng.choice(['exp', 'double_exp', 'power', 'sigmoid',
                      'gauss', 'lin', 'edge', 'mix_exp_uni',
                      'mix_two_exp'])

    if typ == 'exp':
        rate = rng.uniform(0.0002, 0.008)
        arr = np.exp(-rate * np.arange(n))

    elif typ == 'double_exp':
        rate1 = rng.uniform(0.0003, 0.006)
        rate2 = rng.uniform(0.0003, 0.006)
        split = rng.integers(1, n)
        left = np.exp(-rate1 * np.arange(split))
        right = np.exp(-rate2 * np.arange(n - split))
        # shift the right part so the whole vector is continuous
        arr = np.concatenate([left, right * left[-1] / right[0]])

    elif typ == 'power':
        p = rng.uniform(0.2, 2.5)
        arr = (np.arange(1, n + 1, dtype=float) ** (-p))

    elif typ == 'sigmoid':
        mu = rng.uniform(0, n)
        sigma = rng.uniform(20, 400)
        arr = 1.0 / (1.0 + np.exp((np.arange(n) - mu) / sigma))

    elif typ == 'gauss':
        mu = rng.uniform(0, n - 1)
        sigma = rng.uniform(20, 400)
        arr = np.exp(-0.5 * ((np.arange(n) - mu) / sigma) ** 2)

    elif typ == 'lin':
        slope = rng.uniform(-0.0015, 0.0015)
        intercept = rng.uniform(0.0, 1.0)
        arr = intercept + slope * np.arange(n)

    elif typ == 'edge':
        edge_frac = rng.uniform(0.003, 0.04)
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n)
        arr[:m] = 1.0
        arr[-m:] = 1.0

    elif typ == 'mix_exp_uni':
        # mixture of a uniform baseline and a decaying exponential
        alpha = rng.uniform(0.0, 0.5)          # weight of uniform part
        rate = rng.uniform(0.0003, 0.006)
        exp_part = np.exp(-rate * np.arange(n))
        exp_part = np.maximum(exp_part, 0.0)
        exp_part /= exp_part.sum()
        uniform = np.full(n, 1.0 / n)
        arr = alpha * uniform + (1.0 - alpha) * exp_part

    elif typ == 'mix_two_exp':
        # two exponentials with different rates and a mixing weight
        rate1 = rng.uniform(0.0003, 0.008)
        rate2 = rng.uniform(0.0003, 0.008)
        mix = rng.uniform(0.0, 1.0)
        exp1 = np.exp(-rate1 * np.arange(n))
        exp2 = np.exp(-rate2 * np.arange(n))
        arr = mix * exp1 + (1.0 - mix) * exp2

    else:  # fallback – uniform
        arr = np.full(n, 1.0 / n)

    arr = np.maximum(arr, 0.0)
    s = arr.sum()
    if s > 0.0:
        arr /= s
    else:
        arr = np.full(n, 1.0 / n)
    return arr

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000                     # stay within a few thousand entries
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            tot = a0.sum()
            a0 = a0 / tot if tot > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)    # for a*a
    fft_len_g = next_pow2(3 * n - 2)       # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep + large random sweep
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # deterministic families
    add_candidate(np.full(n, 1.0 / n))                     # uniform
    lin = np.arange(n, 0, -1, dtype=float)                # linear decreasing
    add_candidate(lin)
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        add_candidate(np.exp(-rate * np.arange(n)))
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))
    # sinusoid‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # random Dirichlet / Gamma draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # ------------------------------------------------------------------
    # Randomised sweep (time‑boxed)
    # ------------------------------------------------------------------
    sweep_deadline = time.time() + 0.10 * budget_s   # ~10 % of total
    rand_candidates = 0
    while time.time() < sweep_deadline:
        arr = generate_random_candidate(n, rng)
        add_candidate(arr)
        rand_candidates += 1
    print(f"[sweep] deterministic + {rand_candidates:,} random candidates")

    # keep the top seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:12]]
    print(f"[sweep] best raw fast_score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin
    phase1_end = start_time + 0.55 * budget_s       # ~55 % for L‑BFGS
    phase2_end = start_time + 0.80 * budget_s       # next ~25 % for sub‑gradient
    phase3_end = deadline                           # remaining for annealing

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.06, 0.03,
                    0.015, 0.008, 0.004, 0.002, 0.001,
                    0.0005, 0.0002]
    maxiter_per_lam = 250   # a little lighter than the original to allow more restarts

    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (multiple restarts)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sub_deadline = min(phase2_end, deadline - 0.5)
        # we run a few independent sub‑gradient runs to increase chance
        # of escaping shallow minima.
        sg_runs = 3
        for i in range(sg_runs):
            sg_start = best_a + rng.random(n) * 5e-4
            sg_start = np.maximum(sg_start, 0.0)
            sg_start /= sg_start.sum()
            new_a, new_score = subgradient_optimize(sg_start,
                                                    sub_deadline,
                                                    fft_len_conv,
                                                    fft_len_g,
                                                    n)
            if new_score < best_score:
                best_score, best_a = new_score, new_a.copy()
                print(f"[sub‑gradient run {i+1}] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing + hill‑climb
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase3_end)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    # ------------------------------------------------------------------
    # Return the sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
