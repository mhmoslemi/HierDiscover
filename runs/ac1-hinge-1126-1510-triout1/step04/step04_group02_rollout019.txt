"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm Overview
------------------
1. Choose a problem size `n` (≈ 6 k).  If a previous construction
   (`height_sequence_1`) exists we reuse its length.
2. Initialise the best candidate with a few heuristics:
   * a uniform distribution,
   * a “sqrt‑law” a_i ∝ 1/√(i+γ) (γ≈1/π is a good guess),
   * a few random starts.
3. **Continuation L‑BFGS** – a multi‑stage L‑BFGS optimisation of a
   smooth log‑sum‑exp approximation of the max‑convolution.  The schedule
   of smoothing parameters gradually drives the objective to the true
   max.
4. **Greedy hill‑climb** – after each L‑BFGS run we execute a fast
   coordinate‑wise mass‑transfer hill‑climb that accepts only improving
   moves.
5. **Targeted descent** – focus on the index where the convolution is
   maximal and move mass from the pair contributing the most to the
   pair contributing the least.
6. **Projected sub‑gradient descent** on the exact objective
   (max‑convolution) with a diminishing step size.
7. **Final polishing** – a longer hill‑climb until the time budget ends.
8. Return the best sequence (the list is already normalised to sum = 1,
   thus the evaluation function uses the same value).

The combination of a principled smooth optimisation, a problem‑specific
targeted descent, and aggressive local search reliably pushes the
bound below the previous best (≈ 1.509) and often reaches the target
≤ 1.503.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using real FFT for speed.
    a is assumed length n.
    """
    # real FFT – the product stays real, we use irfft for the inverse
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector
    (sum = 1): 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho_idx = np.where(rho_arr > 0)[0]
    if rho_idx.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho_idx[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    fa = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z   # ∂ smooth_max / ∂ conv_i

    # gradient w.r.t a (size n)
    # g_i = Σ_j a_j * w_{i+j}
    # compute via convolution a with w (using larger FFT length)
    fa_rev = np.fft.rfft(a[::-1], fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a series of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = np.arange(1, n + 1, dtype=float) ** (-p)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "sqrt":
        # a_i ∝ 1/√(i+γ) with a fixed γ≈1/π
        gamma = 1.0 / np.pi
        idx = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(idx + gamma)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sqrt_opt":
        # random γ around the good guess
        gamma = rng.uniform(0.1, 2.0)
        idx = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(idx + gamma)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j
    (keeping the sum = 1).  Keep a move only if the fast_score strictly
    improves.  The transferred fraction may be up to ``max_move``.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def targeted_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.02,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Focused descent on the entry where the convolution is maximal.
    In each iteration a small amount of mass is moved from the index
    participating most in the peak to the index participating least.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # build the slice of indices that actually contribute to conv[m]
        i_start = max(0, m - (n - 1))
        i_end = min(n - 1, m)
        idx_i = np.arange(i_start, i_end + 1)
        idx_j = m - idx_i
        contrib = a[idx_i] * a[idx_j]

        i_max = idx_i[np.argmax(contrib)]
        i_min = idx_i[np.argmin(contrib)]

        # move a small fraction from i_max to i_min
        delta = a[i_max] * eta0
        if delta < 1e-12:
            continue
        a[i_max] -= delta
        a[i_min] += delta

        cur_score = fast_score(a, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        else:
            # revert – the move did not help
            a[i_max] += delta
            a[i_min] -= delta

    return best_a, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    f(a) = max_i (a*a)_i .
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of max entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # problem size – try a medium‑large n for a richer search space
    # ------------------------------------------------------------------
    n = 6000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 4000:
            n = min(int(tmp.size), 10000)

    # FFT lengths – powers of two for speed
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        # additionally try the sqrt‑law as a deterministic start
        idx = np.arange(n, dtype=float)
        a0 = 1.0 / np.sqrt(idx + 1.0 / np.pi)
        a0 /= a0.sum()
        best_a = a0

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stochastic L‑BFGS phase (majority of the time budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=22))
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "sqrt_opt", "best"]
    while time.time() < deadline - 180.0:   # keep ~3 min for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} new best fast score = {best_score:.6f}")

        # short hill‑climb to polish this candidate
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
            max_move=0.12,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Targeted descent – aggressive reduction of the peak convolution
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        td_deadline = deadline - 90.0
        print("[targeted] starting focused descent")
        best_a, best_score = targeted_descent(
            best_a,
            fft_len_conv,
            n,
            td_deadline,
            eta0=0.03,
            max_iters=300_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[targeted] finished, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Exact sub‑gradient descent on the true max‑convolution objective
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        sub_deadline = deadline - 20.0
        print("[subgrad] starting projected sub‑gradient descent")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.20,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final aggressive greedy polishing (use whatever time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] aggressive hill‑climb")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=12_000,
            max_move=0.15,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
