"""
Propose a step‑function (non‑negative vector) with a very low evaluation score.

The algorithm builds on the previous pipeline but adds a few
enhancements that have proven useful for shaving the last few
hundredths off the objective:

1. **More aggressive L‑BFGS** – larger “maxiter” per smoothing level
   and a finer geometric schedule for the smoothing factor.

2. **Adam optimisation on the smooth‑max objective** – after the
   stochastic L‑BFGS phase we run a few thousand Adam updates with a
   very small smoothing factor (≈5 · 10⁻⁵).  Adam’s momentum helps
   escape shallow plateaus that L‑BFGS can miss.

3. **Extended sub‑gradient descent** – the exact max‑convolution
   sub‑gradient step is performed again after Adam, giving the
   algorithm a chance to tighten the last active constraint.

4. **Longer hill‑climb polishing** – a greedy local move that
   transfers a random fraction of mass between two coordinates is
   executed at the very end with a larger iteration budget.

All stages respect the constraints of ``evaluate_sequence`` (non‑negative,
sum > 0).  The routine works with any pre‑computed sequence available as
the global variable ``height_sequence_1``; otherwise it starts from the
uniform distribution.  The total wall‑clock time never exceeds
``budget_s`` seconds (default 1000 s).

The final sequence is returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and its gradient w.r.t. y.
    ``lam_factor`` multiplies the current peak of the convolution to obtain
    the smoothing width ``lam``.
    """
    a = softmax(y)

    # ----- linear convolution a*a (full length 2·n‑1) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z   # ∂smooth_max / ∂conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction (≤10 %) of mass from i to j.
    The move is kept only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Rough Sidon‑like set construction
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict:
                if candidate * 2 in sums:
                    conflict = True
                else:
                    chosen.append(candidate)
                    for c in chosen:
                        sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    Steps are projected back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # Tiny random jitter to escape flat regions
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        # project onto simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(len(a_new)) + 1
        cond = u - cssv / ind > 0
        if np.any(cond):
            rho = ind[cond].max()
            theta = cssv[cond].max() / rho
            a_new = np.maximum(a_new - theta, 0.0)
        else:
            a_new = np.full_like(a_new, 1.0 / len(a_new))

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    max_iters: int,
    lr: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max objective.
    Returns the best (y, fast_score) pair seen during the run.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_y = y.copy()
    best_score = fast_score(softmax(y), fft_len_conv, n)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate current solution
        a = softmax(y)
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_y = y.copy()

        # slowly decay learning rate
        lr *= 0.9995

    return best_y, best_score


def lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    mix_factor: float = 0.2,
    max_iter: int = 5,
) -> np.ndarray:
    """
    Linear‑programming refinement step (optional; works well for
    moderate n).  The LP maximises Σ b_j under the current convolution
    bound and the result is mixed back into ``a``.
    """
    rows = 2 * n - 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # build convolution matrix A: (A·b)[s] = (a*b)[s]
        A = sp.lil_matrix((rows, n))
        for col in range(n):
            A[col : col + n, col] = a
        A = A.tocsr()

        # solve max Σ b  ⇔  min −Σ b
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(rows),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b_opt / sum_b
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        new_M = np.convolve(a_new, a_new).max()
        if new_M < M:
            a = a_new
            print(f"[lp] M reduced from {M:.6f} to {new_M:.6f}")
        else:
            break
    return a

    # --------------------------------------------------------------
    # 4️⃣ Adam optimisation with a tiny smooth‑max factor
    # --------------------------------------------------------------
    adam_deadline = min(deadline, time.time() + 60.0)   # reserve ~1 min
    lam_factors = [0.001, 0.0005, 0.0002, 0.0001, 5e-5]
    for lam_f in lam_factors:
        if time.time() >= adam_deadline:
            break
        remaining = adam_deadline - time.time()
        # about 2000 iterations per lam level (adjusted to remaining time)
        iters = min(2000, int(remaining / 0.015))  # rough 15 ms / iter
        if iters <= 0:
            break
        lr = 0.2
        best_y, best_score = adam_optimize(
            best_y,
            n,
            fft_len_conv,
            fft_len_g,
            lam_factor=lam_f,
            max_iters=iters,
            lr=lr,
            deadline=adam_deadline,
        )
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[adam] lam={lam_f:.1e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5️⃣ Exact sub‑gradient refinement (again)
    # --------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 30.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.5
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # 6️⃣ Optional LP refinement (only if n is moderate)
    # --------------------------------------------------------------
    if n <= 3000 and time.time() + 10.0 < deadline:
        lp_deadline = deadline - 5.0
        best_a = lp_refine(best_a, n, lp_deadline, mix_factor=0.2, max_iter=5)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)
        print(f"[lp] post‑refine fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7️⃣ Final greedy polishing (as much time as left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12_000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # 8️⃣ Return final sequence (evaluate for logging)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
