def lp_max_sum(a, M):
    n = len(a)
    m = 2*n - 1
    # Build constraint matrix A: shape (m, n)
    # For k = 0..m-1:
    # columns j where 0 <= j < n and 0 <= k-j < n => i=k-j in range.
    # that entry = a_i
    # Equivalent to A = Toeplitz(a, a[0] repeated?), but we need each row k: a[0:k+1] reversed? Let's do simple loops.
    rows = []
    cols = []
    data = []
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # i in [i_min,i_max], j = k - i
        # add a_i to A[k, j]
        # Instead of iterating each i: we can vectorize.
        # But loop is fine for n~2000: sum of rows lengths ≈ n^2 = 4e6, which might be heavy but okay for few iterations.
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(a[i])
    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    # constraints: A @ g <= M * ones
    # bounds for g: (0, None)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(m, M), bounds=[(0, None)]*n, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    g = res.x
    # scaling
    sum_g = g.sum()
    if sum_g == 0:
        return a  # fallback
    g = g / sum_g  # now sum =1
    return g

"""
Hybrid optimizer for the step‑function scoring problem.

Strategy
--------
1. Initialise a probability vector (sum = 1) on the simplex.
2. Repeated L‑BFGS minimisation of a smooth log‑sum‑exp surrogate for
   max (a*a) with a continuation schedule (large smoothing → small smoothing).
3. Projected sub‑gradient descent with momentum to improve the exact max.
4. Fine‑tune with Adam on the same smooth surrogate.
5. LP‑based mass redistribution:
   given the current sequence a and its maximal convolution value M,
   solve the linear programme
        maximise   1ᵀg
        s.t.      a * g ≤ M   (pointwise)
                 g ≥ 0
   Scale the optimum to sum(g)=1 and mix it with a.
   A few iterations of this step tend to push mass away from the
   “bottleneck’’ indices of the convolution, reducing the peak.
6. Final greedy hill‑climb (mass transfer between two coordinates) to
   polish the result.

All phases respect non‑negativity and the unit‑sum constraint, and the
overall run time is limited to the provided budget (default 1000 s).
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) assuming sum(a)=1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * w) in frequency domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each smoothing parameter in lam_schedule.
    Returns the best (a, score) found before the deadline.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with a simple momentum term.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution (2 * a_rev * mask)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/sqrt(it)
        step = 0.5 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the logit parameters
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimiser on the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Random initialiser
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'geometric'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # geometric
        # pick a decay constant that roughly matches the heuristic 1‑r ≈ 2/n
        c = rng.uniform(1.5, 2.5)
        r = 1.0 - c / n
        if r <= 0.0:
            r = 0.5
        a = (1.0 - r) * (r ** np.arange(n))
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑programming mass redistribution
# ----------------------------------------------------------------------


def lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise 1ᵀg   subject to   a * g ≤ M   (pointwise),   g ≥ 0.
    Returns the optimal g scaled to sum(g)=1.
    """
    n = a.shape[0]
    m = 2 * n - 1

    # Build the sparse matrix that represents convolution a * g.
    # Row k (0 ≤ k < 2n‑1) contains a_i for every i with 0≤i≤n‑1 and 0≤k‑i≤n‑1.
    rows = []
    cols = []
    data = []

    for i in range(n):
        # column i receives a[i] on rows i … i+n‑1
        start = i
        rows.extend(range(start, start + n))
        cols.extend(range(n))
        data.extend([a[i]] * n)

    A = sp.csr_matrix((data, (rows, cols)), shape=(m, n))

    # Solve the LP with the HiGHS interior‑point method (fast for sparse problems)
    res = linprog(c=-np.ones(n),          # minimise -sum(g)  → maximise sum(g)
                  A_ub=A,
                  b_ub=np.full(m, M),
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'presolve': True})
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)

    g = res.x
    total = g.sum()
    if total == 0:
        # fallback – return the current a (already feasible)
        return a.copy()
    # Scale to unit sum (still feasible because constraints are linear)
    g = g / total
    return g


def lp_iterative_improve(a0: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP mass‑redistribution step and mix the result with
    the current vector.  Keeps the best sequence found.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)
    t = 0.5                     # initial mixing weight
    iteration = 0

    while time.time() < deadline:
        # current maximal convolution value
        conv = full_conv(a, fft_len)
        M = conv.max()

        try:
            g = lp_max_sum(a, M)
        except Exception as e:
            print("[LP] solve error:", e)
            break

        # convex combination
        a_new = (1.0 - t) * a + t * g
        # enforce positivity and renormalise
        a_new = np.clip(a_new, 1e-12, None)
        a_new = a_new / a_new.sum()

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best_score, best_a = new_score, a_new.copy()
            print(f"[LP iter {iteration}] improved score = {best_score:.6f}")
            # become more aggressive when we see improvement
            t = min(0.9, t + 0.05)
        else:
            # become more conservative
            t = max(0.1, t * 0.5)

        a = a_new
        iteration += 1

    return best_a, best_score

    max_n = 3000                                  # reasonable size for the budget

        n = max_n // 3

    fft_len = next_pow2(2 * n - 1)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n}, score={best_score:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    phase1_end = start_time + 0.30 * budget_s      # L‑BFGS stage
    phase2_end = start_time + 0.65 * budget_s      # sub‑grad + Adam stage

    # remaining time → final hill‑climb

    # --------------------------------------------------------------

    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
