"""
Propose a low‑score non‑negative step sequence.

The search consists of three stages:

1.  A large number of short L‑BFGS runs on a smooth approximation of the
    max‑convolution (log‑sum‑exp).  The smoothing parameter (``lam``) is
    gradually decreased (continuation) and many random restarts are used.
    This quickly finds good basins of attraction.

2.  A focused projected sub‑gradient descent that uses only the *active*
    constraints (the indices where the convolution attains its maximum)
    for a very sharp gradient.  Momentum and a decreasing step size provide
    a stable descent.

3.  An aggressive greedy hill‑climb that transfers a small random fraction
    of mass between two coordinates and accepts only strictly improving
    moves.  The number of iterations is enlarged compared with the previous
    baseline.

The whole procedure respects the simplex (non‑negative entries, sum = 1) and
runs entirely within the allowed time budget.  The best sequence found is
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (only the useful part is returned)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective 2·n·max(a*a) for a distribution that
    already satisfies sum(a)=1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random probability vector on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (used for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth max based on log‑sum‑exp.
    Returns (objective, gradient) w.r.t. the logits y.
    """
    # soft‑max maps logits to the simplex
    a = np.exp(y - np.max(y))
    a = a / a.sum()

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max (log‑sum‑exp)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution (frequency domain)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (many random restarts)
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # any numerical problem – skip this stage
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Focused projected sub‑gradient descent (only true max constraints)
# ----------------------------------------------------------------------


def subgradient_optimize_focused(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """
    Projected sub‑gradient descent that uses a mask consisting only of the
    (near‑)maximal convolution entries.  Momentum and a 1/√t stepsize are
    employed.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # compute full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        # mask only the truly maximal entries (tiny epsilon)
        eps = max_val * 1e-14 + 1e-16
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ∝ 1/√t
        step = 0.6 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1
    return best_a, best_score

# Adam optimiser on the logit parametrisation (fine tuning)
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.003,
                  lr: float = 0.12):
    """
    Adam optimisation of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = np.exp(y - np.max(y))
    best_a = best_a / best_a.sum()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = np.exp(y - np.max(y))
        a = a / a.sum()
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

# Simulated annealing (short)
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Very short simulated‑annealing phase that randomly moves a fraction of
    mass between two coordinates and accepts worsening moves with a
    geometrically decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 2500 == 0:
            T = max(T * 0.90, 1e-5)
    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000):
    """
    Greedy mass‑transfer hill‑climb.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move a modest fraction (up to 5 %) of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000                       # we use the largest allowed size
    n = max_n

    # initialise from a previous solution if available, otherwise random
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin
    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS phase (55 %)
    phase2_end = start_time + 0.85 * budget_s   # focused sub‑gradient (30 %)
    # remaining ~15 % will be used for Adam / anneal / hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with continuation
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.4, 0.3, 0.22, 0.16, 0.12, 0.09,
                    0.07, 0.05, 0.035, 0.02, 0.012, 0.007,
                    0.004, 0.002, 0.001, 0.0005]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – focused projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = phase2_end - 0.4
        best_a, best_score = subgradient_optimize_focused(best_a,
                                                          sg_deadline,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = deadline - 0.7
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(logits,
                                           n,
                                           fft_len_conv,
                                           fft_len_g,
                                           adam_deadline,
                                           lam_factor=0.002,
                                           lr=0.13)
        print(f"[Adam] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated annealing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        anneal_deadline = deadline - 0.4
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – final greedy hill‑climb
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)
    print(f"[final hill‑climb] best score={best_score:.6f}")

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
