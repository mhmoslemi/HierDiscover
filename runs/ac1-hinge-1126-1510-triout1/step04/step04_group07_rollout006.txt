def projected_gradient_descent(a0, time_limit, start_time):
    n = len(a0)
    a = a0.copy()
    best_seq = a.copy()
    best_val = fast_score(a)
    lr = 0.1
    momentum = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < start_time + time_limit:
        it += 1
        # compute conv and max
        conv = conv via FFT (2n-1)
        M = conv.max()
        cur_val = 2*n*M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute subgradient g
        eps_act = 1e-12
        active = np.where(conv >= M - eps_act)[0]
        # For each active, accumulate
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            inds = np.arange(i_start, i_end+1)
            # j = k - i
            g[inds] += 2.0 * a[k - inds]
        # momentum update
        momentum = beta * momentum + (1-beta) * g
        # adaptive lr: maybe lr = initial / sqrt(it)
        step = lr / np.sqrt(it + 1)
        a_new = a - step * momentum
        # projection onto simplex
        a = project_simplex(a_new)
        # occasional jitter
        if it % 5000 == 0:
            sigma = 0.01/np.sqrt(n)
            a += sigma * np.random.randn(n)
            a = project_simplex(a)
    return best_seq, best_val

def projected_gradient_descent(a0, time_limit, start_time):
    n = a0.size
    a = a0.copy()
    best_seq = a.copy()
    best_val = fast_score(a)
    momentum = np.zeros_like(a)
    beta = 0.9
    lr0 = 1.0
    it = 0
    while True:
        now = time.time()
        if now > start_time + time_limit:
            break
        it += 1
        # convolution
        fft_len = next_pow2(2*n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute subgradient
        eps_act = 1e-12
        active = np.where(conv >= M - eps_act)[0]
        g = np.zeros_like(a)
        # active set loops
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            # indices from i_start to i_end inclusive
            i_range = np.arange(i_start, i_end + 1)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]
        # momentum
        momentum = beta * momentum + (1-beta) * g
        # step size
        lr = lr0 / (np.sqrt(it) + 1.0)
        a_new = a - lr * momentum
        # projection onto simplex
        a = project_simplex(a_new)
        # symmetrize to enforce left-right symmetry
        a = (a + a[::-1]) * 0.5
        # jitter occasional
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            a += sigma * np.random.randn(n)
            a = project_simplex(a)
            a = (a + a[::-1]) * 0.5
    return best_seq, best_val

"""
propose_candidate
-----------------

Search for a non‑negative step‑function with a low value of

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm proceeds in two stages:

1. **Coarse parametric search** – a symmetric two‑sided power‑law
   a_i ∝ (i+1)^(-p) + (n-i)^(-p) is scanned over the exponent `p`
   (grid step 0.02).  The best exponent gives a good starting shape.

2. **Projected gradient descent with momentum** – the sequence is
   optimised directly on the simplex.  In each iteration we
   * compute the convolution via an FFT,
   * locate the active set where the convolution is essentially maximal,
   * build the sub‑gradient `g_i = 2 * Σ_{k∈active} a_{k-i}`,
   * update with a momentum term (β≈0.9) and a diminishing step size,
   * project back onto the probability simplex and enforce left–right
     symmetry.  Small Gaussian jitter is added occasionally to escape
     plateaus.

The whole process respects the 1000 s time budget, uses only NumPy,
and finally returns the best sequence found as a plain Python list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of   2 * n * max(conv(seq, seq)) / (sum(seq)**2)
    using an FFT.  The sequence must be non‑negative and its sum > 0.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators
# ----------------------------------------------------------------------


def generate_two_sided_power_law(n: int, p: float) -> np.ndarray:
    """
    a_i ∝ (i+1)^(-p) + (n-i)^(-p) .
    The result is symmetric and non‑negative.
    """
    i = np.arange(n, dtype=np.float64)
    left = (i + 1.0) ** (-p)
    right = (n - i) ** (-p)
    return left + right


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex
# ----------------------------------------------------------------------


def projected_gradient_descent(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Minimise   2·n·max(conv(a,a))   over the probability simplex.
    Uses sub‑gradient of the max‑convolution, momentum and a
    Euclidean simplex projection after each step.
    Returns the best sequence encountered and its fast_score.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_seq = a.copy()
    best_val = fast_score(best_seq)

    # momentum buffer
    m = np.zeros_like(a)

    it = 0
    while True:
        now = time.time()
        if now > start_time + time_limit:
            break
        it += 1

        # ----- convolution (FFT) -----
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- active set (indices where conv is (almost) maximal) -----
        eps_act = 1e-12
        active = np.where(conv >= M - eps_act)[0]

        # ----- sub‑gradient of the max -----
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # ----- momentum update and step -----
        m = beta * m + (1.0 - beta) * g
        lr = lr0 / (np.sqrt(it) + 1.0)
        a_new = a - lr * m

        # ----- projection onto the simplex -----
        a = project_simplex(a_new)

        # ----- enforce left–right symmetry (does not change the sum) -----
        a = 0.5 * (a + a[::-1])

        # ----- occasional jitter to avoid stagnation -----
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            a += sigma * np.random.randn(n)
            a = project_simplex(a)
            a = 0.5 * (a + a[::-1])

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a low
    ``evaluate_sequence`` value.  The routine respects the 1000 s time
    budget and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 0) Choose a working length (larger lengths tend to lower the
    #    asymptotic constant).  15000 is a good trade‑off for the budget.
    # ------------------------------------------------------------------
    length = 15000

    # ------------------------------------------------------------------
    # 1) Coarse grid search over the exponent p of the symmetric
    #    two‑sided power‑law shape.
    # ------------------------------------------------------------------
    p_grid = np.arange(0.30, 1.60, 0.02)   # modest range around the known 0.55
    best_p = None
    best_score_grid = float("inf")
    for p in p_grid:
        seq = generate_two_sided_power_law(length, p)
        seq = np.maximum(seq, 0.0)
        seq /= seq.sum()
        sc = fast_score(seq)
        if sc < best_score_grid:
            best_score_grid = sc
            best_p = p
    print(f"[grid] best p = {best_p:.4f} → fast_score = {best_score_grid:.6f}")

    # ------------------------------------------------------------------
    # 2) Initialise from the best power‑law shape and optimise with PGD.
    # ------------------------------------------------------------------
    a0 = generate_two_sided_power_law(length, best_p)
    a0 = np.maximum(a0, 0.0)
    a0 /= a0.sum()

    remaining = deadline - time.time()
    if remaining <= 0:
        final_seq = a0
        print("[info] No time left for optimisation – returning grid best.")
    else:
        print(f"[optim] start PGD on length={length} for {remaining:.1f}s")
        seq_opt, val_opt = projected_gradient_descent(a0, remaining, time.time())
        final_seq = seq_opt
        print(f"[optim] PGD finished – best fast_score = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Post‑process: ensure feasibility and evaluate officially.
    # ------------------------------------------------------------------
    final_seq = np.maximum(final_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.full_like(final_seq, 1.0 / final_seq.size)
    else:
        final_seq = final_seq / total

    final_list = final_seq.tolist()

    # final official evaluation (may raise if the environment does not
    # provide ``evaluate_sequence`` – we protect against that).
    try:
        score = evaluate_sequence(final_list)  # type: ignore[name-defined]
        print(f"[final] evaluate_sequence reports {score:.6f}")
    except Exception as exc:  # pragma: no cover
        print("[final] evaluate_sequence failed:", exc)

    return final_list
