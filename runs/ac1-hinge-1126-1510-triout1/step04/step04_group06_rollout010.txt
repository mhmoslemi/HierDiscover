"""...

import time
import math
import numpy as np
from scipy.optimize import minimize

# Helper functions
def next_pow2(x):
    return 1 << ((x - 1).bit_length())

def softmax(y):
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)

def full_conv(a, fft_len):
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]

def fast_score(a, fft_len, n):
    conv = full_conv(a, fft_len)
    return float(2.0 * n * np.max(conv))

def simplex_projection(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
    a = softmax(y)
    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-14
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = np.sum(exp_shift)
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient w.r.t. conv entries
    w = exp_shift / Z            # shape 2n-1

    # compute gradient w.r.t. a: 2 * a * w convolved (as derived)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y

def subgradient_momentum_optimize(a0, deadline, fft_len_conv, fft_len_g, n):
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 1.0 / np.sqrt(it + 1.0)
        a = simplex_projection(a - step * v)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def hill_climb(a, current_score, fft_len, n, deadline, max_iter=2000):
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.04
        if delta <= 0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def simulated_annealing(a_start, score_start, fft_len, n, deadline, rng):
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0
    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        max_delta = cur_a[i] * 0.25
        delta = rng.random() * max_delta
        if delta <= 0:
            continue
        cur_a[i] -= delta
        cur_a[j] += delta
        new_score = fast_score(cur_a, fft_len, n)
        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue
        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()
        T *= decay
        if it % 4000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1
    return best_a, best_score

def build_initial(mode, n, rng):
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    elif mode == 'sin':
        x = np.arange(n)
        a = np.sin(np.pi * (x + 0.5) / n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a

def adam_optimize(a0, n, fft_len_conv, fft_len_g, deadline, rng):
    """Projected Adam on the smoothed max using softmax variables."""
    y = np.log(a0 + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)
    lam_start = 0.3
    lam_end = 0.001
    while time.time() < deadline:
        t += 1
        # linear schedule for smoothing factor
        lam_factor = max(lam_end, lam_start * (1.0 - t / 20000.0))
        f, grad_y = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.5 / (np.sqrt(v_hat) + eps)   # base learning rate
        y -= step * m_hat

        # periodically evaluate true score
        if t % 200 == 0:
            a_cur = softmax(y)
            cur_score = fast_score(a_cur, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_cur.copy()
    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising `evaluate_sequence`.

    The algorithm combines:
        • Multiple restarts with Adam optimisation on a smoothed max objective.
        • A long momentum sub‑gradient descent on the true max.
        • Simulated annealing for non‑local moves.
        • Final greedy hill‑climb polishing.

    It adapts the problem size (n) up to 4000 and respects the time budget.
    """
    rng = np.random.default_rng(seed + 987654)

    # Determine problem size.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        # aim for a larger size but keep it manageable
        target_n = min(4000, max(2500, prev.size * 2))
        if prev.size < target_n:
            # pad with zeros and renormalise (zeros do not affect shape)
            extra = target_n - prev.size
            prev = np.concatenate([prev, np.zeros(extra)])
        a0 = prev / prev.sum() if prev.sum() > 0 else np.full(target_n, 1.0 / target_n)
        n = target_n
    else:
        n = 3000
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    start = time.time()
    deadline = start + budget_s - 0.5

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    # Phase 1: Adam optimisation with several random starts
    phase1_end = start + 0.4 * budget_s

"""
Search for a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The algorithm proceeds in four phases:

1.  Multiple random restarts of Adam optimisation on a smooth
    log‑sum‑exp approximation of the max‑convolution.  A linear
    continuation schedule for the smoothing parameter is used.
2.  A long momentum sub‑gradient descent on the exact max‑convolution.
3.  Simulated annealing for non‑local exploration.
4.  A final greedy hill‑climb polish.

The problem size n is chosen up to 4000 (larger n gives more
degrees of freedom) while respecting the overall time budget.
All updates respect the probability simplex via Euclidean projection.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a already normalised (sum=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by Adam / L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed‑max objective based on log‑sum‑exp and its gradient w.r.t. y.
    y is the unconstrained vector; a = softmax(y) lies on the simplex.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-14
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. a -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (with momentum)
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Directly minimise the true max‑convolution using a momentum
    sub‑gradient method with Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9                    # momentum coefficient
    it = 0

    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        v = beta * v + (1.0 - beta) * g
        step = 1.0 / np.sqrt(it + 1.0)
        a = simplex_projection(a - step * v)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Local optimisation that repeatedly moves a tiny fraction of mass
    between two positions if it improves the (fast) score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that relocates a random fraction of mass.
    Improves exploration after the gradient phases.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.25
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 4000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Initial distribution builders
# ----------------------------------------------------------------------


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n according to the requested mode.
    """
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "edge":
        frac = rng.uniform(0.004, 0.02)
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == "random":
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    elif mode == "sin":
        x = np.arange(n)
        a = np.sin(np.pi * (x + 0.5) / n)
        a = np.maximum(a, 0)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam (via soft‑max variables) on the log‑sum‑exp smooth max.
    A linear continuation schedule for the smoothing factor is employed.
    """
    y = np.log(a0 + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    lam_start = 0.3
    lam_end = 0.001

    while time.time() < deadline:
        t += 1
        lam_factor = max(lam_end, lam_start * (1.0 - t / 20000.0))

        f, grad_y = objective_and_grad(
            y, n, fft_len_conv, fft_len_g, lam_factor
        )

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.5 / (np.sqrt(v_hat) + eps)      # base learning rate
        y -= step * m_hat

        if t % 200 == 0:
            a_cur = softmax(y)
            cur_score = fast_score(a_cur, fft_len_conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_cur.copy()

    return best_a, best_score
