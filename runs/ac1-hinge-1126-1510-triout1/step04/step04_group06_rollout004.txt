"""
propose_candidate – multi‑phase, symmetry‑aware search for a low‑max‑convolution
step function.

The score (ignoring the trivial scaling) is 2·n·max(conv(a,a)) where
a is a non‑negative vector with sum = 1.  The algorithm proceeds as follows:

1.  Initialise a from the previous best solution (if any).  If the previous
    length is < 3000 we up‑sample it by linear interpolation to obtain a
    richer search space (max length capped at 4000).

2.  Repeated stochastic L‑BFGS runs on a smooth log‑sum‑exp surrogate of the
    max‑convolution.  Different random initialisations (uniform, edge‑heavy,
    exponential, power‑law, Dirichlet, asymmetric, two‑peak) and a schedule
    of smoothing factors are used.  After each run a short projected hill‑climb
    is performed.

3.  A long momentum sub‑gradient descent on the true max‑convolution
    (with adaptive step size 1/√t) further reduces the peak.

4.  A brief final L‑BFGS optimisation with a very small smoothing factor
    tightens the solution close to the true objective.

5.  Simulated annealing with large moves explores non‑local regions.

6.  A generous greedy hill‑climb (thousands of swaps) polishes the final
    vector.

All steps maintain the simplex constraint by Euclidean projection and
symmetrise the vector after each update (a_i = a_{n‑1‑i}) because the optimum
is expected to be (approximately) symmetric.  The best vector found is
returned as a plain Python list of floats and stored globally as
`height_sequence_1` for future calls.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a already normalised (sum=1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Returns f (2·n·smooth_max(conv)) and grad_y (gradient w.r.t. y).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c    # for numeric stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # Chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (with momentum)
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """
    Directly minimise the true max‑convolution using a momentum
    sub‑gradient method with Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9                    # momentum coefficient
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active set mask (indices attaining the max, with tiny tolerance)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (see derivation in the original code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ~ 1/√t
        step = 1.0 / np.sqrt(it + 1.0)
        a_new = a - step * v
        a = simplex_projection(a_new)

        # enforce symmetry (optional – helps stability)
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

        # keep best
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Local optimisation that repeatedly moves a tiny fraction of mass
    between two positions if it improves the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simulated annealing that relocates a random fraction of mass.
    Improves exploration after the gradient phases.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # Initial temperature as a modest fraction of the current score
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9993
    it = 0

    while time.time() < deadline:
        # pick a source index with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        # distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction (up to 25 %) of the source mass
        max_delta = cur_a[i] * 0.25
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        # Metropolis acceptance
        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert
                cur_a[i] += delta
                cur_a[j] -= delta
                # temperature update and continue
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        # occasional reheating
        if it % 4000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to build various initial distributions
# ----------------------------------------------------------------------


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n according to the requested mode.
    Modes:
        'uniform' – plain uniform.
        'edge'    – heavier mass near the two ends.
        'exp'     – exponential decay from left (mirrored to right).
        'power'   – power‑law decay.
        'random'  – Dirichlet(0.5) random.
    """
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a

        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 500,
                                'gtol': 1e-10,
                                'disp': False})
        a_candidate = softmax(res.x)
        # Enforce symmetry (helps reduce the peak)
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        a_candidate /= a_candidate.sum()

        cur_score = fast_score(a_candidate, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            print(f"[L‑BFGS {lbfgs_iter}] mode={mode} lam={lam:.4f} "
                  f"score={best_score:.6f}")

        # A short hill‑climb after each L‑BFGS run
        hill_deadline = min(phase1_end, time.time() + 0.12)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hill_deadline,
                                        max_iter=2000)
        lbfgs_iter += 1

    # --------------------------------------------------------------
    # Phase 2 – long momentum sub‑gradient descent on the true max.
    # --------------------------------------------------------------
    phase2_end = start_time + 0.60 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_momentum_optimize(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final L‑BFGS with a very small smoothing factor.
    # --------------------------------------------------------------
    phase3_end = start_time + 0.70 * budget_s
    if time.time() < phase3_end:
        y0 = np.log(best_a + 1e-12)
        tiny_lam = 0.0005  # effectively close to the true max
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, tiny_lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 400,
                                'gtol': 1e-12,
                                'disp': False})
        a_candidate = softmax(res.x)
        a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
        a_candidate /= a_candidate.sum()
        cur_score = fast_score(a_candidate, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_candidate.copy()
            print(f"[final L‑BFGS] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing for non‑local exploration.
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 deadline,
                                                 rng)
        print(f"[SA] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – exhaustive greedy hill‑climb polish.
    # --------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=10000)

    # --------------------------------------------------------------
    # Final evaluation (optional) and return.
    # --------------------------------------------------------------
    final_seq = [float(max(0.0, x)) for x in final_a]
    try:
        eval_val = evaluate_sequence(final_seq)
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # Store globally for possible reuse in later calls
    globals()['height_sequence_1'] = final_seq

    return final_seq
