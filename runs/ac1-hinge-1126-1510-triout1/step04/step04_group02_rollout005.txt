def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine n
    n = 5000  # default
    init_seq = globals().get('height_sequence_1', None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        seq_arr = np.asarray(init_seq, dtype=float)
        if seq_arr.ndim == 1 and seq_arr.size >= 2000:
            n = min(int(seq_arr.size), 8000)  # not too large
    n = max(2000, n)  # ensure at least 2k

    # Compute FFT len
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # Initialise a
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            # Resize to length n via interpolation or random?
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------
    # L-BFGS smoothing phase
    # --------------------
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=30))
    # run a few restarts
    while time.time() < deadline - 150 and random condition? maybe loop limited:
        mode = rng.choice(['random','exp','power','edge','sidon'])
        y0 = generate_initial_y(mode, n, rng) if mode!='best' else best_y + rng.normal(scale=0.03,size=n)
        y_opt = schedule_optimize(y0, n, fft_len_conv, fft_len_g, lam_schedule, maxiter_per_stage=300)
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(f"[L-BFGS] mode={mode} improved score {best_score:.6f}")

        # quick hill climb
        best_a, best_score = hill_climb(best_a, best_score, fft_len_conv, n,
                                        min(deadline, time.time()+0.6), max_iter=3000)
        best_y = np.log(best_a + 1e-12)

    # --------------------
    # LP refinement phase
    # --------------------
    if time.time() < deadline - 30:
        # compute current M
        conv_cur = conv_fft(best_a, fft_len_conv, n)
        M = conv_cur.max()

            g = b_val / sum_b

                conv_cur = conv_fft(best_a, fft_len_conv, n)

"""
propose_candidate
------------------

Search for a non‑negative step‑function (a probability vector) that
minimises  

    2 * n * max(conv(a, a))

which is exactly the value returned by ``evaluate_sequence`` when the
vector is normalised to sum = 1.  

The algorithm consists of three stages:

1. **Smooth L‑BFGS** – minimise a log‑sum‑exp smooth approximation of the
   max‑convolution.  Several random restarts and a geometric schedule of
   smoothing parameters are used.

2. **LP refinement** – given the current vector `a` and its peak
   `M = max(conv(a, a))`, solve the linear programme  

        maximise   Σ b_i
        subject to  conv(a, b) ≤ M ,   b_i ≥ 0

   (the convolution is linear in `b`).  The optimiser (cvxpy/ECOS) returns a
   direction `b`.  After renormalising `b` to sum = 1 we take a convex
   combination with `a`.  This step repeatedly reduces the peak value.

3. **Aggressive hill‑climbing** – local greedy moves that transfer a
   fraction of mass between two coordinates while strictly decreasing the
   fast score.

The whole search is bounded by the provided time budget (≈ 1000 s).  The
best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax that returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2*n‑1) using FFT.
    a is assumed length n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector
    (sum = 1): 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                      # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z   # ∂ smooth_max / ∂ conv_i

    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a series of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j
    (keeping the sum = 1).  Keep a move only if the fast_score strictly
    improves.  The transferred fraction may be up to 10 %.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def lp_step(a: np.ndarray, M: float, solver: str = "ECOS"):
    """
    Solve the LP:
        maximise Σ b_i
        subject to conv(a, b) ≤ M,  b ≥ 0
    Returns the optimal b (numpy array) or None on failure.
    """
    try:
        import cvxpy as cp
    except Exception:
        return None

    n = a.shape[0]

    # cvxpy parameter for the fixed vector a
    a_param = cp.Parameter(n, nonneg=True, value=a)

    b = cp.Variable(n, nonneg=True)
    conv_expr = cp.conv(a_param, b)          # length 2*n-1

    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=solver, warm_start=True, max_iters=2000, verbose=False)
    except Exception:
        return None

    if b.value is None:
        return None
    return np.asarray(b.value, dtype=float)

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Choose problem size n (moderately large but still tractable)
    # --------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 8000)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Initialise best candidate on the probability simplex
    # --------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = tmp / np.sum(tmp)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – smooth L‑BFGS optimisation (majority of time budget)
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=30))
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    while time.time() < deadline - 200.0:   # keep ~3 min for later stages
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=300,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] L‑BFGS mode={mode} improved fast score = {best_score:.6f}")

        # short aggressive hill‑climb after each L‑BFGS run
        hc_dead = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_dead,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Stage 2 – LP refinement (tightening the max‑convolution bound)
    # --------------------------------------------------------------
    if time.time() < deadline - 50.0:
        # current peak of the self‑convolution
        conv_cur = conv_fft(best_a, fft_len_conv, n)
        M = conv_cur.max()
        print(f"[LP] starting refinement, M={M:.8e}, score={best_score:.6f}")

        lp_iters = 0
        while time.time() < deadline - 10.0 and lp_iters < 8:
            lp_iters += 1
            b_val = lp_step(best_a, M, solver="ECOS")
            if b_val is None:
                print("[LP] LP solve failed – exiting LP phase")
                break

            sum_b = b_val.sum()
            if sum_b <= 0:
                print("[LP] degenerate LP solution – exiting")
                break

            g = b_val / sum_b                 # renormalise to sum = 1
            # convex combination (t could be tuned; we start with 0.5)
            t = 0.5
            a_new = (1.0 - t) * best_a + t * g
            a_new = a_new / a_new.sum()
            new_score = fast_score(a_new, fft_len_conv, n)

            if new_score < best_score:
                best_a = a_new
                best_score = new_score
                conv_cur = conv_fft(best_a, fft_len_conv, n)
                M = conv_cur.max()
                print(f"[LP] iter={lp_iters}, improved score={best_score:.6f}, M={M:.8e}")

                # brief hill‑climb to polish the LP update
                hc_dead = min(deadline, time.time() + 0.6)
                best_a, best_score = hill_climb(
                    best_a,
                    best_score,
                    fft_len_conv,
                    n,
                    hc_dead,
                    max_iter=3000,
                )
                best_y = np.log(best_a + 1e-12)
                # recompute M after polishing
                conv_cur = conv_fft(best_a, fft_len_conv, n)
                M = conv_cur.max()
            else:
                # no improvement – reduce the mixing factor and try again
                t *= 0.5
                if t < 0.02:
                    print("[LP] mixing factor exhausted, leaving LP loop")
                    break

        print("[LP] refinement stage finished")

    # --------------------------------------------------------------
    # Stage 3 – final aggressive greedy polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=12000,
        )

    # Final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
