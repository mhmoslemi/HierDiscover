"""
Propose a low‑score step‑function sequence.

The algorithm combines three complementary ideas:

1. **Smart initialisation** – a symmetric “U‑shaped’’ distribution
   (mass concentrated near the ends, low in the centre) that is already
   better than a uniform step.

2. **Simulated‑annealing + projected gradient descent** – a fast Monte‑Carlo
   search that repeatedly transfers a random fraction of the mass between
   two positions and accepts worsening moves with a temperature schedule.
   Interleaved with this, a projected gradient step on a smooth approximation
   of the max‑convolution (using FFT‑based gradients) continuously flattens
   the autocorrelation.  Symmetry is enforced after every update by
   averaging the vector with its reverse, and the simplex projection guarantees
   non‑negativity and unit ℓ₁‑norm.

3. **Polishing** – a short L‑BFGS continuation on the log‑soft‑max
   parametrisation and a final greedy hill‑climb finish the optimisation.

The whole search respects the 1000 s budget, stops a few seconds early,
and returns the best sequence found.  All helper functions are top‑level
and avoid closures, as required by the evaluation harness.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (sum == 1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho_arr[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns a vector of length n (no 2·n scaling).  The smoothing parameter
    is ``lam = max(conv) * lam_factor + 1e-12``.
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z                    # d smooth_max / d conv_i

    # convolve a (reversed) with w
    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    return 2.0 * g_vec  # derivative of smooth max (without the outer 2·n)


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max coordinates ``y``.
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max w.r.t. conv ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max (gradient w.r.t. y)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        # enforce symmetry and simplex
        best = (best + best[::-1]) / 2.0
        best = proj_simplex(best)

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:   # strict improvement
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

        p = rng.uniform(1.5, 3.0)
        a = (dist + 1.0) ** p
        a = proj_simplex(a)

    # enforce symmetry immediately (helps the optimiser)
    a = (a + a[::-1]) / 2.0
    a = proj_simplex(a)

    # --------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # --------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    cur_a = a.copy()
    cur_score = fast_score(cur_a, fft_len, n)
    best_a = cur_a.copy()
    best_score = cur_score
    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Simulated annealing + projected gradient descent (mixed mode)
    # --------------------------------------------------------------
    T = 1.0                 # temperature for SA
    T_min = 1e-8
    cooling = 0.9995
    max_move_frac = 0.30
    lam_factor = 0.30       # smoothing for gradient steps
    lam_decay = 0.999        # slowly tighten the smooth max
    it = 0
    while time.time() < deadline:
        it += 1
        # ---- decide which move to perform ----
        if rng.random() < 0.70:                     # mass‑transfer (SA) move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = cur_a[i] * rng.random() * max_move_frac
            if delta <= 0.0:
                continue
            cand = cur_a.copy()
            cand[i] -= delta
            cand[j] += delta

            # enforce symmetry and simplex
            cand = (cand + cand[::-1]) / 2.0
            cand = proj_simplex(cand)

            cand_score = fast_score(cand, fft_len, n)
            dE = cand_score - cur_score
            accept = False
            if dE < 0:
                accept = True
            else:
                # Metropolis criterion
                accept = rng.random() < math.exp(-dE / max(T, 1e-12))

            if accept:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_a = cand.copy()
                    best_score = cand_score
                    print(f"[{it}] SA improved   fast score = {best_score:.6f}")
            # cool down
            T = max(T * cooling, T_min)

        else:                                        # projected gradient step
            grad = grad_smooth_max(cur_a, lam_factor)
            # simple step‑size schedule
            step = 0.4 / (math.sqrt(it) + 1.0)
            cand = proj_simplex(cur_a - step * grad)
            # enforce symmetry
            cand = (cand + cand[::-1]) / 2.0
            cand = proj_simplex(cand)

            cand_score = fast_score(cand, fft_len, n)
            if cand_score < cur_score:
                cur_a = cand
                cur_score = cand_score
                if cand_score < best_score:
                    best_a = cand.copy()
                    best_score = cand_score
                    print(f"[{it}] grad step improved fast score = {best_score:.6f}")

            # tighten the smoothing gradually
            lam_factor = max(lam_factor * lam_decay, 1e-7)

        # occasional random perturbation to escape plateaus
        if it % 2000 == 0:
            pert = best_a + rng.normal(scale=0.02, size=n)
            pert = np.clip(pert, 0.0, None)
            pert = proj_simplex(pert)
            pert = (pert + pert[::-1]) / 2.0
            pert = proj_simplex(pert)
            pert_score = fast_score(pert, fft_len, n)
            if pert_score < best_score:
                best_a = pert.copy()
                best_score = pert_score
                cur_a = pert.copy()
                cur_score = pert_score
                print(f"[{it}] random perturb improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Short L‑BFGS refinement on the log‑soft‑max parametrisation
    # --------------------------------------------------------------
    if time.time() < deadline - 3.0:
        y0 = np.log(best_a + 1e-12)
        lam_factors = np.geomspace(0.3, 1e-5, num=20)
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors.tolist(),
            maxiter_per_stage=200,
        )
        a_opt = softmax(y_opt)
        a_opt = (a_opt + a_opt[::-1]) / 2.0
        a_opt = proj_simplex(a_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_a = a_opt.copy()
            best_score = score_opt
            print(f"[lbfgs] refined fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final greedy hill‑climb (use any remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len,
            n=n,
            deadline=deadline,
            max_iter=8000,
            max_move_frac=0.25,
        )
        print(f"[final hill] fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
