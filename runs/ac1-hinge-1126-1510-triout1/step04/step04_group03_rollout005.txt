"""
search_step_function.py

Multi‑stage optimizer for the “max‑convolution” metric used by
evaluate_sequence().  The algorithm proceeds as follows

1.  Generate a diverse set of seed distributions (uniform, linear,
    power‑law, exponential, Beta‑shaped, sinusoidal, random Dirichlet,
    and the previous best if available) and keep the best few.

2.  For each seed run a continuation L‑BFGS on a smooth approximation of
    max( a * a ) (log‑sum‑exp) with a decreasing smoothing factor.
    The variables are the unconstrained soft‑max parameters; after each
    λ‑stage a short greedy hill‑climb is performed.

3.  Refine the current best solution with a projected sub‑gradient method
    that uses the exact max‑convolution (the sub‑gradient of the max is
    a linear combination of the active constraints).

4.  Escape possible shallow minima with simulated‑annealing on the
    simplex, accepting worsening moves with a slowly cooling temperature.

5.  End with a longer greedy hill‑climb that transfers small random
    fractions of mass between coordinates while preserving the simplex.

All steps respect the probability simplex (non‑negative, sum = 1) and
use FFT‑based convolution (O(n log n)).  The total wall‑clock budget is
split automatically; the best sequence found before the deadline is
returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a distribution already
    normalised to sum = 1 (the true score is 2·n·max(a*a) / sum(a)²,
    but the ratio is scale‑invariant, so we only need max(a*a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a (full length 2·n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max via log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c          # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. the convolution entries
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the part that corresponds to a_i's gradient (length n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain‑rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing λ schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found
    (still on the simplex) together with its fast_score.
    """
    y = np.log(a_start + 1e-12)          # initialise soft‑max arguments
    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:  # pragma: no cover – very unlikely
            # Occasionally L‑BFGS may fail – ignore and continue.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()

        # a short hill‑climb after each λ can sharpen the result
        hc_deadline = min(deadline, time.time() + 0.35)
        best_a, best_sc = hill_climb(best_a, best_sc,
                                     fft_len_conv, n,
                                     hc_deadline,
                                     max_iter=1200)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation with simple back‑tracking line search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line‑search and respects the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and the active (max) set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_sc = fast_score(a_cand, fft_len_conv, n)
            if cand_sc < best_sc:
                best_sc, best_a = cand_sc, a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               cur_sc: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_sc = cur_sc
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.06   # up to 6 % of a[i]
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_sc = fast_score(best, fft_len, n)
        if new_sc < best_sc:
            best_sc = new_sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_sc


# ----------------------------------------------------------------------
# Simulated‑annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  cur_sc: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_sc = cur_sc
    best_a = a.copy()
    best_sc = cur_sc
    rng = np.random.default_rng()
    T = 0.025
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.12   # up to 12 %
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_sc = fast_score(cand, fft_len, n)
        if new_sc < cur_sc:
            cur_a, cur_sc = cand, new_sc
            if new_sc < best_sc:
                best_a, best_sc = cand.copy(), new_sc
        else:
            prob = np.exp(-(new_sc - cur_sc) / T)
            if rng.random() < prob:
                cur_a, cur_sc = cand, new_sc

        it += 1
        if it % 6000 == 0:
            T = max(T * 0.90, 1e-5)

    return best_a, best_sc


# ----------------------------------------------------------------------
# Candidate generation (parametric families)
# ----------------------------------------------------------------------
def generate_candidates(n: int, rng: np.random.Generator):
    """
    Produce a list of (score, array) tuples for a variety of parametric
    families.  The arrays are already normalised to sum = 1.
    """
    candidates = []

    def add(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, next_pow2(2 * n - 1), n)
        candidates.append((sc, arr))

    # 1) Uniform (baseline)
    add(np.full(n, 1.0 / n))

    # 2) Linear ramps
    add(np.arange(n, 0, -1, dtype=float))          # decreasing
    add(np.arange(1, n + 1, dtype=float))          # increasing

    # 3) Quadratic ramps
    add((np.arange(1, n + 1, dtype=float) ** 2))
    add(((np.arange(n, 0, -1, dtype=float) ** 2)))

    # 4) Cubic ramps
    add((np.arange(1, n + 1, dtype=float) ** 3))
    add(((np.arange(n, 0, -1, dtype=float) ** 3)))

    # 5) Exponential families
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01, 0.02]:
        add(np.exp(-rate * np.arange(n)))

    # 6) Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5, 3.0]:
        add(np.arange(1, n + 1, dtype=float) ** (-p))

    # 7) Edge‑heavy (mass at both ends)
    for edge_frac in [0.003, 0.006, 0.012, 0.024]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add(arr)

    # 8) Sinusoidal shapes (always non‑negative)
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add(sin_arr + 1e-4)   # tiny offset for strict positivity

    cos_arr = np.cos(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add(cos_arr + 1e-4)

    # 9) Beta‑shaped families (discrete approximation of Beta pdf)
    xs = (np.arange(n) + 0.5) / n
    for alpha in [0.4, 0.6, 0.8, 1.2, 1.5, 2.0]:
        for beta in [0.4, 0.6, 0.8, 1.2, 1.5, 2.0]:
            arr = (xs ** (alpha - 1)) * ((1 - xs) ** (beta - 1))
            add(arr)

    # 10) Random Dirichlet / Gamma draws
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
        for _ in range(3):
            add(rng.gamma(alpha, 1.0, size=n))

    # 11) Previous best (if present)
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            # truncate/pad to the chosen length
            if prev.size >= n:
                add(prev[:n])
            else:
                pad = np.zeros(n, dtype=float)
                pad[:prev.size] = prev
                add(pad)

    # keep only the top 15 seeds (lowest fast_score)
    candidates.sort(key=lambda t: t[0])
    return [arr for _sc, arr in candidates[:15]]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a list of floats (the sequence) that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    max_n = 4000  # keep within a few thousand entries
    n = max_n

    # initialise with uniform distribution if nothing else is available
    a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g    = next_pow2(3 * n - 2)   # for gradient / mask convolutions

    best_a   = a0.copy()
    best_sc  = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – parametric sweep
    # ------------------------------------------------------------------
    top_seeds = generate_candidates(n, rng)
    print(f"[sweep] generated {len(top_seeds)} promising seeds (best fast_score≈{best_sc:.6f})")

    # ------------------------------------------------------------------
    # Timing budget
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline   = start_time + budget_s - 0.4   # safety margin
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (many restarts)
    # ------------------------------------------------------------------
    lam_schedule = [0.5, 0.4, 0.3, 0.25, 0.2, 0.15,
                    0.1, 0.07, 0.05, 0.03, 0.02,
                    0.01, 0.008, 0.005, 0.003, 0.001]
    maxiter_per_lam = 400
    restarts = 0

    while time.time() < phase1_end and restarts < 120:
        # sample a seed and add a tiny jitter for diversity
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len_conv,
                                           fft_len_g,
                                           n)
        if sc_opt < best_sc:
            best_sc, best_a = sc_opt, a_opt.copy()
            print(f"[L‑BFGS restart {restarts}] new best fast_score={best_sc:.6f}")

        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.3)
        best_a, best_sc = subgradient_optimize(best_a,
                                               sg_deadline,
                                               fft_len_conv,
                                               fft_len_g,
                                               n)
        print(f"[sub‑gradient] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing / random perturbations
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_sc = random_anneal(best_a,
                                       best_sc,
                                       fft_len_conv,
                                       n,
                                       phase3_end)
        print(f"[anneal] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_sc = hill_climb(best_a,
                                 best_sc,
                                 fft_len_conv,
                                 n,
                                 final_deadline,
                                 max_iter=12000)
    print(f"[final hill‑climb] best fast_score={best_sc:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence (plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # The external evaluator will clamp, scale, etc., but our distribution
    # already respects the constraints (non‑negative, sum≈1).
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
