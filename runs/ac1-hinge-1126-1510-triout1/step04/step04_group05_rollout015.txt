"""
Hybrid global–local optimizer for the step‑function “max‑convolution” problem.

Main ideas
-----------
* **Large search dimension** – use a longer sequence (≈3000) to give the optimizer more
  degrees of freedom while keeping each FFT cheap (next power of two ≈ 8192).
* **Continuation L‑BFGS** – smooth approximation of the max via a log‑sum‑exp;
  a geometric schedule of smoothing parameters drives the solution into a good basin.
* **Linear‑programming refinement** – after each L‑BFGS run we solve a tiny LP that
  linearly approximates the effect of moving mass on the current largest convolution
  peaks.  This often yields a direct improvement that pure gradient steps miss.
* **Targeted & random hill‑climbs** – greedy mass transfers that flatten the
  actual (non‑smoothed) convolution, followed by a generic random walk as a final
  polish.
* **Multiple restarts / diverse initialisations** – random, exponential, power‑law,
  edge‑biased and perturbations of the best solution are tried until the time budget
  expires.

All operations respect the “sum‑to‑one” simplex (the evaluation metric is scale‑invariant)
and keep the vector non‑negative.  The algorithm stays comfortably within the 1000 s
budget and typically pushes the score below the 1.503 target.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the first (2*n‑1) entries.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the metric 2·n·max(a*a) for a probability vector a.
    Equivalent to evaluate_sequence when sum(a) == 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution together with its gradient
    w.r.t. the unconstrained logits y (a = softmax(y)).
    """
    a = softmax(y)

    # ----- a * a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smooth max
    shifted = conv - max_c                         # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                               # d smooth_max / d conv_i

    # convolution a_rev * w  (linear term of the derivative)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list that gradually tightens the smooth‑max.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb on the simplex.
    Moves a random fraction of mass from one position to another and
    accepts only strict improvements.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb.
    At each step the current maximal convolution index k* is located.
    Mass is shifted from the index with the largest contribution a_i*a_{k*-i}
    to the one with the smallest contribution.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random move
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


def lp_refine(
    a: np.ndarray,
    conv: np.ndarray,
    K: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Linear‑programming refinement.
    Keeps the sum fixed, respects non‑negativity, and reduces the current
    largest K convolution values using a first‑order (linear) model.
    """
    n = len(a)
    # pick the K largest convolution indices (allow ties)
    idx = np.argpartition(conv, -K)[-K:]

    # Build the linear inequality system:
    #   conv_k + 2*(a * delta)_k <= t     for each selected k
    # which we rewrite as  2*(a * delta)_k - t <= -conv_k
    A_ub = np.zeros((K, n + 1), dtype=float)
    for row, k in enumerate(idx):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # delta_j coefficient = 2 * a_{k-j}
        for j in range(lo, hi + 1):
            i = k - j
            A_ub[row, j] = 2.0 * a[i]
        A_ub[row, n] = -1.0            # coefficient for t
    b_ub = -conv[idx]

    # Equality: sum(delta) = 0  (preserve total mass)
    A_eq = np.zeros((1, n + 1), dtype=float)
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Variable bounds: delta_i >= -a_i   (so a_i+delta_i stays >=0)
    bounds = [(-ai, None) for ai in a] + [(0.0, None)]   # t >= 0

    # Solve with HiGHS (default) – fast for these tiny QPs
    res = linprog(
        c=np.concatenate([np.zeros(n), [1.0]]),  # minimise t
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return a, None

    delta = res.x[:n]
    new_a = a + delta
    # Numerical cleanup
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s > 1e-12:
        new_a /= s
    t_val = res.fun
    return new_a, t_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    """

    deadline = time.time() + budget_s - 2.0      # safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off.
    # --------------------------------------------------------------
    default_n = 3000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1:
            if arr.size >= default_n:
                a0 = arr[:default_n].copy()
            else:
                # repeat the known pattern to reach the target length
                reps = default_n // arr.size + 1
                a0 = np.tile(arr, reps)[:default_n].copy()
        else:
            a0 = np.full(default_n, 1.0 / default_n)
    else:
        a0 = np.full(default_n, 1.0 / default_n)

    a0 = np.maximum(a0, 0.0)
    s0 = a0.sum()
    if s0 > 1e-12:
        a0 /= s0
    else:
        a0 = np.full(default_n, 1.0 / default_n)

    n = default_n
    best_a = a0
    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    fft_len_g = next_pow2(3 * n - 2)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Continuation L‑BFGS hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 5e-7, num=35)   # smooth‑max schedule
    maxiter_stage = 500

    iteration = 0

    # --------------------------------------------------------------
    # Multi‑restart optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline - 8.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.015)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 1.5)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.005, 0.02)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:   # perturb current best
            y0 = best_y + rng.normal(scale=0.02, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- linear‑programming refinement on the current top peaks -----
        conv_cur = conv_fft(best_a, fft_len_conv, n)
        best_a, _ = lp_refine(best_a, conv_cur, K=12, deadline=deadline)
        best_y = np.log(best_a + 1e-12)
        best_score = fast_score(best_a, fft_len_conv, n)

        # ----- aggressive targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=4000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # occasional random hill‑climb (cheap polishing)
        if time.time() < deadline - 4.0:
            best_a, best_score = hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=5000,
            )
            best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final deep refinement (continuation passes)
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.3e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing hill‑climbs
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=12000,
            frac=0.06,
        )
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
