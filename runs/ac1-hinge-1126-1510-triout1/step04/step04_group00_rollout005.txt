"""
Propose a low‑score step‑function sequence.

The search combines several complementary stages:

1. **Diverse global initialisation** – random, exponential,
   power‑law, Gaussian, triangular, edge‑heavy and Dirichlet
   patterns are generated and optimised with a continuation
   L‑BFGS on a smooth approximation of the max‑convolution.

2. **Projected sub‑gradient descent** – the exact sub‑gradient of the
   (non‑smooth) max‑convolution is cheap to compute (it is just a
   shifted copy of the current vector).  A diminishing‑step
   projected‑sub‑gradient method drives the peak of the auto‑convolution
   down further while staying on the probability simplex.

3. **Aggressive hill‑climb** – a fast greedy mass‑transfer routine
   finishes the optimisation.

All stages respect the time budget (≈1000 s).  The best sequence found
is stored in the global variable ``height_sequence_1`` for possible
re‑use by future runs.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (returns length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    n = a.shape[0]
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the scoring function for a probability vector."""
    conv = conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# L‑BFGS on the smooth‑max approximation (phase 1)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t.
    the unconstrained coordinates y (log‑soft‑max).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z
    # compute g = Σ_j a_j w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Aggressive hill‑climb (phase 3)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:   # strict improvement
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution (phase 2)
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, n: int, fft_len: int):
    """
    Compute a sub‑gradient of the exact max‑convolution at the current
    point a.  Returns (g, max_val, argmax_index).
    """
    conv = conv_fft(a, fft_len)
    max_val = conv.max()
    k = int(conv.argmax())                     # index of the active peak
    # Build the sub‑gradient g_i = 2 * a_{k-i} (zero when out of range)
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    # vector of i indices
    idx = np.arange(i_min, i_max + 1, dtype=int)
    j = k - idx                               # the partner indices
    g[idx] = 2.0 * a[j]
    return g, max_val, k


def subgradient_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iter: int = 20000,
    step0: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient method for minimising the max‑convolution.
    Returns the best vector seen and its fast_score.
    """
    a_curr = a0.copy()
    best_a = a_curr.copy()
    best_score = fast_score(best_a, fft_len, n)

    iter_cnt = 0
    while time.time() < deadline and iter_cnt < max_iter:
        iter_cnt += 1
        g, _, _ = subgradient_step(a_curr, n, fft_len)
        step = step0 / np.sqrt(iter_cnt)
        # additive step followed by simplex projection
        a_new = proj_simplex(a_curr - step * g)
        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()

        # accept only if we improve, otherwise keep the old point
        if new_score < fast_score(a_curr, fft_len, n) - 1e-14:
            a_curr = a_new
        else:
            # occasional random jitter to escape plateaus
            if np.random.rand() < 0.001:
                jitter = np.random.standard_normal(n) * 1e-4
                a_curr = proj_simplex(a_curr + jitter)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    The function respects the time budget and returns the best
    non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 5000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to a probability vector (sum == 1)
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Global search with L‑BFGS (phase 1)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-5, num=30)   # geometric λ‑schedule
    maxiter_stage = 800                           # more iterations per stage

    iteration = 0
    while time.time() < deadline - 5.0:
        # pick a diverse initialisation mode
        mode = rng.choice(["random", "exp", "power", "gaussian",
                           "tri", "edge", "dirichlet", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)

        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)

        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)

        elif mode == "gaussian":
            sigma = rng.uniform(0.05, 0.2) * n
            centre = (n - 1) / 2.0
            a0 = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
            a0 += 1e-12
            a0 /= a0.sum()
            y0 = np.log(a0)

        elif mode == "tri":
            centre = (n - 1) / 2.0
            a0 = 1.0 - np.abs(np.arange(n) - centre) / centre
            a0 = np.maximum(a0, 0.0)
            a0 += 1e-12
            a0 /= a0.sum()
            y0 = np.log(a0)

        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)

        elif mode == "dirichlet":
            # concentration < 1 yields a heavy‑tailed distribution
            alpha = rng.uniform(0.2, 0.8)
            a0 = rng.gamma(alpha, 1.0, size=n)
            a0 += 1e-12
            a0 /= a0.sum()
            y0 = np.log(a0)

        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ----- L‑BFGS continuation -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}"
            )

        # ----- short local hill‑climb to polish each L‑BFGS run -----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=4000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent (phase 2)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        subgrad_deadline = deadline - 2.0
        best_a, best_score = subgradient_descent(
            a0=best_a,
            n=n,
            fft_len=fft_len_conv,
            deadline=subgrad_deadline,
            max_iter=25000,
            step0=5.0,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] best fast score after sub‑gradient = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (use any remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=15000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # the provided black‑box evaluator
    print(f"[final] evaluated score = {final_val:.6f}")

    # store for possible reuse in subsequent runs
    globals()["height_sequence_1"] = final_seq

    return final_seq
