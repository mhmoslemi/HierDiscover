def get_subgradient(a):
    n = a.shape[0]
    fft_len = next_pow2(2*n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
    M = conv.max()
    active = np.where(conv >= M - 1e-12)[0] # maybe relative
    g = np.zeros_like(a)
    # compute contribution for each active k
    for k in active:
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # dp: a[i] gets contribution from a[k-i], so g[i] += 2*a[k-i]
        # vectorized:
        i = np.arange(i_start, i_end+1)
        j = k - i
        g[i] += 2 * a[j]
    return conv, M, active, g

def exponentiated_gradient_optimize(a0, time_limit, start_time, eta0=0.5):
    a = a0.copy()
    y = np.log(a + 1e-16)
    best_seq = a.copy()
    best_val = fast_score(a)
    t = 0
    while time.time() < start_time + time_limit:
        t += 1
        # compute conv & M
        n = a.shape[0]
        fft_len = next_pow2(2*n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # subgradient
        eps_active = 1e-12
        active = np.where(conv >= M - eps_active)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i = np.arange(i_start, i_end+1)
            j = k - i
            g[i] += 2 * a[j]
        # update logit
        eta = eta0 / np.sqrt(t)
        y -= eta * g
        # compute a via softmax
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()
        # jitter
        if t % 2000 == 0:
            sigma = 0.01/np.sqrt(a.shape[0])
            y += sigma * np.random.randn(a.shape[0])
    return best_seq, best_val

def expgrad_optimize(a0, time_limit, start_time, eta0=1.0):
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = fast_score(a)
    it = 0
    while time.time() < start_time + time_limit:
        it += 1
        # compute conv and gradient
        fft_len = next_pow2(2*n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:2*n-1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # subgradient
        eps_active = 1e-12
        active = np.where(conv >= M - eps_active)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n-1))
            i_end = min(n-1, k)
            i = np.arange(i_start, i_end+1)
            j = k - i
            g[i] += 2.0 * a[j]
        # update
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        # ensure nonnegative, then renorm
        sum_a = a.sum()
        if sum_a == 0:
            # reset to uniform
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        # jitter occasionally to escape stagnation
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            a = a + sigma * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
    return best_seq, best_val

"""
Propose a low‑score step function.

Algorithm
---------

* **Seed generation** – a small pool of candidate sequences is built.
  Candidates come from:
    – the previously discovered ‘height_sequence_1’ (if present),
    – a uniform distribution (baseline),
    – random parametrised shapes (power‑law, exponential,
      two‑sided power‑law, Beta pdf) of several lengths,
    – a random Dirichlet draw.
  All candidates are normalised to sum = 1.

* **Fast pre‑selection** – every candidate is scored with a fast FFT based
  evaluation (`fast_score`).  The best few (default 8) are kept for
  refinement.

* **Refinement – Exponentiated Gradient** – each selected seed is improved
  by a simple multiplicative‑weights (exponentiated‑gradient) optimiser.
  The gradient of the objective
      F(a) = 2·n·max(a∗a)
  is the sub‑gradient of the maximal convolution coefficient:
      g_i = 2·∑_{k∈A} a_{k‑i},
  where A is the (tiny) active set of indices attaining the max.
  The update is
      a ← a·exp(‑η·g) ,   a ← a / ∑a,
  with a diminishing step‑size η_t = η₀/√t.
  Occasional Gaussian jitter helps escape plateaus.

* **Optional Adam polishing** – after the EG phase a short Adam run (same
  sub‑gradient, Adam‑style momentum) can further lower the score.

* **Final global polishing** – any remaining time is spent refining the
  best sequence found so far.

All operations respect the simplex constraint (non‑negative, sum = 1)
and are performed with NumPy + FFT, well within the 1000 s budget.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) for a non‑negative sequence that
    already satisfies ∑seq = 1.  Uses an FFT for O(n log n) work.
    """
    n = seq.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return 2.0 * n * float(conv.max())


def compute_conv_and_active(a: np.ndarray, eps: float = 1e-12):
    """
    Return (M, active_indices, full convolution) where M = max(conv) and
    active_indices = {k | conv[k] ≥ M‑eps}.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    M = float(conv.max())
    active = np.where(conv >= M - eps)[0]
    return M, active, conv


def subgradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a for the given active set.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k - i
        g[i] += 2.0 * a[j]
    return g


# ----------------------------------------------------------------------
# Shape generators (used for random initialisation)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(‑p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(‑lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(‑p_left) + w_right·(n‑i)^(‑p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(α‑1)·(n‑i)^(β‑1).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def expgrad_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    eta0: float = 0.5,
    jitter_interval: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (multiplicative‑weights) optimiser.
    Returns the best sequence seen and its `fast_score`.
    """
    a = a0.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        if time.time() > start_time + time_limit:
            break
        it += 1

        # convolution & active set
        M, active, _ = compute_conv_and_active(a)

        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient
        g = subgradient(a, active)

        # step size
        eta = eta0 / np.sqrt(it)

        # multiplicative update + projection onto simplex
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0:
            a = np.full_like(a, 1.0 / n)   # fall back to uniform
        else:
            a = a / s

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            sigma = 0.01 / np.sqrt(n)
            a = a + sigma * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

    return best_seq, best_val


def adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the soft‑max parametrisation (identical to the
    original implementation but kept for polishing).
    """
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)                      # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        if time.time() > start_time + time_limit:
            break
        it += 1

        # soft‑max (probability simplex)
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # convolution
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of max(conv)
        eps_active = 1e-12
        active = np.where(conv >= M - eps_active)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i = np.arange(i_start, i_end + 1, dtype=np.int64)
            j = k - i
            g[i] += 2.0 * a[j]

        # gradient wrt logits
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter
        if it % 2000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights with a low
    `evaluate_sequence` value.  The routine respects the 1000 s budget,
    prints progress, and returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 0) collect raw initial candidates
    # --------------------------------------------------------------
    init_candidates: list[np.ndarray] = []

    # previously discovered sequence, if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s > 1e-12:
                prev = prev / s
                init_candidates.append(prev.copy())
                print("[info] using previously discovered sequence as a start")
    except Exception:
        pass

    # uniform baseline (length 10 000)
    uniform_len = 10000
    uniform_seq = np.full(uniform_len, 1.0 / uniform_len)
    init_candidates.append(uniform_seq)

    # --------------------------------------------------------------
    # 1) quick random shape sampling + fast pre‑selection
    # --------------------------------------------------------------
    length_choices = [4000, 8000, 10000, 12000, 15000]
    families = ["power", "exp", "twosided", "beta"]
    shape_samples: list[tuple[float, np.ndarray]] = []

    num_samples = 120  # enough to explore the space quickly
    for _ in range(num_samples):
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.05, 1.5)
            seq = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)
        elif family == "twosided":
            pl = random.uniform(0.05, 1.5)
            pr = random.uniform(0.05, 1.5)
            wl = random.uniform(0.2, 5.0)
            wr = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        else:  # beta
            a = random.uniform(0.05, 2.0)
            b = random.uniform(0.05, 2.0)
            seq = generate_beta_shape(n, a, b)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        val = fast_score(seq)
        shape_samples.append((val, seq))

    # keep the best few shapes for refinement
    shape_samples.sort(key=lambda x: x[0])
    top_k = 8
    for val, seq in shape_samples[:top_k]:
        init_candidates.append(seq.copy())
        print(f"[sample] seed shape score {val:.6f}")

    # a random Dirichlet draw (adds extra stochasticity)
    dir_n = random.choice(length_choices)
    dir_seq = np.random.gamma(shape=1.0, scale=1.0, size=dir_n)
    dir_seq = dir_seq / dir_seq.sum()
    init_candidates.append(dir_seq)

    # --------------------------------------------------------------
    # 2) allocate time per candidate and run refinements
    # --------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        raise RuntimeError("No time left after initialisation")

    per_candidate = remaining / len(init_candidates)
    print(f"[info] {len(init_candidates)} candidates, ≈{per_candidate:.1f}s each")

    best_overall_seq: np.ndarray | None = None
    best_overall_val = float("inf")

    for idx, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        cand_len = a0.shape[0]
        print(f"[optim] candidate #{idx + 1}/{len(init_candidates)} (len={cand_len})")

        # split the budget: majority → EG, remainder → Adam polishing
        cand_budget = min(per_candidate, deadline - time.time())
        if cand_budget <= 0:
            break
        eg_budget = cand_budget * 0.7
        adam_budget = cand_budget - eg_budget

        # Exponentiated‑gradient phase
        seq_eg, val_eg = expgrad_optimize(a0, eg_budget, time.time(), eta0=0.5)
        print(f"   [EG] best score {val_eg:.6f}")

        # Optional Adam polishing (only if we have a few seconds left)
        if adam_budget > 5.0:
            seq_ad, val_ad = adam_optimize(seq_eg, adam_budget, time.time(), lr0=0.5)
            print(f"   [Adam] best score {val_ad:.6f}")
            if val_ad < val_eg:
                seq_eg, val_eg = seq_ad, val_ad

        # update global best
        if val_eg < best_overall_val:
            best_overall_val = val_eg
            best_overall_seq = seq_eg.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # --------------------------------------------------------------
    # 3) final global polishing with any spare time
    # --------------------------------------------------------------
    final_remaining = deadline - time.time()
    if best_overall_seq is not None and final_remaining > 10.0:
        print(f"[final] polishing best sequence for {final_remaining:.1f}s")
        seq_ref, val_ref = expgrad_optimize(best_overall_seq, final_remaining, time.time(),
                                            eta0=0.3)
        if val_ref < best_overall_val:
            best_overall_val = val_ref
            best_overall_seq = seq_ref.copy()
            print(f"[final] improved global best to {best_overall_val:.6f}")

    # Fallback in the unlikely event we have no sequence
    if best_overall_seq is None:
        best_overall_seq = np.full(10000, 1.0 / 10000)

    # --------------------------------------------------------------
    # 4) final verification with the official evaluator
    # --------------------------------------------------------------
    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"[done] evaluate_sequence reports {final_score:.6f}")

    return final_list
