"""
Improved step‑function optimizer for the evaluation metric.

The algorithm is a hybrid of three ideas:

1. **Fast parametric sweep** – simple families (exponential, power‑law,
   edge‑biased, uniform) are sampled and the best initial shape is chosen
   using the fast FFT‑based score.

2. **Linear‑programming refinement** – given a current vector `f`,
   we solve a small LP that maximises the ℓ₁‑mass of a vector `g`
   under the linear constraints  (f * g)[k] ≤ M where M is the current
   maximal auto‑correlation.  After normalising `g`, an optimal convex
   mixing factor `t∈[0,1]` is selected (grid search) to obtain a new
   vector with a strictly smaller maximal convolution.  This step is
   repeated a few times.

3. **Local polishing** – a targeted hill‑climb that moves mass from the
   largest contributor to the current peak to the smallest contributor,
   followed by a random hill‑climb for fine‑tuning.

All heavy work (FFT, LP) respects the 1000 s budget; the routine falls
back gracefully if time runs out.  The returned sequence is non‑negative
and sums to 1, which matches the fast_score used during the search.
"""

import time
import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the first 2*n‑1 entries (full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector a.
    This is exactly the value returned by ``evaluate_sequence`` when the
    sum of ``a`` equals 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max objective (used by L‑BFGS – kept for optional final passes)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max‑convolution and its gradient w.r.t. logits y.
    """
    a = softmax(y)

    # a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation of L‑BFGS with decreasing smoothing parameters.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climbing utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Simple random hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """Greedy move from the biggest contributor to the peak to the smallest."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        # current convolution and location of the peak
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak: c_i = a_i * a_{k_max-i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback random move
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1
    return best, best_score


# ----------------------------------------------------------------------
# LP‑based refinement utilities
# ----------------------------------------------------------------------


def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix A such that (A @ g)[k] = (f * g)[k] for
    k = 0 … 2·n‑2.  The matrix has shape (2·n‑1, n) and contains O(n²)
    non‑zero entries.
    """
    n = f.size
    rows = []
    cols = []
    data = []

    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        js = np.arange(j_min, j_max + 1, dtype=int)
        is_ = k - js
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        data.extend(f[is_].tolist())

    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def lp_max_l1_under_conv(f: np.ndarray, M: float, time_limit: float = 30.0) -> np.ndarray:
    """
    Solve  max Σ g_j   s.t.   (f * g)_k ≤ M  (∀k) , g ≥ 0.
    Returns the optimal vector g (not normalised).
    """
    n = f.size
    A = build_conv_matrix(f)

    c = -np.ones(n)                       # maximise Σ g_j   → minimise -Σ g_j
    b = np.full(2 * n - 1, M)

    # linprog works with dense bounds, we keep non‑negative constraints
    bounds = [(0, None)] * n

    # Use the high‑performance HiGHS solver; we set a modest time limit.
    opts = {"presolve": True, "time_limit": time_limit}
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options=opts,
    )
    if not res.success:
        # Fallback: return the zero vector (feasible) – the caller will
        # simply keep the current f.
        return np.zeros_like(f)
    return res.x


def optimal_mixing(f: np.ndarray, g: np.ndarray, fft_len: int, n: int) -> tuple[np.ndarray, float]:
    """
    Find the mixing factor t∈[0,1] that minimises the max‑convolution of
    h = (1‑t)·f + t·g.  A small grid search is sufficient.
    Returns the new vector and its fast_score.
    """
    # Normalise g to the same ℓ₁‑mass as f (which is 1)
    sum_g = g.sum()
    if sum_g <= 0:
        return f, fast_score(f, fft_len, n)
    g_norm = g / sum_g

    best_h = f
    best_score = fast_score(f, fft_len, n)

    # grid of 21 points from 0 to 1
    for t in np.linspace(0.0, 1.0, 21):
        h = (1.0 - t) * f + t * g_norm
        score = fast_score(h, fft_len, n)
        if score < best_score:
            best_score = score
            best_h = h
    return best_h, best_score

# Parametric sweep for good initial guesses
# ----------------------------------------------------------------------


def parametric_initial_guess(n: int, fft_len: int) -> tuple[np.ndarray, float]:
    """
    Sample a handful of simple families (exponential, power‑law,
    edge‑biased, uniform) and return the best vector (sum = 1) together
    with its fast_score.
    """
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    # 1) Exponential edge‑biased
    for alpha in np.geomspace(0.001, 0.05, num=12):
        w = np.exp(-alpha * np.arange(n)) + np.exp(-alpha * np.arange(n - 1, -1, -1))
        a = w / w.sum()
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_score, best_a = s, a

    # 2) Power‑law edge‑biased
    for p in np.linspace(0.2, 4.0, num=15):
        w = (np.arange(1, n + 1) ** (-p)) + (np.arange(n, 0, -1) ** (-p))
        a = w / w.sum()
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_score, best_a = s, a

    # 3) Mixed edge / centre (as in the previous script)
    for m_edge_frac in np.linspace(0.01, 0.04, 4):
        m_edge = max(1, int(n * m_edge_frac))
        edge_val = np.sqrt(n / 2.0) / n
        centre_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, centre_val)
        a[:m_edge] = edge_val
        a[-m_edge:] = edge_val
        a /= a.sum()
        s = fast_score(a, fft_len, n)
        if s < best_score:
            best_score, best_a = s, a

    # 4) Uniform (already have) – kept for completeness
    return best_a, best_score
