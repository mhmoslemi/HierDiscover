"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Algorithm (multi‑phase):
1. Initialise a probability vector a (sum = 1) – use the supplied
   `height_sequence_1` if it exists, otherwise uniform.
2. Stochastic L‑BFGS on a smooth log‑sum‑exp approximation of the
   max‑convolution (soft‑max parametrisation).  The smoothing parameter
   is decreased geometrically (≈ 30 stages).  This quickly drives the
   objective down while staying on the probability simplex.
3. Linear‑programming mixing step (inspired by the “extreme‑point”
   update of the reference algorithm):
   * Solve   max Σ b_j   s.t.   a * b ≤ M,   b ≥ 0,
     where M is the current max‑convolution value.
   * Rescale b to the simplex (sum = 1) → g.
   * Replace a ← (1‑t)·a + t·g for a few values of t (0.1‑0.5) and keep
     the best improvement.  Repeat this LP‑mixing a few times.
   The LP is solved with CVXPY (ECOS).  The extreme‑point solution
   tends to flatten the self‑convolution further.
4. Projected sub‑gradient descent on the *exact* objective
   (max‑convolution).  Step size η_t = η₀ / √t, projection onto the
   simplex after each step.
5. Final aggressive greedy hill‑climb that transfers a random fraction
   of mass between coordinates while strictly decreasing the fast
   score.
6. Return the best sequence found (as a plain list).  The harness will
   evaluate it with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * a.size - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast surrogate for the true objective when Σ a = 1:
        2·n·max(a*a)
    """
    n = a.size
    conv = conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x ≥ 0, Σ x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# unconstrained soft‑max parameters y.
# ----------------------------------------------------------------------

    with respect to y (the pre‑soft‑max vector).
    """

    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c       # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z           # ∂ smooth_max / ∂ conv_i
    # gradient w.r.t. a:
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Heuristic initialisation of the soft‑max parameters
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if candidate + c in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Greedy hill‑climb
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    n = a.size

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a) = 2·n·max(conv(a,a)).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)
    n = a.size
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len)
        m = int(np.argmax(conv))

        # sub‑gradient of the active entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑programming mixing step (extreme‑point update)
# ----------------------------------------------------------------------


def lp_mixing_step(a: np.ndarray, M: float, fft_len: int, t_vals: list[float]) -> np.ndarray:
    """
    Perform one LP‑mixing update:
        max Σ b_j   s.t.   a * b ≤ M,   b ≥ 0
    Then rescale b to a probability vector and try a few convex
    combinations (weights in t_vals).  Returns the best new a (or the
    original a if no improvement).
    """
    try:
        import cvxpy as cp
    except Exception:
        # CVXPY not available – skip this phase
        return a

    n = a.size
    # CVXPY objects
    b = cp.Variable(n, nonneg=True)
    a_param = cp.Parameter(n, nonneg=True)
    a_param.value = a
    conv_expr = cp.conv(a_param, b)               # length 2n‑1
    M_vec = np.full(2 * n - 1, M)

    prob = cp.Problem(cp.Maximize(cp.sum(b)), [conv_expr <= M_vec])
    # use ECOS (fast interior‑point) – limit iterations to keep time bounded
    prob.solve(solver=cp.ECOS, max_iters=2000, feastol=1e-7, reltol=1e-7, abstol=1e-7)

    if b.value is None:
        return a  # fallback – LP failed

    b_opt = np.maximum(np.array(b.value).flatten(), 0.0)
    sum_b = b_opt.sum()
    if sum_b <= 1e-12:
        return a

    # Rescale to the simplex (sum = 1)
    g = b_opt / sum_b

    best_a = a
    best_M = M
    for t in t_vals:
        a_new = (1.0 - t) * a + t * g
        cur_M = conv_fft(a_new, fft_len).max()
        if cur_M < best_M - 1e-9:
            best_M = cur_M
            best_a = a_new

    return best_a


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Problem size – use the previous construction size if reasonable
    # ------------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and 2000 <= tmp.size <= 10000:
            n = int(tmp.size)

    # FFT lengths – powers of two for speed
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate on the simplex (Σ a = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(tmp, 0.0)
            if best_a.sum() == 0:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / best_a.sum()
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv)
    best_y = np.log(best_a + 1e-12)   # soft‑max parameters
    print(f"[init] n={n} → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS (≈ 60 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=26))
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    while time.time() < deadline - 180.0:   # reserve ~3 min for later phases
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.04, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=250,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} → fast score = {best_score:.6f}")

        # quick hill‑climb polishing of the L‑BFGS result
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – LP‑mixing extreme‑point updates (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 120.0:
        print("[lp_mix] starting LP‑mixing phase")
        t_vals = [0.10, 0.20, 0.30, 0.40, 0.50]
        mix_deadline = deadline - 110.0
        mix_iter = 0
        while time.time() < mix_deadline and mix_iter < 12:
            M_cur = conv_fft(best_a, fft_len_conv).max()
            a_new = lp_mixing_step(best_a, M_cur, fft_len_conv, t_vals)
            new_score = fast_score(a_new, fft_len_conv)
            if new_score + 1e-9 < best_score:
                best_a = a_new
                best_score = new_score
                print(f"[lp_mix {mix_iter}] improved fast score = {best_score:.6f}")
            else:
                # no improvement – break early
                break
            mix_iter += 1
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (≈ 15 % of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 45.0:
        print("[subgrad] starting projected sub‑gradient descent")
        sub_deadline = deadline - 40.0
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            sub_deadline,
            eta0=0.20,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished → fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final aggressive hill‑climb (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_hc_deadline = deadline
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            final_hc_deadline,
            max_iter=15_000,
            max_move=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Return the best sequence (plain list).  The harness will call
    # evaluate_sequence on the result.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # one final evaluation for logging (uses the provided harness function)
    final_val = evaluate_sequence(final_seq)   # type: ignore
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
