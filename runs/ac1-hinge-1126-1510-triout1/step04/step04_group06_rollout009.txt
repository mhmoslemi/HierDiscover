def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Determine length n; try using prior solution, else uniform.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0)
        n0 = min(prev.size, 4000)  # limit
        a0 = prev[:n0]
        a0 = a0 / a0.sum()
    else:
        n0 = 2500
        a0 = np.full(n0, 1.0/n0)

    # try different lengths: optionally we can try n=2000,3000
    length_options = [n0]
    # maybe try a larger length if within limit up to 5000
    if n0 < 5000:
        length_options.append(min(5000, n0*2))
    # we could try smaller lengths e.g., n0//2
    if n0 > 1000:
        length_options.append(max(1200, n0//2))

    best_a = None
    best_score = np.inf

    for n in length_options:
        # if n != len(a0) we need to interpolate a0 to length n
        if a0.size != n:
            a = np.interp(np.linspace(0, a0.size-1, n), np.arange(a0.size), a0)
        else:
            a = a0.copy()
        a = np.maximum(a, 0)
        a = a / a.sum()
        # compute fft lengths
        fft_len = next_pow2(2*n - 1)
        fft_len_g = next_pow2(3*n - 2)

        # Run multi-stage optimization: L-BFGS schedule
        y = np.log(a + 1e-12)
        lam_seq = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.006, 0.003, 0.001, 5e-4, 2e-4, 1e-4]
        for lam in lam_seq:
            if time.time() >= deadline:
                break
            # minimize with maxiter maybe 800 or 1200 based on remaining time
            remaining = deadline - time.time()
            max_iter = int(max(200, min(1500, remaining * 30)))  # heuristics
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_iter, 'gtol': 1e-12, 'disp': False})
            y = res.x   # warm start
            a_candidate = softmax(y)
            # maybe enforce min positivity: a_candidate = np.maximum(a_candidate, 1e-12)
            score_candidate = fast_score(a_candidate, fft_len, n)
            if score_candidate < best_score:
                best_score = score_candidate
                best_a = a_candidate.copy()
                best_n = n
                print(f"[L-BFGS] n={n} lam={lam} score={best_score:.6f}")

        # After L-BFGS schedule, final subgradient descent
        if time.time() < deadline:
            sub_a, sub_score = subgradient_momentum_optimize(best_a, deadline, fft_len, fft_len_g, n)
            if sub_score < best_score:
                best_score = sub_score
                best_a = sub_a.copy()
                best_n = n
                print(f"[Subgrad] n={n} score={best_score:.6f}")

        # Simulated annealing
        if time.time() < deadline:
            rng_local = np.random.default_rng(seed + n)  # deterministic but varied
            sa_a, sa_score = simulated_annealing(best_a, best_score, fft_len, n, deadline, rng_local)
            if sa_score < best_score:
                best_score = sa_score
                best_a = sa_a.copy()
                best_n = n
                print(f"[SA] n={n} score={best_score:.6f}")

        # final hill climb
        if time.time() < deadline:
            final_a, final_score = hill_climb(best_a, best_score, fft_len, n, deadline, max_iter=5000)
            if final_score < best_score:
                best_score = final_score
                best_a = final_a.copy()
                best_n = n
                print(f"[Hill] n={n} final_score={best_score:.6f}")

    # final evaluate
    try:
        eval_val = evaluate_sequence(best_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}, n={best_n}")

    globals()['height_sequence_1'] = best_a.tolist()
    return best_a.tolist()

def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor):
   a = softmax(y)
   fa = np.fft.fft(a, fft_len_conv)
   conv_full = np.real(np.fft.ifft(fa * fa))[:2*n-1]
   max_c = conv_full.max()
   lam = max_c * lam_factor + 1e-12
   shifted = conv_full - max_c
   exp_shift = np.exp(shifted / lam)
   Z = exp_shift.sum()
   smooth_max = max_c + lam * np.log(Z)
   f = 2.0 * n * smooth_max
   # gradient weight w_k = exp_shift_k / Z
   w = exp_shift / Z
   # compute g_i = sum_j a_j * w_{i+j}
   # compute convolution a_rev * w
   a_rev = a[::-1]
   fa_rev = np.fft.fft(a_rev, fft_len_g)
   fw = np.fft.fft(w, fft_len_g)
   conv_rev = np.real(np.fft.ifft(fa_rev * fw))
   g_vec = conv_rev[(n-1):(n-1+n)]
   # chain rule through softmax
   mean_g = np.dot(a, g_vec)
   grad_y = a * (g_vec - mean_g)
   return f, grad_y
```

            x_old = np.linspace(0, 1, a0.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, a0)

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Score = 2·n·max(conv) for a probability vector a (sum≈1)."""
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed‑max objective based on log‑sum‑exp and its gradient w.r.t. y.

    Parameters
    ----------
    y : np.ndarray
        Unconstrained variables; a = softmax(y) lies on the simplex.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths for the convolution and the gradient convolution.
    lam_factor : float
        Controls the smoothing; smaller → closer to the true max.

    Returns
    -------
    f : float
        2·n·smooth_max(conv).
    grad_y : np.ndarray
        Gradient w.r.t. y (shape n).
    """
    a = softmax(y)

    # full convolution a*a (linear, length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                     # stabilise exp
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. a
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # central part yields the required vector of size n
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (with momentum)
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Directly minimise the true max‑convolution using a momentum
    sub‑gradient method with Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9                    # momentum coefficient
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active‑set mask (indices attaining the max, tolerance)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max(conv) w.r.t. a
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size ~ 1/√t
        step = 1.0 / np.sqrt(it + 1.0)
        a_new = a - step * v
        a = simplex_projection(a_new)

        # keep best
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Local optimiser that repeatedly moves a tiny fraction of mass
    between two positions if it improves the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    score_start: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that relocates a random fraction of mass.
    Improves exploration after the gradient phases.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # Initial temperature as a small fraction of the current score
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9993  # slower cooling for better exploration
    it = 0

    while time.time() < deadline:
        # pick a source index with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction (up to 25 %) of the source mass
        max_delta = cur_a[i] * 0.25
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        # Metropolis acceptance
        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert move
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        # occasional reheating
        if it % 4000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to build various initial distributions
# ----------------------------------------------------------------------


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n according to the requested mode.
    Modes:
        'uniform' – plain uniform.
        'edge'    – heavier mass near the two ends.
        'exp'     – exponential decay from left (mirrored to right).
        'power'   – power‑law decay.
        'random'  – Dirichlet(0.5) random.
    """
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "edge":
        frac = rng.uniform(0.004, 0.02)          # fraction of indices at each edge
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == "random":
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Optimisation for a single length
# ----------------------------------------------------------------------


def optimise_length(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length n starting from init_a (already normalised).
    Returns the best sequence found and its score.
    """
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = init_a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    # ---------- warm‑started L‑BFGS with decreasing smoothing ----------
    y = np.log(best_a + 1e-12)
    lam_seq = [
        0.5,
        0.3,
        0.15,
        0.08,
        0.04,
        0.02,
        0.01,
        0.006,
        0.003,
        0.001,
        5e-4,
        2e-4,
        1e-4,
    ]

    for lam in lam_seq:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        max_iter = int(min(1500, max(200, remaining * 30)))  # heuristic
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "gtol": 1e-12, "disp": False},
        )
        y = res.x
        a_candidate = softmax(y)
        score_candidate = fast_score(a_candidate, fft_len_conv, n)
        if score_candidate < best_score:
            best_score = score_candidate
            best_a = a_candidate.copy()
            print(f"[L‑BFGS] n={n} lam={lam:.5f} score={best_score:.6f}")

        # occasional jitter to shake out of tiny basins
        if rng.random() < 0.1:
            y += rng.normal(scale=0.01, size=y.shape)

    # ---------- momentum sub‑gradient descent ----------
    if time.time() < deadline:
        sub_a, sub_score = subgradient_momentum_optimize(
            best_a, deadline, fft_len_conv, fft_len_g, n
        )
        if sub_score < best_score:
            best_score = sub_score
            best_a = sub_a.copy()
            print(f"[sub‑grad] n={n} score={best_score:.6f}")

    # ---------- simulated annealing ----------
    if time.time() < deadline:
        sa_a, sa_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, deadline, rng
        )
        if sa_score < best_score:
            best_score = sa_score
            best_a = sa_a.copy()
            print(f"[SA] n={n} score={best_score:.6f}")

    # ---------- final greedy hill‑climb ----------
    if time.time() < deadline:
        final_a, final_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=5000,
        )
        if final_score < best_score:
            best_score = final_score
            best_a = final_a.copy()
            print(f"[hill] n={n} final_score={best_score:.6f}")

    # ---------- optional random restarts while time permits ----------
    while time.time() + 5.0 < deadline:
        mode = rng.choice(["uniform", "edge", "exp", "power", "random"])
        a0 = build_initial(mode, n, rng)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        y = np.log(a0 + 1e-12)

        # short L‑BFGS run (coarser schedule)
        for lam in [0.3, 0.1, 0.02, 0.005]:
            if time.time() >= deadline:
                break
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": 600, "gtol": 1e-12, "disp": False},
            )
            y = res.x
            a_candidate = softmax(y)
            score_candidate = fast_score(a_candidate, fft_len_conv, n)
            if score_candidate < best_score:
                best_score = score_candidate
                best_a = a_candidate.copy()
                print(f"[restart L‑BFGS] n={n} lam={lam:.5f} score={best_score:.6f}")

        # brief sub‑gradient + SA refinement
        if time.time() < deadline:
            sub_a, sub_score = subgradient_momentum_optimize(
                best_a, deadline, fft_len_conv, fft_len_g, n
            )
            if sub_score < best_score:
                best_score = sub_score
                best_a = sub_a.copy()
                print(f"[restart sub‑grad] n={n} score={best_score:.6f}")

        if time.time() < deadline:
            sa_a, sa_score = simulated_annealing(
                best_a, best_score, fft_len_conv, n, deadline, rng
            )
            if sa_score < best_score:
                best_score = sa_score
                best_a = sa_a.copy()
                print(f"[restart SA] n={n} score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising `evaluate_sequence`.
    The method evaluates several problem sizes, runs a sequence of
    L‑BFGS → sub‑gradient → simulated annealing → hill‑climb phases,
    and keeps the best result found before the time budget expires.
    """
    rng = np.random.default_rng(seed + 123456)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Retrieve previous construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.array(globals()["height_sequence_1"], dtype=float)
        prev = np.maximum(prev, 0.0)
        max_n_allowed = 4000
        if prev.size > max_n_allowed:
            prev = prev[:max_n_allowed]
        a0 = prev / prev.sum()
        base_n = a0.size
    else:
        base_n = 2500
        a0 = np.full(base_n, 1.0 / base_n)

    # ------------------------------------------------------------------
    # Choose several lengths to explore (current, larger, smaller)
    # ------------------------------------------------------------------
    length_options = [base_n]
    if base_n < 5000:
        length_options.append(min(5000, base_n * 2))
    if base_n > 1200:
        length_options.append(max(1200, base_n // 2))
    length_options = sorted(set(length_options))

    best_global_a = None
    best_global_score = np.inf
    best_global_n = None

    for n in length_options:
        if time.time() >= deadline:
            break

        # interpolate / truncate the previous best to the new size
        if a0.size == n:
            a_start = a0.copy()
        else:
            # linear interpolation preserving total mass
            x_old = np.linspace(0, 1, a0.size)
            x_new = np.linspace(0, 1, n)
            a_start = np.interp(x_new, x_old, a0)
        a_start = np.maximum(a_start, 0.0)
        a_start = a_start / a_start.sum()

        print(f"[init] Optimising length n={n}")
        best_a_n, best_score_n = optimise_length(n, a_start, deadline, rng)

        if best_score_n < best_global_score:
            best_global_score = best_score_n
            best_global_a = best_a_n.copy()
            best_global_n = n
            print(f"[global best] n={n} score={best_global_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the supplied evaluator (optional, for logging)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(best_global_a.tolist())
    except Exception:
        eval_val = float("nan")
    print(f"[final] evaluate_sequence -> {eval_val:.6f}, length={best_global_n}")

    # store globally for possible reuse in a later call
    globals()["height_sequence_1"] = best_global_a.tolist()

    return best_global_a.tolist()
