"""
Propose a step‑function (non‑negative sequence) that minimizes
    2·n·max(conv(a,a)) / (sum(a))²
The evaluation is scale‑invariant, therefore we keep a on the probability
simplex (sum = 1).  The algorithm is a multi‑phase search:

1.  Generate a diverse set of parametric seeds (uniform, linear, exponential,
    power‑law, edge‑heavy, sinusoidal, triangular, random Dirichlet, etc.).
    Keep the best few seeds.

2.  Continuation L‑BFGS on a log‑sum‑exp smoothed version of the max‑convolution.
    The smoothing factor λ is decreased on a long schedule.  Each L‑BFGS run
    starts from a jittered seed; the best result over many restarts is kept.

3.  Long projected sub‑gradient descent on the exact max‑convolution
    (the true non‑smooth objective).  Simple back‑tracking line‑search is used.

4.  A short second L‑BFGS refinement with a very fine λ‑schedule to polish
    the solution obtained from the sub‑gradient phase.

5.  Aggressive hill‑climbing (mass transfers between two coordinates) for final
    polishing.

All phases respect the simplex by projection (or by the soft‑max mapping) and
use FFT‑based convolutions (O(n log n)).  The total wall‑clock time is limited
by *budget_s* (default 1000 s); the best sequence found before the deadline is
returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → the probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1 (the true score is 2·n·max(a*a) / sum(a)², but the ratio is
    scale‑invariant, so we only need max(a*a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient w.r.t. soft‑max logits
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Returns (objective, gradient) where the objective is 2·n·smooth_max,
    with smooth_max = log‑sum‑exp smoothing of the max‑convolution.
    The gradient is with respect to the logits y (softmax arguments).
    """
    a = softmax(y)

    # ----- convolution a*a -------------------------------------------------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth max -------------------------------------------------------
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12          # smoothing scale
    shifted = conv_full - max_c               # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. the convolution entries ------------
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i

    # ----- back‑prop to logits (soft‑max chain rule) -----------------------
    # convolution of a_rev with w gives gradient of conv w.r.t a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # The relevant part corresponds to the original a‑indices
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (smooth max) with a schedule of λ‑factors
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)             # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Occasionally L‑BFGS may fail – ignore and continue.
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"[lbfgs] improved score={best_score:.6f} (λ={lam:.5f})")
        # short hill‑climb after each λ to sharpen the result
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=800)
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via projected sub‑gradient descent.
    Uses a back‑tracking line search and keeps the best sequence seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # ---- convolution and active mask (indices attaining the max) -----
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ---- sub‑gradient of max‑convolution ---------------------------
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # ---- simple back‑tracking line search ---------------------------
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1
        # occasional progress report
        if it % 5000 == 0:
            print(f"[subgrad] it={it}, best_score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two distinct coordinates, move a small random fraction from one
    to the other, accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.07   # up to 7 % of source mass
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1
        if it % 2000 == 0:
            # lightweight progress output
            print(f"[hill] it={it}, best_score={best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats (the sequence) that respects all
    constraints required by `evaluate_sequence`.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Decide problem size and starting point
    # ------------------------------------------------------------------
    max_n = 3000                     # a few thousand entries – manageable & expressive
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            tot = a0.sum()
            a0 = a0 / tot if tot > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)    # for a*a
    fft_len_g = next_pow2(3 * n - 2)       # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep (collect promising seeds)
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # 1) Uniform
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_candidate(lin)

    # 3) Exponential families
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # 4) Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Edge‑heavy (mass at both ends)
    for edge_frac in [0.004, 0.008, 0.012, 0.02]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # 6) Sinusoidal‑squared
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # 7) Symmetric triangular shape (mountain)
    tri = np.minimum(np.arange(n) + 1,
                     n - np.arange(n))
    add_candidate(tri)

    # 8) Random Dirichlet draws with several concentrations
    for alpha in [0.3, 0.6, 1.0, 2.0, 5.0]:
        for _ in range(3):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # keep the best few seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5           # safety margin
    # 55 % for the first L‑BFGS phase, next 30 % for sub‑gradient,
    # remaining for final refinements + hill‑climb.
    phase1_deadline = start_time + 0.55 * budget_s
    phase2_deadline = start_time + 0.85 * budget_s
    phase3_deadline = deadline

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (many restarts, long λ schedule)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.12, 0.09,
        0.07, 0.05, 0.04, 0.03, 0.02, 0.015, 0.01,
        0.008, 0.006, 0.004, 0.002, 0.0015, 0.001, 5e-4
    ]
    maxiter_per_lam = 350      # a bit more than the original 250

    restarts = 0
    while time.time() < phase1_deadline and restarts < 15:
        # Choose a seed and add a tiny jitter for diversity
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 2e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_deadline,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f} after {restarts+1} restarts")
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (longer)
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        sg_deadline = phase2_deadline
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short second L‑BFGS refinement with very fine smoothing
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        low_lam_schedule = [0.001, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]
        # run at most two quick restarts (starting from current best)
        for _ in range(2):
            if time.time() >= phase3_deadline:
                break
            a_opt, score_opt = lbfgs_continuation(best_a,
                                                  low_lam_schedule,
                                                  maxiter_per_lam=150,
                                                  deadline=phase3_deadline,
                                                  fft_len_conv=fft_len_conv,
                                                  fft_len_g=fft_len_g,
                                                  n=n)
            if score_opt < best_score:
                best_score, best_a = score_opt, a_opt.copy()
                print(f"[refine‑L‑BFGS] improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=15000)

    # ------------------------------------------------------------------
    # Return the sequence (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
