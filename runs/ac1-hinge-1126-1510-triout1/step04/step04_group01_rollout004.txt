def beta_distribution(alpha, beta, n):
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    a = np.power(x, alpha-1) * np.power(1 - x, beta - 1)
    a[np.isnan(a)] = 0.0
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.full(n, 1.0 / n)
    return a

"""
Propose a low‑score non‑negative step sequence.

Algorithm Overview
------------------
1. **Initialisation** – start from the previous best if available, otherwise uniform.
2. **Beta‑family sweep** – sample many Beta‑shaped distributions
   (a_i ∝ x^{α‑1}(1‑x)^{β‑1}) on the simplex and keep the best.
3. **Continuation L‑BFGS** – many random restarts, each runs an
   L‑BFGS optimiser on a smooth log‑sum‑exp approximation of the max‑convolution.
   The smoothing factor (`lam_factor`) is decreased in a continuation schedule.
4. **Long sub‑gradient descent** – projected sub‑gradient with momentum
   directly on the true max‑convolution.  A generous iteration budget drives the
   peak down further.
5. **Adam fine‑tuning** – optimisation of the same smooth objective on the
   logit parametrisation for high‑precision refinement.
6. **Final greedy hill‑climb** – tiny random mass transfers that only accept
   strictly improving moves.

The time budget (default 1000 s) is split automatically among the phases.
All iterates respect the simplex (non‑negative, sum = 1).  The best sequence
found before the deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a distribution already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max via log‑sum‑exp.
    Returns (objective, gradient) with respect to logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                           # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (a_rev * mask) in freq. domain
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each lam in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)  # logits

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # optimisation failure – just continue
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int,
                                  step_scale: float = 0.8,
                                  beta: float = 0.9):
    """
    Projected sub‑gradient descent with simple momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    it = 0

    while time.time() < deadline:
        # full convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size decreasing like 1/sqrt(it)
        step = step_scale / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the logit parameters
# ----------------------------------------------------------------------


def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.002,
                  lr: float = 0.15):
    """
    Adam optimisation of the smoothed max objective starting from logits y0.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 8000,
               max_frac: float = 0.10):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a random fraction (up to max_frac) from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * max_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Random initialiser
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:  # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Beta‑family generator
# ----------------------------------------------------------------------


def beta_distribution(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised Beta(α,β) on a symmetric grid.
    a_i ∝ x_i^{α‑1} (1‑x_i)^{β‑1}, where x_i = (i+0.5)/n.
    """
    idx = np.arange(n, dtype=float) + 0.5
    x = idx / n
    # guard against overflow/underflow
    with np.errstate(divide='ignore', invalid='ignore'):
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial distribution
    # ------------------------------------------------------------------
    max_n = 4000
    # start with a moderate size; 2000 works well within the budget
    n = max_n // 2

    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)   # length for a*a
    fft_len_g = next_pow2(3 * n - 2)      # length for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.7   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – sweep Beta‑family (≈10 % of budget)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.10 * budget_s
    while time.time() < phase0_end:
        # sample α,β logarithmically
        alpha = np.exp(rng.uniform(np.log(0.05), np.log(5.0)))
        beta = np.exp(rng.uniform(np.log(0.05), np.log(5.0)))
        a_beta = beta_distribution(alpha, beta, n)
        score = fast_score(a_beta, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a_beta.copy()
            print(f"[beta] α={alpha:.3f}, β={beta:.3f}, new best={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – many L‑BFGS continuations (≈45 % of budget)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.6, 0.5, 0.4, 0.3, 0.2,
        0.15, 0.1, 0.07, 0.045, 0.03,
        0.018, 0.01, 0.006, 0.003, 0.001
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long sub‑gradient descent with momentum (≈30 % of budget)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_optimize_momentum(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
            step_scale=0.9,   # slightly larger steps for faster descent
            beta=0.9
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam fine‑tuning on logits (≈10 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        adam_deadline = deadline - 0.2
        logits = np.log(best_a + 1e-12)
        best_a, best_score = adam_optimize(
            logits,
            n,
            fft_len_conv,
            fft_len_g,
            adam_deadline,
            lam_factor=0.0015,
            lr=0.12
        )
        print(f"[Adam] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (uses remaining time)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=12000,
        max_frac=0.10
    )
    print(f"[final hill‑climb] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (as a plain list of floats)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # sanity check – external evaluation (will be performed by the runner anyway)
    try:
        final_eval = evaluate_sequence(final_seq)
        print(f"[final] evaluated score = {final_eval:.6f}")
    except Exception:
        # if evaluate_sequence is not available in this scope, ignore
        pass

    return final_seq
