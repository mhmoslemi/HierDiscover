"""
Propose a step‑function sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

using a simple “LP‑mix” iteration:
    1. Start from a probability vector a (sum = 1).
    2. Compute M = max(conv(a,a)).
    3. Solve a linear program:
           maximise   Σ g_j
           subject to g ≥ 0
                      (a * g)_k ≤ M  for all k
       (a * g) is the linear convolution of a with the unknown g.
    4. Normalise the LP solution g to unit sum and linearly blend
       it with a (several blend factors are tested).  Keep the blend
       that yields the smallest true score (computed with an FFT‑based
       convolution).
    5. Repeat 2‑4 a few times while improvements are observed.
    6. Finally polish the best vector with a cheap random hill‑climb.

The LP guarantees that the new vector can only increase the total
mass while respecting the current max‑convolution bound; the blending
therefore reduces the bound.  The whole procedure respects the
global time‑budget (default 1000 s) and stores the best sequence in the
global variable ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz
import scipy.sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Compute 2·n·max(conv(a,a)) with a already normalised (sum = 1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum = 1, v ≥ 0)."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Random initialisers
# ----------------------------------------------------------------------


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n.
    Modes:
        'uniform' – all entries equal,
        'edge'    – larger mass at the two ends,
        'exp'     – exponential decay from the left (mirrored),
        'power'   – power‑law decay,
        'random'  – Dirichlet(0.5) random.
    """
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)          # how many entries are “edges”
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        # Dirichlet with concentration 0.5 (spikier than uniform)
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:   # fallback
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Simple hill‑climb (fine‑tuning)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Very cheap random hill‑climb that preserves Σa = 1.
    Moves a tiny fraction of mass from one index to another if it improves
    the fast_score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move at most 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

            bounds = [(0, None)] * n
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=bounds, method='highs',
                          options={'presolve': True})

            if not res.success:
                print(f"[LP {restart_idx}/{iter_idx}] failed: {res.message}")
                break

            g = res.x
            if g.sum() == 0.0:
                break
            g_norm = g / g.sum()          # keep the sum on the simplex

            # --------------------------------------------------------------
            # Try several convex combinations (1‑t)·a + t·g_norm
            # --------------------------------------------------------------
            best_mix_a = local_best_a
            best_mix_score = local_best_score
            for t in [0.01, 0.025, 0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                cand = (1.0 - t) * local_best_a + t * g_norm
                cand_score = fast_score(cand, fft_len, n)
                if cand_score < best_mix_score:
                    best_mix_score = cand_score
                    best_mix_a = cand

            if best_mix_score < local_best_score - 1e-9:
                local_best_a = best_mix_a
                local_best_score = best_mix_score
                print(f"[restart {restart_idx} iter {iter_idx}] improved to {local_best_score:.6f}")
                iter_idx += 1
            else:
                # no improvement – exit the LP‑mix loop
                break

        # ------------------------------------------------------------------
        # Final local polishing with hill‑climb (30 s or remaining time)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            hc_deadline = min(deadline, time.time() + 30.0)
            local_best_a, local_best_score = hill_climb(
                local_best_a,
                local_best_score,
                fft_len,
                n,
                hc_deadline,
                max_iter=3000)

        # update global best
        if local_best_score < best_score:
            best_score = local_best_score
            best_a = local_best_a.copy()
            print(f"[global] new best score {best_score:.6f}")

        restart_idx += 1

    # ----------------------------------------------------------------------
    # Global final hill‑climb (use all remaining budget)
    # ----------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_iter=5000)

    # ----------------------------------------------------------------------
    # Report final evaluation (optional)
    # ----------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence -> {eval_val:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence raised:", exc)

    # store globally for possible reuse
    globals()['height_sequence_1'] = best_a.tolist()

    return best_a.tolist()
