"""
Propose a low‑value step‑function sequence.

The strategy combines several complementary phases:

1. **Parametric sweep** – generate a handful of simple families (uniform,
   linear, exponential, power‑law, edge‑heavy, sinusoid, Dirichlet draws)
   and keep the best few as seeds.

2. **LP‑driven refinement** – given a current distribution f,
   solve a linear program that maximises Σ gᵢ under the constraints
   (f∗g)(k) ≤ M where M = max(f∗f).  The obtained g is mixed with f
   (convex combination) and accepted only if it reduces the max‑convolution.
   This is essentially the “alternating linear‑program” technique described
   in the problem statement.  The LP matrix is built once as a sparse
   Toeplitz operator (O(n²) non‑zeros) and reused each iteration.

3. **Continuation L‑BFGS** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) with a decreasing λ‑schedule, using many
   random restarts and a generous iteration budget.

4. **Projected sub‑gradient descent** – exact max‑convolution sub‑gradients
   with a back‑tracking line‑search.

5. **Simulated‑annealing + greedy hill‑climb** – escape tiny local minima.

6. **Final polishing hill‑climb** – deterministic mass‑transfer moves.

All phases respect the simplex (non‑negative, Σ a = 1).  The code
automatically splits the 1000 s budget; the best sequence found before the
deadline is returned.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1 (the true score is 2·n·max(a*a) / sum(a)², but the ratio is
    scale‑invariant, so we only need max(a*a)).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                # for stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation with simple line‑search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int):
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    Uses a simple back‑tracking line‑search.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing λ schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)    # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # short hill‑climb after each λ
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score


# ----------------------------------------------------------------------
# Linear‑program step (f ∗ g ≤ M) and convex combination update
# ----------------------------------------------------------------------
def build_conv_matrix_indices(n: int):
    """
    For a fixed n, build row/column index arrays for the Toeplitz matrix
    representing the linear operator (f ∗ g)[k] = Σ_i f_i g_{k-i}.
    """
    # length of convolution = 2*n - 1
    size = n * n
    row = np.empty(size, dtype=np.int32)
    col = np.empty(size, dtype=np.int32)
    pos = 0
    for i in range(n):
        row[pos:pos + n] = np.arange(i, i + n, dtype=np.int32)
        col[pos:pos + n] = i
        pos += n
    return row, col


def solve_lp_for_g(f: np.ndarray, M: float, row_idx, col_idx):
    """
    Given current distribution f and a bound M = max(f∗f),
    solve   max Σ g_i   s.t.   (f∗g)_k ≤ M  (∀k) , g_i ≥ 0.
    Returns the optimal g (or None on failure).
    """
    n = f.shape[0]
    # data = kron(f, ones(n)) creates f_i repeated n times
    data = np.kron(f, np.ones(n, dtype=np.float64))
    A = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # minimise –Σg  ⇔ maximise Σg
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print("[LP] linprog exception:", e)
        return None

    if res.success:
        return res.x
    else:
        # occasional numerical issues – treat as failure
        return None


def lp_refinement(a_init: np.ndarray,
                  deadline: float,
                  fft_len_conv: int,
                  n: int):
    """
    Perform the alternating‐LP refinement described in the problem statement.
    Returns a refined distribution and its fast_score.
    """
    a = a_init.copy()
    # Pre‑compute sparse matrix indices once
    row_idx, col_idx = build_conv_matrix_indices(n)

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    # Small set of mixing coefficients to test
    mix_ts = (0.4, 0.2, 0.1, 0.05)

    while time.time() < deadline:
        # current bound M
        conv = full_conv(a, fft_len_conv)
        M = conv.max()

        # solve linear program for g
        g_raw = solve_lp_for_g(a, M, row_idx, col_idx)
        if g_raw is None:
            break

        sum_g = g_raw.sum()
        if sum_g <= 1e-12:
            break

        g = g_raw / sum_g  # normalise to a probability vector

        # try a few convex combinations
        improved = False
        for t in mix_ts:
            cand = (1.0 - t) * a + t * g
            cand_score = fast_score(cand, fft_len_conv, n)
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                improved = True
                break

        if not improved:
            # No improvement for any mixing coefficient – stop early
            break

        it += 1

    print(f"[LP] finished after {it} iterations, best fast_score={best_score:.6f}")
    return best_a, best_score

            a0 = a0 / tot if tot > 0 else np.full(n, 1.0 / n)
        else:
            n = max_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = max_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)    # for a*a
    fft_len_g = next_pow2(3 * n - 2)       # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap parametric sweep to collect promising seeds
    # ------------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # 1) Uniform (baseline)
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    lin = np.arange(n, 0, -1, dtype=float)
    add_candidate(lin)

    # 3) Exponential families
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        add_candidate(np.exp(-rate * np.arange(n)))

    # 4) Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Edge‑heavy (mass at both ends)
    for edge_frac in [0.005, 0.01, 0.02, 0.04]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # 6) Sinusoidal‑squared (always non‑negative)
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-3)

    # 7) Random Dirichlet (Gamma) draws with various concentration
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(4):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # keep the top seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:8]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin
    phase1_end = start_time + 0.45 * budget_s   # ~45 % for LP refinement
    phase2_end = start_time + 0.75 * budget_s   # next ~30 % for L‑BFGS
    phase3_end = deadline                       # remaining for sub‑gradient / anneal

    # ------------------------------------------------------------------
    # Phase 1 – alternating LP refinement (multiple restarts)
    # ------------------------------------------------------------------
    lp_deadline = phase1_end
    restarts = 0
    while time.time() < lp_deadline and restarts < 10:
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()
        a_opt, score_opt = lp_refinement(a_start, lp_deadline, fft_len_conv, n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[LP] new best score={best_score:.6f}")
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.06, 0.03,
                        0.015, 0.008, 0.004, 0.002, 0.001,
                        0.0005, 0.0002, 0.0001]
        maxiter_per_lam = 500
        restarts = 0
        while time.time() < phase2_end and restarts < 30:
            seed_vec = rng.choice(top_seeds)
            jitter = rng.random(n) * 1e-3
            a_start = seed_vec + jitter
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
            a_opt, score_opt = lbfgs_continuation(a_start,
                                                  lam_schedule,
                                                  maxiter_per_lam,
                                                  phase2_end,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
            if score_opt < best_score:
                best_score, best_a = score_opt, a_opt.copy()
                print(f"[L‑BFGS] new best score={best_score:.6f}")
            restarts += 1

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent with line search
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        sg_deadline = phase3_end - 0.3   # keep a little buffer for annealing
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing / random perturbations
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # Re‑use the hill‑climb routine as a very cheap annealer
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        deadline,
                                        max_iter=8000)
        print(f"[anneal/hill] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=12000)

    # ------------------------------------------------------------------
    # Return the sequence (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # evaluate using the external checker (may enforce clipping etc.)
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
