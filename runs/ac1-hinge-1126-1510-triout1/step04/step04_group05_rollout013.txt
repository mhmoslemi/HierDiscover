def subgradient_step(a, step_size):
    conv = conv_fft(a)
    max_b = conv.max()
    K = np.where(conv >= max_b - 1e-12)[0] # indices where near max
    g = np.zeros_like(a)
    for k in K:
        lo = max(0, k - len(a) + 1)
        hi = min(k, len(a)-1)
        # slice a segment reversed
        # a_{k-i} for i in [lo,hi] is a[k - i] = a_segment_rev
        idx = np.arange(lo, hi+1)
        g[idx] += 2 * a[k - idx]
    # Normalize sum of a to 1 after step
    a_new = a - step_size * g
    # project onto simplex
    a_new = simplex_projection(a_new)
    return a_new

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    # choose n
    n = 2000  # can maybe adjust to 2500 if time permits
    # previous best if available
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        arr = np.maximum(arr, 0.0)
        if arr.size >= n:
            a0 = arr[:n]
        else:
            # pad
            a0 = np.full(n, 1.0/n)
        a0_sum = a0.sum()
        if a0_sum > 1e-12:
            a0 /= a0_sum
        else:
            a0 = np.full(n, 1.0/n)
    else:
        a0 = np.full(n, 1.0/n)

    best_a = a0.copy()
    best_score = fast_score(best_a, next_pow2(2*n-1), n)

    # Pre-compute FFT len
    fft_len = next_pow2(2 * n - 1)
    # lambda schedule
    lam_schedule = np.geomspace(0.5, 5e-8, num=40)  # extended deeper

    # Main loop of restarts while time
    iteration = 0
    while time.time() < deadline - 5:
        # choose initialization method randomly
        mode = rng.choice(["random", "exp", "power", "edge", "prev"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n+1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_ratio = rng.uniform(5, 50)  # ratio
            # base value for interior
            interior_val = 1.0
            edge_val = interior_val * edge_ratio
            a = np.full(n, interior_val, dtype=float)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # prev
            a = best_a.copy()
            y0 = np.log(a + 1e-12) + rng.normal(scale=0.02, size=n)

        # Run continuation L-BFGS
        y_opt = schedule_optimize(y_start=y0,
                                 n=n,
                                 fft_len_conv=fft_len,
                                 fft_len_g=fft_len,  # maybe same
                                 lam_factors=lam_schedule,
                                 maxiter_per_stage=800)  # more iterations
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len, n)
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] {mode} L-BFGS improved fast score = {best_score:.6f}")

        # Subgradient mirror descent
        a_tmp, score_tmp = subgradient_mirror_descent(best_a, max_iters=5000,
                                                       step0=0.8, step_decay=0.9995,
                                                       deadline=deadline,
                                                       fft_len=fft_len, n=n)
        if score_tmp < best_score:
            best_score = score_tmp
            best_a = a_tmp.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] after mirror descent fast score = {best_score:.6f}")

        # Targeted hill climb (enhanced)
        best_a, best_score = targeted_hill_climb(best_a, best_score,
                                                 fft_len, n,
                                                 deadline,
                                                 max_moves=5000,
                                                 frac=0.07)
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # Final refinement loops
    # Additional L-BFGS passes with fine lam schedule
    for lam in np.geomspace(1e-5, 1e-8, num=20):
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad, x0=best_y, args=(n, fft_len, fft_len, lam),
                       method="L-BFGS-B", jac=True,
                       options={"maxiter": 1000, "gtol": 1e-14, "disp": False})
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[final L-BFGS] lam={lam:.2e} fast score={best_score:.6f}")

    # final mirror descent deeper

        a_last, score_last = subgradient_mirror_descent(best_a,

                                                        fft_len=fft_len,
                                                        n=n)
        if score_last < best_score:

    # final random hill climb

"""
Propose a low‑score step‑function sequence.

The search pipeline combines several complementary ideas:

* **Continuation L‑BFGS** on a soft‑max parametrisation, using a very
  fine geometric schedule of the smoothing parameter λ.  This finds a
  high‑quality basin rapidly.

* **Mirror‑descent on the true max‑convolution** – a sub‑gradient step
  w.r.t. the exact ℓ∞‑norm of the auto‑correlation, performed with a
  multiplicative (KL‑mirror) update and a simple step‑size decay.  This
  drives the peak down directly, without the smoothing bias of the
  L‑BFGS stage.

* **Multi‑peak greedy hill‑climb** – instead of focusing on a single
  maximal index we consider all (near‑maximal) convolution peaks, 
  compute the total contribution of each coordinate to those peaks and
  transfer a small fraction of mass from the worst contributor to the
  best.  This reduces several peaks simultaneously.

* **Random hill‑climb fallback** for polishing flat spots.

The algorithm runs many restarts (different initialisations) within the
allocated time budget (default 1000 s) and keeps the best sequence seen.
All vectors are kept non‑negative and normalised to sum = 1, which is
compatible with the evaluation routine `evaluate_sequence`.

The implementation is pure NumPy/ SciPy and respects the 2‑CPU limit.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns the full convolution of length ``2*n‑1``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector ``a``.
    This matches the value returned by ``evaluate_sequence`` (the denominator
    equals 1 after normalisation).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smooth max objective (used for continuation L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of the max‑convolution and its gradient w.r.t.
    the unconstrained logits ``y``.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max
    w = exp_shift / Z  # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: run a sequence of optimisations with decreasing
    smoothing λ.  Returns the final logits.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y

# Hill‑climb utilities

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Simple random hill‑climb on the simplex.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves mass from the largest contributor at the
    currently maximal convolution index to the smallest contributor.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


def multi_peak_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that simultaneously considers all (near‑maximal)
    convolution peaks.  The contribution of each coordinate to the set of
    peaks is summed, and mass is transferred from the worst to the best
    contributor.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    eps = 1e-14  # tolerance for “near‑max”

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        max_b = conv.max()
        peak_idxs = np.where(conv >= max_b - eps)[0]

        # total contribution of each coordinate to the selected peaks
        contrib = np.zeros_like(best)
        for k in peak_idxs:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            contrib[idx] += best[idx] * best[k - idx]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random transfer
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Mirror‑descent on the true max‑convolution
# ----------------------------------------------------------------------


def subgradient_mirror_descent(
    a: np.ndarray,
    max_iters: int,
    step0: float,
    step_decay: float,
    deadline: float,
    fft_len: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (KL‑mirror) sub‑gradient descent for the exact
    ``max(a*a)`` objective.  The update is
        a_i ← a_i·exp(−η·g_i) ,   g_i = Σ_{k∈K} 2·a_{k−i},
    where K is the set of maximal convolution indices.
    After each step the vector is re‑normalised to sum = 1.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    eta = step0

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = conv_fft(best, fft_len, n)
        max_b = conv.max()
        # indices attaining the maximum (within tiny tolerance)
        K = np.where(conv >= max_b - 1e-14)[0]

        # sub‑gradient for each coordinate
        g = np.zeros_like(best)
        for k in K:
            lo = max(0, k - n + 1)
            hi = min(k, n - 1)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * best[k - idx]

        # multiplicative update with clipping to avoid overflow
        exp_term = np.exp(np.clip(-eta * g, -50.0, 50.0))
        best = best * exp_term
        s = best.sum()
        if s <= 0.0:
            best = np.full(n, 1.0 / n)
        else:
            best = best / s

        cur_score = fast_score(best, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score

        # decay step‑size
        eta = step0 * (step_decay ** (it + 1))

    return best, best_score

            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            w_frac = rng.uniform(0.01, 0.04)
            m_edge = max(1, int(n * w_frac))
            ratio = rng.uniform(5.0, 50.0)      # edge‑to‑interior ratio
            interior = 1.0
            edge = interior * ratio
            a = np.full(n, interior, dtype=float)
            a[:m_edge] = edge
            a[-m_edge:] = edge
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- mirror‑descent on true max -----
        a_md, score_md = subgradient_mirror_descent(
            best_a,
            max_iters=4000,
            step0=0.9,
            step_decay=0.9995,
            deadline=deadline,
            fft_len=fft_len_conv,
            n=n,
        )
        if score_md < best_score:
            best_score = score_md
            best_a = a_md.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[{iteration}] mirror‑descent improved fast score = {best_score:.6f}")

        # ----- targeted hill‑climb (single peak) -----
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=3000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        # ----- multi‑peak hill‑climb (simultaneous) -----
        best_a, best_score = multi_peak_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=3000,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final deep refinement passes
    # --------------------------------------------------------------
    # A short final L‑BFGS pass with an ultra‑fine λ schedule
    for lam in np.geomspace(1e-5, 1e-9, num=15):
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing (mirror‑descent + random hill‑climb)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = subgradient_mirror_descent(
            best_a,
            max_iters=12000,
            step0=0.5,
            step_decay=0.9999,
            deadline=deadline,
            fft_len=fft_len_conv,
            n=n,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[final‑mirror] fast score = {best_score:.6f}")

    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[final‑random] fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
