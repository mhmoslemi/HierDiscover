def adam_optimize(y_init, n, fft_len_conv, fft_len_g, lambda_end, deadline, alpha=0.05, beta1=0.9, beta2=0.999, eps=1e-8):

    y = y_init.copy()

    m = np.zeros_like(y)

    v = np.zeros_like(y)

    t = 0

    while time.time() < deadline:

        t +=1

        obj, grad = objective_and_grad(y, n, fft_len_conv, fft_len_g, lambda_end)

        m = beta1 * m + (1 - beta1) * grad

        v = beta2 * v + (1 - beta2) * (grad**2)

        m_hat = m / (1 - beta1**t)

        v_hat = v / (1 - beta2**t)

        y = y - alpha * m_hat / (np.sqrt(v_hat) + eps)

    return y

"""
Improved step‑function optimizer.

The optimiser works in several stages:

1.  A cheap parametric sweep (uniform, linear, exponential, power‑law,
    edge‑heavy, sinusoidal, alternating‑high/low, random Dirichlet) to
    collect a few promising seeds.

2.  Continuation L‑BFGS on a smoothed max‑convolution (log‑sum‑exp) with
    a decreasing smoothing factor.  Many restarts (≈30) are performed
    and each λ‑stage is given a generous iteration budget.

3.  Projected Adam on the *exact* max‑convolution (sub‑gradient) for the
    remaining time.  The optimisation is performed in the log‑softmax
    (exponential‑family) space, guaranteeing feasibility (non‑negative,
    sum = 1).

4.  Simulated‑annealing / random perturbations to escape tiny local minima.

5.  A final hill‑climb (greedy mass‑transfer) for polishing.

All stages respect the simplex constraint and use FFT‑based convolutions
(O(n log n)).  The routine returns a plain Python list that can be fed
to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalized
    to sum = 1.  The true score is 2·n·max(a*a) / sum(a)², but the ratio
    is scale‑invariant, so we only need max(a*a).
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c               # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                         # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing λ schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage.  Returns the best sequence found.
    """
    y = np.log(a_start + 1e-12)               # initialise soft‑max arguments
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Occasionally L‑BFGS may fail – ignore and continue.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] λ={lam:.5f} new best score={best_score:.6f}")

        # short hill‑climb after each λ to sharpen
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1200)

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Adam on the exact max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------
def subgradient_adam_optimize(a0: np.ndarray,
                              deadline: float,
                              fft_len_conv: int,
                              fft_len_g: int,
                              n: int,
                              alpha: float = 0.02,
                              beta1: float = 0.9,
                              beta2: float = 0.999,
                              eps: float = 1e-8):
    """
    Adam optimiser on the exact max‑convolution.
    The optimisation variable is y = log(a) (soft‑max space), guaranteeing
    feasibility.  Returns the best a seen together with its fast_score.
    """
    y = np.log(a0 + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        it += 1
        a = softmax(y)

        # convolution and mask for active (max) positions
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_conv = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_conv).astype(np.float64)

        # sub‑gradient of the (exact) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # gradient w.r.t. y (chain rule through soft‑max)
        mean_g = np.dot(a, g)
        grad_y = a * (g - mean_g)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        y = y - alpha * m_hat / (np.sqrt(v_hat) + eps)

        # periodic evaluation (every 30 steps)
        if it % 30 == 0:
            cur_score = fast_score(a, fft_len_conv, n)
            if cur_score < best_score:
                best_score, best_a = cur_score, a.copy()
                print(f"[adam] iter={it} new best score={best_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000):
    """
    Greedy mass‑transfer hill‑climb:
    pick two coordinates, move a small random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing + random perturbations
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        # exponential cooling every few thousand iterations
        if it % 5000 == 0:
            T = max(T * 0.92, 1e-5)

    return best_a, best_score
