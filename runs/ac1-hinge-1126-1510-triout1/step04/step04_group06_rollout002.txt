def solve_lp_for_g(f, M):
    n=len(f)
    K = 2*n-1
    rows=[]
    cols=[]
    data=[]
    for i in range(n):
        fi = f[i]
        # row = i + j, column=j for j in range(n)
        # we can fill entire slice
        for j in range(n):
            rows.append(i+j)
            cols.append(j)
            data.append(fi)
    A = csr_matrix((data, (rows, cols)), shape=(K, n))
    b = np.full(K, M)
    c = -np.ones(n) # minimize negative sum = maximize sum
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        raise RuntimeError('LP failed')

"""
Propose a step‑function with a low evaluate_sequence() value.

Algorithm
---------
1. Initialise a probability vector `f` (sum = 1).  Use the previous
   construction if available, otherwise uniform.
2. Repeatedly perform the “linear‑program (LP) update” described in the
   problem statement:
        * Compute M = max(conv(f,f)).
        * Solve   maximise   Σ g_j
                  subject to  g ≥ 0  and  conv(f,g) ≤ M  (pointwise)
          with a sparse linear program (SciPy/HiGHS).  The convolution
          constraints are linear in `g`; they are stored as a Toeplitz
          matrix built from `f`.
        * Rescale `g` to keep Σ g = 1 and replace `f` by a convex
          combination f←(1‑t)f+t g (t∈(0,1)).  Accept the new `f` only if the
          objective improves; otherwise shrink `t`.
   This iteration provably never increases the max‑convolution and
   converges to a vector with a smaller peak.
3. After the LP phase, polish the best vector with a cheap greedy hill‑climb
   that moves a tiny amount of mass between two positions if it lowers the
   fast score.
4. Return the final vector (list of floats).  The vector is kept on the
   probability simplex, so `evaluate_sequence` receives a sum = 1 and
   the returned score equals `2·n·max(conv)`.
"""

import time
import math
import numpy as np

# optional imports – if SciPy is not available we fall back to a simple uniform
try:
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix
except Exception:  # pragma: no cover
    linprog = None
    csr_matrix = None


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a already normalised (sum=1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Local optimisation that repeatedly moves a tiny fraction of mass
    between two positions if it improves the (fast) score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def build_conv_matrix(f: np.ndarray) -> csr_matrix:
    """
    Build the Toeplitz matrix A such that (A @ g) = conv(f,g) (full length).
    A has shape (2*len(f)-1, len(f)).
    """
    n = f.shape[0]
    # rows = i + j for all i,j
    i = np.arange(n)[:, None]                # shape (n,1)
    j = np.arange(n)                         # shape (n,)
    rows = (i + j).ravel()                   # length n*n
    cols = np.tile(j, n)                     # repeat 0..n-1 n times
    data = np.repeat(f, n)                   # repeat each f_i n times
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def lp_update(f: np.ndarray, M: float, t: float) -> np.ndarray:
    """
    Perform one LP‑based update.
    Returns the new f (already normalised) or the original f if the LP fails.
    """
    n = f.shape[0]
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)            # minimise -sum(g)  == maximise sum(g)
    bounds = [(0, None)] * n

    # Solve the LP with HiGHS (fast interior‑point/simplex)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if not res.success:
        return None
    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return None
    # Rescale to keep Σg = 1 (simplex)
    g = g / sum_g
    # Convex combination
    return (1.0 - t) * f + t * g
