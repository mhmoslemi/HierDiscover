Pseudo:

def proj_simplex(v):
    """Projection of vector v onto probability simplex."""
    # ensure array
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def proj_simplex(v):
    """Projection of v onto the probability simplex {x >=0, sum x = 1}."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive after subtracting; fallback: uniform
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def exponentiated_gradient_descent(a, n, fft_len, max_iters, eta0, deadline):

"""
Propose a low‑score step‑function sequence.

The search is based on three complementary ideas:

1. **Continuation L‑BFGS** – the probability vector `a` is parametrised by
   unconstrained logits `y` (soft‑max).  A smoothed version of the max‑convolution
   is minimised while gradually reducing the smoothing parameter.  This quickly
   finds a good basin of attraction.

2. **Exponentiated‑gradient fine‑tuning** – after each L‑BFGS run we perform a
   sub‑gradient descent on the true max‑convolution:
   the gradient of the peak is simple (`2·a[k‑j]`), so we can apply an
   exponentiated‑gradient step (`a ← a·exp(‑η·g)`) and renormalise.
   This directly reduces the true peak without any smooth approximation.

3. **Targeted & random hill‑climbs** – a greedy move that transfers mass from the
   largest contributor to the current peak to the smallest contributor, followed
   by a generic random hill‑climb for polishing.

The algorithm runs many restarts with diverse initialisations,
keeps the best sequence seen, and finishes with a deep refinement phase.
All candidates are kept on the probability simplex (sum = 1, non‑negative),
so the returned list can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic linear‑convolution utilities (FFT based)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns the first ``2*n‑1`` entries (the full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector ``a``.
    This equals the value returned by ``evaluate_sequence`` when ``sum(a)=1``.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Soft‑max parametrisation + smoothed max objective (for L‑BFGS)
# ----------------------------------------------------------------------


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution and its gradient w.r.t.
    the unconstrained logits ``y`` (the log‑softmax coordinates).

    The smoothing is performed by a log‑sum‑exp with a width proportional to
    ``lam_factor`` times the current maximum of the convolution.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a continuation of L‑BFGS optimisations.
    ``lam_factors`` is a decreasing list that gradually reduces the smoothing.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Simplex projection (not used directly but kept for completeness)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection of vector ``v`` onto the probability simplex
    {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full(n, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exponentiated_gradient_descent(
    a: np.ndarray,
    n: int,
    fft_len: int,
    max_iters: int,
    eta0: float,
    deadline: float,
) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution using an
    exponentiated‑gradient (mirror‑descent) step.
    """
    a = a.copy()
    for t in range(max_iters):
        if time.time() > deadline:
            break
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # gradient of c_k = a * a at index k:
        #   ∂c_k/∂a_j = 2 * a_{k-j}  (if 0 ≤ k‑j < n, otherwise 0)
        g = np.zeros_like(a)
        low = max(0, k - (n - 1))
        high = min(n - 1, k)
        if low <= high:
            idx = np.arange(low, high + 1)
            g[idx] = 2.0 * a[k - idx]

        # decreasing step size
        eta = eta0 / (1.0 + t * 0.01)

        # exponentiated update, renormalise to keep a on the simplex
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Hill‑climbing utilities (same as the previous version)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Random hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """Greedy move based on the contributors to the current peak."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions of each coordinate to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score

# Main entry point used by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the evaluation score.  Returns the best sequence found within the time
    budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – aim for a modest but effective length
    # --------------------------------------------------------------
    default_n = 2000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            n = default_n
            a0 = arr[:n].copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 1e-12:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    best_a = a0
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    fft_len_conv = fft_len
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Continuation L‑BFGS hyper‑parameters
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.8, 5e-7, num=40)   # finer schedule
    maxiter_stage = 600                           # more iterations per stage

    iteration = 0

    # --------------------------------------------------------------
    # Multi‑restart optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            edge_val = np.sqrt(n / 2.0) / n
            centre_val = 1.0 / np.sqrt(2.0) / n
            a = np.full(n, centre_val)
            a[:m_edge] = edge_val
            a[-m_edge:] = edge_val
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} L‑BFGS improved fast score = {best_score:.6f}")

        # ----- exponentiated‑gradient fine‑tuning (quick) -----
        eg_deadline = min(deadline, time.time() + 0.7)
        best_a = exponentiated_gradient_descent(
            best_a,
            n,
            fft_len_conv,
            max_iters=2000,
            eta0=0.8,
            deadline=eg_deadline,
        )
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)

        # ----- targeted hill‑climb ---------------------------------
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=3000,
            frac=0.08,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final deep refinement (another continuation pass)
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.4g} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final exponentiated‑gradient + hill‑climb polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        # long exponentiated‑gradient run
        best_a = exponentiated_gradient_descent(
            best_a,
            n,
            fft_len_conv,
            max_iters=8000,
            eta0=0.5,
            deadline=deadline,
        )
        best_score = fast_score(best_a, fft_len_conv, n)

    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=6000,
            frac=0.06,
        )
        # safety random hill‑climb in case of flat spots
        if time.time() < deadline:
            best_a, best_score = hill_climb(
                best_a,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=5000,
            )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
