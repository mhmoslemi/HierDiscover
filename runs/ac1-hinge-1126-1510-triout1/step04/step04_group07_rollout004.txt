"""
Propose a step‑function with a low value of evaluate_sequence.

Algorithm
---------

1. **Initial pool** – generate a handful of diverse shapes
   (two‑sided power‑law, pure power‑law, exponential, beta‑pdf)
   of moderate length (≈8k‑12k).  If a previous best sequence
   `height_sequence_1` is present it is added to the pool.

2. **Smooth Adam optimisation** – the coefficients are represented by
   a soft‑max `a = softmax(y)`, guaranteeing non‑negativity and a
   fixed ℓ₁‑norm (∑a = 1).  Instead of the non‑smooth max‑convolution
   we minimise a smooth surrogate

        φ_α(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

   whose gradient is dense:
        ∂φ/∂a = 2·(w ⋆ a) ,
   where `w` is the soft‑max of the convolution entries and `⋆`
   denotes cross‑correlation.  All convolutions are performed with FFTs,
   giving an O(n log n) per‑iteration cost.

   An Adam optimiser (β₁=0.9, β₂=0.999) updates the logits `y`.  The
   smoothing parameter α is increased during optimisation (1 → 30)
   to first explore a smooth landscape and then focus on the true max.

3. **Time allocation** – the total 1000 s budget is split evenly among
   the candidates; each optimisation runs until its slice of time is
   exhausted.  The best sequence (lowest true score) seen overall is
   kept.

4. **Final refinement** – any remaining time is spent further
   optimising the current global best with the same optimiser.

5. **Verification** – the obtained sequence is finally evaluated with
   the provided `evaluate_sequence` function and returned as a plain
   Python list.

The code uses only NumPy, runs on a single thread (the FFTs are multithreaded
by NumPy) and respects the 2‑CPU restriction.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute the exact evaluate_sequence value (2·n·max(conv)/sum²)
    via an FFT – much faster than the naïve O(n²) version.
    The sequence must be non‑negative and sum > 0.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Shape generators (used for random starts)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(α-1)·(n-i)^(β-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Optimiser (Adam on a smooth max‑convolution surrogate)
# ----------------------------------------------------------------------


def adam_smooth_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    alpha_start: float = 1.0,
    alpha_end: float = 30.0,
    alpha_step_iters: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth surrogate φ_α(a) = (1/α)·log Σ exp(α·(a*a)_k).
    Returns the best sequence (∑a = 1) encountered and its *exact* score
    (2·n·max(conv)/∑a²) as computed by fast_score.
    """
    n = a0.shape[0]

    # Initialise logits (log‑softmax variables) from a0
    y = np.log(a0 + 1e-16)

    # Adam moments
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # Pre‑compute FFT sizes that stay constant
    fft_len_conv = next_pow2(2 * n)                 # for a*a
    len_w = 2 * n - 1                               # length of convolution
    len_a_rev = n
    fft_len_cross = next_pow2(len_w + len_a_rev - 1)  # for w ⋆ a

    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ------------------- soft‑max -------------------
        y_shift = y - np.max(y)          # numeric safety
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()          # ℓ₁‑norm = 1, a_i ≥ 0

        # ------------------- convolution a*a -------------------
        fa = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(fa * fa, n=fft_len_conv)[: 2 * n - 1]

        # ------------------- smooth max (φ) -------------------
        c_max = conv.max()
        # soft‑max weights (stable)
        exp_term = np.exp(alpha_start * (conv - c_max))
        Z = exp_term.sum()
        # initial α (will be increased later)
        phi = c_max + (1.0 / alpha_start) * np.log(Z)

        # ------------------- gradient of φ wrt a -------------------
        w = exp_term / Z                     # w_k = ∂φ/∂c_k
        a_rev = a[::-1]

        # cross‑correlation w ⋆ a via FFT
        fw = np.fft.rfft(w, n=fft_len_cross)
        fra = np.fft.rfft(a_rev, n=fft_len_cross)
        corr_full = np.fft.irfft(fw * fra, n=fft_len_cross)
        start_idx = len_a_rev - 1            # = n‑1
        g = 2.0 * corr_full[start_idx : start_idx + n]   # ∂φ/∂a

        # ------------------- gradient wrt logits -------------------
        ag = np.dot(a, g)                     # scalar ⟨a,g⟩
        grad_y = a * (g - ag)                 # Jᵀ·g for soft‑max Jacobian

        # ------------------- Adam update -------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ------------------- (optional) increase smoothing -------------------
        if it % alpha_step_iters == 0:
            # linear increase of α up to alpha_end
            cur_alpha = min(alpha_end,
                            alpha_start + (alpha_end - alpha_start) *
                            (it // alpha_step_iters) * 0.5)
            # recompute phi and w with the new α on the next loop
            alpha_start = cur_alpha

        # ------------------- jitter to escape plateaus -------------------
        if it % 4000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

        # ------------------- keep best exact score -------------------
        cur_score = fast_score(a)
        if cur_score < best_val:
            best_val = cur_score
            best_seq = a.copy()

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low evaluate_sequence
    value.  The routine respects the 1000 s budget, prints progress,
    and finally returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # 0) Build a pool of initial candidates (including any prior best)
    # --------------------------------------------------------------
    init_candidates = []

    # previous best, if the variable exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            total = prev.sum()
            if total < 1e-12:
                prev = np.full(prev.shape, 1.0 / prev.shape[0])
            else:
                prev = prev / total
            init_candidates.append(prev.copy())
            print("[info] using previously discovered sequence as a start")
    except Exception:
        pass

    # a default robust start (two‑sided power‑law)
    default_len = 10000
    default_seq = generate_two_sided_power_law(default_len, 0.55, 0.55, 1.0, 1.0)
    default_seq = np.maximum(default_seq, 0.0)
    default_seq = default_seq / default_seq.sum()
    init_candidates.append(default_seq.copy())

    # --------------------------------------------------------------
    # 1) Randomised family search to fill the pool (≈8 seeds)
    # --------------------------------------------------------------
    length_choices = [8000, 10000, 12000, 15000]
    families = ["power", "exp", "twosided", "beta"]
    desired_pool = 8

    while len(init_candidates) < desired_pool and time.time() < deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.05, 1.5)
            seq = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(1e-5, 5e-2)
            seq = generate_exponential(n, lam)
        elif family == "twosided":
            pl = random.uniform(0.05, 1.5)
            pr = random.uniform(0.05, 1.5)
            wl = random.uniform(0.2, 5.0)
            wr = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        else:  # beta
            a = random.uniform(0.05, 2.0)
            b = random.uniform(0.05, 2.0)
            seq = generate_beta_shape(n, a, b)

        seq = np.maximum(seq, 0.0)
        total = seq.sum()
        if total < 1e-12:
            continue
        seq = seq / total
        init_candidates.append(seq)

    # --------------------------------------------------------------
    # 2) Optimise each candidate with the smooth Adam optimiser
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left after initialisation")

    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, a0 in enumerate(init_candidates):
        # allocate the *remaining* budget evenly among the still‑to‑run candidates
        now = time.time()
        remaining = deadline - now
        if remaining <= 0:
            break
        candidates_left = len(init_candidates) - idx
        per_candidate = remaining / candidates_left

        print(
            f"[optim] start candidate #{idx + 1}/{len(init_candidates)} "
            f"(len={a0.size}) – time slice {per_candidate:.1f}s"
        )
        seq, val = adam_smooth_optimize(
            a0,
            time_limit=per_candidate,
            start_time=time.time(),
            lr0=0.6,
            alpha_start=1.0,
            alpha_end=30.0,
            alpha_step_iters=2000,
        )
        print(f"[optim] candidate #{idx + 1} best exact score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # --------------------------------------------------------------
    # 3) Final refinement with any leftover time
    # --------------------------------------------------------------
    leftover = deadline - time.time()
    if best_overall_seq is not None and leftover > 5.0:
        print(f"[refine] final refinement using {leftover:.1f}s")
        seq, val = adam_smooth_optimize(
            best_overall_seq,
            time_limit=leftover,
            start_time=time.time(),
            lr0=0.5,
            alpha_start=5.0,
            alpha_end=30.0,
            alpha_step_iters=1500,
        )
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"[refine] improved global best to {best_overall_val:.6f}")

    # --------------------------------------------------------------
    # 4) Final verification with the official evaluator
    # --------------------------------------------------------------
    if best_overall_seq is None:
        # fallback: uniform distribution of length 8000
        n = 8000
        best_overall_seq = np.full(n, 1.0 / n)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")

    return final_list
