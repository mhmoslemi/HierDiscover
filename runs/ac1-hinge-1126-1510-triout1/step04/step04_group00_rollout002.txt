"""
Propose a low‑score step‑function sequence.

The search proceeds in three stages:

1. **Global exploration** – a few multi‑restart L‑BFGS optimisations of a
   smooth approximation of the max‑convolution.  A geometric λ‑schedule
   (λ small → true max) is used.  Different random initialisations
   (exponential, power‑law, edge‑biased, perturbation of the current best)
   diversify the search.

2. **Projected sub‑gradient descent on the true max** – after the L‑BFGS
   stage we minimise the exact objective 2 n · max(a∗a) using a
   sub‑gradient (obtained from the active peak positions) together with
   projection onto the probability simplex.  The step size decays as
   1/√t.

3. **Aggressive hill‑climb** – a final random‑mass‑transfer hill‑climb
   (up to 20 % of the source mass per move) finishes the optimisation
   using any remaining time budget.

All intermediate candidates are kept on the simplex (a_i ≥ 0,
∑a_i = 1).  The routine respects the 1000 s budget, prints progress
information and stores the best sequence in the global variable
`height_sequence_1` for possible reuse by later runs.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (∑a = 1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho_arr[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (log‑soft‑max coordinates).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # convolution a with w (only the part aligned with a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
    time_deadline: float = None,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    Stops early if the global time budget is exhausted.
    """
    y = y_start.copy()
    for lam in lam_factors:
        if time_deadline is not None and time.time() > time_deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb (random mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.  Randomly moves a fraction of the
    mass from one index to another and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to max_move_frac of the source mass
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:   # strict improvement
            best_score = new_score
            # keep the move
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max (λ = 0)
# ----------------------------------------------------------------------


def projected_subgradient_descent(
    a: np.ndarray,
    n: int,
    fft_len: int,
    init_score: float,
    deadline: float,
    step0: float = 0.2,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on 2·n·max(a∗a) with projection onto the simplex.
    """
    best = a.copy()
    best_score = init_score
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = np.convolve(best, best)
        max_val = conv.max()
        # mask of active peak positions (could be multiple)
        mask = (conv == max_val).astype(float)

        # gradient of max(C) w.r.t. a: 2 * (a * mask) centred on a
        grad = 2.0 * np.convolve(best, mask)[n - 1 : 2 * n - 1]

        # diminishing step size
        step = step0 / np.sqrt(it)

        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
        # else keep the current point
    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – reuse a previously found construction if available
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = rng.random(n)
    else:
        n = 5000
        best_a = rng.random(n)

    # Normalise to the probability simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Phase 0 – Global search with L‑BFGS (few restarts)
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-6, num=30)   # λ‑schedule
    maxiter_stage = 500                            # iterations per λ‑stage
    it = 0

    while time.time() < deadline - 10.0:
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.015)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.2, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a0 = np.full(n, 1.0 / n)
            edge_val = min(0.5, a0[:m_edge].mean() * 3.0)
            a0[:m_edge] = edge_val
            a0[-m_edge:] = edge_val
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturb current best
            y0 = best_y + rng.normal(scale=0.04, size=n)

        # ----- run continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
            time_deadline=deadline - 5.0,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{it}] mode={mode} improved fast score = {best_score:.6f}")

        # ----- aggressive local hill‑climb (short burst) -----
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=3000,
            max_move_frac=0.15,
        )
        best_y = np.log(best_a + 1e-12)

        it += 1

    # --------------------------------------------------------------
    # Phase 1 – Projected sub‑gradient descent on the exact max
    # --------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.3)
    best_a, best_score = projected_subgradient_descent(
        a=best_a,
        n=n,
        fft_len=fft_len_conv,
        init_score=best_score,
        deadline=sub_deadline,
        step0=0.25,
        max_iter=2500,
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] best score after sub‑gradient = {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Final aggressive hill‑climb using remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=12000,
            max_move_frac=0.10,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    # store for possible reuse by later runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
