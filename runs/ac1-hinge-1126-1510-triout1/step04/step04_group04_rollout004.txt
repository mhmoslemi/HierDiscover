   L‑BFGS iterations.  This yields a very accurate point close to a local
   optimum of the true max‑convolution.

3. **Stochastic restarts** – dozens of random initialisations (uniform,
   power‑law, exponential and Dirichlet draws) are tried; the best result
   is kept.

4. **Exact sub‑gradient descent** – after the smooth optimisation we
   perform a fast sub‑gradient method on the *exact* max‑convolution.
   The gradient is the sum of the contributing coefficients at the
   active (maximising) index.  The iterate is projected onto the probability
   simplex after every step.

5. **Aggressive hill‑climbing** – a greedy mass‑transfer move (move a
   fraction of the probability from one entry to another) is repeatedly
   applied while it strictly improves the fast‑score.  This is cheap and
   cleans up any remaining irregularities.

6. **Final evaluation** – the best vector is returned as a plain list of
   non‑negative floats.  `evaluate_sequence` will scale it automatically,
   but we keep the sum close to 1 to avoid the “sum too close to zero”
   guard.

The whole pipeline is time‑aware; it stops well before the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions – all top level, no closures or lambdas
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax; returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT, returning the first 2·n‑1 entries."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.001,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max objective (2·n·smooth_max) and its gradient w.r.t. y.
    ``lam_factor`` controls the smoothness: smaller → closer to true max.
    """
    a = softmax(y)

    # ----- convolution a*a (FFT) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12  # smoothing parameter
    shifted = conv - max_c               # max becomes zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. a -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # convolution of a_rev with w gives d conv / d a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through the soft‑max -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 2000,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "dirichlet":
        # very sparse Dirichlet draws
        alpha = rng.uniform(0.05, 0.3, size=n)
        a0 = rng.gamma(shape=alpha, scale=1.0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: transfer a small random fraction of mass from i to j.
    Keep the move only if the fast score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.05  # up to 5 % of mass at i
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-9,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution,
    projected onto the probability simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        # gradient of the (single) active convolution entry
        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # small jitter to escape flat plateaus
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad

        # project onto the simplex (see Chen & Ye 2011)
        # -------------------------------------------------
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / (np.arange(len(u)) + 1) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        a_new = np.maximum(a_new - theta, 0.0)
        # -------------------------------------------------

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.3, 2.0)  # accelerate when successful
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  Returns the sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # Choose problem size – try to improve on the previously stored
    # construction if one is available, otherwise start with n=8000.
    # --------------------------------------------------------------
    n = 8000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 500:
            # use the longer of the two vectors
            n = max(n, int(tmp.size))
            best_a = tmp.copy()
            best_a = np.clip(best_a, 0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    # Ensure a valid probability vector
    best_a = np.maximum(best_a, 0.0)
    best_a /= best_a.sum()
    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS cascades – many random restarts with a deep schedule
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.4, 0.0005, num=26))
    maxiter_stage = 2000
    modes = ["random", "exp", "power", "dirichlet", "edge", "best"]
    iteration = 0

    while time.time() < deadline - 150:  # leave time for refinements
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # local hill‑climb after each L‑BFGS run (cheap and often helpful)
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=5_000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Sub‑gradient optimisation on the exact max‑convolution
    # --------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 100.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.6
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgradient] improved fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final fine‑grained L‑BFGS passes (very low smoothing)
    # --------------------------------------------------------------
    fine_lam_schedule = list(np.geomspace(0.2, 2e-6, num=30))
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 1000, "gtol": 1e-13, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.2e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Aggressive hill‑climb using any remaining time
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=30_000,
        )
        print(f"[final hill‑climb] fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation (the true black‑box objective)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
