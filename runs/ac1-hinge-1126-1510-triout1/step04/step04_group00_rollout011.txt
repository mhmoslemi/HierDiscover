def mirror_descent_optimize(a_start, n, fft_len_conv, fft_len_g, time_deadline, max_iter=20000, lam_coef=0.3, eta0=1.0):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    for it in range(max_iter):
        if time.time() > time_deadline:
            break
        # compute conv
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
        max_c = conv.max()
        # smoothing lambda: use relative factor
        lam = max_c * lam_coef + 1e-12

        shifted = conv - max_c
        exp_shift = np.exp(shifted / lam)
        Z = exp_shift.sum()
        w = exp_shift / Z

        # gradient w.r.t a:
        # gradient of smoothed max (not multiplied by 2n)
        # compute g = 2 * convolution of a with w reversed.
        fa_rev = np.fft.fft(a[::-1], fft_len_g)
        fw = np.fft.fft(w, fft_len_g)
        conv_rev = np.real(np.fft.ifft(fa_rev * fw))
        grad = 2.0 * conv_rev[(n-1):(2*n-1)]

        # Mirror descent step via exponentiated gradient
        eta = eta0 / np.sqrt(it+1)
        # update in log domain to avoid underflow:
        a = a * np.exp(-eta * grad)
        sum_a = a.sum()
        if sum_a == 0 or np.isnan(sum_a):
            a = np.full(n, 1.0/n)
        else:
            a = a / sum_a

        # evaluate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # possible print
            #print(f"[md] it={it} best score {best_score:.6f}")
    return best_a, best_score

"""
Propose a low‑score step‑function sequence.

The search proceeds in three phases:

1. **Global exploration** – a few multi‑restart L‑BFGS optimisations on the
   smooth (log‑sum‑exp) approximation to the max‑convolution.  The soft‑max
   parametrisation guarantees non‑negativity and the simplex constraint.
2. **Fine‑grained refinement** – projected (exponentiated) gradient descent on
   the same smooth objective with a slowly decreasing temperature.  This
   phase gets the remaining 0.4‒0.5 % of the budget and is able to push the
   score below the previous best.
3. **Aggressive hill‑climb** – a simple random mass‑transfer hill‑climb on the
   simplex that polishes the final solution.

All helper functions are top‑level (no closures) and the code respects the
1000 s time budget.
"""

import time
import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (∑a = 1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth max objective (log‑sum‑exp) and its gradient w.r.t. log‑soft‑max
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed maximum of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (log‑soft‑max coordinates).
    """
    a = softmax(y)

    # ---- convolution a*a ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c                       # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max                     # same scaling as fast_score

    # ---- gradient of smooth max ----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max (gradient w.r.t. y)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 500,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Gradient of the smooth max w.r.t. the probability vector (no log‑soft‑max)
# ----------------------------------------------------------------------


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns a vector of length n (no outer 2·n scaling).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # d(smooth_max)/da = 2 * g
    return 2.0 * g


# ----------------------------------------------------------------------
# Simple greedy hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that randomly transfers a fraction of the mass from
    one index to another and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    The algorithm follows the three‑phase strategy described in the module
    docstring.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – reuse a previous construction if available
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= 1000:
            n = int(arr.size)
            best_a = arr.copy()
        else:
            n = 5000
            best_a = np.full(n, 1.0 / n)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to probability simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Phase 1: Global exploration with L‑BFGS (≈30 % of the budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + budget_s * 0.30
    lam_factors = np.geomspace(0.5, 1e-5, num=25)   # decreasing smoothing
    maxiter_stage = 500

    iteration = 0
    while time.time() < phase1_deadline:
        # pick a diverse starting point
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
            a_edge = np.sqrt(n / 2.0) / n
            a_center = 1.0 / np.sqrt(2.0) / n
            a0 = np.full(n, a_center)
            a0[:m_edge] = a_edge
            a0[-m_edge:] = a_edge
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.05, size=n)

        # ----- L‑BFGS continuation -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}"
            )

        # ----- short hill‑climb to polish the run -----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=3000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2: Projected (exponentiated) gradient descent on smooth max
    # ------------------------------------------------------------------
    pg_deadline = start_time + budget_s * 0.85  # leave ~15% for final polish
    lam_factor_pg = 0.30          # start relatively smooth
    step0 = 0.60                  # initial step size for exponentiated update
    pg_iter = 0

    while time.time() < pg_deadline and pg_iter < 5000:
        pg_iter += 1
        # gradient of the smooth max (no log‑soft‑max)
        grad = grad_smooth_max(best_a, lam_factor_pg)

        # exponentiated (mirror) descent step
        eta = step0 / np.sqrt(pg_iter)
        a_candidate = best_a * np.exp(-eta * grad)
        # projection onto simplex
        a_candidate_sum = a_candidate.sum()
        if a_candidate_sum == 0 or np.isnan(a_candidate_sum):
            a_candidate = np.full(n, 1.0 / n)
        else:
            a_candidate = a_candidate / a_candidate_sum

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = a_candidate
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            # tighten smoothing when we make progress
            lam_factor_pg = max(lam_factor_pg * 0.9, 1e-6)

    print(f"[pg] iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3: Final aggressive hill‑climb using any remaining time
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
