"""
Step‑function optimizer for the “max‑auto‑convolution / (sum²)” score.

The algorithm combines several complementary heuristics:

1. **Multi‑restart smoothed‑max L‑BFGS** – the max‑operator is
   approximated by a log‑sum‑exp with a geometric λ‑schedule.  Each
   run starts from a diverse family of initial shapes (random,
   exponential, power‑law, edge‑heavy, centre‑heavy, etc.).  The best
   solution found is kept.

2. **Local greedy hill‑climbing** – random transfers of a fraction of
   mass between two positions are accepted only if they strictly
   improve the fast score (computed via FFT).

3. **Symmetrisation** – the current best vector is averaged with its
   reverse; this often reduces the peak of the convolution.

4. **Projected sub‑gradient descent** – using the exact arg‑max of the
   convolution we form a sub‑gradient of the max‑convolution and take
   diminishing projected‑gradient steps on the simplex.

5. **Final polishing hill‑climb** – leftover time is spent on a last
   greedy hill‑climb.

All operations respect the simplex constraint (non‑negative,
∑aᵢ = 1).  The routine runs for at most ``budget_s`` seconds (default
1000 s) and returns the best sequence it discovered.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that produces a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the score for a probability vector ``a`` (∑a = 1).

    Returns 2·n·max(a*a) which equals the value produced by
    ``evaluate_sequence`` for the same normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used inside L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑soft‑max parameters ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smoothed max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain‑rule through the soft‑max (gradient w.r.t. y)
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisation over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Greedy hill‑climb on the simplex
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of the mass from one index to another.
    Keep the move only if it yields a strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def compute_full_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * len(a) - 1]
    return conv


def subgradient_descent(
    a: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    max_iter: int = 2000,
    init_step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)
    step0 = init_step
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        conv = compute_full_conv_fft(best, fft_len_conv)
        M = conv.max()
        p = conv.argmax()

        # gradient of the max‑convolution w.r.t. a (sub‑gradient)
        grad = np.zeros_like(best)
        lo = max(0, p - (n - 1))
        hi = min(n - 1, p)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * best[p - idx]

        # diminishing step size
        step = step0 / np.sqrt(it)
        cand = proj_simplex(best - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)

        if cand_score < best_score - 1e-12:
            best = cand
            best_score = cand_score
            # optionally shrink the step size more aggressively after a good move
            step0 *= 0.97
    return best, best_score

        elif mode == "midrect":
            width = rng.uniform(0.05, 0.30)
            m_mid = max(1, int(n * width))
            a0 = np.zeros(n, dtype=float)
            start = (n - m_mid) // 2
            a0[start : start + m_mid] = 1.0
            a0 /= a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.06, size=n)

        # ---- continuation L‑BFGS ----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(
                f"[{iteration}] mode={mode:9s} improved fast score = {best_score:.6f}"
            )

        # ---- short greedy hill‑climb ----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=6000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Symmetrisation – many good solutions are close to symmetric.
    # ------------------------------------------------------------------
    sym_a = (best_a + best_a[::-1]) * 0.5
    sym_a = sym_a / sym_a.sum()
    sym_score = fast_score(sym_a, fft_len_conv, n)
    if sym_score < best_score:
        best_a = sym_a
        best_score = sym_score
        best_y = np.log(best_a + 1e-12)
        print(f"[sym] symmetrised score improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Projected sub‑gradient descent on the exact max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = min(deadline, time.time() + 12.0)
        best_a, best_score = subgradient_descent(
            a=best_a,
            n=n,
            fft_len_conv=fft_len_conv,
            deadline=sub_deadline,
            max_iter=2500,
            init_step=0.6,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] after sub‑gradient descent score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (use any remaining time).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # ------------------------------------------------------------------
    # Official evaluation and return.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
