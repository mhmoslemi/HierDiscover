"""
Simple analytic construction for a low‑value step function.

The continuous density p(x)=2·(1−x) on [0,1] has a self‑convolution
with supremum 2/3, which yields the asymptotic evaluation value

    2·max(conv) = 2·(2/3) = 4/3 ≈ 1.3333.

Discretising this density on a grid of length *n* (we use n≈2000 or
the length of a previously supplied sequence) gives a sequence
`a_i ∝ (n‑i)`.  Because the evaluation formula is homogeneous
the absolute scale does not matter – we normalise the sum to 1
so that the result is comparable with `evaluate_sequence`.

Optionally a very cheap random hill‑climb is performed for a few
seconds to polish the result without breaking the time budget.
The final sequence is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(arr: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast computation of the evaluation numerator:
        2·n·max(a*a)   (the denominator Σa_i² cancels after normalisation).

    Parameters
    ----------
    arr : ndarray
        The candidate probability vector (already normalised, sum=1).
    fft_len : int
        Length of the FFT (a power of two ≥ 2·n−1).
    n : int
        Length of the vector.

    Returns
    -------
    float
        The quantity 2·n·max(conv) (the denominator of ``evaluate_sequence``
        is 1 because ``arr`` is normalised).
    """
    fa = np.fft.fft(arr, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Build a linearly decreasing step‑function (the discretised version of
    p(x)=2·(1−x) on [0,1]), optionally polish it with a tiny random hill‑climb,
    and return the best sequence found before the time budget expires.

    Returns
    -------
    list[float]
        The candidate height sequence.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # keep a small safety margin

    # --------------------------------------------------------------
    # Choose the length of the sequence.
    # Prefer the length of a previously‑found solution if it exists,
    # otherwise fall back to 2000 (a comfortable size for FFT).
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 100:
        n = len(prev)
    else:
        n = 2000

    # --------------------------------------------------------------
    # Analytic construction: a_i ∝ (n - i)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=float)
    raw = n - idx                # decreasing from n to 1
    a = raw / raw.sum()          # normalise to Σa_i = 1

    # --------------------------------------------------------------
    # Very light polishing: random moves that strictly improve the fast
    # score.  Because the fast score is exactly the numerator of the
    # official metric (the denominator is 1), any improvement is
    # guaranteed to improve the final value.
    # --------------------------------------------------------------
    fft_len = next_pow2(2 * n - 1)
    best = a.copy()
    best_score = fast_score(best, fft_len, n)

    # simple random hill‑climb – stops when time runs out
    max_move_frac = 0.05
    while time.time() < deadline:
        i_src, i_dst = rng.integers(0, n, size=2)
        if i_src == i_dst:
            continue
        delta = best[i_src] * max_move_frac * rng.random()
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i_src] -= delta
        cand[i_dst] += delta
        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score

    # --------------------------------------------------------------
    # Final evaluation (the official wrapper will clamp/normalise as
    # required).  Print the achieved value for logging.
    # --------------------------------------------------------------
    final_seq = best.tolist()
    final_val = evaluate_sequence(final_seq)  # the provided function
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
