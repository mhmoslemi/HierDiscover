def smooth_max_and_grad(a, lam_factor):
    n = a.shape[0]
    fft_len = next_pow2(2*n-1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z   # weights for each convolution term
    # compute g: derivative of smooth max wrt a
    # (a * w convolve)
    fft_len_g = next_pow2(3*n-2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n-1):(2*n-1)]  # length n
    grad = 2 * g
    smooth_max_val = max_c + lam * np.log(Z)
    return smooth_max_val, grad, max_c

"""
Propose a low‑score step‑function sequence.

The algorithm consists of three stages:

1. **Initialisation** – obtain a starting sequence (previous best,
   a uniform distribution, an exponential, an “edge‑heavy’’ pattern or a
   random vector).  All candidates are normalised to the probability
   simplex (∑aᵢ = 1).

2. **Smoothed‑max sub‑gradient descent** – minimise the maximum value of
   the auto‑convolution  a∗a  directly on the simplex.
   The non‑smooth max is replaced by a smooth soft‑max with a decreasing
   smoothing factor λ (λ = α·max) so that the gradient is well‑defined.
   At each iteration a projected step with a diminishing step‑size is
   performed.  The descent is run for as many iterations as the time
   budget allows (≈ 30 000 – 50 000 iterations for n≈5000).

3. **Aggressive hill‑climb** – a simple stochastic mass‑transfer hill‑climb
   (up to 20 % of the source mass per move) cleans up the remaining
   imperfections.

The whole search runs under the supplied time budget (default 1000 s) and
returns the best sequence found.  The code is fully self‑contained; the
provided ``evaluate_sequence`` function is called only once at the end.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (∑a = 1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def smooth_max_grad(
    a: np.ndarray,
    lam_factor: float,
    fft_len_conv: int,
    fft_len_g: int,
) -> tuple[float, np.ndarray, float]:
    """
    Compute a smooth approximation of max(a*a) and its gradient w.r.t. a.
    Returns (smooth_max, gradient, max_c) where gradient is the derivative
    of the smooth max (not yet multiplied by the outer factor 2·n).
    """
    n = a.shape[0]

    # ----- a*a (auto‑convolution) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_c = conv.max()

    # smoothing parameter λ = lam_factor · max_c
    lam = max_c * lam_factor + 1e-12

    # soft‑max weights
    shifted = conv - max_c            # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # ----- gradient of smooth max w.r.t. a -----
    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (2 * n - 1)]

    grad = 2.0 * g_vec                # derivative of max(a*a) w.r.t. a
    smooth_val = max_c + lam * np.log(Z)

    return float(smooth_val), grad, max_c


def subgradient_descent(
    a_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    deadline: float,
    rng: np.random.Generator,
    max_iter: int = 30000,
    lam_start: float = 0.15,
    lam_end: float = 1e-7,
    step0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Smoothed‑max projected sub‑gradient descent on the simplex.
    The smoothing factor λ is decreased geometrically from ``lam_start``
    to ``lam_end``.  The step size follows a 1/√t schedule with back‑track.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    if max_iter > 1:
        lam_decay = (lam_end / lam_start) ** (1.0 / (max_iter - 1))
    else:
        lam_decay = 1.0

    lam = lam_start
    safety = 1.0  # keep a second margin for the final evaluation

    for it in range(1, max_iter + 1):
        # time‑check
        if time.time() + safety >= deadline:
            break

        # smooth max and gradient
        _, grad, _ = smooth_max_grad(a, lam, fft_len_conv, fft_len_g)

        # step size (1/√t) – we also normalise by the gradient norm to keep
        # the absolute movement moderate.
        step = step0 / sqrt(it)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            step = step / grad_norm

        # simple back‑tracking line search (max 5 reductions)
        for _ in range(5):
            a_new = proj_simplex(a - step * grad)
            new_score = fast_score(a_new, fft_len_conv, n)
            if new_score < best_score - 1e-13:   # strict improvement
                a = a_new
                best_a = a_new.copy()
                best_score = new_score
                break
            else:
                step *= 0.5

        # decrease λ
        lam = max(lam * lam_decay, lam_end)

    return best_a, best_score


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 8000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy stochastic hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:   # keep only strict improvements
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta
    return best, best_score

            a0 /= a0.sum()
        else:   # random
            a0 = rng.random(n)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

        # ----- smoothed‑max descent -----
        a_opt, score_opt = subgradient_descent(
            a_start=a0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            deadline=deadline,
            rng=rng,
            max_iter=20000,
            lam_start=0.2,
            lam_end=1e-7,
            step0=0.8,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[init-{mode}] improved fast score = {best_score:.6f}")

        # ----- brief hill‑climb to polish -----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=4000,
            max_move_frac=0.20,
        )

    # ------------------------------------------------------------------
    # Stage 2 – long focused projected sub‑gradient descent on the best
    # ------------------------------------------------------------------
    if time.time() + safety < deadline:
        remaining_iters = 40000
        best_a, best_score = subgradient_descent(
            a_start=best_a,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            deadline=deadline,
            rng=rng,
            max_iter=remaining_iters,
            lam_start=0.12,
            lam_end=1e-8,
            step0=0.7,
        )
        print(f"[refine] after long descent fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final aggressive hill‑climb using any remaining time
    # ------------------------------------------------------------------
    if time.time() + safety < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.20,
        )
        print(f"[final‑hill] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore  # provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
