def adam_optimize_softmax(a0, time_limit, start_time, lr0=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_seq = a0.copy()
    best_val = fast_score(best_seq)
    it = 0
    beta_temp = 0.5
    beta_max = 200.0
    while time.time() < start_time + time_limit:
        it += 1
        # compute a via softmax
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()
        # compute convolution C
        C = conv_self(a)  # length 2n-1
        M = C.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute softmax weights
        # Use temperature beta_temp
        # subtract max for stability
        C_shift = C - M
        exp_beta = np.exp(beta_temp * C_shift)
        w = exp_beta / exp_beta.sum()
        # compute gradient g = 2 * correlation(a, w)
        g = correlation_a_w(a, w)  # shape n
        # compute gradient w.r.t. logits
        ag = np.dot(a, g)
        grad_y = a * (g - ag)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # update temperature
        beta_temp = min(beta_max, beta_temp * 1.001)
        # jitter
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)
        # early exit if best_val close to theoretical lower bound
        if best_val < 2.0 * n * (1.0/(2*n-1)) * 1.01: # near optimal 1% above theoretical lower bound
            # could break early
            pass
    return best_seq, best_val

"""
Propose a low-score step‑function.

Algorithm
---------

* **Initial pool** – a handful of diversified shapes (uniform, power‑law,
  exponential, two‑sided power‑law, Beta‑pdf, triangular).  If a previous
  solution ``height_sequence_1`` exists it is added to the pool.

* **Soft‑max parametrisation** – the coefficients are stored as logits ``y``.
  The vector ``a = softmax(y)`` is always non‑negative and sums to one,
  satisfying the constraints of ``evaluate_sequence`` without an explicit
  projection.

* **Smooth‑max surrogate** – we minimise a differentiable approximation of the
  objective ``max(conv(a,a))``:
      L(a) = (1/β)·log Σ_k exp(β·c_k)          (c = a*a)
  The temperature β is increased slowly (annealing) so the surrogate
  converges to the true max.

* **Gradient computation** – for the surrogate:
      ∂L/∂a = Σ_k w_k·∂c_k/∂a ,  w_k = softmax_β(c_k)
  Using the identity ∂c_k/∂a_i = a_{k‑i}+a_{i‑k} (zero‑padded) we obtain
  the gradient via a single FFT‑based cross‑correlation:
      g = 2·corr(a, w)          (length‑n vector)

  The gradient w.r.t. the logits is
      ∇_y = a * (g – ⟨a,g⟩).

* **Adam optimiser** – the logits are updated with Adam (β₁=0.9, β₂=0.999,
  learning‑rate decaying as 1/√t).  Every few thousand steps a tiny Gaussian
  jitter is added to escape plateaus.

* **Time budgeting & restarts** – the total time (≈1000 s) is split over a
  small number of independent restarts.  Each restart runs as many Adam
  steps as the allocated budget permits.  The best sequence ever seen is
  returned and verified with the official ``evaluate_sequence`` function.

The code respects the 2‑CPU, no‑IO constraints and uses only NumPy.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1).  a is assumed non‑negative."""
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def correlation_a_w(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Cross‑correlation (a ⋆ w)[i] = Σ_j a_j·w_{i+j} for i=0..n‑1.
    Returns 2·(a ⋆ w) as the gradient of Σ_k w_k·(a*a)_k.
    """
    n = a.shape[0]
    L = w.shape[0]                # = 2·n‑1
    fft_len = next_pow2(n + L - 1)
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    # correlation via inverse FFT of Fa * conj(Fw)
    corr_full = np.fft.irfft(fa * np.conj(fw), n=fft_len)[: n + L - 1]
    start = L - 1                 # lag 0 is at this index
    grad = 2.0 * corr_full[start:start + n]
    return grad


def fast_score(seq: np.ndarray) -> float:
    """
    Fast evaluation of 2·n·max(conv(seq,seq)) / (sum(seq)**2)
    for a non‑negative sequence.  The sequence is assumed to already
    satisfy sum(seq) > 0.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = conv_self(seq)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Shape generators (used for random starts)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^{-p_left} + w_right·(n-i)^{-p_right}.
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta‑pdf shape."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


def generate_triangular(n: int) -> np.ndarray:
    """Simple symmetric triangular shape."""
    i = np.arange(n, dtype=np.float64)
    return (i + 1) * (n - i)


# ----------------------------------------------------------------------
# Core optimiser (Adam on the soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that minimises the smooth‑max surrogate of
    max(conv(a,a)).  Returns the best sequence seen and its (exact)
    fast_score.
    """
    n = a0.shape[0]

    # initialise logits (log‑softmax variables)
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # bookkeeping
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    beta_temp = 0.5          # start temperature
    beta_max = 200.0

    while True:
        now = time.time()
        if now > start_time + time_limit:
            break
        it += 1

        # ---------- soft‑max ----------
        y_shift = y - np.max(y)          # stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()          # stays on the simplex

        # ---------- convolution ----------
        C = conv_self(a)                 # length 2·n‑1
        M = float(C.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---------- smooth max ----------
        # w_k = softmax_β(C_k)
        C_shift = C - M
        # exponentiate with current temperature
        exp_beta = np.exp(beta_temp * C_shift)
        w = exp_beta / exp_beta.sum()

        # ---------- gradient ----------
        g = correlation_a_w(a, w)       # length n

        # gradient w.r.t. logits (soft‑max Jacobian)
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- temperature schedule ----------
        beta_temp = min(beta_max, beta_temp * 1.001)

        # ---------- occasional jitter ----------
        if it % 3000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the 1000 s budget, prints progress, and returns the
    best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 0) Build a pool of initial candidates (including any previous best)
    # ------------------------------------------------------------------
    init_candidates = []

    # previous best, if the variable exists in the global namespace
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() < 1e-12:
                prev = np.full(prev.shape, 1.0 / prev.shape[0])
            else:
                prev = prev / prev.sum()
            init_candidates.append(prev.copy())
            print("[info] Using the previously discovered sequence as a start")
    except Exception:
        pass

    # a guaranteed seed – symmetric two‑sided power law
    default_len = 12000
    default_seq = generate_two_sided_power_law(default_len, 0.6, 0.6)
    default_seq = np.maximum(default_seq, 0.0)
    default_seq = default_seq / default_seq.sum()
    init_candidates.append(default_seq.copy())

    # ------------------------------------------------------------------
    # 1) Populate the pool with random shapes (≈8 total)
    # ------------------------------------------------------------------
    length_choices = [8000, 10000, 12000, 15000, 18000]
    families = ["power", "exp", "twosided", "beta", "triangular"]
    desired_pool = 8

    while len(init_candidates) < desired_pool and time.time() < deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.1, 1.4)
            seq = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(5e-4, 4e-2)
            seq = generate_exponential(n, lam)
        elif family == "twosided":
            pl = random.uniform(0.1, 1.4)
            pr = random.uniform(0.1, 1.4)
            wl = random.uniform(0.3, 3.0)
            wr = random.uniform(0.3, 3.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        elif family == "beta":
            a = random.uniform(0.2, 2.2)
            b = random.uniform(0.2, 2.2)
            seq = generate_beta_shape(n, a, b)
        else:  # triangular
            seq = generate_triangular(n)

        seq = np.maximum(seq, 0.0)
        s = seq.sum()
        if s < 1e-12:
            continue
        seq = seq / s
        init_candidates.append(seq)

    # ------------------------------------------------------------------
    # 2) Optimise each candidate with the Adam‑softmax routine
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left after initialisation")

    per_candidate = remaining / len(init_candidates)
    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        print(f"[optim] start candidate #{idx + 1}/{len(init_candidates)} "
              f"(len={a0.size}) – budget {per_candidate:.1f}s")
        seq, val = adam_optimize(a0, per_candidate, time.time())
        print(f"[optim] candidate #{idx + 1} best score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback: uniform distribution of length 10000
        n = 10000
        best_overall_seq = np.full(n, 1.0 / n)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")

    return final_list
