"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

The search consists of three phases:

1️⃣  **Exploratory random restarts** – many cheap L‑BFGS runs on the
    smooth‐max surrogate (log‑sum‑exp) with a geometric smoothing schedule.
    Each restart starts from a different heuristic distribution on the simplex
    (random, exponential, power‑law, edge‑biased, Sidon‑like).

2️⃣  **Focused refinement** – for the best candidate so far we run
    *sub‑gradient descent* on the exact max‑convolution, a *linear‑programming*
    step that maximises the total mass of an auxiliary vector `b` while keeping
    the current convolution peak `M` as a constraint (active‑set LP), and a final
    aggressive *hill‑climb* (simulated‑annealing style).

3️⃣  **Final evaluation** – the obtained vector is returned as a plain Python
    list.  All numbers are non‑negative and the sum is positive, satisfying the
    security checks of ``evaluate_sequence``.

The algorithm respects the overall time budget (default 1000 s) by checking the
deadline after each logical block.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax: returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the quantity 2·n·max(a*a) for a probability vector
    (sum = 1).  The exact evaluation used by the challenge is scale‑invariant,
    so with sum = 1 the score equals this expression.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of `v` onto the probability simplex
    (non‑negative, sum = 1).  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# 1️⃣  Smooth‑max L‑BFGS helpers
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective `2·n·smooth_max( a*a )` and its gradient w.r.t. the
    pre‑soft‑max parameters `y`.  The smooth approximation uses a log‑sum‑exp
    with smoothing parameter `lam = lam_factor * max(conv)`.
    """
    a = softmax(y)

    # ----- convolution a*a (exact, via FFT) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                 # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                      # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 200,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothing.
    Each stage uses the `objective_and_grad` routine.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        # `lam` is passed as the last argument of `objective_and_grad`
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point `y` (pre‑soft‑max) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # a simple greedy Sidon‑like construction (few ones, no repeated sums)
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict:
                if candidate * 2 in sums:
                    conflict = True
                else:
                    chosen.append(candidate)
                    for c in chosen:
                        sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k if k > 0 else 0
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that moves a random fraction of mass from i → j.
    Only moves that strictly improve the fast score are kept.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 15 % of the mass at i
        delta = best[i] * rng.random() * 0.15
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.7,
    min_step: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    Uses the true max index to compute the sub‑gradient.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        # compute exact convolution via FFT
        conv = conv_fft(best, fft_len, n)
        s_star = int(np.argmax(conv))

        # sub‑gradient: g_i = 2 * a_{s_star - i}  if the index exists
        grad = np.zeros_like(best)
        lo = max(0, s_star - (n - 1))
        hi = min(n - 1, s_star)
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * best[s_star - idx]

        # tiny jitter to escape plateaus
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.3, 3.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def lp_refine_active(
    a: np.ndarray,
    n: int,
    deadline: float,
    epsilon_rel: float = 0.02,
    mix_factor: float = 0.15,
    max_iter: int = 5,
) -> np.ndarray:
    """
    Linear‑programming refinement using only the *active* constraints:
    indices `s` where the convolution is within `epsilon_rel` of the current
    maximum `M`.  The LP maximises Σ b_j subject to (a*b)[s] ≤ M for those
    active `s`.  The resulting `b` is scaled to unit sum and mixed back into `a`.
    """
    rows_full = 2 * n - 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = conv_fft(a, next_pow2(2 * n - 1), n)
        M = conv.max()
        epsilon = epsilon_rel * M
        active = np.where(conv >= M - epsilon)[0]
        if len(active) == 0:
            break

        # Build sparse constraint matrix A (|active| × n)
        A = sp.lil_matrix((len(active), n))
        for r, s in enumerate(active):
            lo = max(0, s - (n - 1))
            hi = min(n - 1, s)
            cols = np.arange(lo, hi + 1)
            A[r, cols] = a[s - cols]    # coefficient a_{s-j}
        A = A.tocsr()

        # LP: maximise Σ b_j  ⇔ minimise −Σ b_j
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(len(active)),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            break   # no improvement possible

        g = b_opt / sum_b          # scale to unit sum
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        new_M = conv_fft(a_new, next_pow2(2 * n - 1), n).max()
        if new_M < M:
            a = a_new
            print(f"[lp] M reduced from {M:.6e} to {new_M:.6e}")
        else:
            break
    return a

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[fine L‑BFGS] fast score = {best_score:.6f}")

    # sub‑gradient descent on the exact max‑convolution
    sub_deadline = min(deadline, time.time() + 120.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.7
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[sub‑grad] fast score = {best_score:.6f}")

    # active‑set LP refinement (works also for n=5000)
    if time.time() < deadline - 30.0:
        lp_deadline = deadline - 20.0
        best_a = lp_refine_active(
            best_a,
            n,
            lp_deadline,
            epsilon_rel=0.015,
            mix_factor=0.15,
            max_iter=7,
        )
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[lp‑active] post‑refine fast score = {best_score:.6f}")

    # final aggressive hill‑climb with a simulated‑annealing flavour
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[final hill‑climb] fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Return the sequence; the external harness will compute the real score
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    # Export the best sequence for possible reuse in subsequent runs
    globals()["height_sequence_1"] = final_seq
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
