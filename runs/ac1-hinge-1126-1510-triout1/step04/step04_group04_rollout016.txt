    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline:

        k = int(np.argmax(conv))

        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        g[i[mask]] = 2.0 * best[j[mask]]
        # find source and target indices
        src = np.argmax(g)   # highest gradient
        tgt = np.argmin(g)   # lowest gradient
        if src == tgt:

        delta = best[src] * rng.random() * 0.2
        if delta <= 0:

        best[src] -= delta
        best[tgt] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[src] += delta
            best[tgt] -= delta
    return best, best_score

def subgradient_optimize(a, n, fft_len, deadline, init_step=0.5, min_step=1e-8):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    while time.time() < deadline:
        # compute conv and max index
        conv = conv_full(best, fft_len, n)
        k = int(np.argmax(conv))
        # compute gradient
        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(best)
        grad[mask] = 2.0 * best[j_arr[mask]]
        # add jitter
        grad += rng.normal(scale=1e-4, size=n)
        # step
        a_new = best - step * grad
        # project onto simplex
        a_new = project_to_simplex(a_new)
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score

def lp_refine(a, n, deadline, mix_factor=0.2, max_iter=30):
    rows = 2*n - 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv = np.convolve(a, a)
        M = conv.max()
        # build sparse convolution matrix A
        A = sp.lil_matrix((rows, n))
        for col in range(n):
            A[col:col+n, col] = a
        A = A.tocsr()
        # solve LP
        res = linprog(c=-np.ones(n), A_ub=A, b_ub=M * np.ones(rows), bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1 + 1e-12:
            break
        g = b_opt / sum_b
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        new_M = np.convolve(a_new, a_new).max()
        if new_M < M:
            a = a_new
            print(f"[lp] M reduced from {M:.6f} to {new_M:.6f}")
        else:
            break
    return a

def simulated_annealing(a, score, fft_len, n, deadline):
    cur_a = a.copy()
    cur_score = score
    rng = np.random.default_rng()
    init_T = 0.01 * cur_score  # scale
    final_T = 1e-6 * cur_score
    total_steps = 0
    while time.time() < deadline and total_steps < 500000:
        # linear cooling schedule
        elapsed = time.time() - start_anneal
        fraction = elapsed / (deadline - start_anneal)
        T = init_T * (1 - fraction) + final_T * fraction
        # propose move
        i, j = rng.integers(0, n, size=2)
        if i == j: continue
        delta = cur_a[i] * rng.random() * 0.2

        # evaluate
        cand_score = fast_score(cand, fft_len, n)

            cur_a = cand
            cur_score = cand_score
            if cur_score < best_score: update best.
        total_steps += 1
    return best_a, best_score

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm outline
-----------------
1. **Problem size** – we use n = 3000 (small enough for LP refinements,
   still large enough that the theoretical lower bound of the score is ~1).

2. **Initialisation** – start from a known construction
   (`height_sequence_1`) if available, otherwise from a uniform distribution.

3. **Stochastic L‑BFGS** – many random restarts on the smooth max‑convolution
   (softmax re‑parameterisation).  The smoothing factor (lam_factor) is
   decreased geometrically, each stage performing a few hundred L‑BFGS steps.
   After each optimisation a brief hill‑climb fine‑tunes the solution.

4. **Sub‑gradient descent** – exact max‑convolution is used; at each step
   the gradient of the active (largest) constraint is taken, a step is made,
   and the vector is projected onto the probability simplex.

5. **Linear‑programming refinement** – for n ≤ 3000 we solve the LP
   `max Σ b_j  s.t.  b ≥ 0, (a * b) ≤ M` where `M` is the current convolution
   peak.  The optimal `b` is mixed back into `a`.  This is repeated until
   no further reduction of `M` is observed.

6. **Guided hill‑climb** – repeatedly move a small amount of mass from the
   index with the largest sub‑gradient (most responsible for the current
   peak) to the index with the smallest sub‑gradient, accepting only
   improving moves.

7. **Simulated annealing** – a final polish that performs many tiny random
   mass transfers, accepting worse moves with a temperature schedule.

All steps respect the non‑negativity and sum‑to‑one constraints required by
`evaluate_sequence`.  The routine runs for at most `budget_s` seconds
(the default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable softmax; returns a probability vector (sum=1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·n‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a probability vector:
    2·n·max(a*a).  The vector is assumed to sum to 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (sum=1, v≥0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (smooth max) and its gradient – used by L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max approximation of 2·n·max(a*a) and its gradient w.r.t. y.
    `lam_factor` is the smoothing parameter (fraction of the current max).
    """
    a = softmax(y)

    # ----- linear convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c               # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Run a cascade of L‑BFGS optimisations with decreasing smoothness.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Random initial points on the simplex
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # simple greedy Sidon‑like set
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = any((candidate + c) in sums for c in chosen) or (candidate * 2 in sums)
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Simple hill‑climb using the exact max‑convolution (FFT based)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    A move is kept only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution.
    Steps are projected back onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros_like(best)
        grad[mask] = 2.0 * best[j_arr[mask]]
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (only practical for n ≤ ~3000)
# ----------------------------------------------------------------------


def lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    mix_factor: float = 0.15,
    max_iter: int = 15,
) -> np.ndarray:
    """
    Linear‑programming refinement.
    Solve   max Σ b_j  s.t. b ≥ 0, (a * b)[s] ≤ M  (M = max(a*a)).
    The optimum b is scaled to unit sum and mixed back into a.
    """
    rows = 2 * n - 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # Build sparse convolution matrix A with (A·b)[s] = (a*b)[s]
        A = sp.lil_matrix((rows, n))
        for col in range(n):
            A[col : col + n, col] = a
        A = A.tocsr()

        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(rows),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b_opt / sum_b
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        new_M = np.convolve(a_new, a_new).max()
        if new_M < M:
            a = a_new
            print(f"[lp] M reduced from {M:.6f} to {new_M:.6f}")
        else:
            break
    return a


# ----------------------------------------------------------------------
# Guided hill‑climb (uses the sub‑gradient of the current peak)
# ----------------------------------------------------------------------


def guided_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Move mass from the index with the largest sub‑gradient (most responsible
    for the current peak) to the index with the smallest sub‑gradient.
    Only improving moves are accepted.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        i_arr = np.arange(n)
        j_arr = k - i_arr
        mask = (j_arr >= 0) & (j_arr < n)
        grad = np.zeros(n)
        grad[mask] = 2.0 * best[j_arr[mask]]

        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # fall back to random move
            src, tgt = rng.integers(0, n, size=2)
            if src == tgt:
                continue

        delta = best[src] * rng.random() * 0.15
        if delta <= 0.0:
            continue

        best[src] -= delta
        best[tgt] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[src] += delta
            best[tgt] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (final polishing)
# ----------------------------------------------------------------------


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Many tiny random mass transfers; worse moves are occasionally accepted
    according to a linear temperature schedule.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    start = time.time()

    # temperature goes from 0.1·score down to 1e‑6·score
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        frac = elapsed / total
        T = ((1 - frac) * 0.1 + frac * 1e-6) * cur_score
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.08
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = fast_score(cand, fft_len, n)
        if cand_score < best_score:
            best = cand
            best_score = cand_score
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):
                best = cand
                best_score = cand_score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # problem size – use n = 3000 (allows LP) but fall back to a larger
    # existing construction if it is already available.
    # --------------------------------------------------------------
    n = 3000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1_000:
            # truncate / pad to the chosen size
            if tmp.size >= n:
                best_a = tmp[:n]
            else:
                best_a = np.concatenate([tmp, np.full(n - tmp.size, 0.0)])
            best_a = np.clip(best_a, 0, None)
            s = best_a.sum()
            if s > 0:
                best_a = best_a / s
            else:
                best_a = np.full(n, 1.0 / n)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_a = best_a / best_a.sum()
    best_y = np.log(best_a + 1e-12)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – many L‑BFGS restarts on the smoothed objective
    # --------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-5, num=22))
    maxiter_stage = 600
    modes = ["random", "exp", "power", "edge", "sidon", "best"]
    iteration = 0

    # reserve about 350 s for this stage
    lbgfs_deadline = min(deadline, start_time + 350.0)
    while time.time() < lbgfs_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_y = y_opt
            best_score = score_opt
            print(f"[{iteration}] L‑BFGS mode={mode} improved fast_score={best_score:.6f}")

        # short aggressive hill‑climb to clean up the result
        hc_deadline = min(lbgfs_deadline, time.time() + 0.5)
        best_a, best_score = hill_climb(
            best_a, best_score, fft_len_conv, n, hc_deadline, max_iter=2000
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Stage 2 – sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 200.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.5
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgradient] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – linear‑programming refinement (possible because n≤3000)
    # --------------------------------------------------------------
    if n <= 3000 and time.time() + 90.0 < deadline:
        lp_deadline = deadline - 90.0
        best_a = lp_refine(best_a, n, lp_deadline, mix_factor=0.15, max_iter=15)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[lp] post‑refine fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – guided hill‑climb using the active‑set sub‑gradient
    # --------------------------------------------------------------
    guide_deadline = min(deadline, time.time() + 30.0)
    if time.time() < guide_deadline:
        best_a, best_score = guided_hill_climb(
            best_a, best_score, fft_len_conv, n, guide_deadline
        )
        print(f"[guided] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Stage 5 – simulated‑annealing polishing (uses remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, deadline
        )
        print(f"[anneal] final fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # final evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
