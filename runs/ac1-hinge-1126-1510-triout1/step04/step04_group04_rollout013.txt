"""
Search for low‐evaluation step‑function using a blend of
- stochastic L‑BFGS on a smooth max‑convolution,
- systematic exploration of simple parametric families (exponential edge, beta‐type),
- sub‑gradient descent on the exact max,
- a mirror‑descent refinement.
All steps respect the non‑negative simplex constraints required by
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# Helper functions
def next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def softmax(y: np.ndarray) -> np.ndarray:
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()

def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]

def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())

def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    a = softmax(y)

    # linear convolution a*a via FFT
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt conv
    w = exp_shift / Z

    # g = d conv / d a  (size n)
    # we need g_j = sum_i w_i * (a_{i-j} + a_{i-(i-j)}) = sum_i w_i * a_{i-j} + ...
    # Our earlier code computes via convolution of a reversed with w.
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def hill_climb(a: np.ndarray, cur_score: float, fft_len: int, n: int,
               deadline: float, max_iter: int = 2000) -> tuple[np.ndarray, float]:
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score

def schedule_optimize(y_start: np.ndarray, n: int,
                     fft_len_conv: int, fft_len_g: int,
                     lam_schedule: list[float],
                     maxiter_per_stage: int = 300) -> np.ndarray:
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method="L-BFGS-B",
                       jac=True,
                       options={"maxiter": maxiter_per_stage,
                                "gtol": 1e-12,
                                "disp": False})
        y = res.x
    return y

def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "exp_edge":
        # exponential decay from the edges, parameter k drawn randomly
        k = rng.uniform(0.0005, 0.02)
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-k * dist)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "beta":
        # symmetric beta‑type shape: a_i ∝ (i+δ)^{p} + (n-1-i+δ)^{p}
        p = rng.uniform(-0.9, 2.0)
        delta = 0.5
        i = np.arange(n, dtype=float) + delta
        a0 = (i ** p) + ((n - 1 - np.arange(n, dtype=float) + delta) ** p)
        # clip extreme values that might overflow
        a0 = np.clip(a0, 0, np.max(a0))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def subgradient_optimize(a: np.ndarray, n: int, fft_len: int,
                         deadline: float,
                         init_step: float = 0.5,
                         min_step: float = 1e-8) -> tuple[np.ndarray, float]:
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = np.convolve(best, best)
        i_star = int(np.argmax(conv))
        # gradient of max conv w.r.t a
        grad = np.zeros_like(best)
        # positions where a contributes to conv_i_star
        idx = np.arange(n)
        j = i_star - idx
        mask = (j >= 0) & (j < n)
        grad[idx[mask]] = 2.0 * best[j[mask]]
        # add tiny jitter
        grad += rng.normal(scale=1e-4, size=n)
        a_new = best - step * grad
        a_new = project_to_simplex(a_new)
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score

def mirror_descent_optimize(a: np.ndarray, n: int, fft_len: int,
                            deadline: float,
                            init_eta: float = 0.05,
                            decay: float = 0.99) -> tuple[np.ndarray, float]:
    """Multiplicative (KL‑mirror) descent on the exact max convolution."""
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    eta = init_eta
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = np.convolve(best, best)
        i_star = int(np.argmax(conv))
        grad = np.zeros_like(best)
        idx = np.arange(n)
        j = i_star - idx
        mask = (j >= 0) & (j < n)
        grad[idx[mask]] = 2.0 * best[j[mask]]
        # multiplicative update (ensure positivity)
        update = np.exp(-eta * grad)
        a_new = best * update
        # project onto simplex (renormalize)
        a_new = project_to_simplex(a_new)
        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
        # schedule eta
        eta *= decay
        # stop when eta becomes too small
        if eta < 1e-6:
            break
    return best, best_score

def parametric_search(a_best: np.ndarray, n: int, fft_len: int,
                      deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Randomly explore two low‑dimensional families (exp‑edge and beta).
    """
    best = a_best.copy()
    best_score = fast_score(best, fft_len, n)
    # a small budget for this stage
    while time.time() < deadline:
        mode = rng.choice(["exp_edge", "beta"])
        if mode == "exp_edge":
            k = rng.uniform(0.0005, 0.03)  # broader range
            dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
            a0 = np.exp(-k * dist)
        else:  # beta
            p = rng.uniform(-0.9, 2.5)
            delta = 0.5
            i = np.arange(n, dtype=float) + delta
            a0 = (i ** p) + ((n - 1 - np.arange(n, dtype=float) + delta) ** p)
        # normalize, clip negative just in case
        a0 = np.clip(a0, 0, None)
        s = a0.sum()
        if s < 1e-12:
            continue
        a0 /= s
        score = fast_score(a0, fft_len, n)
        if score < best_score:
            best = a0
            best_score = score
    return best, best_score

"""
Search for a low‐evaluation step‑function.

The pipeline combines:
* stochastic L‑BFGS on a smooth max‑convolution (log‑softmax parametrisation),
* random exploration of two low‑dimensional families (exponential‑edge and beta‑type),
* a sub‑gradient descent on the exact max convolution,
* a mirror‑descent (multiplicative) refinement on the simplex,
* a final fine L‑BFGS sweep with a very small smoothing parameter,
* a short greedy hill‑climb polish.

All iterates live on the probability simplex, satisfying the constraints
imposed by ``evaluate_sequence``.
"""
import time
import numpy as np
from scipy.optimize import minimize
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum = 1.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    inds = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if inds.size == 0:
        # return uniform fallback
        return np.full_like(v, 1.0 / len(v))
    rho = inds[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective and its gradient (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max objective (2·n·smooth_max) and gradient w.r.t. y.
    The smoothing parameter is ``lam = max(conv) * lam_factor``.
    """
    a = softmax(y)

    # ---- convolution a*a (FFT) ----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                     # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---- gradient of smooth max ----
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # compute g_i = Σ_j a_j * w_{i+j}
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Cascade of L‑BFGS runs with a decreasing smoothing parameter.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    A move is kept only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)

    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "sidon":
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (candidate * 2 not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)

    elif mode == "exp_edge":
        # exponential decay from the edges; k is drawn randomly
        k = rng.uniform(0.0005, 0.02)
        dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
        a0 = np.exp(-k * dist)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    elif mode == "beta":
        # symmetric beta‑type shape: a_i ∝ (i+δ)^p + (n-1-i+δ)^p
        p = rng.uniform(-0.9, 2.0)
        delta = 0.5
        i = np.arange(n, dtype=float) + delta
        a0 = (i ** p) + ((n - 1 - np.arange(n, dtype=float) + delta) ** p)
        a0 = np.clip(a0, 0, None)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)

    else:
        raise ValueError(f"unknown mode {mode}")

    return y0


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution, projected onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        i_star = int(np.argmax(conv))

        grad = np.zeros_like(best)
        idx = np.arange(n)
        j = i_star - idx
        mask = (j >= 0) & (j < n)
        grad[idx[mask]] = 2.0 * best[j[mask]]

        # small random jitter helps escaping plateaus
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break

    return best, best_score


def mirror_descent_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_eta: float = 0.05,
    decay: float = 0.99,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror (multiplicative) descent on the exact max‑convolution.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    eta = init_eta
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        i_star = int(np.argmax(conv))

        grad = np.zeros_like(best)
        idx = np.arange(n)
        j = i_star - idx
        mask = (j >= 0) & (j < n)
        grad[idx[mask]] = 2.0 * best[j[mask]]

        # multiplicative update, then renormalise
        mult = np.exp(-eta * grad)
        a_new = best * mult
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score

        eta *= decay
        if eta < 1e-6:
            break

    return best, best_score


def parametric_search(
    a_best: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Random exploration of two low‑dimensional families:
    * exponential‑edge   a_i ∝ exp(-k·dist(i,edge))
    * beta‑type          a_i ∝ (i+δ)^p + (n-1-i+δ)^p
    """
    best = a_best.copy()
    best_score = fast_score(best, fft_len, n)

    while time.time() < deadline:
        if rng.random() < 0.5:
            # exponential‑edge
            k = rng.uniform(0.0005, 0.03)
            dist = np.minimum(np.arange(n), n - 1 - np.arange(n))
            a0 = np.exp(-k * dist)
        else:
            # beta‑type
            p = rng.uniform(-0.9, 2.5)
            delta = 0.5
            i = np.arange(n, dtype=float) + delta
            a0 = (i ** p) + ((n - 1 - np.arange(n, dtype=float) + delta) ** p)

        a0 = np.clip(a0, 0, None)
        s = a0.sum()
        if s < 1e-12:
            continue
        a0 /= s
        score = fast_score(a0, fft_len, n)
        if score < best_score:
            best = a0
            best_score = score

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for the final evaluation
    final_deadline = start_time + budget_s - 2.0

    # -------------------- problem size --------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float)
        if tmp.ndim == 1 and tmp.size >= 1_000:
            n = max(n, int(tmp.size))
            best_a = tmp.copy()
            best_a = np.clip(best_a, 0, None)
            s = best_a.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = best_a / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # -------------------- hyper‑parameters --------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    lam_schedule = list(np.geomspace(0.4, 0.001, num=20))
    maxiter_stage = 350
    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "exp_edge",
        "beta",
        "best",
    ]

    # -------------------- 1. Random L‑BFGS + hill‑climb --------------------
    # reserve ~250 s for the subsequent refinement phases
    random_deadline = start_time + budget_s - 250
    iteration = 0
    while time.time() < random_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # quick local hill‑climb
        hc_deadline = min(random_deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # -------------------- 2. Parametric family search --------------------
    param_deadline = min(final_deadline, time.time() + 30.0)
    best_a, best_score = parametric_search(
        best_a, n, fft_len_conv, param_deadline, rng
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[parametric] best fast score = {best_score:.6f}")

    # -------------------- 3. Exact sub‑gradient descent --------------------
    sub_deadline = min(final_deadline, time.time() + 30.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.5
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[subgrad] best fast score = {best_score:.6f}")

    # -------------------- 4. Mirror‑descent refinement --------------------
    md_deadline = min(final_deadline, time.time() + 20.0)
    best_a, best_score = mirror_descent_optimize(
        best_a, n, fft_len_conv, md_deadline, init_eta=0.08, decay=0.98
    )
    best_y = np.log(best_a + 1e-12)
    print(f"[mirror] best fast score = {best_score:.6f}")

    # -------------------- 5. Very fine L‑BFGS sweep --------------------
    fine_lam_schedule = list(np.geomspace(0.2, 5e-6, num=25))
    for lam in fine_lam_schedule:
        if time.time() >= final_deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 800, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.5g} fast score = {best_score:.6f}")

    # -------------------- 6. Final greedy polishing --------------------
    if time.time() < final_deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=8000,
        )
        best_y = np.log(best_a + 1e-12)

    # -------------------- 7. Official evaluation and return --------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
