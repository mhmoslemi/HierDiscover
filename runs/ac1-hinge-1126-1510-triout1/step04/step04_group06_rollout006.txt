"""
Propose a step‑function (non‑negative sequence) minimizing the
`evaluate_sequence` score.

Approach
--------
1. Choose a manageable length `n` (≈4000).  The score equals
   `2*n*max(conv(a,a))` when `a` is normalized to sum‑1.
2. Search a few simple parametric families (edge‑constant,
   exponential decay, power‑law) using a cheap golden‑section
   optimisation.  These give a good starting point.
3. Refine the candidate with a long projected sub‑gradient descent
   on the true `max(conv)` (momentum + simplex projection).
4. Perform a short simulated‑annealing phase for non‑local moves.
5. Finish with a greedy hill‑climb to polish the best vector.
6. Return the final sequence as a plain list of floats and store it
   globally as `height_sequence_1` for possible reuse.

All steps respect the total time budget (default 1000 s) and use only
NumPy / SciPy (no filesystem or network access).
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σa_i = 1, a_i≥0."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback (should not happen)
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len_conv: int, n: int) -> float:
    """Score = 2·n·max(conv) for a already normalised (sum=1)."""
    conv = full_conv(a, fft_len_conv)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parametric families – cheap golden‑section search
# ----------------------------------------------------------------------


def edge_array(w: int, a_edge: float, n: int) -> np.ndarray:
    """Construct piecewise‑constant edge shape."""
    if w * 2 >= n:
        return None
    interior_len = n - 2 * w
    a_int = (1.0 - 2.0 * w * a_edge) / interior_len
    if a_int < 0.0:
        return None
    a = np.full(n, a_int, dtype=np.float64)
    a[:w] = a_edge
    a[-w:] = a_edge
    # normalise for safety against rounding errors
    a /= a.sum()
    return a


def exponential_array(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * idx) + np.exp(-alpha * (n - 1 - idx))
    a /= a.sum()
    return a


def powerlaw_array(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law decay from both ends."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p) + (n + 1 - i) ** (-p)
    a /= a.sum()
    return a


def golden_section_search(func, low: float, high: float, iters: int = 30):
    """Find approx. minimum of `func` on [low,high] (unimodal)."""
    phi = (math.sqrt(5) - 1) / 2  # ≈0.618
    x1 = low + (1 - phi) * (high - low)
    x2 = low + phi * (high - low)
    f1 = func(x1)
    f2 = func(x2)
    for _ in range(iters - 1):
        if f1 < f2:
            high = x2
            x2 = x1
            f2 = f1
            x1 = low + (1 - phi) * (high - low)
            f1 = func(x1)
        else:
            low = x1
            x1 = x2
            f1 = f2
            x2 = low + phi * (high - low)
            f2 = func(x2)
    if f1 < f2:
        return x1, f1
    else:
        return x2, f2


def best_edge_family(n: int, fft_len_conv: int) -> tuple[np.ndarray, float]:
    """Return best edge‑constant shape over a grid of widths."""
    best_score = np.inf
    best_a = None
    # candidate widths (integers)
    widths = [1, 2, 3, 5, 7, 10, 15, 20, 30, 50, 70, 100, 150, 200]
    for w in widths:
        if 2 * w >= n:
            continue
        # feasible a_edge lies in (0, 1/(2*w))
        high = 1.0 / (2.0 * w)
        # objective for a given a_edge
        def obj(a_edge):
            a = edge_array(w, a_edge, n)
            if a is None:
                return np.inf
            return fast_score(a, fft_len_conv, n)

        a_edge_opt, sc = golden_section_search(obj, 0.0, high, iters=30)
        # final evaluation at the optimum tip
        a_opt = edge_array(w, a_edge_opt, n)
        if a_opt is not None:
            sc = fast_score(a_opt, fft_len_conv, n)
            if sc < best_score:
                best_score = sc
                best_a = a_opt
    return best_a, best_score


def best_exponential_family(n: int, fft_len_conv: int) -> tuple[np.ndarray, float]:
    """Search best symmetric exponential decay."""
    # alpha in [0, 0.02] – 0 gives uniform, larger concentrates ends.
    high = 0.02
    def obj(alpha):
        a = exponential_array(alpha, n)
        return fast_score(a, fft_len_conv, n)

    alpha_opt, sc = golden_section_search(obj, 0.0, high, iters=30)
    a_opt = exponential_array(alpha_opt, n)
    sc = fast_score(a_opt, fft_len_conv, n)
    return a_opt, sc


def best_powerlaw_family(n: int, fft_len_conv: int) -> tuple[np.ndarray, float]:
    """Search best symmetric power‑law decay."""
    # p in [0, 4]
    low, high = 0.0, 4.0
    def obj(p):
        a = powerlaw_array(p, n)
        return fast_score(a, fft_len_conv, n)

    p_opt, sc = golden_section_search(obj, low, high, iters=30)
    a_opt = powerlaw_array(p_opt, n)
    sc = fast_score(a_opt, fft_len_conv, n)
    return a_opt, sc


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (momentum)
# ----------------------------------------------------------------------


def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Momentum sub‑gradient descent on the true max‑convolution.
    Keeps a on the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9                    # momentum coefficient
    it = 0

    while time.time() < deadline:
        # ---------- convolution ----------
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # ---------- active set mask ----------
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ---------- sub‑gradient ----------
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size
        step = 0.4 / math.sqrt(it + 1.0)
        a = a - step * v
        a = simplex_projection(a)

        # keep the best seen so far
        cur_sc = fast_score(a, fft_len_conv, n)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated annealing (non‑local moves)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len_conv: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated annealing that moves a random fraction of mass
    between two positions.
    """
    cur_a = a_start.copy()
    cur_sc = score_start
    best_a = cur_a.copy()
    best_sc = cur_sc

    T = max(cur_sc * 0.02, 1e-8)
    decay = 0.9994
    it = 0

    while time.time() < deadline:
        # pick source with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        # pick a distinct target
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # propose moving a fraction (up to 25 %) of source mass
        max_delta = cur_a[i] * 0.25
        delta = rng.random() * max_delta
        if delta <= 0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_sc = fast_score(cur_a, fft_len_conv, n)

        if new_sc < cur_sc:
            cur_sc = new_sc
        else:
            prob = math.exp(-(new_sc - cur_sc) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_sc = new_sc
            else:
                # revert move
                cur_a[i] += delta
                cur_a[j] -= delta
                # temperature update and continue
                it += 1
                T *= decay
                continue

        # update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = cur_a.copy()

        # anneal temperature
        T *= decay
        if it % 4000 == 0 and T < best_sc * 1e-6:
            T = max(best_sc * 0.02, 1e-8)
        it += 1

    return best_a, best_sc


# ----------------------------------------------------------------------
# Greedy hill‑climb (local polishing)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len_conv: int,
               n: int,
               deadline: float,
               max_iter: int = 3000) -> tuple[np.ndarray, float]:
    """
    Tiny mass transfers between two positions while the score improves.
    Operates entirely on the simplex.
    """
    best = a.copy()
    best_sc = cur_score
    rng = np.random.default_rng()
    iter_left = max_iter

    while time.time() < deadline and iter_left > 0:
        iter_left -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move (up to 4 % of source)
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_sc = fast_score(best, fft_len_conv, n)
        if new_sc < best_sc:
            best_sc = new_sc
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_sc


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises `evaluate_sequence`.

    The method:
      * builds a length `n` (≈4000) probability vector,
      * explores three parametric families (edge, exponential, power‑law) via
        golden‑section search,
      * refines the best candidate with a long momentum sub‑gradient descent,
      * adds simulated annealing and a final hill‑climb for polishing,
      * returns the best list of floats found.
    """
    rng = np.random.default_rng(seed + 777)

    # --------------------------------------------------------------
    # Determine problem size n
    # --------------------------------------------------------------
    max_n = 4000
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        if prev.size > max_n:
            prev = prev[:max_n]
        if prev.sum() > 0:
            a0 = prev / prev.sum()
            n = a0.size
        else:
            n = 2500
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2500
        a0 = np.full(n, 1.0 / n)

    # If we have room, increase to the max allowed size (more degrees of freedom)
    if n < max_n:
        # pad with tiny uniform mass to reach max_n without altering shape much
        pad = max_n - n
        a0 = np.concatenate([a0, np.full(pad, 1e-12)])
        a0 = a0 / a0.sum()
        n = max_n

    # --------------------------------------------------------------
    # FFT lengths (powers of two for speed)
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient convolutions

    # --------------------------------------------------------------
    # Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin
    # Phase 1 – cheap parametric search (≈10 % of budget)
    phase1_end = start_time + 0.12 * budget_s

    # --------------------------------------------------------------
    # Phase 1 – parametric families
    # --------------------------------------------------------------
    best_a = a0.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_sc:.6f}")

    # Edge family
    if time.time() < phase1_end:
        a_edge, sc_edge = best_edge_family(n, fft_len_conv)
        if a_edge is not None and sc_edge < best_sc:
            best_a, best_sc = a_edge, sc_edge
            print(f"[param] edge family best_score={best_sc:.6f}")

    # Exponential family
    if time.time() < phase1_end:
        a_exp, sc_exp = best_exponential_family(n, fft_len_conv)
        if sc_exp < best_sc:
            best_a, best_sc = a_exp, sc_exp
            print(f"[param] exponential best_score={best_sc:.6f}")

    # Power‑law family
    if time.time() < phase1_end:
        a_pow, sc_pow = best_powerlaw_family(n, fft_len_conv)
        if sc_pow < best_sc:
            best_a, best_sc = a_pow, sc_pow
            print(f"[param] power‑law best_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sub‑gradient optimisation (major chunk of budget)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    # allocate ~60 % of remaining time to sub‑gradient descent
    sub_end = time.time() + remaining * 0.6
    if time.time() < sub_end:
        best_a, best_sc = subgradient_optimize(best_a,
                                               sub_end,
                                               fft_len_conv,
                                               fft_len_g,
                                               n)
        print(f"[sub‑gradient] best_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing (moderate time)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    sa_end = time.time() + remaining * 0.8  # leave ~20 % for final hill‑climb
    if time.time() < sa_end:
        best_a, best_sc = simulated_annealing(best_a,
                                              best_sc,
                                              fft_len_conv,
                                              n,
                                              sa_end,
                                              rng)
        print(f"[SA] best_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (remaining budget)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_a, final_sc = hill_climb(best_a,
                                       best_sc,
                                       fft_len_conv,
                                       n,
                                       deadline,
                                       max_iter=5000)
        best_a, best_sc = final_a, final_sc
        print(f"[final] hill‑climb score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Evaluate with the supplied evaluator (optional logging)
    # --------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(best_a.tolist())
        print(f"[final] evaluate_sequence -> {eval_val:.6f}")
    except Exception:
        eval_val = float('nan')
        print("[final] evaluate_sequence raised an exception")

    # store globally for possible reuse in a later call
    globals()['height_sequence_1'] = best_a.tolist()

    return best_a.tolist()
