def generate_initial_candidates(n, rng):
    candidates = []
    # uniform
    a = np.full(n, 1.0 / n)
    candidates.append(a)
    # exponential rates
    for rate in np.linspace(0.001, 0.02, num=5):
        a = np.exp(-rate * np.arange(n))
        a /= a.sum()
        candidates.append(a)
    # power law exponents
    for p in np.linspace(0.3, 3.0, num=6):
        a = (np.arange(1, n+1, dtype=float) ** (-p))
        a /= a.sum()
        candidates.append(a)
    # polynomial decreasing (n-i)^p
    for p in np.linspace(0.5, 3.0, num=5):
        a = ((n - np.arange(n) - 1) ** p) + 1e-12  # +eps to avoid 0
        a /= a.sum()
        candidates.append(a)
    # linear decreasing (triangular)
    a = (n - np.arange(n) - 1)
    a = np.maximum(a, 0)
    a /= a.sum()
    candidates.append(a)
    # random Dirichlet (alpha 0.1 to 1)
    for alpha in [0.05, 0.1, 0.2, 0.5]:
        a = rng.gamma(shape=alpha, scale=1.0, size=n)
        a /= a.sum()
        candidates.append(a)
    # near-sparse random: randomly choose K points
    for K in [50, 80, 100, 150]:
        a = np.zeros(n)
        idx = rng.choice(n, size=K, replace=False)
        # equal weight
        a[idx] = 1.0 / K
        a /= a.sum()
        candidates.append(a)
    return candidates
```

"""
A fast, multi‑stage optimiser for the step‑function evaluation problem.

The algorithm proceeds as follows:

1. **Initial candidates** – A diverse set of monotone (or sparse)
   distributions (uniform, exponential, power‑law, polynomial,
   Dirichlet‑sampled) is generated and the best one is kept.

2. **L‑BFGS refinement** – The best candidate is locally optimised
   using a continuation scheme on a smooth approximation of the
   max‑convolution (soft‑max with decreasing temperature λ).

3. **Aggressive hill‑climb** – A pure greedy hill‑climb on the simplex
   finishes each L‑BFGS run.

4. **Simulated annealing** – The current best solution is refined with
   a temperature‑controlled random‑mass‑transfer walk that can accept
   occasional uphill moves, allowing it to escape local minima.

5. **Projected‑gradient polishing** – A few projected‑gradient steps
   on the smooth max further tidy the solution.

6. **Final hill‑climb** – One last deterministic hill‑climb uses any
   remaining time.

All steps respect the non‑negative, ℓ₁‑normalised constraint; the
objective value printed during the run is `2·n·max(conv)/(sum)²`,
identical to `evaluate_sequence`.  The function returns the best
sequence found before the time budget expires.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max, producing a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective on a probability vector (sum == 1).

    Returns 2·n·max(conv(a,a)), which matches evaluate_sequence for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    { x | x_i ≥ 0, Σ x_i = 1 }.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained vector y (log‑soft‑max coordinates).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # peak becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # compute ∂conv_i / ∂a using the structure of convolution
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation of L‑BFGS optimisations over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:   # strict improvement
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns a vector of length n (without the 2·n scaling).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # derivative of smooth max w.r.t a is 2 * g
    return 2.0 * g


def generate_initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Assemble a collection of diverse, monotone (or sparse) probability
    vectors to seed the optimiser.
    """
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # exponential families
    for rate in np.linspace(0.001, 0.02, 5):
        a = np.exp(-rate * np.arange(n))
        a = a / a.sum()
        cand.append(a)

    # power‑law families a_i ∝ (i+1)^(-p)
    for p in np.linspace(0.3, 3.0, 6):
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = a / a.sum()
        cand.append(a)

    # polynomial decreasing a_i ∝ (n-i)^p
    for p in np.linspace(0.5, 3.0, 5):
        a = ((n - np.arange(n) - 1) ** p) + 1e-12
        a = a / a.sum()
        cand.append(a)

    # linear (triangular) decreasing
    a = (n - np.arange(n) - 1).astype(float)
    a = a / a.sum()
    cand.append(a)

    # Dirichlet‑sampled (alpha < 1 gives sparse vectors)
    for alpha in [0.05, 0.1, 0.2, 0.5]:
        a = rng.gamma(shape=alpha, scale=1.0, size=n)
        a = a / a.sum()
        cand.append(a)

    # Random sparse patterns with equal weights
    for K in [30, 60, 90, 130]:
        a = np.zeros(n)
        idx = rng.choice(n, size=K, replace=False)
        a[idx] = 1.0 / K
        a = a / a.sum()
        cand.append(a)

    return cand

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Total time budget, in seconds.  The function returns before the
        budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Choose problem size (reuse previous construction if possible)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, list) and len(prev) >= 2000:
        n = len(prev)
        best_a = np.asarray(prev, dtype=float)
    else:
        n = 5000
        best_a = np.full(n, 1.0 / n)

    # Normalise to the probability simplex
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / best_a.sum()

    fft_len_conv = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1. Scan a wide range of handcrafted candidates
    # ------------------------------------------------------------------
    for cand in generate_initial_candidates(n, rng):
        sc = fast_score(cand, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            best_y = np.log(best_a + 1e-12)
            print(f"[init‑cand] improved fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2. L‑BFGS continuation on a smoothed max‑convolution
    # ------------------------------------------------------------------
    fft_len_g = next_pow2(3 * n - 2)
    lam_factors = np.geomspace(0.5, 1e-6, num=40)   # finer schedule
    maxiter_stage = 800

    iteration = 0
    while time.time() < deadline - 5.0:
        # randomised start for the next L‑BFGS run
        mode = rng.choice(["best", "random", "exp", "power", "poly", "sparse"])
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.05, size=n)
        elif mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a0 = np.exp(-rate * np.arange(n))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.3, 3.0)
            a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        elif mode == "poly":
            p = rng.uniform(0.5, 3.0)
            a0 = ((n - np.arange(n) - 1) ** p) + 1e-12
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)
        else:   # sparse heavy‑tailed start
            K = int(rng.integers(40, 180))
            a0 = np.zeros(n)
            idx = rng.choice(n, size=K, replace=False)
            a0[idx] = 1.0 / K
            a0 = a0 / a0.sum()
            y0 = np.log(a0 + 1e-12)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.12f}")

        # ----- aggressive deterministic hill‑climb -----
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=2000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing refinement (exploratory, accepts uphill)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4.0:
        cur = best_a.copy()
        cur_score = best_score
        global_best = best_a.copy()
        global_best_score = best_score

        T = 0.05 * best_score          # initial temperature
        cooling = 0.9995               # exponential cooling factor
        move_frac = 0.20

        while time.time() < deadline - 2.0:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1
            if cur[i] == 0.0:
                continue
            delta = cur[i] * rng.random() * move_frac
            if delta <= 0.0:
                continue

            old_i, old_j = cur[i], cur[j]
            cur[i] = old_i - delta
            cur[j] = old_j + delta
            new_score = fast_score(cur, fft_len_conv, n)

            if new_score < cur_score:
                cur_score = new_score
            else:
                # accept uphill with probability exp(-Δ/T)
                accept_prob = np.exp(-(new_score - cur_score) / max(T, 1e-12))
                if rng.random() < accept_prob:
                    cur_score = new_score
                else:
                    # revert move
                    cur[i] = old_i
                    cur[j] = old_j

            # keep track of the overall best ever seen
            if cur_score < global_best_score:
                global_best_score = cur_score
                global_best = cur.copy()

            T *= cooling

        best_a = global_best
        best_score = global_best_score
        best_y = np.log(best_a + 1e-12)
        print(f"[sa] SA best fast score = {best_score:.12f}")

    # ------------------------------------------------------------------
    # 4. Projected‑gradient descent on the smooth max (optional polishing)
    # ------------------------------------------------------------------
    lam_factor_pg = 1e-5
    step0 = 0.4
    max_pg_iter = 1500
    pg_iter = 0
    while time.time() < deadline - 2.0 and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)   # gradient of smooth max
        step = step0 / np.sqrt(pg_iter)
        cand = proj_simplex(best_a - step * grad)
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            lam_factor_pg = max(lam_factor_pg * 0.9, 1e-7)

    print(f"[pg] projected‑gradient iterations={pg_iter} best fast score={best_score:.12f}")

    # ------------------------------------------------------------------
    # 5. Final deterministic hill‑climb (use any remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=15000,
            max_move_frac=0.20,
        )
        print(f"[final‑hc] final fast score = {best_score:.12f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.12f}")
    return final_seq
