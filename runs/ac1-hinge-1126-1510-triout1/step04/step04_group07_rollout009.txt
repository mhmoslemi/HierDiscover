"""
Propose a low‑score step‑function.

Main ideas
----------
* Initialise several diverse candidates (power‑law, exponential, two‑sided,
  Dirichlet draws, previous best if available) of length ~10‑15k.
* Optimise each candidate with a *smooth* approximation of the ∞‑norm:
      Lβ(a)= (1/β)·log∑_k exp(β·(a*a)_k) .
  For β→∞ the loss tends to max(a*a).  The gradient of Lβ is a simple
  convolution that can be computed with two FFTs per iteration.
* Parameters are kept in a log‑softmax representation (logits y) – this
  automatically enforces non‑negativity and the simplex constraint.
  Adam is used for the updates; the temperature β is increased gradually
  (annealing) so the optimisation first sees a smooth landscape and later
  focuses on the true max‑value.
* The whole run is split among a few restarts, each respecting the global
  1000 s budget.  The best sequence seen overall is finally verified with
  the supplied `evaluate_sequence` function.
"""

# ----------------------------------------------------------------------
# Imports & utilities
# ----------------------------------------------------------------------
import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)**2) using an FFT.
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape generators (used for random starting points)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64)
    return np.exp(-lam * i)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_dirichlet(n: int, alpha: float = 0.5) -> np.ndarray:
    """
    Draw a random point on the simplex using a Dirichlet distribution.
    Smaller alpha gives sparser vectors.
    """
    x = np.random.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Optimiser: smooth‑max (softmax) loss with Adam on logits
# ----------------------------------------------------------------------


def smooth_max_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    beta_schedule: list[float] | None = None,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Minimises  max(conv(a,a))  using a smooth soft‑max approximation.
    The parameters are kept as logits y so that a = softmax(y) stays on the
    simplex automatically.

    Returns the best sequence (as a np.ndarray) seen during the run and its
    fast_score value.
    """
    n = a0.shape[0]

    # initialise logits (log‑softmax variables)
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # best‑so‑far bookkeeping
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    # temperature schedule
    if beta_schedule is None:
        beta_schedule = [5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0]

    beta_idx = 0
    steps_at_current_beta = 0
    steps_per_beta = 3000   # after this many steps we move to the next β

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # --------------------------------------------------------------
        # soft‑max re‑parameterisation (a lies on the simplex)
        # --------------------------------------------------------------
        y_shift = y - np.max(y)          # improve numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()          # always non‑negative, sum=1

        # --------------------------------------------------------------
        # convolution a*a   (linear, not circular)
        # --------------------------------------------------------------
        conv_len = 2 * n - 1
        fft_len = next_pow2(conv_len + n - 1)          # enough for cross‑corr later
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # --------------------------------------------------------------
        # smooth approximation of max – soft‑max with current β
        # --------------------------------------------------------------
        beta = beta_schedule[beta_idx]
        max_c = conv.max()
        # subtract max to avoid overflow
        exp_beta = np.exp(beta * (conv - max_c))
        w = exp_beta / exp_beta.sum()         # weights sum to 1

        # --------------------------------------------------------------
        # gradient of the smooth loss w.r.t. a
        #   g = 2 * (a ⋆ w)  (cross‑correlation)
        # --------------------------------------------------------------
        fw = np.fft.rfft(w, n=fft_len)
        # cross‑correlation via FFT (a * conj(w))
        cross = np.fft.irfft(fa * np.conj(fw), n=fft_len)[:n]
        g = 2.0 * cross                       # shape (n,)

        # --------------------------------------------------------------
        # back‑prop through the soft‑max (Jacobian)
        # --------------------------------------------------------------
        ag = np.dot(a, g)                     # scalar ⟨a,g⟩
        grad_y = a * (g - ag)                 # shape (n,)

        # --------------------------------------------------------------
        # Adam update on logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # bookkeeping: keep the best sequence seen so far (based on true max)
        # --------------------------------------------------------------
        cur_val = 2.0 * n * max_c               # sum(a)=1 ⇢ denominator =1
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # --------------------------------------------------------------
        # annealing: after a fixed number of steps increase β
        # --------------------------------------------------------------
        steps_at_current_beta += 1
        if steps_at_current_beta >= steps_per_beta and beta_idx + 1 < len(beta_schedule):
            beta_idx += 1
            steps_at_current_beta = 0
            # small jitter to help escaping flat regions when we change temperature
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

        # occasional random jitter (even without temperature change)
        if it % 5000 == 0:
            sigma = 0.005 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000) -> list[float]:
    """
    Search for a non‑negative step‑function with a low value of
    `evaluate_sequence`.  The function respects the 1000 s time budget,
    runs several independent restarts, and returns the best sequence found
    as a plain Python list.
    """
    # --------------------------------------------------------------
    # deterministic randomness
    # --------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5        # safety margin

    # --------------------------------------------------------------
    # 0) build a pool of initial candidates
    # --------------------------------------------------------------
    init_candidates: list[np.ndarray] = []

    # use the previously discovered sequence if it exists
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            s = prev.sum()
            if s < 1e-12:
                prev = np.full(prev.shape, 1.0 / prev.shape[0])
            else:
                prev = prev / s
            init_candidates.append(prev.copy())
            print("[info] using previously discovered sequence as a start")
    except Exception:
        pass

    # a default two‑sided power‑law (moderate length)
    default_len = 12000
    default_seq = generate_two_sided_power_law(
        default_len, p_left=0.55, p_right=0.55, w_left=1.0, w_right=1.0
    )
    default_seq = np.maximum(default_seq, 0.0)
    default_seq /= default_seq.sum()
    init_candidates.append(default_seq.copy())

    # a handful of random Dirichlet draws (different lengths)
    length_options = [8000, 10000, 12000, 15000]
    while len(init_candidates) < 7 and time.time() < deadline:
        n = random.choice(length_options)
        alpha = random.uniform(0.2, 1.5)       # control sparsity
        seq = generate_dirichlet(n, alpha=alpha)
        init_candidates.append(seq)

    # --------------------------------------------------------------
    # 1) Optimise each candidate with the smooth‑max Adam optimiser
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left after initialisation")

    per_candidate = remaining / len(init_candidates)

    best_overall_seq: np.ndarray | None = None
    best_overall_val = float("inf")

    for idx, a0 in enumerate(init_candidates):
        if time.time() > deadline:
            break
        print(f"[optim] start candidate #{idx+1}/{len(init_candidates)} "
              f"(len={a0.size}) – time budget {per_candidate:.1f}s")

        seq, val = smooth_max_optimize(
            a0,
            time_limit=per_candidate,
            start_time=time.time(),
            beta_schedule=[5, 10, 20, 50, 100, 200, 500],
            lr0=0.7,
        )
        print(f"[optim] candidate #{idx+1} best score (fast) {val:.6f}")

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # --------------------------------------------------------------
    # 2) Final verification with the official evaluator
    # --------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – uniform distribution of length 10000
        n_fallback = 10000
        best_overall_seq = np.full(n_fallback, 1.0 / n_fallback)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")

    return final_list
