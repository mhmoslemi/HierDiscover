"""
Propose a low‑evaluation step‑function by a multi‑phase optimisation.

The algorithm
----------------
1.  Generate many parametric seeds (uniform, linear, exponential, power‑law,
    sinusoidal, Hann/Blackman windows, Gaussian, edge‑heavy, random Dirichlet).
    Keep the best few (by the fast surrogate 2·n·max(a*a)).
2.  For each seed run a continuation L‑BFGS on a smoothed max‑convolution
    (log‑sum‑exp) with a decreasing smoothing factor (λ‑schedule).  The
    variables are the log‑weights y, the actual distribution a = softmax(y)
    stays on the probability simplex.
3.  Refine the best result with a projected sub‑gradient descent that uses the
    exact max‑convolution to compute a sub‑gradient.
4.  Apply a simulated‑annealing phase that accepts occasional worsening moves.
5.  End with a greedy hill‑climb that transfers a small fraction of mass
    between random coordinates (always respecting the simplex).

All heavy operations use FFT‑based convolutions (O(n log n)).  The total time
budget (default 1000 s) is split automatically between the phases.  The final
sequence is returned as a plain Python list; it satisfies the non‑negativity
and sum‑positivity requirements of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1.  This is exactly the score returned by ``evaluate_sequence``
    for sum(a)=1, because the denominator cancels.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the autocorrelation using log‑sum‑exp.
    Returns (objective, gradient) where objective = 2·n·smooth_max.
    ``lam_factor`` multiplies the current max to obtain the smoothing λ.
    """
    a = softmax(y)

    # ---------- convolution a*a ----------
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ---------- smooth max ----------
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c  # for numerical stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ---------- gradient w.r.t. a ----------
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # slice to obtain the n‑dimensional gradient vector
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ---------- chain rule through soft‑max ----------
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS (multiple λ values)
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule: list[float],
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Runs L‑BFGS for each λ in ``lam_schedule`` starting from ``a_start``.
    The best sequence seen (lowest surrogate score) is returned.
    """
    # Initialise log‑weights y (soft‑max will enforce positivity)
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as exc:  # pragma: no cover
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score = score
            best_a = a.copy()

        # A short greedy hill‑climb after each λ can sharpen the result
        hc_deadline = min(deadline, time.time() + 0.3)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=1200,
        )

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (exact max‑convolution)
# ----------------------------------------------------------------------


def subgradient_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent with a simple
    back‑tracking line search.  All iterates stay on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and mask of active (max) indices
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass transfer)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    current_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that transfers a small random fraction of mass from one
    coordinate to another.  Moves are accepted only if they strictly improve the
    surrogate score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # transfer up to 5 % of the source mass
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / random perturbation
# ----------------------------------------------------------------------


def random_anneal(
    a: np.ndarray,
    score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10  # up to 10 % of source
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            # Metropolis acceptance
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 5000 == 0:
            T = max(T * 0.92, 1e-5)

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns a plain Python list of floats that respects all constraints.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (increase length for a potentially better bound)
    # --------------------------------------------------------------
    max_n = 4000
    # If a previous solution exists, we may reuse its length, but we prefer
    # the maximum allowed length for more flexibility.
    n = max_n
    a0 = np.full(n, 1.0 / n)  # start from uniform

    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for gradient / mask convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – cheap parametric sweep to collect promising seeds
    # --------------------------------------------------------------
    candidates = []

    def add_candidate(arr):
        arr = np.asarray(arr, dtype=float)
        if arr.sum() <= 0:
            return
        arr = np.maximum(arr, 0.0)
        arr = arr / arr.sum()
        sc = fast_score(arr, fft_len_conv, n)
        candidates.append((sc, arr))

    # 1) Uniform
    add_candidate(np.full(n, 1.0 / n))

    # 2) Linear decreasing
    lin = np.linspace(1.0, 0.0, n, dtype=float)
    add_candidate(lin)

    # 3) Exponential families
    for rate in [0.0005, 0.001, 0.0025, 0.005, 0.01]:
        add_candidate(np.exp(-rate * np.arange(n, dtype=float)))

    # 4) Power‑law families
    for p in [0.3, 0.6, 1.0, 1.5, 2.0, 2.5]:
        add_candidate((np.arange(1, n + 1, dtype=float) ** (-p)))

    # 5) Edge‑heavy
    for edge_frac in [0.004, 0.008, 0.016]:
        m = max(1, int(n * edge_frac))
        arr = np.zeros(n, dtype=float)
        arr[:m] = 1.0
        arr[-m:] = 1.0
        add_candidate(arr)

    # 6) Sinusoidal‑squared (always non‑negative)
    sin_arr = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    add_candidate(sin_arr + 1e-4)

    # 7) Hann window (raised cosine)
    hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    add_candidate(hann + 1e-5)

    # 8) Blackman window (more aggressive taper)
    a = 0.16
    b = 0.5
    c = 0.34
    n_idx = np.arange(n)
    black = (
        a
        - b * np.cos(2 * np.pi * n_idx / (n - 1))
        + c * np.cos(4 * np.pi * n_idx / (n - 1))
    )
    add_candidate(black + 1e-5)

    # 9) Gaussian bump centred in the middle
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((np.arange(n) - (n - 1) / 2) / sigma) ** 2)
    add_candidate(gauss + 1e-5)

    # 10) Random Dirichlet draws (various concentrations)
    for alpha in [0.3, 0.7, 1.5, 3.0]:
        for _ in range(3):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # Keep the top few seeds
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # --------------------------------------------------------------
    # Timing budget split
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.3   # safety margin
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    # Remaining time will be used for annealing + final hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # --------------------------------------------------------------
    lam_schedule = [
        0.5,
        0.35,
        0.25,
        0.15,
        0.10,
        0.07,
        0.05,
        0.03,
        0.02,
        0.015,
        0.008,
        0.004,
        0.002,
        0.001,
        0.0005,
    ]
    maxiter_per_lam = 500  # more iterations per λ than the previous version

    restarts = 0
    while time.time() < phase1_end and restarts < 15:
        # pick a promising seed and add a tiny jitter for diversity
        seed_vec = rng.choice(top_seeds)
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

        restarts += 1

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline)
        best_a, best_score = subgradient_optimize(
            best_a, sg_deadline, fft_len_conv, fft_len_g, n
        )
        print(f"[sub‑gradient] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing / random perturbations
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = random_anneal(
            best_a, best_score, fft_len_conv, n, deadline
        )
        print(f"[anneal] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing hill‑climb
    # --------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        final_deadline,
        max_iter=12000,
    )

    # --------------------------------------------------------------
    # Return the sequence as a plain Python list
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)  # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
