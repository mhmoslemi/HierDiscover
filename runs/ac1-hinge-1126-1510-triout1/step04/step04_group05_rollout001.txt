"""
Propose a low‑value step‑function using a hybrid of
* Adam optimisation on a smooth approximation of the max‑convolution,
* a geometric λ‑schedule (smooth‑max continuation),
* greedy targeted hill‑climbing that directly reduces the true max‑convolution,
* occasional random hill‑climbing for polishing.

The optimisation works on the unconstrained logits `y` of a soft‑max
parameterisation `a = softmax(y)`.  The objective is the smoothed version
of `2·n·max(a*a)`, which is exactly the score returned by
`evaluate_sequence` when the probabilities `a` sum to 1.
A simple Adam optimiser replaces the L‑BFGS step used in the reference
implementation; this allows many more cheap iterations inside the
1000 s time budget, especially when combined with a moderate sequence
length `n≈3000`.  After each continuation stage we perform a fast greedy
move that transfers mass from the biggest contributor to the smallest
contributor of the current convolution peak – this step reduces the
true (non‑smoothed) max‑convolution directly.  Finally a short random
hill‑climb polishes the result.

The algorithm keeps the best sequence seen across many random restarts
and returns it as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns the first ``2*n‑1`` entries (the full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of ``2·n·max(a*a)`` for a probability vector ``a``.
    This matches the value returned by ``evaluate_sequence`` (the denominator
    ``sum(a)**2`` equals 1).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of the max‑convolution and its gradient w.r.t.
    the unconstrained logits ``y`` (the log‑softmax coordinates).

    Parameters
    ----------
    y : ndarray
        Unconstrained parameters; a = softmax(y).
    n : int
        Length of the probability vector.
    fft_len_conv, fft_len_g : int
        FFT lengths (powers of two) for the convolution and the auxiliary
        convolution needed for the gradient.
    lam_factor : float
        Multiplicative factor for the smoothing width λ.

    Returns
    -------
    f : float
        Smoothed version of ``2·n·max(a*a)``.
    grad_y : ndarray
        Gradient of ``f`` w.r.t. ``y``.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c          # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of the smooth max -----
    w = exp_shift / Z                 # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def adam_optimize(
    y0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float,
    max_iter: int = 80,
    lr: float = 0.06,
) -> np.ndarray:
    """
    Perform Adam optimisation on the smooth max objective.

    Parameters
    ----------
    y0 : ndarray
        Starting logits.
    lam_factor : float
        Smoothing factor (see ``objective_and_grad``).
    max_iter : int
        Number of Adam steps.
    lr : float
        Base learning rate.

    Returns
    -------
    y : ndarray
        Optimised logits after ``max_iter`` Adam steps.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    for t in range(1, max_iter + 1):
        _, grad = objective_and_grad(
            y, n, fft_len_conv, fft_len_g, lam_factor=lam_factor
        )
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Generic random hill‑climb on the simplex.
    Randomly transfers a fraction of mass between two indices and keeps
    the move only if ``fast_score`` strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb.
    At each step the index contributing most to the current peak
    (k* = argmax conv) donates a small fraction of its mass to the index
    with the smallest contribution.  This directly reduces the true
    ``max(a*a)``.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        # current convolution and its peak location
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions to that peak: a_i * a_{k_max-i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # fallback to a random move if the contribution vector is flat
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta

        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    ``evaluate_sequence`` score.

    The routine runs many random restarts of an Adam optimiser with a
    geometric smooth‑max continuation schedule, followed by targeted and
    random hill‑climbs.  The best sequence encountered before the time
    limit is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Choose a problem size.
    # Larger n gives a smaller theoretical lower bound, but costs more
    # per iteration.  3000 works well within the budget.
    # --------------------------------------------------------------
    n = 3000
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # Initialise a first candidate.
    # If a previous construction exists, up‑sample it; otherwise start
    # from a simple power‑law shape.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        # Simple linear up‑sampling to length n
        x_old = np.linspace(0, 1, arr.size)
        x_new = np.linspace(0, 1, n)
        a0 = np.interp(x_new, x_old, arr)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 1e-12:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # power‑law decay (p≈1) – a common good starter
        p = 1.0
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()

    best_a = a0
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Continuation schedule for the smooth‑max factor λ.
    # --------------------------------------------------------------
    lam_factors = np.geomspace(0.3, 5e-6, num=26)   # decreasing λ‑factor
    adam_iters_per_stage = 70                      # cheap Adam steps per stage

    iteration = 0

    # --------------------------------------------------------------
    # Main multi‑restart optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline - 6.0:
        # ----- pick an initial point for this restart -----
        mode = rng.choice(["random", "exp", "power", "edge", "perturb"])

        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.018)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "power":
            p = rng.uniform(0.4, 1.8)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        elif mode == "edge":
            # concentrate a modest amount on both ends, flat centre
            edge_frac = rng.uniform(0.02, 0.07)
            a = np.full(n, (1.0 - 2 * edge_frac) / n)
            m_edge = max(1, int(n * edge_frac))
            a[:m_edge] = edge_frac / m_edge
            a[-m_edge:] = edge_frac / m_edge
            a /= a.sum()
            y0 = np.log(a + 1e-12)
        else:  # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        # ----- continuation Adam optimisation -----
        y_cur = y0
        for lam in lam_factors:
            y_cur = adam_optimize(
                y_start=y_cur,
                n=n,
                fft_len_conv=fft_len_conv,
                fft_len_g=fft_len_g,
                lam_factor=lam,
                max_iter=adam_iters_per_stage,
                lr=0.07,
            )
            # quick check after each stage in case a dramatic improvement appears
            a_tmp = softmax(y_cur)
            score_tmp = fast_score(a_tmp, fft_len_conv, n)
            if score_tmp < best_score:
                best_score = score_tmp
                best_a = a_tmp.copy()
                best_y = y_cur.copy()
                print(
                    f"[{iteration}] mode={mode} stage lam={lam:.2e} improved fast score={best_score:.6f}"
                )

        # ----- translate logits back to probabilities -----
        a_opt = softmax(y_cur)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_cur.copy()
            print(f"[{iteration}] mode={mode} L‑Adam improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb -----
        hc_deadline = min(deadline, time.time() + 0.7)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=2500,
            frac=0.07,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final deep refinement with a finer λ‑schedule
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        best_y = adam_optimize(
            y_start=best_y,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factor=lam,
            max_iter=120,
            lr=0.05,
        )
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑Adam] lam={lam:.2e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing: targeted then random hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=7000,
            frac=0.05,
        )
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=5000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Convert to list, evaluate with the official routine and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
