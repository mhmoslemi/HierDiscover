def edge_shape(n, m_edge, ratio):
    a = np.full(n, 1.0)
    # We'll solve for interior value b such that total sum = 1

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm consists of four phases

1. **Pattern exploration** – generate a few hundred simple
   parametric families (exponential, power‑law, linear decay,
   edge‑spikes, random Dirichlet).  The best of these gives a
   reasonably good starting point.

2. **Continuated L‑BFGS** – a multi‑restart L‑BFGS optimisation on a
   smooth approximation of the max‑convolution.  The smoothing
   parameter λ is decreased geometrically (continuation).

3. **Projected gradient / multiplicative descent** – fine‑tune the best
   candidate with many cheap gradient steps on the simplex.

4. **Aggressive hill‑climb + optional smoothing** – a greedy random
   mass‑transfer hill‑climb finishes the optimisation.  If a short
   moving‑average smoothing improves the score, the smoother version
   is kept.

The routine respects the 1000 s budget, periodically checks the
deadline and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max that returns a probability vector."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a probability vector (∑a = 1).
    This equals the value returned by ``evaluate_sequence`` for the same
    normalised sequence.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1.0 - cssv) / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / len(v)
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of the auto‑convolution and its gradient w.r.t. the
    unconstrained log‑softmax parameters y.
    Returns (value, grad_y).
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max = max_c + lam * log Σ exp((c_i - max_c)/lam)
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. conv -----
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j},   i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return float(f), grad_y.astype(np.float64)


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS optimisation over a decreasing λ‑schedule.
    """
    y = y_start.copy()
    for lam in lam_factors:
        # L‑BFGS on the smoothed objective
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move_frac: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb on the simplex.
    Randomly transfers a fraction of the mass from one index to another
    and keeps any strict improvement.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
    return best, best_score


def grad_smooth_max(a: np.ndarray, lam_factor: float) -> np.ndarray:
    """
    Gradient of the smooth max of the auto‑convolution w.r.t. a.
    Returns a vector of length n (no 2·n scaling).
    """
    n = a.shape[0]
    fft_len_conv = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    w = exp_shift / Z

    fft_len_g = next_pow2(3 * n - 2)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g = conv_rev[(n - 1) : (2 * n - 1)]

    # derivative of smooth max w.r.t a is 2 * g
    return 2.0 * g


# ----------------------------------------------------------------------
# Pattern generators (simple parametric families)
# ----------------------------------------------------------------------


def gen_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def gen_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.0005, 0.02)
    a = np.exp(-rate * np.arange(n, dtype=float))
    a /= a.sum()
    return a


def gen_powerlaw(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(0.1, 4.0)
    a = (np.arange(1, n + 1, dtype=float) ** (-p))
    a /= a.sum()
    return a


def gen_linear_decay(n: int, rng: np.random.Generator) -> np.ndarray:
    a = (np.arange(n, 0, -1, dtype=float))
    a /= a.sum()
    return a


def gen_linear_increase(n: int, rng: np.random.Generator) -> np.ndarray:
    a = (np.arange(1, n + 1, dtype=float))
    a /= a.sum()
    return a


def gen_edge(n: int, rng: np.random.Generator) -> np.ndarray:
    # two parameters: number of edge entries and ratio between edge /
    # interior value.
    m = max(1, int(n * rng.uniform(0.005, 0.07)))  # edge block size
    ratio = rng.uniform(2.0, 20.0)                # edge / interior factor
    # interior value b, edge value = ratio * b
    denom = (n - 2 * m) + 2 * m * ratio
    b = 1.0 / denom
    a = np.full(n, b)
    a[:m] = ratio * b
    a[-m:] = ratio * b
    return a


def gen_random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # gamma shape < 1 favours sparsity
    shape = rng.uniform(0.2, 1.0)
    a = rng.gamma(shape, 1.0, size=n)
    a /= a.sum()
    return a


def gen_two_spikes(n: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=float)
    i, j = rng.choice(n, size=2, replace=False)
    # allocate random split of total mass
    w = rng.random()
    a[i] = w
    a[j] = 1.0 - w
    return a


PATTERN_FUNCS = [
    gen_uniform,
    gen_exponential,
    gen_powerlaw,
    gen_linear_decay,
    gen_linear_increase,
    gen_edge,
    gen_random_dirichlet,
    gen_two_spikes,
]

# Main entry point used by the evaluation harness

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget (seconds).  The function stops before the budget expires.

    Returns
    -------
    list[float]
        The best non‑negative sequence found.
    """
    rng = np.random.default_rng(seed)
    # safety margin – we want to exit a few seconds before the hard deadline
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # Choose a problem size.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float).ravel()
        if arr.ndim == 1 and arr.size >= 5000:
            # keep the original length unless we want to enlarge it
            # but a modest increase often helps – cap at 20000 for speed
            n = min(max(arr.size, 12000), 20000)
        else:
            n = 15000
    else:
        n = 15000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (uniform – a safe fallback)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)
    print(f"[init] n={n}, start fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣  Pattern exploration (cheap, diverse starts)
    # ------------------------------------------------------------------
    pattern_budget = min(0.25 * budget_s, 60.0)               # seconds
    pattern_deadline = time.time() + pattern_budget
    attempts = 0
    while time.time() < pattern_deadline:
        attempts += 1
        # pick a random generator from the list
        gen = rng.choice(PATTERN_FUNCS)
        a = gen(n, rng) if gen.__code__.co_argcount == 2 else gen(n)
        cur = fast_score(a, fft_len_conv, n)
        if cur < best_score:
            best_score = cur
            best_a = a
            best_y = np.log(best_a + 1e-12)
            print(f"[pattern] attempt={attempts} new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Continuated L‑BFGS optimisation (global search)
    # ------------------------------------------------------------------
    lam_factors = np.geomspace(0.5, 1e-6, num=22)   # decreasing λ‑schedule
    maxiter_per_stage = 500                         # a little less than before
    iteration = 0
    while time.time() < deadline - 5.0:
        mode = rng.choice(
            ["random", "exp", "power", "edge", "linear", "reverse", "best"]
        )
        if mode == "random":
            y0 = rng.standard_normal(n)
        elif mode == "exp":
            a0 = gen_exponential(n, rng)
            y0 = np.log(a0 + 1e-12)
        elif mode == "power":
            a0 = gen_powerlaw(n, rng)
            y0 = np.log(a0 + 1e-12)
        elif mode == "edge":
            a0 = gen_edge(n, rng)
            y0 = np.log(a0 + 1e-12)
        elif mode == "linear":
            a0 = gen_linear_decay(n, rng)
            y0 = np.log(a0 + 1e-12)
        elif mode == "reverse":
            a0 = gen_linear_increase(n, rng)
            y0 = np.log(a0 + 1e-12)
        else:                     # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=n)

        y_opt = schedule_optimize(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=maxiter_per_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt
            best_y = y_opt
            print(
                f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}"
            )

        # ----- local hill‑climb (short) -----
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=hc_deadline,
            max_iter=4000,
            max_move_frac=0.20,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # 3️⃣  Fine‑grained projected‑gradient descent (multiplicative update)
    # ------------------------------------------------------------------
    lam_factor_pg = 1e-6
    step0 = 0.25
    max_pg_iter = 2000
    pg_iter = 0
    while time.time() < deadline - 2.0 and pg_iter < max_pg_iter:
        pg_iter += 1
        grad = grad_smooth_max(best_a, lam_factor_pg)   # gradient of smooth max
        # multiplicative (mirror‑descent) step on the simplex
        step = step0 / np.sqrt(pg_iter)
        cand = best_a * np.exp(-step * grad)
        cand /= cand.sum()
        cand_score = fast_score(cand, fft_len_conv, n)
        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            best_y = np.log(best_a + 1e-12)
            # tighten λ a little when we make progress
            lam_factor_pg = max(lam_factor_pg * 0.85, 1e-8)

    print(f"[pg] projected‑gradient iterations={pg_iter} best fast score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final polishing: hill‑climb + optional smoothing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            a=best_a,
            cur_score=best_score,
            fft_len=fft_len_conv,
            n=n,
            deadline=deadline,
            max_iter=20000,
            max_move_frac=0.25,
        )
        best_y = np.log(best_a + 1e-12)

    # Try a few moving‑average smoothings – keep the improvement if any
    for width in (3, 5, 7, 9):
        if time.time() >= deadline:
            break
        kernel = np.ones(width) / width
        smooth = np.convolve(best_a, kernel, mode="same")
        smooth = np.maximum(smooth, 0.0)
        smooth /= smooth.sum()
        s_score = fast_score(smooth, fft_len_conv, n)
        if s_score < best_score:
            best_a = smooth
            best_score = s_score
            best_y = np.log(best_a + 1e-12)
            print(f"[smooth] width={width} improved score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
