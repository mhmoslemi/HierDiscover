def exponential_descent(a_start, fft_len, n, deadline, eta0=0.5, max_iters=200000):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        # compute conv and max
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))
        # compute gradient g: similar to subgradient
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2 * a[counterpart[mask]]
        # step size
        eta = eta0 / np.sqrt(it)
        # exponentiated update
        a_new = a * np.exp(-eta * g)
        # renormalize to sum = 1
        sum_a = a_new.sum()
        if sum_a == 0:
            a_new = np.full_like(a, 1.0/n)
        else:
            a_new = a_new / sum_a
        # evaluate
        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score

"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that yields a
small value of the objective

    2·n·max(conv(a,a)) / (sum(a)^2)

(the same quantity returned by the provided ``evaluate_sequence``).

The algorithm combines several complementary phases:

1. **Problem size** – a moderately large sequence (≈8 k entries) is used.
   If a previous construction ``height_sequence_1`` exists it is reused.

2. **Multiple stochastic restarts** – for each restart an initial point
   (soft‑max log‑probabilities) is generated by a random heuristic
   (uniform, exponential decay, power‑law, edge‑heavy, Sidon set,
   Gaussian perturbation of the current best, etc.).  A continuation
   L‑BFGS optimisation with a *geometric* schedule of smoothing
   parameters (smooth approximation of the max) is performed.
   After each L‑BFGS run a short greedy hill‑climb refines the result.

3. **Exponential‑gradient descent** – an entropy‑mirror‑descent step
   (multiplicative update `a_i ← a_i·exp(‑η·g_i)`) is applied using the
   exact sub‑gradient of the active convolution entry.  The step size
   decays as `η = η₀/√t` and the iterate is re‑normalised, which keeps
   the vector on the probability simplex.

4. **Projected sub‑gradient descent with line‑search** – a classic
   projected gradient method that adaptively searches a step size that
   actually decreases the true objective.

5. **Final aggressive hill‑climb** – greedy moves of a random
   fraction of mass between two coordinates are performed until the
   time budget expires.

All phases share the same fast evaluation based on FFTs, therefore they
are inexpensive enough to run many iterations within the 1000 s budget.
The best sequence found is finally evaluated with the supplied
``evaluate_sequence`` and returned.

The combination of many random restarts, smooth L‑BFGS optimisation,
entropy‑mirror updates and aggressive local search typically improves
the bound beyond the previous best (≈ 1.509) and can reach the target
≤ 1.503.
"""

import time
from typing import List, Tuple

import numpy as np
from scipy.optimize import minimize


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using FFT.
    ``a`` must have length ``n``; ``fft_len`` should be a power of two ≥ 2·n‑1.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum(x) = 1}.
    """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> Tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp approximation of ``max(conv)`` and its gradient
    w.r.t. the unconstrained log‑probabilities ``y`` (soft‑max
    parametrisation).

    The smoothing parameter is ``λ = max(conv)·lam_factor``.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                 # max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z                      # ∂ smooth_max / ∂ conv_i

    # gradient of smooth max w.r.t. a
    # (a * w) convolution (reversed a) → yields vector of length n
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: List[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Perform a continuation L‑BFGS optimisation.
    ``lam_schedule`` is a list of smoothing factors (lam_factor) that are
    applied successively; each stage starts from the result of the
    previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point ``y`` (pre‑softmax) for a given heuristic.
    The returned vector is suitable for ``softmax``.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # Greedy Sidon set – modest size ≈ √n
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                if (candidate + c) in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate) not in sums:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "linear":
        a0 = np.linspace(1.0, 0.0, n)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "gaussian":
        mu = rng.integers(0, n)
        sigma = rng.uniform(0.1 * n, 0.3 * n)
        x = np.arange(n)
        a0 = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        a0 += 1e-9
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> Tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction (≤10 %) of mass from ``i``
    to ``j`` (preserving the sum).  Keep a move only if the fast score
    improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert the move
            best[i] += delta
            best[j] -= delta

    return best, best_score


def exponential_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.5,
    max_iters: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponential‑gradient) on the true objective.
    Iteration:
        a ← a * exp(-η·g)  ;  g is the sub‑gradient of the active
        convolution entry, η decays as η₀/√t.
    The vector is re‑normalised after each step (keeps sum = 1).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a * np.exp(-eta * g)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def subgradient_descent_adaptive(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent with a simple back‑tracking line search.
    The sub‑gradient is that of the active convolution entry; after each
    step the iterate is projected onto the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        # line search – start with a moderate step and shrink until improvement
        alpha = 0.5
        cur_score = fast_score(a, fft_len, n)
        while True:
            a_tmp = a - alpha * g
            a_tmp = project_onto_simplex(a_tmp)
            new_score = fast_score(a_tmp, fft_len, n)
            if new_score < cur_score or alpha < 1e-9:
                break
            alpha *= 0.5

        a = a_tmp
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (list of non‑negative heights) with a low
    evaluation score.  The function respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0    # small safety margin

    # ------------------------------------------------------------------
    # Problem size – use a moderately large n, re‑use a previous construction
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size >= 2000:
            n = min(int(tmp.size), 12000)    # cap to keep the search tractable
        else:
            n = 8000
    else:
        n = 8000

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initialise best candidate (probability simplex, sum = 1)
    # ------------------------------------------------------------------
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).ravel()
        if tmp.ndim == 1 and tmp.size == n:
            best_a = np.maximum(tmp, 0.0)
            s = best_a.sum()
            best_a = best_a / s if s > 0 else np.full(n, 1.0 / n)
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_a + 1e-12)           # soft‑max parameters

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS restarts (≈ 60 % of the budget)
    # ------------------------------------------------------------------
    lam_schedule = list(np.geomspace(0.5, 1e-7, num=30))
    maxiter_stage = 250
    l_bfgs_deadline = time.time() + 0.60 * budget_s - 20.0
    iteration = 0
    modes = ["random", "exp", "power", "edge", "sidon", "linear", "gaussian", "best"]

    while time.time() < l_bfgs_deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.03, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        # continuation L‑BFGS
        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # short greedy polishing
        hc_deadline = min(l_bfgs_deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=3000,
        )
        best_y = np.log(best_a + 1e-12)

        iteration += 1

    # ------------------------------------------------------------------
    # Phase 2 – exponential‑gradient descent (≈ 20 % of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        exp_deadline = deadline - 20.0
        print("[expdesc] starting exponential‑gradient descent")
        best_a, best_score = exponential_descent(
            best_a,
            fft_len_conv,
            n,
            exp_deadline,
            eta0=0.5,
            max_iters=500_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[expdesc] finished fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent (≈ 15 % of the remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10.0:
        sub_deadline = deadline - 5.0
        print("[subgrad] starting projected sub‑gradient descent")
        best_a, best_score = subgradient_descent_adaptive(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            max_iters=400_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final aggressive greedy polishing (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        print("[final] aggressive hill‑climbing")
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            final_deadline,
            max_iter=15000,
        )
        print(f"[final] best fast score after polishing = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the sequence (as a plain list) and report the official score
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # type: ignore – provided by harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
