"""
Propose a low‑score non‑negative step sequence.

The optimizer is a multi‑phase pipeline.
1.  Many cheap L‑BFGS runs on a smooth (log‑sum‑exp) surrogate of the
    max‑convolution, using a long continuation schedule (λ down to 1e‑4).
2.  A projected momentum sub‑gradient descent.
3.  Adam on the same smooth surrogate (fine‑tuning).
4.  One or two linear‑programming refinement steps based on the
    “LP‑refinement” described in the statement (maximise ‖g‖₁ subject
    to ‖f★g‖∞ ≤ M).  The LP is solved with SciPy’s high‑performance
    ‘highs’ solver.
5.  An extended simulated‑annealing / greedy hill‑climb (many ×10⁴
    mass‑transfer moves).  Targeted moves that directly attack the
    current maximal convolution index are also tried.

All intermediate iterates stay on the probability simplex
(∑aᵢ = 1, aᵢ ≥ 0), so the evaluation score
    2·n·max(a★a) / (∑a)²
reduces to 2·n·max(a★a).  The routine runs for the full time budget
(≈1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """2·n·max(a★a) for a already normalised to sum = 1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smooth‑max of the convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                     # ∂smooth_max / ∂conv_i

    # back‑propagation through convolution (a on both sides)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # only the central n entries correspond to derivative w.r.t. a
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Continuation L‑BFGS
# ----------------------------------------------------------------------


def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous one.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(fun=objective_and_grad,
                       x0=y,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': maxiter_per_lam,
                                'gtol': 1e-12,
                                'disp': False})
        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent with momentum
# ----------------------------------------------------------------------


def subgradient_optimize_momentum(a0: np.ndarray,
                                  deadline: float,
                                  fft_len_conv: int,
                                  fft_len_g: int,
                                  n: int):
    """
    Projected sub‑gradient descent with simple momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # linear convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # identify (approximately) active constraints
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # step size ~ 1/√it
        step = 0.5 / np.sqrt(it + 1.0)
        a_candidate = simplex_projection(a - step * v)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
        a = a_candidate
        it += 1

    return best_a, best_score

def adam_optimize(y0: np.ndarray,
                  n: int,
                  fft_len_conv: int,
                  fft_len_g: int,
                  deadline: float,
                  lam_factor: float = 0.002,
                  lr: float = 0.15):
    """
    Adam optimiser of the smoothed max objective.
    """
    y = y0.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        f, grad = objective_and_grad(y, n,
                                     fft_len_conv, fft_len_g,
                                     lam_factor)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        y = y - step

        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Random initialiser
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random distribution on the simplex.
    Several families are mixed to increase diversity.
    """
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal', 'spike'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # spike – a very short heavy front, long tail
        a = np.full(n, 1.0 / n)
        pos = rng.integers(0, n)
        a[pos] = 0.5
        a = a / a.sum()
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Simulated annealing
# ----------------------------------------------------------------------


def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float):
    """
    Randomly move a fraction of mass between two coordinates,
    accepting worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score
        it += 1
        if it % 4000 == 0:
            T *= 0.92
            T = max(T, 1e-5)

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb (mass‑transfer) – extended version
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 40000):
    """
    Greedy hill‑climb with many random mass transfers.
    Accepts only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # move a fraction (up to 5 %) of the mass at i
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (the “LP” step from the statement)
# ----------------------------------------------------------------------


def lp_refine(f: np.ndarray,
              target_sum: float,
              deadline: float,
              max_iters: int = 5):
    """
    One‑step LP refinement: maximise ‖g‖₁ under ‖f★g‖∞ ≤ M,
    mix the result back into f.  Returns a possibly improved f.
    """
    n = f.shape[0]
    fft_len = next_pow2(2 * n - 1)

    best_f = f.copy()
    best_score = fast_score(best_f, fft_len, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current bound M
        conv_f = full_conv(best_f, fft_len)
        M = conv_f.max()

        # Build constraint matrix A where A[k, j] = f_{k-j} if 0≤k-j<n else 0
        rows = []
        cols = []
        data = []
        for k in range(2 * n - 1):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            idxs = np.arange(start, end + 1, dtype=int)
            rows.extend([k] * len(idxs))
            cols.extend(idxs.tolist())
            vals = best_f[k - idxs]
            data.extend(vals.tolist())
        A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # Solve LP: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status != 0:
            # LP failed – abort refinement
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        # Normalise g to the target sum (our sequences always sum to 1)
        g = g * (target_sum / sum_g)

        # Mix – try a few mixing ratios, keep the best improvement
        improved = False
        for t in (0.5, 0.2, 0.1, 0.05):
            f_candidate = (1.0 - t) * best_f + t * g
            sc = fast_score(f_candidate, fft_len, n)
            if sc < best_score:
                best_f = f_candidate
                best_score = sc
                improved = True
                # early exit from mixing loop
                break
        if not improved:
            # No improvement with any mixing ratio ⇒ stop LP refinement
            break
        it += 1

    return best_f, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use a moderate size – 2000 works well)
    # --------------------------------------------------------------
    max_n = 4000
    n = max_n // 2                # 2000 by default

    # If a previous solution exists, start from it
    if 'height_sequence_1' in globals():
        init_seq = np.array(globals()['height_sequence_1'], dtype=float)
        if init_seq.ndim == 1 and init_seq.size > 0:
            n = min(init_seq.size, max_n)
            a0 = init_seq[:n].copy()
            a0[a0 < 0] = 0.0
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Time budget split
    # --------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s   # L‑BFGS + random restarts
    phase2_end = start_time + 0.80 * budget_s   # sub‑gradient + Adam
    # remaining time → LP + anneal + hill‑climb

    # --------------------------------------------------------------
    # Phase 1 – many L‑BFGS restarts with a fine continuation schedule
    # --------------------------------------------------------------
    lam_schedule = [0.5, 0.35, 0.25, 0.15, 0.10,
                    0.07, 0.045, 0.03, 0.018, 0.010,
                    0.006, 0.003, 0.0015, 0.0007,
                    0.0003, 0.0001]
    maxiter_per_lam = 350

    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)
        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – momentum sub‑gradient descent followed by Adam
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = phase2_end - 0.5
        best_a, best_score = subgradient_optimize_momentum(best_a,
                                                            sg_deadline,
                                                            fft_len_conv,
                                                            fft_len_g,
                                                            n)
        print(f"[sub‑gradient] score={best_score:.6f}")

        if time.time() < phase2_end:
            adam_deadline = phase2_end - 0.2
            logits = np.log(best_a + 1e-12)
            best_a, best_score = adam_optimize(logits,
                                               n,
                                               fft_len_conv,
                                               fft_len_g,
                                               adam_deadline,
                                               lam_factor=0.002,
                                               lr=0.15)
            print(f"[Adam] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – LP refinement (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        lp_deadline = time.time() + min(30.0, deadline - time.time() - 5.0)
        best_a, best_score = lp_refine(best_a, 1.0, lp_deadline, max_iters=6)
        print(f"[LP‑refine] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing (short)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        anneal_deadline = time.time() + min(8.0, deadline - time.time() - 2.0)
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           anneal_deadline)
        print(f"[anneal] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – extended greedy hill‑climb (many iterations)
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_deadline = deadline
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len_conv,
                                        n,
                                        final_deadline,
                                        max_iter=80000)
        print(f"[final hill‑climb] score={best_score:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
