"""
Propose a step‑function (non‑negative sequence) with a low evaluation score.

Improvements over the reference implementation
------------------------------------------------
* Larger problem size (n≈8000) gives a richer search space.
* Longer L‑BFGS continuation (more smoothing stages, more iterations per stage).
* Additional “mix” start mode that perturbs the current best candidate.
* Two hill‑climb procedures:
    - a fast random move (as before) with a larger transfer budget,
    - a *targeted* move that preferentially draws mass from indices that
      contribute most to the current maximal convolution entry.
* Sub‑gradient descent is given a larger time budget and a slightly larger
  initial step size.
* All time‑budget splits are slightly relaxed to spend more time on the
  stochastic optimisation phase (the part that gave most of the improvements).

The algorithm respects the 1000‑second budget, works on the probability
simplex (∑a_i = 1) and returns a plain list of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax yielding a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    # keep only the real part (numerical noise may introduce a tiny imag)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector (∑a=1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection onto the probability simplex {x≥0, sum(x)=1}. """
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective 2·n·smooth_max(conv) and its gradient wrt y.
    lam_factor controls the amount of smoothing: lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c            # max shifted to zero
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient wrt conv entries
    w = exp_shift / Z                 # ∂smooth_max / ∂conv_i

    # convolution of a with w (needed for ∂conv/∂a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # back‑prop through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 300,
) -> np.ndarray:
    """
    Perform a series of L‑BFGS runs with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y

# Initialisation helpers
# ----------------------------------------------------------------------


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a start point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set (size ≈ √n)
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "mix":
        # will be handled outside – placeholder
        raise ValueError("mix mode handled separately")
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Hill‑climbing utilities
# ----------------------------------------------------------------------


def hill_climb_random(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    max_move: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Simple greedy hill‑climb.  Randomly move a fraction of mass from i to j.
    The move is kept only if the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def hill_climb_targeted(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
    max_move: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that concentrates on the index responsible for the
    current maximal convolution entry.  At each step we pick a source
    coordinate that contributes heavily to the max entry and move a
    random fraction of its mass to a random destination.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        # compute convolution and the index of its maximum
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # contribution of each coordinate to conv[m]
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[counterpart[mask]]
        # bias source selection towards large contributions
        if contrib.sum() == 0:
            src = rng.integers(0, n)
        else:
            prob = contrib / contrib.sum()
            src = rng.choice(n, p=prob)

        dst = rng.integers(0, n)
        if src == dst:
            continue

        delta = best[src] * rng.random() * max_move
        if delta <= 0.0:
            continue

        best[src] -= delta
        best[dst] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[src] += delta
            best[dst] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.30,
    max_iters: int = 600_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on f(a)=max_i (a*a)_i.
    Uses diminishing step size η_t = η0/√t and projects onto the simplex.
    Returns the best iterate encountered.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the max entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()
        a = a_new
    return best_a, best_score
