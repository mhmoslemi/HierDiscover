    rows = []
    cols = []

        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)

    b = res.x
    return b

def solve_lp_for_b(f, M, b_max=1000.0):
    n = len(f)
    # Build matrix A via diags
    offsets = np.arange(n)  # 0..n-1
    data = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, b_max) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros
        return np.zeros(n)
    return res.x
```

def simplex_projection(v):
    # v is array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid sub‑gradient / LP optimiser for the max‑auto‑convolution problem.

The algorithm works on a probability vector a (sum = 1, a_i ≥ 0) of length n.
It alternates three ingredients inside the time budget:

1. **Projected sub‑gradient step** – the exact sub‑gradient of the current
   maximal convolution entry is computed and a step is taken in the opposite
   direction.  The step is projected back onto the simplex (non‑negative,
   sum = 1) using an O(n log n) projection.

2. **LP refinement** – for the current a we solve a linear programme that
   maximises Σ b_i under the linear constraints
        (a * b)_k ≤ M   (k = 0 … 2n‑2) ,
   where M = max_k (a * a)_k.  The solution b is normalised to sum 1 and
   mixed with a (mixing factor t≈0.1).  This “direction” is guaranteed not to
   raise the maximal convolution value for a sufficiently small t.

3. **Targeted hill‑climb** – a fast greedy move that transfers a small
   fraction of mass from the biggest contributor to the smallest one for
   the current peak index.  It directly reduces the true max‑convolution
   and polishes the solution after each major update.

All three phases are repeated while time remains.  The best sequence seen
is returned and finally evaluated by the supplied ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
from math import log2, ceil

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution of a with itself via FFT (real part returned)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective (2·n·max(conv))."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full(n, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Construct the (2n‑1)×n matrix A such that (A @ b) = conv(f,b).
    Each diagonal offset d (0 ≤ d < n) contains the constant f[d].
    """
    n = f.shape[0]
    offsets = np.arange(n)                     # 0 … n‑1
    data = [np.full(n, f[d]) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def solve_lp_for_b(f: np.ndarray, M: float, b_ub: float = 1000.0) -> np.ndarray:
    """
    Linear programme:
        maximise Σ b_i
        s.t.   (f * b)_k ≤ M   for all k
               0 ≤ b_i ≤ b_ub
    Returns the optimal vector b (may be all zeros if the LP fails).
    """
    n = f.shape[0]
    A = build_conv_matrix(f)                     # shape (2n‑1, n)
    c = -np.ones(n)                              # maximise Σb  → minimise –Σb
    bounds = [(0.0, b_ub) for _ in range(n)]
    rhs = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=rhs, bounds=bounds, method="highs")
    if not res.success:
        # fall back to a feasible point (zero vector)
        return np.zeros(n, dtype=float)
    return res.x


# ----------------------------------------------------------------------
# Targeted hill‑climb (same as in the reference code)
# ----------------------------------------------------------------------


def conv_full(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Convenience wrapper returning the full linear convolution."""
    return conv_fft(a, fft_len, n)


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb:
    transfer a small fraction of mass from the largest contributor
    to the smallest one for the current peak index.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_moves > 0:
        conv = conv_full(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contributions a_i * a_{k*‑i}
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            # flat, pick two random indices
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1
    return best, best_score


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence with a low
    ``evaluate_sequence`` score.  The routine mixes projected sub‑gradient
    descent, a specialised LP step and a targeted hill‑climb within the
    allotted time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    # --------------------------------------------------------------
    # Problem size and initialisation
    # --------------------------------------------------------------
    default_n = 2000
    # Use the provided construction if it exists; otherwise start uniform.
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            n = default_n
            a = arr[:n].copy()
        else:
            n = default_n
            a = np.full(n, 1.0 / n)
    else:
        n = default_n
        a = np.full(n, 1.0 / n)

    # Force strict positivity (helps the LP) and normalise to sum = 1.
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()

    best_a = a.copy()
    fft_len = next_pow2(2 * n - 1)
    best_score = fast_score(best_a, fft_len, n)
    print(f"[init] n={n} score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    subgrad_step = 0.2  # initial step size for the sub‑gradient phase

    while time.time() < deadline - 5.0:
        # ----------------------------------------------------------
        # 1) Projected sub‑gradient descent on the exact max‑conv.
        # ----------------------------------------------------------
        conv = conv_full(best_a, fft_len, n)
        k_max = int(np.argmax(conv))

        # sub‑gradient of conv[k_max] w.r.t. a
        grad = np.zeros_like(best_a)
        # a_{k_max-i} term
        i_low = max(0, k_max - (n - 1))
        i_high = min(k_max, n - 1)
        for i in range(i_low, i_high + 1):
            j = k_max - i
            grad[i] += best_a[j]
            grad[j] += best_a[i]  # symmetric contribution
        # The above double‑counts the diagonal term when i==j; that's fine
        # because the exact derivative of a_i*a_i appears twice in the sum.

        # Take a step opposite to the sub‑gradient and project back.
        candidate = best_a - subgrad_step * grad
        candidate = simplex_projection(candidate)

        cand_score = fast_score(candidate, fft_len, n)
        if cand_score < best_score:
            best_a = candidate
            best_score = cand_score
            # slowly shrink step size when we make progress
            subgrad_step = max(0.01, subgrad_step * 0.9)
            print(f"[{iteration}] sub‑grad improved → {best_score:.6f}")
        else:
            # no improvement – reduce step size more aggressively
            subgrad_step *= 0.6
            if subgrad_step < 1e-4:
                subgrad_step = 0.01

        # ----------------------------------------------------------
        # 2) LP refinement (once every few iterations)
        # ----------------------------------------------------------
        if iteration % 5 == 0:
            conv = conv_full(best_a, fft_len, n)
            M = conv.max()
            b = solve_lp_for_b(best_a, M, b_ub=1000.0)
            sum_b = b.sum()
            if sum_b > 0:
                g = b / sum_b                     # normalised to sum 1
                t = 0.1
                mixed = (1.0 - t) * best_a + t * g
                mixed = simplex_projection(mixed)
                mixed_score = fast_score(mixed, fft_len, n)
                if mixed_score < best_score:
                    best_a = mixed
                    best_score = mixed_score
                    print(f"[{iteration}] LP mix improved → {best_score:.6f}")

        # ----------------------------------------------------------
        # 3) Targeted hill‑climb polishing
        # ----------------------------------------------------------
        if time.time() < deadline - 2.0:
            best_a, best_score = targeted_hill_climb(
                best_a,
                best_score,
                fft_len,
                n,
                deadline,
                max_moves=1500,
                frac=0.08,
            )

        iteration += 1

    # --------------------------------------------------------------
    # Final polishing stage (more hill‑climb moves)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len,
            n,
            deadline,
            max_moves=8000,
            frac=0.06,
        )
        # Final safety projection to guarantee non‑negativity and sum 1.
        best_a = simplex_projection(best_a)

    # --------------------------------------------------------------
    # Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
