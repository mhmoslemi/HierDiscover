"""
propose_candidate
=================

Search for a non‑negative step‑function (a list of heights) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

which is exactly the value returned by ``evaluate_sequence`` when the
sequence is normalised to sum = 1.  The algorithm is a two‑phase
gradient‑based optimisation:

1.  **Random initialisation** – a small pool of candidates is generated
    from several analytic families (power‑law, exponential, two‑sided
    power‑law, Beta‑pdf) with different lengths.  The previously found
    sequence (`height_sequence_1`) is added if it exists.

2.  **Smoothed Adam optimisation** – the coefficients are parametrised by
    logits ``y`` and the soft‑max ``a = softmax(y)`` guarantees non‑negative
    entries that sum to one.  At each iteration the true convolution
    ``c = a * a`` (computed by FFT) is used to build a smooth approximation
    of the max:

        M_eps = eps * log( Σ_k exp(c_k/eps) )

    with a gradually decaying ``eps`` (initially a fraction of the current
    maximum and then reduced exponentially).  The gradient of the
    smoothed objective is obtained analytically; it uses a single additional
    FFT to compute the cross‑correlation between the weight vector
    (soft‑max of the convolution values) and ``a``.  Adam updates are
    applied to the logits.  Occasionally a small Gaussian jitter is added
    to escape plateaus.

The total budget (≈ 1000 s) is split equally among the pool of candidates.
The best sequence seen over all runs is finally verified with the provided
``evaluate_sequence`` function and returned as a plain Python list.

The implementation uses only NumPy and the standard library, respects the
2‑CPU limit, performs no I/O and prints progress information.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Quick evaluation of the objective 2·n·max(conv(seq, seq)) / (sum(seq)**2).
    The sequence is assumed to be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Shape generators (used for random starts)
# ----------------------------------------------------------------------


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    return idx ** (-p)


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    return np.exp(-lam * idx)


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))


# ----------------------------------------------------------------------
# Smoothed gradient helpers
# ----------------------------------------------------------------------


def compute_gradient_smooth(a: np.ndarray, conv: np.ndarray, eps: float) -> np.ndarray:
    """
    Gradient of the smoothed maximum with respect to the probabilities a.
    ``conv`` is the full convolution a * a (length 2n‑1), ``eps`` is the
    smoothing parameter.
    Returns a vector g of size n where
        g_i = Σ_k w_k * d(conv_k)/da_i,
    w_k = exp(conv_k/eps) / Σ_j exp(conv_j/eps).
    """
    n = a.shape[0]
    L = conv.shape[0]                # = 2n‑1
    # ----- weights for the smoothed max -----
    M = conv.max()
    # shift for numerical stability
    w = np.exp((conv - M) / eps)
    w_sum = w.sum()
    if w_sum == 0 or not np.isfinite(w_sum):
        # fallback to uniform over the (tiny) active set
        active = np.where(conv >= M - 1e-12)[0]
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
    else:
        w /= w_sum

    # ----- cross‑correlation a ⋆ w  (g_i needs Σ_j a_j w_{i+j}) -----
    w_rev = w[::-1]
    nfft = next_pow2(n + L - 1)
    fa = np.fft.rfft(a, n=nfft)
    fw = np.fft.rfft(w_rev, n=nfft)
    cross_full = np.fft.irfft(fa * fw, n=nfft)[: n + L - 1]

    # indices to extract Σ_j a_j w_{i+j}
    idx = L - 1 - np.arange(n)
    cross_vals = cross_full[idx]                 # shape (n,)

    # ----- derivative of each quadratic term -----
    # for k = 2*i the derivative is a_i (not doubled)
    diag_term = w[2 * np.arange(n)] * a

    g = 2.0 * cross_vals - diag_term
    return g


# ----------------------------------------------------------------------
# Core optimiser (Adam on the soft‑max parametrisation with smoothing)
# ----------------------------------------------------------------------


def smooth_adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.4,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps_adam: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed max‑convolution objective.
    Returns the best sequence seen and its exact (unsmoothed) score.
    """
    n = a0.shape[0]

    # initialise logits (soft‑max variables)
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # bookkeeping
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0

    # schedule parameters for the smoothing epsilon
    rel_eps_start = 0.30          # initial ε ≈ 0.30 * current max
    rel_eps_min = 1e-4
    decay = 0.9995                # exponential decay per iteration

    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ----- soft‑max (guarantees sum = 1) -----
        y_shift = y - np.max(y)                # for numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # ----- convolution a * a (FFT) -----
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- smoothing parameter ε -----
        rel_eps = max(rel_eps_min, rel_eps_start * (decay ** it))
        eps = max(rel_eps * M, 1e-12)

        # ----- gradient of the smoothed objective -----
        g = compute_gradient_smooth(a, conv, eps)

        # ----- gradient wrt logits (soft‑max Jacobian) -----
        ag = float(np.dot(a, g))               # scalar ⟨a,g⟩
        grad_y = a * (g - ag)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional jitter to escape possible plateaus
        if it % 2000 == 0:
            sigma = 0.02 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

            break
        print(f"[optim] candidate #{idx+1}/{len(init_candidates)} "
              f"(len={a0.size}) – time budget {per_candidate:.1f}s")
        seq, val = smooth_adam_optimize(a0, per_candidate, time.time())
        print(f"[optim] candidate #{idx+1} best score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final verification with the official evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        # fallback – uniform distribution of a modest size
        n = 12000
        best_overall_seq = np.full(n, 1.0 / n)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")
    return final_list
