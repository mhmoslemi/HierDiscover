"""
Improved search for a low‑value step function.

The algorithm builds on the previous L‑BFGS + hill‑climb + sub‑gradient
pipeline but makes several upgrades:

* A much broader smoothing schedule (lam_factor from 20 down to 1e‑7)
  and more L‑BFGS iterations per stage.
* Richer initial‑point generators (Gaussian, sinusoidal, etc.).
* A targeted greedy hill‑climb that moves mass from indices that
  contribute heavily to the current maximal convolution entry to
  indices that contribute little.
* Faster and more aggressive projected sub‑gradient descent
  (larger initial step, more iterations).
* A final simulated‑annealing polishing phase.

All vectors are kept on the probability simplex (sum = 1); the
objective 2·n·max(conv) is therefore identical to the value returned by
``evaluate_sequence``.  The routine runs for up to the allotted time
(≈ 1000 s) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed log‑sum‑exp objective (2·n·smooth_max) and its gradient
    w.r.t. the unconstrained vector y (soft‑max parametrisation).

    lam = max(conv) * lam_factor.
    """
    a = softmax(y)

    # forward convolution
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c                    # max = 0 now
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    w = exp_shift / Z   # ∂ smooth_max / ∂ conv_i

    # gradient w.r.t. a via convolution with w
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # entries (n‑1) … (n‑1+n‑1) correspond to the gradient map
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Run a series of L‑BFGS optimisations with decreasing smoothing.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-12,
                "disp": False,
            },
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a starting point y (pre‑softmax) for a given mode.
    """
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # modest greedy Sidon set (≈√n)
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * candidate not in sums):
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "gaussian":
        sigma = rng.uniform(0.1, 0.5) * n
        x = np.arange(n)
        centre = n // 2
        a0 = np.exp(-((x - centre) ** 2) / (2.0 * sigma ** 2))
        a0 += 1e-12
        a0 /= a0.sum()
        y0 = np.log(a0)
    elif mode == "sin":
        freq = rng.uniform(1, 5)
        a0 = np.abs(np.sin(np.linspace(0, freq * np.pi, n)))
        a0 += 1e-12
        a0 /= a0.sum()
        y0 = np.log(a0)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb focusing on the indices that dominate the current
    maximal convolution entry.  Moves transfer a fraction of mass from a
    “heavy” source index to a “light” target index and are kept only if
    the fast_score improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move = 0.30               # up‑to‑30 % of source mass per move
    source_k = 30                 # consider the top‑k contributors
    target_k = 30                 # and the bottom‑k contributors

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        # current convolution and the index of its maximum
        conv = conv_fft(best, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of the max entry (2·a_{m‑i})
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(best)
        g[mask] = 2.0 * best[counterpart[mask]]

        # source candidates: large gradient & non‑zero mass
        possible_src = np.where(best > 1e-12)[0]
        if possible_src.size == 0:
            break
        src_order = np.argsort(-g[possible_src])
        src_candidates = possible_src[src_order[:source_k]]

        # target candidates: small gradient (preferably also small mass)
        tgt_order = np.argsort(g)
        tgt_candidates = tgt_order[:target_k]

        i = rng.choice(src_candidates)
        j = rng.choice(tgt_candidates)
        if i == j:
            continue

        delta = best[i] * rng.random() * max_move
        if delta <= 0.0:
            continue

        # propose move
        trial = best.copy()
        trial[i] -= delta
        trial[j] += delta
        trial_score = fast_score(trial, fft_len, n)

        if trial_score < best_score:
            best = trial
            best_score = trial_score

    return best, best_score


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.40,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        m = int(np.argmax(conv))

        # sub‑gradient of max entry
        idx = np.arange(n)
        counterpart = m - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing polishing.  Random mass transfers are
    accepted with probability exp(−Δ/ T); temperature T decays linearly.
    """
    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = start_score
    rng = np.random.default_rng()
    total = max(deadline - time.time(), 1e-12)

    while time.time() < deadline:
        # linear cooling schedule
        elapsed = time.time() - (deadline - total)
        frac = elapsed / total
        T = (1.0 - frac) * 0.01 + 1e-12

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.20
        if delta <= 0.0:
            continue

        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)

        dE = new_score - cur_score
        if dE < 0 or rng.random() < np.exp(-dE / T):
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best = cur.copy()
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta

    return best, best_score

            best_y = y_opt.copy()
            print(f"[{it:04d}] mode={mode} improved fast score = {best_score:.6f}")

        # quick targeted hill‑climb
        hc_deadline = min(deadline, time.time() + 0.8)
        best_a, best_score = targeted_hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=2000,
        )
        best_y = np.log(best_a + 1e-12)
        it += 1

    # ------------------------------------------------------------------
    # Exact sub‑gradient descent phase
    # ------------------------------------------------------------------
    if time.time() < deadline - 60.0:
        sub_deadline = deadline - 30.0
        print("[subgrad] starting projected sub‑gradient descent")
        best_a, best_score = subgradient_descent(
            best_a,
            fft_len_conv,
            n,
            sub_deadline,
            eta0=0.40,
            max_iters=800_000,
        )
        best_y = np.log(best_a + 1e-12)
        print(f"[subgrad] finished; best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final annealing / polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[anneal] final simulated annealing")
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
        )
        print(f"[final] best fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)   # provided by the harness
    print(f"[result] evaluated score = {final_val:.6f}")
    return final_seq
