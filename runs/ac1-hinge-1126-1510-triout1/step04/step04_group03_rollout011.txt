"""
search for a low‑score step sequence.

The algorithm works in several stages:

1. **Parametric sweep** – generate a handful of promising seeds from simple
   families (uniform, linear, exponential, power‑law, edge‑heavy,
   sinusoid‑squared, raised‑cosine, beta‑distribution and random Dirichlet).

2. **Continuation L‑BFGS** – minimise a smooth approximation of the
   max‑convolution (log‑sum‑exp) using many λ‑values (geometric schedule)
   and a generous number of iterations per λ.  Each restart starts from a
   different seed with a tiny jitter.

3. **Long sub‑gradient descent** – direct minimisation of the exact
   max‑convolution using the sub‑gradient obtained from the active set.
   A simple back‑tracking line‑search and Euclidean projection onto the
   probability simplex are applied.

4. **Simulated annealing** – random mass‑transfer moves are accepted
   according to a decreasing temperature schedule, helping to escape any
   remaining shallow local minima.

5. **Final hill‑climb** – deterministic greedy moves that only keep strictly
   improving candidates are performed until the time budget expires.

All stages respect the constraints required by ``evaluate_sequence``:
non‑negative entries, non‑empty list, and the sequence may be of any length.
The total running time is limited to ``budget_s`` seconds (default 1000 s);
the best sequence found so far is printed and finally returned as a plain
Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)‑1) via FFT.
    Returns the real part truncated to the needed length.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of 2·n·max(a*a) for a distribution already normalised
    to sum = 1.
    """
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x ≥ 0, sum x = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------
def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float) -> tuple[float, np.ndarray]:
    """
    Smoothed max based on log‑sum‑exp.
    Returns (objective, gradient) where the objective is 2·n·smooth_max.
    """
    a = softmax(y)                     # a lies on the simplex

    # convolution a*a (full length 2n‑1)
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth max = max + λ·log Σ exp((x‑max)/λ)
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv_full
    w = exp_shift / Z                     # ∂smooth_max/∂conv_i

    # convolution of a_rev with w gives ∂smooth_max/∂a
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]  # size n

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation with simple back‑tracking line‑search
# ----------------------------------------------------------------------
def subgradient_optimize(a0: np.ndarray,
                         deadline: float,
                         fft_len_conv: int,
                         fft_len_g: int,
                         n: int) -> tuple[np.ndarray, float]:
    """
    Directly minimise 2·n·max(a*a) via sub‑gradient descent.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # current convolution and active set mask
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # back‑tracking line search
        step = 0.5 / np.sqrt(it + 1.0)
        improved = False
        for _ in range(5):
            a_cand = simplex_projection(a - step * g)
            cand_score = fast_score(a_cand, fft_len_conv, n)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_cand.copy()
                a = a_cand
                improved = True
                break
            step *= 0.5
        if not improved:
            a = simplex_projection(a - step * g)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------
def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Greedy mass‑transfer hill‑climb:
    pick two indices, move a tiny random fraction from one to the other.
    Accept only strictly improving moves.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * 0.05
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Continuation L‑BFGS using a decreasing λ schedule
# ----------------------------------------------------------------------
def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam: int,
                       deadline: float,
                       fft_len_conv: int,
                       fft_len_g: int,
                       n: int) -> tuple[np.ndarray, float]:
    """
    Runs L‑BFGS for each λ in lam_schedule, starting each stage from the
    result of the previous stage. Returns the best sequence encountered.
    """
    # initialise the soft‑max argument vector
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:
            # Rare failures of L‑BFGS are ignored.
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best score={best_score:.6f}")

        # short hill‑climb after each λ to sharpen the result
        hc_deadline = min(deadline, time.time() + 0.4)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hc_deadline,
                                        max_iter=1500)

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing / random perturbation phase
# ----------------------------------------------------------------------
def random_anneal(a: np.ndarray,
                  score: float,
                  fft_len: int,
                  n: int,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass between two coordinates.
    Accept worsening moves with a decreasing temperature.
    """
    cur_a = a.copy()
    cur_score = score
    best_a = a.copy()
    best_score = score
    rng = np.random.default_rng()
    T = 0.02
    it = 0

    while time.time() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur_a[i] * rng.random() * 0.10   # up to 10 % of a[i]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[i] -= delta
        cand[j] += delta

        new_score = fast_score(cand, fft_len, n)
        if new_score < cur_score:
            cur_a, cur_score = cand, new_score
            if new_score < best_score:
                best_a, best_score = cand.copy(), new_score
        else:
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                cur_a, cur_score = cand, new_score

        it += 1
        if it % 5000 == 0:
            T = max(T * 0.92, 1e-5)

    return best_a, best_score

    # 8) Raised cosine – high at the borders, low in the centre
    rc = (1.0 + np.cos(np.pi * np.arange(n) / (n - 1))) * 0.5
    add_candidate(rc + 1e-6)

    # 9) Beta‑distribution shapes (including edge‑biased)
    xs = np.arange(n) / (n - 1)
    for α, β in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        beta_arr = (xs ** α) * ((1 - xs) ** β) + 1e-6
        add_candidate(beta_arr)

    # 10) Random Dirichlet draws
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        for _ in range(6):
            add_candidate(rng.gamma(alpha, 1.0, size=n))

    # keep the top few seeds for the heavy optimisation phases
    candidates.sort(key=lambda t: t[0])
    top_seeds = [arr for _score, arr in candidates[:10]]
    print(f"[sweep] best sweep score = {candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # a small safety margin

    # allocate a generous fraction to the L‑BFGS phase
    phase1_end = start_time + 0.55 * budget_s
    phase2_end = start_time + 0.80 * budget_s
    phase3_end = deadline

    # ------------------------------------------------------------------
    # Phase 1 – continuation L‑BFGS (multiple restarts)
    # ------------------------------------------------------------------
    lam_schedule = np.geomspace(0.5, 0.001, num=30)   # geometric decay
    maxiter_per_lam = 500    # more iterations than the baseline

    restarts = 0
    while time.time() < phase1_end and restarts < 30:
        # pick a promising seed and add a tiny jitter for diversity
        seed_vec = top_seeds[rng.integers(len(top_seeds))]
        jitter = rng.random(n) * 1e-3
        a_start = seed_vec + jitter
        a_start = np.maximum(a_start, 0.0)
        a_start /= a_start.sum()

        a_opt, score_opt = lbfgs_continuation(a_start,
                                              lam_schedule,
                                              maxiter_per_lam,
                                              phase1_end,
                                              fft_len_conv,
                                              fft_len_g,
                                              n)

        if score_opt < best_score:
            best_score, best_a = score_opt, a_opt.copy()
            print(f"[L‑BFGS] new global best score={best_score:.6f}")

        # early exit if we already beat the target
        if best_score <= 1.503:
            print("[L‑BFGS] target reached, exiting early")
            return best_a.tolist()

        restarts += 1

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent with line search
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        sg_deadline = min(phase2_end, deadline - 0.5)
        best_a, best_score = subgradient_optimize(best_a,
                                                  sg_deadline,
                                                  fft_len_conv,
                                                  fft_len_g,
                                                  n)
        print(f"[sub‑gradient] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing / random perturbations
    # ------------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = random_anneal(best_a,
                                           best_score,
                                           fft_len_conv,
                                           n,
                                           phase3_end)
        print(f"[anneal] score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing hill‑climb (as much as time permits)
    # ------------------------------------------------------------------
    final_deadline = deadline
    best_a, best_score = hill_climb(best_a,
                                    best_score,
                                    fft_len_conv,
                                    n,
                                    final_deadline,
                                    max_iter=8000)

    # ------------------------------------------------------------------
    # Return the sequence (as a plain Python list)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # external function provided by the runner
    print(f"[final] evaluated score = {final_eval:.6f}")
    return final_seq
