def upsample_sequence(seq, target_n):
    if seq.size == target_n:
        return seq / seq.sum()
    # Interpolate using np.interp
    x_old = np.linspace(0, 1, seq.size)
    x_new = np.linspace(0, 1, target_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq[new_seq < 0] = 0
    if new_seq.sum() == 0:
        new_seq = np.full(target_n, 1.0 / target_n)
    else:
        new_seq = new_seq / new_seq.sum()
    return new_seq

def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a vector already on the simplex."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective based on log‑sum‑exp and its gradient
    w.r.t. the unconstrained variables y (softmax(y) lies on the simplex).

    Parameters
    ----------
    y : np.ndarray
        Unconstrained variables.
    n : int
        Length of the sequence.
    fft_len_conv, fft_len_g : int
        FFT lengths for the convolution and the gradient convolution.
    lam_factor : float
        Controls the smoothing; smaller → closer to the true max.
    """
    # map to simplex
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    # compute g_i = Σ_k a_k * w_{i+k}
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Projected sub‑gradient optimisation (with momentum)
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """
    Momentum sub‑gradient descent on the true max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)
    beta = 0.9
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active set (indices attaining the max)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size ~ 1/√t
        step = 1.0 / np.sqrt(it + 1.0)
        a_new = a - step * v
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Local optimisation: repeatedly move a tiny fraction of mass
    between two positions if it improves the fast score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Simulated annealing (non‑local search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simulated annealing that periodically relocates a random
    fraction of mass.  Uses a slowly cooling temperature.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # start temperature proportional to current score
    T = max(cur_score * 0.05, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        # source index with non‑negligible mass
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        # target index distinct from i
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # move a random fraction (up to 25 %) of the source mass
        max_delta = cur_a[i] * 0.25
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                # revert the move
                cur_a[i] += delta
                cur_a[j] -= delta
                # temperature update and continue
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        # occasional reheating to avoid freezing too early
        if it % 3000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.05, 1e-8)
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Helper to build various initial distributions
# ----------------------------------------------------------------------


def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n according to the requested mode.
    """
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a


def upsample_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a new length.
    The result is re‑normalised to sum to one.
    """
    if seq.size == target_n:
        return seq / seq.sum()
    x_old = np.linspace(0.0, 1.0, seq.size)
    x_new = np.linspace(0.0, 1.0, target_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq[new_seq < 0] = 0.0
    s = new_seq.sum()
    if s == 0.0:
        return np.full(target_n, 1.0 / target_n)
    return new_seq / s


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising `evaluate_sequence`.

    The method combines:
        * up‑sampling of a previous best solution,
        * many stochastic L‑BFGS runs on a log‑sum‑exp surrogate,
        * a long momentum sub‑gradient descent on the true max,
        * an enhanced simulated annealing,
        * a final exhaustive hill‑climb.

    It respects the time budget and returns the best list of floats found.
    """
    rng = np.random.default_rng(seed + 987654)

    # --------------------------------------------------------------
    # Determine problem size and initial vector
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        # enlarge up to 4000 entries (more degrees of freedom)
        target_n = min(4000, max(prev.size, int(prev.size * 1.5)))
        a0 = upsample_sequence(prev, target_n)
        n = a0.size
    else:
        n = 2500
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()

    # FFT lengths (powers of two for speed)
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # for auxiliary convolutions

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Phase 1 – extensive stochastic L‑BFGS on a smooth surrogate
    # --------------------------------------------------------------
    phase1_end = start_time + 0.20 * budget_s
    lam_factors = [0.5, 0.3, 0.15, 0.08, 0.04,
                   0.02, 0.01, 0.005, 0.002, 0.001]

    lbfgs_iter = 0
    while time.time() < phase1_end:
        mode = rng.choice(['uniform', 'edge', 'exp', 'power', 'random'])
        a_start = build_initial(mode, n, rng) if mode != 'prev' else best_a
        # add a small Gaussian perturbation to break symmetry
        y0 = np.log(a_start + 1e-12) + rng.normal(scale=0.05, size=n)
        lam = lam_factors[lbfgs_iter % len(lam_factors)]

        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 1200,
                                'gtol': 1e-12,
                                'disp': False})

        a_candidate = softmax(res.x)
        score_candidate = fast_score(a_candidate, fft_len_conv, n)
        if score_candidate < best_score:
            best_score = score_candidate
            best_a = a_candidate.copy()
            print(f"[L‑BFGS {lbfgs_iter}] mode={mode} lam={lam:.4f} "
                  f"score={best_score:.6f}")

        # quick local polishing
        hill_deadline = min(phase1_end, time.time() + 0.10)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hill_deadline,
                                        max_iter=800)
        lbfgs_iter += 1

    # --------------------------------------------------------------
    # Phase 2 – prolonged momentum sub‑gradient descent
    # --------------------------------------------------------------
    phase2_end = start_time + 0.65 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_momentum_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n)
        print(f"[sub‑gradient] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – enhanced simulated annealing
    # --------------------------------------------------------------
    phase3_end = deadline
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a, best_score, fft_len_conv, n, phase3_end, rng)
        print(f"[SA] best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final exhaustive hill‑climb polishing
    # --------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=5000)
    print(f"[final] fast_score={final_score:.6f}")

    # --------------------------------------------------------------
    # Optional evaluation via the supplied evaluator
    # --------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[evaluate_sequence] -> {eval_val:.6f}")

    # store for possible reuse in later calls
    globals()['height_sequence_1'] = final_a.tolist()

    return final_a.tolist()
