    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0

    n = 3000  # maybe we can increase to 5000 if time left

    a = np.full(n, 1.0 / n)
    best = a.copy()
    best_score = fast_score(a)
    # Set eta schedule: start with eta = 0.5, gradually reduce
    eta = 0.5
    iteration = 0
    while time.time() < deadline:
        # compute convolution
        conv = conv_fft(a, fft_len, n)  # returns length 2n-1
        s_star = int(np.argmax(conv))
        max_conv = conv[s_star]
        # compute gradient
        idx = s_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]
        # multiplicative update
        factor = np.exp(-eta * grad)
        a_new = a * factor
        a_new_sum = a_new.sum()
        if a_new_sum == 0:
            # fallback to uniform
            a_new = np.full(n, 1.0 / n)
            a_new_sum = 1.0
        a_new = a_new / a_new_sum
        # evaluate new max conv
        conv_new = conv_fft(a_new, fft_len, n)
        new_max = conv_new.max()
        if new_max < max_conv - 1e-12:
            # improvement
            a = a_new
            if new_max < best_score:
                best = a_new.copy()
                best_score = new_max
                print(f"[expgrad] iter {iteration} new best max conv {new_max:.6e}")
            # possibly increase eta a bit
            eta = min(eta * 1.05, 1.0)
        else:
            # no improvement, maybe reduce eta
            eta = eta * 0.7
            # optionally random perturbation for exploration
            if rng.random() < 0.01:
                # random jitter
                pert = rng.normal(scale=0.01, size=n)
                a = np.clip(a + pert, 0, None)
                a = a / a.sum()
        iteration += 1

    # After loop, we may have final best conv and compute final score
    final_seq = best.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] conv max = {best_score:.6e}, evaluated score = {final_val:.6f}")
    return final_seq

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1) of length *n*.
   Several random starts (uniform, exponential, power‑law) are tried and
   the best one (lowest fast score) is kept.

2. **Exponential‑gradient descent** on the exact max‑convolution.
   At each iteration:
       • compute the linear convolution a∗a with an FFT,
       • locate the index `s*` where the convolution attains its maximum,
       • the sub‑gradient of the max is `g_i = 2·a_{s*‑i}` (out‑of‑range entries are 0),
       • update multiplicatively   a_i ← a_i·exp(‑η·g_i)   (η – learning rate),
       • renormalise to the simplex.
   Small random jitter is added periodically to escape flat plateaus.
   The learning rate is increased after a successful step and decreased
   after a failure.

3. **Greedy hill‑climb** – a few passes of the standard “move a random
   fraction of mass from i to j” optimisation, accepting only strict
   improvements of the fast score.

4. **Linear‑programming refinement** (only for moderate sizes, n ≤ 3000):
   maximise Σ b_j subject to b ≥ 0, (a ∗ b)[s] ≤ M (M = max (a∗a)).
   The optimal `b` is rescaled to unit sum and mixed back into `a`.

5. Return the best sequence (as a plain Python list) found before the
   time budget expires.  The score is finally computed with the provided
   `evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·n‑1) using a real‑FFT.
    Returns the full convolution (real values).
    """
    # real FFT is a bit faster and uses less memory
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation 2·n·max(a*a) for a probability vector (sum = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Core optimisation stages
# ----------------------------------------------------------------------


def exponential_gradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_eta: float = 0.5,
    eta_max: float = 1.0,
    jitter_scale: float = 0.002,
) -> tuple[np.ndarray, float]:
    """
    Multiplicative (exponential) gradient descent on the exact max‑convolution.
    Returns the best vector seen and its max‑convolution value.
    """
    best_a = a.copy()
    best_max = conv_fft(a, fft_len, n).max()
    eta = init_eta
    rng = np.random.default_rng()
    no_improve_cnt = 0

    while time.time() < deadline:
        # exact convolution
        conv = conv_fft(a, fft_len, n)
        s_star = int(np.argmax(conv))
        max_conv = conv[s_star]

        # sub‑gradient of the max (ordered pairs)
        idx = s_star - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # multiplicative update
        factor = np.exp(-eta * grad)
        a_new = a * factor
        # keep numerical stability – enforce a tiny floor before renorm
        a_new = np.maximum(a_new, 1e-15)
        a_new /= a_new.sum()

        new_max = conv_fft(a_new, fft_len, n).max()

        if new_max < max_conv - 1e-14:
            # improvement – keep the step and try a slightly larger step size
            a = a_new
            max_conv = new_max
            eta = min(eta * 1.07, eta_max)
            no_improve_cnt = 0
            if new_max < best_max:
                best_a = a_new.copy()
                best_max = new_max
                print(f"[exp‑grad] new best max conv = {best_max:.6e}")
        else:
            # no improvement – shrink step size
            eta *= 0.6
            no_improve_cnt += 1

        # occasional random jitter to escape plateaus
        if no_improve_cnt >= 30:
            jitter = rng.normal(scale=jitter_scale, size=n)
            a = a * (1.0 + jitter)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            no_improve_cnt = 0

    return best_a, best_max


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
    move_fraction: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: randomly move a fraction of mass from i to j.
    Improves the fast score only when strictly better.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * move_fraction
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
            # keep the improvement
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


def lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    mix_factor: float = 0.2,
    max_iter: int = 4,
) -> np.ndarray:
    """
    Linear‑programming refinement for moderate n.
    Solve:
        maximise   Σ b_j
        s.t.      b ≥ 0,
                  (a * b)[s] ≤ M    (M = max (a*a))
    The optimal b is scaled to unit sum and mixed back into a.
    """
    rows = 2 * n - 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # Build sparse convolution matrix A where (A·b)[s] = (a*b)[s].
        A = sp.lil_matrix((rows, n), dtype=float)
        for col in range(n):
            A[col : col + n, col] = a
        A = A.tocsr()

        # solve max Σ b   ⇔   min −Σ b
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(rows),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b_opt = res.x
        total = b_opt.sum()
        if total <= 1.0 + 1e-12:
            break

        g = b_opt / total
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        new_M = np.convolve(a_new, a_new).max()
        if new_M < M:
            a = a_new
            print(f"[LP] reduced M from {M:.6f} to {new_M:.6f}")
        else:
            break
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # problem size – 3000 gives a good trade‑off between speed and quality
    # ------------------------------------------------------------------
    n = 3000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1️⃣  Initialise – try a few simple factories and keep the best
    # ------------------------------------------------------------------
    def make_candidate(mode: str) -> np.ndarray:
        if mode == "uniform":
            a = np.full(n, 1.0 / n)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.02)
            a = np.exp(-rate * np.arange(n))
            a /= a.sum()
        elif mode == "power":
            p = rng.uniform(0.3, 2.0)
            a = (np.arange(1, n + 1, dtype=float) ** (-p))
            a /= a.sum()
        elif mode == "random":
            a = rng.random(n)
            a = a / a.sum()
        else:
            a = np.full(n, 1.0 / n)
        return a

    best_a = None
    best_score = np.inf
    for mode in ["uniform", "exp", "power", "random"]:
        cand = make_candidate(mode)
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
        print(f"[init] mode={mode} fast score = {sc:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Exponential‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    exp_deadline = min(deadline, time.time() + 0.70 * (deadline - time.time()))
    best_a, best_max = exponential_gradient_optimize(
        best_a, n, fft_len, exp_deadline, init_eta=0.5, eta_max=1.0
    )
    best_score = 2.0 * n * best_max
    print(f"[exp‑grad] final max‑conv = {best_max:.6e}, fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Aggressive local hill‑climb
    # ------------------------------------------------------------------
    hc_deadline = min(deadline, time.time() + 0.25 * (deadline - time.time()))
    best_a, best_score = hill_climb(
        best_a,
        best_score,
        fft_len,
        n,
        hc_deadline,
        max_iter=12_000,
        move_fraction=0.12,
    )
    print(f"[hill‑climb] post‑climb fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Linear‑programming refinement (only if n is modest)
    # ------------------------------------------------------------------
    if n <= 3000 and time.time() + 5.0 < deadline:
        lp_deadline = deadline - 5.0
        best_a = lp_refine(best_a, n, lp_deadline, mix_factor=0.2, max_iter=5)
        best_score = fast_score(best_a, fft_len, n)
        print(f"[LP] post‑LP fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
