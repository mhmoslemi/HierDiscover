def flattening_optimize(a_start, deadline, fft_len_conv, fft_len_g, n):
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(a, fft_len_conv, n)
    t = 0
    while time.time() < deadline:
        # compute a*a conv
        fa = np.fft.fft(a, fft_len_conv)
        c = np.real(np.fft.ifft(fa * fa))[:2*n-1]  # conv
        # compute gradient of sum(c^2)
        # gradient = 4 * conv(c, a) central part
        fc = np.fft.fft(c, fft_len_g)
        fa_full = np.fft.fft(a, fft_len_g)
        grad_full = np.real(np.fft.ifft(fc * fa_full))
        grad = 4 * grad_full[n-1:n-1+n]
        # step size: schedule maybe 0.01 / sqrt(iter+1)
        step = 0.1 / np.sqrt(t + 1)
        a_new = a - step * grad
        # project onto simplex
        a = simplex_projection(a_new)
        # keep best
        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        t += 1
    return best, best_score

"""
Propose a step‑function sequence with a low
    2·n·max(conv(a,a)) / (sum(a)**2)

The pipeline:
  0️⃣  Gradient‑flattening: minimise Σ(conv²) to make the self‑convolution as flat
  1️⃣  Stochastic L‑BFGS on a smoothed max (log‑sum‑exp) objective
  2️⃣  Long momentum sub‑gradient descent on the true max
  3️⃣  Simulated annealing for non‑local moves
  4️⃣  Final greedy hill‑climb polish

All steps respect the given time budget (default 1000 s) and work on the
probability simplex via Euclidean projection.  The best sequence found
is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max mapping ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Score = 2·n·max(conv) for a already normalised (sum=1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed‑max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """
    Smoothed‑max objective based on log‑sum‑exp and its gradient w.r.t. y.
    """
    a = softmax(y)

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # smooth approximation of the max using log‑sum‑exp
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c                     # for numeric stability
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d(smooth_max)/d conv_k
    # compute g_i = Σ_k a_k * w_{i+k}
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through the soft‑max
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Gradient‑flattening (minimise Σ conv²)
# ----------------------------------------------------------------------


def flattening_optimize(a_start: np.ndarray,
                        deadline: float,
                        fft_len_conv: int,
                        fft_len_g: int,
                        n: int):
    """
    Projected gradient descent on L = Σ(conv(a,a)²).
    This flattens the self‑convolution and usually lowers the max value.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = fast_score(best, fft_len_conv, n)
    it = 0

    while time.time() < deadline:
        # convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        c = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]   # full conv

        # gradient = 4 * (c * a_rev) (central part)
        a_rev = a[::-1]
        fc = np.fft.fft(c, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_rev = np.real(np.fft.ifft(fc * fa_rev))
        grad = 4.0 * conv_rev[(n - 1):(n - 1 + n)]

        # step size schedule
        step = 0.30 / np.sqrt(it + 1.0)
        a_new = a - step * grad
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        it += 1

    return best, best_score

# Projected sub‑gradient optimisation (with momentum)
# ----------------------------------------------------------------------


def subgradient_momentum_optimize(a0: np.ndarray,
                                 deadline: float,
                                 fft_len_conv: int,
                                 fft_len_g: int,
                                 n: int):
    """
    Directly minimise the true max‑convolution using a momentum
    sub‑gradient method with Euclidean projection onto the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    v = np.zeros_like(a)          # momentum buffer
    beta = 0.9                    # momentum coefficient
    it = 0

    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()

        # active set mask (indices attaining the max)
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient (see derivation in the original code)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1):(n - 1 + n)]

        # momentum update
        v = beta * v + (1.0 - beta) * g

        # diminishing step size
        step = 1.0 / np.sqrt(it + 1.0)
        a_new = a - step * v
        a = simplex_projection(a_new)

        cur_score = fast_score(a, fft_len_conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple greedy hill‑climb (keeps sum = 1)
# ----------------------------------------------------------------------


def hill_climb(a: np.ndarray,
               current_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               max_iter: int = 2000):
    """
    Local optimisation that repeatedly moves a tiny fraction of mass
    between two positions if it improves the (fast) score.
    """
    best = a.copy()
    best_score = current_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move up to 4 % of the source mass
        delta = best[i] * rng.random() * 0.04
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score

# Simulated annealing (non‑local search)

def simulated_annealing(a_start: np.ndarray,
                        score_start: float,
                        fft_len: int,
                        n: int,
                        deadline: float,
                        rng: np.random.Generator):
    """
    Simulated annealing that relocates a random fraction of mass.
    Improves exploration after the gradient phases.
    """
    cur_a = a_start.copy()
    cur_score = score_start
    best_a = cur_a.copy()
    best_score = cur_score

    # Initial temperature as a small fraction of the current score
    T = max(cur_score * 0.02, 1e-8)
    decay = 0.9995
    it = 0

    while time.time() < deadline:
        i = rng.integers(0, n)
        if cur_a[i] < 1e-14:
            continue

        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        max_delta = cur_a[i] * 0.30
        delta = rng.random() * max_delta
        if delta <= 0.0:
            continue

        cur_a[i] -= delta
        cur_a[j] += delta

        new_score = fast_score(cur_a, fft_len, n)

        if new_score < cur_score:
            cur_score = new_score
        else:
            prob = math.exp(-(new_score - cur_score) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_score = new_score
            else:
                cur_a[i] += delta
                cur_a[j] -= delta
                it += 1
                T *= decay
                continue

        if cur_score < best_score:
            best_score = cur_score
            best_a = cur_a.copy()

        T *= decay
        if it % 5000 == 0 and T < best_score * 1e-6:
            T = max(best_score * 0.02, 1e-8)
        it += 1

    return best_a, best_score

def build_initial(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a probability vector of length n according to the requested mode.
    """
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'edge':
        frac = rng.uniform(0.004, 0.02)          # fraction of indices at each edge
        m = max(1, int(frac * n))
        edge_val = np.sqrt(n / 2.0) / n
        interior_val = 1.0 / np.sqrt(2.0) / n
        a = np.full(n, interior_val)
        a[:m] = edge_val
        a[-m:] = edge_val
        a /= a.sum()
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        x = np.exp(-rate * np.arange(n))
        a = x / x.sum()
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        x = (np.arange(1, n + 1, dtype=float) ** (-p))
        a = x / x.sum()
    elif mode == 'random':
        a = rng.gamma(0.5, 1.0, size=n)
        a = a / a.sum()
    else:
        a = np.full(n, 1.0 / n)
    return a

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence minimising `evaluate_sequence`.

    The algorithm combines a gradient‑flattening stage, stochastic L‑BFGS,
    a long momentum sub‑gradient descent, simulated annealing and a final
    greedy hill‑climb.  It respects the time budget and returns the best
    list of floats found.
    """
    rng = np.random.default_rng(seed + 123456)

    # ------------------------------------------------------------------
    # Choose problem size (n) and initialise a starting point.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        prev = np.maximum(prev, 0.0)
        max_n = 4000
        if prev.size > max_n:
            prev = prev[:max_n]
        n = max(prev.size, 2000)               # ensure not too small
        if prev.sum() > 0:
            a0 = prev / prev.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2000
        a0 = np.full(n, 1.0 / n)

    a0 = a0 / a0.sum()                         # enforce simplex

    # ------------------------------------------------------------------
    # FFT lengths (powers of two for speed)
    # ------------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)   # for a*a
    fft_len_g = next_pow2(3 * n - 2)      # auxiliary convolutions

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – gradient flattening (Σ conv² minimisation)
    # ------------------------------------------------------------------
    phase0_end = start_time + 0.15 * budget_s
    if time.time() < phase0_end:
        best_a, best_score = flattening_optimize(best_a,
                                                 phase0_end,
                                                 fft_len_conv,
                                                 fft_len_g,
                                                 n)
        print(f"[flatten] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – stochastic L‑BFGS on smoothed max
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_factors = [0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.006, 0.003, 0.001]
    lbfgs_iter = 0
    while time.time() < phase1_end:
        mode = rng.choice(['uniform', 'edge', 'exp', 'power', 'random'])
        a_start = build_initial(mode, n, rng)
        y0 = np.log(a_start + 1e-12)
        lam = lam_factors[lbfgs_iter % len(lam_factors)]

        # L‑BFGS on the smoothed objective
        res = minimize(fun=objective_and_grad,
                       x0=y0,
                       args=(n, fft_len_conv, fft_len_g, lam),
                       method='L-BFGS-B',
                       jac=True,
                       options={'maxiter': 600,
                                'gtol': 1e-12,
                                'disp': False})

        a_candidate = softmax(res.x)
        score_candidate = fast_score(a_candidate, fft_len_conv, n)

        if score_candidate < best_score:
            best_score = score_candidate
            best_a = a_candidate.copy()
            print(f"[L‑BFGS {lbfgs_iter}] mode={mode} lam={lam:.4f} score={best_score:.6f}")

        # short hill‑climb after each L‑BFGS run
        hill_deadline = min(phase1_end, time.time() + 0.12)
        best_a, best_score = hill_climb(best_a, best_score,
                                        fft_len_conv, n,
                                        hill_deadline,
                                        max_iter=800)
        lbfgs_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – long momentum sub‑gradient descent on the true max.
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_momentum_optimize(best_a,
                                                          phase2_end,
                                                          fft_len_conv,
                                                          fft_len_g,
                                                          n)
        print(f"[sub‑gradient] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – simulated annealing for non‑local moves.
    # ------------------------------------------------------------------
    phase3_end = deadline
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(best_a,
                                                 best_score,
                                                 fft_len_conv,
                                                 n,
                                                 phase3_end,
                                                 rng)
        print(f"[SA] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb polish.
    # ------------------------------------------------------------------
    final_a, final_score = hill_climb(best_a, best_score,
                                      fft_len_conv, n,
                                      deadline,
                                      max_iter=5000)

    # ------------------------------------------------------------------
    # Evaluation with the provided evaluator (optional)
    # ------------------------------------------------------------------
    try:
        eval_val = evaluate_sequence(final_a.tolist())
    except Exception:
        eval_val = float('nan')
    print(f"[final] evaluate_sequence -> {eval_val:.6f}")

    # store globally for possible reuse
    globals()['height_sequence_1'] = final_a.tolist()

    return final_a.tolist()
