def targeted_hill_climb(a, cur_score, fft_len, n, deadline, move_limit=0.1, max_iter=5000):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter>0:
        max_iter -= 1
        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))
        # compute contributions
        # compute counterpart array: b = best[::-1]? For each i, counterpart = k-i if valid else -1
        indices = np.arange(n)
        counterpart = k - indices
        valid = (counterpart >= 0) & (counterpart < n)
        prod = np.zeros(n)
        prod[valid] = best[indices[valid]] * best[counterpart[valid]]
        # choose i_max with largest product
        i = int(np.argmax(prod))
        # choose j minimal a counterpart (or random among low) 
        # pick indices where counterpart valid, and product small.
        # To avoid zero denominator, find set of candidate j where counterpart valid and product low.
        low_candidates = np.where(valid)[0]
        # compute counterpart values for these candidates
        counterpart_vals = best[counterpart[low_candidates]]
        # choose j with smallest counterpart
        j = int(low_candidates[np.argmin(counterpart_vals)])
        if i == j:
            continue
        delta = best[i] * rng.random() * move_limit
        if delta <= 0:
            continue
        # move delta from i to j
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score

    mean_g = np.dot(a, g_vec)
    grad_y = 2.0 * n * a * (g_vec - mean_g)
    return f, grad_y

"""
Improved step‑function search.

The code fixes a bug in the previous implementation (the smoothing
parameter was ignored) and adds a targeted hill‑climb that moves mass
from indices that contribute strongly to the current maximal
convolution entry to indices that contribute weakly.  A geometric
schedule of smoothing parameters (σ) is used with L‑BFGS, followed by
projected sub‑gradient descent and a final aggressive polishing pass.
All steps respect the non‑negativity and simplex constraints required
by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax; returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2*n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a*a) for a probability vector."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = (cssv[-1] - 1.0) / n
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑softmax) for a given heuristic mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # greedy Sidon set – size ≈ √n
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                s = cand + c
                if s in sums:
                    conflict = True
                    break
            if not conflict and (2 * cand not in sums):
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


# ----------------------------------------------------------------------
# Smoothed objective (σ‑log‑sum‑exp) and its gradient w.r.t. y
# ----------------------------------------------------------------------


def objective_and_grad_smooth(y: np.ndarray, n: int, fft_len: int, sigma: float):
    """
    Smoothed objective 2·n·smooth_max and gradient.
    smooth_max = max_c + sigma·log Σ exp((c_i-max_c)/sigma)
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    shifted = conv - max_c                    # max = 0 now
    exp_shift = np.exp(shifted / sigma)
    Z = exp_shift.sum()
    smooth_max = max_c + sigma * np.log(Z)
    f = 2.0 * n * smooth_max

    # derivative of smooth_max w.r.t. conv entries
    w = exp_shift / Z                          # softmax over conv entries

    # ∂ smooth_max / ∂ a_j = Σ_i w_i * 2 * a_{i-j}
    # Compute via convolution of w with reversed a
    fa_rev = np.fft.fft(a[::-1], fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = 2.0 * conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through soft‑max parametrisation
    mean_g = np.dot(a, g_vec)
    grad_y = 2.0 * n * a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize_smooth(
    y_start: np.ndarray,
    n: int,
    fft_len: int,
    sigma_schedule,
    maxiter_per_stage: int = 200,
):
    """
    Run a multi‑stage L‑BFGS optimisation with decreasing smoothing σ.
    Each stage starts from the result of the previous one.
    """
    y = y_start.copy()
    for sigma in sigma_schedule:
        res = minimize(
            fun=objective_and_grad_smooth,
            x0=y,
            args=(n, fft_len, sigma),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Targeted hill‑climb (mass transfer guided by the maximal convolution entry)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    move_limit: float = 0.10,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb that looks at the current maximal convolution index.
    It moves a random fraction of mass from an index that heavily contributes
    to that maximum to an index that contributes little.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1

        conv = conv_fft(best, fft_len, n)
        k = int(np.argmax(conv))

        idx = np.arange(n)
        counterpart = k - idx
        mask = (counterpart >= 0) & (counterpart < n)
        # contribution product a_i * a_{k-i}
        prod = np.zeros(n)
        prod[mask] = best[idx[mask]] * best[counterpart[mask]]

        i = int(np.argmax(prod))          # heavy contributor
        # among valid positions pick one whose counterpart value is minimal
        valid_idxs = idx[mask]
        counterpart_vals = best[counterpart[mask]]
        j = int(valid_idxs[np.argmin(counterpart_vals)])

        if i == j:
            continue

        delta = best[i] * rng.random() * move_limit
        if delta <= 0.0:
            continue

        best[i] -= delta
        best[j] += delta

        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta

    return best, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    fft_len: int,
    n: int,
    deadline: float,
    eta0: float = 0.20,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Classic projected sub‑gradient descent on 2·n·max(conv)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len, n)
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        conv = conv_fft(a, fft_len, n)
        k = int(np.argmax(conv))

        # sub‑gradient of the maximal entry: g_i = 2·a_{k-i} (if in range)
        idx = np.arange(n)
        counterpart = k - idx
        mask = (counterpart >= 0) & (counterpart < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[counterpart[mask]]

        eta = eta0 / np.sqrt(it)
        a_new = a - eta * g
        a_new = project_onto_simplex(a_new)

        cur_score = fast_score(a_new, fft_len, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        a = a_new

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
