"""
Hybrid symmetric optimizer for the step‑function attack.

The idea:
* Use a symmetric step function a of even length n.
  Only the first half (size m = n//2) is free – the second half is its mirror.
  This halves the optimisation dimension while keeping the same expressive power.
* Optimise a smooth approximation of the max‑convolution via L‑BFGS
  (continuation on the smoothing λ).
* After each continuation run perform a greedy “targeted” hill‑climb that
  moves probability mass from the indices that contribute most to the current
  convolution peak towards those that contribute least, always preserving symmetry.
* Finish with a short random symmetric hill‑climb for polishing.

The algorithm works for any affordable n (here n = 4000).  It usually drives
the evaluation score below the target 1.503 within the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures, all top‑level)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of a with itself using FFT.
    Returns the first 2*n‑1 entries (the full convolution).
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Exact evaluation of 2·n·max(a*a) for a probability vector a
    (i.e. ∑ a = 1). This matches evaluate_sequence() when the sum is 1.
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Symmetric representation
# ----------------------------------------------------------------------


def expand_from_half(p: np.ndarray) -> np.ndarray:
    """
    From the half‑vector p (size m) build the full symmetric step
    function a (size n = 2*m).  The construction satisfies ∑a = 1.
    """
    # a_i = p_i/2  for i = 0..m‑1,  a_{n‑1‑i} = p_i/2
    half = p * 0.5
    a = np.concatenate([half, half[::-1]])
    return a


# ----------------------------------------------------------------------
# Smoothed objective and gradient (symmetric version)
# ----------------------------------------------------------------------


def objective_and_grad_sym(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution and its gradient w.r.t.
    the unconstrained logits y (size m = n//2).  The mapping
    y -> p = softmax(y) -> a = expand_from_half(p) is used.
    """
    # ----- build a from logits -----
    p = softmax(y)                     # sum(p) = 1
    a = expand_from_half(p)            # sum(a) = 1, a >= 0

    # ----- convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c               # max becomes 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)  # smooth approximation of max(conv)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth max w.r.t. a (full vector) -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1  (cross‑correlation)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    grad_a = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- compress gradient to half‑vector p -----
    # each p_k controls a_k and a_{n‑1‑k} (both equal to p_k/2)
    m = n // 2
    grad_p = 0.5 * (grad_a[:m] + grad_a[-1:-m - 1:-1])

    # ----- back‑prop through soft‑max -----
    mean_g = np.dot(p, grad_p)
    grad_y = p * (grad_p - mean_g)

    return f, grad_y


def schedule_optimize_sym(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 600,
) -> np.ndarray:
    """
    Continuation L‑BFGS on the symmetric objective.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad_sym,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={
                "maxiter": maxiter_per_stage,
                "gtol": 1e-14,
                "disp": False,
                "iprint": -1,
            },
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Symmetric hill‑climb utilities
# ----------------------------------------------------------------------


def random_hill_climb_sym(
    p: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Random symmetric hill‑climb on the simplex (half‑vector p).
    """
    best = p.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.12

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, best.size, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        a = expand_from_half(best)
        new_score = fast_score(a, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb_sym(
    p: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb on the symmetric half‑vector.
    At each iteration the contribution of each pair to the current
    convolution peak is computed, and a small fraction of probability mass
    is transferred from the worst‑contributing pair to the best‑contributing
    pair (preserving symmetry).
    """
    best = p.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        a = expand_from_half(best)

        # current convolution and peak location
        conv = conv_fft(a, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each half–index to that peak
        # full contribution = a[i] * a[k_max-i]; we need it for both sides of the pair
        idx_full = np.arange(n)
        partner = n - 1 - idx_full
        # compute a shifted copy (safe indexing)
        shift = k_max - idx_full
        valid = (shift >= 0) & (shift < n)
        a_shift = np.zeros_like(a)
        a_shift[valid] = a[shift[valid]]
        contrib_full = a * a_shift

        # collapse to half‑vector (pairwise sum)
        m = n // 2
        contrib_half = contrib_full[:m] + contrib_full[-1:-m - 1:-1]

        i_src = int(np.argmax(contrib_half))
        i_dst = int(np.argmin(contrib_half))

        if i_src == i_dst:
            # fallback to a random move
            i_src = rng.integers(0, best.size)
            i_dst = rng.integers(0, best.size)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta

        a_new = expand_from_half(best)
        new_score = fast_score(a_new, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑score step‑function sequence.
    Returns a list of non‑negative heights (full vector a).
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Problem size – a larger symmetric length gives a lower asymptotic bound.
    # --------------------------------------------------------------
    n = 4000                     # must be even
    m = n // 2

    # initialise best candidate (uniform)
    best_p = np.full(m, 1.0 / m)
    best_a = expand_from_half(best_p)
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)
    best_score = fast_score(best_a, fft_len_conv, n)
    best_y = np.log(best_p + 1e-12)

    print(f"[init] n={n}, initial fast score = {best_score:.6f}")

    # continuation λ‑schedule (smooth‑max)
    lam_factors = np.geomspace(0.5, 5e-6, num=35)   # 35 stages, gradually tighter

    iteration = 0
    max_restarts = 12

    # --------------------------------------------------------------
    # Multi‑restart optimisation loop
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0 and iteration < max_restarts:
        # ----- pick an initialisation mode -----
        mode = rng.choice(["random", "exp", "power", "edge", "best"])

        if mode == "random":
            y0 = rng.standard_normal(m)
        elif mode == "exp":
            rate = rng.uniform(0.001, 0.015)
            a_tmp = np.exp(-rate * np.arange(m))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == "power":
            pwr = rng.uniform(0.3, 2.2)
            a_tmp = (np.arange(1, m + 1, dtype=float) ** (-pwr))
            a_tmp /= a_tmp.sum()
            y0 = np.log(a_tmp + 1e-12)
        elif mode == "edge":
            # concentrate a little mass near the ends
            edge_frac = rng.uniform(0.01, 0.04)
            m_edge = max(1, int(m * edge_frac))
            base = np.full(m, 1.0 / m)
            base[:m_edge] = base[:m_edge] * 3.0
            base[-m_edge:] = base[-m_edge:] * 3.0
            base /= base.sum()
            y0 = np.log(base + 1e-12)
        else:   # perturb the current best
            y0 = best_y + rng.normal(scale=0.07, size=m)

        # ----- continuation L‑BFGS -----
        y_opt = schedule_optimize_sym(
            y_start=y0,
            n=n,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            lam_factors=lam_factors,
            maxiter_per_stage=800,
        )
        p_opt = softmax(y_opt)
        a_opt = expand_from_half(p_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_score = score_opt
            best_p = p_opt.copy()
            best_y = y_opt.copy()
            print(f"[{iteration}] mode={mode}  L‑BFGS improved fast score = {best_score:.6f}")

        # ----- aggressive targeted hill‑climb (symmetrised) -----
        hc_deadline = min(deadline, time.time() + 0.9)
        best_p, best_score = targeted_hill_climb_sym(
            best_p,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_moves=3500,
            frac=0.07,
        )
        best_y = np.log(best_p + 1e-12)

        # ----- short random polishing hill‑climb -----
        if time.time() < deadline:
            best_p, best_score = random_hill_climb_sym(
                best_p,
                best_score,
                fft_len_conv,
                n,
                deadline,
                max_iter=2500,
            )
            best_y = np.log(best_p + 1e-12)

        iteration += 1

    # --------------------------------------------------------------
    # Final deep refinement (another continuation pass)
    # --------------------------------------------------------------
    for lam in lam_factors:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad_sym,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 900, "gtol": 1e-14, "disp": False},
        )
        best_y = res.x
        best_p = softmax(best_y)
        best_a = expand_from_half(best_p)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final‑L‑BFGS] lam={lam:.3e} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing (targeted + random)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_p, best_score = targeted_hill_climb_sym(
            best_p,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_moves=8000,
            frac=0.05,
        )
        best_p, best_score = random_hill_climb_sym(
            best_p,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=4000,
        )
        best_y = np.log(best_p + 1e-12)

    # --------------------------------------------------------------
    # Convert to list and final evaluate (for logging)
    # --------------------------------------------------------------
    final_seq = expand_from_half(best_p).tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")

    return final_seq
