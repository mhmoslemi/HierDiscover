def proj_simplex(v):
    # v: array of any length, can be negative.
    # returns w: nonnegative, sum=1.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def subgradient_descent(a, best_score, fft_len_conv, fft_len_grad, deadline, max_iters=20000):
    n = len(a)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute conv
        conv = conv_fft(a, fft_len_conv, n)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
        # compute active set
        tol = max_b * 1e-12 + 1e-14  # tolerance relative
        active = np.where(conv >= max_b - tol)[0]
        if active.size == 0:
            w = np.zeros(2*n-1)
            w[np.argmax(conv)] = 1.0
        else:
            w = np.zeros(2*n-1)
            w[active] = 1.0 / active.size
        # gradient
        fa_rev = np.fft.fft(a[::-1], fft_len_grad)
        fw = np.fft.fft(w, fft_len_grad)
        g_full = np.real(np.fft.ifft(fa_rev * fw))
        grad = 2.0 * g_full[(n-1):(n-1+n)]
        # step size strategy
        # use diminishing step size e.g., eta = 1.0 / np.sqrt(it+1)
        eta = 1.0 / np.sqrt(it + 1)
        a_new = a - eta * grad
        # project onto simplex
        a_new = proj_simplex(a_new)
        # compute new score
        conv_new = conv_fft(a_new, fft_len_conv, n)
        max_b_new = conv_new.max()
        new_score = 2.0 * n * max_b_new
        # accept if improved else maybe keep a unchanged (or random perturb)
        if new_score < cur_score:
            a = a_new
        else:
            # maybe try smaller step (line search)
            # reduce eta multiplicatively up to few attempts
            accepted = False
            for factor in [0.5, 0.25, 0.125]:
                a_tmp = proj_simplex(a - eta * factor * grad)
                conv_tmp = conv_fft(a_tmp, fft_len_conv, n)
                new_score_tmp = 2.0 * n * conv_tmp.max()
                if new_score_tmp < cur_score:
                    a = a_tmp
                    accepted = True
                    break
            if not accepted:
                # random perturbation
                i, j = np.random.randint(0, n, size=2)
                if i != j and a[i] > 0:
                    delta = a[i] * 0.05
                    a[i] -= delta
                    a[j] += delta
                    a = proj_simplex(a)
        # maybe update best global
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score

def targeted_hill_climb(a, cur_score, fft_len, n, deadline, max_moves=5000, frac=0.06):
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]
        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))
        if i_src == i_dst:
            i_src, i_dst = rng.integers(0, n, size=2)
            if i_src == i_dst:
                max_moves -= 1
                continue

"""
Hybrid local‑search for the step‑function attack.

The algorithm proceeds in four stages:

1. **Initialised by a cheap parametric sweep** – we look at mixtures of a
   symmetric exponential edge profile and a uniform background.
   The best candidate (according to the fast FFT based score) seeds the
   optimisation.

2. **Projected sub‑gradient descent on the exact max‑convolution** – the
   objective  f(a)=2·n·max(a*a) is non‑smooth; we compute a true sub‑gradient
   from the active convolution indices (those attaining the maximum) and
   perform a diminishing‑step gradient step followed by a projection onto the
   probability simplex.

3. **Targeted hill‑climb** – a greedy move that transfers a small fraction of
   mass from the index contributing most to the current peak to the index
   contributing the least.  This directly reduces the true max‑convolution.

4. **Random hill‑climb polish** – a generic move that swaps mass between two
   randomly chosen positions, accepted only if it improves the fast score.

All stages respect the non‑negative and normalisation constraints required
by `evaluate_sequence`.  The routine runs until the supplied time budget
(≈1000 s) expires and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the first 2*n‑1 entries.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast version of `evaluate_sequence` for a vector that sums to 1.
    Returns 2·n·max(a*a).
    """
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    v = v.astype(np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric initialisation
# ----------------------------------------------------------------------


def param_sweep_initialisation(
    a_cur: np.ndarray,
    cur_score: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Scan a two‑parameter family
        a_i = w * exp(-beta * min(i,n-1-i)) + (1-w)/n
    and keep the best vector according to `fast_score`.
    """
    best_a = a_cur.copy()
    best_score = cur_score

    dist = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))

    w_vals = np.linspace(0.0, 1.0, 21)
    beta_vals = np.logspace(-4, -1, 30)  # 1e‑4 … 1e‑1

    for w in w_vals:
        for beta in beta_vals:
            a = w * np.exp(-beta * dist) + (1.0 - w) / n
            a = np.maximum(a, 0.0)
            a /= a.sum()
            sc = fast_score(a, fft_len, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    print(f"[param sweep] best fast score = {best_score:.6f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def subgradient_descent(
    a_start: np.ndarray,
    start_score: float,
    n: int,
    fft_len_conv: int,
    fft_len_grad: int,
    deadline: float,
    max_iters: int = 30000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent for  f(a)=2·n·max(a*a).
    A diminishing step size  η_t = 0.5/√t is used; after each step we project
    onto the simplex.  Simple back‑tracking is employed if the move does not
    improve the score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score

    it = 0
    while it < max_iters and time.time() < deadline:
        # current convolution and max
        conv = conv_fft(a, fft_len_conv, n)
        max_b = conv.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()

        # active set (indices attaining the max, within a tiny tolerance)
        tol = max_b * 1e-12 + 1e-14
        active = np.where(conv >= max_b - tol)[0]
        w = np.zeros(2 * n - 1)
        if active.size == 0:
            # fallback – take the single argmax
            w[int(np.argmax(conv))] = 1.0
        else:
            w[active] = 1.0 / active.size

        # sub‑gradient (see derivation in the analysis)
        fa_rev = np.fft.fft(a[::-1], fft_len_grad)
        fw = np.fft.fft(w, fft_len_grad)
        g_full = np.real(np.fft.ifft(fa_rev * fw))
        grad = 2.0 * g_full[(n - 1) : (n - 1 + n)]

        # diminishing step size
        eta = 0.5 / np.sqrt(it + 1)

        # try a simple projected step
        a_new = proj_simplex(a - eta * grad)
        conv_new = conv_fft(a_new, fft_len_conv, n)
        new_score = 2.0 * n * conv_new.max()

        if new_score < cur_score:
            a = a_new
        else:
            # back‑track – try smaller factors
            improved = False
            for factor in (0.5, 0.25, 0.125):
                a_tmp = proj_simplex(a - eta * factor * grad)
                conv_tmp = conv_fft(a_tmp, fft_len_conv, n)
                if 2.0 * n * conv_tmp.max() < cur_score:
                    a = a_tmp
                    improved = True
                    break
            if not improved:
                # random jitter to escape a flat region
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 0:
                    delta = a[i] * rng.random() * 0.05
                    a[i] -= delta
                    a[j] += delta
                    a = proj_simplex(a)

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Targeted hill‑climb (as in the reference code)
# ----------------------------------------------------------------------


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 8000,
    frac: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Greedy mass transfer from the coordinate that contributes most to the
    current peak of a*a to the one that contributes the least.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src, i_dst = rng.integers(0, n, size=2)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Simple random hill‑climb (fallback)
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Randomly move a fraction of mass between two positions; keep the move
    only if it strictly improves the fast score.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            best[i] += delta
            best[j] -= delta
        max_iter -= 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence achieving a low
    `evaluate_sequence` score.  Returns the best sequence found within the
    allocated time.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Problem size and initial vector
    # ------------------------------------------------------------------
    default_n = 2000
    prev = globals().get("height_sequence_1")
    if prev is not None:
        arr = np.asarray(prev, dtype=float)
        if arr.ndim == 1 and arr.size >= default_n:
            n = default_n
            a0 = arr[:n].copy()
            a0 = np.maximum(a0, 0.0)
            s = a0.sum()
            if s > 1e-12:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            n = default_n
            a0 = np.full(n, 1.0 / n)
    else:
        n = default_n
        a0 = np.full(n, 1.0 / n)

    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_grad = next_pow2(3 * n - 2)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – cheap parametric sweep
    # ------------------------------------------------------------------
    best_a, best_score = param_sweep_initialisation(
        best_a, best_score, n, fft_len_conv, rng
    )

    # ------------------------------------------------------------------
    # Stage 2 – projected sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    best_a, best_score = subgradient_descent(
        best_a,
        best_score,
        n,
        fft_len_conv,
        fft_len_grad,
        deadline,
        max_iters=30000,
    )
    print(f"[sub‑grad] best fast score after descent = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – targeted hill‑climb for polishing
    # ------------------------------------------------------------------
    best_a, best_score = targeted_hill_climb(
        best_a,
        best_score,
        fft_len_conv,
        n,
        deadline,
        max_moves=15000,
        frac=0.06,
    )
    print(f"[targeted] best fast score after hill‑climb = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final random hill‑climb polish (if time permits)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            max_iter=8000,
        )
        print(f"[random] final fast score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation and return
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
