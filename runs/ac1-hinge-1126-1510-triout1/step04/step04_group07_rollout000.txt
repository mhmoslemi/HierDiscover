def compute_conv(a): # returns conv length 2n-1 using FFT.
    n = len(a)
    fft_len = next_pow2(2*n)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
    return conv

def compute_smooth_max(conv, tau):
    # stable log-sum-exp
    max_c = conv.max()
    exp_term = np.exp(tau * (conv - max_c))
    sum_exp = exp_term.sum()
    smooth_max = max_c + (1.0 / tau) * np.log(sum_exp)
    probs = exp_term / sum_exp
    return smooth_max, probs

assistantfinal**Solution Overview**

The program searches for a non‑negative step function `a = (a₀,…,a_{n‑1})` that minimizes  

\[
E(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` denotes the linear convolution.  
After normalising the sequence to sum = 1 the objective reduces to  

\[
E(a)=2\,n\;\max (a*a).
\]

The main difficulties are

* the `max` is non‑differentiable,
* the search space is the simplex (non‑negative numbers that sum to 1),
* a good solution usually needs **tens of thousands** of coefficients.

The algorithm combines three ideas that improve on the previous state‑of‑the‑art:

1. **Smooth approximation of the maximum** –  
   `Mτ(a) = (1/τ)·log ∑ₖ exp(τ·cₖ)` with `c = a*a`.  
   For a small temperature `τ` the function is smooth; for a large `τ` it approaches the true maximum.  
   The gradient of `Mτ` can be written in closed form and evaluated with two FFT‑based convolutions.

2. **Soft‑max parametrisation + Adam** –  
   The coefficients are written as `a = softmax(y)`.  
   This guarantees non‑negativity and the simplex constraint without any projection step.  
   Adam (with bias‑correction) is used to update the logits `y`.

3. **Annealing & multi‑restart scheme** –  
   * τ is increased linearly from `τ_start` to `τ_end` during the run, allowing the optimiser to first explore a smooth landscape and then concentrate on the true maximum.  
   * Several random initial shapes (power‑law, exponential, two‑sided power‑law, Beta‑pdf) and several lengths (`8 k` … `30 k`) are tried.  
   * The best sequence is finally refined by a cheap stochastic local‑search (multiplicative Gaussian jitter on a few coordinates).

All heavy work is performed with NumPy’s FFT (`rfft`/`irfft`).  
For a length `n = 30 000` the FFT size is only `131 072`, so many thousands of Adam steps comfortably fit into the 1000 s budget.

The program implements the required `propose_candidate` entry point, prints progress, and finally returns the best sequence as a plain Python list.

```python
"""
Propose a low‑score step‑function.

Algorithm
---------

* Randomly generate a modest pool of initial shapes (power‑law,
  exponential, two‑sided power‑law, Beta‑pdf) with several lengths
  (8 k … 30 k).  The previously discovered sequence `height_sequence_1`
  is used if it exists.

* Represent the coefficients as a soft‑max of logits `y`.  This puts
  the vector automatically on the probability simplex (non‑negative,
  sum = 1).

* Optimise the *smoothed* maximum of the convolution using Adam.
  For a temperature τ the smoothed quantity is

      Mτ(a) = (1/τ)·log Σₖ exp(τ·cₖ) ,   c = a*a .

  Its gradient w.r.t. a is 2·(p ⊛ a_rev) where pₖ ∝ exp(τ·cₖ)
  (probabilities) and a_rev is the reversed a.  Both the convolution
  `c` and the gradient are computed with FFTs.

* τ is annealed linearly from τ_start to τ_end during the optimisation,
  allowing a smooth start and a precise final focus on the true maximum.

* After the Adam phase a cheap stochastic local‑search (random multiplicative
  perturbation of a few coordinates, renormalisation) is performed on the
  best candidate.

* The best sequence found is evaluated with the official
  `evaluate_sequence` function and returned as a plain Python list.

The whole routine respects the 1000 s time budget, runs on a single
process, and uses only NumPy (no I/O, no external solvers).

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Exact objective for a normalised sequence (sum(seq) = 1):
        2 * n * max(conv(seq, seq))

    Uses an FFT for speed.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


def convolve_self(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of seq with itself (full length)."""
    n = seq.shape[0]
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return conv


def smooth_max_and_probs(conv: np.ndarray, tau: float):
    """
    Return the smoothed maximum Mτ(conv) and the probability vector
    pₖ = exp(τ·convₖ) / Σⱼ exp(τ·convⱼ) in a numerically stable way.
    """
    max_c = conv.max()
    # stabilised exponentials
    exp_term = np.exp(tau * (conv - max_c))
    sum_exp = exp_term.sum()
    M = max_c + (1.0 / tau) * np.log(sum_exp)
    probs = exp_term / sum_exp
    return M, probs


def gradient_wrt_a(a: np.ndarray, probs: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max w.r.t. a (before the soft‑max Jacobian).
    Implements   g = 2 * (probs ⊛ a_rev)   where ⊛ is linear convolution.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    # length of probs = 2n‑1, length of a_rev = n
    fft_len = next_pow2(len(probs) + n - 1)
    fp = np.fft.rfft(probs, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fp * fa_rev, n=fft_len)
    # pick the n entries corresponding to lags 0 … n‑1
    start = n - 1
    g = 2.0 * corr_full[start:start + n]
    return g


def adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    tau_start: float = 5.0,
    tau_end: float = 150.0,
    lr0: float = 0.3,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the soft‑max parametrisation of `a`.
    Returns the best sequence (normalised to sum = 1) and its exact
    evaluate‑sequence value (computed with `fast_score`).
    """
    n = a0.shape[0]
    # initialise logits from a0 (avoid log(0) with a tiny offset)
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # bookkeeping
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # --- soft‑max (ensures sum = 1) ---
        y_shift = y - np.max(y)               # improve numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()               # a lies on the simplex

        # --- convolution a*a ---
        conv = convolve_self(a)               # length 2n‑1

        # anneal temperature
        frac = min((now - start_time) / time_limit, 1.0)
        tau = tau_start + (tau_end - tau_start) * frac

        # --- smoothed maximum and probabilities ---
        M_smooth, probs = smooth_max_and_probs(conv, tau)

        # exact objective (use actual max for bookkeeping)
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # --- gradient wrt a (smooth) ---
        g_a = gradient_wrt_a(a, probs)       # shape (n,)

        # Jacobian of soft‑max:
        inner = np.dot(a, g_a)                # scalar ⟨a,g_a⟩
        grad_y = a * (g_a - inner)            # ∇_y

        # --- Adam update (bias‑corrected) ---
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # slowly decaying learning rate
        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape plateaus
        if it % 3000 == 0:
            sigma = 0.02 / math.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val


def random_initial_shape(n: int, rng: random.Random) -> np.ndarray:
    """
    Produce a random shape from a family of simple parametric forms.
    Families: power‑law, exponential, two‑sided power‑law, Beta‑pdf.
    The returned vector is normalised to sum = 1.
    """
    family = rng.choice(["power", "exp", "twosided", "beta"])
    if family == "power":
        p = rng.uniform(0.05, 1.5)
        idx = np.arange(1, n + 1, dtype=np.float64)
        seq = idx ** (-p)
    elif family == "exp":
        lam = rng.uniform(1e-4, 5e-2)
        idx = np.arange(n, dtype=np.float64)
        seq = np.exp(-lam * idx)
    elif family == "twosided":
        pl = rng.uniform(0.05, 1.5)
        pr = rng.uniform(0.05, 1.5)
        wl = rng.uniform(0.2, 5.0)
        wr = rng.uniform(0.2, 5.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        left = wl * i ** (-pl)
        right = wr * (n - i + 1) ** (-pr)
        seq = left + right
    else:  # beta
        a = rng.uniform(0.05, 2.0)
        b = rng.uniform(0.05, 2.0)
        i = np.arange(1, n + 1, dtype=np.float64)
        seq = (i ** (a - 1.0)) * ((n - i + 1) ** (b - 1.0))

    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s < 1e-12:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / s
    return seq


def stochastic_local_refine(
    a: np.ndarray,
    time_limit: float,
    start_time: float,
    sigma: float = 0.02,
    batch: int = 30,
) -> np.ndarray:
    """
    Simple stochastic hill‑climbing on the simplex.
    Multiplicative Gaussian noise on a random subset of coordinates,
    followed by renormalisation.  The best candidate seen is returned.
    """
    n = a.shape[0]
    best = a.copy()
    best_val = fast_score(best)

    it = 0
    while True:
        now = time.time()
        if now > start_time + time_limit:
            break
        it += 1

        # pick a random subset of coordinates
        idx = np.random.choice(n, size=batch, replace=False)
        delta = np.random.randn(batch) * sigma
        a_new = best.copy()
        a_new[idx] *= np.exp(delta)          # multiplicative perturbation
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()                 # re‑normalise

        val = fast_score(a_new)
        if val < best_val:
            best_val = val
            best = a_new.copy()

        # occasional larger jump
        if it % 1000 == 0:
            sigma = min(0.1, sigma * 1.05)

    return best
```
