def objective_and_grad(y, n, fft_len_conv, fft_len_g, lam_factor=0.3, beta=0.0):
    a = softmax(y)
    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    
    # gradient wrt a for smooth max
    w = exp_shift / Z
    # convolution a with w (reversed a)
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n-1 : n-1 + n]  # length n
    
    grad_smooth = a * (g_vec - np.dot(a, g_vec))
    
    # variance penalty term
    if beta != 0.0:
        J = np.sum(conv * conv)
        # gradient of J w.r.t a
        # need convolution conv with reversed a
        conv_fft_full = np.fft.fft(conv, fft_len_g)
        a_rev_fft = np.fft.fft(a[::-1], fft_len_g)
        conv_a_rev = np.real(np.fft.ifft(conv_fft_full * a_rev_fft))
        grad_J = 4.0 * conv_a_rev[n-1 : n-1 + n]
        total_grad = grad_smooth + beta * grad_J
    else:
        total_grad = grad_smooth
    
    # gradient w.r.t y
    mean_total = np.dot(a, total_grad)
    grad_y = a * (total_grad - mean_total)
    
    # objective value (only smooth max, ignore beta term for objective to keep scale)
    # but we may incorporate beta term into objective value to guide optimizer.
    obj = 2.0 * n * smooth_max
    if beta != 0.0:
        obj += beta * J
    return obj, grad_y
```

def generate_initial_y(mode, n, rng):
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n+1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n/2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # simple deterministic Sidon-like set
        chosen = []
        sums = set()
        for candidate in range(n):
            conflict = False
            for c in chosen:
                s = candidate + c
                if s in sums:
                    conflict = True
                    break
                if c*2 in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(candidate)
                for c in chosen:
                    sums.add(candidate + c)
            if len(chosen) >= int(np.sqrt(n)*1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "hann":
        a0 = np.hanning(n)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "hamming":
        a0 = np.hamming(n)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "blackman":
        a0 = np.blackman(n)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0

def project_to_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w
```

def simulated_annealing(a, cur_score, fft_len, n, deadline, init_temp=1e-2, final_temp=1e-6, steps=20000):
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    total_steps = steps
    start_time = time.time()
    while time.time() < deadline and total_steps > 0:
        total_steps -= 1
        # linear cooling based on elapsed time fraction
        elapsed = time.time() - start_time
        t_frac = elapsed / (deadline - start_time)
        T = init_temp * (1 - t_frac) + final_temp * t_frac
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.05
        if delta <= 0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # revert
            cur[i] += delta
            cur[j] -= delta
    return best, best_score

def subgradient_optimize(a, n, fft_len, deadline, init_step=0.5, min_step=1e-8):
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()
    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))
        # gradient for max: grad_i = 2 * best[s_star - i] if partner exists else 0
        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]
        grad += rng.normal(scale=1e-4, size=n)  # jitter

        a_new = project_to_simplex(a_new)
        new_score = fast_score(a_new, fft_len, n)

            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score

"""
Propose a step‑function sequence with a low evaluation score.

The algorithm combines several complementary ingredients:

1️⃣  Multiple random restarts on the probability simplex.
2️⃣  L‑BFGS optimisation of a smooth approximation of the max‑convolution
    with an *additional variance penalty* (∑(a∗a)²).  The penalty encourages
    a flatter autocorrelation which helps to lower the true max‑convolution.
3️⃣  Fast hill‑climbing moves (random mass transfers) and a short simulated‑annealing
    polishing phase to escape shallow local minima.
4️⃣  (When the problem size permits) a linear‑programming refinement that
    maximises the total mass of a partner vector under the current convolution bound
    and mixes it back into the current candidate.

All constraints required by `evaluate_sequence` are respected: the vector is
non‑negative, lives on the simplex (∑a = 1) and the code works entirely with
floating‑point numbers.  The search runs for the allotted time (≈1000 s) and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """Linear convolution a*a (length 2·n‑1) using FFT."""
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Quick evaluation of 2·n·max(a*a) for a probability vector (∑a = 1)."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
    beta: float = 0.0,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max‑convolution objective plus an optional variance penalty.
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # ----- linear convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    # smooth max via log‑sum‑exp
    shifted = conv - max_c                # now max = 0
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    # ----- gradient of smooth max -----
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # convolution a with w (reversed a) gives g_vec
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    grad_smooth = a * (g_vec - np.dot(a, g_vec))

    # ----- optional variance penalty (∑ conv²) -----
    if beta != 0.0:
        J = np.sum(conv * conv)                               # scalar
        # gradient of J w.r.t. a : 4 * (conv * a_rev) restricted to n entries
        conv_fft_full = np.fft.fft(conv, fft_len_g)
        a_rev_fft = np.fft.fft(a[::-1], fft_len_g)
        conv_a_rev = np.real(np.fft.ifft(conv_fft_full * a_rev_fft))
        grad_J = 4.0 * conv_a_rev[(n - 1) : (n - 1 + n)]
        total_grad = grad_smooth + beta * grad_J
    else:
        J = 0.0
        total_grad = grad_smooth

    # chain rule through the soft‑max
    mean_total = np.dot(a, total_grad)
    grad_y = a * (total_grad - mean_total)

    # objective (the variance term is only a heuristic, not part of the true score)
    obj = 2.0 * n * smooth_max + beta * J
    return obj, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_schedule: list[float],
    beta: float = 0.0,
    maxiter_per_stage: int = 250,
) -> np.ndarray:
    """
    Cascade of L‑BFGS runs with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_schedule:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam, beta),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
        )
        y = res.x
    return y


def generate_initial_y(mode: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a starting point y (pre‑soft‑max) for a given mode."""
    if mode == "random":
        y0 = rng.standard_normal(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a0 = np.exp(-rate * np.arange(n))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "power":
        p = rng.uniform(0.3, 2.0)
        a0 = (np.arange(1, n + 1, dtype=float) ** (-p))
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "edge":
        m_edge = max(1, int(n * rng.uniform(0.01, 0.04)))
        a_edge = np.sqrt(n / 2.0) / n
        a_center = 1.0 / np.sqrt(2.0) / n
        a0 = np.full(n, a_center)
        a0[:m_edge] = a_edge
        a0[-m_edge:] = a_edge
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "sidon":
        # simple deterministic Sidon‑like construction
        chosen = []
        sums = set()
        for cand in range(n):
            conflict = False
            for c in chosen:
                if cand + c in sums:
                    conflict = True
                    break
                if c * 2 in sums:
                    conflict = True
                    break
            if not conflict:
                chosen.append(cand)
                for c in chosen:
                    sums.add(cand + c)
            if len(chosen) >= int(np.sqrt(n) * 1.5):
                break
        k = len(chosen)
        a0 = np.zeros(n)
        a0[chosen] = 1.0 / k
        y0 = np.log(a0 + 1e-12)
    elif mode == "hann":
        a0 = np.hanning(n)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "hamming":
        a0 = np.hamming(n)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    elif mode == "blackman":
        a0 = np.blackman(n)
        a0 = np.maximum(a0, 0)
        a0 /= a0.sum()
        y0 = np.log(a0 + 1e-12)
    else:
        raise ValueError(f"unknown mode {mode}")
    return y0


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (non‑negative, sum = 1)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 3000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climb: move a random fraction of mass from i to j.
    Keep a move only if the fast_score strictly improves.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * 0.10
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def simulated_annealing(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    init_temp: float = 1e-2,
    final_temp: float = 1e-7,
    steps: int = 15000,
) -> tuple[np.ndarray, float]:
    """
    Stochastic mass‑transfer with a temperature schedule.
    Allows occasional uphill moves to escape local minima.
    """
    best = a.copy()
    best_score = cur_score
    cur = a.copy()
    cur_score = cur_score
    rng = np.random.default_rng()
    start = time.time()
    while time.time() < deadline and steps > 0:
        steps -= 1
        # linear cooling based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (deadline - start + 1e-12)
        T = init_temp * (1 - frac) + final_temp * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = cur[i] * rng.random() * 0.05
        if delta <= 0.0:
            continue
        cur[i] -= delta
        cur[j] += delta
        new_score = fast_score(cur, fft_len, n)
        diff = new_score - cur_score
        if diff < 0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # revert move
            cur[i] += delta
            cur[j] -= delta
    return best, best_score


def subgradient_optimize(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    init_step: float = 0.5,
    min_step: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution with projection onto the simplex.
    """
    best = a.copy()
    best_score = fast_score(best, fft_len, n)
    step = init_step
    rng = np.random.default_rng()

    while time.time() < deadline:
        conv = np.convolve(best, best)
        s_star = int(np.argmax(conv))

        i = np.arange(n)
        j = s_star - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(best)
        grad[i[mask]] = 2.0 * best[j[mask]]

        # small random jitter to avoid cycling
        grad += rng.normal(scale=1e-4, size=n)

        a_new = best - step * grad
        a_new = project_to_simplex(a_new)

        new_score = fast_score(a_new, fft_len, n)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            step = min(step * 1.2, 2.0)
        else:
            step *= 0.5
            if step < min_step:
                break
    return best, best_score


def lp_refine(
    a: np.ndarray,
    n: int,
    deadline: float,
    mix_factor: float = 0.2,
    max_iter: int = 5,
) -> np.ndarray:
    """
    Linear‑programming refinement.
    Solve maximise Σ b_j subject to b ≥ 0 and (a*b)[s] ≤ M,
    where M = max(a*a).  The optimal b is scaled to sum = 1 and mixed back.
    """
    rows = 2 * n - 1
    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        conv_aa = np.convolve(a, a)
        M = conv_aa.max()

        # build the convolution matrix A such that (A·b)[s] = (a*b)[s]
        A = sp.lil_matrix((rows, n))
        for col in range(n):
            A[col : col + n, col] = a
        A = A.tocsr()

        # maximise Σ b  ⇔  minimise −Σ b
        res = linprog(
            c=-np.ones(n),
            A_ub=A,
            b_ub=M * np.ones(rows),
            bounds=[(0, None)] * n,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            break

        g = b_opt / sum_b
        a_new = (1.0 - mix_factor) * a + mix_factor * g
        new_M = np.convolve(a_new, a_new).max()
        if new_M < M:
            a = a_new
            print(f"[lp] M reduced from {M:.6f} to {new_M:.6f}")
        else:
            break
    return a

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a plain list of non‑negative floats (sum > 0) ready for
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # safety margin

    # --------------------------------------------------------------
    # problem dimension – try to reuse an existing construction
    # --------------------------------------------------------------
    n = 5000
    init_seq = globals().get("height_sequence_1", None)
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        tmp = np.asarray(init_seq, dtype=float).flatten()
        if tmp.ndim == 1 and tmp.size >= 1000:
            if tmp.size < n:
                # pad with zeros
                tmp = np.concatenate([tmp, np.zeros(n - tmp.size)])
            else:
                tmp = tmp[:n]
            tmp = np.clip(tmp, 0, None)
            s = tmp.sum()
            if s < 1e-12:
                best_a = np.full(n, 1.0 / n)
            else:
                best_a = tmp / s
        else:
            best_a = np.full(n, 1.0 / n)
    else:
        best_a = np.full(n, 1.0 / n)

    best_score = fast_score(best_a, next_pow2(2 * n - 1), n)
    best_y = np.log(best_a + 1e-12)

    print(f"[init] n={n} initial fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # optimisation hyper‑parameters
    # --------------------------------------------------------------
    fft_len_conv = next_pow2(2 * n - 1)
    fft_len_g = next_pow2(3 * n - 2)

    lam_schedule = list(np.geomspace(0.4, 0.001, num=18))
    fine_lam_schedule = list(np.geomspace(0.3, 5e-5, num=22))
    maxiter_stage = 250       # moderate L‑BFGS work per stage
    beta = 0.5                # weight for the variance penalty

    modes = [
        "random",
        "exp",
        "power",
        "edge",
        "sidon",
        "hann",
        "hamming",
        "blackman",
    ]

    iteration = 0

    # --------------------------------------------------------------
    # Stochastic L‑BFGS + hill‑climb loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        mode = rng.choice(modes)
        if mode == "best":
            y0 = best_y + rng.normal(scale=0.02, size=n)
        else:
            y0 = generate_initial_y(mode, n, rng)

        y_opt = schedule_optimize(
            y0,
            n,
            fft_len_conv,
            fft_len_g,
            lam_schedule,
            beta=beta,
            maxiter_per_stage=maxiter_stage,
        )
        a_opt = softmax(y_opt)
        score_opt = fast_score(a_opt, fft_len_conv, n)

        if score_opt < best_score:
            best_a = a_opt
            best_score = score_opt
            best_y = y_opt
            print(f"[{iteration}] mode={mode} improved fast score = {best_score:.6f}")

        # aggressive local hill‑climb
        hc_deadline = min(deadline, time.time() + 0.6)
        best_a, best_score = hill_climb(
            best_a,
            best_score,
            fft_len_conv,
            n,
            hc_deadline,
            max_iter=4000,
        )
        best_y = np.log(best_a + 1e-12)
        iteration += 1

    # --------------------------------------------------------------
    # Sub‑gradient refinement on the exact max‑convolution
    # --------------------------------------------------------------
    sub_deadline = min(deadline, time.time() + 30.0)
    best_a, best_score = subgradient_optimize(
        best_a, n, fft_len_conv, sub_deadline, init_step=0.5
    )
    best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # Optional linear‑programming refinement (only for moderate n)
    # --------------------------------------------------------------
    if n <= 3000 and time.time() + 10.0 < deadline:
        lp_deadline = deadline - 5.0
        best_a = lp_refine(best_a, n, lp_deadline, mix_factor=0.25, max_iter=6)
        best_score = fast_score(best_a, fft_len_conv, n)
        best_y = np.log(best_a + 1e-12)
        print(f"[lp] post‑refine fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final fine L‑BFGS run with a very small smoothing factor
    # --------------------------------------------------------------
    for lam in fine_lam_schedule:
        if time.time() >= deadline:
            break
        res = minimize(
            fun=objective_and_grad,
            x0=best_y,
            args=(n, fft_len_conv, fft_len_g, lam, beta),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 600, "gtol": 1e-12, "disp": False},
        )
        best_y = res.x
        best_a = softmax(best_y)
        best_score = fast_score(best_a, fft_len_conv, n)
        print(f"[final L‑BFGS] lam={lam:.5g} fast score = {best_score:.6f}")

    # --------------------------------------------------------------
    # final simulated‑annealing polishing (use whatever time remains)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            deadline,
            init_temp=1e-2,
            final_temp=1e-7,
            steps=20000,
        )
        best_y = np.log(best_a + 1e-12)

    # --------------------------------------------------------------
    # official evaluation and return
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_val = evaluate_sequence(final_seq)  # type: ignore – provided by the harness
    print(f"[final] evaluated score = {final_val:.6f}")
    return final_seq
