# ---------- Utility ----------
def next_pow2(x: int) -> int:
    p = 1
    while p < x:
        p <<= 1
    return p

def fast_score(seq: np.ndarray) -> float:
    """Fast evaluation of evaluate_sequence (ignores clamping)."""
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))

# ---------- Shape generators ----------
def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)

def generate_power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    return seq / seq.sum()

def generate_exponential(n: int, lam: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    return seq / seq.sum()

def generate_two_sided_power_law(n: int,
                                 p_left: float,
                                 p_right: float,
                                 w_left: float = 1.0,
                                 w_right: float = 1.0) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    seq = left + right
    return seq / seq.sum()

def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0:

    else:

    return seq

    else:

def adam_optimize(a0: np.ndarray,
                  time_limit: float,
                  start_time: float,
                  lr0: float = 0.6,
                  beta1: float = 0.9,
                  beta2: float = 0.999,
                  eps: float = 1e-12,
                  jitter_every: int = 1500,
                  jitter_scale: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex (soft‑max parametrisation).
    Returns the best sequence discovered and its fast_score.
    """
    n = a0.shape[0]
    # initialise logits
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ---------- soft‑max ----------
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # ---------- convolution ----------
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M          # sum(a)=1 by construction
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---------- sub‑gradient ----------
        eps_act = 1e-12
        active = np.where(conv >= M - eps_act)[0]
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # ---------- gradient w.r.t logits ----------
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to keep diversity
        if it % jitter_every == 0:
            sigma = jitter_scale / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

    Simple simulated‑annealing style refinement that perturbs the

    y = np.log(a + 1e-16)
    best_y = y.copy()

        val = fast_score(a_new)
        if val < best_val:
            best_val = val
            best_y = y_new

    # return best sequence (soft‑max of best_y)
    exp_y = np.exp(best_y - np.max(best_y))

"""
Search for a non‑negative step‑function that minimizes the provided
``evaluate_sequence``.  The algorithm works in three phases:

1. **Random seeding** – generate a diverse pool of candidates from several
   parametric families (power‑law, exponential, two‑sided power‑law,
   Beta‑pdf, linear ramps, sine‑wave) with lengths between 8 k and 20 k.
   The best few (by a fast FFT‑based surrogate) are kept.

2. **Deep optimisation** – each selected seed is refined with Adam on the
   soft‑max (log‑soft‑max) parametrisation.  The objective is the exact
   quantity ``2·n·max(conv(a,a))/sum(a)^2`` (the sum is constantly 1 because
   of the soft‑max).  The gradient is the sub‑gradient of the max‑entry of
   the convolution; Adam’s adaptive steps and occasional Gaussian jitter
   help escape plateaus.

3. **Local refinement** – if time remains, a cheap simulated‑annealing
   style perturbation of the logits is performed that accepts any
   improvement, further polishing the best solution.

All work respects the 1000 s budget, prints progress, and finally returns
the best sequence as a plain list of floats.
"""

# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Fast surrogate for ``evaluate_sequence`` using FFT.
    The sequence is assumed to be non‑negative and already normalised
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    s = seq.sum()

        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))

def generate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n)


def generate_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = idx ** (-p)
    return seq / seq.sum()


def generate_exponential(n: int, lam: float) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    idx = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * idx)
    return seq / seq.sum()


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    seq = left + right
    return seq / seq.sum()


def generate_beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta‑pdf shape:
        a_i ∝ (i+1)^(alpha-1) * (n-i)^(beta-1)
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    seq = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    seq = np.maximum(seq, 0.0)
    s = seq.sum()
    if s == 0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / s
    return seq


def generate_linear_ramp(n: int, increasing: bool = True) -> np.ndarray:
    """Linear ramp (either increasing or decreasing)."""
    if increasing:
        seq = np.arange(1, n + 1, dtype=np.float64)
    else:
        seq = np.arange(n, 0, -1, dtype=np.float64)
    return seq / seq.sum()


def generate_sine_wave(n: int) -> np.ndarray:
    """Half‑sine shape (always positive)."""
    x = np.linspace(0, np.pi, n, dtype=np.float64)
    seq = np.sin(x) + 0.1  # shift to stay >0
    return seq / seq.sum()


# ----------------------------------------------------------------------
# Core optimiser (Adam on soft‑max parametrisation)
# ----------------------------------------------------------------------


def adam_optimize(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.6,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
    jitter_every: int = 1500,
    jitter_scale: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation on ``a`` (initialised from ``a0``) to minimise
    2·n·max(conv(a,a))/sum(a)^2.  The sequence is kept on the probability
    simplex via a log‑softmax parametrisation.
    Returns the best sequence encountered and its fast_score.
    """
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)          # logits

    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # ---- soft‑max ----
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()          # always on the simplex

        # ---- convolution ----
        fft_len = next_pow2(2 * n)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ---- sub‑gradient of the max ----
        eps_act = 1e-12
        active = np.where(conv >= M - eps_act)[0]
        g = np.zeros_like(a)

        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            j_range = k - i_range
            g[i_range] += 2.0 * a[j_range]

        # ---- gradient w.r.t. logits (soft‑max Jacobian) ----
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- occasional jitter to escape flat plateaus ----
        if it % jitter_every == 0:
            sigma = jitter_scale / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val


def local_refine(
    a: np.ndarray,
    time_budget: float,
    start_time: float,
    temp0: float = 0.1,
    temp_decay: float = 0.95,
) -> tuple[np.ndarray, float]:
    """
    Light simulated‑annealing style refinement that perturbs the log‑softmax
    variables and keeps any improvement.  Works on the simplex.
    """
    n = a.shape[0]
    y = np.log(a + 1e-16)
    best_y = y.copy()
    best_val = fast_score(a)

    t0 = time.time()
    while time.time() < start_time + time_budget:
        elapsed = time.time() - t0
        frac = elapsed / time_budget
        temp = max(temp0 * (1.0 - frac), 1e-8)
        sigma = temp / np.sqrt(n)

        y_new = best_y + sigma * np.random.randn(n)
        exp_y = np.exp(y_new - np.max(y_new))
        a_new = exp_y / exp_y.sum()
        val = fast_score(a_new)

        if val < best_val:
            best_val = val
            best_y = y_new

    exp_y = np.exp(best_y - np.max(best_y))
    best_seq = exp_y / exp_y.sum()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: int = 1000) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    ``evaluate_sequence`` value.  The routine respects the 1000 s time budget,
    mixes random initialisation, Adam optimisation and a final refinement,
    prints progress information and finally returns the best sequence as a
    plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # small safety margin

    # ------------------------------------------------------------------
    # 0) Build a pool of initial candidates (including a possible previous best)
    # ------------------------------------------------------------------
    init_candidates = []

    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)   # type: ignore[name-defined]
        if prev.ndim == 1 and prev.size > 0:
            prev = np.maximum(prev, 0.0)
            if prev.sum() < 1e-12:
                prev = generate_uniform(prev.size)
            else:
                prev = prev / prev.sum()
            init_candidates.append(prev.copy())
            print("[info] using previously discovered sequence as a start")
    except Exception:
        pass

    # guaranteed fallback – uniform distribution of moderate length
    default_len = 10000
    init_candidates.append(generate_uniform(default_len))
    print(f"[info] fallback uniform sequence added (len={default_len})")

    # ------------------------------------------------------------------
    # 1) Populate the pool with random shapes
    # ------------------------------------------------------------------
    length_choices = [8000, 10000, 12000, 15000, 20000]
    families = ["power", "exp", "twosided", "beta", "linear", "sine"]
    desired_pool = 30

    while len(init_candidates) < desired_pool and time.time() < deadline:
        n = random.choice(length_choices)
        family = random.choice(families)

        if family == "power":
            p = random.uniform(0.05, 2.0)
            seq = generate_power_law(n, p)
        elif family == "exp":
            lam = random.uniform(1e-4, 5e-2)
            seq = generate_exponential(n, lam)
        elif family == "twosided":
            pl = random.uniform(0.05, 2.0)
            pr = random.uniform(0.05, 2.0)
            wl = random.uniform(0.2, 5.0)
            wr = random.uniform(0.2, 5.0)
            seq = generate_two_sided_power_law(n, pl, pr, wl, wr)
        elif family == "beta":
            a = random.uniform(0.05, 3.0)
            b = random.uniform(0.05, 3.0)
            seq = generate_beta_shape(n, a, b)
        elif family == "linear":
            inc = random.choice([True, False])
            seq = generate_linear_ramp(n, inc)
        else:  # sine
            seq = generate_sine_wave(n)

        init_candidates.append(seq)

    # ------------------------------------------------------------------
    # 2) Quick evaluation & keep the most promising seeds
    # ------------------------------------------------------------------
    scores = []
    for i, seq in enumerate(init_candidates):
        sc = fast_score(seq)
        scores.append(sc)
        print(f"[init] candidate {i+1}/{len(init_candidates)} len={seq.size} score={sc:.6f}")

    top_k = 5
    top_indices = np.argsort(scores)[:top_k]
    print("[info] selected top", top_k, "candidates for deep optimisation")

    # ------------------------------------------------------------------
    # 3) Deep optimisation (Adam) on the selected seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining <= 0:
        raise RuntimeError("No time left after initialisation")
    per_candidate = remaining / len(top_indices)

    best_overall_seq = None
    best_overall_val = float("inf")

    for rank, idx in enumerate(top_indices):
        if time.time() > deadline:
            break
        a0 = init_candidates[int(idx)]
        print(
            f"[optim] start deep optimisation #{rank+1}/{len(top_indices)} "
            f"(len={a0.size}) – budget {per_candidate:.1f}s"
        )
        seq, val = adam_optimize(a0, per_candidate, time.time())
        print(f"[optim] candidate #{rank+1} best fast_score {val:.6f}")
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"   >>> new global best fast_score = {best_overall_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final local refinement if time permits
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0 and best_overall_seq is not None:
        print("[refine] running final local refinement with budget",
              f"{remaining:.1f}s")
        seq, val = local_refine(best_overall_seq, remaining, time.time())
        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print("[refine] improvement after refinement:", val)

    # ------------------------------------------------------------------
    # 5) Verification with the official evaluator
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = generate_uniform(10000)

    final_list = best_overall_seq.tolist()
    final_score = evaluate_sequence(final_list)  # type: ignore[name-defined]
    print(f"Finished. evaluate_sequence reports {final_score:.6f}")
    return final_list
