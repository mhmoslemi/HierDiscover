def adam_optimize_smooth(a0, time_limit, start_time, lr0=0.1, beta1=0.9, beta2=0.999, eps=1e-8, tau0=0.1, tau_decay=0.9995, min_tau=1e-4):

    n = a0.shape[0]
    y = np.log(a0 + 1e-16) # logits
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_seq = a0.copy()
    best_val = fast_score(best_seq)
    it = 0
    while True:
        it+=1
        t = time.time()
        if t > start_time+time_limit:
            break

        # compute a via softmax
        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # compute conv a*a
        fft_len = next_pow2(2*n)
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa*Fa, n=fft_len)[:2*n-1]

        M = conv.max()
        cur_val = 2.0*n*M  # sum =1 due to softmax
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # compute smoothed gradient
        tau = max(tau0 * (tau_decay**it), min_tau)
        # weights
        w = np.exp((conv - M) / tau)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w = w / w_sum

        # gradient of weighted sum: g = 2 * (a * (w_rev convolve))
        # compute convolution of a with reversed w
        w_rev = w[::-1]
        # conv2 length n + len(w_rev) -1 = n + (2n-1)-1 = 3n-2
        fft_len2 = next_pow2(n + (2*n-1) - 1)
        Fa2 = np.fft.rfft(a, n=fft_len2)
        Wf = np.fft.rfft(w_rev, n=fft_len2)
        g = np.fft.irfft(Fa2 * Wf, n=fft_len2)[:n] * 2.0

        # gradient w.r.t softmax logits
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # Adam update
        m = beta1 * m + (1-beta1) * grad_y
        v = beta2 * v + (1-beta2) * (grad_y**2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to avoid plateau
        if it % 5000 == 0:
            sigma = 0.01/np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

def refine_seq(seq, time_limit, start_time, step_size=0.01):
    n = len(seq)
    best_seq = seq.copy()
    best_score = fast_score(best_seq)
    while time.time() < start_time+time_limit:
        # propose random move: pick i!=j, delta = step_size * best_seq[i] * random_uniform(0,1)
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        if best_seq[i] < 1e-12:
            continue
        delta = (np.random.rand()*step_size) * best_seq[i]
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure nonneg.
        # compute score quickly using fast_score
        val = fast_score(new_seq)
        if val < best_score:
            best_score = val
            best_seq = new_seq
    return best_seq, best_score

def adam_optimize_smooth(a0, time_limit, start_time):
    n = a0.shape[0]
    y = np.log(a0 + 1e-16)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    it = 0
    tau0 = 0.05
    tau_min = 1e-4
    tau_decay = 0.9995
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        y_shift = y - np.max(y)
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # compute convolution via FFT
        L = next_pow2(3*n)  # Ensure enough length for both conv and correlation
        Fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(Fa * Fa, n=L)[:2*n-1]
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # smoothed weights
        tau = max(tau_min, tau0 * (tau_decay ** it))
        # stable exp: shift by max
        w = np.exp((conv - M) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # compute gradient: g = 2 * correlation(a, w)
        w_rev = w[::-1]
        W_rev_f = np.fft.rfft(w_rev, n=L)
        corr_full = np.fft.irfft(Fa * W_rev_f, n=L)
        # extract segment n-1 .. 2*n-2 (inclusive)
        segment = corr_full[n-1:2*n-1]
        g = 2.0 * segment[::-1]   # reverse to align with formula

        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = 0.5 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # jitter occasionally
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

"""
Propose a low‑score step‑function.

Algorithm
---------

* **Initial seeds** – we start from the previously known best
  `height_sequence_1` (if it exists), a deterministic two‑sided
  power‑law shape and a few random Dirichlet vectors.  All seeds are
  normalised to sum = 1 (soft‑max parametrisation).

* **Smoothed‑max Adam optimisation** – the objective
  ``2·n·max(conv(a,a))`` is approximated by a smoothed max
  ``(1/τ)·log∑exp(τ·conv)``.  The gradient of the weighted sum
  ``∑ w_k·conv_k`` (with ``w_k∝exp(τ·conv_k)``) can be written as a
  correlation and computed with two FFTs per iteration.
  The parameters are optimised in the log‑softmax space with Adam.
  The temperature ``τ`` is decreased gradually so that the optimisation
  first sees a smooth landscape and finally focuses on the true max.

* **Local random‑swap refinement** – after the main optimisation the
  best sequence is refined by many tiny mass‑redistribution moves
  (simulated‑annealing style) that are evaluated with the fast FFT
  based score.

* **Time‑budget handling** – the whole 1000 s budget is split between a
  few restarts of the Adam optimiser and a final refinement phase.
  Progress and intermediate scores are printed.

The code respects the 2‑CPU limit, uses only NumPy and the supplied
`evaluate_sequence` and returns the best sequence found as a plain list.
"""

import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    p = 1
    while p < x:
        p <<= 1
    return p


def fast_score(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) / (sum(seq)^2) using an FFT.
    The sequence must be non‑negative.
    """
    n = seq.shape[0]
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    fft_len = next_pow2(2 * n)
    fa = np.fft.rfft(seq, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Shape generators (used for random starts)
# ----------------------------------------------------------------------


def generate_two_sided_power_law(
    n: int,
    p_left: float,
    p_right: float,
    w_left: float = 1.0,
    w_right: float = 1.0,
) -> np.ndarray:
    """
    a_i = w_left·(i+1)^(-p_left) + w_right·(n-i)^(-p_right).
    """
    i = np.arange(1, n + 1, dtype=np.float64)
    left = w_left * i ** (-p_left)
    right = w_right * (n - i + 1) ** (-p_right)
    return left + right


# ----------------------------------------------------------------------
# Smoothed‑max Adam optimiser (FFT based gradient)
# ----------------------------------------------------------------------


def adam_optimize_smooth(
    a0: np.ndarray,
    time_limit: float,
    start_time: float,
    lr0: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau0: float = 0.05,
    tau_min: float = 1e-4,
    tau_decay: float = 0.9995,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the simplex (soft‑max parametrisation) using a
    log‑sum‑exp smoothing of the max‑convolution.
    Returns the best sequence encountered and its (unsmoothed) score.
    """
    n = a0.shape[0]
    # logits, initialise from the current probabilities
    y = np.log(a0 + 1e-16)

    # Adam state
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    best_seq = a0.copy()
    best_val = fast_score(best_seq)

    # length needed for the correlation (a * w_rev) – at most 3·n
    fft_len = next_pow2(3 * n)

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > start_time + time_limit:
            break

        # --- soft‑max (stays on the probability simplex) ---
        y_shift = y - np.max(y)                # numerical stability
        exp_y = np.exp(y_shift)
        a = exp_y / exp_y.sum()

        # --- convolution a * a (full length 2·n‑1) ---
        Fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]
        M = float(conv.max())
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # --- smoothed weights (log‑sum‑exp) ---
        tau = max(tau_min, tau0 * (tau_decay ** it))
        # stable exponentiation
        w = np.exp((conv - M) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of weighted sum = 2·corr(a, w) ---
        w_rev = w[::-1]                     # reverse for correlation
        W_rev_f = np.fft.rfft(w_rev, n=fft_len)
        corr_full = np.fft.irfft(Fa * W_rev_f, n=fft_len)
        # relevant slice: indices n‑1 … 2·n‑2  (length n)
        segment = corr_full[n - 1 : 2 * n - 1]
        g = 2.0 * segment[::-1]             # reverse to obtain g_i

        # --- gradient w.r.t. logits (soft‑max Jacobian) ---
        ag = np.dot(a, g)
        grad_y = a * (g - ag)

        # --- Adam update ---
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (np.sqrt(it) + 1.0)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to escape flat plateaus
        if it % 5000 == 0:
            sigma = 0.01 / np.sqrt(n)
            y += sigma * np.random.randn(n)

    return best_seq, best_val

# ----------------------------------------------------------------------


def refine_seq(
    seq: np.ndarray,
    time_limit: float,
    start_time: float,
    step_frac: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Tiny mass‑redistribution moves (move a random fraction of the mass
    from one index to another).  Accept a move only if it improves the
    score.  Returns the best sequence found during the allocated time.
    """
    n = seq.shape[0]
    best_seq = seq.copy()
    best_val = fast_score(best_seq)

    while True:
        now = time.time()
        if now > start_time + time_limit:
            break

        i = np.random.randint(n)
        if best_seq[i] < 1e-12:
            continue
        # pick a different destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        delta = (np.random.rand() * step_frac) * best_seq[i]  # up to step_frac of the source
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        val = fast_score(new_seq)
        if val < best_val:
            best_val = val
            best_seq = new_seq

    return best_seq, best_val
