def proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]

def low_pass(a, alpha):
    n=len(a)
    # compute FFT of a with length next_pow2.
    fft_len = next_pow2(n)
    A = np.fft.fft(a, fft_len)
    freqs = np.fft.fftfreq(fft_len)
    # compute Gaussian filter
    filter = np.exp(-alpha * (freqs**2))
    A_filt = A * filter
    a_smooth = np.real(np.fft.ifft(A_filt))[:n]
    a_smooth = np.maximum(a_smooth, 0)
    a_smooth /= a_smooth.sum()

"""
Hybrid optimiser for the step‑function attack.

The optimiser consists of three phases:

1. **Continuation L‑BFGS** – a smooth approximation of the max‑convolution
   is minimised on the soft‑max logits.  A geometric λ‑schedule drives the
   smoothing to zero, quickly locating a good basin.

2. **Intensive local search** – a mixture of greedy targeted hill‑climb,
   random hill‑climb and a simple simulated‑annealing step.  The gradient
   of the true max‑convolution is used directly, and mass is repeatedly
   transferred from indices that contribute most to the current peak to
   those that contribute least.

3. **Low‑pass smoothing** – occasional Gaussian smoothing of the
   probability vector helps to escape shallow local minima.

The algorithm runs as long as time permits (budget_s seconds) and returns
the best non‑negative sequence found.  The returned list sums to one,
which makes the fast_score (`2·n·max(a*a)`) equal to the value produced
by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max producing a probability vector (sum = 1)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def conv_fft(a: np.ndarray, fft_len: int, n: int) -> np.ndarray:
    """
    Linear convolution of ``a`` with itself using FFT.
    Returns the full convolution of length ``2*n-1``.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.fft.ifft(fa * fa)
    return np.real(conv)[: 2 * n - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of ``2·n·max(a*a)`` for a probability vector ``a``."""
    conv = conv_fft(a, fft_len, n)
    return float(2.0 * n * conv.max())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    w >= 0, sum w = 1.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def low_pass(a: np.ndarray, alpha: float = 0.01) -> np.ndarray:
    """
    Gentle Gaussian low‑pass smoothing of a probability vector.
    The smoothing strength is controlled by ``alpha`` (larger = more smoothing).
    """
    n = a.shape[0]
    fft_len = next_pow2(n)
    A = np.fft.fft(a, fft_len)
    freqs = np.fft.fftfreq(fft_len)
    filt = np.exp(-alpha * (freqs ** 2))
    A_filt = A * filt
    a_s = np.real(np.fft.ifft(A_filt))[:n]
    a_s = np.maximum(a_s, 0.0)
    s = a_s.sum()
    if s > 0:
        a_s /= s
    else:
        a_s = np.full(n, 1.0 / n)
    return a_s


# ----------------------------------------------------------------------
# Smooth max objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of the max‑convolution and its gradient w.r.t.
    the unconstrained logits ``y``.
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam = max_c * lam_factor + 1e-12

    shifted = conv - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max
    w = exp_shift / Z                           # d smooth_max / d conv_i

    # g_i = Σ_j a_j * w_{i+j}, i = 0..n‑1
    fa_rev = np.fft.fft(a[::-1], fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def schedule_optimize(
    y_start: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factors: list[float],
    maxiter_per_stage: int = 400,
) -> np.ndarray:
    """
    Continuation L‑BFGS: a sequence of L‑BFGS runs with decreasing smoothing.
    """
    y = y_start.copy()
    for lam in lam_factors:
        res = minimize(
            fun=objective_and_grad,
            x0=y,
            args=(n, fft_len_conv, fft_len_g, lam),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter_per_stage, "gtol": 1e-14, "disp": False},
        )
        y = res.x
    return y


# ----------------------------------------------------------------------
# Hill‑climb utilities
# ----------------------------------------------------------------------


def hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple[np.ndarray, float]:
    """Fast random hill‑climb on the simplex."""
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()
    max_move_frac = 0.10

    while time.time() < deadline and max_iter > 0:
        max_iter -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = best[i] * rng.random() * max_move_frac
        if delta <= 0.0:
            continue
        best[i] -= delta
        best[j] += delta
        new_score = fast_score(best, fft_len, n)
        if new_score < best_score:
            best_score = new_score
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_score


def targeted_hill_climb(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    max_moves: int = 3000,
    frac: float = 0.10,
) -> tuple[np.ndarray, float]:
    """
    Greedy sub‑gradient hill‑climb: mass is moved from the worst
    contributor to the best one for the current peak.
    """
    best = a.copy()
    best_score = cur_score
    rng = np.random.default_rng()

    while time.time() < deadline and max_moves > 0:
        conv = conv_fft(best, fft_len, n)
        k_max = int(np.argmax(conv))

        # contribution of each index to that peak
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        contrib = np.zeros_like(best)
        contrib[mask] = best[mask] * best[idx[mask]]

        i_src = int(np.argmax(contrib))
        i_dst = int(np.argmin(contrib))

        if i_src == i_dst:
            i_src = rng.integers(0, n)
            i_dst = rng.integers(0, n)
            if i_src == i_dst:
                max_moves -= 1
                continue

        delta = best[i_src] * rng.random() * frac
        if delta <= 0.0:
            max_moves -= 1
            continue

        best[i_src] -= delta
        best[i_dst] += delta
        new_score = fast_score(best, fft_len, n)

        if new_score < best_score - 1e-12:
            best_score = new_score
        else:
            # revert
            best[i_src] += delta
            best[i_dst] -= delta

        max_moves -= 1

    return best, best_score


def simulated_annealing_step(
    a: np.ndarray,
    cur_score: float,
    fft_len: int,
    n: int,
    rng: np.random.Generator,
    temperature: float,
) -> tuple[np.ndarray, float]:
    """
    Perform a single SA move: transfer a small random amount of mass
    from one index to another and accept according to the Metropolis rule.
    """
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    if a[i] == 0.0:
        return a, cur_score
    # transfer up to 5 % of a[i]
    delta = a[i] * rng.random() * 0.05
    cand = a.copy()
    cand[i] -= delta
    cand[j] += delta
    new_score = fast_score(cand, fft_len, n)
    diff = new_score - cur_score
    if diff < 0 or rng.random() < np.exp(-diff / (temperature + 1e-16)):
        return cand, new_score
    return a, cur_score
