"""
Simple two‑block construction.

We take a long step function and put equal mass on the first
k positions and the last k positions, leaving a gap of at least
one zero in the middle.  With k = ⌊(n‑1)/2⌋ this gives a
convolution whose maximum value is k·(1/(2k))² = 1/(4k),
hence the official score is

    2·n·max(conv) = n/(2k) ≈ 1.0005  for n = 4000.

The construction is deterministic, runs in milliseconds and
produces a score far below the target 1.503.
If a previous construction (`height_sequence_1`) is available we
compare both and keep the better one.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a non‑negative step sequence with a very low evaluation score.
    The sequence length is fixed to 4000 (the size used by the reference
    code), but the method works for any n.

    Parameters
    ----------
    seed : int (unused, kept for API compatibility)
    budget_s : int (unused, the algorithm finishes instantly)

    Returns
    -------
    list[float]
        The best sequence found.
    """
    # --------------------------------------------------------------
    # 1️⃣  Build the two‑block candidate
    # --------------------------------------------------------------
    n = 4000                                 # length of the step function
    k = (n - 1) // 2                         # size of each non‑zero block
    if k == 0:                               # degenerate case (n very small)
        a = np.full(n, 1.0 / n)
    else:
        a = np.zeros(n, dtype=np.float64)
        w = 1.0 / (2 * k)                    # each non‑zero entry gets the same weight
        a[:k] = w
        a[-k:] = w
        # normalise to eliminate possible rounding drift
        a /= a.sum()

    # --------------------------------------------------------------
    # 2️⃣  Evaluate the candidate (the official evaluator will be called
    #     by the judging system, but we also print the score for debugging)
    # --------------------------------------------------------------
    try:
        # the global `evaluate_sequence` function is provided by the environment
        score = evaluate_sequence(a.tolist())
        print(f"[candidate] n={n}, k={k}, evaluate_sequence = {score:.6f}")
    except Exception as exc:                # pragma: no cover – defensive
        print("[candidate] evaluation failed:", exc)
        score = None

    # --------------------------------------------------------------
    # 3️⃣  If a previous best construction is available, keep the better one
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # make sure the old sequence obeys the same non‑negative, normalised constraints
            old = np.maximum(old, 0.0)
            if old.sum() == 0.0:
                old = np.full_like(old, 1.0 / old.size)
            else:
                old /= old.sum()
            old_score = evaluate_sequence(old.tolist())
            print(f"[old] previous evaluate_sequence = {old_score:.6f}")
            if old_score is not None and (score is None or old_score < score):
                print("[candidate] keeping the older better construction")
                return old.tolist()

    return a.tolist()
