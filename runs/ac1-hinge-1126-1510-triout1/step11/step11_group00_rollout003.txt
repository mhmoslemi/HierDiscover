"""
Propose a low‑score step function.

Algorithm
---------
* Choose a relatively large length `n` (power of two) – a finer
  discretisation gives the optimiser more freedom to lower the
  continuous limit of the objective.
* Build a diverse pool of initial probability vectors (uniform,
  spectral, Hann, Blackman, Gaussian, random Dirichlet‑like, power‑law…)
  and keep the best according to the exact objective
          F(a) = 2·n·max(a*a)      (a on the simplex).
* **Phase 1 – multi‑restart L‑BFGS on a smoothed max**.
  The “max’’ is replaced by a log‑sum‑exp with an aggressive schedule of
  smoothing parameters λ → 0.  Each run starts from a convex combination
  of the current best and a fresh random vector.  The best result of all
  restarts is retained.
* **Phase 2 – Adam on the exact max‑convolution**.
  The gradient of the exact max is obtained by convolving the reversed
  vector with a mask that is 1 on the active (max‑valued) indices.
  Adam works on the logits `y` (softmax(y) lives on the simplex) and
  includes occasional Gaussian jitter to escape plateaux.
* **Phase 3 – greedy hill‑climb** (tiny mass‑transfer moves) for a
  final polish.
* Return the best sequence found.  The official `evaluate_sequence`
  rescales internally, therefore the returned list is left normalised
  (sum = 1) and consists of non‑negative floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast exact evaluation (sum=1 ⇒ objective = 2·n·max(conv))
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective for a probability vector a (sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max objective (log‑sum‑exp) and its gradient (L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len_conv: int,
                       fft_len_g: int, lam_factor: float = 0.30):
    """
    Smoothed max of the self‑convolution using log‑sum‑exp.
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of the smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (all return a probability vector, sum=1)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Inverse‑FFT based spectral guess (projected onto simplex)."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power',
                       'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape) normalised to sum = 1."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def lbfgs_continuation(a_start: np.ndarray, lam_schedule, maxiter_per_lam,
                       deadline, fft_len_conv, fft_len_g, n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            # Should never happen, but keep the loop robust.
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a, best_sc


def adam_optimize(a_start: np.ndarray, deadline: float,
                  n: int, fft_len: int, rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y so that a = softmax(y) stays on the simplex.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_sc = fast_score(best_a, fft_len, n)

    t = 0
    while time.time() < deadline:
        t += 1
        a = softmax(y)

        # convolution and exact max
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        # mask of (almost) active indices
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        # gradient wrt a (exact for the max)
        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)

        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]          # d max / d a
        grad_a = 2.0 * n * g_a                               # d F / d a

        # chain rule through softmax
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        # Adam bias‑corrected updates
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # decaying learning rate (aggressive early, gentle later)
        lr = 0.3 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional jitter to hop out of flat plateaux
        if t % 4000 == 0:
            y += rng.normal(scale=5e-4, size=y.shape)

        # keep the best ever seen
        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = a_curr.copy()

    return best_a, best_sc


def hill_climb(a: np.ndarray, cur_score: float,
               fft_len: int, n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 50000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb with tiny mass‑transfer moves.
    """
    best = a.copy()
    best_sc = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        # propose moving a fraction of mass from i to j
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0.0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_sc:
            best_sc = sc
        else:
            # revert – we only accept strict improvements
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_sc

    candidates = []

    # If a previous solution is available, interpolate it to the current size.
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Randomised families (multiple draws)
    for _ in range(6):
        candidates.append(random_initial(n, rng))
    for _ in range(3):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Choose the best initial candidate according to the exact objective.
    best_a = candidates[0].copy()
    best_sc = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Phase 1 – multi‑restart L‑BFGS on a smoothed max
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.35 * budget_s
    lam_schedule = [0.5, 0.3, 0.2, 0.1, 0.05, 0.025, 0.01,
                    0.005, 0.002, 0.001, 5e-4, 2e-4, 1e-4,
                    5e-5, 2e-5, 1e-5]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        # Blend current best with a fresh random vector – this keeps diversity.
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.25
        a_start = (1.0 - alpha) * best_a + alpha * rand_vec
        a_start /= a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_end,
                                           fft_len, fft_len, n)
        if sc_opt < best_sc:
            best_sc = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_sc:.8f}")

        # Early exit if we have already beaten the target.
        if best_sc <= 1.503:
            break

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline and best_sc > 1.503:
        print("[Adam] start exact optimisation")
        best_a, best_sc = adam_optimize(best_a,
                                        deadline,
                                        n,
                                        fft_len,
                                        rng)
        print(f"[Adam] best fast_score={best_sc:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline and best_sc > 1.503:
        remaining = deadline - time.time()
        hill_deadline = time.time() + min(remaining, 30.0)
        best_a, best_sc = hill_climb(best_a,
                                     best_sc,
                                     fft_len,
                                     n,
                                     hill_deadline,
                                     rng,
                                     max_iter=40000,
                                     delta_frac=0.05)
        print(f"[final hill‑climb] best fast_score={best_sc:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
