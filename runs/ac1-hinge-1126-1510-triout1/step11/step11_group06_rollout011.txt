    n = len(a)
    conv = np.convolve(a, a) # length 2n-1

    best_a = a.copy()
    best_score = score

    total_time = deadline - time.time()
    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2 * n * cur_max
        # If cur_score is already <= best_score, already stored.
        # Compute subgradient mask and g

        src = np.argmax(g)
        dst = np.argmin(g)
        if src == dst or a[src] <= 0:
            # random fallback
            src = int(rng.choice(np.where(a > 1e-12)[0]))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue
        # step factor schedule
        remaining = max(0.0, deadline - time.time())

        delta_max = a[src]
        # start with delta = delta_max * factor

        if delta < 1e-15:

        # we will do line search decreasing delta if needed
        # compute linear coeff once
        lin = np.zeros_like(conv)
        # indices for src
        lin[src:src+n] -= 2.0 * a
        # indices for dst

        # quadratic coeff

        idx2dst = 2*dst

                    best_a = a.copy()
                break

"""
Improved step‑function search for the min‑max self‑convolution objective.

The algorithm proceeds in three stages:

1.  Load an existing good construction (if any).  Optionally perform a short
    exponentiated‑gradient pass to escape obvious plateaus.
2.  A *gradient‑guided hill‑climb* that moves a fraction of mass from the
    index with the largest sub‑gradient (most responsible for the current
    peak) to the index with the smallest sub‑gradient.  The move size is
    chosen by a simple line‑search that evaluates the exact convolution after
    the transfer (including the quadratic delta² terms).  The step factor
    decays linearly with the remaining time, guaranteeing fine adjustments
    at the end.
3.  A final Adam optimisation on the smooth log‑sum‑exp surrogate to polish
    the result.

All steps respect the time budget (default 1000 s).  The returned sequence
is a non‑negative probability vector; ``evaluate_sequence`` returns the
objective value, which should be ≤ 1.503 for a successful run.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of max(conv).
    Returns (objective, gradient) w.r.t. logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max wrt convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len)
    fw = np.fft.fft(w, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb with exact line‑search
# ----------------------------------------------------------------------


def guided_hill_climb_opt(a_start: np.ndarray,
                          start_score: float,
                          deadline: float,
                          rng: np.random.Generator,
                          start_factor: float = 0.25,
                          end_factor: float = 0.001):
    """
    Gradient‑guided hill‑climb.
    At each iteration a fraction of mass is moved from the index with the
    largest sub‑gradient to the index with the smallest.  The step size
    (δ) is chosen by a simple back‑tracking line‑search that evaluates the
    *exact* convolution after the move (linear term + quadratic δ² term).
    The step factor decays linearly with the remaining time.
    Returns the best a and its fast_score before the deadline.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact linear convolution, length 2n‑1
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score
    total_time = max(1e-12, deadline - time.time())

    steps_without_improve = 0
    max_no_improve = 5000   # after many stagnant steps we inject randomness

    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # sub‑gradient of the max‑convolution
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback random move
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz, p=a[nz] / a[nz].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # line‑search parameters
        remaining_frac = max(0.0, (deadline - time.time()) / total_time)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta_max = a[src]
        delta = delta_max * step_factor
        if delta < 1e-15:
            continue

        # pre‑compute linear and quadratic change terms (using the *current* a)
        lin = np.zeros_like(conv)
        lin[src:src + n] -= 2.0 * a
        lin[dst:dst + n] += 2.0 * a

        quad = np.zeros_like(conv)
        idx2src = 2 * src
        idx2dst = 2 * dst
        idx_sum = src + dst
        quad[idx2src] += 1.0
        quad[idx2dst] += 1.0
        quad[idx_sum] -= 2.0

        improved = False
        # back‑track until the move reduces the peak (or delta becomes tiny)
        for _ in range(20):
            new_conv = conv + delta * lin + (delta * delta) * quad
            new_max = new_conv.max()
            new_score = 2.0 * n * new_max
            if new_score <= cur_score:
                # accept
                a[src] -= delta
                a[dst] += delta
                conv = new_conv
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_a = a.copy()
                improved = True
                steps_without_improve = 0
                break
            else:
                delta *= 0.5

        if not improved:
            steps_without_improve += 1
            # occasional tiny random perturbation to escape stagnation
            if steps_without_improve >= max_no_improve:
                # move a miniscule amount between two random indices
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 1e-12:
                    tiny = min(a[i], 1e-9)
                    a[i] -= tiny
                    a[j] += tiny
                    conv = np.convolve(a, a)    # exact recompute
                steps_without_improve = 0

        # periodic exact recompute to keep numerical drift in check
        if np.mod(time.time(), 100.0) < 0.01:   # cheap periodic trigger
            conv = np.convolve(a, a)

    return best_a, best_score

    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from a previously stored construction if available
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # Linear interpolation to required length
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – optional exp‑gradient to shake off obvious plateaus
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_tmp, sc_tmp = exp_gradient_optimize(best_a, deadline, n,
                                             fft_len, rng, eta0=12.0)
        if sc_tmp < best_score:
            best_score = sc_tmp
            best_a = a_tmp.copy()
            print(f"[exp‑grad] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – gradient‑guided hill‑climb with exact line‑search
    # --------------------------------------------------------------
    if time.time() < deadline:
        a_hc, sc_hc = guided_hill_climb_opt(best_a, best_score,
                                            deadline, rng,
                                            start_factor=0.25,
                                            end_factor=0.001)
        if sc_hc < best_score:
            best_score = sc_hc
            best_a = a_hc.copy()
            print(f"[guided_hc_opt] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final Adam refinement on the smoothed surrogate
    # --------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = [0.4, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(y0, lam_sched,
                                           max_steps_per_lam=1500,
                                           deadline=deadline,
                                           n=n, fft_len=fft_len,
                                           lr0=0.13)
        print(f"[Adam final] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final evaluation using the provided security‑checked function
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
