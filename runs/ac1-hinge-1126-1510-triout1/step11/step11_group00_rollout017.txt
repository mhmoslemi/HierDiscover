def lp_step(a):
    n = len(a)
    M = np.convolve(a, a).max()
    # Build matrix A
    offsets = np.arange(n)
    diag_data = [np.full(n, a[i]) for i in range(n)]
    A = diags(diag_data, offsets, shape=(2*n-1, n), format='csr')
    # Upper bounds: M for each row
    ub = np.full(2*n-1, M)
    # bounds for g variables: (0, None)
    bounds = [(0, None)] * n
    # objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success: return a
    g = res.x
    # Scale g to sum=1
    g_norm = g / g.sum()
    return g_norm

def lp_refinement(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  max_iter: int = 10,
                  rng: np.random.Generator = None):
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute max conv
        conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
        M = conv.max()
        # Build A
        offsets = np.arange(n)
        data = np.tile(a, (n, 1)).T
        A = diags(data, offsets, shape=(2*n-1, n), format='csr')
        ub = np.full(2*n-1, M)
        c = -np.ones(n)
        res = linprog(c, A_ub=A, b_ub=ub, bounds=[(0, None)] * n, method='highs')
        if not res.success:

        g = res.x
        # Normalize g to sum 1

        if sum_g <= 0:

        t = 0.15 * rng.random()  # random
        a = (1 - t) * a + t * g_norm
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[LP] new best fast_score={best_score:.8f}")
    return best_a, best_score

"""
Propose a low‑score step function.

Algorithm Overview
------------------
1. **Initial pool** – generate a diverse set of probability vectors
   (spectral, Hann, Blackman, Gaussian, random Dirichlet‑like …) and
   keep the best one according to the exact objective
   `fast_score = 2·n·max(conv)` (the sum is fixed to 1).

2. **Phase 1 – L‑BFGS refinements** – repeatedly initialise a start
   vector by blending the current best with a fresh random vector and
   run a short L‑BFGS optimiser on the *smoothed* max‑convolution
   (`log‑sum‑exp`).  A long decreasing λ‑schedule is used (down to 1e‑6)
   and a modest number of L‑BFGS iterations per λ.

3. **Phase 2 – Adam on the exact max‑convolution** – optimise the true
   objective while staying on the simplex via a soft‑max
   parametrisation.  The update uses Adam with a 1/√t learning‑rate
   schedule and occasional jitter to escape plateaux.

4. **Phase 3 – Linear‑programming refinement** – for the current
   vector `a` compute the maximal convolution value `M`.  Solve the
   LP

        maximise   Σ g_j
        subject to  (a * g)_k ≤ M   for all k
                    g_j ≥ 0

   (the convolution constraints are linear in `g`).  The solution `g`
   is normalised to a probability vector and blended with `a`
   (small step size).  A few LP iterations dramatically reduce the
   peak of the self‑convolution.

5. **Phase 4 – Greedy hill‑climb** – a final local‑search that moves
   a small fraction of mass between two random positions only if it
   strictly improves the score.

The whole procedure respects the 1000 s time budget and returns the
best sequence found (the official `evaluate_sequence` rescales the
vector internally, so the returned list does not need to be normalised).
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Stable soft‑max ℝⁿ → probability simplex."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σ x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast exact score (sequence is on the simplex → sum = 1)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(convolution) for a probability vector."""
    a = np.asarray(a, dtype=np.float64)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) and its gradient
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len_conv: int, fft_len_g: int,
                       lam_factor: float = 0.30):
    """Return (smoothed objective, gradient) w.r.t. logits y."""
    a = softmax(y)

    # full self‑convolution (length 2·n‑1)
    fa = np.fft.rfft(a, fft_len_conv)
    conv_full = np.fft.irfft(fa * fa, fft_len_conv)[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = (rev_a * w)
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, fft_len_g)
    fw = np.fft.rfft(w, fft_len_g)
    conv_rev = np.fft.irfft(fa_rev * fw, fft_len_g)

    # extract gradient wrt a (size n)
    g_vec = conv_rev[(n - 1):(n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Initialisation helpers (produce probability vectors)
# ----------------------------------------------------------------------


def spectral_init(n: int) -> np.ndarray:
    """Spectral guess – inverse‑FFT of sqrt‑spectrum of a uniform window."""
    L = next_pow2(2 * n - 1)
    target_len = 2 * n - 1
    uniform = np.full(target_len, 1.0 / target_len)
    u_padded = np.zeros(L)
    u_padded[:target_len] = uniform
    S = np.fft.fft(u_padded)
    amp = np.abs(S)
    phase = np.angle(S)
    sqrt_amp = np.sqrt(np.maximum(amp, 0.0))
    sqrt_phase = phase / 2.0
    A = sqrt_amp * np.exp(1j * sqrt_phase)
    a_complex = np.fft.ifft(A)
    a = np.real(a_complex[:n])
    a[a < 0] = 0.0
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Diversified random draws on the simplex."""
    mode = rng.choice(['uniform', 'random', 'exp', 'power', 'edge', 'bimodal'])
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    else:   # bimodal
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    a /= a.sum()
    return a


def triangular_init(n: int) -> np.ndarray:
    """Linear rise then fall (triangular shape)."""
    half = n // 2
    inc = np.linspace(0, 1, half, endpoint=False)
    dec = np.linspace(1, 0, n - half)
    a = np.concatenate([inc, dec])
    a /= a.sum()
    return a


def hann_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def blackman_init(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1)) +
         0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def kaiser_init(n: int, beta: float = 8.0) -> np.ndarray:
    a = np.kaiser(n, beta)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_init(n: int, sigma_factor: float = 0.25) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a

# Optimisation primitives

def lbfgs_continuation(a_start: np.ndarray,
                       lam_schedule,
                       maxiter_per_lam,
                       deadline,
                       fft_len_conv,
                       fft_len_g,
                       n):
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage from the
    previous result. Returns the best a found and its fast_score.
    """
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len_conv, fft_len_g, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue
        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  rng: np.random.Generator):
    """
    Adam optimiser on the exact max‑convolution objective.
    Operates on logits y so that a = softmax(y) stays on the simplex.
    """
    y = np.log(a_start + 1e-12)
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    best_a = softmax(y)
    best_score = fast_score(best_a, fft_len, n)
    t = 0

    while time.time() < deadline:
        t += 1
        a = softmax(y)

        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        mask = (conv >= max_c - 1e-14).astype(np.float64)

        a_rev = a[::-1]
        fa_rev = np.fft.rfft(a_rev, fft_len)
        fm = np.fft.rfft(mask, fft_len)
        rev_grad = np.fft.irfft(fa_rev * fm, fft_len)
        g_a = 2.0 * rev_grad[(n - 1):(n - 1 + n)]

        grad_a = 2.0 * n * g_a
        mean_g = np.dot(a, grad_a)
        grad_y = a * (grad_a - mean_g)

        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y * grad_y)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)

        lr = 0.2 / np.sqrt(t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            y += rng.normal(scale=1e-4, size=y.shape)

        a_curr = softmax(y)
        sc = fast_score(a_curr, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()
    return best_a, best_score


def hill_climb(a: np.ndarray,
               cur_score: float,
               fft_len: int,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               max_iter: int = 80000,
               delta_frac: float = 0.06):
    """
    Strictly improving greedy hill‑climb.
    """
    best = a.copy()
    best_score = cur_score
    it = 0
    while time.time() < deadline and it < max_iter:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            it += 1
            continue
        delta = best[i] * rng.random() * delta_frac
        if delta <= 0:
            it += 1
            continue
        best[i] -= delta
        best[j] += delta
        sc = fast_score(best, fft_len, n)
        if sc < best_score:
            best_score = sc
        else:
            best[i] += delta
            best[j] -= delta
        it += 1
    return best, best_score


def lp_refinement(a_start: np.ndarray,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  max_iter: int = 8,
                  rng: np.random.Generator = None):
    """
    Linear‑programming refinement.
    For the current a compute M = max(a*a).  Solve
        maximise Σ g_j
        subject to (a * g)_k ≤ M   (∀k)
                    g_j ≥ 0
    The solution is normalised to a probability vector and blended with a.
    A few such LP iterations usually lower the peak of the self‑convolution.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len, n)

    for it in range(max_iter):
        if time.time() >= deadline:
            break

        # current maximal convolution value
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()

        # Build the constraint matrix A where A[k, j] = a_{k-j} (if in range)
        offsets = np.arange(n)
        # each diagonal i contains the constant a[i] (repeated n times)
        # build a dense (n,n) array of the diagonals – still modest memory (~130 MiB)
        data = np.tile(a, (n, 1)).T  # shape (n, n); row i = a_i repeated
        A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

        # Solve linear program: maximise Σ g_j  ⇔ minimise −Σ g_j
        c = -np.ones(n)
        ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g_norm = g / sum_g

        # Blend a little towards the LP solution
        t = 0.12 * (rng.random() if rng is not None else 0.12)
        a = (1 - t) * a + t * g_norm
        a /= a.sum()

        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[LP] new best fast_score={best_score:.8f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    Returns the best list of floats before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (trade‑off between granularity and speed)
    # ------------------------------------------------------------------
    n = 4096                       # power‑of‑two size – good FFT behaviour
    fft_len = next_pow2(2 * n - 1)   # length for zero‑padded FFTs

    # ------------------------------------------------------------------
    # Build initial candidate pool
    # ------------------------------------------------------------------
    candidates = []

    # Interpolate a previous solution if available
    if 'height_sequence_1' in globals():
        prev = np.array(globals()['height_sequence_1'], dtype=float)
        if prev.ndim == 1 and prev.size > 0:
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, prev)
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full(n, 1.0 / n)
            candidates.append(a0)

    # Deterministic families
    candidates.append(spectral_init(n))
    candidates.append(triangular_init(n))
    candidates.append(hann_init(n))
    candidates.append(blackman_init(n))
    candidates.append(gaussian_init(n))

    # Randomised families
    for _ in range(4):
        candidates.append(random_initial(n, rng))
    for _ in range(2):
        beta = rng.uniform(4.0, 12.0)
        candidates.append(kaiser_init(n, beta))

    # Select the best of the pool (fast_score is exact for sum=1)
    best_a = candidates[0].copy()
    best_score = fast_score(best_a, fft_len, n)
    for cand in candidates[1:]:
        sc = fast_score(cand, fft_len, n)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
    print(f"[init] n={n}, best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Time budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline_total = start_time + budget_s - 0.5   # safety margin

    phase1_deadline = start_time + 0.35 * budget_s
    phase2_deadline = phase1_deadline + 0.35 * budget_s
    phase3_deadline = phase2_deadline + 0.20 * budget_s
    # the remaining time is for the final hill‑climb

    # ------------------------------------------------------------------
    # Phase 1 – short L‑BFGS refinements
    # ------------------------------------------------------------------
    lam_schedule = [0.30, 0.20, 0.10, 0.05, 0.02, 0.01,
                    0.005, 0.002, 0.001, 5e-4, 2e-4,
                    1e-4, 5e-5, 2e-5, 1e-5, 5e-6]
    maxiter_per_lam = 800

    while time.time() < phase1_deadline:
        rand_vec = random_initial(n, rng)
        alpha = rng.random() * 0.3
        a_start = (1 - alpha) * best_a + alpha * rand_vec
        a_start = a_start / a_start.sum()
        a_opt, sc_opt = lbfgs_continuation(a_start,
                                           lam_schedule,
                                           maxiter_per_lam,
                                           phase1_deadline,
                                           fft_len, fft_len, n)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation on the exact max‑convolution
    # ------------------------------------------------------------------
    if time.time() < phase2_deadline:
        print("[Adam] start exact optimisation")
        best_a, best_score = adam_optimize(best_a,
                                           phase2_deadline,
                                           n,
                                           fft_len,
                                           rng)
        print(f"[Adam] best fast_score={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 3 – Linear‑programming refinement
    # ------------------------------------------------------------------
    if time.time() < phase3_deadline:
        print("[LP] start refinement")
        best_a, best_score = lp_refinement(best_a,
                                           phase3_deadline,
                                           n,
                                           fft_len,
                                           max_iter=8,
                                           rng=rng)
        print(f"[LP] best fast_score after refinement={best_score:.8f}")

    # ------------------------------------------------------------------
    # Phase 4 – final greedy hill‑climb (polish)
    # ------------------------------------------------------------------
    if time.time() < deadline_total:
        remaining = deadline_total - time.time()
        hill_deadline = time.time() + min(remaining, 30.0)
        best_a, best_score = hill_climb(best_a,
                                        best_score,
                                        fft_len,
                                        n,
                                        hill_deadline,
                                        rng,
                                        max_iter=50000,
                                        delta_frac=0.04)
        print(f"[final hill‑climb] best fast_score={best_score:.8f}")

    # --------------------------------------------------------------
    # Return the result (official evaluation rescales internally)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # official score
    print(f"[final] evaluated score = {final_eval:.8f}")
    return final_seq
