def multiplicative_weights_optimize(a0, deadline, ...):
    a = a0.copy()
    best_score = fast_score(a)
    best_a = a.copy()
    t = 0
    while time.time() < deadline:
        t += 1
        # compute conv
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[:2*n -1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()
        # subgradient
        a_rev = a[::-1]
        # compute gradient
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2 * conv_g[n-1:2*n-1]  # size n
        # step size
        alpha = 0.5 / np.sqrt(t) # maybe.
        multiplicative = np.exp(-alpha * g)
        a = a * multiplicative
        a = a / a.sum()
        # evaluate
        score = fast_score(a, fft_len_conv, n)
        if score < best_score:
            best_score, best_a = score, a.copy()
    return best_a, best_score

"""
Propose a low‑score non‑negative step sequence.

The algorithm mixes several complementary heuristics:
  * systematic and random exploration of many classic window families
    (exponential, power‑law, logistic, Gaussian, cosine, Hann, Hamming,
    Blackman, triangular, bimodal, sparse, etc.);
  * aggressive L‑BFGS on a smooth log‑sum‑exp approximation of the
    max‑convolution, with a long λ‑schedule and many jittered restarts;
  * projected Adam on the true max‑convolution (sub‑gradient);
  * simulated annealing with several move types;
  * a final multiplicative‑weights sub‑gradient refinement that
    directly reduces the exact max‑convolution.

All phases respect a global deadline (default 1000 s).  The best
candidate encountered is finally evaluated with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)−1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for `evaluate_sequence` when sum(a)==1."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (softmax → simplex)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)  # logits – stays on the simplex via softmax

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution (sub‑gradient of max).
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.35  # tuned a bit lower for stability

    while time.time() < deadline:
        it += 1
        # true convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()  # convex combination of active constraints

        # sub‑gradient of the max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (2 * n - 1)]

        # Adam update on the raw vector (project later)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with square‑root cooling.
    Moves:
      – transfer a random fraction of mass from i → j,
      – swap two positions,
      – redistribute a small fraction among three random positions.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # temperature initialization
    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # square‑root cooling
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            # simple transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            # triple redistribution (keep sum)
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score


def multiplicative_weights_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (multiplicative‑weights) refinement on the exact
    max‑convolution.  Guarantees non‑negativity and constant sum.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    it = 0
    while time.time() < deadline:
        it += 1
        # convolution and max
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()

        # sub‑gradient of the max (same as in Adam)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (2 * n - 1)]

        # step size decreasing like 1/√t
        alpha = 0.45 / np.sqrt(it)

        multiplicative = np.exp(-alpha * g)
        a = a * multiplicative
        a = a / a.sum()

        # evaluate occasionally (every 50 iterations)
        if it % 50 == 0:
            score = fast_score(a, fft_len_conv, n)
            if score < best_score:
                best_score, best_a = score, a.copy()
                print(f"[mw] new best fast_score={best_score:.6f} (it={it})")

    return best_a, best_score

def generate_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.0002, 0.0015)
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, rng: np.random.Generator) -> np.ndarray:
    p = rng.uniform(-5.0, -0.5)
    inc = rng.choice([True, False])
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, rng: np.random.Generator) -> np.ndarray:
    mu = rng.uniform(0.2 * n, 0.8 * n)
    sigma = rng.uniform(200.0, 1200.0)
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = rng.uniform(200.0, 2000.0)
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    width = rng.uniform(200.0, n / 2.0)
    i = np.arange(n, dtype=float)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    mu1 = rng.uniform(0.0, n / 2.0)
    mu2 = rng.uniform(n / 2.0, n - 1)
    sigma1 = rng.uniform(100.0, 500.0)
    sigma2 = rng.uniform(100.0, 500.0)
    i = np.arange(n, dtype=float)
    a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
    a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    a = a1 + a2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_binary_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    prob = rng.uniform(0.02, 0.25)
    mask = rng.random(n) < prob
    if not np.any(mask):
        mask[0] = True
    a = mask.astype(float)
    a /= a.sum()
    return a


def generate_double_exp(n: int, rng: np.random.Generator) -> np.ndarray:
    rate = rng.uniform(0.0002, 0.0012)
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_window(n: int, kind: str, rng: np.random.Generator) -> np.ndarray:
    """
    Generate Hann, Hamming or Blackman windows, optionally raised to a random
    exponent to increase flexibility.
    """
    i = np.arange(n, dtype=float)
    if kind == "hann":
        w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    elif kind == "hamming":
        w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    elif kind == "blackman":
        w = (
            0.42
            - 0.5 * np.cos(2 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4 * np.pi * i / (n - 1))
        )
    else:
        w = np.ones_like(i)

    # random power to shape the window (0.5 .. 3.0)
    power = rng.uniform(0.5, 3.0)
    w = np.maximum(w, 0.0) ** power
    w /= w.sum()
    return w


def generate_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.minimum(i + 1, n - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_random(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a named parametric family."""
    if family == "exp":
        return generate_exp(n, rng)
    if family == "power":
        return generate_power(n, rng)
    if family == "logistic":
        return generate_logistic(n, rng)
    if family == "gauss":
        return generate_gauss(n, rng)
    if family == "cos":
        return generate_cosine(n, rng)
    if family == "bimodal":
        return generate_bimodal(n, rng)
    if family == "binary_sparse":
        return generate_binary_sparse(n, rng)
    if family == "double_exp":
        return generate_double_exp(n, rng)
    if family == "hann":
        return generate_window(n, "hann", rng)
    if family == "hamming":
        return generate_window(n, "hamming", rng)
    if family == "blackman":
        return generate_window(n, "blackman", rng)
    if family == "triangular":
        return generate_triangular(n)
    if family == "random":
        return generate_random(n, rng)
    # default fallback
    return generate_random(n, rng)


# ----------------------------------------------------------------------
# Systematic grid search over a few well‑known families
# ----------------------------------------------------------------------


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Explore systematic grids for several classic families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential rates
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        i = np.arange(n, dtype=float)
        a = np.exp(-rate * i)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power‑law exponents
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        i = np.arange(n, dtype=float) + 1.0
        a = i ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic parameters (coarse)
    mus = np.linspace(0.2 * n, 0.8 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            i = np.arange(n, dtype=float)
            a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
            a = np.maximum(a, 0.0)
            a /= a.sum()
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # window families with a few exponent choices
    for kind in ("hann", "hamming", "blackman"):
        for power in (0.7, 1.0, 1.5, 2.0):
            if time.time() >= deadline:
                return best_a, best_sc
            i = np.arange(n, dtype=float)
            if kind == "hann":
                w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
            elif kind == "hamming":
                w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
            else:  # blackman
                w = (
                    0.42
                    - 0.5 * np.cos(2 * np.pi * i / (n - 1))
                    + 0.08 * np.cos(4 * np.pi * i / (n - 1))
                )
            w = np.maximum(w, 0.0) ** power
            w /= w.sum()
            sc = fast_score(w, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, w.copy()
                print(f"[grid] {kind} power={power:.2f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Quick random exploration over many families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10  # 10 % of total budget

    families = [
        "exp",
        "power",
        "gauss",
        "cos",
        "bimodal",
        "logistic",
        "binary_sparse",
        "double_exp",
        "hann",
        "hamming",
        "blackman",
        "triangular",
        "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # Problem dimension (can be changed, but 4000 works well)
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # ------------------------------------------------------------------
    # Initial candidate (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Warm‑start from a historic construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            # interpolate to the current length
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Global timing layout
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # Phase 0 – systematic grid + cheap random sweep
    phase0_end = start_time + 0.20 * budget_s
    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, phase0_end)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many jittered restarts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.7, 0.5, 0.35, 0.25, 0.18, 0.12, 0.08, 0.05,
        0.03, 0.018, 0.01, 0.006, 0.003, 0.0015,
        0.0008, 0.0005, 0.0002, 0.0001, 5e-05,
        2e-05, 1e-05, 5e-06, 2e-06, 5e-07, 2e-07,
    ]
    maxiter_per_lam = 2500

    while time.time() < phase1_end:
        # 30 % chance to jitter the current best, otherwise start from a fresh exp
        if rng.random() < 0.30:
            jitter = rng.normal(1.0, 0.03, size=n)
            a_start = best_a * jitter
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()
        else:
            a_start = generate_exp(n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected Adam on the true max‑convolution
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.70 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam_optimize(
            best_a, phase2_end, fft_len_conv, fft_len_g, n
        )
        print(f"[phase2] Adam sub‑gradient best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing (rich move set)
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.90 * budget_s
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final multiplicative‑weights refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = multiplicative_weights_optimize(
            best_a,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[final‑mw] best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Official evaluation (scale does not matter, keep sum≈1)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
