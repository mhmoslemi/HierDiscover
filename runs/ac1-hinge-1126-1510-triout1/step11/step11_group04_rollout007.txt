    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
    max_val = conv.max()

    # compute subgradient
    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))  # length n+ len-...

    # exponentiated update
    exp_term = np.exp(np.clip(-eta * g, -50, 50))
    a = a * exp_term
    a = a / a.sum()
    # symmetrize
    a = (a + a[::-1]) / 2
    a = a / a.sum()

"""
Step‑function optimiser – 1000 s budget (enhanced)

* Uses a larger problem size (n≈8000) to gain extra degrees of freedom.
* Builds a very diverse pool of start vectors (random families, classic windows,
  parameterised Gaussians, tophats, DPSS etc.).
* Phase‑1: a cascade of L‑BFGS on the smooth log‑sum‑exp surrogate with a
  long λ‑schedule and many iterations per λ.
* Phase‑2: mirror‑SGD (exponentiated‑gradient) on the true L∞ objective,
  with a symmetric projection after each update to keep the search on the
  “mirror‑symmetric” sub‑space (the optimum is known to be almost symmetric).
* Phase‑3: a very low‑λ L‑BFGS polish followed by a short simulated‑annealing
  hill‑climb.
* All phases respect the 1000 s deadline; the best sequence encountered is
  stored in the global variable ``height_sequence_1`` and returned.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int = None) -> float:
    """
    Fast proxy for the official score when ``sum(a) == 1``:
        2·n·max(conv(a,a))
    where the convolution is evaluated by FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = next_pow2(2 * n - 1)
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:   # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def gaussian_initial(n: int, sigma_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Centered Gaussian window (σ = sigma_frac·n)."""
    sigma = sigma_frac * n
    x = np.arange(n)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def tophat_initial(n: int, frac: float, rng: np.random.Generator) -> np.ndarray:
    """Uniform block (fraction of the total length) placed at a random offset."""
    m = max(1, int(n * frac))
    a = np.zeros(n, dtype=float)
    start = rng.integers(0, n - m + 1)
    a[start:start + m] = 1.0
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry a[i]=a[n‑1‑i] while staying on the simplex."""
    a_sym = (a + a[::-1]) / 2.0
    a_sym /= a_sym.sum()
    return a_sym


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns smoothed max‑convolution value and its gradient w.r.t. the
    log‑variables y (so that a = softmax(y)).
    """
    # --- softmax ---
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # --- convolution via FFT ---
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                      # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the exact L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0,
                          max_iters: int = 10_000_000):
    """
    Mirror‑SGD (exponentiated‑gradient) on the true max‑convolution.
    Each update is symmetrised to keep the search in the (almost) optimal
    symmetric subspace.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the hard max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped to avoid overflow)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
            a_sum = a.sum()
        a = a / a_sum

        # symmetric projection (keeps the iterates mirror‑symmetric)
        a = symmetrize(a)

        # occasional cheap score update
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers (Metropolis rule) with a linear cooling schedule.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a finer description of the optimal shape.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a rich pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Re‑use the previous best (if any) and interpolate to the new length
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Random families (12 variants)
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # 3) Classic windows (if SciPy is available)
    try:
        from scipy.signal import windows as win

        # Chebyshev windows
        for att in (20, 30, 40, 50, 60):
            w = win.chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        # Kaiser windows
        for beta in (2, 4, 6, 8, 10, 12, 14, 16):
            w = win.kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        # Tukey windows
        for alpha in (0.2, 0.4, 0.6, 0.8):
            w = win.tukey(n, alpha)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        # Standard cosine‑type windows
        for name in ('hann', 'hamming', 'blackman', 'triang'):
            w = getattr(win, name)(n)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

        # DPSS (Slepian) window – if present
        if hasattr(win, 'dpss'):
            # reasonable time‑bandwidth product
            w = win.dpss(n, NW=4, Kmax=1)[0]
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)

    except Exception:
        pass

    # 4) Parameterised Gaussians
    for sigma_frac in (0.05, 0.08, 0.12, 0.15, 0.2, 0.25):
        pool.append(gaussian_initial(n, sigma_frac, rng))

    # 5) Tophat blocks with different fill fractions
    for frac in (0.05, 0.10, 0.20, 0.30):
        pool.append(tophat_initial(n, frac, rng))

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    phase1_end = start_time + 0.20 * budget_s       # L‑BFGS coarse
    phase2_end = start_time + 0.80 * budget_s       # long EG phase
    # remaining time → final polish + anneal

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = np.geomspace(0.8, 1e-5, num=24)   # smooth to very low λ
    maxiter_per_lam = 1500

    best_a = None
    best_score = float('inf')
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        # apply a little random jitter to diversify the basin entry
        if rng.random() < 0.35:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.008), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS coarse] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        # make it symmetric before the EG stage
        best_a = symmetrize(best_a)
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=30.0,
            max_iters=25_000_000   # very generous upper bound; loop stops by time
        )
        print(f"[exp‑grad] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polish (very low λ L‑BFGS) + short anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # final symmetrisation ensures the start is perfectly mirror‑symmetric
        best_a = symmetrize(best_a)

        fine_lam_schedule = [5e-6, 2.5e-6, 1.2e-6, 6e-7, 3e-7, 1e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[final‑lbfgs] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a = symmetrize(best_a)          # keep the symmetry invariant
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
