"""
Propose a low‑score non‑negative step sequence.

The algorithm combines several complementary phases:

0️⃣  Initialise with the uniform distribution (and an optional historic best).
1️⃣  Systematic grid search over a few analytic families (exponential,
    power‑law, logistic) followed by a cheap random sweep over many
    parametric families.
2️⃣  Aggressive multi‑restart L‑BFGS optimisation on a *smooth* surrogate
    of the max‑convolution (log‑sum‑exp).  The gradient is now exact –
    it includes the missing factor 2 and the outer factor 2·n from the
    official objective.  A fine λ‑schedule with longer inner iterations
    is used.
3️⃣  Projected Adam on the true non‑smooth max‑convolution (sub‑gradient).
4️⃣  Simulated annealing with richer moves (single transfer, swap,
    triple redistribution) for global exploration.
5️⃣  Final ultra‑fine L‑BFGS polishing (λ down to 1e‑9).

All phases respect a global 1000 s deadline; the best vector seen at any
moment is returned and finally evaluated with ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate of the official evaluation (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and exact gradient
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed version of 2·n·max(a*a) where a = softmax(y).

    Returns (objective, gradient w.r.t. y).  The gradient now contains the
    correct factor 2 from d(conv)/da and the outer factor 2·n from the
    objective.
    """
    a = softmax(y)

    # ----- convolution a*a (full) -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smooth approximation of the max -----
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-14          # avoid zero
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max                  # objective

    # ----- gradient of smooth_max w.r.t. the convolution entries -----
    w = exp_shift / Z                         # d smooth_max / d conv_i

    # ----- back‑propagation through the bilinear convolution -----
    # derivative conv_i w.r.t. a_j is 2·a_{i‑j} (zero outside the range)
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    # extract the segment that corresponds to Σ_i w_{i+j}·a_i
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # factor 2 from d(conv)/da and factor 2·n from the outer objective
    grad_a = 4.0 * n * g_vec

    # ----- chain rule through the soft‑max -----
    mean_g = np.dot(a, grad_a)
    grad_y = a * (grad_a - mean_g)

    return float(f), grad_y


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in ``lam_schedule``, starting each stage from
    the result of the previous stage.  Returns the best vector seen and
    its fast_score.
    """
    y = np.log(a_start + 1e-12)   # logits – stay on the simplex via softmax

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12,
                         "disp": False},
            )
        except Exception as e:   # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_adam_optimize(

    Projected Adam on the true (non‑smooth) max‑convolution.
    This is a robust sub‑gradient method for the original objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)   # first moment
    v = np.zeros_like(a)   # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.4          # empirically good for this problem

    while time.time() < deadline:
        it += 1
        # true convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                        # any convex comb. is a valid sub‑grad

        # sub‑gradient of max (2·a_rev * mask)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update on the raw vector (project later)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with square‑root cooling.
    Move set:
      – transfer a random fraction between two indices,
      – swap two indices,
      – redistribute mass among three indices (Dirichlet step).
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            # simple transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            # triple redistribution (keeps total mass)
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_gauss(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_cos(n: int, width: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu1 = rng.uniform(0.0, n / 2.0)
    mu2 = rng.uniform(n / 2.0, n - 1)
    sigma1 = rng.uniform(100.0, 500.0)
    sigma2 = rng.uniform(100.0, 500.0)
    a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
    a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    a = a1 + a2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_binary_sparse(n: int, rng: np.random.Generator) -> np.ndarray:
    prob = rng.uniform(0.02, 0.25)
    mask = rng.random(n) < prob
    if not np.any(mask):
        mask[0] = True
    a = mask.astype(float)
    a /= a.sum()
    return a


def generate_double_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_random(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.random(n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from one of the predefined families."""
    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = generate_power(n, p, inc)
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * np.arange(n, dtype=float))
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        a = generate_gauss(n, sigma)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = generate_cos(n, width)
    elif family == "bimodal":
        a = generate_bimodal(n, rng)
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = generate_logistic(n, mu, sigma)
    elif family == "binary_sparse":
        a = generate_binary_sparse(n, rng)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = generate_double_exp(n, rate)
    else:   # plain random
        a = generate_random(n, rng)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Systematic sweep over exponential, power‑law and logistic families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential grid
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power‑law grid
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic grid
    mus = np.linspace(0.2 * n, 0.8 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    return best_a, best_sc


def cheap_random_sweep(
    n: int,
    fft_len: int,
    budget_s: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Quick random exploration over many parametric families."""
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len, n)

    start = time.time()
    deadline = start + budget_s * 0.10   # 10 % of total budget

    families = [
        "power", "exp", "gauss", "cos",
        "bimodal", "logistic", "binary_sparse",
        "double_exp", "random",
    ]

    while time.time() < deadline:
        fam = rng.choice(families)
        a = generate_family(fam, n, rng)
        sc = fast_score(a, fft_len, n)
        if sc < best_score:
            best_score, best_a = sc, a.copy()
            print(f"[sweep] new best fast_score={best_score:.6f} (family={fam})")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem parameters
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # for a*a
    fft_len_g = next_pow2(3 * n - 2)         # for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate (uniform)
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_score = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Re‑use a previously discovered construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        hs = np.array(globals()["height_sequence_1"], dtype=float)
        if hs.ndim == 1 and hs.size > 0:
            x_old = np.linspace(0, 1, hs.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, hs)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            score1 = fast_score(a1, fft_len_conv, n)
            if score1 < best_score:
                best_a, best_score = a1, score1
                print(f"[init] previous best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 0 – systematic grid search + cheap random sweep
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    a_grid, sc_grid = parametric_grid_search(n, fft_len_conv, rng, deadline)
    if sc_grid < best_score:
        best_a, best_score = a_grid, sc_grid
        print(f"[grid] best fast_score={best_score:.6f}")

    a_rand, sc_rand = cheap_random_sweep(n, fft_len_conv, budget_s, rng)
    if sc_rand < best_score:
        best_a, best_score = a_rand, sc_rand
        print(f"[rand] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Time allocation for the remaining phases
    # --------------------------------------------------------------
    phase1_end = start_time + 0.40 * budget_s   # L‑BFGS restarts
    phase2_end = start_time + 0.70 * budget_s   # Adam on true max
    phase3_end = start_time + 0.95 * budget_s   # Simulated annealing
    # Remaining time → final polishing L‑BFGS with ultra‑tiny λ

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS with many random restarts
    # --------------------------------------------------------------
    lam_schedule = [
        0.6, 0.4, 0.25, 0.15, 0.09, 0.05, 0.03, 0.015,
        0.008, 0.004, 0.002, 0.001, 5e-04, 2e-04,
        1e-04, 5e-05, 2e-05, 1e-05, 5e-06, 2e-06,
        1e-06, 5e-07, 2e-07, 1e-07,
    ]
    maxiter_per_lam = 3000   # more inner work per λ

    while time.time() < phase1_end:
        # jitter the current best or draw a fresh random family
        if rng.random() < 0.35:
            a_start = best_a * rng.normal(1.0, 0.025, size=n)
            a_start = np.maximum(a_start, 0.0)
            s = a_start.sum()
            if s == 0.0:
                a_start = np.full(n, 1.0 / n)
            else:
                a_start /= s
        else:
            fam = rng.choice(
                ["exp", "power", "gauss", "cos",
                 "bimodal", "logistic", "binary_sparse",
                 "double_exp", "random"]
            )
            a_start = generate_family(fam, n, rng)

        a_opt, sc_opt = lbfgs_continuation(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_score:
            best_score, best_a = sc_opt, a_opt.copy()
            print(f"[phase1] improved fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Projected Adam on the true max‑convolution
    # --------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = subgradient_adam_optimize(
            best_a,
            phase2_end,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] Adam best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing (rich move set)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing – ultra‑fine L‑BFGS run with a tiny λ schedule
    # --------------------------------------------------------------
    if time.time() < deadline:
        tiny_sched = [5e-08, 2e-08, 1e-08, 5e-09, 2e-09, 1e-09]
        a_final, sc_final = lbfgs_continuation(
            best_a,
            tiny_sched,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_score:
            best_score, best_a = sc_final, a_final.copy()
            print(f"[final‑lbfgs] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
