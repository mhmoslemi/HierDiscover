def simulated_annealing(a_start, start_score, deadline, rng, init_temp=0.01, final_temp=1e-5):
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    total = deadline - time.time()
    steps = 0
    while time.time() < deadline:
        remaining = deadline - time.time()
        # compute temperature as linear interpolation from init_temp to final_temp
        frac = max(0.0, remaining / total)
        T = final_temp + (init_temp - final_temp) * frac

        # choose source and destination
        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # choose delta proportion
        step_factor = 0.1 * (rng.random())  # random between 0 and 0.1
        delta = a[src] * step_factor
        if delta == 0:
            continue
        # store old copies
        a_old = a.copy()
        conv_old = conv.copy()

        # update conv approx
        conv[src: src + n] -= 2.0 * delta * a_old
        conv[dst: dst + n] += 2.0 * delta * a_old

        a[src] -= delta
        a[dst] += delta

        # recompute exact conv periodically
        if steps % 1000 == 0:
            conv = np.convolve(a, a)

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        # accept?
        if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / max(T, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # reject
            a = a_old
            conv = conv_old

        steps += 1

    return best_a, best_score

Propose a low‑evaluation step‑function sequence.

The search mixes several optimization ingredients that together can drive the
objective 2·n·max(conv(a,a)) (with sum(a)=1) below the previous best
≈1.5064 and approach the theoretical lower bound 1.5.

1. **Diverse random starts** – uniform, exponential, power‑law, edge,
   bimodal, sparse, Gaussian and beta‑shaped initial vectors.
2. **L‑BFGS on a log‑sum‑exp smoothed max‑convolution** – a cascade of
   decreasing smoothing parameters (λ) drives the solution toward a flat
   self‑convolution while keeping the simplex (∑a=1, a≥0) constraints via a
   soft‑max parametrisation.
3. **Adam refinement** – the same smoothed objective with an adaptive
   learning‑rate optimiser.
4. **Gradient‑guided hill‑climb** – moves a small fraction of mass from the
   index with the largest sub‑gradient (most responsible for the current peak)
   to the index with the smallest sub‑gradient; the linear‑approximation of the
   convolution is updated incrementally, with periodic exact recomputation.
5. **Simulated‑annealing fine‑tuning** – occasional uphill moves are accepted
   according to a temperature schedule, helping to escape shallow local minima.

All phases respect the 1000 s budget; the best sequence seen so far is finally
checked with the provided ``evaluate_sequence`` function.  The returned list
contains non‑negative numbers and is already normalised (∑a = 1), which is
invariant for the objective.

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector (∑a=1)."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "gaussian",
            "beta",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "gaussian":
        # centred Gaussian truncated to the interval
        sigma = rng.uniform(0.15, 0.35) * n
        xs = np.arange(n) - n / 2.0
        a = np.exp(-0.5 * (xs / sigma) ** 2)
    elif mode == "beta":
        # beta distribution on [0,1] stretched to n points
        a = rng.beta(0.5 + rng.random() * 2.0, 0.5 + rng.random() * 2.0, size=n)
    else:
        a = np.full(n, 1.0 / n)

    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp approx of max)
# ----------------------------------------------------------------------


def smooth_obj_and_grad(y: np.ndarray, n: int, fft_len: int, lam_factor: float = 0.30):
    """
    Return (objective, gradient) for the log‑sum‑exp smoothed max‑convolution.
    The optimisation variables are logits y; a = softmax(y) guarantees a≥0,
    sum(a)=1.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]  # gradient w.r.t. a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the smoothed objective
# ----------------------------------------------------------------------


def lbfgs_optimize(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    # initialise logits
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=smooth_obj_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective
# ----------------------------------------------------------------------


def adam_optimize(
    y_start: np.ndarray,
    lam_schedule,
    max_steps_per_lam: int,
    deadline: float,
    n: int,
    fft_len: int,
    lr0: float = 0.12,
):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = smooth_obj_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb
# ----------------------------------------------------------------------


def guided_hill_climb(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
    start_factor: float = 0.15,
    end_factor: float = 0.0005,
    recompute_every: int = 2000,
):
    """
    Move a fraction of mass from the index with the largest sub‑gradient to the
    index with the smallest sub‑gradient, accepting only non‑increasing moves.
    Step size shrinks linearly with remaining time; the exact convolution is
    recomputed every ``recompute_every`` accepted moves.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)  # exact linear convolution, length 2n‑1
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score

    total = deadline - time.time()
    accepted = 0
    steps = 0

    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        g_full = np.real(
            np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a_rev, fft_len))
        )
        g = 2.0 * g_full[n - 1 : n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 1e-15:
            # fall back to a small random move
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz, p=a[nz] / a[nz].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step factor
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        delta = min(delta, a[src])
        if delta < 1e-14:
            continue

        a_old = a.copy()
        conv_old = conv.copy()

        # linear update of convolution (ignore O(delta²) terms)
        conv[src : src + n] -= 2.0 * delta * a_old
        conv[dst : dst + n] += 2.0 * delta * a_old

        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score:
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a = a_old
            conv = conv_old

        steps += 1

        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        if steps % 10000 == 0:
            elapsed = time.time() - (deadline - total)
            print(
                f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} best_score={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
    init_temp: float = 0.02,
    final_temp: float = 1e-6,
):
    """
    Simple SA that moves a random fraction of mass from a source index to a
    destination index.  Moves that increase the score are sometimes accepted
    according to the current temperature, allowing the search to escape shallow
    minima.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    total = deadline - time.time()
    steps = 0

    while time.time() < deadline:
        # temperature – linear cooling
        remaining = max(0.0, deadline - time.time())
        frac = remaining / total
        T = final_temp + (init_temp - final_temp) * frac

        # pick source and destination
        src_candidates = np.where(a > 1e-15)[0]
        if src_candidates.size == 0:
            break
        src = int(rng.choice(src_candidates))
        dst = int(rng.integers(0, n))
        if src == dst:
            continue

        # random move size (up to a modest fraction of src mass)
        delta = a[src] * rng.random() * 0.3
        if delta < 1e-15:
            continue

        a_old = a.copy()
        conv_old = conv.copy()

        # incremental update (ignoring O(delta²) terms)
        conv[src : src + n] -= 2.0 * delta * a_old
        conv[dst : dst + n] += 2.0 * delta * a_old

        a[src] -= delta
        a[dst] += delta

        # occasional exact recompute to keep drift small
        if steps % 500 == 0:
            conv = np.convolve(a, a)

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a = a_old
            conv = conv_old

        steps += 1

        if steps % 20000 == 0:
            elapsed = time.time() - (deadline - total)
            print(
                f"[SA] elapsed={elapsed:.1f}s steps={steps} best_score={best_score:.6f}"
            )
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function.
    Returns a list of floats (the heights) before the deadline.
    """
    rng = np.random.default_rng(seed)

    # problem size – a few thousand entries give a good trade‑off
    n = 6000
    fft_len = next_pow2(2 * n - 1)

    # try to reuse a previously stored construction if present
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # safety margin

    # --------------------------------------------------------------
    # Phase 1 – random starts + L‑BFGS
    # --------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = np.geomspace(0.3, 1e-6, num=20)
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start,
            lam_schedule,
            maxiter_per_lam=400,
            deadline=phase1_end,
            fft_len=fft_len,
            n=n,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam refinement on the smoothed objective
    # --------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = np.geomspace(0.2, 1e-7, num=30)
        best_a, best_score = adam_optimize(
            y0,
            lam_sched,
            max_steps_per_lam=4000,
            deadline=deadline,
            n=n,
            fft_len=fft_len,
            lr0=0.15,
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – gradient‑guided hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = guided_hill_climb(
            best_a,
            best_score,
            deadline,
            rng,
            start_factor=0.15,
            end_factor=0.0005,
            recompute_every=3000,
        )
        print(f"[guided_hc] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – simulated annealing for final tweaks
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline, rng, init_temp=0.02, final_temp=1e-6
        )
        print(f"[SA] final fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation with the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
