"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
1.  Build a diverse set of initial candidates (uniform, arcsine,
    random, previous best, triangular).
2.  Keep the best candidate according to the exact score
    `2·n·max(conv)` (fast to compute via FFT).
3.  Run three optimisation phases within the time budget:
    * **Phase A – Smoothed‑max gradient descent**.
      A log‑sum‑exp surrogate of the max‑convolution is minimised.
      The gradient is obtained analytically in the frequency domain
      and the update is projected onto the probability simplex.
    * **Phase B – Exponentiated‑gradient descent** (as in the reference
      code) to accelerate convergence on the active set.
    * **Phase C – Projected sub‑gradient descent** for fine‑tuning.
4.  Finish with a short simulated‑annealing walk followed by a greedy
    hill‑climb that uses an exact incremental convolution update.
5.  Return the best list found; the harness will finally evaluate it
    with the supplied `evaluate_sequence` routine.

All phases respect the 1000 s total budget and keep the vector
non‑negative and normalised (∑aᵢ = 1), which is the natural scale for
the supplied evaluation function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n) + 1)
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def tent_initial(n: int) -> np.ndarray:
    """Triangular (tent) distribution – peak in the centre."""
    x = np.linspace(-1, 1, n)
    a = 1.0 - np.abs(x)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (new phase)
# ----------------------------------------------------------------------


def smoothed_max_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          alpha_start: float = 0.5,
                          alpha_end: float = 30.0,
                          step0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Minimise `log‑sum‑exp(alpha·conv)` with an increasing smoothing
    parameter `alpha`.  The true objective `max(conv)` is approached as
    `alpha → ∞`.  Gradient is computed via FFT, then projected onto the
    simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        # --- convolution and smoothing parameter ---------------------------------
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # linear fraction of elapsed time determines current alpha
        fraction = (time.time() - (deadline - (deadline - time.time()))) / (deadline - (deadline - (deadline - time.time())) + 1e-12)
        # Simpler: use iteration count scaling
        alpha = min(alpha_end, alpha_start + it * (alpha_end - alpha_start) / max(1, int(0.5 * (deadline - time.time()))))

        # --- soft‑max weights ----------------------------------------------------
        # stability: subtract max
        max_c = conv.max()
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            # degenerate – fall back to plain max mask
            mask = (conv >= max_c * (1.0 - 1e-12)).astype(np.float64)
            w_norm = mask / mask.sum()
        else:
            w_norm = w / w_sum

        # --- gradient of the smoothed objective ---------------------------------
        a_rev = a[::-1]
        fw = np.fft.fft(w_norm, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fw))
        grad = 2.0 * grad_full[n - 1: n - 1 + n]   # length‑n gradient

        # --- simple step‑size schedule and projection ---------------------------
        step = step0 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = simplex_projection(a)

        # --- keep the best -------------------------------------------------------
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent (reference implementation)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 30.0) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on max(conv)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (reference implementation)
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent with Euclidean projection onto the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        improved = False
        for _ in range(5):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated annealing (global search)
# ----------------------------------------------------------------------


def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Temperature‑controlled random walk on the simplex."""
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12

    while time.time() < deadline:
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        max_frac = 0.02 + 0.18 * fraction

        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb – moves a tiny amount of mass from a heavy index to a lighter one."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental convolution update (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Returns the best list of non‑negative floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few thousand entries give a good trade‑off
    # ------------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1. Build a set of diverse starting points
    # ------------------------------------------------------------------
    init_candidates = [
        np.full(n, 1.0 / n),               # uniform
        arcsine_initial(n),                # heavy edges
        tent_initial(n),                   # triangular peak
        random_initial(n, rng),            # random
    ]

    # include previous best if available
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            init_candidates.append(a_old)

    # keep the best primitive (cheap evaluation)
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # spare 0.05 s for cleanup

    # ------------------------------------------------------------------
    # 2. Phase A – smoothed‑max gradient descent (≈ 35 % of time)
    # ------------------------------------------------------------------
    phaseA_deadline = time.time() + 0.35 * (deadline - time.time())
    best_a, best_score = smoothed_max_optimize(
        best_a, phaseA_deadline, n, fft_len, rng,
        alpha_start=0.5, alpha_end=30.0, step0=0.25
    )
    print(f"[smooth‑max] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Phase B – exponentiated‑gradient descent (≈ 30 % of time)
    # ------------------------------------------------------------------
    phaseB_deadline = time.time() + 0.30 * (deadline - time.time())
    best_a, best_score = exp_gradient_optimize(
        best_a, phaseB_deadline, n, fft_len, rng, eta0=30.0
    )
    print(f"[exp‑grad] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Phase C – projected sub‑gradient descent (≈ 20 % of time)
    # ------------------------------------------------------------------
    phaseC_deadline = time.time() + 0.20 * (deadline - time.time())
    best_a, best_score = projected_subgradient_optimize(
        best_a, phaseC_deadline, n, fft_len, rng,
        eps_frac=0.001, step0=0.2
    )
    print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Global simulated annealing (spare a few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline - 5.0, rng
        )
        print(f"[SA] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6. Final greedy hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation (the harness will also call it)
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
