propose_candidate – 1000 s budget (enhanced)

The optimiser works in three stages:

1️⃣ **Diverse warm‑starts + aggressive L‑BFGS**.
   A cascade of L‑BFGS runs on a log‑sum‑exp smooth approximation of the
   L∞ objective drives many random starts into good basins.  The smoothing
   parameter λ is decreased continuously and the final low‑λ run is given

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when sum(a)=1:
        2·n·max(conv(a,a))
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "power",
            "edge",
            "bimodal",
            "sparse",
            "inv_sqrt",
            "sin",
            "hanning",
            "bartlett",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == "inv_sqrt":
        idx = np.arange(1, n + 1, dtype=float)
        a = 1.0 / np.sqrt(idx) + 1.0 / np.sqrt(n - idx + 1)
    elif mode == "sin":
        a = np.sin(np.pi * np.arange(n) / (n - 1)) + 0.01
    elif mode == "hanning":
        a = np.hanning(n) + 0.001
    elif mode == "bartlett":
        a = np.bartlett(n) + 0.001
    else:
        a = rng.random(n)
    # ensure non‑negative and normalise
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Inject isotropic gaussian noise and re‑project onto simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp surrogate (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns smooth approximation to ``2·n·max(conv)`` and its gradient
    w.r.t. the log‑variables y (a = softmax(y)).
    """
    # softmax → probability vector
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12  # relative smoothing

    shifted = conv_full - max_c          # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1 : n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            # informative output
            print(f"[L‑BFGS λ={lam:.1e}] new best fast_score={best_score:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑SGD) on the true L∞ objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
    max_iters: int = 20_000_000,
):
    """
    Mirror‑SGD (exponentiated gradient) on the exact max‑convolution.
    A tiny momentum term (previous gradient) is mixed in to accelerate.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    prev_g = np.zeros_like(a)

    while time.time() < deadline and it < max_iters:
        # convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

        # momentum (exponential moving average)
        if it == 0:
            m = g
        else:
            m = 0.9 * prev_g + 0.1 * g
        prev_g = m.copy()

        # step‑size schedule (slow decay)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (clipped)
        exp_term = np.exp(np.clip(-eta * m, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        # cheap score update every 64 steps
        if (it & 63) == 0:
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[exp‑grad] it={it} fast_score={best_score:.9f}")

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature cools linearly.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.20

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000                     # keep the FFT cheap enough for many iterations
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Re‑use the previous best (if any)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Random families (including several window shapes)
    for _ in range(16):
        pool.append(random_initial(n, rng))

    # 3) Classic windows (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser

        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # safety margin

    phase1_end = start_time + 0.12 * budget_s   # L‑BFGS coarse
    phase2_end = start_time + 0.78 * budget_s   # EG long
    # remaining time → final polish + anneal

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6,
        0.45,
        0.33,
        0.24,
        0.18,
        0.13,
        0.09,
        0.06,
        0.04,
        0.025,
        0.015,
        0.009,
        0.005,
        0.003,
        0.0015,
        0.0008,
        0.0004,
        0.0002,
        0.0001,
        0.00005,
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float("inf")
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        if rng.random() < 0.35:
            a_start = add_noise(
                a_start, sigma=rng.uniform(0.001, 0.01), rng=rng
            )
        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[Phase 1] new global best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=22.0,
            max_iters=15_000_000,
        )
        print(f"[Phase 2] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final low‑λ L‑BFGS polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-5, 2.5e-5, 1.2e-5, 6e-6, 3e-6, 1.5e-6, 8e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[Phase 3] polished fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – quick simulated annealing finish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a, best_score, deadline, n, rng
        )
        print(f"[Phase 4] final annealed fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final official evaluation
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()["height_sequence_1"] = final_seq
    return final_seq
