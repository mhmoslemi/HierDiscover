"""
Improved search for a low‑score non‑negative step sequence.

Main ideas
----------
* Use a soft‑max parametrisation (logits → probabilities) to keep the
  vector on the simplex (∑a = 1, a_i ≥ 0).
* Optimise a *smoothed* version of the true objective:
      f(a) = 2·n·max(a*a) .
  The max is approximated by a log‑sum‑exp (LSE) with a decreasing
  smoothing factor λ.  Additionally a variance term
      λ₂·Var(a*a)
  is added early on to encourage a flat convolution.
* A cheap multi‑restart L‑BFGS loop explores many random initial
  points in parallel with a modest iteration budget.
* A short projected‑Adam stage on the *true* max uses a sub‑gradient
  (mask of the largest convolution entries) to polish the best
  candidate.
* Finally a simulated‑annealing phase with richer moves (targeted
  mass transfer, swaps, three‑way redistribution) explores the
  neighbourhood of the current best.
* All phases respect a global deadline (default 1000 s).  The best
  sequence seen at any moment is returned.

The code is completely self‑contained and uses only numpy / scipy.

import math
import numpy as np
from scipy.optimize import minimize

# Helper utilities

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (maps ℝⁿ → simplex)."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the official evaluation (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, ∑x=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for random initialisation
# ----------------------------------------------------------------------


def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on {0,…,n‑1}."""
    x = np.linspace(0, 1, n)
    a = (x ** (alpha - 1.0)) * ((1 - x) ** (beta - 1.0))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Gamma‑based Dirichlet draw with concentration α (scalar)."""
    x = rng.gamma(alpha, 1.0, size=n)
    if x.sum() == 0.0:
        x = np.full(n, 1.0 / n)
    else:
        x = x / x.sum()
    return x


# ----------------------------------------------------------------------
# Smoothed max + optional variance term (log‑sum‑exp)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
    var_weight: float = 0.0,
) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) for a soft‑max parametrised vector.

    Parameters
    ----------
    y : np.ndarray
        Logits (unconstrained).
    n : int
        Length of the underlying step sequence.
    fft_len_conv : int
        Length of the FFT used for the forward convolution.
    fft_len_g : int
        Length of the FFT used for the back‑propagation step.
    lam_factor : float
        Multiplicative factor controlling the LSE smoothing.
    var_weight : float
        Weight of the variance term Var(a*a) added to the objective.
        Used only in early stages to flatten the convolution.
    """
    # vector on the simplex
    a = softmax(y)

    # ----- forward convolution a*a -----
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # ----- smoothed max (log‑sum‑exp) -----
    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12                     # smoothing coefficient
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    # ----- optional variance term -----
    if var_weight != 0.0:
        mean_c = conv_full.mean()
        var = ((conv_full - mean_c) ** 2).mean()
    else:
        var = 0.0

    # overall objective (the factor 2·n is irrelevant for the optimisation direction)
    objective = float(2.0 * n * smooth_max + var_weight * var)

    # ----- gradient w.r.t. convolution entries -----
    # derivative of smooth_max
    w = exp_shift / Z                                   # d smooth_max / d conv_i
    # derivative of variance term (∂Var/∂c_i = 2·(c_i-mean)/N, mean is constant because Σc = 1)
    if var_weight != 0.0:
        N = conv_full.shape[0]
        w += var_weight * (2.0 * (conv_full - mean_c) / N)

    # ----- back‑propagation through convolution -----
    # a_rev * w  (linear convolution, see derivation in comments above)
    a_rev = a[::-1]
    fm = np.fft.fft(w, fft_len_g)
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fm))
    # the slice corresponds to the gradient ∂F/∂a_i for i=0…n‑1
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # ----- chain rule through soft‑max -----
    mean_g = np.dot(a, g_vec)          # because ∂softmax/∂y = diag(a) - a aᵀ
    grad_y = a * (g_vec - mean_g)

    return objective, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation with a schedule of (λ, var_weight)
# ----------------------------------------------------------------------


def lbfgs_schedule(
    a_start: np.ndarray,
    schedule,
    maxiter_per_stage: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS through a sequence of smoothing/variance schedules.
    ``schedule`` is a list of (lam_factor, var_weight) pairs.
    Returns the best vector (as np.ndarray) and its fast_score.
    """
    # start from logits so that the simplex constraint is handled automatically
    y = np.log(a_start + 1e-12)

    best_a = a_start.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    for lam_factor, var_weight in schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam_factor, var_weight),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_stage, "gtol": 1e-12, "disp": False},
            )
        except Exception as exc:          # pragma: no cover
            print("[lbfgs] exception:", exc)
            continue

        y = res.x
        a = softmax(y)
        sc = fast_score(a, fft_len_conv, n)

        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[lbfgs] new best fast_score={best_sc:.6f}")

    return best_a, best_sc


# ----------------------------------------------------------------------
# Projected Adam on the *true* max‑convolution (sub‑gradient)
# ----------------------------------------------------------------------


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """Projected Adam using the true max‑convolution sub‑gradient."""
    a = a0.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    base_lr = 0.4

    while time.time() < deadline:
        it += 1
        # full convolution
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()                     # any convex combination is a valid sub‑gradient

        # back‑propagation of the sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam step (project onto simplex afterwards)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_sc = fast_score(a_candidate, fft_len_conv, n)
        if cand_sc < best_sc:
            best_sc, best_a = cand_sc, a_candidate.copy()
        a = a_candidate

    return best_a, best_sc


# ----------------------------------------------------------------------
# Simulated annealing with richer moves
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climb with square‑root cooling.
    Moves:
      - targeted transfer (from a large entry to a small one)
      - simple swap
      - three‑way redistribution
    """
    a = a_start.copy()
    best_a = a.copy()
    best_sc = start_score
    cur_sc = start_score

    T0 = max(0.08 * start_score, 1e-6)       # initial temperature
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    # pre‑compute index sets for fast “large → small” transfers
    top_frac = 0.05
    low_frac = 0.5

    while time.time() < deadline:
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)  # sqrt cooling

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            # targeted transfer: move mass from a relatively large index to a relatively small one
            top_k = max(2, int(top_frac * n))
            low_k = max(2, int(low_frac * n))

            top_idx = np.argpartition(-a, top_k)[:top_k]
            low_idx = np.argpartition(a, low_k)[:low_k]

            i = rng.choice(top_idx)
            j = rng.choice(low_idx)
            if i == j or a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # simple swap
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            # three‑way redistribution (keeps total mass)
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            w = rng.random(3)
            w /= w.sum()
            a_new[idx] = total * w

        new_sc = fast_score(a_new, fft_len, n)
        diff = new_sc - cur_sc

        if diff < 0.0 or rng.random() < math.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a_new.copy()

    return best_a, best_sc


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    evaluate_sequence(...).  Returns the best list of floats found
    before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (the original runner used n = 4000)
    # --------------------------------------------------------------
    n = 4000
    fft_len_conv = next_pow2(2 * n - 1)      # length for a*a
    fft_len_g = next_pow2(3 * n - 2)         # length for back‑propagation

    # --------------------------------------------------------------
    # Initial candidate – uniform, possibly improved by a historic vector
    # --------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len_conv, n)
    print(f"[init] uniform fast_score={best_sc:.6f}")

    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a1 = np.interp(x_new, x_old, old)
            a1 = np.maximum(a1, 0.0)
            a1 /= a1.sum()
            sc1 = fast_score(a1, fft_len_conv, n)
            if sc1 < best_sc:
                best_a, best_sc = a1, sc1
                print(f"[init] historic fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Global deadline
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # tiny safety margin

    # --------------------------------------------------------------
    # Phase 0 – systematic parametric sweep (adds cheap candidates)
    # --------------------------------------------------------------
    # Exponential sweep (denser than previous version)
    for rate in np.linspace(0.00015, 0.0015, 71):
        if time.time() >= deadline:
            break
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] exp rate={rate:.5f} → fast_score={best_sc:.6f}")

    # Power‑law sweep
    for p in np.linspace(-6.0, -0.3, 29):
        if time.time() >= deadline:
            break
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] power p={p:.2f} → fast_score={best_sc:.6f}")

    # Beta‑distribution sweep (adds a 2‑parameter family)
    for _ in range(30):
        if time.time() >= deadline:
            break
        alpha = rng.uniform(0.3, 5.0)
        beta = rng.uniform(0.3, 5.0)
        a = generate_beta(n, alpha, beta)
        sc = fast_score(a, fft_len_conv, n)
        if sc < best_sc:
            best_a, best_sc = a, sc
            print(f"[grid] beta α={alpha:.2f} β={beta:.2f} → fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – many random L‑BFGS restarts with a λ/variance schedule
    # --------------------------------------------------------------
    # Schedule: start with strong smoothing + variance, finish with almost exact max
    schedule = [
        (0.6, 1.0),
        (0.4, 0.5),
        (0.3, 0.2),
        (0.2, 0.1),
        (0.1, 0.05),
        (0.05, 0.02),
        (0.02, 0.005),
        (0.01, 0.001),
        (0.005, 0.0),
        (0.002, 0.0),
        (0.001, 0.0),
        (0.0005, 0.0),
        (0.0002, 0.0),
    ]
    maxiter_per_stage = 600                     # modest budget per λ‑stage

    while time.time() < deadline * 0.55:        # allocate ~55 % of total time
        # draw a fresh Dirichlet start (concentration randomised)
        alpha = rng.uniform(0.2, 3.0)
        a_start = random_dirichlet(n, alpha, rng)

        # add a small jitter to the best vector with 30 % probability
        if rng.random() < 0.30:
            jitter = rng.normal(1.0, 0.02, size=n)
            a_start = best_a * jitter
            a_start = np.maximum(a_start, 0.0)
            a_start /= a_start.sum()

        a_opt, sc_opt = lbfgs_schedule(
            a_start,
            schedule,
            maxiter_per_stage,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        if sc_opt < best_sc:
            best_a, best_sc = a_opt, sc_opt
            print(f"[phase1] improved fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected Adam on true max‑convolution
    # --------------------------------------------------------------
    if time.time() < deadline * 0.80:
        best_a, best_sc = subgradient_adam_optimize(
            best_a,
            deadline,
            fft_len_conv,
            fft_len_g,
            n,
        )
        print(f"[phase2] Adam sub‑gradient best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – simulated annealing with targeted moves
    # --------------------------------------------------------------
    if time.time() < deadline * 0.95:
        best_a, best_sc = simulated_annealing(
            best_a,
            best_sc,
            fft_len_conv,
            n,
            deadline,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Final polishing – a short L‑BFGS run with an ultra‑small λ
    # --------------------------------------------------------------
    if time.time() < deadline:
        tiny_sched = [(5e-07, 0.0), (2e-07, 0.0), (1e-07, 0.0), (5e-08, 0.0)]
        a_final, sc_final = lbfgs_schedule(
            best_a,
            tiny_sched,
            maxiter_per_stage=1200,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_sc:
            best_a, best_sc = a_final, sc_final
            print(f"[final‑lbfgs] best fast_score={best_sc:.6f}")

    # --------------------------------------------------------------
    # Official evaluation (the heavy call – performed once)
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
