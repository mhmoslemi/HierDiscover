def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(

    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':

        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':

        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':

        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy

    elif mode == 'bimodal':

    idx = np.arange(n, dtype=np.float64) + 0.5

def exp_gradient_optimize(a_start: np.ndarray,

def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,

    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set consists of all indices whose convolution value is within
    `eps_frac` of the current maximum.
    """

    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:

        # sub‑gradient of the (non‑smoothed) max
        a_rev = a[::-1]

            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score

def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Temperature‑controlled random walk on the simplex."""
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12  # initial temperature

    while time.time() < deadline:
        # linear cooling
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        max_frac = 0.02 + 0.18 * fraction

        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score

def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score
