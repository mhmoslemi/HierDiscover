"""
Propose a low‑evaluation step‑function.
The algorithm builds several diverse starting points (uniform,
arcsine, power‑law, exponential, random, previous best),
picks the best by a cheap max‑convolution score and then runs a
long Adam‑style projected gradient descent on a smooth
log‑sum‑exp approximation of the max‑convolution.
The smoothing parameter α is increased from 1 to ≈200 during the
run, allowing the optimizer to first see a smooth landscape and
then focus on the true maximum.
A final greedy hill‑climb fine‑tunes the solution.
All operations stay on the probability simplex, so the final
evaluation equals the true objective 2·n·max(conv).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Cheap score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Initial distributions
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point on the simplex."""
    a = rng.random(n) + 1e-9
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Arcsine‑shaped distribution (edges heavier)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def powerlaw_initial(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """
    a_i ∝ (dist_to_edge + offset)^{-p},
    where dist_to_edge = min(i+0.5, n+0.5-i).
    """
    idx = np.arange(n, dtype=np.float64) + 0.5
    dist = np.minimum(idx, n + 0.5 - idx) + offset
    a = dist ** (-p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def exponential_initial(n: int, rate: float) -> np.ndarray:
    """Symmetric exponential decay from the edges."""
    d = np.minimum(np.arange(n), n - 1 - np.arange(n))
    a = np.exp(-rate * d)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected Adam optimisation on a smooth max‑convolution
# ----------------------------------------------------------------------


def smooth_adam_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         alpha_max: float = 200.0,
                         base_lr: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Adam‑style projected gradient descent on
    `log‑sum‑exp(alpha * conv) / alpha`,
    with a linearly increasing smoothing parameter alpha.
    """
    a = a_start.copy()
    best_a = a.copy()
    # initial score (exact max, cheap to compute)
    conv = full_conv(a, fft_len)
    best_score = 2.0 * n * conv.max()

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 1
    start_time = time.time()

    while time.time() < deadline:
        # fraction of budget already used
        frac = (time.time() - start_time) / (deadline - start_time)
        alpha = 1.0 + frac * (alpha_max - 1.0)          # 1 → alpha_max

        # ---------- current convolution ----------
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_c = conv.max()
        cur_score = 2.0 * n * max_c
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # ---------- soft‑max weights ----------
        # shift for numerical stability
        shifted = conv - max_c          # ≤ 0
        exp_terms = np.exp(alpha * shifted)      # safe: all ≤ 1
        w = exp_terms / exp_terms.sum()          # shape (2n‑1,)

        # ---------- gradient ----------
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        fw = np.fft.fft(w, fft_len)
        grad_full = np.real(np.fft.ifft(fa_rev * fw))
        # extract the middle n entries (linear convolution result)
        g = 2.0 * grad_full[n - 1:n - 1 + n]

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = simplex_projection(a_new)

        a = a_new
        it += 1

    # final check (in case the loop ended before an evaluation)
    conv = full_conv(a, fft_len)
    final_score = 2.0 * n * conv.max()
    if final_score < best_score:
        best_score = final_score
        best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb using exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a tiny fraction of mass from a heavy index to a lighter one."""
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact linear convolution, length 2n‑1
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)

    return best, best_score

            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3. Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # a tiny safety margin

    # --------------------------------------------------------------
    # 4. Long projected Adam optimisation (≈ 70 % of remaining time)
    # --------------------------------------------------------------
    phase1_deadline = time.time() + 0.70 * (deadline - time.time())
    best_a, best_score = smooth_adam_optimize(
        best_a, phase1_deadline, n, fft_len, rng,
        alpha_max=200.0, base_lr=0.25
    )
    print(f"[adam] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Final greedy hill‑climb (the remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
