"""
Improved local‑search for the step‑function self‑convolution problem.

The algorithm works in two phases:

1. **Gradient‑guided hill‑climb** – at each iteration we compute the exact
   self‑convolution (FFT) and the sub‑gradient of the current maximum.
   Mass is transferred from the index with largest gradient (most responsible
   for the peak) to the index with smallest gradient.  The step size decays
   from a relatively large fraction (≈40 %) to a tiny one (10⁻⁶), and every
   move is accepted only if the max‑convolution does not increase.

2. **Simulated‑annealing fine‑tuning** – once the hill‑climb has stalled we
   perform many small random redistributions of mass.  An exponentially‑
   decaying temperature allows occasional uphill moves to escape shallow
   local minima.  The convolution is recomputed exactly after every trial.

Both phases use FFT‑based convolution (O(n log n)) and run until the allotted
time (default 1000 s) is exhausted.  The best sequence found is finally
checked with the provided `evaluate_sequence` function and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Exact linear convolution a*a using FFT (length 2·len(a)‑1).
    Returns the first 2·len(a)-1 entries of the circular convolution.
    """
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))
    return conv_full[: 2 * a.shape[0] - 1]


def max_score_from_conv(conv: np.ndarray, n: int) -> float:
    """Given the self‑convolution array, compute the evaluation score."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation phases
# ----------------------------------------------------------------------


def gradient_guided_hill_climb(a: np.ndarray,
                               start_score: float,
                               deadline: float,
                               rng: np.random.Generator,
                               start_factor: float = 0.40,
                               end_factor: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    Move mass from the index with the largest sub‑gradient to the one with the
    smallest sub‑gradient.  The step size decays linearly from `start_factor`
    to `end_factor`.  Every move is accepted only if the maximal convolution
    entry does not increase.
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score
    cur_a = a.copy()
    cur_score = start_score

    # Pre‑compute total time for the linear decay of the step factor.
    total_hill_time = deadline - time.time()
    start_time = time.time()

    while time.time() < deadline:
        # Current convolution and its maximum.
        conv = conv_fft(cur_a, fft_len)
        cur_max = conv.max()
        cur_score = max_score_from_conv(conv, n)

        # Mask of entries attaining (or near attaining) the maximum.
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # Sub‑gradient of the max‑convolution w.r.t. a.
        a_rev = cur_a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                      np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or cur_a[src] <= 0.0:
            # Degenerate case – fall back to a random feasible move.
            nonzero = np.where(cur_a > 1e-15)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero,
                                 p=cur_a[nonzero] / cur_a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # Linearly decaying step factor.
        elapsed = time.time() - start_time
        remaining_frac = max(0.0, (deadline - time.time()) / total_hill_time)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = cur_a[src] * step_factor
        if delta < 1e-15:
            continue
        if delta > cur_a[src]:
            delta = cur_a[src]

        # Apply the move.
        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        # Exact convolution for the trial state.
        conv_new = conv_fft(new_a, fft_len)
        new_score = max_score_from_conv(conv_new, n)

        # Accept only non‑worsening moves.
        if new_score <= cur_score + 1e-14:
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                elapsed_total = time.time() - start_time
                print(f"[hill] t={elapsed_total:.1f}s best_score={best_score:.6f}")

    return best_a, best_score


def simulated_annealing_fine_tune(a: np.ndarray,
                                  cur_score: float,
                                  deadline: float,
                                  rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Small random redistributions of mass with a simple temperature schedule.
    Accepts uphill moves with probability exp(‑Δ/temperature).
    """
    n = a.shape[0]
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()
    cur_score = cur_score

    # Lower bound 2 n/(2 n‑1) – used to set an initial temperature.
    lower_bound = 2.0 * n / (2 * n - 1)
    temp_start = max(1e-6, (cur_score - lower_bound) * 0.05)
    temp_end = 0.0
    start_time = time.time()

    while time.time() < deadline:
        # Time‑dependent temperature and maximal transfer fraction.
        t_elapsed = time.time() - start_time
        fraction = t_elapsed / (deadline - start_time)
        temperature = temp_start * (1.0 - fraction) + temp_end * fraction
        max_frac = 0.2 * (1.0 - fraction) + 0.001 * fraction  # ≤20 %

        # Choose a source index proportionally to its mass.
        src_candidates = np.where(cur_a > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = int(rng.choice(src_candidates,
                             p=cur_a[src_candidates] / cur_a[src_candidates].sum()))
        dst = int(rng.integers(0, n))
        if src == dst:
            continue

        # Random fraction of the source mass.
        delta = cur_a[src] * max_frac * rng.random()
        if delta < 1e-15:
            continue

        new_a = cur_a.copy()
        new_a[src] -= delta
        new_a[dst] += delta

        conv_new = conv_fft(new_a, fft_len)
        new_score = max_score_from_conv(conv_new, n)

        accept = False
        if new_score <= cur_score:
            accept = True
        else:
            if temperature > 0.0:
                prob = math.exp(-(new_score - cur_score) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a = new_a
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = cur_a.copy()
                print(f"[anneal] t={t_elapsed:.1f}s best_score={best_score:.6f}")

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Produce a non‑negative step‑function (list of floats) with a low
    `evaluate_sequence` score.  The routine runs a gradient‑guided hill‑climb
    followed by a simulated‑annealing fine‑tuning phase, staying within the
    given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a longer sequence gives a lower theoretical bound.
    # 8000 provides a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 8000
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise – start from a previously discovered construction if
    # available, otherwise fall back to a uniform distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                # Linear interpolation to the new length.
                x_old = np.linspace(0.0, 1.0, old.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, old)
            else:
                a = old.copy()
            a = np.maximum(a, 0.0)
            total = a.sum()
            if total == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= total
        else:
            a = np.full(n, 1.0 / n)
    else:
        a = np.full(n, 1.0 / n)

    # Initial score (exact).
    cur_conv = conv_fft(a, fft_len)
    cur_score = max_score_from_conv(cur_conv, n)
    best_a = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Time management.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – deterministic gradient‑guided hill‑climb.
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.55 * budget_s
    best_a, best_score = gradient_guided_hill_climb(
        best_a,
        best_score,
        phase1_end,
        rng,
        start_factor=0.40,
        end_factor=1e-6,
    )
    print(f"[phase1] best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – stochastic simulated‑annealing fine‑tuning.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_fine_tune(
            best_a,
            best_score,
            deadline,
            rng,
        )
        print(f"[phase2] final_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided security‑checked function.
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated_score={final_eval:.6f}")

    # Store the result for possible reuse by later runs.
    globals()["height_sequence_1"] = final_seq

    return final_seq
