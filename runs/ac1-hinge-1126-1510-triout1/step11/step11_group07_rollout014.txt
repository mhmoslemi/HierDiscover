"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
1. Build a diverse pool of initial candidates:
   - uniform, arcsine, several power‑law “β‑shapes”, tilted arcsine,
     random vectors and the previous best if available.
   - Keep the candidate with the smallest cheap score
     (2·n·max(conv)).

2. Repeatedly launch short “smooth‑max” gradient runs with different
   random starts.  The objective is the smoothed maximum
   `softmax_α(conv) = (1/α)·log∑exp(α·conv)`.  α is increased linearly
   from a small value to a large one, turning the smooth objective into
   the true max.  Each iteration:
   * compute the convolution by FFT,
   * form the soft‑max weights,
   * obtain a sub‑gradient of the weighted sum,
   * take a diminishing step and project onto the probability simplex.
   The best solution over all runs is kept.

3. Refine the current best with a projected sub‑gradient method that
   works directly on the true max (active‑set mask).  A simple back‑
   tracking line‑search is used.

4. A final greedy hill‑climb with exact convolution updates makes tiny
   local improvements.

All phases respect the 1000 s budget, checking the wall‑clock time
regularly.  The returned list is evaluated by the supplied
`evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Cheap score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def beta_power_initial(n: int, p: float) -> np.ndarray:
    """Power‑law family a_i ∝ (i·(n−i))^(−p) (sym‑metric)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    a = (i * (n + 0.5 - i)) ** (-p)
    a /= a.sum()
    return a


def tilt_distribution(a: np.ndarray, tilt: float) -> np.ndarray:
    """Apply a linear tilt; tilt in [−1,1] (negative → left‑heavy)."""
    n = a.shape[0]
    x = (np.arange(n) - (n - 1) / 2) / (n / 2)   # range roughly [‑1,1]
    a_t = a * (1.0 + tilt * x)
    a_t = np.clip(a_t, 0.0, None)
    if a_t.sum() == 0.0:
        a_t = np.full_like(a_t, 1.0 / n)
    else:
        a_t /= a_t.sum()
    return a_t


# ----------------------------------------------------------------------
# Smooth‑max (softmax) gradient descent
# ----------------------------------------------------------------------


def smooth_max_optimize(a_start: np.ndarray,
                        deadline: float,
                        n: int,
                        fft_len: int,
                        rng: np.random.Generator,
                        eta0: float = 0.01,
                        alpha_start: float = 5.0,
                        alpha_end: float = 2000.0,
                        max_iter: int = 5000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smoothed max (soft‑max) of the convolution.
    The smoothing parameter alpha is increased linearly from alpha_start
    to alpha_end, turning the objective into the true max.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    start = time.time()
    it = 0
    while time.time() < deadline and it < max_iter:
        # convolution a*a
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        # linearly increase smoothing
        fraction = it / max_iter
        alpha = alpha_start + (alpha_end - alpha_start) * fraction

        # soft‑max weights (stable version)
        scaled = alpha * conv
        scaled -= scaled.max()                     # shift for numerical stability
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.shape[0])
        else:
            w /= w_sum

        # sub‑gradient of the weighted sum  ∑ w_i·conv_i  w.r.t. a
        a_rev = a[::-1]
        fw = np.fft.fft(w, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fw))
        g = 2.0 * grad_conv[n - 1:n - 1 + n]

        # simple diminishing step size
        step = eta0 / np.sqrt(it + 1.0)
        a_candidate = a - step * g
        a_candidate = simplex_projection(a_candidate)

        # keep best (cheap evaluation using fast_score)
        score = fast_score(a_candidate, fft_len)
        if score < best_score:
            best_score = score
            best_a = a_candidate.copy()

        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgradient_optimize(a_start: np.ndarray,
                                   deadline: float,
                                   n: int,
                                   fft_len: int,
                                   rng: np.random.Generator,
                                   eps_frac: float = 0.001,
                                   step0: float = 0.2) -> tuple[np.ndarray, float]:
    """Additive sub‑gradient descent with Euclidean projection onto the simplex."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(a, fft_len)

    it = 0
    while time.time() < deadline:
        # current convolution
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        # sub‑gradient of the (non‑smoothed) max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1:n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        # back‑tracking line‑search (few attempts)
        improved = False
        for _ in range(4):
            a_candidate = a - step * g
            a_candidate = simplex_projection(a_candidate)
            cand_score = fast_score(a_candidate, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
            if cand_score < fast_score(a, fft_len):
                a = a_candidate
                improved = True
                break
            step *= 0.5
        if not improved:
            # small random jitter to escape shallow plateaus
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score

def simulated_annealing(a_start: np.ndarray,
                        start_score: float,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Temperature‑controlled random walk on the simplex."""
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12  # initial temperature

    while time.time() < deadline:
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        max_frac = 0.02 + 0.18 * fraction

        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(a_start: np.ndarray,
                           start_score: float,
                           deadline: float,
                           rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src:src + n] -= best
        diff[dst:dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            # periodically recompute exactly to avoid drift
            conv = np.convolve(best, best)

    return best, best_score

        init_candidates.append(random_initial(n, rng))

    # previous best, if the harness provided one
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            if a_old.sum() == 0.0:
                a_old = np.full_like(a_old, 1.0 / n)
            else:
                a_old /= a_old.sum()
            init_candidates.append(a_old)

    # ------------------------------------------------------------------
    # Pick the best primitive candidate (cheap score)
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2. Time bookkeeping
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05    # safety margin

    # --------------------------------------------------------------
    # 3. Multiple short smooth‑max gradient runs (exploration)
    # --------------------------------------------------------------
    # Allocate roughly 70 % of the remaining budget to these runs.
    remaining = deadline - time.time()
    allocation = 0.7 * remaining
    end_of_phase1 = time.time() + allocation

    while time.time() < end_of_phase1 - 5.0:
        # pick a start point: either a jittered copy of the current best
        # or a fresh random vector
        if rng.random() < 0.6:
            a_start = best_a.copy()
            jitter = rng.normal(scale=0.001, size=n)
            a_start = a_start + jitter
            a_start = np.clip(a_start, 0.0, None)
            if a_start.sum() == 0.0:
                a_start = np.full(n, 1.0 / n)
            else:
                a_start /= a_start.sum()
        else:
            a_start = random_initial(n, rng)

        # Give each run a modest slice of time (≈2 % of the total budget)
        run_deadline = min(end_of_phase1, time.time() + 0.02 * allocation)

        a_opt, sc_opt = smooth_max_optimize(
            a_start, run_deadline, n, fft_len, rng,
            eta0=0.02, alpha_start=5.0, alpha_end=2000.0, max_iter=4000
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[smooth‑run] improved fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient refinement (focus on true max)
    # --------------------------------------------------------------
    phase2_deadline = deadline - 2.0   # reserve a couple of seconds for later phases
    best_a, best_score = projected_subgradient_optimize(
        best_a, phase2_deadline, n, fft_len, rng,
        eps_frac=0.0005, step0=0.15
    )
    print(f"[proj‑subgrad] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5. Global simulated annealing (light‑weight)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        best_a, best_score = simulated_annealing(
            best_a, best_score, deadline - 1.0, rng
        )
        print(f"[SA] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Final greedy hill‑climb (exact convolution updates)
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = hill_climb_incremental(
            best_a, best_score, deadline, rng
        )
        print(f"[hill‑climb] fast_score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
