```
def exp_gradient_momentum(a_start, deadline, n, fft_len, rng, eta0=20.0, beta=0.9, max_iters=10_000_000):
    a = a_start.copy()
    best_a = a.copy()
    best_sc = fast_score(best_a, fft_len)
    m = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv and mask same as before
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa*fa))[:2*n-1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        grad = 2.0 * grad_conv[n-1:n-1+n]   # length n
        # momentum
        m = beta * m + (1-beta) * grad
        # step size
        eta = eta0 / np.sqrt(it + 1.0)
        # mirror update
        exp_term = np.exp(np.clip(-eta * m, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0:
            a = best_a.copy()
        else:
            a /= s
        # quick score check
        if (it & 31) == 0:
            sc = fast_score(a, fft_len)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        it+=1
    return best_a, best_sc
```

Multi‑phase step‑function optimiser (≈1000 s budget).

1. Build a diverse pool of candidate probability vectors (uniform, random,
   exponential, power, edge‑heavy, Gaussian, Hann, sine, triangular,
   logistic, sparse, …).  The best pool element is taken as a warm‑start.
2. Coarse L‑BFGS optimisation of a smooth log‑sum‑exp surrogate of the
   true L∞ objective.  A cascade of decreasing smoothing parameters drives
   the iterate into a good basin.
3. Mirror‑SGD (exponentiated‑gradient) on the exact max‑convolution,
   enhanced with a momentum term and a slowly decaying step‑size.
4. Fine L‑BFGS polishing with very small smoothing parameters and a large
   number of iterations.
5. A short simulated‑annealing hill‑climb to escape remaining shallow
   local minima.
All phases respect the global 1000 s deadline; the best sequence found is
stored in the global variable ``height_sequence_1`` and returned.
"""
import time
import numpy as np
from scipy.optimize import minimize

# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when the vector a sums to one:
        2·n·max(conv(a,a))
    The convolution is evaluated by FFT for speed.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def interpolate_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D vector to a new length."""
    if old.size == 0:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    a = np.interp(x_new, x_old, old)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian‐shaped probability vector with random width."""
    mu = n / 2.0
    sigma = rng.uniform(0.1, 0.35) * n
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def hann_initial(n: int) -> np.ndarray:
    """Classic Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def sine_initial(n: int) -> np.ndarray:
    """Sine window (half‑sine)."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def triangular_initial(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def logistic_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Logistic‑shaped window with random scale."""
    mu = n / 2.0
    scale = rng.uniform(0.08, 0.25) * n
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp(-(i - mu) / scale))
    a -= a.min()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution objective and its gradient w.r.t. the log‑variables y.
    The mapping a = softmax(y) guarantees a ∈ simplex.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # smoothed max (log‑sum‑exp)
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

# Exponentiated‑gradient descent with momentum on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_momentum(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 22.0,
                         beta: float = 0.9,
                         max_iters: int = 8_000_000):
    """
    Mirror‑SGD (exponentiated‑gradient) using the exact sub‑gradient of the
    max‑convolution, enhanced with a momentum term.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    m = np.zeros_like(a)               # momentum buffer
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        m = beta * m + (1.0 - beta) * g

        # stepsize schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step (clipped exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * m, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # periodic score check
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        it += 1

    return best_a, best_score

# Projected Euclidean gradient descent on the true max‑convolution

def projected_gradient_descent(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              rng: np.random.Generator,
                              init_step: float = 0.05,
                              max_iters: int = 300_000):
    """
    Simple projected gradient descent (Euclidean) on the exact max‑convolution.
    Step size is adapted based on success/failure of the move.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    step = init_step
    it = 0

    while time.time() < deadline and it < max_iters:
        # compute mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # length n

        # gradient descent step (move opposite to gradient)
        a_candidate = a - step * g
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            # if improvement, allow a modest increase of step size
            step *= 1.02
        else:
            # no improvement – shrink step
            step *= 0.5
            # fallback to old a (already stored)

        a = a_candidate
        it += 1

    return best_a, best_score

# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature linearly decreases from 0.05·best_score to a tiny floor.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.25  # maximal fraction of source mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


def interpolate_sequence(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D vector to a new length."""
    if old.size == 0:
        return np.full(new_n, 1.0 / new_n)
    x_old = np.linspace(0.0, 1.0, old.size)
    x_new = np.linspace(0.0, 1.0, new_n)
    a = np.interp(x_new, x_old, old)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def gaussian_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Gaussian‑shaped probability vector with random width."""
    mu = n / 2.0
    sigma = rng.uniform(0.1, 0.35) * n
    x = np.arange(n, dtype=float)
    a = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    a /= a.sum()
    return a


def hann_initial(n: int) -> np.ndarray:
    """Classic Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=float)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def sine_initial(n: int) -> np.ndarray:
    """Half‑sine window."""
    i = np.arange(n, dtype=float)
    a = np.sin(np.pi * (i + 0.5) / n)
    a /= a.sum()
    return a


def triangular_initial(n: int) -> np.ndarray:
    """Symmetric triangular window."""
    i = np.arange(n, dtype=float)
    a = 1.0 - np.abs(i - (n - 1) / 2.0) / ((n - 1) / 2.0)
    a /= a.sum()
    return a


def logistic_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Logistic‑shaped window with random scale."""
    mu = n / 2.0
    scale = rng.uniform(0.08, 0.25) * n
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp(-(i - mu) / scale))
    a -= a.min()
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth surrogate (log‑sum‑exp) for L‑BFGS
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution objective and its gradient w.r.t. the log‑variables y.
    The mapping a = softmax(y) guarantees a ∈ simplex.
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    # smoothed max (log‑sum‑exp)
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                     # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]   # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent with momentum on the exact max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_momentum(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 22.0,
                         beta: float = 0.9,
                         max_iters: int = 8_000_000):
    """
    Mirror‑SGD (exponentiated‑gradient) using the exact sub‑gradient of the
    max‑convolution, enhanced with a momentum term.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    m = np.zeros_like(a)               # momentum buffer
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # momentum update
        m = beta * m + (1.0 - beta) * g

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step (clipped exponent to avoid overflow)
        exp_term = np.exp(np.clip(-eta * m, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # periodic score check
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Projected Euclidean gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_gradient_descent(a_start: np.ndarray,
                              deadline: float,
                              n: int,
                              fft_len: int,
                              rng: np.random.Generator,
                              init_step: float = 0.05,
                              max_iters: int = 300_000):
    """
    Simple projected gradient descent (Euclidean) on the exact max‑convolution.
    Step size is adapted based on success/failure of the move.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    step = init_step
    it = 0

    while time.time() < deadline and it < max_iters:
        # mask of maximal entries
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # sub‑gradient of the (hard) max
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # length n

        # gradient step (move opposite to gradient) and projection
        a_candidate = a - step * g
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            step *= 1.02          # modest step‑size increase on success
        else:
            step *= 0.5           # shrink step on failure

        a = a_candidate
        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature linearly decreases from 0.05·best_score to a tiny floor.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score
    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-8)
    max_frac = 0.25  # maximal fraction of source mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger entry to smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000  # works well within the budget; can be changed if desired
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Re‑use the previous best (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            pool.append(interpolate_sequence(old, n))

    # 2) Deterministic windows
    pool.append(np.full(n, 1.0 / n))     # uniform
    pool.append(hann_initial(n))
    pool.append(sine_initial(n))
    pool.append(triangular_initial(n))

    # 3) Parametric families with random parameters
    for _ in range(4):
        pool.append(gaussian_initial(n, rng))
    for _ in range(4):
        pool.append(logistic_initial(n, rng))

    # 4) Randomly generated families
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # ------------------------------------------------------------------
    # Timing windows
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # tiny safety margin

    phase0_end = start_time + 0.05 * budget_s       # quick scan
    phase1_end = start_time + 0.20 * budget_s       # coarse L‑BFGS
    phase2_end = start_time + 0.85 * budget_s       # long EG‑momentum
    # remaining time → fine L‑BFGS, PGD, anneal

    # ------------------------------------------------------------------
    # Phase 0 – cheap evaluation of the pool
    # ------------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in pool:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[phase0] best fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 1 – coarse L‑BFGS cascade
    # ------------------------------------------------------------------
    lam_schedule = [
        0.5, 0.35, 0.24, 0.16, 0.11,
        0.075, 0.051, 0.034, 0.023, 0.015,
        0.01, 0.006, 0.004, 0.0025, 0.0015,
        0.001, 0.0007, 0.0005, 0.0003, 0.0002
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        if rng.random() < 0.30:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.015), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient descent with momentum
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_momentum(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=24.0,
            beta=0.92,
            max_iters=10_000_000
        )
        print(f"[exp‑grad‑mom] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – fine L‑BFGS polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-5, 2.5e-5, 1.2e-5, 6e-6, 3e-6, 1.5e-6, 8e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[fine‑L‑BFGS] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – projected gradient descent (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = projected_gradient_descent(
            best_a,
            deadline,
            n,
            fft_len,
            rng,
            init_step=0.07,
            max_iters=400_000
        )
        print(f"[PGD] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 5 – short simulated‑annealing hill‑climb
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
