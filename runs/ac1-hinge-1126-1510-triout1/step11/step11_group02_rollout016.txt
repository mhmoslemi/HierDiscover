"""
Search for a low‑score non‑negative step sequence.

The algorithm works in several stages:
  0️⃣  A systematic sweep over several parametric families
      (exponential, power‑law, logistic, Beta‑distribution).  The best
      candidate is kept.
  1️⃣  A cheap random sweep over many handcrafted families (including the
      new “beta” family and bimodal / cosine shapes).
  2️⃣  Aggressive multi‑restart L‑BFGS on a smooth log‑sum‑exp
      surrogate of max(a∗a).  The temperature λ is decreased according to a
      long schedule, each stage started from the previous one.
  3️⃣  Projected Adam on the true (non‑smooth) max‑convolution – a robust
      sub‑gradient method.
  4️⃣  Simulated annealing with three move types (transfer, swap,
      triple‑redistribution) and a square‑root cooling schedule.
  5️⃣  Final polishing L‑BFGS with a very small λ.

All phases respect a global 1000 s deadline; the best sequence seen at any
time is returned.  The code is pure NumPy / SciPy, uses no I/O, and works
with the already‑imported ``evaluate_sequence`` function.
"""

import time
import numpy as np
from scipy.optimize import minimize, linprog

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast surrogate for the official evaluation (∑a = 1)."""
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, ∑x = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed max (log‑sum‑exp) and its gradient (logits → probabilities)
# ----------------------------------------------------------------------


def objective_and_grad(
    y: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_g: int,
    lam_factor: float = 0.30,
) -> tuple[float, np.ndarray]:
    """
    Smoothed max of a*a where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    a = softmax(y)

    # convolution a*a
    fa = np.fft.fft(a, fft_len_conv)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z  # d smooth_max / d conv_i

    # back‑propagation through convolution
    a_rev = a[::-1]
    fa_rev = np.fft.fft(a_rev, fft_len_g)
    fw = np.fft.fft(w, fft_len_g)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[(n - 1) : (n - 1 + n)]

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y

def lbfgs_continuation(
    a_start: np.ndarray,
    lam_schedule,
    maxiter_per_lam: int,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS for each λ in lam_schedule, starting each stage
    from the result of the previous stage.
    """
    y = np.log(a_start + 1e-12)  # logits – stays on the simplex

    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len_conv, fft_len_g, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception as e:  # pragma: no cover
            print("[lbfgs] exception:", e)
            continue

        y = res.x
        a = softmax(y)
        score = fast_score(a, fft_len_conv, n)

        if score < best_score:
            best_score, best_a = score, a.copy()
            print(f"[lbfgs] new best fast_score={best_score:.6f}")

    return best_a, best_score


def subgradient_adam_optimize(
    a0: np.ndarray,
    deadline: float,
    fft_len_conv: int,
    fft_len_g: int,
    n: int,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true max‑convolution (sub‑gradient of max).
    Very robust for long runs.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len_conv, n)

    m = np.zeros_like(a)  # first moment
    v = np.zeros_like(a)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    base_lr = 0.5  # a bit larger than the previous version

    while time.time() < deadline:
        it += 1
        # true convolution a*a
        fa = np.fft.fft(a, fft_len_conv)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps_val = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps_val).astype(np.float64)
        mask /= mask.sum()  # a sub‑gradient of the max

        # back‑propagation of the sub‑gradient
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len_g)
        fa_rev = np.fft.fft(a_rev, fft_len_g)
        conv_g = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * conv_g[(n - 1) : (n - 1 + n)]

        # Adam update (project later)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = a - step
        a_candidate = simplex_projection(a_candidate)

        cand_score = fast_score(a_candidate, fft_len_conv, n)
        if cand_score < best_score:
            best_score, best_a = cand_score, a_candidate.copy()
        a = a_candidate

    return best_a, best_score


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    fft_len: int,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climb with square‑root cooling.
    Moves:
      – transfer a random fraction of mass from i → j,
      – swap the values of two indices,
      – redistribute a small fraction among three random positions.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score

    # temperature initialization
    T0 = max(0.08 * start_score, 1e-6)
    t_start = time.time()
    total_time = max(deadline - t_start, 1e-6)

    while time.time() < deadline:
        # square‑root cooling
        remain = max(deadline - time.time(), 0.0)
        T = T0 * np.sqrt(remain / total_time)

        move_type = rng.integers(3)
        a_new = a.copy()

        if move_type == 0:
            # simple transfer
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if a[i] <= 0.0:
                continue
            delta = a[i] * rng.random() * 0.35
            if delta <= 0.0:
                continue
            a_new[i] -= delta
            a_new[j] += delta

        elif move_type == 1:
            # swap two positions
            i, j = rng.choice(n, size=2, replace=False)
            a_new[i], a_new[j] = a_new[j], a_new[i]

        else:
            # triple redistribution (keep sum)
            idx = rng.choice(n, size=3, replace=False)
            total = a_new[idx].sum()
            weights = rng.random(3)
            weights /= weights.sum()
            a_new[idx] = total * weights

        new_score = fast_score(a_new, fft_len, n)
        diff = new_score - cur_score

        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

    return best_a, best_score

def generate_exp(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.exp(-rate * i)
    a /= a.sum()
    return a


def generate_power(n: int, p: float, inc: bool = True) -> np.ndarray:
    i = np.arange(n, dtype=float) + 1.0
    if not inc:
        i = n - i + 1
    a = i ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_logistic(n: int, mu: float, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def generate_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) on the interior of the interval, with a tiny shift to avoid 0/1."""
    x = (np.arange(n, dtype=float) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    # replace NaNs that arise from 0**negative by 0
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def generate_gauss(n: int, sigma: float, mu: float = None) -> np.ndarray:
    i = np.arange(n, dtype=float)
    if mu is None:
        mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def generate_cos(n: int, width: float) -> np.ndarray:
    i = np.arange(n, dtype=float)
    a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def generate_bimodal(n: int, rng: np.random.Generator) -> np.ndarray:
    i = np.arange(n, dtype=float)
    mu1 = rng.uniform(0.0, n / 2.0)
    mu2 = rng.uniform(n / 2.0, n - 1)
    sigma1 = rng.uniform(100.0, 500.0)
    sigma2 = rng.uniform(100.0, 500.0)
    a1 = np.exp(-0.5 * ((i - mu1) / sigma1) ** 2)
    a2 = np.exp(-0.5 * ((i - mu2) / sigma2) ** 2)
    a = a1 + a2
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def generate_family(family: str, n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random candidate from a parametric family."""
    i = np.arange(n, dtype=float)

    if family == "power":
        p = rng.uniform(-5.0, -0.5)
        inc = rng.choice([True, False])
        a = (i + 1) ** p if inc else (n - i) ** p
    elif family == "exp":
        rate = rng.uniform(0.0002, 0.0015)
        a = np.exp(-rate * i)
    elif family == "gauss":
        sigma = rng.uniform(200.0, 2000.0)
        a = np.exp(-0.5 * ((i - (n - 1) / 2.0) / sigma) ** 2)
    elif family == "cos":
        width = rng.uniform(200.0, n / 2.0)
        a = np.cos(np.pi * (i - (n - 1) / 2.0) / width)
        a = np.maximum(a, 0.0)
    elif family == "bimodal":
        a = generate_bimodal(n, rng)
    elif family == "logistic":
        mu = rng.uniform(0.0, n - 1)
        sigma = rng.uniform(200.0, 1200.0)
        a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    elif family == "beta":
        alpha = rng.uniform(0.2, 5.0)
        beta = rng.uniform(0.2, 5.0)
        a = generate_beta(n, alpha, beta)
    elif family == "binary_sparse":
        prob = rng.uniform(0.02, 0.25)
        mask = rng.random(n) < prob
        if not np.any(mask):
            mask[0] = True
        a = mask.astype(float)
    elif family == "double_exp":
        rate = rng.uniform(0.0002, 0.0012)
        a = np.exp(-rate * i) + np.exp(-rate * (n - 1 - i))
    else:  # completely random
        a = rng.random(n)

    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def parametric_grid_search(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Systematic search over exponential, power‑law, logistic and Beta families."""
    best_a = np.full(n, 1.0 / n)
    best_sc = fast_score(best_a, fft_len, n)

    # exponential grid
    for rate in np.linspace(0.0002, 0.0010, 41):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_exp(n, rate)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] exp rate={rate:.5f} → score={best_sc:.6f}")

    # power‑law grid
    for p in np.linspace(-5.0, -0.5, 19):
        if time.time() >= deadline:
            return best_a, best_sc
        a = generate_power(n, p, inc=True)
        sc = fast_score(a, fft_len, n)
        if sc < best_sc:
            best_sc, best_a = sc, a.copy()
            print(f"[grid] power p={p:.2f} → score={best_sc:.6f}")

    # logistic grid
    mus = np.linspace(0.2 * n, 0.8 * n, 7)
    sigmas = np.linspace(200.0, 1200.0, 7)
    for mu in mus:
        for sigma in sigmas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_logistic(n, mu, sigma)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] logistic mu={mu:.0f} sigma={sigma:.0f} → score={best_sc:.6f}")

    # Beta family grid (α and β in [0.2, 5.0])
    alphas = np.linspace(0.2, 5.0, 9)
    betas = np.linspace(0.2, 5.0, 9)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                return best_a, best_sc
            a = generate_beta(n, alpha, beta)
            sc = fast_score(a, fft_len, n)
            if sc < best_sc:
                best_sc, best_a = sc, a.copy()
                print(f"[grid] beta α={alpha:.2f} β={beta:.2f} → score={best_sc:.6f}")

    return best_a, best_sc

    # Phase 3 – simulated annealing (rich move set)
    # --------------------------------------------------------------
    if time.time() < phase3_end:
        best_a, best_score = simulated_annealing(
            best_a,
            best_score,
            fft_len_conv,
            n,
            phase3_end,
            rng,
        )
        print(f"[phase3] annealing best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing – short L‑BFGS run with an ultra‑small λ
    # --------------------------------------------------------------
    if time.time() < deadline:
        final_start = best_a.copy()
        tiny_sched = [1e-07, 5e-08, 2e-08, 1e-08]
        a_final, sc_final = lbfgs_continuation(
            final_start,
            tiny_sched,
            maxiter_per_lam=1500,
            deadline=deadline,
            fft_len_conv=fft_len_conv,
            fft_len_g=fft_len_g,
            n=n,
        )
        if sc_final < best_score:
            best_score, best_a = sc_final, a_final.copy()
            print(f"[final‑lbfgs] best fast_score={best_score:.6f}")

    # --------------------------------------------------------------
    # Official evaluation
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
