def guided_hc_optimal(a_start, start_score, deadline, rng):
    n = len(a_start)
    a = a_start.copy()
    conv = np.convolve(a, a)  # length 2n-1
    best_a = a.copy()
    best_score = start_score
    cur_score = start_score
    total = deadline - time.time()
    accepted = 0
    steps = 0
    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) * np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n-1:n-1+n]
        src = int(np.argmax(g))
        dst = int(np.argmin(g))
        if src == dst or a[src] < 1e-15:
            # fallback random movement
            nonzero = np.where(a > 1e-12)[0]
            if nonzero.size == 0:
                break
            src = int(rng.choice(nonzero, p=a[nonzero] / a[nonzero].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue
        # compute optimal delta using ternary search
        delta_opt = optimal_move_delta(src, dst, a, conv, a[src])
        if delta_opt <= 1e-15:
            # cannot move, break to avoid infinite loop
            break
        a_old = a.copy()
        conv_old = conv.copy()
        # apply move
        a[src] -= delta_opt
        a[dst] += delta_opt
        # update conv exactly
        # linear part
        conv[src:src+n] -= 2.0 * delta_opt * a_old
        conv[dst:dst+n] += 2.0 * delta_opt * a_old
        # quadratic part
        idx2src = 2*src
        if idx2src < conv.shape[0]:
            conv[idx2src] += delta_opt**2
        idx2dst = 2*dst
        if idx2dst < conv.shape[0]:
            conv[idx2dst] += delta_opt**2
        idx_srcdst = src + dst
        if idx_srcdst < conv.shape[0]:
            conv[idx_srcdst] -= 2.0 * delta_opt**2
        # recompute max
        new_max = conv.max()
        new_score = 2.0 * n * new_max
        if new_score <= cur_score + 1e-12:
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # should not happen if delta_opt optimal; revert
            a = a_old
            conv = conv_old
        steps += 1
        # optional periodic recompute
        if steps % 5000 == 0:
            conv = np.convolve(a, a)
    return best_a, best_score

def optimal_move_delta(src, dst, a, conv, a_src_val):
    n = len(a)
    L = 2*n - 1
    # Precompute slices for linear term
    src_slice_start = src
    src_slice_end = src + n
    dst_slice_start = dst
    dst_slice_end = dst + n
    def new_max(delta):
        # Using current conv and a (original) to compute updated conv's max
        # We'll compute incremental updates rather than copy conv for speed
        # But need copy anyway to compute max; simplest copy
        new_conv = conv.copy()
        # linear updates
        new_conv[src_slice_start:src_slice_end] -= 2.0 * delta * a
        new_conv[dst_slice_start:dst_slice_end] += 2.0 * delta * a
        # quadratic
        idx2src = 2*src
        if idx2src < L:
            new_conv[idx2src] += delta*delta
        idx2dst = 2*dst
        if idx2dst < L:
            new_conv[idx2dst] += delta*delta
        idx_srcdst = src + dst
        if idx_srcdst < L:
            new_conv[idx_srcdst] -= 2.0 * delta*delta
        return new_conv.max()
    lo = 0.0
    hi = a_src_val
    # ternary search for minimum
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = new_max(m1)
        f2 = new_max(m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0
```

def compute_subgradient(a, conv):

"""
Improved step‑function search for the evaluation function.

The algorithm works in several stages:

1. **Random initialisation & L‑BFGS** – many short L‑BFGS runs on a
   log‑sum‑exp smoothed version of the objective, each from a
   randomly generated distribution on the simplex.

2. **Adam** – fine‑tuning of the same smooth objective with an adaptive
   learning‑rate scheme.

3. **Exponentiated‑gradient on the true max‑convolution** – a
   multiplicative update that directly reduces the peak of the
   self‑convolution.

4. **Gradient‑guided hill‑climb with optimal line‑search** – the
   sub‑gradient of the maximal convolution entries is used to identify a
   source (largest sub‑gradient) and a destination (smallest
   sub‑gradient).  The amount of mass transferred is chosen by a cheap
   ternary‑search on the exact max‑convolution after the move, guaranteeing
   a monotone decrease of the score.

5. **Random perturbations + extra refinement** – if time permits the
   best solution is slightly perturbed and the hill‑climb is run again,
   allowing the search to escape shallow local minima.

All work is performed on a normalized distribution (`sum(a)=1`), so the
fast score `2·n·max(conv(a,a))` coincides with the value returned by
`evaluate_sequence`.  The code respects the 1000 s time limit and never
uses illegal types.

The final candidate is returned as a plain list of Python floats ready
for the harness.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random probability vector of length n."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) for L‑BFGS / Adam
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int,
                       fft_len: int, lam_factor: float = 0.30):
    """
    Smoothed objective based on a log‑sum‑exp approximation of the max
    self‑convolution.  Returns (value, grad) w.r.t. logits y where a = softmax(y).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth max w.r.t. conv entries
    w = exp_shift / Z                     # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]    # gradient wrt a before softmax

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_optimize(a_start: np.ndarray,
                   lam_schedule,
                   maxiter_per_lam: int,
                   deadline: float,
                   fft_len: int,
                   n: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best a and its fast_score.
    """
    y = np.log(a_start + 1e-12)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


def adam_optimize(y_start: np.ndarray,
                  lam_schedule,
                  max_steps_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int,
                  lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective.
    Returns the best a encountered and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = np.exp(y - np.max(y))
    best_a /= best_a.sum()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad(y, n, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            a = np.exp(y - np.max(y))
            a /= a.sum()
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent on the hard max‑convolution.
    Returns the best a found and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score

def _new_max_after_move(delta: float, src: int, dst: int,
                       a: np.ndarray, conv: np.ndarray) -> float:
    """Return the maximal convolution value after moving 'delta' mass."""
    n = a.shape[0]
    L = 2 * n - 1
    new_conv = conv.copy()

    # linear part (uses the *old* a)
    new_conv[src:src + n] -= 2.0 * delta * a
    new_conv[dst:dst + n] += 2.0 * delta * a

    # quadratic part
    idx = 2 * src
    if idx < L:
        new_conv[idx] += delta * delta
    idx = 2 * dst
    if idx < L:
        new_conv[idx] += delta * delta
    idx = src + dst
    if idx < L:
        new_conv[idx] -= 2.0 * delta * delta

    return float(new_conv.max())


def optimal_move_delta(src: int, dst: int,
                       a: np.ndarray,
                       conv: np.ndarray,
                       a_src_val: float) -> float:
    """
    Find (by ternary search) a mass delta ∈ [0, a_src_val] that minimises
    max(conv after moving delta from src to dst).
    """
    lo, hi = 0.0, a_src_val
    for _ in range(25):          # 25 iterations ≈ 1e‑7 relative precision
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = _new_max_after_move(m1, src, dst, a, conv)
        f2 = _new_max_after_move(m2, src, dst, a, conv)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0


def guided_hc_optimal(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided hill‑climb that moves an *optimally chosen* amount of
    mass from the index with the largest sub‑gradient to the index with the
    smallest sub‑gradient.  The move size is obtained by a cheap ternary
    search on the exact convolution, guaranteeing monotone improvement.
    """
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    a = a_start.copy()
    conv = np.convolve(a, a)               # exact linear convolution, length 2n‑1
    best_a = a.copy()
    best_score = start_score

    steps = 0
    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        # fall back to a random direction if the gradient is degenerate
        if src == dst or a[src] < 1e-14:
            nz = np.where(a > 1e-12)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz, p=a[nz] / a[nz].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        delta_opt = optimal_move_delta(src, dst, a, conv, a[src])
        if delta_opt <= 1e-15:
            break      # nothing useful can be moved

        a_old = a.copy()
        conv_old = conv.copy()

        # apply the optimal move exactly
        a[src] -= delta_opt
        a[dst] += delta_opt

        # linear term update
        conv[src:src + n] -= 2.0 * delta_opt * a_old
        conv[dst:dst + n] += 2.0 * delta_opt * a_old

        # quadratic term update
        L = conv.shape[0]
        idx = 2 * src
        if idx < L:
            conv[idx] += delta_opt * delta_opt
        idx = 2 * dst
        if idx < L:
            conv[idx] += delta_opt * delta_opt
        idx = src + dst
        if idx < L:
            conv[idx] -= 2.0 * delta_opt * delta_opt

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score + 1e-12:
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            # numerical safeguard – revert
            a = a_old
            conv = conv_old

        steps += 1

        # occasional exact recompute to keep drift under control
        if steps % 5000 == 0:
            conv = np.convolve(a, a)

    return best_a, best_score

            if old.size != n:
                x_old = np.linspace(0, 1, old.size)
                x_new = np.linspace(0, 1, n)
                a0 = np.interp(x_new, x_old, old)
            else:
                a0 = old.copy()
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)

    best_a = a0.copy()
    best_score = fast_score(best_a, fft_len)
    print(f"[init] n={n}, fast_score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 0.05   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – many short L‑BFGS runs from random starts
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.08 * budget_s   # ~80 s
    lam_schedule = np.geomspace(1.0, 1e-6, num=15)
    maxiter_per_lam = 300
    while time.time() < phase1_end:
        a_start = random_initial(n, rng)
        a_opt, score_opt = lbfgs_optimize(
            a_start, lam_schedule, maxiter_per_lam,
            phase1_end, fft_len, n
        )
        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on the smoothed objective
    # ------------------------------------------------------------------
    if time.time() < deadline:
        y0 = np.log(best_a + 1e-12)
        lam_sched = [0.5, 0.3, 0.2, 0.1, 0.05,
                     0.02, 0.01, 0.005, 0.002, 0.001]
        best_a, best_score = adam_optimize(
            y0, lam_sched, max_steps_per_lam=5000,
            deadline=deadline, n=n, fft_len=fft_len, lr0=0.15
        )
        print(f"[Adam] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exponentiated gradient on the true max‑convolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = exp_gradient_optimize(
            best_a, deadline, n, fft_len, rng, eta0=18.0
        )
        print(f"[exp‑grad] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – gradient‑guided hill‑climb with optimal line‑search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = guided_hc_optimal(
            best_a, best_score, deadline, rng
        )
        print(f"[guided_hc_opt] fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – random perturbations + extra hill‑climb if time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:   # keep a few seconds for final eval
        sigma = 0.005 + 0.02 * rng.random()   # modest jitter
        pert = best_a * (1.0 + sigma * rng.standard_normal(n))
        pert = np.clip(pert, 0.0, None)
        if pert.sum() == 0.0:
            continue
        pert /= pert.sum()
        sub_deadline = min(deadline, time.time() + 20.0)
        a_ref, score_ref = guided_hc_optimal(
            pert, fast_score(pert, fft_len), sub_deadline, rng
        )
        if score_ref < best_score:
            best_score = score_ref
            best_a = a_ref.copy()
            print(f"[perturb‑hc] new best fast_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the provided security‑checked function
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.6f}")

    return final_seq
