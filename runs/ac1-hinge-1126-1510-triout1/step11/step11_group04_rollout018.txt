        if (it & 31)==0:
            sc = fast_score(a, fft_len)
            if sc<best_score:
                best_score=sc; best_a=a.copy()
        it+=1
    return best_a,best_score

"""
step‑function optimiser – 1000 s budget (enhanced 2)

The optimiser consists of four stages

1. **Broad L‑BFGS sweep** – many cheap L‑BFGS runs on a
   log‑sum‑exp surrogate of the L∞ objective.  The λ‑schedule
   is long (≈20 values) and each sub‑run performs up to 1500
   BFGS iterations.  Random starts are heavily jittered,
   producing very diverse basins.

2. **Long mirror‑SGD (entropy‑mirror / exponentiated‑gradient)**
   on the exact max‑convolution.  An AdaGrad‑style per‑coordinate
   learning‑rate adaption is used, together with a schedule
   η₀ / √t.  This phase dominates the time‑budget.

3. **A short projected‑gradient descent** – Euclidean
   sub‑gradient steps on the true L∞ objective, followed by a
   Euclidean projection onto the probability simplex.
   This cleans up the remaining wiggles.

4. **Fine BFGS polish + a very brief simulated‑annealing
   hill‑climb**.  The final L‑BFGS uses an ultra‑small λ
   (≈10⁻⁶) and many iterations; a tiny anneal then
   scratches out any remaining local defects.

All stages respect the overall 1000 s deadline, keep track of
the best sequence seen (both in “fast_score” and the official
`evaluate_sequence`) and finally store it in the global variable
`height_sequence_1`.
"""

import time
import numpy as np
from scipy.optimize import minimize

def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(conv(a,a)) for vectors that sum to 1.
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a diverse probability vector.
    Several families (uniform, power‑law, exponential, etc.) are mixed.
    """
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal',
         'sparse', 'logistic', 'gaussian']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.0005, 0.003)          # decay rate
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.005, 0.025)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.015 * n), int(0.045 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.015 * n), int(0.055 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'logistic':
        # logistic (sigmoid) shape – steeper than exponential
        s = rng.uniform(0.001, 0.01)
        x = np.arange(n) - n / 2
        a = 1.0 / (1.0 + np.exp(s * x))
    elif mode == 'gaussian':
        sigma = rng.uniform(0.1 * n, 0.3 * n)
        x = np.arange(n) - n / 2
        a = np.exp(-0.5 * (x / sigma) ** 2)
    else:   # fallback – random
        a = rng.random(n)

    a += 1e-12                     # avoid exact zeros before normalisation
    a /= a.sum()
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    return simplex_projection(noisy)

def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution.
    y – log‑variables (softmax ensures a lies on the simplex)
    Returns f (smooth objective) and its gradient w.r.t. y.
    """
    # softmax → a
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    # smoothing parameter scaled with the current max to keep it relative
    lam_eff = max_c * lam + 1e-12

    # log‑sum‑exp around the max
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                                 # ∂smooth_max / ∂c_i
    # back‑propagation through convolution (c = a * a)
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]               # size‑n gradient w.r.t. a

    # chain‑rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space (softmax)
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score

def maxconv_gradient(a: np.ndarray, n: int, fft_len: int):
    """
    Sub‑gradient of the hard max‑convolution.
    Returns a vector g of size n.
    """
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    max_val = conv.max()
    eps = max_val * 1e-12 + 1e-12
    mask = (conv >= max_val - eps).astype(np.float64)

    a_rev = a[::-1]
    fm = np.fft.fft(mask, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fm * fa_rev))
    g = 2.0 * grad_conv[n - 1: n - 1 + n]
    return g, max_val


def exp_adagrad_optimize(a_start: np.ndarray,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator,
                         eta0: float = 15.0,
                         max_iters: int = 15_000_000):
    """
    Entropy‑mirror SGD with per‑coordinate AdaGrad adaptation.
    Operates directly on the simplex via exponentiated updates.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    # AdaGrad accumulator
    acc = np.zeros_like(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        g, _ = maxconv_gradient(a, n, fft_len)
        acc += g * g
        # effective step size per coordinate
        step = eta0 / (np.sqrt(acc) + 1e-12)

        # exponentiated update (mirror move)
        exp_term = np.exp(-step * g)
        a = a * exp_term
        a_sum = a.sum()
        if a_sum <= 0.0:
            a = best_a.copy()
            a_sum = a.sum()
        a /= a_sum

        if (it & 63) == 0:          # every 64 iterations a cheap score check
            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        it += 1
    return best_a, best_score

def proj_grad_optimize(a_start: np.ndarray,
                       deadline: float,
                       n: int,
                       fft_len: int,
                       max_iters: int = 2_000_000):
    """
    Projected gradient descent on the exact max‑convolution.
    Uses a simple diminishing step‑size and Euclidean projection onto
    the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        g, _ = maxconv_gradient(a, n, fft_len)
        eta = 1.0 / np.sqrt(it + 1.0)          # 1/√t schedule
        a_new = a - eta * g
        a = simplex_projection(a_new)

        if (it & 63) == 0:
            sc = fast_score(a, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        it += 1
    return best_a, best_score

def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature decays linearly.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.04, 1e-8)
    max_frac = 0.18  # max fraction of source mass moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score
        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split (more generous early L‑BFGS)
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.07   # small safety margin

    phase1_end = start_time + 0.30 * budget_s   # L‑BFGS heavy
    phase2_end = start_time + 0.80 * budget_s   # long EG / AdaGrad
    # remaining time → final polish + anneal

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS cascade (coarse, many restarts)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.6, 0.45, 0.35, 0.25, 0.18, 0.12,
        0.08, 0.05, 0.032, 0.020, 0.013,
        0.008, 0.005, 0.003, 0.0018, 0.001,
        0.0006, 0.00035, 0.0002, 0.0001,
        0.00007, 0.00004, 0.000025, 0.000015,
        0.000008, 0.000004, 0.000002, 0.000001,
        0.0000005, 0.0000002
    ]
    maxiter_per_lam = 1500

    best_a = None
    best_score = float('inf')
    while time.time() < phase1_end:
        a_start = pool[rng.integers(len(pool))]
        if rng.random() < 0.40:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.008), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient with AdaGrad
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_adagrad_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=18.0,
            max_iters=20_000_000
        )
        print(f"[exp‑AdaGrad] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – short projected‑gradient clean‑up
    # ------------------------------------------------------------------
    if time.time() < deadline:
        # allocate a few seconds for this tidy‑up
        pg_deadline = min(deadline, time.time() + max(5.0, 0.04 * budget_s))
        best_a, best_score = proj_grad_optimize(best_a, pg_deadline, n, fft_len)
        print(f"[proj‑grad] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 4 – final L‑BFGS polish (very low λ) + quick anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-5, 2.5e-5, 1.2e-5, 6e-6, 3e-6, 1.5e-6, 7e-7]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2500,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[final‑lbfgs] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
