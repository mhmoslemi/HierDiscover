"""
Optimisation of a non‑negative step‑function (length ~4000) to minimise
the evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of four stages inside the allowed 1000 s budget:

1. **Broad exploration** – generate a large diverse pool of candidate
   probability vectors (uniform, exponential, power‑law, Gaussian,
   symmetric‑exponential, *etc.*) and run a short L‑BFGS cascade on each
   with a relatively coarse smoothing schedule.  The best few candidates
   are kept for the next stage.

2. **Fine L‑BFGS polish** – the retained candidates are refined with a
   long cascade that drives the smoothing parameter λ down to 1e‑6
   (essentially the true L∞ objective).  The best vector from this stage
   becomes the current incumbent.

3. **Momentum exponentiated‑gradient (mirror‑SGD)** – a long
   mirror‑descent on the true max‑convolution using a moving‑average of
   sub‑gradients.  The step‑size decays as η₀/√t.  This stage typically
   yields the final improvement.

4. **Intensive simulated‑annealing hill‑climb** – a final stochastic
   search that moves small amounts of mass between positions with a
   linear cooling schedule.  The best solution found by any stage is
   returned.

All stages respect the global deadline; intermediate best scores are
printed.  The final sequence is stored in the global variable
``height_sequence_1`` and also returned as a plain ``list`` of ``float``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast proxy for the official score when sum(a) == 1:
    2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    total = noisy.sum()
    if total <= 0.0:
        return a.copy()
    return noisy / total


# ----------------------------------------------------------------------
# Random initialisations (large, diverse pool)
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a random probability vector using a variety of families.
    The returned vector sums to 1 and contains only non‑negative entries.
    """
    mode = rng.choice(
        [
            "uniform",
            "random",
            "exp",
            "sym_exp",
            "power",
            "sym_power",
            "gaussian",
            "edge",
            "bimodal",
            "sparse",
            "sinc",
        ]
    )
    if mode == "uniform":
        a = np.full(n, 1.0 / n)
    elif mode == "random":
        a = rng.random(n)
    elif mode == "exp":
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == "sym_exp":
        rate = rng.uniform(0.001, 0.02)
        idx = np.arange(n) - (n - 1) / 2.0
        a = np.exp(-rate * np.abs(idx))
    elif mode == "power":
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == "sym_power":
        p = rng.uniform(0.2, 2.5)
        idx = np.abs(np.arange(n) - (n - 1) / 2.0) + 1.0
        a = idx ** (-p)
    elif mode == "gaussian":
        sigma = rng.uniform(0.07, 0.20) * n
        centre = rng.uniform(0, n - 1)
        a = np.exp(-0.5 * ((np.arange(n) - centre) / sigma) ** 2)
    elif mode == "edge":
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == "bimodal":
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == "sparse":
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    else:  # sinc
        x = np.linspace(-5, 5, n)
        a = np.abs(np.sinc(x))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (used by L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Returns the smoothed max‑convolution value (f) and its gradient w.r.t.
    the log‑variables y (such that a = softmax(y)).
    """
    # ----- softmax -----
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # ----- convolution (FFT) -----
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # ----- gradient of smooth_max w.r.t. convolution entries -----
    w = exp_shift / Z                      # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]     # size n

    # ----- chain rule through softmax -----
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(
    a_start: np.ndarray,
    lam_sched,
    maxiter_per_lam: int,
    deadline: float,
    fft_len: int,
    n: int,
):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector seen in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # start in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method="L-BFGS-B",
                jac=True,
                options={"maxiter": maxiter_per_lam, "gtol": 1e-12, "disp": False},
            )
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror SGD) with momentum on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 12.0,
    max_iters: int = 10_000_000,
    beta: float = 0.9,
):
    """
    Mirrored SGD with exponential update and a moving‑average of sub‑gradients.
    The update stays on the probability simplex.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    g_bar = np.zeros_like(a)  # momentum term
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- compute convolution and mask of (almost) maximal entries -----
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        # ----- sub‑gradient of the (hard) max (via convolution with reversed a) -----
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]   # size n

        # ----- momentum (exponential moving average) -----
        g_bar = beta * g_bar + (1.0 - beta) * g

        # ----- step‑size schedule -----
        eta = eta0 / np.sqrt(it + 1.0)

        # ----- exponentiated‑gradient update (clipped to avoid overflow) -----
        exp_term = np.exp(np.clip(-eta * g_bar, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # ----- occasional cheap score update -----
        if (it & 31) == 0:  # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climb (more intensive than the brief one earlier)
# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    n: int,
    rng: np.random.Generator,
):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature decays linearly to almost zero.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    # Initial temperature: a small fraction of the current score
    temp0 = max(best_score * 0.08, 1e-8)
    max_frac = 0.30  # allow moving up to 30 % of the source mass

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from larger to smaller entry
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.

    # Problem size – a trade‑off between expressivity and speed
    # --------------------------------------------------------------

    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # --------------------------------------------------------------
    pool = []

    # 1) Warm‑start from the previously best sequence (if any)
    if "height_sequence_1" in globals():
        old = np.array(globals()["height_sequence_1"], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to current length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) A fairly large set of random families (≈60 vectors)
    for _ in range(60):
        pool.append(random_initial(n, rng))

    # 3) Classic window functions, if SciPy is available
    try:
        from scipy.signal import chebwin, kaiser

        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # --------------------------------------------------------------
    # Split the total budget into stages
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05  # tiny safety margin

    phase1_end = start_time + 0.18 * budget_s

            phase1_end,

        # keep global best
        if sc_opt < best_score:
            best_score = sc_opt

        # maintain a short list of the best few vectors
        if len(top_candidates) < 5:
            top_candidates.append((sc_opt, a_opt.copy()))

        elif sc_opt < top_candidates[-1][0]:

    if best_a is None:

    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    # Store for possible warm‑starts in future runs
    globals()["height_sequence_1"] = final_seq
    return final_seq
