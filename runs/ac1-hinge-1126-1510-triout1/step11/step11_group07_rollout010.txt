def lse_gradient(a, fft_len, beta):

    n = len(a)

    a_rev = a[::-1]

    # compute conv for weight calculation
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    # compute weight w = exp(beta * conv) normalized
    # Might overflow so subtract max conv for stability
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp
    # Compute gradient = 2 * conv of a_rev with w
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad = 2.0 * np.real(np.fft.ifft(fa_rev * fw))[:n]  # conv index n-1 to n-1 + n-1

    return grad, conv

def lse_gradient(a, fft_len, beta):
    n = a.shape[0]
    a_rev = a[::-1]
    # compute convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]
    # compute softmax weights
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    sum_exp = exp_terms.sum()
    w = exp_terms / sum_exp
    # gradient conv

"""
Propose a low‑evaluation step‑function sequence.

Algorithm
---------
1. Build a diverse set of initial probability vectors:
   - uniform, deterministic arcsine, random, several symmetric Beta shapes.
   - reuse a previously found sequence if available.
2. Pick the best according to a cheap FFT‑based score.
3. Refine it with three optimisation phases:
   * (A) Smoothed‑max gradient descent.
     The objective B_β(a)= (1/β)·log Σ exp(β·conv(a,a)) smoothly
     approximates the true max.  β is increased from a small to a large
     value while a simple projected gradient descent with a diminishing
     step size is applied.
   * (B) Exponentiated‑gradient descent on the exact max (active‑set mask).
   * (C) Projected sub‑gradient descent with a tighter active‑set tolerance.
4. A short simulated‑annealing walk explores the neighbourhood globally.
5. A final greedy hill‑climb with exact convolution updates fine‑tunes the
   solution.
6. The best sequence found before the time limit is returned; the harness
   will evaluate it with the supplied `evaluate_sequence` function.

All optimisation phases respect the remaining budget and always stay on the
probability simplex, thus the final score equals `2·n·max(conv)`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def full_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))
    return conv[: 2 * a.shape[0] - 1]


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) (n = len(a))."""
    n = a.shape[0]
    conv = full_conv(a, fft_len)
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Diverse initial distributions
# ----------------------------------------------------------------------


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


def arcsine_initial(n: int) -> np.ndarray:
    """Deterministic arcsine‑shaped distribution (heavy at the edges)."""
    idx = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    a /= a.sum()
    return a


def beta_initial(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density discretised on the simplex."""
    x = (np.arange(n) + 0.5) / n
    # avoid exact 0/1 to stay finite
    a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    a += 1e-12
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed‑max (log‑sum‑exp) gradient
# ----------------------------------------------------------------------


def lse_gradient(a: np.ndarray, fft_len: int, beta: float) -> np.ndarray:
    """
    Gradient of the smooth max B_β(a) = (1/β)·log Σ exp(β·conv_i).
    Returns ∇B_β(a) (shape n).
    """
    n = a.shape[0]
    a_rev = a[::-1]

    # convolution a*a
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    # soft‑max weights (stable)
    maxc = conv.max()
    exp_terms = np.exp(beta * (conv - maxc))
    w = exp_terms / exp_terms.sum()          # shape (2n‑1)

    # grad of Σ w_k·conv_k  w.r.t a
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    grad_conv = np.real(np.fft.ifft(fa_rev * fw))[:n]
    grad = 2.0 * grad_conv
    return grad


def smoothed_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smooth surrogate of max‑convolution.
    β increases from beta_start to beta_end following a time‑based schedule.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    phase_start = time.time()
    phase_len = deadline - phase_start

    while time.time() < deadline:
        # fraction of phase elapsed – used for β schedule
        frac = (time.time() - phase_start) / max(phase_len, 1e-12)
        beta = beta_start * ((beta_end / beta_start) ** frac)

        grad = lse_gradient(a, fft_len, beta)

        # diminishing step size
        step = eta0 / np.sqrt(it + 1.0)

        cand = a - step * grad
        cand = simplex_projection(cand)

        cand_score = fast_score(cand, fft_len)

        # keep track of the best found so far
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()

        # accept if we made progress, otherwise jitter occasionally
        cur_score = fast_score(a, fft_len)
        if cand_score <= cur_score:
            a = cand
        elif rng.random() < 0.01:          # tiny random jump to escape flat spots
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def exp_gradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eta0: float = 30.0,
) -> tuple[np.ndarray, float]:
    """Multiplicative (exponentiated) gradient descent on max(conv)."""
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = best_a.copy()

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the true max‑convolution
# ----------------------------------------------------------------------


def projected_subgradient_optimize(
    a_start: np.ndarray,
    deadline: float,
    n: int,
    fft_len: int,
    rng: np.random.Generator,
    eps_frac: float = 0.0005,
    step0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    The active set consists of all indices whose convolution value is within
    `eps_frac` of the current maximum.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

        max_val = conv.max()
        mask = (conv >= max_val - eps_frac * max_val).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fa_rev * fm))
        g = 2.0 * grad_conv[n - 1 : n - 1 + n]

        step = step0 / np.sqrt(it + 1.0)

        # simple back‑tracking line‑search (max 5 reductions)
        improved = False
        for _ in range(5):
            cand = a - step * g
            cand = simplex_projection(cand)
            cand_score = fast_score(cand, fft_len)
            if cand_score < best_score:
                best_score = cand_score
                best_a = cand.copy()
            if cand_score < fast_score(a, fft_len):
                a = cand
                improved = True
                break
            step *= 0.5

        if not improved:
            # stochastic jitter to escape possible stalls
            jitter = rng.normal(scale=1e-4, size=n)
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        it += 1
    return best_a, best_score

# ----------------------------------------------------------------------


def simulated_annealing(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Temperature‑controlled random walk on the simplex."""
    n = a_start.shape[0]
    fft_len = next_pow2(2 * n - 1)

    cur = a_start.copy()
    cur_score = start_score
    best = cur.copy()
    best_score = cur_score

    total_time = deadline - time.time()
    T0 = cur_score * 0.05 + 1e-12  # initial temperature

    while time.time() < deadline:
        fraction = (deadline - time.time()) / total_time
        T = T0 * fraction
        max_frac = 0.02 + 0.18 * fraction

        src = rng.choice(n, p=cur / cur.sum())
        dst = rng.integers(0, n)
        if src == dst or cur[src] <= 0.0:
            continue

        delta = cur[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        new = cur.copy()
        new[src] -= delta
        new[dst] += delta

        new_score = fast_score(new, fft_len)

        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Greedy hill‑climb with exact convolution updates
# ----------------------------------------------------------------------


def hill_climb_incremental(
    a_start: np.ndarray,
    start_score: float,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climb that moves a small fraction of mass from a heavy
    index to a lighter one, accepting only improving moves.
    """
    n = a_start.shape[0]
    best = a_start.copy()
    best_score = start_score
    conv = np.convolve(best, best)          # exact full convolution
    start_time = time.time()
    accepted = 0

    while time.time() < deadline:
        elapsed = time.time() - start_time
        total = deadline - start_time
        max_frac = 0.02 + 0.18 * (1.0 - elapsed / total)

        src = rng.choice(n, p=best / best.sum())
        dst = rng.integers(0, n)
        if src == dst or best[src] <= 0.0:
            continue

        delta = best[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # incremental update of the convolution (linear term only)
        diff = np.zeros_like(conv)
        diff[src : src + n] -= best
        diff[dst : dst + n] += best
        conv_new = conv + 2.0 * delta * diff
        new_max = conv_new.max()
        new_score = 2.0 * n * new_max

        if new_score < best_score:
            best[src] -= delta
            best[dst] += delta
            conv = conv_new
            best_score = new_score
            accepted += 1

        if accepted and accepted % 2000 == 0:
            conv = np.convolve(best, best)   # occasional re‑sync

    return best, best_score

    """

    # and speed for the 1000 s budget.
    n = 4000

    # --------------------------------------------------------------
    # 1. Build a diverse collection of starting points
    # --------------------------------------------------------------
    init_candidates = []

    init_candidates.append(arcsine_initial(n))

    # a few symmetric Beta shapes (α in (0.2, 1.5))
    for alpha in rng.uniform(0.2, 1.5, size=6):

                x_new = np.linspace(0, 1, n)
                a_old = np.interp(x_new, x_old, old)
            else:
                a_old = old.copy()
            a_old = np.maximum(a_old, 0.0)
            a_old /= a_old.sum()
            init_candidates.append(a_old)

    # pick the best according to the cheap FFT‑based score
    best_a = None
    best_score = float('inf')
    for a in init_candidates:
        sc = fast_score(a, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    print(f"[init] best fast_score = {best_score:.6f}")

    # --------------------------------------------------------------

    # 4. Projected sub‑gradient descent (≈ 20 % of remaining time)

    best_a, best_score = projected_subgradient_optimize(
