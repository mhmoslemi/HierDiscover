    a = a_start.copy()
    best_a = a.copy()

        conv = np.real(np.fft.ifft(fa * fa))[:2*n-1]

"""
step‑function optimiser – 1000 s budget (enhanced version)

The optimiser works in four stages:

1. **Diverse warm‑starts + L‑BFGS on a smoothed log‑sum‑exp surrogate**.
   A cascade of L‑BFGS runs with a decreasing λ‑schedule quickly pushes a
   random start into a good basin.  After each L‑BFGS run the vector is
   symmetrised (a←(a+a[::-1])/2) because the optimal solution is symmetric.

2. **AdaGrad‑mirrored‑SGD on the true L∞‑objective**.
   The exact sub‑gradient of max (conv(a,a)) is computed via FFT.
   An AdaGrad‑style exponentiated‑gradient update stays on the probability
   simplex and automatically adapts per‑coordinate step sizes.

3. **Fine L‑BFGS polish** with a very low λ‑schedule to refine the best
   candidate found so far.

4. **Short simulated‑annealing hill‑climb** for a final polish.

All stages respect the 1000 s deadline; the best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain list
of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast proxy for the official score when sum(a)=1:
    2·n·max(conv(a,a)) computed by FFT."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry (a[i]=a[n‑1‑i]) and keep the scale."""
    return 0.5 * (v + v[::-1])


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a diverse random probability vector."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse',
         'semicircle', 'gauss', 'cosine']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    elif mode == 'sparse':
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    elif mode == 'semicircle':
        # a_i ∝ sqrt(1 - ((2*i/(n-1))-1)^2)
        x = np.linspace(-1.0, 1.0, n)
        a = np.sqrt(np.clip(1.0 - x ** 2, 0.0, None))
    elif mode == 'gauss':
        centre = n // 2
        sigma = rng.uniform(0.1, 0.4) * n
        xs = np.arange(n)
        a = np.exp(-0.5 * ((xs - centre) / sigma) ** 2)
    else:   # cosine hill (zero at ends, peak in centre)
        xs = np.arange(n)
        a = np.sin(np.pi * xs / (n - 1))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Add isotropic Gaussian noise and re‑project onto the simplex."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray, n: int, fft_len: int, lam: float):
    """
    Smoothed max‑convolution (log‑sum‑exp surrogate) and its gradient w.r.t.
    log‑variables y (so that a = softmax(y)).
    """
    # softmax
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv_full.max()
    lam_eff = max_c * lam + 1e-12

    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam_eff)
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # gradient of smooth_max w.r.t. convolution entries
    w = exp_shift / Z                           # d smooth_max / d conv_i
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]          # size n

    # chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)

    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  fft_len: int,
                  n: int):
    """
    Run a cascade of L‑BFGS optimisations over a decreasing λ‑schedule.
    Returns the best vector found in this phase and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score(best_a, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(
                fun=objective_and_grad,
                x0=y,
                args=(n, fft_len, lam),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': maxiter_per_lam,
                         'gtol': 1e-12,
                         'disp': False}
            )
        except Exception:
            continue
        y = res.x
        # recover a from y (softmax)
        a = np.exp(y - np.max(y))
        a /= a.sum()
        # enforce symmetry (helps the objective)
        a = symmetrize(a)
        a /= a.sum()
        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# AdaGrad‑mirrored SGD on the true L∞‑objective
# ----------------------------------------------------------------------


def exp_gradient_adagrad_optimize(a_start: np.ndarray,
                                  deadline: float,
                                  n: int,
                                  fft_len: int,
                                  rng: np.random.Generator,
                                  eta0: float = 12.0,
                                  max_iters: int = 10_000_000):
    """
    Mirror‑SGD with AdaGrad‑style per‑coordinate step size on the exact
    max‑convolution objective.  The iterates stay on the probability simplex
    via a multiplicative‑weights update.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)
    grad_sq = np.zeros_like(a)
    eps = 1e-12
    it = 0

    while time.time() < deadline and it < max_iters:
        # convolution and mask of entries attaining (or nearly attaining) the max
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        tol = max_val * 1e-12 + eps
        mask = (conv >= max_val - tol).astype(np.float64)

        # sub‑gradient of the (hard) max‑convolution
        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]   # size n

        # AdaGrad accumulation
        grad_sq += g ** 2

        # step‑size schedule (slowly decaying)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update with per‑coordinate scaling
        step = -eta * g / (np.sqrt(grad_sq) + eps)
        exp_term = np.exp(np.clip(step, -50, 50))
        a = a * exp_term

        # enforce symmetry and re‑normalise
        a = symmetrize(a)
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a = a / s

        # occasional cheap score update
        if (it & 31) == 0:          # every 32 iterations
            score = fast_score(a, fft_len)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        it += 1

    return best_a, best_score

# Simple simulated‑annealing hill‑climb
# ----------------------------------------------------------------------


def simulated_annealing_simple(a_start: np.ndarray,
                               start_score: float,
                               deadline: float,
                               n: int,
                               rng: np.random.Generator):
    """
    Random mass transfers accepted with a Metropolis rule.
    Temperature is linearly cooled.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-10)
    max_frac = 0.15        # maximal fraction of src mass that may be moved

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # move mass from the larger entry to the smaller one
        if a[i] < a[j]:
            src, dst = j, i
        else:
            src, dst = i, j
        if a[src] <= 0.0:
            continue

        delta = a[src] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # symmetrise candidate (helps keep the objective low)
        cand = symmetrize(cand)
        cand /= cand.sum()

        new_score = fast_score(cand, next_pow2(2 * n - 1))
        dE = new_score - cur_score

        if dE < 0.0 or rng.random() < np.exp(-dE / temp):
            a = cand
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function with a low evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a moderate size gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000                     # can be tuned, 4000 works well on the sandbox
    fft_len = next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Build a pool of diverse starting vectors
    # ------------------------------------------------------------------
    pool = []

    # 1) Re‑use the previous best (if any)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # interpolate to the current length
            x_old = np.linspace(0, 1, old.size)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            pool.append(a0)

    # 2) Randomly generated families
    for _ in range(12):
        pool.append(random_initial(n, rng))

    # 3) Classic window functions (if SciPy is available)
    try:
        from scipy.signal import chebwin, kaiser
        for att in (30, 50, 70, 90):
            w = chebwin(n, at=att)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
        for beta in (4, 8, 12, 16):
            w = kaiser(n, beta)
            w = np.maximum(w, 0.0)
            w /= w.sum()
            pool.append(w)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # Timing budget split
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05          # safety margin

    phase1_end = start_time + 0.20 * budget_s       # L‑BFGS warm‑up
    phase2_end = start_time + 0.85 * budget_s       # long EG‑AdaGrad phase
    # remaining time → final polish + anneal

    # ------------------------------------------------------------------
    # Phase 1 – L‑BFGS cascade (coarse)
    # ------------------------------------------------------------------
    lam_schedule = [
        0.55, 0.42, 0.31, 0.22, 0.16, 0.12,
        0.08, 0.05, 0.03, 0.018, 0.011,
        0.006, 0.0035, 0.0018, 0.0009,
        0.00045, 0.00022, 0.00011, 5e-05, 2e-05
    ]
    maxiter_per_lam = 1200

    best_a = None
    best_score = float('inf')
    while time.time() < phase1_end:
        # pick a random start; occasionally jitter it a bit
        a_start = pool[rng.integers(len(pool))]
        if rng.random() < 0.35:
            a_start = add_noise(a_start, sigma=rng.uniform(0.001, 0.01), rng=rng)

        a_opt, sc_opt = lbfgs_cascade(
            a_start,
            lam_schedule,
            maxiter_per_lam,
            phase1_end,
            fft_len,
            n,
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new best fast_score={best_score:.9f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score(best_a, fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – long AdaGrad mirrored‑SGD on the true L∞‑objective
    # ------------------------------------------------------------------
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_adagrad_optimize(
            best_a,
            phase2_end,
            n,
            fft_len,
            rng,
            eta0=18.0,
            max_iters=12_000_000
        )
        print(f"[AdaGrad‑EG] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Phase 3 – final L‑BFGS polish (very low λ) + short anneal
    # ------------------------------------------------------------------
    if time.time() < deadline:
        fine_lam_schedule = [5e-5, 2.5e-5, 1.2e-5, 6e-6, 3e-6, 1.5e-6]
        a_refined, sc_refined = lbfgs_cascade(
            best_a,
            fine_lam_schedule,
            maxiter_per_lam=2000,
            deadline=deadline,
            fft_len=fft_len,
            n=n,
        )
        if sc_refined < best_score:
            best_score = sc_refined
            best_a = a_refined.copy()
            print(f"[final‑lbfgs] fast_score={best_score:.9f}")

    if time.time() < deadline:
        best_a, best_score = simulated_annealing_simple(
            best_a,
            best_score,
            deadline,
            n,
            rng,
        )
        print(f"[anneal] fast_score={best_score:.9f}")

    # ------------------------------------------------------------------
    # Final evaluation with the official scorer
    # ------------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
