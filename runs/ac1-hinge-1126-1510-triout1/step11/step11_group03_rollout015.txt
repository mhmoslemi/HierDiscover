def generate_exponential(n, alpha, eps):
    a = np.exp(-alpha * np.arange(n)) + eps
    a /= a.sum()

def generate_power_law(n, p, eps):
    a = (np.arange(1, n+1, dtype=float) ** (-p)) + eps

    right = np.exp(-beta * np.arange(n)[::-1])
    a = w * left + (1-w) * right

"""
Propose a low‑scoring non‑negative step function.

Algorithm
---------
1.  **Initial candidates** – use the previous best sequence (if any) and a
    systematic sweep over a few low‑dimensional parametric families:
    exponential decay, power‑law decay and “edge‑heavy’’ blocks.  All vectors
    are normalised to sum 1 and evaluated with a fast FFT‑based score
    ``2·n·max(conv)``.

2.  **Refinement with smoothed L‑BFGS** – the best candidate is fed to a
    cascade of L‑BFGS optimisations on a log‑sum‑exp smooth approximation
    of the true L∞ objective.  The λ‑schedule goes down to 1e‑5 and we
    allow many inner iterations.

3.  **Exact sub‑gradient descent** – mirrored exponentiated‑gradient descent
    on the true max‑convolution is run for a while (step size ∝1/√t).

4.  **Fine‑tuned L‑BFGS** – a second short L‑BFGS pass with an even smaller
    λ‑schedule.

5.  **Short annealing** – a Metropolis mass‑transfer annealer makes the last
    tweaks.

All stages respect a 1000 s global deadline.  The final sequence is stored
in the global ``height_sequence_1`` and returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x>0)."""
    return 1 << ((x - 1).bit_length())


def fast_score_vector(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(linear convolution of a with itself).  a is assumed to sum to 1."""
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def add_noise(a: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Gaussian perturbation followed by simplex projection."""
    noisy = a + rng.normal(0.0, sigma, size=a.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return a.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Parametric families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def exponential_family(n: int, alpha: float, eps: float) -> np.ndarray:
    """a_i ∝ exp(-α·i) + eps  (eps is an absolute additive term)."""
    a = np.exp(-alpha * np.arange(n)) + eps
    a /= a.sum()
    return a


def powerlaw_family(n: int, p: float, eps: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p} + eps."""
    a = (np.arange(1, n + 1, dtype=float) ** (-p)) + eps
    a /= a.sum()
    return a


def edge_heavy_family(n: int, block_frac: float, block_mass: float, rng: np.random.Generator) -> np.ndarray:
    """
    Two heavy edge blocks.
    block_frac  – fraction of the array occupied by each block (≥0, ≤0.5)
    block_mass  – total mass assigned to *each* block   (0 ≤ block_mass ≤ 0.5)
    The remaining mass is spread uniformly in the interior.
    """
    m = max(1, int(n * block_frac))
    # mass per entry inside a block
    val = block_mass / m
    a = np.full(n, 0.0, dtype=float)
    # randomise inside the block a little – this helps escaping plateaus
    left_vals = rng.random(m) * 0.5 + 0.5
    right_vals = rng.random(m) * 0.5 + 0.5
    a[:m] = left_vals * (block_mass / left_vals.sum())
    a[-m:] = right_vals * (block_mass / right_vals.sum())

    interior = n - 2 * m
    if interior > 0:
        remaining = 1.0 - 2.0 * block_mass
        if remaining < 0.0:
            remaining = 0.0
        a[m:-m] = remaining / interior
    a /= a.sum()
    return a


def mix_two_exp_family(n: int, alpha: float, beta: float, w: float) -> np.ndarray:
    """a = w·exp(-α·i) + (1‑w)·exp(-β·(n‑1‑i)).  Normalised."""
    left = np.exp(-alpha * np.arange(n))
    right = np.exp(-beta * np.arange(n)[::-1])
    a = w * left + (1.0 - w) * right
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed log‑sum‑exp objective (for L‑BFGS)
# ----------------------------------------------------------------------


def objective_and_grad(y: np.ndarray,
                       n: int,
                       fft_len: int,
                       lam: float):
    """
    Smooth approximation of 2·n·max(conv(a,a)) where a = softmax(y).
    Returns (objective, gradient w.r.t. y).
    """
    # Softmax to stay on the simplex
    m = np.max(y)
    e = np.exp(y - m)
    a = e / e.sum()

    # Convolution via FFT
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]

    max_c = conv.max()
    lam_eff = max_c * lam + 1e-14

    shifted = conv - max_c                     # ≤ 0
    exp_shift = np.exp(shifted / lam_eff)      # safe: exponents ≤ 0
    Z = exp_shift.sum()
    smooth_max = max_c + lam_eff * np.log(Z)
    f = 2.0 * n * smooth_max

    # Gradient of smooth_max wrt convolution entries
    w = exp_shift / Z
    # Back‑prop through convolution a*a → a
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_vec = conv_rev[n - 1: n - 1 + n]          # size n

    # Chain rule through softmax
    mean_g = np.dot(a, g_vec)
    grad_y = a * (g_vec - mean_g)
    return f, grad_y


def lbfgs_cascade(a_start: np.ndarray,
                  lam_sched,
                  maxiter_per_lam: int,
                  deadline: float,
                  n: int,
                  fft_len: int):
    """
    Run L‑BFGS over a decreasing λ‑schedule, always staying on the simplex.
    Returns the best vector found and its fast_score.
    """
    y = np.log(a_start + 1e-12)               # work in log‑space
    best_a = a_start.copy()
    best_score = fast_score_vector(best_a, n, fft_len)

    for lam in lam_sched:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad,
                           x0=y,
                           args=(n, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue
        y = res.x
        a = np.exp(y - np.max(y))
        a /= a.sum()
        sc = fast_score_vector(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[L‑BFGS] λ={lam:.1e} → score={sc:.9f}")
    return best_a, best_score


# ----------------------------------------------------------------------
# Exact max‑convolution sub‑gradient (mirrored exponentiated gradient)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 20.0):
    """
    Mirrored exponentiated‑gradient descent on the true L∞ objective.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score_vector(a, n, fft_len)
    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(float)

        fm = np.fft.fft(mask, fft_len)
        a_rev = a[::-1]
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n - 1: n - 1 + n]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
            s = a.sum()
        a = a / s

        sc = fast_score_vector(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        it += 1
    return best_a, best_score


# ----------------------------------------------------------------------
# Simple Metropolis annealer for final tweaks
# ----------------------------------------------------------------------


def simulated_annealing_improved(a_start: np.ndarray,
                                 start_score: float,
                                 deadline: float,
                                 n: int,
                                 rng: np.random.Generator):
    """
    Mass‑transfer / swap moves with a linearly cooled temperature.
    """
    a = a_start.copy()
    cur_score = start_score
    best_a = a.copy()
    best_score = cur_score

    start_time = time.time()
    temp0 = max(best_score * 0.05, 1e-12)
    max_frac = 0.25

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        temp = max(temp0 * (1.0 - frac), 1e-12)

        move_type = rng.choice(['transfer', 'swap', 'local'])

        if move_type == 'swap':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            i = rng.integers(0, n)
            if move_type == 'local':
                offset = rng.integers(-5, 6)
                j = (i + offset) % n
            else:   # transfer
                j = rng.integers(0, n)
            if i == j:
                continue
            src, dst = (i, j) if a[i] > a[j] else (j, i)
            if a[src] <= 0.0:
                continue
            delta = a[src] * rng.random() * max_frac
            if delta <= 0.0:
                continue
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta

        cand_score = fast_score_vector(cand, n, next_pow2(2 * n - 1))
        dE = cand_score - cur_score
        accept = False
        if dE < 0.0:
            accept = True
        else:
            if rng.random() < np.exp(-dE / temp):
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with minimal evaluation score.
    Returns the best list of floats found before the deadline.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = 4000
    fft_len = next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Gather initial candidates (old result + parametric families)
    # --------------------------------------------------------------
    candidates = []

    # 0) previous global best (if present)
    if 'height_sequence_1' in globals():
        old = np.array(globals()['height_sequence_1'], dtype=float)
        if old.ndim == 1 and old.size > 0:
            # Resample to current n via linear interpolation
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, old)
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
            candidates.append(a0)

    # --------------------------------------------------------------
    # Parameter sweep (takes roughly the first 15 % of the budget)
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.05               # safety margin
    sweep_deadline = start_time + 0.15 * budget_s

    # Exponential decay family
    alphas = np.linspace(0.001, 0.02, 30)
    eps_factors = [0.0, 1e-4, 1e-3, 1e-2]                 # absolute eps = factor / n
    for alpha in alphas:
        if time.time() > sweep_deadline:
            break
        for ef in eps_factors:
            eps = ef / n
            a = exponential_family(n, alpha, eps)
            candidates.append(a)

    # Power‑law family
    ps = np.linspace(0.2, 3.0, 30)
    for p in ps:
        if time.time() > sweep_deadline:
            break
        for ef in eps_factors:
            eps = ef / n
            a = powerlaw_family(n, p, eps)
            candidates.append(a)

    # Edge‑heavy blocks
    block_fracs = np.linspace(0.001, 0.02, 10)
    block_masses = np.linspace(0.01, 0.45, 10)   # per‑edge mass, 2·block_mass ≤ 1
    for bf in block_fracs:
        if time.time() > sweep_deadline:
            break
        for bm in block_masses:
            if 2.0 * bm > 1.0:
                continue
            a = edge_heavy_family(n, bf, bm, rng)
            candidates.append(a)

    # Mixture of two exponentials (asymmetric)
    mix_alphas = np.linspace(0.001, 0.02, 8)
    mix_betas = np.linspace(0.001, 0.02, 8)
    mix_ws = [0.2, 0.5, 0.8]
    for a1 in mix_alphas:
        if time.time() > sweep_deadline:
            break
        for a2 in mix_betas:
            for w in mix_ws:
                a = mix_two_exp_family(n, a1, a2, w)
                candidates.append(a)

    # --------------------------------------------------------------
    # Pick the best candidate from the sweep
    # --------------------------------------------------------------
    best_a = None
    best_score = float('inf')
    for a in candidates:
        sc = fast_score_vector(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[sweep] new best score = {sc:.9f}")

    # If sweep failed for any reason fallback to uniform
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = fast_score_vector(best_a, n, fft_len)

    # --------------------------------------------------------------
    # Phase 1 – aggressive L‑BFGS on the smoothed objective
    # --------------------------------------------------------------
    phase1_end = start_time + 0.45 * budget_s
    lam_schedule = [
        0.6, 0.45, 0.33, 0.24, 0.18, 0.13,
        0.09, 0.06, 0.04, 0.025, 0.015,
        0.009, 0.005, 0.003, 0.0015, 0.0008,
        0.0004, 0.0002, 0.0001, 5e-5, 2e-5, 1e-5,
    ]
    maxiter_per_lam = 800

    while time.time() < phase1_end:
        a0 = best_a.copy()
        # occasional jitter to escape shallow plateaus
        if rng.random() < 0.3:
            sigma = rng.uniform(0.001, 0.01)
            a0 = add_noise(a0, sigma, rng)
        a_opt, sc_opt = lbfgs_cascade(a0,
                                      lam_schedule,
                                      maxiter_per_lam,
                                      phase1_end,
                                      n,
                                      fft_len)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"[L‑BFGS] new global best score = {sc_opt:.9f}")

    # --------------------------------------------------------------
    # Phase 2 – exact max‑convolution exponentiated‑gradient descent
    # --------------------------------------------------------------
    phase2_end = start_time + 0.80 * budget_s
    if time.time() < phase2_end:
        best_a, best_score = exp_gradient_optimize(best_a,
                                                   phase2_end,
                                                   n,
                                                   fft_len,
                                                   rng,
                                                   eta0=25.0)
        print(f"[exp‑grad] score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Phase 3 – fine L‑BFGS with a very low λ‑schedule
    # --------------------------------------------------------------
    phase3_end = start_time + 0.95 * budget_s
    if time.time() < phase3_end:
        fine_lam_sched = [5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
        a_fine, sc_fine = lbfgs_cascade(best_a,
                                         fine_lam_sched,
                                         maxiter_per_lam=1200,
                                         deadline=phase3_end,
                                         n=n,
                                         fft_len=fft_len)
        if sc_fine < best_score:
            best_score = sc_fine
            best_a = a_fine.copy()
            print(f"[fine‑L‑BFGS] score = {sc_fine:.9f}")

    # --------------------------------------------------------------
    # Phase 4 – short simulated annealing for the final polish
    # --------------------------------------------------------------
    if time.time() < deadline:
        best_a, best_score = simulated_annealing_improved(best_a,
                                                          best_score,
                                                          deadline,
                                                          n,
                                                          rng)
        print(f"[anneal] score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Final evaluation with the official scorer
    # --------------------------------------------------------------
    final_seq = best_a.tolist()
    final_eval = evaluate_sequence(final_seq)   # provided by the framework
    print(f"[final] evaluated score = {final_eval:.9f}")

    globals()['height_sequence_1'] = final_seq
    return final_seq
