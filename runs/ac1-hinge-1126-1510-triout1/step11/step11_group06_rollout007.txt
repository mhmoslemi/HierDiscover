def adam_optimize_half(y_start, lam_schedule, max_steps_per_lam, deadline, n_full, fft_len, lr0=0.12):
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1=0.9;beta2=0.999;eps=1e-8
    t=0
    # compute initial x and best_score
    e = np.exp(y - np.max(y))
    x = e / e.sum()
    a_full = 0.5*symmetrize_half(x)
    best_a = a_full.copy()
    best_score = fast_score(best_a, fft_len)
    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad_half(y, n_full, fft_len, lam)
            t += 1
            m = beta1 * m + (1-beta1) * grad_y
            v = beta2 * v + (1-beta2) * (grad_y**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            lr = lr0 / (np.sqrt(v_hat) + eps)
            y = y - lr * m_hat
            # compute score
            e = np.exp(y - np.max(y))
            x = e / e.sum()
            a_full = 0.5*symmetrize_half(x)
            score = fast_score(a_full, fft_len)
            if score < best_score:
                best_score = score
                best_a = a_full.copy()
            steps += 1
        if time.time() >= deadline:
            break
    return best_a, best_score

"""
symmetry‑enhanced step‑function optimizer

The search works on an even length n and enforces a mirror symmetry
(a[i] = a[n‑1‑i]).  This reduces the dimensionality by a factor two,
allowing a more thorough optimisation of the “flattened”
self‑convolution.  The optimisation pipeline:

1.  Initialise the half‑vector (size n/2) with a few diverse random
    distributions and optionally from the previously discovered
    construction (global height_sequence_1).

2.  Run short L‑BFGS bursts on a smooth log‑sum‑exp approximation of the
    max‑convolution.  The smoothness parameter (lam_factor) follows a
    geometric schedule (0.5 → 1e‑5).

3.  Refine the best half‑vector with Adam on the same smoothed objective.

4.  Expand to the full symmetric sequence and improve it further with
    exponentiated‑gradient descent on the true max‑convolution.

5.  Finally perform a gradient‑guided hill‑climb that moves a fraction of
    probability mass from the index with the largest sub‑gradient to the
    index with the smallest sub‑gradient, accepting only non‑increasing
    moves.

All steps respect the time budget (default 1000 s).  The resulting
sequence is returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def fast_score(a: np.ndarray, fft_len: int) -> float:
    """Score = 2·n·max(conv(a,a)) for a probability vector."""
    n = a.shape[0]
    fa = np.fft.fft(a, fft_len)
    conv = np.real(np.fft.ifft(fa * fa))[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def symmetrize_half(x: np.ndarray) -> np.ndarray:
    """Mirror‑symmetrize a half‑vector (even total length)."""
    return np.concatenate([x, x[::-1]])


def random_initial(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a diverse random distribution on the simplex."""
    mode = rng.choice(
        ['uniform', 'random', 'exp', 'power', 'edge', 'bimodal', 'sparse']
    )
    if mode == 'uniform':
        a = np.full(n, 1.0 / n)
    elif mode == 'random':
        a = rng.random(n)
    elif mode == 'exp':
        rate = rng.uniform(0.001, 0.02)
        a = np.exp(-rate * np.arange(n))
    elif mode == 'power':
        p = rng.uniform(0.2, 2.5)
        a = (np.arange(1, n + 1, dtype=float) ** (-p))
    elif mode == 'edge':
        edge_frac = rng.uniform(0.006, 0.03)
        m_edge = max(1, int(n * edge_frac))
        a = np.full(n, 1.0 / n)
        heavy = np.sqrt(n / 2.0) / n
        a[:m_edge] = heavy
        a[-m_edge:] = heavy
    elif mode == 'bimodal':
        centre = n // 2
        width = rng.integers(int(0.02 * n), int(0.07 * n))
        peak = rng.uniform(0.5, 1.5) / n
        a = np.full(n, 1.0 / n)
        left = max(0, centre - width)
        right = min(n, centre + width)
        a[left:right] = peak
    else:  # sparse
        m = rng.integers(int(0.02 * n), int(0.07 * n))
        idx = rng.choice(n, size=m, replace=False)
        vals = rng.random(m) + 1e-6
        a = np.zeros(n, dtype=float)
        a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed objective (log‑sum‑exp) for the half‑vector
# ----------------------------------------------------------------------


def objective_and_grad_half(y: np.ndarray,
                            n_full: int,
                            fft_len: int,
                            lam_factor: float = 0.30):
    """
    Smoothed objective for the symmetric step‑function.
    Parameters
    ----------
    y : np.ndarray
        Log‑softmax parameters for the half‑vector (size = n_full/2).
    n_full : int
        Full length of the sequence (must be even).
    fft_len : int
        Length of the FFT (power of two ≥ 2*n_full‑1).
    lam_factor : float
        Smoothing factor (multiplied by current max of the convolution).

    Returns
    -------
    f : float
        Objective value = 2·n_full·smooth_max(conv).
    grad_y : np.ndarray
        Gradient w.r.t. the log‑softmax parameters (same shape as y).
    """
    # softmax on the half‑vector
    m = np.max(y)
    e = np.exp(y - m)
    x = e / e.sum()                # x sums to 1, non‑negative
    # build the full symmetric distribution (still sums to 1)
    a = 0.5 * symmetrize_half(x)   # length n_full, sum = 1

    # linear convolution a*a via FFT
    fa = np.fft.fft(a, fft_len)
    conv_full = np.real(np.fft.ifft(fa * fa))[: 2 * n_full - 1]

    max_c = conv_full.max()
    lam = max_c * lam_factor + 1e-12

    # log‑sum‑exp smooth max
    shifted = conv_full - max_c
    exp_shift = np.exp(shifted / lam)
    Z = exp_shift.sum()
    smooth_max = max_c + lam * np.log(Z)

    f = 2.0 * n_full * smooth_max

    # gradient of smooth max w.r.t. convolution entries
    w = exp_shift / Z                       # d smooth_max / d conv_i

    # back‑propagation through convolution: d conv / d a = rev_a * w
    a_rev = a[::-1]
    fw = np.fft.fft(w, fft_len)
    fa_rev = np.fft.fft(a_rev, fft_len)
    conv_rev = np.real(np.fft.ifft(fa_rev * fw))
    g_full = conv_rev[n_full - 1: n_full - 1 + n_full]   # length n_full

    # gradient w.r.t. the half‑vector x (since a = 0.5*symmetrize(x))
    m_half = n_full // 2
    grad_half = 0.5 * (g_full[:m_half] + g_full[n_full - m_half:][::-1])

    # chain rule through softmax
    mean_g = np.dot(x, grad_half)
    grad_y = x * (grad_half - mean_g)

    return f, grad_y


# ----------------------------------------------------------------------
# L‑BFGS optimisation on the half‑vector
# ----------------------------------------------------------------------


def lbfgs_optimize_half(x_start: np.ndarray,
                        lam_schedule,
                        maxiter_per_lam: int,
                        deadline: float,
                        fft_len: int,
                        n_full: int):
    """
    Run a cascade of L‑BFGS optimisations for each λ in lam_schedule.
    Returns the best half‑vector and its fast_score (computed on the full
    symmetric sequence).
    """
    y = np.log(x_start + 1e-12)
    best_x = x_start.copy()
    best_a = 0.5 * symmetrize_half(best_x)
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        try:
            res = minimize(fun=objective_and_grad_half,
                           x0=y,
                           args=(n_full, fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': maxiter_per_lam,
                                    'gtol': 1e-12,
                                    'disp': False})
        except Exception:
            continue

        y = res.x
        # recover the half‑vector from y
        e = np.exp(y - np.max(y))
        x = e / e.sum()
        a_full = 0.5 * symmetrize_half(x)
        score = fast_score(a_full, fft_len)

        if score < best_score:
            best_score = score
            best_x = x.copy()
            print(f"[L‑BFGS] lam={lam:.2e} new best fast_score={best_score:.6f}")

    return best_x, best_score


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed objective (half‑vector)
# ----------------------------------------------------------------------


def adam_optimize_half(y_start: np.ndarray,
                       lam_schedule,
                       max_steps_per_lam: int,
                       deadline: float,
                       n_full: int,
                       fft_len: int,
                       lr0: float = 0.12):
    """
    Adam optimiser on the log‑sum‑exp smoothed objective for the half‑vector.
    Returns the best full symmetric sequence and its fast_score.
    """
    y = y_start.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    e = np.exp(y - np.max(y))
    x = e / e.sum()
    best_a = 0.5 * symmetrize_half(x)
    best_score = fast_score(best_a, fft_len)

    for lam in lam_schedule:
        steps = 0
        while time.time() < deadline and steps < max_steps_per_lam:
            _, grad_y = objective_and_grad_half(y, n_full, fft_len, lam)

            t += 1
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            y = y - lr * m_hat

            e = np.exp(y - np.max(y))
            x = e / e.sum()
            a_full = 0.5 * symmetrize_half(x)
            score = fast_score(a_full, fft_len)

            if score < best_score:
                best_score = score
                best_a = a_full.copy()

            steps += 1
        if time.time() >= deadline:
            break

    return best_a, best_score


# ----------------------------------------------------------------------
# Exponentiated‑gradient descent on the true max‑convolution (full)
# ----------------------------------------------------------------------


def exp_gradient_optimize(a_start: np.ndarray,
                          deadline: float,
                          n_full: int,
                          fft_len: int,
                          rng: np.random.Generator,
                          eta0: float = 10.0):
    """
    Exponentiated‑gradient descent directly on the hard max‑convolution.
    Returns the best full sequence and its fast_score.
    """
    a = a_start.copy()
    best_a = a.copy()
    best_score = fast_score(best_a, fft_len)

    it = 0
    while time.time() < deadline:
        fa = np.fft.fft(a, fft_len)
        conv = np.real(np.fft.ifft(fa * fa))[: 2 * n_full - 1]

        max_val = conv.max()
        eps = max_val * 1e-12 + 1e-12
        mask = (conv >= max_val - eps).astype(np.float64)

        a_rev = a[::-1]
        fm = np.fft.fft(mask, fft_len)
        fa_rev = np.fft.fft(a_rev, fft_len)
        grad_conv = np.real(np.fft.ifft(fm * fa_rev))
        g = 2.0 * grad_conv[n_full - 1: n_full - 1 + n_full]

        eta = eta0 / np.sqrt(it + 1.0)
        exp_term = np.exp(np.clip(-eta * g, -50, 50))
        a = a * exp_term
        s = a.sum()
        if s <= 0.0:
            a = best_a.copy()
        else:
            a /= s

        score = fast_score(a, fft_len)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        it += 1

    return best_a, best_score


# ----------------------------------------------------------------------
# Gradient‑guided hill‑climb (full sequence)
# ----------------------------------------------------------------------


def guided_hill_climb(a_start: np.ndarray,
                      start_score: float,
                      deadline: float,
                      rng: np.random.Generator,
                      start_factor: float = 0.20,
                      end_factor: float = 0.001,
                      recompute_every: int = 2000):
    """
    Gradient‑guided hill‑climb that moves a fraction of mass from the index
    with the largest sub‑gradient (most responsible for the current peak) to
    the index with the smallest sub‑gradient.  Step size decays linearly.
    """
    n = a_start.shape[0]
    a = a_start.copy()
    conv = np.convolve(a, a)               # exact linear convolution, length 2n‑1
    fft_len = next_pow2(2 * n - 1)

    best_a = a.copy()
    best_score = start_score

    phase_start = time.time()
    total = deadline - phase_start
    accepted = 0
    steps = 0

    while time.time() < deadline:
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        eps = cur_max * 1e-12 + 1e-12
        mask = (conv >= cur_max - eps).astype(np.float64)

        # sub‑gradient of the max‑convolution
        a_rev = a[::-1]
        g_full = np.real(np.fft.ifft(np.fft.fft(mask, fft_len) *
                                     np.fft.fft(a_rev, fft_len)))
        g = 2.0 * g_full[n - 1: n - 1 + n]   # length n

        src = int(np.argmax(g))
        dst = int(np.argmin(g))

        if src == dst or a[src] <= 0.0:
            # fallback random move
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            src = int(rng.choice(nz, p=a[nz] / a[nz].sum()))
            dst = int(rng.integers(0, n))
            if src == dst:
                continue

        # linearly decaying step size
        remaining_frac = max(0.0, (deadline - time.time()) / total)
        step_factor = end_factor + (start_factor - end_factor) * remaining_frac

        delta = a[src] * step_factor
        if delta < 1e-14:
            continue
        delta = min(delta, a[src])

        a_old = a.copy()
        conv_old = conv.copy()

        # incremental update of convolution (linear term only)
        conv[src:src + n] -= 2.0 * delta * a_old
        conv[dst:dst + n] += 2.0 * delta * a_old

        # move mass
        a[src] -= delta
        a[dst] += delta

        new_max = conv.max()
        new_score = 2.0 * n * new_max

        if new_score <= cur_score:
            accepted += 1
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        else:
            a = a_old
            conv = conv_old

        steps += 1

        if accepted and accepted % recompute_every == 0:
            conv = np.convolve(a, a)

        if steps % 10000 == 0:
            elapsed = time.time() - phase_start
            print(f"[guided_hc] elapsed={elapsed:.1f}s steps={steps} "
                  f"best_score={best_score:.6f}")

    return best_a, best_score
